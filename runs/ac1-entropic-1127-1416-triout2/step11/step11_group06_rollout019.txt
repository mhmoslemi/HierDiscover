def smoothed_mirror_optimize(start_a, time_deadline, max_iters=500_000, eta0=0.8, beta_max=50.0, beta_start=0.5):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iters+1):
        if time.time() >= time_deadline: break

        conv = convolution_fft(a, a)  # size 2n-1
        # compute smoothing parameter beta based on iteration
        beta = beta_start + (beta_max - beta_start) * min(1.0, t / (max_iters * 0.5))  # ramp up to half iterations

        # compute softmax weights
        # Using stability: subtract max before exponentiation
        max_c = conv.max()
        exp_arg = beta * (conv - max_c)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        w = w / w_sum

        # compute gradient g = 2*(a * w + a * w_rev_slice)
        # term2 = convolution a * w (full) & slice first n entries
        term2 = convolution_fft(a, w)[:n]
        # term1 via reversed w
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        # slice for term1: indices (2n-2) - i for i=0..n-1
        idx = (2 * n - 2) - np.arange(n)
        term1 = conv_rev[idx]

        g = 2.0 * (term1 + term2)

        avg = np.dot(a, g)

        eta = eta0 / np.sqrt(t)

        # exponentiated gradient update (KL mirror)
        # a_i <- a_i * exp( -eta * (g_i - avg) )
        exp_update = -eta * (g - avg)
        # clip for stability
        exp_update = np.clip(exp_update, -50, 50)
        a = a * np.exp(exp_update)
        # renormalize onto simplex: sum to 1
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # track best
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check after loop
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score

def solve_lp_for_g(a, M, time_limit=None):
    n = a.shape[0]
    # Build A_ub as sparse matrix via diags
    from scipy.sparse import diags
    data = [np.full(n, a[i]) for i in range(n)]
    offsets = np.arange(n)
    A_ub = diags(data, offsets, shape=(2*n-1, n), format='csc')
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    # objective: maximize sum b => minimize -sum b
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'time_limit': time_limit})
    if not res.success:
        return None
    b = res.x
    return b

"""
search for a non‑negative step‑function (height sequence) that minimises
evaluate_sequence(sequence).

Approach
--------
1. Build a diverse pool of analytic seeds (uniform, central‑binomial,
   power‑law, geometric, symmetric Beta, random Dirichlet draws).
2. Run a *smoothed* KL‑mirror descent:
   – the sub‑gradient is taken w.r.t. a soft‑max of the convolution
     (β‑smooth approximation of the true max);
   – β is increased gradually so the optimiser first sees a smooth
     landscape and then focuses on the true peak.
3. Optionally boost the current best distribution with a tiny linear
   programme (maximise total mass while keeping the convolution ≤ the
   current bound).  This step usually improves the bound further.
4. Finish with a cheap stochastic hill‑climb (swap / fractional mass move)
   on the simplex to squeeze out the last bits of improvement.
5. All phases respect the global 1000 s wall‑clock budget and keep the
   best sequence encountered.  The returned list is already on the simplex.
"""

import time
import numpy as np

# optional linear‑programming boost (requires scipy)
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags
    SCIPY_AVAILABLE = True
except Exception:            # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


def generate_random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a random point from Dirichlet(concentration⋅1)."""
    vec = np.random.gamma(concentration, 1.0, size=n)
    return project_to_simplex(vec)


# ----------------------------------------------------------------------
# Smoothed KL‑mirror optimiser (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smoothed_mirror_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
    beta_start: float = 0.5,
    beta_max: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on the smooth approximation
        Sβ(a) = (1/β)·log Σ_k exp(β·conv_k(a,a))

    The gradient of Sβ is a weighted sum of the usual max‑gradient.
    Larger β concentrates the weight on the true peak.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # 1) convolution of a with itself
        conv = convolution_fft(a, a)                     # length 2n‑1

        # 2) soft‑max weights
        beta = beta_start + (beta_max - beta_start) * min(1.0, t / max_iters)
        max_c = conv.max()
        exp_w = np.exp(beta * (conv - max_c))
        w = exp_w / exp_w.sum()                         # ∑ w = 1, w ≥0

        # 3) gradient of the smooth max (vector g)
        # term2 = Σ_j a_j·w[i‑j]  (ordinary convolution, slice 0..n‑1)
        term2 = convolution_fft(a, w)[:n]

        # term1 = Σ_j a_j·w[i+j]  (cross‑correlation, obtained via reversed w)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)            # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # mapping i → 2n‑2‑i
        term1 = conv_rev[idx]

        g = 2.0 * (term1 + term2)                       # sub‑gradient of Sβ

        # 4) KL‑mirror (exponentiated‑gradient) update
        avg = np.dot(a, g)
        eta = eta0 / np.sqrt(t)

        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)          # safety
        a = a * np.exp(exp_arg)

        # 5) project back onto the simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # occasional check for improvement
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Optional linear‑programming boost (maximise total mass under the same bound)
# ----------------------------------------------------------------------


def lp_boost(a: np.ndarray, M: float, time_limit: float = 5.0) -> np.ndarray | None:
    """
    Solve:
        maximise Σ_j b_j
        s.t.    a * b ≤ M   (element‑wise, 2n‑1 constraints)
                b ≥ 0
    Returns the optimal b (or None on failure).  The LP is solved with
    SciPy's HiGHS interior‑point solver.  A tiny time limit prevents
    overrunning the global budget.
    """
    if not SCIPY_AVAILABLE:                     # pragma: no cover
        return None
    n = a.shape[0]
    # each diagonal k of the constraint matrix has constant a_k
    offsets = np.arange(n)                     # 0 … n‑1
    diag_data = [np.full(n, a[k]) for k in range(n)]
    A_ub = diags(diag_data, offsets, shape=(2 * n - 1, n), format="csc")
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # maximise Σb → minimise –Σb
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"time_limit": time_limit},
        )
        if res.success:
            return res.x
    except Exception:                          # pragma: no cover
        return None
    return None


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        print(f"Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        print(f"No external sequence provided – using default length {n}.")

    # --------------------------------------------------------------
    # 1. Build a collection of analytic starters
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # classic central‑binomial
    init_candidates.append(generate_central_binomial(n))

    # power‑law family
    for p in np.linspace(0.2, 1.5, 13):
        init_candidates.append(generate_powerlaw(n, float(p)))

    # geometric family (including more aggressive decays)
    for r in np.linspace(0.8, 0.999, 12):
        init_candidates.append(generate_geometric(n, float(r)))

    # symmetric Beta shapes (α from very spiky to rather flat)
    for alpha in np.linspace(0.05, 0.85, 16):
        init_candidates.append(generate_beta_shape(n, float(alpha)))

    # random Dirichlet draws (various concentration parameters)
    for conc in [0.3, 0.5, 1.0, 2.0, 5.0]:
        init_candidates.append(generate_random_dirichlet(n, float(conc)))

    # previously found construction (if any)
    try:
        init_candidates.append(np.array(height_sequence_1, dtype=np.float64))
        print("Added provided height_sequence_1 as a starter.")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2. Evaluate starters and keep a few best
    # --------------------------------------------------------------
    evaluated = []
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        evaluated.append((val, cand))
        print(f"Starter {idx:02d}: score = {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [cand for _, cand in evaluated[:top_k]]
    best_val, best_seq = evaluated[0]
    print(f"Best starter score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Smoothed mirror‑descent optimisation (main workhorse)
    # --------------------------------------------------------------
    opt_deadline = deadline - 80.0   # reserve time for LP boost & hill‑climb
    if time.time() < opt_deadline:
        # allocate iterations proportionally to remaining time
        remaining_time = opt_deadline - time.time()
        # very rough estimate: ~0.002 s per iteration → 500 iters per sec
        base_iters = int(min(200_000, remaining_time * 480))
        print(f"Smoothed mirror‑descent phase – up to {base_iters} iterations per seed.")
        for i, start_a in enumerate(top_candidates):
            if time.time() >= opt_deadline:
                break
            # a slightly larger budget for the very best starter
            iters = base_iters if i > 0 else int(base_iters * 1.2)
            a_opt, sc_opt = smoothed_mirror_optimize(
                start_a,
                time_deadline=opt_deadline,
                max_iters=iters,
                eta0=0.7,
                beta_start=0.5,
                beta_max=80.0,
            )
            if sc_opt < best_val:
                best_val, best_seq = sc_opt, a_opt.copy()
                print(f"Seed {i} improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Optional random‑restart bursts (cheap)
    # --------------------------------------------------------------
    restart_deadline = deadline - 55.0
    while time.time() < restart_deadline:
        pert = best_seq + 0.02 * np.random.randn(n)
        pert = project_to_simplex(pert)
        sub_dead = time.time() + 7.0
        a_tmp, s_tmp = smoothed_mirror_optimize(
            pert,
            time_deadline=sub_dead,
            max_iters=120_000,
            eta0=0.8,
            beta_start=0.5,
            beta_max=80.0,
        )
        if s_tmp < best_val:
            best_val, best_seq = s_tmp, a_tmp.copy()
            print(f"Random restart yielded improvement: {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Linear‑programming boost (if scipy is available)
    # --------------------------------------------------------------
    if SCIPY_AVAILABLE and time.time() < deadline - 40.0:
        current_M = convolution_fft(best_seq, best_seq).max()
        print(f"LP boost: current bound M = {current_M:.6e}")
        b = lp_boost(best_seq, current_M, time_limit=6.0)
        if b is not None and b.sum() > 0.0:
            g = b / b.sum()
            g = project_to_simplex(g)
            g_score = compute_score(g)
            if g_score < best_val:
                best_val, best_seq = g_score, g.copy()
                print(f"LP boost improvement: {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # --------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 400_000
        print(f"Final hill‑climb for up to {steps} steps.")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val, best_seq = hill_score, np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard – fall back to uniform if something went wrong
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 8. Final reporting via the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
