"""
Propose a coefficient sequence minimizing `evaluate_sequence`.

Algorithm
---------
0. **Fallback** – try the known construction `height_sequence_1`.

1. **Rich parametric sweep** – explore many analytic families
   (geometric, power‑law, symmetric U‑shaped power‑law, stretched
   exponentials and convex mixtures with the uniform vector).  The
   sweep runs until a configurable time budget is exhausted and keeps
   the best (lowest 2·n·max(conv)) vector found.

2. **Frank‑Wolfe LP refinement** – if cvxpy is available, solve the
   LP that maximises Σg under the current max‑convolution bound and
   perform a line‑search.  Several FW iterations are performed.

3. **Projected sub‑gradient polishing** – minimise the non‑smooth
   objective `M = max(conv(a,a))` by sub‑gradient descent on the
   simplex.  A slowly decreasing step size, occasional jitter and an
   aggressive iteration count give a fine‑grained polishing.

4. **Stochastic hill‑climb / simulated‑annealing** – tiny mass‑transfer
   moves are greedily accepted; when progress stalls a small random
   perturbation is added (simulated‑annealing style) to escape plateaus.

All phases respect the remaining wall‑clock budget (default 1000 s).
The returned Python list sums to 1, so `evaluate_sequence` sees the same
normalisation as the internal `compute_score` helper.
"""

import time
import numpy as np

# Optional cvxpy – used only for the Frank‑Wolfe LP step
try:
    import cvxpy as cp
    _HAVE_CVX = True
except Exception:   # pragma: no cover
    _HAVE_CVX = False


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT. Length = 2·n‑1."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – exhaustive parametric sweep, many families
# ----------------------------------------------------------------------
def best_initial_vector(n: int, deadline: float) -> np.ndarray:
    """Search a collection of analytic families and return the best vector."""
    best_a = None
    best_val = float("inf")

    # ---------- uniform (baseline) ----------
    uniform = np.full(n, 1.0 / n)
    val = compute_score(uniform)
    if val < best_val:
        best_val, best_a = val, uniform

    # ---------- power‑law family (monotone decreasing) ----------
    for p in np.linspace(0.20, 1.30, 57):   # step ≈0.02
        if time.time() >= deadline:
            return best_a
        a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ---------- geometric decay ----------
    # coarse sweep
    for r in np.linspace(0.50, 0.99, 81):
        if time.time() >= deadline:
            return best_a
        a = r ** np.arange(n, dtype=np.float64)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # fine sweep with r extremely close to 1
    epsilons = np.logspace(-2, -5, num=7)   # 1e‑2 … 1e‑5
    for eps in epsilons:
        if time.time() >= deadline:
            return best_a
        r = 1.0 - eps
        if r <= 0.0:
            continue
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ---------- symmetric U‑shaped power law:
    # a_i ∝ ((i+1)*(n-i))^{-p}
    for p in np.linspace(0.10, 1.20, 23):
        if time.time() >= deadline:
            return best_a
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = (idx * (n + 1 - idx)) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ---------- stretched exponential family:
    # a_i ∝ exp(-c * i^γ)
    gammas = [0.5, 0.75, 1.0, 1.25, 1.5]
    cs = [0.001, 0.005, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0]
    for gamma in gammas:
        i = np.arange(n, dtype=np.float64)
        i_pow = i ** gamma
        for c in cs:
            if time.time() >= deadline:
                return best_a
            a = np.exp(-c * i_pow)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    # ---------- convex mixtures with the uniform vector ----------
    # mix each discovered candidate with uniform to smooth the shape
    # (mixing ratios 0.1 … 0.9)
    if best_a is None:
        best_a = uniform
    for lam in np.linspace(0.1, 0.9, 9):
        if time.time() >= deadline:
            return best_a
        mixed = lam * best_a + (1.0 - lam) * uniform
        # already sums to 1, just re‑normalise for safety
        mixed /= mixed.sum()
        val = compute_score(mixed)
        if val < best_val:
            best_val, best_a = val, mixed

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – Frank‑Wolfe (LP) refinement
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build matrix A such that A @ g = conv(f, g) (full convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve max Σ g_i  s.t. conv(f,g) ≤ M , g ≥ 0."""
    if not _HAVE_CVX:
        return None
    n = f.shape[0]
    A = _build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=1500, abstol=1e-9, reltol=1e-9)
    except Exception:   # pragma: no cover
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return np.asarray(g.value).flatten()


def frank_wolfe_refine(f0: np.ndarray, deadline: float,
                       max_iter: int = 150) -> np.ndarray:
    """Conditional‑gradient (Frank‑Wolfe) improvement."""
    f = f0.copy()
    best_f = f.copy()
    best_M = full_convolution(f).max()
    # a modest set of candidate mixing factors – include finer steps
    t_candidates = np.array([0.01, 0.02, 0.05, 0.10, 0.15,
                             0.20, 0.25, 0.30, 0.40])

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        cur_M = full_convolution(f).max()

        g = _solve_lp_for_g(f, cur_M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_hat = g / sum_g

        best_t = 0.0
        best_new_M = cur_M
        for t in t_candidates:
            f_cand = (1.0 - t) * f + t * g_hat
            M_cand = full_convolution(f_cand).max()
            if M_cand < best_new_M:
                best_new_M = M_cand
                best_t = t

        if best_t == 0.0:          # no improvement found
            break

        f = (1.0 - best_t) * f + best_t * g_hat
        cur_M = best_new_M

        if cur_M < best_M:
            best_M = cur_M
            best_f = f.copy()

    return best_f


# ----------------------------------------------------------------------
# Phase 3 – Sub‑gradient polishing
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.20) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max entry: sum of 2*a shifted for each active k
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
            # diagonal terms counted twice when i == j; the above already does it.

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape tiny plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.02 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Phase 4 – hill‑climb with simulated‑annealing flavour
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_500_000) -> np.ndarray:
    """Greedy mass‑transfer moves with occasional temperature‑based jumps."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.12                # initial magnitude of transferred mass
    it = 0
    no_improve = 0
    temperature = 1.0                # starts warm, cools slowly

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # transfer a fraction of the source mass
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        improve = M_cand < best_M

        # simulated‑annealing acceptance
        if not improve:
            prob = np.exp(-(M_cand - best_M) / max(temperature, 1e-12))
            improve = np.random.rand() < prob

        if improve:
            if M_cand < best_M:
                best_M = M_cand
                best_a = cand.copy()
                no_improve = 0
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)   # slowly enlarge step
            temperature *= 0.9995                     # cool down a bit
        else:
            step_scale *= 0.985
            temperature *= 1.0002                     # a tiny reheating
            no_improve += 1

        it += 1

        # occasional random jitter when stuck
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.0015
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    n = 2048   # length of the step function (kept moderate for speed)

    # -----------------------------------------------------------------
    # 1. Global parametric sweep (rich families)
    # -----------------------------------------------------------------
    pll_deadline = deadline - 400.0
    if time.time() < pll_deadline:
        a_initial = best_initial_vector(n, pll_deadline)

            val = compute_score(a_initial)
            print(f"[global sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Frank‑Wolfe LP refinement (if cvxpy is available)
    # -----------------------------------------------------------------
    fw_deadline = deadline - 200.0

        f_refined = frank_wolfe_refine(f0, fw_deadline, max_iter=150)
        val = compute_score(f_refined)
        print(f"[Frank‑Wolfe] refined score = {val:.6f}")
        if val < best_val:

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing – run longer than the previous version
    # -----------------------------------------------------------------
    sub_deadline = deadline - 80.0

        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, sub_deadline, step0=0.20)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:

    # -----------------------------------------------------------------
    # 4. Final stochastic hill‑climb / simulated annealing
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
