def additive_subgrad_optimize(start_a, time_deadline, max_iters=300000, eta0=0.5):
   a = start_a.copy()
   best_a, best_score = a.copy(), compute_score(a)

       if time.time() >= time_deadline:

       conv = convolution_fft(a,a)
       k = int(np.argmax(conv))

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

What it does
------------
1.  Builds a rich library of analytic “seed” sequences (uniform,
    central‑binomial, power‑law, geometric, beta‑shaped, triangular,
    exponential and a few random Dirichlet draws).  All seeds are
    normalised to the probability simplex (sum = 1).

2.  Picks the best seed by the exact objective
    ``score = 2·n·max(conv(a,a))`` (the same value returned by
    ``evaluate_sequence`` up to the harmless sum‑normalisation).

3.  Refines the best seed with three complementary optimisation phases,
    each respecting the wall‑clock deadline (default 1000 s).

    • **Mirror descent** – exponentiated‑gradient (KL‑mirror) updates on
      the true non‑smooth objective.  A relatively large step‑size
      (η₀≈0.9) and a generous iteration budget push the solution far
      into a low‑peak region.

    • **Additive sub‑gradient descent** – after the mirror phase the
      non‑smoothness is handled by a classic projected‑sub‑gradient step
      (a←a−η·g,   g the sub‑gradient of the active convolution entry).
      This phase is very good at polishing the result because the
      additive update can make tiny, balanced adjustments that the
      multiplicative update cannot.

    • **Targeted hill‑climb** – a fast stochastic local search that
      explicitly reduces the product causing the current maximal
      convolution entry.  At each iteration the pair (i,j) that contributes
      most to the peak is identified and a small fraction of mass is
      moved from the worst index to a more “benign’’ one.  The move is
      accepted only if it decreases the exact score (or with a tiny
      probability to escape plateaus).

   Each phase keeps the best sequence ever seen; the final answer is the
   best among all phases.

4.  Returns the best sequence as a plain list of floats (already on the
   simplex).  ``evaluate_sequence`` is called only once at the end for
   reporting.

The whole optimiser runs in ≤ budget‑seconds (default 1000) and is
designed to beat the previous best score (≈ 1.511 ) by a comfortable
margin (target ≈ 1.503).  The implementation uses only ``numpy`` and
standard Python libraries, works on any length (the code doubles the
provided length if time permits), and contains no hidden state or I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety – renormalise exactly
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """
    a_i = C(2i,i)/4**i (monotonically decreasing), normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
    Symmetric when α=β.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (beta - 1.0)
    a = left * right
    return project_to_simplex(a)


def generate_triangular(n: int) -> np.ndarray:
    """A symmetric triangular shape rising to the centre."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    return project_to_simplex(a)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) sub‑gradient descent on
    f(a)=2·n·max(conv(a,a)).  a stays on the simplex.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # current convolution and active peak
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (g_i = 2 a_{k-i} if defined)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # centre the sub‑gradient (mirrored update)
        avg = np.dot(a, g)                # = 2·max(conv)
        eta = eta0 / np.sqrt(t)

        exp_arg = -eta * (g - avg)
        # protect against overflow in the exponent
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # project back onto the simplex
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        if t % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    # final check
    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()

    return best_a, best_score


def additive_subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Classical projected additive sub‑gradient descent on the same objective.
    The method is especially good for fine‑grained polishing once the
    iterate is already near a local optimum.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diminishing additive step
        eta = eta0 / np.sqrt(t)
        a_candidate = a - eta * g
        a_candidate = project_to_simplex(a_candidate)

        cur = compute_score(a_candidate)
        if cur < best_score:
            best_score = cur
            best_a = a_candidate.copy()

        a = a_candidate

    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score


def targeted_hill_climb(
    start_a: np.ndarray,
    time_deadline: float,
    max_steps: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Very local stochastic moves that try to weaken the active convolution
    peak directly.  In each iteration we locate the pair (i, k‑i) that
    contributes most to the current maximum and shift a small fraction of
    mass from the worst index to a more benign one.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    step_scale = 0.03          # initial fraction of mass transferred

    while max_steps > 0 and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contributions to the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        prod = np.zeros_like(a)
        prod[mask] = a[idx[mask]] * a[j[mask]]

        i_max = int(np.argmax(prod))
        # choose a target index that likely lowers the peak
        # we try a few random candidates and keep the first that
        # pairs with a much smaller counterpart.
        candidate_j = None
        for _ in range(6):
            cand = np.random.randint(n)
            if cand == i_max:
                continue
            j_cand = k - cand
            if 0 <= j_cand < n:
                prod_cand = a[cand] * a[j_cand]
            else:
                prod_cand = 0.0
            if prod_cand < prod[i_max] * 0.5:
                candidate_j = cand
                break
        if candidate_j is None:
            candidate_j = np.random.randint(n)

        # move a tiny fraction of mass
        delta = step_scale * a[i_max]
        if delta > a[i_max]:
            delta = a[i_max] * 0.5
        if delta <= 0:
            break

        cand_seq = a.copy()
        cand_seq[i_max] -= delta
        cand_seq[candidate_j] += delta
        cand_seq = np.maximum(cand_seq, 0.0)
        cand_seq = project_to_simplex(cand_seq)

        cur_score = compute_score(cand_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand_seq.copy()
            a = cand_seq
            step_scale = min(step_scale * 1.06, 0.2)
        else:
            # occasional uphill move to escape flat plateaus
            if np.random.rand() < 0.004:
                a = cand_seq
            step_scale = max(step_scale * 0.96, 2e-4)

        max_steps -= 1

    final = compute_score(best_a)
    if final < best_score:
        best_score, best_a = final, best_a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------

    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]

    except Exception:
        base_n = 2048
        print(f"No external sequence – using default length {base_n}.")

    # If we have enough slack we double the length – this gives the optimiser
    # a richer space and often yields a lower peak.
    if deadline - time.time() > 250:  # enough time to afford the extra work
        n = base_n * 2
        print(f"Allocating a larger problem size n={n} (double).")
    else:
        n = base_n
        print(f"Working with n={n}.")

    # ------------------------------------------------------------------
    # 1. Build a diverse collection of analytic starters
    # ------------------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # central binomial
    seeds.append(generate_central_binomial(n))

    # power‑law family (wide sweep)
    for p in np.linspace(0.2, 1.3, 12):
        seeds.append(generate_powerlaw(n, float(p)))

    # geometric family (r close to 1)
    for r in [0.96, 0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:

    # Beta‑shaped symmetric (α=β)

    # Beta‑shaped asymmetric (α≠β) – a couple of skewed variants
    for a, b in [(0.15, 0.35), (0.35, 0.15), (0.1, 0.4), (0.4, 0.1)]:
