def gaussian_family(n, sigma):

def sine_power_family(n, p):
    i = np.arange(n)

    sin_vals = np.sin(np.pi * (i + 0.5) / n)

    if p >= 0:

"""
Propose a non‑negative step‑function (probability vector) that minimises

    2·n·max(convolution(a,a)) / (sum(a)**2)

The algorithm works under a 1000 s wall‑clock budget and proceeds in three
phases:

1. **Broad seeding** – a large set of analytic families (product, power,
   exponential‑U, Gaussian, sine‑power, linear ramp) and a few random Dirichlet
   draws are evaluated on a chosen length (4096 or 8192).  The best few seeds
   are kept.

2. **Smooth‑max Adam optimisation** – each seed is refined by minimising a
   smooth approximation of the max‑convolution.  The approximation is the
   log‑sum‑exp surrogate with an increasing smoothing parameter β.
   Adam updates are performed on the soft‑max logits (so the simplex and
   non‑negativity are satisfied automatically).  A schedule of β values
   drives the solution towards the true max.

3. **Exact‑max sub‑gradient polishing** – the best solution from phase 2 is
   further improved by projected sub‑gradient descent on the true max‑convolution.
   After each step the vector is projected onto the probability simplex.

All stages respect the global deadline and continuously keep the best
sequence seen so far.  The function finally returns a plain Python list of
floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast for large vectors)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v_i ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑ a = 1).
    Returns 2·n·max(convolution(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for fast seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised to a simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against infs for extreme α
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian bell centred at the middle of the support."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return w / w.sum()


def sine_power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p (π (i+½) / n), symmetric."""
    i = np.arange(n, dtype=np.float64)
    sin_vals = np.sin(np.pi * (i + 0.5) / n)
    # sin_vals > 0 for all i, so power is safe even for negative p
    w = sin_vals ** p
    return w / w.sum()


def linear_ramp_family(n: int, r: float) -> np.ndarray:
    """a_i ∝ (i+1)^r (r can be negative for left‑heavy shapes)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** r
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where
        f   = smooth max = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z  (gradient w.r.t. the soft‑max parameters)
    The gradient is computed analytically via convolution operations.
    """
    a = softmax(z)                     # a lies on the simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights over convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta
    w = expw / sumexp                  # normalised weights, sum to 1

    # gradient w.r.t. a : 2 * (a_rev * w) evaluated on the central slice
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)          # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]      # shape (n,)

    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(z0: np.ndarray,
                  beta: float,
                  max_iters: int,
                  deadline: float,
                  lr: float = 0.1):
    """
    Adam optimisation of the smooth‑max surrogate.
    Operates on the logits `z` (soft‑max parameters) and returns the best
    probability vector found as well as its exact score.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(z)
    best_a = (best_a + best_a[::-1]) * 0.5
    best_a = project_to_simplex(best_a)
    best_score = compute_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        f, grad = smooth_max_and_grad(z, beta, n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if t % 200 == 0 or t == max_iters:
            a = softmax(z)
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Adam β={beta:.0f} iter={t} new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         max_iters: int = 30000,
                         step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true `max(convolution(a,a))`.
    Works directly on the probability vector `a` (sum = 1).
    Returns the best vector found and its exact score.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        idx = int(conv.argmax())

        # gradient of the selected entry (indicator weight)
        w = np.zeros_like(conv)
        w[idx] = 1.0
        conv_rev = convolution_fft(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # simple diminishing step size
        step = step0 / np.sqrt(it)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        if it % 500 == 0 or it == max_iters:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Sub‑grad iter={it} new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided ``evaluate_sequence`` objective.  The routine respects the
    wall‑clock budget, prints progress information and finally returns a
    plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the evaluator
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback using a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives a tighter theoretical bound)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 700:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # 1. Broad scan of analytic families (cheap)
    # ------------------------------------------------------------------
    seeds = []

    # Uniform baseline
    seeds.append(np.full(n, 1.0 / n))

    # Symmetric product family
    for alpha in np.linspace(0.0, 4.0, 17):
        seeds.append(symmetric_product_family(n, alpha))

    # Left‑heavy power family
    for beta in np.linspace(0.0, 4.0, 17):
        seeds.append(left_heavy_power_family(n, beta))

    # Exponential U‑shape
    for gamma in np.linspace(0.0, 4.5, 10):
        seeds.append(exp_u_family(n, gamma))

    # Gaussian bells (σ as a fraction of n)
    for frac in np.linspace(0.05, 0.45, 9):
        sigma = frac * n
        seeds.append(gaussian_family(n, sigma))

    # Sine‑power family (p can be negative or positive)
    for p in np.linspace(-3.0, 3.0, 13):
        seeds.append(sine_power_family(n, p))

    # Linear ramp (right‑heavy if r>0, left‑heavy if r<0)
    for r in np.linspace(-2.5, 2.5, 11):
        seeds.append(linear_ramp_family(n, r))

    # A handful of random Dirichlet draws
    for _ in range(12):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # Evaluate all seeds (stop early if we run out of time)
    scored_seeds = []
    for vec in seeds:
        if time.time() >= deadline - 120.0:
            break
        # numerical safety – ensure strict positivity before log‑softmax later
        vec = np.maximum(vec, 1e-20)
        sc = compute_score(vec)
        scored_seeds.append((sc, vec))
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best from seeding = {best_score:.6f}")

    # Keep the top‑K seeds for intensive optimisation
    scored_seeds.sort(key=lambda x: x[0])
    top_k = min(5, len(scored_seeds))
    top_seeds = [vec for (_, vec) in scored_seeds[:top_k]]
    print(f"[{time.time():.1f}] proceeding with {top_k} top seeds (best = {best_score:.6f})")

    # ------------------------------------------------------------------
    # 2. Adam optimisation on smooth‑max surrogate (multiple β stages)
    # ------------------------------------------------------------------
    beta_schedule = [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0, 32000.0]

    for idx, a_start in enumerate(top_seeds):
        if time.time() >= deadline - 30.0:
            break
        print(f"[{time.time():.1f}] Optimising seed {idx + 1}/{top_k}")

        # Initialise logits from the probability vector
        z = np.log(a_start + 1e-12)

        # Run through the β schedule
        for beta in beta_schedule:
            if time.time() >= deadline - 10.0:
                break
            # allocate a modest number of Adam steps; adapt to remaining time
            time_left = deadline - time.time()
            max_iters = int(min(2500, max(300, time_left * 3)))  # a few thousand at most
            a_opt, sc_opt = adam_optimize(z, beta, max_iters, deadline, lr=0.12)
            # update the starting point for the next β stage
            z = np.log(a_opt + 1e-12)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}]   Global best improved to {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3. Sub‑gradient polishing of the currently best candidate
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            a_current = np.array(best_seq, dtype=np.float64)
            a_refined, sc_refined = subgradient_optimize(a_current, deadline, max_iters=30000,
                                                         step0=0.07)
            if sc_refined < best_score:
                best_score = sc_refined
                best_seq = a_refined.tolist()
                print(f"[{time.time():.1f}]   Post‑sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
