"""
Advanced optimiser for the non‑negative step‑function problem.

The goal is to minimise
    2·n·max(conv(a, a))   with   a_i ≥ 0 , Σ a_i = 1

This implementation:
* generates a diverse set of initial seeds (deterministic shapes,
  beta‑family PDFs, Dirichlet samples, sparse spikes and a novel
  random‑phase “flat‑spectrum” seed);
* evaluates every seed exactly (via FFT convolution) and keeps the best;
* runs a cascade of optimisation phases tuned for the remaining time:
    1. annealed soft‑max (log‑sum‑exp) gradient descent;
    2. Adam optimiser on the same smooth surrogate;
    3. sub‑gradient descent with momentum (updates the true max‑objective);
    4. simulated‑annealing hill‑climb that can accept occasional
       uphill moves, using targeted transfers from large‑gradient indices;
    5. final lightweight stochastic hill‑climb polish.
All updates are projected onto the probability simplex, so the objective
remains comparable to ``evaluate_sequence`` throughout the search.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (Σ a = 1):
        2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Seed generators

def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def generate_flat_spectrum_random(n: int) -> np.ndarray:
    """
    Random‑phase spectrum with unit magnitude (flat power spectrum).
    The inverse FFT yields a real vector; negatives are clipped and the
    result is normalised to a probability simplex.
    """
    # Length of the rfft output for a real signal of length n
    m = n // 2 + 1
    phases = np.random.rand(m) * 2.0 * np.pi
    phases[0] = 0.0                     # DC component must be real and positive
    if n % 2 == 0:
        phases[-1] = 0.0                # Nyquist frequency also real
    spectrum = np.exp(1j * phases)      # unit magnitude, random phase
    a = np.fft.irfft(spectrum, n=n)     # real signal
    a = np.maximum(a, 0.0)              # enforce non‑negativity
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

# Optimisation phases

def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    step_coef: float = 0.0015,
) -> tuple[np.ndarray, float]:
    """
    Projected GD on the smooth surrogate
        L_β(a) = log Σ exp(β·conv(a,a)).
    Anneals β from ``beta_start`` to ``beta_end``.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    it = 0
    t0 = time.time()
    while time.time() < deadline:
        # exact score for bookkeeping
        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # annealed temperature
        elapsed = time.time() - t0
        fraction = elapsed / (deadline - t0)
        beta = beta_start + fraction * (beta_end - beta_start)

        # current convolution
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()

        # soft‑max weights (stable)
        shifted = conv - max_conv                 # max → 0
        exp_vals = np.exp(beta * shifted)        # ≤ 1
        w = exp_vals / exp_vals.sum()

        # gradient of the surrogate
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # step size (decays with √iteration and with β)
        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        # projected update
        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 5000 == 0:
            print(
                f"[soft‑max GD] it {it:6d} | best {best_score:.6f} | β {beta:.1f}"
            )
    return best_a, best_score


def adam_softmax_optimize(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    lr0: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the same smooth surrogate used by
    ``softmax_gradient_descent``.  The learning‑rate is decayed
    automatically via the Adam moment estimates.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur_a)
    v = np.zeros_like(cur_a)

    it = 0
    t0 = time.time()
    while time.time() < deadline:
        it += 1
        # annealed β
        elapsed = time.time() - t0
        fraction = elapsed / (deadline - t0)
        beta = beta_start + fraction * (beta_end - beta_start)

        # conv & smooth gradient
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_vec = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        cur_a = cur_a - step_vec
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if it % 3000 == 0:
            print(f"[Adam] it {it:6d} | best {best_score:.6f}")

    return best_a, best_score


def subgradient_momentum_optimize(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.07,
    mu: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑norm,
    augmented with classical momentum (parameter μ).
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    momentum = np.zeros_like(cur_a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        # momentum update
        momentum = mu * momentum + (1.0 - mu) * g

        step = step0 / np.sqrt(it)
        cur_a = cur_a - step * momentum
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # occasional jitter to shake out of shallow plates
        if it % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(it))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climb (only accepts strict improvements)."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand /= cand.sum()                     # re‑normalise numerically
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96
        steps += 1
    return best_a, best_score


def simulated_annealing_optimize(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    start_temp: float = 5e-4,
    end_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb that may accept uphill moves.
    Moves consist of directed mass transfers (targeting the current
    max‑convolution gradient) and small random jitters.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    t0 = time.time()
    total = deadline - t0
    it = 0
    while time.time() < deadline:
        it += 1
        # exponential temperature schedule
        frac = (time.time() - t0) / total
        temp = start_temp * ((end_temp / start_temp) ** frac)

        cand = cur_a.copy()
        move_type = np.random.rand()

        if move_type < 0.6:
            # targeted transfer from a high‑gradient index to a low‑gradient one
            conv = convolution_fft(cand, cand)
            k = int(np.argmax(conv))
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(cand)
            grad[mask] = 2.0 * cand[j[mask]]
            i_src = np.argmax(grad)                     # where the max‑conv is most sensitive
            # choose a destination with small gradient (or simply random)
            i_dst = np.random.randint(n)
            while i_dst == i_src:
                i_dst = np.random.randint(n)
            if cand[i_src] > 0.0:
                eps = 0.02 * cand[i_src]                # small fraction
                cand[i_src] -= eps
                cand[i_dst] += eps
        else:
            # tiny Gaussian jitter on all coordinates
            noise = np.random.randn(n) * 0.001
            cand += noise
            cand = np.maximum(cand, 0.0)

        # renormalise
        if cand.sum() == 0.0:
            continue
        cand /= cand.sum()

        new_score = compute_score(cand)
        delta = new_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / max(temp, 1e-12)):
            cur_a = cand
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()

        # occasional progress report
        if it % 5000 == 0:
            print(f"[Anneal] it {it:6d} | best {best_score:.6f} | temp {temp:.2e}")
    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 1 – pick a moderate sequence length (2048 works well; 4096 if time allows)
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 850.0:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2 – generate a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic shapes
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # Beta‑pdf families
    for alpha, beta in [
        (0.2, 0.2), (0.5, 0.5), (1.0, 1.0),
        (2.0, 2.0), (5.0, 5.0), (0.5, 2.0), (2.0, 0.5)
    ]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    # random Dirichlet (various concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # very sparse spikes – often a good starting point
    for k in (30, 50, 80):
        seeds.append(random_spike(n, k=k))

    # flat‑spectrum random‑phase seeds
    for _ in range(4):
        seeds.append(generate_flat_spectrum_random(n))

    # optional fallback from a previously discovered sequence
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        if len(fallback) == n:
            seeds.append(np.asarray(fallback, dtype=float))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3 – evaluate seeds and keep the best one
    # ------------------------------------------------------------------
    best_a = None
    best_val = float("inf")
    for s in seeds:
        a = np.asarray(s, dtype=float)
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[seed] improvement → score = {best_val:.6f}")

    cur_a = best_a.copy()
    cur_score = best_val

    # ------------------------------------------------------------------
    # 4 – allocate time for the optimisation phases
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    phase1_deadline = now + remaining * 0.25
    phase2_deadline = phase1_deadline + remaining * 0.15
    phase3_deadline = phase2_deadline + remaining * 0.15
    phase4_deadline = phase3_deadline + remaining * 0.20
    # the rest (if any) goes to a final light hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent
    # ------------------------------------------------------------------
    print("\n[Phase 1] Soft‑max gradient descent")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a, cur_score, phase1_deadline,
        beta_start=5.0,
        beta_end=5000.0,
        step_coef=0.0015,
    )
    if cur_score < best_val:
        best_val = cur_score
        best_a = cur_a.copy()
        print(f"[after Phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("\n[Phase 2] Adam optimiser (smooth surrogate)")
        cur_a, cur_score = adam_softmax_optimize(
            cur_a, cur_score, phase2_deadline,
            beta_start=5.0,
            beta_end=5000.0,
            lr0=0.01,
        )
        if cur_score < best_val:
            best_val = cur_score
            best_a = cur_a.copy()
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        print("\n[Phase 3] Sub‑gradient descent with momentum")
        cur_a, cur_score = subgradient_momentum_optimize(
            cur_a, cur_score, phase3_deadline,
            step0=0.07,
            mu=0.9,
        )
        if cur_score < best_val:
            best_val = cur_score
            best_a = cur_a.copy()
            print(f"[after Phase 3] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        print("\n[Phase 4] Simulated‑annealing polishing")
        cur_a, cur_score = simulated_annealing_optimize(
            cur_a, cur_score, phase4_deadline,
            start_temp=5e-4,
            end_temp=1e-6,
        )
        if cur_score < best_val:
            best_val = cur_score
            best_a = cur_a.copy()
            print(f"[after Phase 4] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final lightweight hill‑climb (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 5] Final hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=1_000_000)
        if cur_score < best_val:
            best_val = cur_score
            best_a = cur_a.copy()
            print(f"[after Phase 5] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
