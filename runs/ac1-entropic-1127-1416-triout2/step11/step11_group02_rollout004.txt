import time, numpy as np, math, random

def full_convolution(a):
    n = a.shape[0]
    m = 2*n -1
    nfft = 1<<((m-1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A*A, n=nfft)[:m]
    return conv

import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Sub‑gradient polishing (few iterations – fast improvement)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of the max‑convolution entry
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # tiny jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def simulated_annealing(start_a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Stochastic hill‑climbing with a temperature schedule.
    Moves consist of transferring a random fraction of mass from one
    index to another while keeping the total mass = 1.
    """
    a = start_a.copy()
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    # theoretical lower bound (uniform convolution)
    lower_bound = 2.0 * n / (2 * n - 1)   # ≈ 1 for large n
    est_gap = max(cur_score - lower_bound, 1e-6)
    T = max(est_gap * 0.3, 1e-4)          # initial temperature
    step_factor = 0.1                    # initial relative step size
    it = 0
    last_print = time.time()

    while time.time() < deadline:
        it += 1

        # pick a source proportionally to its current mass (more mass ⇒ more likely)
        i = np.random.choice(n, p=a)
        j = np.random.randint(n)
        if i == j:
            continue

        # propose a transfer from i to j
        delta = step_factor * a[i] * np.random.rand()
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate
        cand_score = compute_score(cand)
        dE = cand_score - cur_score

        # Metropolis acceptance
        if dE < 0.0 or np.random.rand() < math.exp(-dE / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        # linear cooling based on elapsed fraction
        elapsed = time.time() - (deadline - (deadline - time.time()))
        fraction = (time.time() - (deadline - (deadline - time.time()))) / max(
            deadline - (deadline - (deadline - time.time())), 1e-9
        )
        #Simple exponential cooling
        T *= 0.99995
        # occasionally shrink step size to allow fine tuning
        if it % 5000 == 0:
            step_factor = max(step_factor * 0.95, 0.001)

        # occasional progress print (does not affect optimisation)
        if time.time() - last_print > 30:
            print(f"[SA] iter {it:,d}, best score {best_score:.6f}, cur score {cur_score:.6f}, T={T:.6e}")
            last_print = time.time()

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 5.0

    n = 2048     # length of the step function
    best_seq = None
    best_val = float("inf")

    # 0. Fallback (previous best construction, if available)
    # ------------------------------------------------------------------

        val = evaluate_sequence(fallback)
        print(f"[fallback] evaluate_sequence = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Simple families – power‑law, exponential, symmetric U‑shape
    # ------------------------------------------------------------------
    def make_u_shape(p):
        """Symmetric U‑shape: decay from both ends with exponent p."""
        left = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        right = left[::-1]
        a = left + right
        a = np.maximum(a, 0.0)
        a /= a.sum()
        return a

    sweep_deadline = deadline - 550.0          # reserve ~550 s for later phases
    if time.time() < sweep_deadline:
        best_a = None
        best_score = float("inf")

        # power‑law
        for p in np.linspace(0.10, 2.00, 96):
            if time.time() >= sweep_deadline:
                break
            a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            a /= a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a

        # geometric decay
        for r in np.linspace(0.40, 0.99, 120):
            if time.time() >= sweep_deadline:
                break
            a = (r ** np.arange(n, dtype=np.float64))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a

        # symmetric U‑shape
        for p in np.linspace(0.10, 2.00, 96):
            if time.time() >= sweep_deadline:
                break
            a = make_u_shape(p)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a

        if best_a is not None:
            print(f"[global sweep] best score = {best_score:.6f}")
            if best_score < best_val:
                best_val = best_score
                best_seq = best_a.tolist()

    # ------------------------------------------------------------------
    # 2. Sub‑gradient polishing (fast convex‑ish improvement)
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 250.0
    if best_seq is not None and time.time() < subgrad_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, subgrad_deadline, step0=0.10)
        sc = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {sc:.6f}")
        if sc < best_val:
            best_val = sc
            best_seq = a_polished.tolist()

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing refinement (main work‑horse)
    # ------------------------------------------------------------------
    sa_deadline = deadline - 5.0
    if best_seq is not None and time.time() < sa_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_sa = simulated_annealing(a_start, sa_deadline)
        sc = compute_score(a_sa)
        print(f"[SA] final SA score = {sc:.6f}")
        if sc < best_val:
            best_val = sc
            best_seq = a_sa.tolist()

    # ------------------------------------------------------------------
    # 4. Safeguard – uniform distribution if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Report using the official evaluation routine
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
