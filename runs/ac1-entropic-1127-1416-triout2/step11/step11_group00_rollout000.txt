"""
nonnegative_step_optimizer.py

A faster, higher‑precision optimizer for the “evaluate_sequence” objective.
The algorithm consists of three stages

0.  Very cheap global sweep over a few parametric families
    (beta‑U‑shapes, power‑laws, exponentials and random Dirichlet draws)
    – the best 5 candidates are kept.

1.  Smooth‑max gradient descent (Adam) on the exact convolution.
    The true max is approximated by a soft‑max (log‑sum‑exp) with a
    temperature β that is increased during the run.  After each step the
    vector is projected onto the probability simplex, guaranteeing
    non‑negativity and sum = 1.

2.  A short simulated‑annealing polishing phase (mass‑transfer moves)
    to escape possible shallow plateaus left by the gradient phase.

All work is bounded by the supplied wall‑clock budget (≈ 1000 s);
intermediate best scores are printed for monitoring.  The returned
object is a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of length power‑of‑2."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised (sum=1) vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the cheap global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """Return up to ``max_keep`` vectors with the lowest exact score."""
    best = []  # list of (score, vector)

    def consider(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Beta‑U‑shape – go deeper into the (α,β)<1 regime
    al_vals = np.concatenate([np.linspace(0.05, 0.3, 6),
                              np.linspace(0.4, 2.0, 9)])
    be_vals = al_vals
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    # 2) Power‑law (including steeper tails)
    for p in np.linspace(0.2, 3.0, 151):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # 3) Exponential (wide range of decay rates)
    for lam in np.linspace(0.00005, 0.04, 400):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 4) Random Dirichlet draws with a variety of concentrations
    concs = [0.05, 0.1, 0.2, 0.5, 1.0]
    for conc in concs:
        for _ in range(120):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            consider(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# 1. Smoothed‑max gradient descent (Adam + simplex projection)
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max of β·conv (returns a probability vector)."""
    v = beta * conv
    m = np.max(v)
    expv = np.exp(v - m)
    return expv / np.sum(expv)


def gradient_softmax_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of 2·n·S_β(a) where
    S_β(a) = (1/β)·log Σ exp(β·conv_i) .
    Returns a dense vector of size n.
    """
    n = a.shape[0]
    conv = full_convolution(a)                     # length 2n‑1
    w = softmax_weights(conv, beta)                # same length

    # ------- FFT setup (linear convolutions) -------------------------
    la = len(a)
    lw = len(w)
    L = 1 << ((la + lw - 1 - 1).bit_length())  # next power‑of‑2 ≥ 3n‑2

    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)

    # term2 = (w * a)[k]   (standard linear convolution)
    conv_w_a = np.fft.irfft(W * A, n=L)[: la + lw - 1]   # length 3n‑2

    # term1 = correlation a ⋆ w  = ifft( A * conj(W) )
    corr = np.fft.irfft(A * np.conj(W), n=L)[: la + lw - 1]

    # gradient for each index 0 … n‑1
    grad = 2.0 * (conv_w_a[:n] + corr[:n])
    return grad


def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  beta_start: float = 20.0,
                  beta_end: float = 200.0,
                  lr_start: float = 0.05,
                  lr_end: float = 0.008) -> np.ndarray:
    """
    Adam‑style projected gradient descent on the smoothed max.
    The temperature β is increased linearly from ``beta_start`` to ``beta_end``.
    The learning rate is decreased linearly from ``lr_start`` to ``lr_end``.
    Returns the best vector (lowest exact score) seen during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        it += 1
        # linear schedules
        frac = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - (deadline - time.time())))
        # (the above formula is deliberately simple – we just compute fraction of elapsed time)
        elapsed = max(0.0, time.time() - (deadline - (deadline - time.time())))

        # simpler: compute progress as it / max_iters (unknown)
        # Instead use time fraction directly:
        prog = (time.time() - (deadline - (deadline - time.time())))/ (deadline - (deadline - time.time()))
        prog = min(1.0, prog)  # safety
        beta = beta_start + prog * (beta_end - beta_start)
        lr = lr_start + prog * (lr_end - lr_start)

        # gradient of smoothed max
        g = gradient_softmax_max(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # occasional exact evaluation (every 250 iterations)
        if it % 250 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[adam] it {it:6d}  β={beta:6.2f}  lr={lr:7.5f}  score={cur_val:.6f}")

        # early exit if gradient norm is tiny
        if np.linalg.norm(step) < 1e-12:
            break

    return best_a


# ----------------------------------------------------------------------
# 2. Simulated annealing polishing (identical to the reference version)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # global jitter every few thousand steps
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     n: int = 2048,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for the wrapper that calls us
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. optional fallback (if a previous construction is present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Very cheap global exploration (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline - 300.0, time.time() + 150.0)
    if time.time() < baseline_deadline:
        print("[search] running baseline exploration")
        candidates = baseline_search(n, baseline_deadline, max_keep=5)
        for vec in candidates:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
        print(f"[baseline] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smoothed‑max Adam optimisation (≈ 400 s)
    # -----------------------------------------------------------------
    adam_deadline = min(deadline - 120.0, time.time() + 400.0)
    if best_seq is not None and time.time() < adam_deadline:
        print("[adam] starting smooth‑max descent")
        a0 = np.array(best_seq, dtype=float)
        a_refined = adam_optimize(a0, adam_deadline)
        val = compute_score(a_refined)
        print(f"[adam] refined exact score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 3. Simulated‑annealing polishing (remaining time)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        print("[anneal] final stochastic polishing")
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final exact score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – if everything failed, fall back to a uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final reporting using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
