"""
Propose a step‑function sequence that minimises the evaluation function
``evaluate_sequence``.  The optimisation proceeds in three phases:

1.  **Fast analytic seeding** – scan a few low‑dimensional families
    (symmetric product, left‑heavy power, exponential U‑shape) on a
    reasonably large length (up to 8192) and keep the best candidate.

2.  **Smooth‑max L‑BFGS refinement** – minimise a log‑sum‑exp
    (smooth) approximation of the max‑convolution.  The parameters are
    optimised in the soft‑max (log‑probability) space, which enforces
    non‑negativity and the simplex constraint automatically.  A schedule
    of increasing smoothing parameters `β` drives the solution towards
    the true max.

3.  **Projected sub‑gradient descent** – after the L‑BFGS phase we
    perform a long run of cheap projected sub‑gradient steps directly
    on the true max‑convolution (the non‑smooth objective).  The step
    size decays as 1/√t, and after every step we project onto the
    probability simplex and enforce symmetry (a_i = a_{n‑1‑i}).  This
    final stage typically yields the last few hundredths of improvement
    needed to beat the target bound.

All phases respect the global time budget (default 1000 s) and the best
sequence seen so far is constantly updated.  The function returns a
plain ``list[float]`` that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (fast for large n)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x : x>=0, sum x = 1}.
    """
    n = v.shape[0]
    # shift to make the problem centered
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (without the factor 2·n) for a probability vector.
    Returns max(conv) where conv = a * a.
    """
    conv = convolution_fft(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for fast seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # catch potential infinities when alpha=0
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth max and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where

        f   = smooth max = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z  (gradient w.r.t. the soft‑max parameters)

    The gradient is computed analytically via convolution with the soft‑max
    weights on the convolution values.
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta                 # smooth approximation of max
    w = expw / sumexp                                # normalised weights (shape 2n‑1)

    # gradient of the weighted sum of convolutions w.r.t. a:
    #   d/da  Σ_k w_k conv_k = 2 * (a_rev * w)[central slice]
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation with a schedule of β values
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS runs on the smooth‑max surrogate.
    Returns the best probability vector (still on the simplex) found
    within the given time budget.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = 2.0 * n * compute_score(a)          # exact objective (2·n·max)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        # initialise optimisation variables (log‑probabilities)
        z0 = np.log(a + 1e-12)

        # wrapper returning smooth objective and gradient
        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun, z0, method="L-BFGS-B", jac=True, options=opts)
        except Exception as e:
            print(f"  L‑BFGS failure: {e}")
            continue

        a = softmax(res.x)

        # enforce symmetry (helps numerical stability)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        cur_score = 2.0 * n * compute_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        step0: float = 0.4,
                        max_iters: int = 10_000_000):
    """
    Simple projected sub‑gradient method directly on max(conv).
    Returns the best vector and its exact objective (2·n·max).
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    maxc = compute_score(a)
    best_score = 2.0 * n * maxc

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # current convolution and its max
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # active set (indices attaining the max, tolerance for floating point)
        active = np.where(conv >= maxc - 1e-14)[0]
        # weight vector for sub‑gradient (= uniform over the active set)
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)

        # gradient of the max w.r.t. a = 2 * (a_rev * w)[central slice]
        conv_rev = convolution_fft(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # step size (classic 1/√t schedule)
        step = step0 / np.sqrt(it)

        # gradient step and projection onto simplex
        a = a - step * grad
        a = project_to_simplex(a)

        # enforce perfect symmetry after projection
        a = (a + a[::-1]) * 0.5

        cur_max = compute_score(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            # optionally accelerate step size a little when we improve
            step0 = min(step0 * 1.03, 1.0)

        # occasional progress report
        if it % 200_000 == 0:
            print(f"[{time.time():.1f}] sub‑grad iter {it:,} best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Short simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    50 % swaps, 50 % mass transfers, temperature decays linearly.
    Returns the best vector seen and its exact objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = 2.0 * n * compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        cand_score = 2.0 * n * compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """

    ``evaluate_sequence`` under the simplex constraint.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------

    # -----------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_obj = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_obj:.6f}")
        if fb_obj < best_score:
            best_score = fb_obj
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose problem size
    # -----------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] problem size n = {n}")

    # -----------------------------------------------------------
    # 2. Scan analytic families (cheap, gives good start points)
    # -----------------------------------------------------------
    # list of families to explore
    families = []

    # product family (α)
    for alpha in np.linspace(0.0, 3.0, 13):
        families.append(("sym‑α", symmetric_product_family(n, alpha)))

    # left‑heavy power (β)
    for beta in np.linspace(0.0, 3.0, 13):
        families.append(("left‑β", left_heavy_power_family(n, beta)))

    # exponential U‑shape (γ)
    for gamma in np.linspace(0.0, 4.0, 11):
        families.append(("exp‑γ", exp_u_family(n, gamma)))

    # a few random Dirichlet draws for diversity
    for _ in range(8):
        vec = np.random.rand(n)
        vec = vec / vec.sum()
        families.append(("rand", vec))

    for name, vec in families:
        if time.time() >= deadline - 150.0:
            break
        sc = 2.0 * n * compute_score(vec)   # exact objective
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best ({name}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. L‑BFGS refinement on the smooth‑max surrogate
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        a_start = np.array(best_seq, dtype=np.float64)
        # a fairly dense β schedule (log‑space)
        betas = [0.5, 1, 2, 5, 10, 20, 40, 80, 150, 300,

        a_opt, sc_opt = l_bfgs_optimize(a_start,
                                        betas=betas,
                                        max_iter_per_stage=300,
                                        deadline=deadline - 80.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Projected sub‑gradient descent (long fine‑tuning phase)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = subgradient_descent(a_start,
                                            deadline=deadline - 10.0,
                                            step0=0.3,
                                            max_iters=5_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Final short simulated annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,

        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------

    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
