def gradient_from_conv_weights(a, w):

    # a shape (n), w shape (2n-1)
    n = a.shape[0]
    m = w.shape[0]  # = 2n-1

    L = 1 << ((n + m - 2).bit_length())   # next pow2 >= n+m-1

    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)

    # correlation: a cross-correlate w yields r = ifft(A * conj(W))
    r = np.fft.irfft(Af * np.conj(Wf), n=L)

    # first n entries correspond to sum_j a_j w[i+j]
    g = 2.0 * r[:n]

    return g

def adam_optimize(a0, deadline, init_lr=0.2, alpha_start=5.0, alpha_max=200.0, alpha_inc=1.001):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_score = compute_score(a)
    L = 1 << ((n + (2*n-1) - 2).bit_length())  # same as before
    α = alpha_start
    while time.time() < deadline:
        t += 1
        # compute FFT of a
        A = np.fft.rfft(a, n=L)

        # compute convolution
        conv = np.fft.irfft(A * np.conj(A), n=L)[:2*n-1]

        max_conv = conv.max()
        # compute softmax weights
        # stability: subtract max_conv or use alpha scaling
        logits = α * (conv - max_conv)  # max logit = 0
        w = np.exp(logits)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w /= w_sum

        # compute gradient correlation a with w
        W = np.fft.rfft(w, n=L)
        # gradient = 2 * correlation a ⋆ w = 2 * irfft(A * conj(W))[:n]
        grad = 2.0 * np.fft.irfft(A * np.conj(W), n=L)[:n]

        # adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a = a - init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = project_to_simplex(a)

        # compute objective
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] iter {t} improved score {cur_score:.6f}")

        # adjust learning rate maybe small gradually
        if t % 500 == 0:
            # Maybe reduce lr
            init_lr *= 0.9
            # increase α
            α = min(α * alpha_inc, alpha_max)

        # occasional jitter
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (init_lr * 0.001 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

Enhanced non‑negative step‑function optimiser.

The algorithm consists of three phases:

1. **Fine‑grained parametric sweep** – exhaustive search over power‑law,
   exponential and Beta‑shaped families (including very small shape
   parameters) plus a pool of random Dirichlet draws.  The best few
   candidates (according to the exact objective 2·n·max(conv(a,a))) are
   retained.

2. **Soft‑max Adam polishing** – starting from the retained candidates we
   minimise a smooth approximation of the max‑convolution
   (log‑sum‑exp).  The gradient of this surrogate can be written as a
   correlation, which we evaluate by FFT.  Adam (first‑moment /
   second‑moment) together with Euclidean projection onto the probability
   simplex provides a fast and stable descent.

3. **Stochastic hill‑climb** – a final simulated‑annealing pass
   (mass‑transfer moves) refines the best solution found.

All phases respect the overall 1000 s wall‑clock budget.  The function
`propose_candidate` is the required entry point.
"""

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # The projection of the zero vector is the uniform distribution.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full, length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of parametric sequences
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}.  Positive p gives decreasing, negative p increasing."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i).  Positive lam → decreasing."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α-1}·(n‑i)^{β-1} (discrete Beta density)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global baseline exploration (fine grid)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 10,
) -> list[np.ndarray]:
    """Return up to ``max_keep`` best vectors found by exhaustive sweeps."""
    best = []  # list of (score, vector)

    def maybe_add(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)

        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep (covers decreasing and increasing shapes)
    for p in np.linspace(-0.5, 1.2, 161):  # finer grid, includes small negative p
        if time.time() >= deadline:
            break
        maybe_add(generate_powerlaw(n, p))

    # 2) Exponential sweep
    for lam in np.linspace(0.00001, 0.10, 401):
        if time.time() >= deadline:
            break
        maybe_add(generate_exponential(n, lam))

    # 3) Beta sweep (dense grid, includes very sharp edge shapes)
    al_vals = np.linspace(0.05, 2.0, 40)          # step ≈0.05
    be_vals = np.linspace(0.05, 2.0, 40)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            maybe_add(generate_beta(n, alpha, beta))

    # 4) Random Dirichlet draws with several concentration levels
    conc_levels = [0.05, 0.15, 0.5, 1.0, 2.0]
    draws_per = 120  # total ≈ 600 samples
    for conc in conc_levels:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            maybe_add(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Soft‑max + Adam optimisation
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    init_lr: float = 0.12,
    alpha_start: float = 5.0,
    alpha_target: float = 250.0,
) -> np.ndarray:
    """
    Adam‑based minimisation of a smooth surrogate of max(conv(a,a)).
    The surrogate uses a soft‑max (log‑sum‑exp) with a temperature that
    increases from ``alpha_start`` to ``alpha_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    L = 1 << ((n + (2 * n - 1) - 2).bit_length())  # next power of two ≥ 3·n‑3

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = a.copy()
    best_score = compute_score(a)

    # schedule for the soft‑max temperature
    total_iters_est = 20000  # rough upper bound for step size scheduling
    alpha = alpha_start

    while time.time() < deadline:
        t += 1

        # FFT of the current vector
        A = np.fft.rfft(a, n=L)

        # linear convolution a*a (full length 2n‑1)
        conv_full = np.fft.irfft(A * np.conj(A), n=L)[: 2 * n - 1]

        # --- soft‑max weights ---
        max_c = conv_full.max()
        # stability: shift by max_c before exponentiation
        logits = alpha * (conv_full - max_c)
        w = np.exp(logits)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv_full, 1.0 / conv_full.size)
        else:
            w /= w_sum

        # --- gradient (correlation a ⋆ w) ---
        W = np.fft.rfft(w, n=L)
        grad = 2.0 * np.fft.irfft(A * np.conj(W), n=L)[:n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a -= init_lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps sum=1 and a≥0)
        a = project_to_simplex(a)

        # tracking best candidate
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] iter {t:5d}  score = {cur_score:.6f}")

        # schedule adjustments
        if t % 500 == 0:
            # slowly increase the soft‑max temperature
            alpha = min(alpha * 1.03, alpha_target)

        if t % 2000 == 0:
            # tiny jitter to escape possible plateaus
            jitter = np.random.randn(n) * (init_lr * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 1e-3,
    final_temp: float = 1e-7,
    cooling: float = 0.9994,
) -> np.ndarray:
    """Simple Metropolis hill‑climb with temperature decay."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    it = 0
    step_scale = 0.04
    step_min = 1e-6

    while time.time() < deadline:
        it += 1
        # pick a source index with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        # pick a different target index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            # accept
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_min)

        # temperature schedule
        T = max(final_temp, T * cooling)

        # occasional global jitter
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.008)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Fine‑grained global exploration
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + (budget_s * 0.30))
    if time.time() < baseline_deadline:
        candidates = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam polishing (soft‑max minimisation of the max‑conv)
    # -----------------------------------------------------------------
    polish_deadline = min(deadline, time.time() + (budget_s * 0.45))
    if best_seq is not None and time.time() < polish_deadline:
        # keep only the top few candidates for polishing
        top_candidates = [np.array(best_seq, dtype=float)]
        # also try a few other high‑rank baselines if we have them
        if 'candidates' in locals():
            for vec in candidates[:4]:
                top_candidates.append(vec.copy())
        for idx, a0 in enumerate(top_candidates):
            if time.time() >= polish_deadline:
                break
            a_refined = adam_optimize(a0, polish_deadline)
            val = compute_score(a_refined)
            print(f"[adam] candidate {idx} refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 3. Simulated annealing final pass
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 2.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safety net – uniform vector if nothing else succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
