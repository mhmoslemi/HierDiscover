def hill_climb_targeted(start_a, deadline, max_steps=10_000_000):
    a = start_a.copy()
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    # initial temperature relative to score:
    T0 = cur_score * 1e-4
    step_scale = 0.15
    # run steps
    steps = 0
    while steps < max_steps and time.time() < deadline:
        steps += 1
        # compute conv and max index
        conv = convolution_fft(a, a)
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # contributions
        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]
        total_contrib = contrib.sum()
        if total_contrib == 0:
            # fallback random move
            i = np.random.randint(n)
        else:
            # probabilities proportional to contribution
            # add small epsilon to avoid zero
            p = contrib / total_contrib
            i = np.random.choice(n, p=p)
        # choose target j2 distinct
        j2 = np.random.randint(n-1)
        if j2 >= i:
            j2 += 1
        # transfer mass
        delta = step_scale * a[i]
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j2] += delta
        # enforce nonnegativity via projection
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        delta_score = cand_score - cur_score
        # temperature decreasing linearly
        T = T0 * (1.0 - steps / max_steps)
        if delta_score < 0 or np.random.rand() < math.exp(-delta_score / (T + 1e-15)):
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97
        # occasional perturbation if no improvement for many steps
        if steps % 50000 == 0:
            # small random perturbation to escape plateaus
            if np.random.rand() < 0.2:
                pert = np.random.rand(n)
                pert = pert / pert.sum()
                a = project_to_simplex(0.9 * a + 0.1 * pert)
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm – three phases
------------------------
0. Load a fallback sequence if one exists.
1. Quick analytic / random scan to obtain a decent starter.
2. A cheap projected sub‑gradient descent (with the *correct* sub‑gradient)
   for a few hundred seconds.
3. A long simulated‑annealing hill‑climb that *targets* the indices that
   actually contribute to the current maximal convolution entry.
   Moves consist of a small mass transfer from a high‑contributing index
   to a random partner, with a linearly‑cooled temperature.
All phases respect the global wall‑clock limit (default 1000 s) and
the vector is projected onto the probability simplex after every change.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (with correct sub‑gradient)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     step0: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the true max‑objective.
    Uses the exact sub‑gradient of the maximal convolution entry.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_idx = int(conv.argmax())

        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(a)
        # correct sub‑gradient: g_i = a_i + a_{k-i} (or 2*a_i on diagonal)
        g[mask] = a[idx[mask]] + a[j[mask]]

        step = step0 / math.sqrt(it)
        a = project_to_simplex(a - step * g)

        if it % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement step (simple version)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weights: list[float]) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M  (all k),   g ≥ 0.
    Afterwards return the convex mixture f_new = (1‑t)·f + t·g_norm
    where t is chosen from *mix_weights* to minimise the true score.
    """
    n = f.shape[0]
    m_len = 2 * n - 1

    # Build convolution matrix A where (A @ g) = f * g
    A = np.zeros((m_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ↔ minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(m_len, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f, compute_score(f)

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, compute_score(f)

    g_norm = g / sum_g

    # line‑search for the best mixing weight
    best_f = f
    best_sc = compute_score(f)
    for t in mix_weights:
        cand = (1.0 - t) * f + t * g_norm
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_f = cand
    return best_f, best_sc


# ----------------------------------------------------------------------
# Targeted simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_targeted(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 8_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk that preferentially transfers mass away from indices that
    contribute most to the current maximal convolution entry.
    """
    a = start_a.copy()
    n = a.size
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    # initialise a small temperature proportional to current score
    T0 = cur_score * 5e-5
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_idx = int(conv.argmax())

        # contributions a[i] * a[max_idx-i] (zero where out‑of‑range)
        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]
        total_contrib = contrib.sum()

        # pick a source index proportional to its contribution
        if total_contrib > 0:
            p = contrib / total_contrib
            i = np.random.choice(n, p=p)
        else:
            i = np.random.randint(n)

        # pick a distinct target index uniformly at random
        j2 = np.random.randint(n - 1)
        if j2 >= i:
            j2 += 1

        # transfer a fraction of the mass from i to j2
        delta = step_scale * a[i]
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j2] += delta

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta_score = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)   # linear cooling
        if delta_score < 0 or np.random.rand() < math.exp(-delta_score / (T + 1e-18)):
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # occasional small random perturbation to escape plateaus
        if step % 50000 == 0 and np.random.rand() < 0.2:
            pert = np.random.rand(n)
            pert /= pert.sum()
            a = project_to_simplex(0.9 * a + 0.1 * pert)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Load fallback (if present)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Determine working length (fallback length or 2048)
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2. Quick analytic / random scan (≈20 s)
    # ------------------------------------------------------------------
    if time.time() + 20.0 < deadline:
        print(f"[{time.time():.1f}] quick analytic scan (n={n})")
        alphas = np.linspace(0.0, 12.0, 13)
        betas = np.linspace(0.0, 12.0, 13)
        gammas = np.linspace(0.0, 6.0, 13)

        for alpha in alphas:
            if time.time() >= deadline - 5.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym α={alpha:.2f}) = {best_score:.6f}")

        for beta in betas:
            if time.time() >= deadline - 5.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left β={beta:.2f}) = {best_score:.6f}")

        for gamma in gammas:
            if time.time() >= deadline - 5.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp γ={gamma:.2f}) = {best_score:.6f}")

        # a handful of random starts
        for _ in range(60):
            if time.time() >= deadline - 5.0:
                break
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise starter to a probability vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. Sub‑gradient descent – allocate up to ~380 s
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 150.0:
        sub_time = min(380.0, remaining * 0.5)
        sub_deadline = time.time() + sub_time
        print(f"[{time.time():.1f}] Sub‑gradient descent start (≈{sub_time:.0f}s)")
        a, sc = subgrad_optimize(a, sub_deadline, step0=0.08)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Light LP‑refinement (≈20 s)
    # ------------------------------------------------------------------
    if time.time() + 20.0 < deadline:
        print(f"[{time.time():.1f}] LP‑refinement start")
        mix_weights = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        for it in range(8):
            if time.time() >= deadline - 5.0:
                break
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new, sc = lp_refine_step(a, M, mix_weights)
            if sc < best_score - 1e-9:
                best_score = sc
                a = a_new
                print(f"[{time.time():.1f}] LP step {it} improved to {best_score:.6f}")
            else:
                # tighten mix if not improving
                mix_weights = [max(w * 0.5, 0.001) for w in mix_weights]

    # ------------------------------------------------------------------
    # 5. Final aggressive targeted hill‑climb (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (targeted SA) start")
        a, sc = hill_climb_targeted(a, deadline, max_steps=8_000_000)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
