2. Generate a diverse set of seed sequences:
   * the provided fallback `height_sequence_1`
   * uniform, several analytic families (product, left‑heavy power, exponential‑U)
   * random Dirichlet draws with several concentration parameters.
   Keep the best few seeds (according to the exact objective).
3. Refine the best seed with a **projected exponentiated‑gradient**
   descent on the exact max‑convolution (one FFT per iteration).  
   The update moves mass away from positions that contribute most to the
   currently maximal convolution value.
4. Further polish the result with **L‑BFGS** on a smooth‑max surrogate
   (log‑sum‑exp) using a schedule of increasing smoothing parameters.
   No artificial symmetry is enforced – the optimizer is free to exploit
   any asymmetry that lowers the max.
5. Finish with a short **simulated‑annealing hill‑climb** (swap / mass‑transfer
   moves) to escape any remaining plateaus.
6. Return the best sequence found (as a plain Python list of floats).

All steps respect the wall‑clock deadline and continuously keep the best
sequence encountered.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation for a probability vector (∑a = 1):
    2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for cheap seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """ a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex. """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """ a_i ∝ (i+½)^{‑β}, monotone decreasing. """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def random_dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using a Dirichlet distribution."""
    w = rng.gamma(conc, 1.0, size=n)
    if w.sum() == 0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w.sum()
    return w


# ----------------------------------------------------------------------
# Exact max‑convolution and its sub‑gradient
# ----------------------------------------------------------------------


def max_conv_and_grad(a: np.ndarray):
    """
    Compute the current maximum of a*a and a sub‑gradient of the max w.r.t. a.
    Gradient is given by
        g_j = 2 * a[i_max - j]   if 0 ≤ i_max - j < n, else 0,
    where i_max is the index attaining the maximal convolution value.
    """
    conv = convolution_fft(a, a)
    maxc = conv.max()
    idx = conv.argmax()
    n = a.shape[0]
    shift = idx - (n - 1)               # offset relative to zero shift
    grad = np.zeros_like(a)

    if shift >= 0:
        # grad[shift + t] = 2 * a_rev[t]  for t = 0 .. n-1-shift
        grad[shift:] = 2.0 * a[::-1][: n - shift]
    else:
        s = -shift
        # grad[t] = 2 * a_rev[s + t] for t = 0 .. n-1-s
        grad[: n - s] = 2.0 * a[::-1][s:]

    return maxc, grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (exact max)
# ----------------------------------------------------------------------


def exp_grad_refine(a_start: np.ndarray,
                    deadline: float,
                    max_iters: int = 300_000,
                    eta0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent on the exact max‑convolution.
    The update is a_i ← a_i·exp(−η·g_i)  followed by simplex projection.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        _, grad = max_conv_and_grad(a)
        eta = eta0 / np.sqrt(it)

        # multiplicative update (clip exponent to avoid overflow)
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -30.0, 30.0)
        a = a * np.exp(log_factor)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / s

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]   EG iter {it:,} → score {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Smooth approximation of max(conv) and its gradient w.r.t. the soft‑max
    parameters `z`.  Returns (f, grad_z) where

        f   = maxc + log(∑exp(β·(conv−maxc))) / β
        grad_z = a·grad_a − (a·grad_a)·a   (back‑prop through soft‑max)
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta                 # smooth max
    w = expw / sumexp                                # soft‑max weights over positions

    a_rev = a[::-1]
    # weighted convolution of a_rev with the soft‑max weights
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1: n - 1 + n]       # gradient w.r.t a (shape n)

    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS runs on the smooth‑max surrogate.
    No symmetry enforcement – the optimizer may exploit asymmetry.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        # initialise the optimisation variables
        z0 = np.log(a + 1e-12)

        def fun(z):
            return smooth_max_and_grad(z, beta, n)

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as e:
            print(f"   L‑BFGS failure: {e}")
            continue

        a = softmax(res.x)
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]   improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simple SA walk on the simplex:
        * 50 % swap two entries,
        * 50 % transfer a fraction of mass from i to j.
    Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4          # initial temperature proportional to score
    step_scale = 0.12               # controls how much mass is moved

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # accept if better or according to Metropolis criterion
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T0 * (1.0 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # adapt step size slowly towards larger moves when improving
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # shrink step scale when move is rejected
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence minimising
    `evaluate_sequence` under the simplex constraint.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Fallback using the previously discovered sequence (if any)
    # -----------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fb_score = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose problem size
    # -----------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 650 else 2048
    print(f"[{time.time():.1f}] using length n = {n}")

    # -----------------------------------------------------------
    # 2. Build a pool of seed sequences
    # -----------------------------------------------------------
    seed_cands = []

    # uniform seed
    uniform = np.full(n, 1.0 / n)
    seed_cands.append(uniform)

    # analytic families (coarse grid)
    for alpha in np.linspace(0.0, 3.0, 7):
        if time.time() >= deadline - 120.0:
            break
        vec = symmetric_product_family(n, alpha)
        seed_cands.append(vec)

    for beta in np.linspace(0.0, 3.0, 7):
        if time.time() >= deadline - 120.0:
            break
        vec = left_heavy_power_family(n, beta)
        seed_cands.append(vec)

    for gamma in np.linspace(0.0, 4.0, 9):
        if time.time() >= deadline - 120.0:
            break
        vec = exp_u_family(n, gamma)
        seed_cands.append(vec)

    # random Dirichlet draws
    for conc in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(8):
            if time.time() >= deadline - 120.0:
                break
            vec = random_dirichlet_seed(n, conc, rng)
            seed_cands.append(vec)

    # evaluate all seeds and keep the best K of them
    scored_seeds = []
    for idx, vec in enumerate(seed_cands):
        sc = compute_score(vec)
        scored_seeds.append((sc, vec))
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new initial best (seed {idx}) = {best_score:.6f}")

    # keep only the top few (e.g. 10) seeds for refinement
    scored_seeds.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored_seeds[:10]]

    # -----------------------------------------------------------
    # 3. Exponentiated‑gradient refinement on the best seed
    # -----------------------------------------------------------
    if top_seeds and time.time() < deadline - 300.0:
        a0 = top_seeds[0].copy()
        max_iters = 400_000
        print(f"[{time.time():.1f}] EG refinement start (max_iters={max_iters})")
        a_ref, sc_ref = exp_grad_refine(a0,
                                        deadline=deadline - 150.0,
                                        max_iters=max_iters,
                                        eta0=0.3)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}] after EG refinement best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. L‑BFGS optimisation on smooth‑max surrogate
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        a_start = np.array(best_seq, dtype=np.float64)
        betas = [5, 30, 150, 500, 2000, 8000, 30000, 120000]
        a_opt, sc_opt = l_bfgs_optimize(a_start,
                                        betas=betas,
                                        max_iter_per_stage=250,
                                        deadline=deadline - 70.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=3_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Safety fallback (should never happen)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
