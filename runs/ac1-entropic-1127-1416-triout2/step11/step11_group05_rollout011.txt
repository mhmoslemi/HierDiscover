"""
Improved step‑function search.

The algorithm proceeds in several phases while respecting a wall‑clock
budget (default 1000 s).

1. **Fast analytic families** – a cheap scan of three one‑parameter
   families (symmetric product, left‑heavy power, exponential U‑shape)
   to obtain a decent initial candidate.

2. **Parametric log‑linear model** – the log of the coefficients is
   expressed as a linear combination of a few simple basis functions
   (log‑product, log‑index, distance‑to‑edge, cosine).  The few
   parameters are optimised with L‑BFGS‑B on the exact objective
   (max‑convolution).  This often discovers a better shape than the
   pure families.

3. **Multi‑restart L‑BFGS on the smooth‑max surrogate** – the smooth
   approximation of the maximum (log‑sum‑exp) is minimised using the
   exact gradient derived analytically.  Several random simplex starts
   are processed, each with an increasing schedule of smoothing
   parameters β.  The best solution across all restarts is retained.

4. **Final simulated‑annealing hill‑climb** – a short SA walk with
   symmetric “swap” and “mass‑transfer” moves refines the best
   sequence and removes any remaining asymmetry.

All stages keep the globally best sequence seen so far and stop
shortly before the deadline to avoid a timeout.  The returned value is
a plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑ a = 1).

    score = 2 * n * max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection and softmax
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (coarse seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i)), on the simplex."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Log‑linear parametric family
# ----------------------------------------------------------------------


def generate_shape_from_params(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a symmetric probability vector from a low‑dimensional
    parameter vector.

    log a[i] = Σ_k params[k] * φ_k(i)   (φ are simple basis functions)
    The result is exponentiated and projected onto the simplex,
    then symmetrised.
    """
    i = np.arange(n, dtype=np.float64)

    # Basis functions (all symmetric)
    phi0 = np.log((i + 0.5) * (n - 0.5 - i) + 1e-12)        # product term
    phi1 = np.log(i + 0.5)                                 # index term
    phi2 = np.minimum(i, n - 1 - i).astype(np.float64)     # distance to edge
    phi3 = np.cos(np.pi * i / (n - 1))                     # cosine term
    # Stack them
    Phi = np.stack([phi0, phi1, phi2, phi3])               # shape (4, n)

    log_a = np.dot(params, Phi)                           # (n,)
    # Stabilise exponentiation
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    a = a / a.sum()
    # Enforce exact symmetry
    a = (a + a[::-1]) * 0.5
    return a


def param_optimize_shape(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Optimise the 4‑parameter log‑linear model (low‑dimensional)."""
    best_a = None
    best_score = float("inf")
    # bounds chosen wide enough to explore diverse shapes
    bounds = [(-10.0, 10.0)] * 4
    # we run a few restarts with different seeds
    for restart in range(5):
        if time.time() >= deadline - 5.0:
            break
        # start near zero (uniform) with a tiny random perturbation
        init = np.random.uniform(-0.5, 0.5, size=4)
        res = minimize(
            lambda w: compute_score(generate_shape_from_params(w, n)),
            init,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 250, "disp": False},
        )
        a = generate_shape_from_params(res.x, n)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] param‑opt restart {restart} improved to {best_score:.6f}")
    # fallback to uniform if something went wrong
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = compute_score(best_a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_obj(z: np.ndarray, beta: float, n: int):
    """
    Return (f, grad) where f is a smooth approximation of max(conv)
    and grad = ∂f/∂z for the soft‑max parametrisation a = softmax(z).
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights over convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta
    w = expw / sumexp

    # gradient w.r.t. a (see derivation in the description)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 2n‑1
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # shape (n,)

    # back‑prop through the soft‑max (Jacobian of softmax)
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS with a schedule of β values
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Run successive L‑BFGS minimisations of the smooth‑max surrogate,
    gradually sharpening the approximation (increasing β).  The routine
    respects a global deadline and returns the best vector found.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        # initialise z = log(a)  (softmax parametrisation)
        eps = 1e-12
        z0 = np.log(a + eps)

        # SciPy minimise will call our objective with extra args (beta,n)
        res = minimize(
            smooth_max_obj,
            z0,
            args=(beta, n),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter_per_stage, "disp": False},
        )

        a = softmax(res.x)

        # enforce exact symmetry at each stage (helps stability)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a basic linear cooling schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # move a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # Metropolis acceptance
        if delta < 0 or np.random.rand() < math.exp(
            -delta / (T0 * (1 - step / max_steps) + 1e-18)
        ):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase step size when improving
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # shrink step size on rejections
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Utility to obtain a random point on the simplex
# ----------------------------------------------------------------------


def random_simplex(n: int) -> np.ndarray:
    """Draw a random probability vector (Dirichlet(1) distribution)."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()

            print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

    # exponential U‑shape (γ)
    for gamma in np.linspace(0.0, 4.0, 11):

    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:

        a_final, sc_final = hill_climb_sa(

    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
