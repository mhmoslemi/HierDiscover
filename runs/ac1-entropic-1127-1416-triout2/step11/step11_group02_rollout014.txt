def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    n = 2048
    best_seq = None
    best_val = float('inf')
    
    # baseline: uniform
    a0 = np.full(n, 1.0 / n)
    seq0 = a0.tolist()
    val0 = evaluate_sequence(seq0)
    best_seq, best_val = seq0, val0
    print(f"[baseline] score = {val0:.6f}")

    # generate initial candidates
    candidates = []
    # param sweep for power law
    for p in np.linspace(0.4, 1.2, 21):
        a = np.arange(1, n+1) ** (-p)
        a /= a.sum()
        candidates.append(a)
    # geometric
    for r in np.linspace(0.85, 0.99, 21):
        a = r ** np.arange(n)
        a /= a.sum()
        candidates.append(a)
    # combined power+geom
    for p in np.linspace(0.5, 1.0, 11):
        for r in np.linspace(0.88, 0.98, 11):
            a = (np.arange(1, n+1) ** (-p)) * (r ** np.arange(n))
            a /= a.sum()
            candidates.append(a)
    # random variations of best found candidate etc.
    # We'll limit candidate count to maybe 200.

    # Evaluate each candidate quickly (use compute_score, not evaluate_sequence)
    # Choose top few for further refinement
    top_k = 10
    scored = []
    for a in candidates:
        if time.time() >= deadline:
            break
        score = compute_score(a)
        scored.append((score, a))
    scored.sort(key=lambda x: x[0])
    top_candidates = [a for _,a in scored[:top_k]]

    # Add some random perturbations around each top candidate
    for base in top_candidates:
        for _ in range(3):
            if time.time() >= deadline:
                break
            perturb = base + 0.01 * np.random.randn(n)
            perturb = np.maximum(perturb, 0)
            perturb /= perturb.sum()
            top_candidates.append(perturb)

    # now run smooth gradient descent on each candidate
    for idx, a_start in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # allocate time per run: maybe  (deadline - time.time()) / (remaining candidates)
        remaining = len(top_candidates) - idx
        local_deadline = time.time() + (deadline - time.time()) * 0.8 / remaining  # give 80% of remaining time
        a_refined = smooth_gradient_descent(a_start, local_deadline, max_iter=100000, step0=1e-5, alpha_start=1.0, alpha_max=64.0, alpha_inc_iter=2000)
        val = compute_score(a_refined)
        print(f"[candidate {idx}] refined score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_refined.tolist()
    
    # final hill climb (greedy) from best_seq for a while
    if time.time() < deadline - 5 and best_seq is not None:
        a = np.array(best_seq)
        a_final = hill_climb(a, deadline - 5, max_iter=500000)  # re-use hill_climb defined earlier
        val = compute_score(a_final)
        print(f"[final hill climb] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. final reported score = {final_score:.6f}")
    return best_seq

def smooth_gradient_descent(a0, time_deadline, max_iter=200000, step0=1e-5, alpha_start=1.0, alpha_max=64.0, alpha_inc_iter=2000):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    alpha = alpha_start
    iteration = 0
    while time.time() < time_deadline and iteration < max_iter:
        # compute conv
        conv = full_convolution(a)  # length 2n-1
        # smooth max and weights
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sumexp = np.sum(exp_scaled)
        w = exp_scaled / sumexp
        # compute gradient via cross-correlation
        # linear correlation w with a: use FFT
        L = w.shape[0]
        N = 1 << ((L + n - 1).bit_length())
        W_fft = np.fft.rfft(w, n=N)
        A_fft = np.fft.rfft(a, n=N)
        corr = np.fft.irfft(W_fft * np.conj(A_fft), n=N)[:L + n - 1]
        grad = 4.0 * n * corr[:n]  # gradient
        # step size
        step = step0 / np.sqrt(iteration + 1)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)
        # evaluate with true objective
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        # update a
        a = a_new
        iteration += 1
        if iteration % alpha_inc_iter == 0 and alpha < alpha_max:
            alpha = min(alpha * 2, alpha_max)
            print(f"[smooth GD] iteration {iteration}, alpha {alpha}, best_score {best_score:.6f}")
    return best_a

"""
Optimiser for `evaluate_sequence`.

The optimiser proceeds in several stages:

1. **Initial families** – generate a modest pool of candidates from
   simple parametric families (power‐law, geometric, mixed power+geometric)
   and keep the best few (according to the exact objective
   `2·n·max(conv)`).

2. **Smooth‐max gradient descent** – minimise a smooth approximation of the
   max‑convolution objective:
        O_α(a) = (2·n/α)·log∑_k exp(α·(a*a)[k])
   The gradient of O_α is obtained analytically and the update is a
   projected gradient step on the probability simplex.  The smoothing
   parameter α is increased geometrically during the run, driving the
   approximation towards the true max.  The step size is chosen so that
   no coordinate moves by more than a prescribed amount (which decays as
   1/√iteration).

3. **Final stochastic hill‑climb** – a cheap greedy neighbourhood search
   (mass transfer between two random coordinates) is applied to the best
   vector found by the gradient stage.

All phases respect the overall time budget (default 1000 s).  The
function `propose_candidate` returns a plain Python list of non‑negative
floats whose sum is 1, which is exactly what `evaluate_sequence` expects.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Correlation needed for the smooth‑max gradient
# ----------------------------------------------------------------------
def linear_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation (w ⋆ a)[i] = Σ_j w[i+j]·a[j] for i=0..len(a)-1.
    Implemented via FFT for O(N log N) time.
    """
    L = w.shape[0]
    n = a.shape[0]
    N = 1 << ((L + n - 1).bit_length())         # power of two ≥ L+n‑1
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a, n=N)
    corr_full = np.fft.irfft(np.conj(W) * A, n=N)
    # keep only the first n entries (the values we need)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent
# ----------------------------------------------------------------------
def smooth_gradient_descent(a0: np.ndarray,
                           time_deadline: float,
                           max_iter: int = 200_000,
                           step0: float = 0.5,
                           alpha_start: float = 1.0,
                           alpha_max: float = 64.0,
                           alpha_inc_iter: int = 5_000) -> np.ndarray:
    """
    Minimise a smooth approximation of the max‑convolution objective.
    Parameters
    ----------
    a0            : initial feasible vector (non‑negative, Σ=1)
    time_deadline : absolute time at which the routine must stop
    max_iter      : safety cap on the number of iterations
    step0         : controls the maximum coordinate change (≈ step0/√t)
    alpha_start   : initial smoothing factor
    alpha_max     : final smoothing factor (larger → closer to true max)
    alpha_inc_iter: how many iterations between α‑doublings
    Returns
    -------
    best feasible vector found (still Σ=1)
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    n = a.shape[0]
    alpha = alpha_start
    it = 0

    while time.time() < time_deadline and it < max_iter:
        # ----- 1. convolution and smooth‑max weights -----
        conv = full_convolution(a)                 # length 2n‑1
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)   # stable exponentiation
        sumexp = np.sum(exp_scaled)
        w = exp_scaled / sumexp                    # soft‑max weights (sum to 1)

        # ----- 2. gradient of the smooth objective -----
        # grad = 4·n· Σ_k w_k·shift_k(a)  → implemented via correlation
        corr = linear_correlation(w, a)            # length n
        grad = 4.0 * n * corr

        # ----- 3. step size (max coordinate move decays as 1/√t) -----
        max_abs = np.max(np.abs(grad))
        if max_abs == 0:
            break
        # desired maximum change of any coordinate
        desired_change = step0 / np.sqrt(it + 1.0)
        step = desired_change / max_abs
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        # ----- 4. keep the best true objective -----
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        # ----- 5. increase smoothing factor slowly -----
        if it % alpha_inc_iter == 0 and alpha < alpha_max:
            alpha = min(alpha * 2.0, alpha_max)
            print(f"[smooth‑GD] it={it:,} α={alpha:.2f} best={best_score:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (same as before)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               time_deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Tiny stochastic moves with greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        if no_improve > 100_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

    # -----------------------------------------------------------------

    best_seq = None

    # -----------------------------------------------------------------
    # 1. Generate a pool of simple‑family candidates (very cheap)

    n = 2048
    top_k = 10                     # keep only the best few for refinement
    pool = []                      # list of (score, vector)

    def consider(vec):
        nonlocal pool
        s = compute_score(vec)
        if len(pool) < top_k:
            pool.append((s, vec))
            pool.sort(key=lambda x: x[0])
        elif s < pool[-1][0]:
            pool[-1] = (s, vec)
            pool.sort(key=lambda x: x[0])

    # uniform baseline (useful as a sanity check)
    uniform = np.full(n, 1.0 / n)
    consider(uniform)

    # power‑law family
    for p in np.linspace(0.4, 1.2, 21):
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        consider(a)

    # geometric (exponential) family
    for r in np.linspace(0.85, 0.99, 21):
        a = (r ** np.arange(n, dtype=np.float64))
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        consider(a)

    # mixed power‑law + geometric (a_i = (i+1)^{-p} * r^{i})
    for p in np.linspace(0.5, 1.0, 11):
        for r in np.linspace(0.88, 0.98, 11):

            a = (idx ** (-p)) * (r ** np.arange(n, dtype=np.float64))
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            consider(a)

    # keep the best few vectors

    # -----------------------------------------------------------------

            cand /= cand.sum()
            perturbed.append(cand)
    top_candidates.extend(perturbed)
