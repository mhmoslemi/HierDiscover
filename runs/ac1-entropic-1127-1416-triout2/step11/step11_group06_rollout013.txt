The evaluate_sequence function does:

5. sum_a = sum(sequence).

def smooth_mirror_optimize(start_a, time_deadline, max_iters=300_000, eta0=0.6, lambda_start=0.5, lambda_end=30.0):

    a = start_a.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        # lambda schedule
        lam = lambda_start + (lambda_end - lambda_start) * (t / max_iters)
        # compute softmax weights
        # shift conv by max for stability
        exp_arg_w = lam * (conv - max_conv)
        # Clip exponent to avoid overflow
        exp_arg_w = np.clip(exp_arg_w, -50, 50)
        w = np.exp(exp_arg_w)
        w_sum = w.sum()
        w = w / w_sum
        # gradient
        a_rev = a[::-1]
        conv2 = convolution_fft(w, a_rev)
        g = 2.0 * conv2[n-1:n-1+n]
        # compute avg = a dot g
        avg = np.dot(a, g)
        # step size
        eta = eta0 / np.sqrt(t)
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50, 50)
        a = a * np.exp(exp_arg)
        # symmetrize
        a = (a + a[::-1]) / 2.0
        # renormalize onto simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / sum_a
        # best check
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Core ideas
----------
* Build many analytic seeds (uniform, central‑binomial, power‑law,
  geometric, Beta‑shaped) and keep the best.
* Refine the best seed with a *smooth* KL‑mirror descent that minimises
  a log‑sum‑exp approximation of  max(conv(a,a)).  The smoothing
  (temperature λ) grows from a gentle value to a large one, so the
  optimiser first moves in a well‑behaved landscape and then focuses
  on the true max‑convolution.
* Perform a few short random‑restart bursts (perturb the current best
  and run a cheap optimisation) to escape possible traps.
* Finish with a cheap stochastic hill‑climb (swap / fractional mass
  transfer) to squeeze out the final few bits.
* All phases respect a global wall‑clock deadline (default 1000 s) and
  keep the best sequence seen.  The final sequence is returned as a
  plain Python list of floats (already normalised to sum = 1), ready
  for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Symmetric and heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth KL‑mirror optimiser (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------


def smooth_mirror_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.6,
    lambda_start: float = 0.5,
    lambda_end: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) descent on a smooth
    approximation of max(conv(a,a)):
        f_λ(a) = (1/λ)·log Σ_k exp(λ·conv_k(a)).
    ``a`` stays on the simplex (sum = 1, a≥0) and is symmetrised
    after each update (a_i ← (a_i + a_{n‑1‑i})/2) because the optimal
    solution is known to be symmetric.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # pre‑allocate reversed view to avoid repeated copies
    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ----- convolution and smoothing -----
        conv = convolution_fft(a, a)                     # length 2n‑1
        max_conv = conv.max()

        # exponential temperature schedule (smooth → sharp)
        lam = lambda_start * (lambda_end / lambda_start) ** (t / max_iters)

        # soft‑max weights (stable via shift by max)
        exp_arg = lam * (conv - max_conv)
        exp_arg = np.clip(exp_arg, -50.0, 0.0)           # avoid underflow/overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        a_rev = a[::-1]
        conv2 = convolution_fft(w, a_rev)               # length 3n‑2
        g = 2.0 * conv2[n - 1 : n - 1 + n]               # gradient (size n)

        # average of the gradient under the current a (used by KL‑mirror)
        avg = np.dot(a, g)

        # ----- mirror (exponentiated‑gradient) update -----
        eta = eta0 / np.sqrt(t)
        exp_update = -eta * (g - avg)
        exp_update = np.clip(exp_update, -50.0, 50.0)
        a *= np.exp(exp_update)

        # enforce symmetry
        a = (a + a[::-1]) * 0.5

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional progress check
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine problem size (use provided sequence length if available)
    # ------------------------------------------------------------------
    try:
        n0 = len(height_sequence_1)          # type: ignore[name-defined]
        print(f"Detected provided height_sequence_1 of length {n0}.")
    except Exception:
        n0 = 2048
        print(f"No external sequence provided – using default length {n0}.")

    # We will try the native length and, if time permits, double it.
    lengths = [n0]
    if time.time() < deadline - 150:
        if n0 * 2 <= 8192:
            lengths.append(n0 * 2)
            print(f"Will also explore length {lengths[-1]}.")

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Loop over candidate problem sizes
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"\n--- Optimising for length n = {n} ---")

        # ------------------------------------------------------------------
        # 0. Build a pool of analytic starters
        # ------------------------------------------------------------------
        seeds = []

        # uniform
        seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

        # central binomial
        seeds.append(generate_central_binomial(n))

        # power‑law (wide sweep)
        for p in np.linspace(0.1, 1.0, 12):
            seeds.append(generate_powerlaw(n, float(p)))

        # geometric (r close to 1)
        for r in [0.96, 0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:
            seeds.append(generate_geometric(n, float(r)))

        # Beta‑shaped (α from very small up to 0.6)
        for alpha in np.linspace(0.02, 0.6, 13):
            seeds.append(generate_beta_shape(n, float(alpha)))

        # possibly reuse the previously found construction
        try:
            if len(height_sequence_1) == n:   # type: ignore[name-defined]
                seeds.append(np.array(height_sequence_1, dtype=np.float64))
                print("Added provided height_sequence_1 as a starter.")
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 1. Pick the best starter by exact evaluation
        # ------------------------------------------------------------------
        local_best = None
        local_best_val = float("inf")
        for idx, cand in enumerate(seeds):
            cand = project_to_simplex(cand)
            val = compute_score(cand)
            print(f"Starter {idx:02d}: score = {val:.6f}")
            if val < local_best_val:
                local_best_val = val
                local_best = cand.copy()
        print(f"Best starter for n={n}: score = {local_best_val:.6f}")

        # ------------------------------------------------------------------
        # 2. Smooth mirror‑descent optimisation (main workhorse)
        # ------------------------------------------------------------------
        mirror_deadline = deadline - 50.0   # reserve time for later phases
        if time.time() < mirror_deadline:
            remaining = mirror_deadline - time.time()
            # estimate how many iterations we can afford (≈0.0007 s/iter)
            max_iters = int(min(1_200_000, remaining * 1400))
            print(f"Smooth mirror phase – up to {max_iters} iterations.")
            a_opt, opt_score = smooth_mirror_optimize(
                start_a=local_best,
                time_deadline=mirror_deadline,
                max_iters=max_iters,
                eta0=0.6,
                lambda_start=0.5,
                lambda_end=30.0,
            )
            if opt_score < local_best_val:
                local_best_val, local_best = opt_score, a_opt.copy()
                print(f"Smooth phase improved to {local_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3. Short random‑restart bursts (cheap)
        # ------------------------------------------------------------------
        restart_deadline = deadline - 30.0
        if time.time() < restart_deadline:
            base = local_best.copy()
            for r in range(4):
                if time.time() >= restart_deadline:
                    break
                pert = base + 0.015 * np.random.randn(n)
                pert = project_to_simplex(pert)
                sub_dead = time.time() + 9.0
                a_tmp, s_tmp = smooth_mirror_optimize(
                    pert,
                    time_deadline=sub_dead,
                    max_iters=100_000,
                    eta0=0.5,
                    lambda_start=0.5,
                    lambda_end=20.0,
                )
                if s_tmp < local_best_val:
                    local_best_val, local_best = s_tmp, a_tmp.copy()
                    print(f"Restart {r} gave improvement: {local_best_val:.6f}")

        # ------------------------------------------------------------------
        # 4. Final stochastic hill‑climb
        # ------------------------------------------------------------------
        hill_deadline = deadline - 5.0
        if time.time() < hill_deadline:
            steps = 250_000
            print(f"Final hill‑climb (≤{steps} steps).")
            polished, hill_score = hill_climb_permutation(
                local_best.tolist(),
                time_deadline=hill_deadline,
                max_steps=steps,
            )
            if hill_score < local_best_val:
                local_best_val, local_best = hill_score, np.array(polished, dtype=np.float64)
                print(f"Hill‑climb improved to {local_best_val:.6f}")

        # ------------------------------------------------------------------
        # 5. Record globally best sequence
        # ------------------------------------------------------------------
        if local_best_val < best_score:
            best_score, best_seq, best_n = local_best_val, local_best.copy(), n
            print(f"New global best (n={best_n}) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard – fall back to uniform if something went wrong
    # ------------------------------------------------------------------
    if best_seq is None:
        best_n = n0
        best_seq = np.full(best_n, 1.0 / best_n, dtype=np.float64)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 7. Final reporting using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq.tolist()
