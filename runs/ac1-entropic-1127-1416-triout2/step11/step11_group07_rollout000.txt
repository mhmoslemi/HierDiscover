def softmax_adam(a, start_score, deadline, beta_start=5., beta_end=5000., lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    cur = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = cur.copy()
    best_score = start_score
    it = 0
    start = time.time()
    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2*cur.shape[0]*max_conv
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
        # anneal beta
        fraction = (time.time() - start) / (deadline - start)
        beta = beta_start + fraction*(beta_end - beta_start)
        # compute w for gradient
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        grad = 2.0 * beta * np.correlate(w, cur, mode='valid')
        # Adam update
        it += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cur = project_to_simplex(cur - step)
        if it % 2000 == 0:
            print(f"[softmax_ADAM] it {it:6d} | time {time.time()-start:6.1f}s | best {best_score:.6f} | beta {beta:.1f}")
    return best, best_score

def softmax_adam(a, start_score, deadline, beta_start=5., beta_end=4000., lr0=0.05):
    cur = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = cur.copy()
    best_score = start_score
    it = 0
    start = time.time()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0 * cur.shape[0] * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
        # beta schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        beta = beta_start + frac*(beta_end - beta_start)
        # weights
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        # gradient
        grad = 2.0 * beta * np.correlate(w, cur, mode='valid')
        # Adam update
        it += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr0 / np.sqrt(it)  # decaying learning rate
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cur = project_to_simplex(cur - step)
        # periodic logging
        if it % 5000 == 0:
            print(f"[softmax Adam] it {it:6d} | time {elapsed:6.1f}s | best {best_score:.6f} | beta {beta:.1f}")
    return best, best_score

     optimiser (annealed temperature β, decaying learning‑rate,
     Euclidean projection onto the simplex).
   * Phase B – variance flattening (minimise the variance of the
     autocorrelation) with a simple projected gradient descent.
   * Phase C – classic sub‑gradient descent on the true max‑objective.
   * Phase D – stochastic hill‑climb polishing (swap / transfer moves).
4. Return the final sequence as a plain list of Python floats.
   The harness will call ``evaluate_sequence`` on it.

All arithmetic is performed with NumPy and the code respects the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: choose `k` distinct positions and assign random weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase A – soft‑max (log‑sum‑exp) minimisation with Adam
# ----------------------------------------------------------------------


def softmax_adam(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 4000.0,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        Lβ(a)=log Σ exp(β·conv(a,a))
    using an Adam optimiser with an annealed temperature β.
    Returns the best (lowest exact) score encountered.
    """
    cur = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best = cur.copy()
    best_score = start_score

    it = 0
    start = time.time()
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0 * cur.shape[0] * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # annealed β
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max weights
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)  # numerically stable (max shifted to 0)
        w = exp_vals / exp_vals.sum()

        # gradient of the surrogate (valid correlation)
        grad = 2.0 * beta * np.correlate(w, cur, mode="valid")

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)               # decaying step size
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cur = project_to_simplex(cur - step)

        if it % 5000 == 0:
            print(
                f"[softmax‑Adam] it {it:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.1f}"
            )

    return best, best_score


# ----------------------------------------------------------------------
# Phase B – variance‑flattening gradient descent
# ----------------------------------------------------------------------


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation (length = len(a)):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    Implemented via one FFT‑convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]


def variance_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the auto‑convolution.
    Gradient: ∇V = 4·((conv-avg) ⋆ a) .
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        e = conv - avg
        grad = 4.0 * correlation_valid(e, cur)

        it += 1
        step = step0 / np.sqrt(it)
        nxt = cur - step * grad
        cur = project_to_simplex(nxt)

        if it % 3000 == 0:
            print(
                f"[var‑GD] it {it:6d} | time {time.time():6.1f}s | best {best_score:.6f}"
            )

    return best, best_score


# ----------------------------------------------------------------------
# Phase C – sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑objective."""
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        step = step0 / np.sqrt(t)
        cur -= step * g
        cur = project_to_simplex(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # occasional jitter to escape shallow plateaux
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur += jitter
            cur = project_to_simplex(cur)

    return best, best_score


# ----------------------------------------------------------------------
# Phase D – hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    """Stochastic moves kept only when they improve the exact score."""
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()

        if np.random.rand() < 0.5:                # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑normalise to counteract floating‑point drift
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1
        if steps % 200_000 == 0:
            print(
                f"[hill‑climb] step {steps:,} | time {time.time():6.1f}s | best {best_score:.6f}"
            )

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Reserve a small safety margin so the harness never times out
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # 0 – optional fallback from a previously obtained sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1 – choose sequence length (larger lengths give a tighter theoretical bound)
    # ------------------------------------------------------------------
    if total_time > 940.0:
        n = 8192
    elif total_time > 650.0:
        n = 4096
    else:
        n = 2048
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2 – generate a diverse set of seeds and keep the best one
    # ------------------------------------------------------------------
    seeds = []

    # deterministic shapes
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # beta pdf families
    for alpha, beta in [
        (0.2, 0.2),
        (0.5, 0.5),
        (1.0, 1.0),
        (2.0, 2.0),
        (5.0, 5.0),
        (0.5, 2.0),
        (2.0, 0.5),
    ]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    # Dirichlet draws (different concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # sparse spikes – often a good start for flattening the autocorrelation
    for k in (30, 60, 120):
        seeds.append(random_spike(n, k=k))

    # include fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # evaluate seeds and retain the best
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)               # ensure a valid probability vector
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] improvement → score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – start optimisation from the best seed
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------------------------------------------------------
    # 4 – allocate time for the four optimisation phases
    # ------------------------------------------------------------------
    t0 = time.time()
    phase1_deadline = t0 + total_time * 0.35
    phase2_deadline = phase1_deadline + total_time * 0.20
    phase3_deadline = phase2_deadline + total_time * 0.20
    # remaining time goes to hill‑climb
    phase4_deadline = deadline - 0.5  # tiny buffer

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max Adam optimisation
    # ------------------------------------------------------------------
    print("\n[Phase 1] soft‑max Adam optimisation")
    cur_a, cur_score = softmax_adam(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=4000.0,
        lr0=0.03,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after Phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – variance‑flattening gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("\n[Phase 2] variance‑flattening gradient descent")
        cur_a, cur_score = variance_gradient_descent(
            cur_a,
            cur_score,
            phase2_deadline,
            step0=0.04,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        print("\n[Phase 3] sub‑gradient refinement")
        cur_a, cur_score = subgradient_refine(
            cur_a,
            cur_score,
            phase3_deadline,
            max_iter=600_000,
            step0=0.07,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 3] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – hill‑climb polishing (uses whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        print("\n[Phase 4] hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, phase4_deadline, max_steps=2_000_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after hill‑climb] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
