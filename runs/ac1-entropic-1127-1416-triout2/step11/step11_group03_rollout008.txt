"""Propose a step‐function sequence minimizing the auto‑convolution peak.

The algorithm
---------------

1. **Coarse scan** of several analytic families (product‑power, power‑law,
   exponential‑U, beta‑type and a simple triangular shape) on a moderate
   length `n_coarse`.  The best parameter of the product‑power family is
   remembered for a later golden‑section refinement.

2. **Refinement** of the product‑power family by golden‑section search on the
   same length.

3. **Upscaling** – the best coarse vector is interpolated to a larger length
   `n_big` (8192).  The vector is re‑projected onto the probability simplex
   (sum = 1).

4. **Smooth‑max L‑BFGS** – the objective `2·n·max(conv)` is approximated by a
   log‑sum‑exp (smooth maximum) with a large temperature `β`.  The optimisation
   is performed on the soft‑max parametrisation `a = softmax(z)` using
   SciPy’s L‑BFGS‑B (fallback to Adam if SciPy is unavailable).

5. **Exact sub‑gradient descent** on the true `max(conv)` objective for a
   few hundred‑thousand steps.

6. **Final simulated‑annealing hill‑climb** for a last polishing on the
   simplex.

All steps respect the 1000 s wall‑clock budget and keep the current best
solution.  The returned sequence is a plain list of non‑negative floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS smooth‑max routine
# ----------------------------------------------------------------------
_curr_n = None          # current length of the optimisation problem
_smooth_beta = None     # temperature of the smooth‑max surrogate


# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (fast O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1).

    Returns 2·n·max(conv) where conv = a * a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric product family: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the grid: (i+½)^{p‑1}(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


def triangular_family(n: int) -> np.ndarray:
    """Simple triangular (piecewise linear) shape, symmetric."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.minimum(i, n - 1 - i) + 0.5
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Random starts (fallbacks)
# ----------------------------------------------------------------------
def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start: ~5 % non‑zero entries, uniform mass on them."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort a powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return project_to_simplex(a)


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Golden‑section search (for a single scalar parameter)
# ----------------------------------------------------------------------
def golden_section_optimize(func, a: float, b: float, max_iter: int = 12):
    """Minimise `func` on [a,b] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Smooth‑max objective used by L‑BFGS (and optionally Adam)
# ----------------------------------------------------------------------
def smooth_obj_grad(z: np.ndarray):
    """Smooth‑max (log‑sum‑exp) objective and its gradient w.r.t. logits z."""
    n = _curr_n
    beta = _smooth_beta
    a = softmax(z)                     # probability vector, sum=1
    conv = convolution_fft(a, a)       # full linear convolution
    maxc = conv.max()
    # stable log‑sum‑exp of beta·conv
    exp_term = np.exp(beta * (conv - maxc))
    sumexp = exp_term.sum()
    value = maxc + np.log(sumexp) / beta
    obj = 2.0 * n * value               # exact objective for sum=1

    # weights of the smooth‑max (sum to 1)
    w = exp_term / sumexp

    # gradient w.r.t. a (derivative of value)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

    # back‑prop through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    grad_z = 2.0 * n * grad_z          # chain rule for the outer factor
    return obj, grad_z


def smooth_obj_wrapper(z):
    """Wrapper required by SciPy’s `minimize` (returns f and grad)."""
    return smooth_obj_grad(z)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate (fallback)
# ----------------------------------------------------------------------
def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 10.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2500):
    """Adam optimisation of the smooth‑max surrogate."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)               # initialise log‑weights

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.045) -> tuple[np.ndarray, float]:
    """Classical sub‑gradient descent for the exact max‑objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))               # index of the current peak

        # sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polish)
# ----------------------------------------------------------------------
def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a tiny temperature schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4          # initial temperature
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def l_bfgs_optimize(start_a: np.ndarray,
                    beta: float,
                    max_iter: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """L‑BFGS‑B optimisation of the smooth‑max objective.

    Falls back to the Adam optimiser if SciPy is not available.
    """
    global _curr_n, _smooth_beta
    n = start_a.shape[0]
    _curr_n = n
    _smooth_beta = beta

    z0 = np.log(start_a + 1e-12)

    try:
        from scipy.optimize import minimize  # type: ignore
    except Exception:
        # SciPy missing – use Adam as a safe fallback
        return adam_smooth_optimize(start_a,
                                    deadline,
                                    max_iters=max_iter,
                                    beta_start=beta,
                                    beta_end=beta,
                                    lr=0.015,
                                    sym_interval=0)

    # L‑BFGS‑B (bounded, but we keep the variables unconstrained)
    opts = {'maxiter': max_iter, 'disp': False}
    res = minimize(fun=smooth_obj_wrapper,
                   x0=z0,
                   jac=True,
                   method='L-BFGS-B',
                   options=opts)

    a_opt = softmax(res.x)
    a_opt = project_to_simplex(a_opt)
    return a_opt, compute_score(a_opt)


# ----------------------------------------------------------------------
# Upscaling helper (linear interpolation)
# ----------------------------------------------------------------------
def upscale_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Upsample `v` to length `new_n` by linear interpolation and renormalise."""
    old_n = v.shape[0]
    if old_n == new_n:
        return v.copy()
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    new_v = np.interp(new_x, old_x, v)
    new_v = new_v / new_v.sum()
    return new_v

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence minimizing ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    safety = 2.0                       # keep a small margin before the hard deadline
    deadline = start_time + budget_s - safety

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback to a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Coarse scan on a moderate length
    # ------------------------------------------------------------------
    n_coarse = 2048
    print(f"[{time.time():.1f}] starting coarse scan (n={n_coarse})")
    best_alpha = None

    # product‑power family
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n_coarse, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")

    # left‑heavy power law
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n_coarse, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")

    # exponential U shape
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n_coarse, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")

    # triangular shape (simple unimodal)
    vec = triangular_family(n_coarse)
    sc = compute_score(vec)
    if sc < best_score:
        best_score = sc
        best_seq = vec.tolist()
        print(f"[{time.time():.1f}] new best (triangular) = {best_score:.6f}")

    # beta family grid
    for p in np.linspace(0.1, 3.0, 13):
        if time.time() >= deadline - 30.0:
            break
        for q in np.linspace(0.1, 3.0, 13):
            vec = beta_family(n_coarse, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (beta p={p:.2f},q={q:.2f}) = {best_score:.6f}")

    # mixture uniform + U shape
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n_coarse, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Fine golden‑section optimisation of the product family (if possible)
    # ------------------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] golden‑section refinement around α={best_alpha:.3f}")

        def score_of_alpha(alpha_val: float) -> float:
            vec = symmetric_product_family(n_coarse, alpha_val)
            return compute_score(vec)

        lo = max(0.0, best_alpha - 0.6)
        hi = min(12.0, best_alpha + 0.6)
        best_alpha, best_alpha_score = golden_section_optimize(score_of_alpha,
                                                               lo, hi, max_iter=12)
        vec = symmetric_product_family(n_coarse, best_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] refined product‑α={best_alpha:.6f} → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Upscale to a larger resolution for finer optimisation
    # ------------------------------------------------------------------
    n_big = 8192
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] up‑scaling best coarse solution to n={n_big}")
        a_big = upscale_vector(np.array(best_seq, dtype=np.float64), n_big)
        best_seq = a_big.tolist()
        best_score = compute_score(a_big)
        print(f"[{time.time():.1f}] score after up‑scaling = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. L‑BFGS optimisation with a smooth‑max surrogate (first pass)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] L‑BFGS smooth‑max optimisation (first pass)")
        a_start = np.array(best_seq, dtype=np.float64)
        remaining = deadline - time.time()
        # give a modest number of L‑BFGS iterations (each iteration is cheap)
        max_iter = 300
        a_opt, sc_opt = l_bfgs_optimize(a_start,
                                        beta=200.0,
                                        max_iter=max_iter,
                                        deadline=deadline - 120.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS 1st pass best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Exact sub‑gradient descent (refine the true max‑objective)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] sub‑gradient descent refinement")
        a_sub, sc_sub = subgrad_optimize(np.array(best_seq, dtype=np.float64),
                                         deadline - 60.0,
                                         max_iters=1_000_000,
                                         step0=0.045)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second L‑BFGS pass with a higher temperature for tighter smooth‑max
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        print(f"[{time.time():.1f}] second L‑BFGS (higher β) pass")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = l_bfgs_optimize(a_start,
                                          beta=1000.0,
                                          max_iter=200,
                                          deadline=deadline - 30.0)
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS 2nd pass best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Enforce exact symmetry (often reduces the peak a little)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetry step improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final simulated‑annealing hill‑climb for polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (simulated annealing)")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline,
                                          max_steps=5_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Safety fallback (uniform distribution) – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] used uniform fallback, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Final official evaluation (includes security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
