"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Main ideas
----------
* Generate a rich collection of analytic “seed” distributions
  (uniform, central‑binomial, power‑law, geometric, asymmetric‑Beta,
  random Dirichlet draws).

* Refine the most promising seeds with a *multi‑peak* KL‑mirror (exponentiated‑gradient)
  descent.  Instead of using only the single active convolution index,
  the update aggregates the top‑k convolution peaks (weighted by their
  values).  An exponential moving‑average (momentum) of the gradient is
  kept, providing a smoother descent direction and often escaping shallow
  local minima.

* A short “random‑restart + short mirror” phase explores neighbouring basins.

* Finally a cheap stochastic hill‑climb (swap / fractional mass moves)
  polishes the best solution.

All phases are time‑boxed against the wall‑clock deadline (≈ 1000 s) and
the best sequence seen so far is always retained.  The returned list is
already on the simplex, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete (α,β) Beta shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
    Not necessarily symmetric.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (beta - 1.0)
    a = left * right
    return project_to_simplex(a)


def generate_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a random point on the simplex from a Dirichlet(α,…,α)."""
    x = np.random.gamma(concentration, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Multi‑peak KL‑mirror optimiser
# ----------------------------------------------------------------------


def weighted_subgradient(a: np.ndarray,
                         conv: np.ndarray,
                         top_k: int = 5) -> np.ndarray:
    """
    Return a sub‑gradient of the *smooth* max‑convolution built from the
    top_k largest convolution entries (weighted by their values).

    For each selected lag k we add the contribution 2·a_{k‑j} (if it exists)
    multiplied by the normalised weight w_k ∝ conv[k].
    """
    n = a.shape[0]
    L = conv.shape[0]

    if top_k >= L:
        idxs = np.arange(L)
        weights = np.full(L, 1.0 / L)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idxs].astype(np.float64)
        w_sum = vals.sum()
        if w_sum == 0.0:
            weights = np.full(top_k, 1.0 / top_k)
        else:
            weights = vals / w_sum

    g = np.zeros_like(a)

    for idx, w in zip(idxs, weights):
        if idx < n:
            seg = a[:idx + 1][::-1]               # a[0..idx] reversed
            g[:idx + 1] += 2.0 * w * seg
        else:
            start = idx - (n - 1)                  # first index where shift lands inside a
            seg = a[start:][::-1]                  # a[start..] reversed
            g[start:] += 2.0 * w * seg

    return g


def mirror_multi_peak_optimize(start_a: np.ndarray,
                               time_deadline: float,
                               max_iters: int = 1_000_000,
                               top_k: int = 5,
                               eta0: float = 0.8,
                               momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) descent that uses the top_k
    convolution peaks to form a weighted sub‑gradient.  An exponential
    moving‑average of the gradient (momentum) is employed for smoother
    updates.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    # momentum accumulator (zero‑initialised)
    m_acc = np.zeros_like(a)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        g = weighted_subgradient(a, conv, top_k=top_k)

        # average term (ensures the update stays on the simplex)
        avg = np.dot(a, g)

        # exponential moving‑average of the centred gradient
        grad_centered = g - avg
        m_acc = momentum * m_acc + (1.0 - momentum) * grad_centered

        # diminishing stepsize
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update (KL‑mirror)
        exp_arg = -eta * m_acc
        exp_arg = np.clip(exp_arg, -50.0, 50.0)      # safety
        a = a * np.exp(exp_arg)

        # renormalise onto the simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # keep track of the best encountered point
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb(seq,
               time_deadline,
               max_steps: int = 300_000) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                         # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
