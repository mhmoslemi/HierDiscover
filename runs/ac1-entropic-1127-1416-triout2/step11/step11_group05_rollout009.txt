def subgradient_descent(a_start, deadline, max_iters=20000, init_eta=1.0):
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    eta = init_eta
    eps = 1e-12
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # find active indices (within tolerance)
        mask = (conv >= maxc - eps).astype(np.float64)
        active = mask.sum()
        if active == 0:
            # fallback to exact max index
            mask = np.zeros_like(conv)
            mask[conv.argmax()] = 1.0
            active = 1.0
        w = mask / active
        # gradient using convolution with reversed a
        grad = 2.0 * convolution_fft(a[::-1], w)[n-1:n-1 + n]
        # exponentiated gradient (mirror descent)
        # log update
        log_a = np.log(a + 1e-16)
        log_a -= eta * grad
        a = np.exp(log_a)
        # renormalize to simplex
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)
        # optional symmetry enforcement
        a = (a + a[::-1]) * 0.5
        # evaluate
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            # perhaps increase eta a little
            # track improvement
        # step size schedule: diminishing
        eta = init_eta / np.sqrt(it + 1)
    return best_a, best_score

import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (fast seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate (for L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) with
        f   = smooth max = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z  (gradient w.r.t. the soft‑max parameters)

    Uses the exact analytic gradient.
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta                 # smooth approximation of max
    w = expw / sumexp                                # normalised soft‑max over positions
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]        # shape (n,)
    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation (smooth surrogate)
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS runs on the smooth‑max surrogate.
    Returns the best vector (still on the simplex) and its exact score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        z0 = np.log(a + 1e-12)  # initialise soft‑max logits

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as e:
            print(f"  L‑BFGS failure: {e}")
            continue

        a = softmax(res.x)
        # enforce symmetry (helps numerically)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient mirror descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        max_iters: int = 20000,
                        init_eta: float = 1.0):
    """
    Mirror‑descent on the exact max‑convolution using the analytical
    sub‑gradient (computed via a single FFT).  The iterate stays on the
    probability simplex by means of an exponentiated‑gradient step.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # keep previous gradient for a simple momentum term
    prev_grad = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # Active set (indices where conv is within a tiny tolerance of the max)
        eps = 1e-13
        mask = (conv >= maxc - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:       # should not happen, but be safe
            mask[conv.argmax()] = 1.0
            cnt = 1.0
        w = mask / cnt      # uniform weights over the active set

        # Sub‑gradient of the max‑convolution (exact)
        grad = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

        # Simple momentum (helps stability)
        gamma = 0.9
        grad_mom = gamma * prev_grad + (1.0 - gamma) * grad
        prev_grad = grad_mom

        # Mirror‑descent step in log‑space (exponentiated gradient)
        eta = init_eta / math.sqrt(it)   # diminishing step‑size
        log_a = np.log(a + 1e-16)
        log_a -= eta * grad_mom
        a = np.exp(log_a)

        # Re‑normalise to the simplex and enforce symmetry
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        a = (a + a[::-1]) * 0.5   # keep the vector exactly symmetric
        a = project_to_simplex(a)  # tiny numerical correction

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            # occasional aggressive step‑size increase when improvement occurs
            init_eta = min(init_eta * 1.05, 2.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Short SA walk with symmetric moves (swap / mass transfer).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(
                -delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` under the simplex constraint.
    The routine respects the wall‑clock budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # give a 2 s safety margin
    best_seq = None

            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose problem dimension
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 650 else 2048
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Quick scan of analytic families
    # ------------------------------------------------------------------
    for alpha in np.linspace(0.0, 3.0, 13):
        if time.time() >= deadline - 120.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

    for beta in np.linspace(0.0, 3.0, 13):
        if time.time() >= deadline - 120.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

    for gamma in np.linspace(0.0, 4.0, 11):
        if time.time() >= deadline - 120.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Sub‑gradient mirror descent (the main workhorse)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] starting sub‑gradient descent")
        a_start = np.array(best_seq, dtype=np.float64)
        # allocate up to ~300 s for this stage
        sg_deadline = deadline - 200.0
        a_sg, sg_score = subgradient_descent(a_start,
                                              deadline=sg_deadline,
                                              max_iters=40000,
                                              init_eta=1.2)
        if sg_score < best_score:
            best_score = sg_score
            best_seq = a_sg.tolist()
            print(f"[{time.time():.1f}] sub‑gradient best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  L‑BFGS refinement on the smooth max surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting L‑BFGS refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        # a fairly aggressive schedule of smoothing parameters
        betas_stage = [5, 30, 150, 500, 2000, 8000, 20000, 50000]
        a_opt, opt_score = l_bfgs_optimize(a_start,
                                            betas=betas_stage,
                                            max_iter_per_stage=300,
                                            deadline=deadline - 80.0)
        if opt_score < best_score:
            best_score = opt_score
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Enforce perfect symmetry and re‑project onto simplex
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_arr = (a_arr + a_arr[::-1]) * 0.5
        a_arr = project_to_simplex(a_arr)
        sc_sym = compute_score(a_arr)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_arr.tolist()
            print(f"[{time.time():.1f}] symmetry projection improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final short simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=2_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Fallback in the (unlikely) event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
