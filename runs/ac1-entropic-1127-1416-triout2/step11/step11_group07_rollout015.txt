"""
Improved step‑function optimiser.

Main ideas
----------
* Use a longer sequence when time permits (n = 4096).
* Initialise from a diverse set of analytically‑generated seeds.
* Phase‑1: smooth surrogate (log‑sum‑exp) minimisation with Adam
            and a linearly‑annealed temperature β → large.
* Phase‑2: projected sub‑gradient descent on the true max‑objective.
* Phase‑3: cheap stochastic hill‑climb polishing.
All iterates are projected onto the probability simplex (non‑negative,
unit‑sum).  The optimisation time is split dynamically according to the
remaining budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real‑FFT (full length)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid (non‑padded) correlation of e with a.
    Returns length = len(a) array:
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max( a ∗ a ).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def proj_simplex_fast(v: np.ndarray) -> np.ndarray:
    """
    Simple, cheap projection onto the simplex:
        clip negatives, then renormalise.
    This is not the Euclidean projection but works well with
    gradient‑descent on the simplex and is much faster.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v[:] = 1.0 / v.shape[0]
    return v


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1))."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Sparse vector with k random non‑zero entries."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def softmax_adam(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 8000.0,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp surrogate of the max‑norm using Adam.
    The temperature β is increased linearly from beta_start to beta_end.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    start = time.time()

    while time.time() < deadline:
        # ---- evaluation (cheap because we already have the convolution) ----
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)  # sum(cur)=1 by construction
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            # occasional report
            print(f"[softmax‑Adam] new best {best_score:.6f}")

        # ---- anneal β ----------------------------------------------------
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # ---- soft‑max weights --------------------------------------------
        shifted = conv - max_conv               # stability
        exp_vals = np.exp(beta * shifted)      # ≤ 1
        w = exp_vals / exp_vals.sum()          # probability vector, len = 2n‑1

        # ---- gradient of the surrogate -----------------------------------
        # g = 2·β·(w ⋆ a)  (valid correlation, length n)
        grad = 2.0 * beta * correlation_valid(w, cur)

        # ---- Adam update -------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size scaled down with β (high β → smaller steps)
        step = (lr / (beta + 1.0)) * m_hat / (np.sqrt(v_hat) + eps)

        nxt = cur - step
        cur = proj_simplex_fast(nxt)

    return best, best_score


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective.
    Uses the simple non‑negative‑renormalise projection for speed.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))          # index of the current maximum

        # sub‑gradient of max(conv) w.r.t. a:
        # g_i = 2 * a_{k-i}  whenever the shifted index is inside [0,n)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        step = step0 / np.sqrt(t)
        cur -= step * g
        cur = proj_simplex_fast(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            # occasional report
            if t % 10000 == 0:
                print(f"[subgrad] it {t:6d} | best {best_score:.6f}")

        # small jitter every 5000 iterations to escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            cur += jitter
            cur = proj_simplex_fast(cur)

    return best, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that only keeps improvements.
    Two move types are used:
        * swapping two entries,
        * transferring a fraction of mass from one index to another.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.1
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = proj_simplex_fast(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1
    return best, best_score
