"""
Advanced step‑function optimiser (v2).

Goal
----
Search for a non‑negative sequence a with minimal
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum a)² .
For a probability vector (sum = 1) this is simply
    2·n·max(conv(a,a)) .
The optimizer works in four stages:

1️⃣  Seed generation – a large, diverse collection of analytic
    shapes (sin², Beta‑polynomials, power‑/exponential‑decay,
    Dirichlet samples, sparse spikes).  The best seed is kept.

2️⃣  Smooth surrogate optimisation – soft‑max (log‑sum‑exp) gradient
    descent with an annealed temperature.  This quickly drives the
    max‑convolution down.

3️⃣  Exact sub‑gradient descent – projected sub‑gradient on the true
    max‑convolution objective, with a decreasing step‑size and occasional
    jitter to escape flat regions.

4️⃣  Simulated‑annealing polishing – stochastic mass‑transfer moves
    accepted with a Metropolis probability, allowing occasional
    uphill steps and thorough exploration of the remaining time budget.

All updates are projected onto the probability simplex, so the objective
remains comparable to ``evaluate_sequence``.  The routine respects the
global time limit (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation (length = len(a)):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    Implemented via one FFT‑convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Seed generators (analytic families)
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n  # points in (0,1)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def generate_power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i ∝ (i+1)^{-p}, normalised."""
    i = np.arange(n, dtype=float) + 1.0
    w = i ** (-p)
    a = w / w.sum()
    return a


def generate_exponential_decay(n: int, rate: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-rate·i), normalised."""
    i = np.arange(n, dtype=float)
    w = np.exp(-rate * i)
    a = w / w.sum()
    return a

# Optimisation phases

def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    step_coef: float = 0.002,
) -> tuple[np.ndarray, float]:
    """Smooth surrogate minimisation via projected gradient descent."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    it = 0
    start = time.time()
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # temperature schedule (linear in elapsed fraction)
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of surrogate (valid correlation)
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")
        it += 1
        step = step_coef / (np.sqrt(it) * beta)
        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 5000 == 0:
            print(
                f"[soft‑max GD] it {it:6d} | time {elapsed:6.1f}s |"
                f" best {best_score:.6f} | β {beta:.1f}"
            )
    return best_a, best_score


def variance_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.10,
) -> tuple[np.ndarray, float]:
    """Flatten the convolution profile via projected gradient descent."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        e = conv - avg
        grad = 4.0 * correlation_valid(e, cur_a)

        it += 1
        step = step0 / np.sqrt(it)
        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 2000 == 0:
            print(
                f"[var‑GD] it {it:6d} | time {time.time():6.1f}s | best {best_score:.6f}"
            )
    return best_a, best_score


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))  # shift achieving the max

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # occasional random jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

        if t % 20000 == 0:
            print(
                f"[sub‑grad] it {t:6d} | time {time.time():6.1f}s | best {best_score:.6f}"
            )
    return best_a, best_score


def simulated_annealing(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    initial_temp: float = 0.02,
    final_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with Metropolis acceptance.
    Moves transfer a random fraction of mass between two indices.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    start = time.time()
    while time.time() < deadline:
        # linear temperature schedule based on remaining fraction
        remaining = deadline - time.time()
        frac = max(0.0, remaining / (deadline - start))
        T = final_temp + (initial_temp - final_temp) * frac

        cand = cur_a.copy()
        i, j = np.random.randint(n, size=2)
        if cand[i] > 0 and i != j:
            # transfer a random fraction (1‑10 % of the donor mass)
            delta = (0.01 + 0.09 * np.random.rand()) * cand[i]
            cand[i] -= delta
            cand[j] += delta

        # re‑normalise (projection also restores feasibility)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            cur_a, cur_score = cand, cand_score
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_score - cur_score) / (T + 1e-12))
            if np.random.rand() < prob:
                cur_a, cur_score = cand, cand_score

        if cur_score < best_score:
            best_score, best_a = cur_score, cur_a.copy()

        # occasional status line
        if int(time.time() - start) % 30 == 0:
            print(
                f"[SA] time {time.time() - start:6.1f}s | T {T:.2e}"
                f" | cur {cur_score:.6f} | best {best_score:.6f}"
            )
    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Fallback (if a previous good sequence is available globally)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a moderate length – 2048 works fast; 4096 if we have ample time
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 850.0:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 1 – generate a rich set of seeds
    # ------------------------------------------------------------------
    seeds = []

    # analytic families
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-5.0, 5.0, 11):
        seeds.append(generate_beta_family(n, p))
    for alpha, beta in [(0.2, 0.2), (0.5, 2.0), (2.0, 0.5), (5.0, 5.0), (1.0, 1.0)]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    # power‑law decays
    for p in np.linspace(0.2, 2.5, 12):
        seeds.append(generate_power_decay(n, p))

    # exponential decays
    for rate in np.linspace(0.001, 0.02, 12):
        seeds.append(generate_exponential_decay(n, rate))

    # random Dirichlet samples
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(random_dirichlet(n, conc=conc))

    # sparse spikes (often good starting points)
    for k in (15, 30, 50, 80, 120):
        seeds.append(random_spike(n, k=k))

    # include fallback if length matches
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.asarray(best_seq, dtype=float))

    # ------------------------------------------------------------------
    # Evaluate seeds (exact objective) and keep the best
    # ------------------------------------------------------------------
    for idx, s in enumerate(seeds):
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)           # guard against tiny drift
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed {idx}] new best → score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2 – start optimisation from the best seed
    # ------------------------------------------------------------------
    cur_a = np.asarray(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)
    print(f"Starting score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3 – time allocation for the remaining phases
    # ------------------------------------------------------------------
    t0 = time.time()
    phase1_deadline = t0 + total_time * 0.25
    phase2_deadline = phase1_deadline + total_time * 0.15
    phase3_deadline = phase2_deadline + total_time * 0.45
    phase4_deadline = deadline - 0.5          # keep a tiny safety buffer

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent
    # ------------------------------------------------------------------
    print("\n[Phase 1] soft‑max gradient descent")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=5000.0,
        step_coef=0.002,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after Phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – variance‑flattening gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("\n[Phase 2] variance‑flattening gradient descent")
        cur_a, cur_score = variance_gradient_descent(
            cur_a,
            cur_score,
            phase2_deadline,
            step0=0.10,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        print("\n[Phase 3] sub‑gradient refinement")
        cur_a, cur_score = subgradient_refine(
            cur_a,
            cur_score,
            phase3_deadline,
            step0=0.12,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 3] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing polishing (uses the remaining time)
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        print("\n[Phase 4] simulated‑annealing polishing")
        cur_a, cur_score = simulated_annealing(
            cur_a,
            cur_score,
            phase4_deadline,
            initial_temp=0.02,
            final_temp=1e-6,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 4] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
