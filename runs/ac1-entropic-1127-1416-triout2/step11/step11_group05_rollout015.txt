```
    n = len(a_start)
    z = np.log(a_start + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    beta_mp = 0.9; beta_v = 0.999; eps = 1e-8
    for beta in betas:
         for i in range(iterations_per_beta):
             if time.time() >= deadline: break outer

             f, g = smooth_max_and_grad(z, beta, n)
             t += 1
             m = beta_mp*m + (1-beta_mp)*g

             v_hat = v/(1 - beta_v**t)
```

* The score is scale‑invariant, therefore we keep every candidate on the simplex (`∑a_i = 1`).
* A fast analytic seed is obtained from a few simple families (uniform, arcsine‑like,
  power‑law, exponential‑U and a few random Dirichlet draws).  
* Starting from each seed we run a **smooth‑max Adam optimiser**.  
  The non‑smooth term `max(conv)` is replaced by a log‑sum‑exp surrogate with a
  temperature β that is increased from a very soft approximation (β≈5) to a
  very sharp one (β≈5·10⁴).  The Adam optimiser works in the log‑softmax
  parametrisation `a = softmax(z)`, guaranteeing non‑negativity and the simplex
  constraint automatically.
* After the Adam phase we perform a short **L‑BFGS** refinement (still on the
  same smooth‑max surrogate) – this gives a second, more accurate pass.
* Finally a lightweight **simulated‑annealing hill‑climb** cleans up any remaining
  irregularities.
* All steps respect the 1000 s wall‑clock budget; the best sequence seen so far
  is continuously reported.

The code below is completely self‑contained and only uses `numpy` and
`scipy.optimize`.  It can be dropped into the evaluation harness; the function
`propose_candidate` will be called by the competition driver.

```python
"""Advanced search for a step‑function that minimises the evaluation score.

The algorithm proceeds as follows

1. Build a small library of analytic seeds (uniform, arcsine‑like, power‑law,
   exponential‑U, random Dirichlet).
2. For each seed run a long Adam optimisation on the smooth‑max surrogate.
   The temperature β is increased gradually (5 → 5·10⁴) while the Adam
   parameters (m, v) are kept across stages.
3. Run a short L‑BFGS optimisation on the same surrogate with a finer β
   schedule.
4. Symmetrise and re‑project on the simplex.
5. Finish with a brief simulated‑annealing hill‑climb.

All candidates are kept on the simplex (∑a_i = 1) using the soft‑max
parametrisation, so the final score returned by ``evaluate_sequence`` is
exactly ``2·n·max(conv)``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities (FFT convolution, projection, soft‑max)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast for large n)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (∑=1, v≥0)."""
    if np.all(v <= 0):
        # return uniform vector if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for fast seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def random_dirichlet(n: int, concentration: float = 1.0) -> np.ndarray:
    """Random point on the simplex (Dirichlet with identical concentration)."""
    raw = np.random.gamma(concentration, 1.0, size=n)
    return raw / raw.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by Adam & L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where

        f   = smooth max = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z   (gradient w.r.t. the soft‑max parameters)

    The gradient is derived analytically, avoiding finite‑difference noise.
    """
    a = softmax(z)
    conv = convolution_fft(a, a)           # length 2n‑1
    maxc = conv.max()
    # stable soft‑max over the convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta

    w = expw / sumexp                     # normalized soft‑max weights over positions

    # gradient w.r.t. a (vector of length n)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]   # central n entries

    # back‑propagation through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  betas,
                  iters_per_beta: int,
                  deadline: float):
    """
    Adam optimisation in the soft‑max (log‑probability) space.
    Returns the best a and its exact score.
    """
    n = a_start.shape[0]
    # initialise Adam state
    z = np.log(a_start + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0
    base_lr = 0.05   # learning‑rate before scaling by v̂

    for beta in betas:
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            f, g = smooth_max_and_grad(z, beta, n)
            t += 1
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_lr / (np.sqrt(v_hat) + eps)
            z = z - step * m_hat
        # re‑project onto the simplex after each temperature stage
        a = softmax(z)
        a = (a + a[::-1]) * 0.5          # enforce symmetry (helps the max)
        a = project_to_simplex(a)
        z = np.log(a + 1e-12)           # reset logits for the next stage
        # early‑exit if budget exhausted
        if time.time() >= deadline:
            break
    a_opt = softmax(z)
    return a_opt, compute_score(a_opt)


# ----------------------------------------------------------------------
# L‑BFGS optimisation (uses the same smooth‑max surrogate)
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS runs on the smooth‑max surrogate.
    Returns the best a and its exact score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        # initialise from current a
        z0 = np.log(a + 1e-12)

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": max_iter_per_stage,
                                    "disp": False})
        except Exception as e:
            print(f"  L‑BFGS failure (β={beta}): {e}")
            continue

        a = softmax(res.x)
        # enforce symmetry and simplex
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta} improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    50 % swap, 50 % mass transfer.  Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        temperature = T0 * (1.0 - step / max_steps)
        if delta < 0 or np.random.rand() < math.exp(-delta / (temperature + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` under the simplex constraint.
    The routine respects the wall‑clock budget and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback on a previously discovered sequence (if available)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size (larger n yields a lower theoretical bound)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 600 else 2048
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # 2. Build a modest pool of analytic seeds
    # ------------------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # arcsine‑like (product family with α≈0.5)
    seeds.append(symmetric_product_family(n, 0.5))

    # a few α values around 0.5 (fine grid)
    for alpha in np.linspace(0.2, 1.0, 9):
        seeds.append(symmetric_product_family(n, alpha))

    # left‑heavy power law
    for beta in [0.5, 1.0, 1.5]:
        seeds.append(left_heavy_power_family(n, beta))

    # exponential U‑shape
    for gamma in [0.2, 0.6, 1.0]:
        seeds.append(exp_u_family(n, gamma))

    # random Dirichlet draws
    for _ in range(5):
        seeds.append(random_dirichlet(n, concentration=0.8))

    # ------------------------------------------------------------------
    # Helper to record improvements
    # ------------------------------------------------------------------
    def register(a_arr: np.ndarray):
        nonlocal best_seq, best_score
        sc = compute_score(a_arr)
        if sc < best_score:
            best_score = sc
            best_seq = a_arr.tolist()
            print(f"[{time.time():.1f}] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Evaluate all seeds (quick pass)
    # ------------------------------------------------------------------
    for a0 in seeds:
        if time.time() >= deadline - 200.0:
            break
        a0 = a0 / a0.sum()
        register(a0)

    # ------------------------------------------------------------------
    # 4. Adam optimisation on each seed (most time‑consuming part)
    # ------------------------------------------------------------------
    betas_adam = [5, 30, 150, 500, 1500, 5000, 15000, 50000]
    iters_per_beta = 300   # enough iterations for a smooth descent

    for a0 in seeds:
        if time.time() >= deadline - 200.0:
            break
        a_opt, sc_opt = adam_optimize(a0, betas_adam, iters_per_beta, deadline - 20.0)
        register(a_opt)

    # ------------------------------------------------------------------
    # 5. L‑BFGS refinement (fewer iterations, but more precise second‑order steps)
    # ------------------------------------------------------------------
    betas_lbfgs = [500, 2000, 8000, 20000, 50000]
    a_start = np.array(best_seq, dtype=np.float64) if best_seq is not None else np.full(n, 1.0 / n)
    a_opt, sc_opt = l_bfgs_optimize(a_start,
                                    betas=betas_lbfgs,
                                    max_iter_per_stage=200,
                                    deadline=deadline - 10.0)
    register(a_opt)

    # ------------------------------------------------------------------
    # 6. Enforce perfect symmetry and re‑project (tiny but sometimes useful)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_arr = (a_arr + a_arr[::-1]) * 0.5
        a_arr = project_to_simplex(a_arr)
        register(a_arr)

    # ------------------------------------------------------------------
    # 7. Final short simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=1_500_000)
        register(a_final)

    # ------------------------------------------------------------------
    # 8. Fallback in the extremely unlikely case we found nothing
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return best_seq
```
