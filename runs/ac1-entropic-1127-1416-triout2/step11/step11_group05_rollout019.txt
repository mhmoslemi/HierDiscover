def max_conv_gradient(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    # i the index in a
    # range where k - i is valid
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start <= i_end:
        i_range = np.arange(i_start, i_end + 1)

    return g

def subgradient_descent(a_start, deadline, max_iters=8000, init_lr=0.5):
    a = a_start.copy()
    n = a.shape[0]
    conv, max_idx = max_convolution_and_index(a)
    cur_score = 2.0 * n * float(conv[max_idx])
    best_a = a.copy()
    best_score = cur_score
    lr = init_lr
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # gradient of current max conv
        g = max_conv_gradient(a, max_idx)
        # step
        cand = a - lr * g
        cand = project_to_simplex(cand)
        # evaluate
        conv_cand = convolution_fft(cand, cand)
        cand_score = 2.0 * n * float(conv_cand.max())
        # accept if improves
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            # recompute max index
            max_idx = int(np.argmax(conv_cand))
            # enlarge step size a bit
            lr *= 1.02
        else:
            # reject, reduce step size
            lr *= 0.7
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # occasional reporting
        if it % 500 == 0:
            print(f"[{time.time():.1f}] subgrad iter {it}, best={best_score:.6f}, lr={lr:.6e}")

def hill_climb_sa(start_a, deadline, max_steps=5_000_000):
    n = start_a.size
    cur_a = start_a.copy()
    conv = convolution_fft(cur_a, cur_a)
    cur_score = 2.0 * n * float(conv.max())
    best_a = cur_a.copy()
    best_score = cur_score
    T0 = best_score * 1e-4
    step_scale = 0.12
    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        conv_cand = convolution_fft(cand, cand)
        cand_score = 2.0 * n * float(conv_cand.max())
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):

"""
Search for a non‑negative step‑function (probability vector) that minimises
the provided ``evaluate_sequence`` objective.

Strategy
--------
1.  **Analytic seeds** – scan a few low‑dimensional families (product,
    left‑heavy power, exponential U) on a reasonably large length
    (2048/4096) to obtain a good starting point.

2.  **Smooth‑max L‑BFGS** – minimise a log‑sum‑exp surrogate of the
    max‑convolution using the exact gradient.  An increasing schedule of
    the smoothness parameter `β` drives the solution towards the true
    max.  No forced symmetry is applied (the optimum need not be symmetric).

3.  **Non‑smooth sub‑gradient descent** – repeatedly identify the shift
    `k` where the convolution attains its maximum, compute the exact
    gradient of that term and take a projected gradient step on the simplex.
    A simple adaptive step‑size scheme is used.

4.  **Final simulated‑annealing hill‑climb** – a short SA walk with
    symmetric “swap / mass‑transfer’’ moves refines the result.

All steps respect the 1000 s wall‑clock budget and continuously keep the
best sequence seen so far.  The routine returns a plain Python list of
floats (the step heights).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (cheap seeds)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where
        f   = smooth max = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z  (gradient w.r.t. the soft‑max parameters)
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta
    w = expw / sumexp                                # soft‑max weights over shifts

    # derivative of ∑ w_k conv_k w.r.t. a
    # conv_k = a * a  => ∂conv_k/∂a = 2 * (a convolved with shift‑k basis)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]        # central slice, shape (n,)

    # back‑prop through the soft‑max (standard formula)
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS runs on the smooth‑max surrogate.
    `betas` is a list of smoothing parameters (increasing).
    No symmetry enforcement – the full search space is retained.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        z0 = np.log(a + 1e-12)          # initialise logits

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as e:
            print(f"  L‑BFGS failed: {e}")
            continue

        a = softmax(res.x)
        a = project_to_simplex(a)        # keep on the simplex

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution gradient (sub‑gradient descent)
# ----------------------------------------------------------------------


def max_conv_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the convolution term at shift k:
        g_i = ∂/∂a_i  Σ_j a_j a_{k‑j}
    It equals  a_{k‑i} + a_i (when the pair (i,k‑i) is valid);
    the vectorised implementation updates both ends of each pair.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start <= i_end:
        i = np.arange(i_start, i_end + 1)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]          # adds the symmetric contribution (double‑counts i=j correctly)
    return g


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        max_iters: int = 8000,
                        init_lr: float = 0.5):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Each iteration reduces the current maximal convolution value.
    """
    a = a_start.copy()
    n = a.shape[0]
    conv = convolution_fft(a, a)
    cur_idx = int(np.argmax(conv))
    cur_score = 2.0 * n * float(conv.max())

    best_a = a.copy()
    best_score = cur_score
    lr = init_lr

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        g = max_conv_gradient(a, cur_idx)

        cand = a - lr * g
        cand = project_to_simplex(cand)

        conv_cand = convolution_fft(cand, cand)
        cand_score = 2.0 * n * float(conv_cand.max())

        if cand_score < cur_score - 1e-12:
            # accept improvement
            a = cand
            cur_score = cand_score
            cur_idx = int(np.argmax(conv_cand))
            lr *= 1.02                # modestly increase step size
        else:
            # reject – shrink step size
            lr *= 0.70

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if it % 500 == 0:
            print(f"[{time.time():.1f}] sub‑grad iter {it}, best={best_score:.6f}, lr={lr:.2e}")

        # keep step size from collapsing
        if lr < 1e-9:
            lr = 1e-9

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    50 % swap, 50 % mass transfer.  Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    ``evaluate_sequence`` objective.  The routine respects the wall‑clock
    budget, builds analytic seeds, refines them with L‑BFGS on a smooth
    max‑approximation, applies a dedicated sub‑gradient descent on the
    exact max‑convolution and finishes with a short simulated‑annealing
    hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0           # safety margin
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback (previously discovered best)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 650 else 2048
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # 2. Coarse scan of simple analytic families (very cheap)
    # ------------------------------------------------------------------
    for alpha in np.linspace(0.0, 3.0, 13):
        if time.time() >= deadline - 120.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

    for beta in np.linspace(0.0, 3.0, 13):
        if time.time() >= deadline - 120.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

    for gamma in np.linspace(0.0, 4.0, 11):
        if time.time() >= deadline - 120.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS refinement (no symmetry enforcement)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 300.0:
        a_start = np.array(best_seq, dtype=np.float64)
        betas = [5, 20, 100, 500, 2000, 8000, 20000]
        a_opt, sc_opt = l_bfgs_optimize(a_start,
                                        betas=betas,
                                        max_iter_per_stage=200,
                                        deadline=deadline - 200.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = subgradient_descent(a_start,
                                            deadline=deadline - 70.0,
                                            max_iters=8000,
                                            init_lr=0.5)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final short simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=2_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Fallback in the extremely unlikely case we found nothing
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
