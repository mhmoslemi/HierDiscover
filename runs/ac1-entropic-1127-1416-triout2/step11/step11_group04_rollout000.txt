"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm (single‑thread, 1000 s budget)
-------------------------------------------------
1. Load the fallback sequence if present.
2. Quick analytic / random scan (≈15 s) to collect a few promising
   candidates (analytic families + random starts) – evaluate the exact
   objective and keep the best few.
3. For each promising seed run a multi‑stage quasi‑Newton optimisation:
   * Parameterise a = softmax(z) (automatically non‑negative, sum‑1).
   * Minimise the smooth “log‑sum‑exp” surrogate of the max‑convolution
     fβ(a) = (1/β)·log Σk exp(β·(a*a)k) with increasing β
     (β = 5 → 50 → 500 → 2000).
   * Use L‑BFGS‑B on the free variable z; gradient is obtained
     analytically (see code).  Each β‑stage is limited to a few
     hundred L‑BFGS iterations, keeping the total wall‑clock well
     under the budget.
   * After each stage optionally symmetrise a = (a+a[::-1])/2.
4. Run a short projected sub‑gradient descent from the best result
   (true max‑objective) to tighten the final value.
5. Final aggressive simulated‑annealing hill‑climb (mass‑transfer /
   swaps) using all remaining time.
6. Return the best sequence found (plain list of floats).

All phases respect the global deadline; the code works for any length
≤ 3000 (default 2048).
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variable used by the smooth‑max surrogate (updated before each call)
# ----------------------------------------------------------------------
CURRENT_BETA = 1.0


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
    2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families – used for the fast initial scan
# ----------------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smooth‑max oracle for L‑BFGS (value + gradient w.r.t. z)
# ----------------------------------------------------------------------
def smooth_obj_and_grad(z: np.ndarray):
    """
    Returns (f, grad_z) where
        f = (1/β)·log Σk exp(β·(a*a)k)
    and grad_z is ∇f(z) for a = softmax(z).
    """
    a = softmax(z)
    conv = convolution_fft(a, a)

    # log‑sum‑exp with a shift for numerical stability
    m = conv.max()
    exp_term = np.exp(CURRENT_BETA * (conv - m))
    Z = exp_term.sum()
    # smooth surrogate value
    f = (np.log(Z) + CURRENT_BETA * m) / CURRENT_BETA

    # weights for the smooth max
    w = exp_term / Z

    # gradient w.r.t. a : 2 * (a_rev * w) cross‑correlation
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)
    grad_a = 2.0 * conv_rev[len(a) - 1: len(a) - 1 + len(a)]

    # back‑prop through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return float(f), grad_z.astype(np.float64)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def subgrad_optimize(start_a: np.ndarray, deadline: float,
                     max_iters: int = 4_000_000, step0: float = 0.1):
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (shift k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal terms get factor 1 instead of 2
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = a - step * g
        # project back onto the probability simplex
        # (Euclidean projection; fast O(n log n))
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        a = np.maximum(a - theta, 0.0)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (mass‑transfer / swaps)
# ----------------------------------------------------------------------
def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    # small initial temperature proportional to current score
    T0 = cur_score * 5e-5
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑project onto simplex (soft‑max is cheap but we keep vector form)
        # Euclidean projection
        u = np.sort(cand)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        cand = np.maximum(cand - theta, 0.0)

        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score

# Main driver – respects the 1000 s wall‑clock limit
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    evaluate_sequence(...).  Returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 4.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback candidate if present
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working length (fallback length or default 2048)
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048
    if n > 3000:
        n = 2048

    # ------------------------------------------------------------------
    # 2. Very fast analytic / random scan (≈15 s) – collect a few seeds
    # ------------------------------------------------------------------
    scan_deadline = start_time + 15.0
    candidates = []

    # analytic families
    for alpha in np.linspace(0.0, 12.0, 13):
        if time.time() >= scan_deadline:
            break
        cand = symmetric_product_family(n, alpha)
        sc = compute_score(cand)
        candidates.append((sc, cand))
        if sc < best_score:
            best_score = sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] new best (sym α={alpha:.2f}) = {best_score:.6f}")

    for beta in np.linspace(0.0, 12.0, 13):
        if time.time() >= scan_deadline:
            break
        cand = left_heavy_power_family(n, beta)
        sc = compute_score(cand)
        candidates.append((sc, cand))
        if sc < best_score:
            best_score = sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] new best (left β={beta:.2f}) = {best_score:.6f}")

    for gamma in np.linspace(0.0, 6.0, 13):
        if time.time() >= scan_deadline:
            break
        cand = exp_u_family(n, gamma)
        sc = compute_score(cand)
        candidates.append((sc, cand))
        if sc < best_score:
            best_score = sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] new best (exp γ={gamma:.2f}) = {best_score:.6f}")

    # random starts
    for _ in range(30):
        if time.time() >= scan_deadline:
            break
        if np.random.rand() < 0.5:
            cand = random_monotone_start(n)
        else:
            cand = random_u_start(n)
        sc = compute_score(cand)
        candidates.append((sc, cand))
        if sc < best_score:
            best_score = sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # Keep the best few distinct seeds
    candidates.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in candidates[:7]]
    # ensure we also try the current best (if any)
    if best_seq is not None:
        seed_vectors.insert(0, np.array(best_seq, dtype=np.float64))

    # ------------------------------------------------------------------
    # 3. Multi‑stage L‑BFGS‑B optimisation on the smooth surrogate
    # ------------------------------------------------------------------
    beta_schedule = [5.0, 50.0, 500.0, 2000.0]
    iters_per_stage = [250, 250, 250, 250]  # total ≈ 1 k L‑BFGS iterations per seed
    eps = 1e-12

    for idx, start_vec in enumerate(seed_vectors):
        if time.time() >= deadline - 200.0:
            break
        print(f"[{time.time():.1f}] L‑BFGS start #{idx + 1}")

        a = np.array(start_vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        z = np.log(a + eps)

        for beta, maxit in zip(beta_schedule, iters_per_stage):
            if time.time() >= deadline - 200.0:
                break
            global CURRENT_BETA
            CURRENT_BETA = beta

            res = minimize(smooth_obj_and_grad,
                           z,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxit,
                                    'disp': False,
                                    'gtol': 1e-12})
            z = res.x
            a = softmax(z)

            # optional symmetrisation after each β‑stage
            a = (a + a[::-1]) * 0.5
            # (already normalised)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] improved (β={beta:.0f}) to {best_score:.6f}")

        # final sub‑gradient polish for this seed
        if time.time() < deadline - 120.0:
            a, sc = subgrad_optimize(a, deadline - 120.0,
                                     max_iters=1_000_000,
                                     step0=0.07)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] after sub‑grad (seed {idx+1}) best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Global hill‑climb (simulated annealing) with remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a0 = np.array(best_seq, dtype=np.float64) if best_seq is not None else np.full(n, 1.0 / n)
        a, sc = hill_climb_sa(a0, deadline, max_steps=6_000_000)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq if best_seq is not None else np.full(n, 1.0 / n).tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
