def fft_convolve(a,b):
    # compute linear convolution a*b (full) using FFT, possibly zero padding to next power of two.
    m = len(a) + len(b) - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    return np.fft.irfft(C, n=n_fft)[:m]

"""
step‑function optimiser – smoothed‐max gradient descent + hill‑climb

The objective of `evaluate_sequence` (for a non‑negative vector a)
is
    2·n·max_k (a * a)[k]     with    Σ a_i = 1 .
The maximum of the auto‑convolution is a non‑smooth function.
We first minimise a smooth approximation
    fβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
whose gradient can be written analytically as a correlation
between a and the soft‑max weights w_k ∝ exp(β·(a*a)[k]).
A simple projected gradient descent with an annealing schedule for β
produces a much flatter convolution than the plain uniform vector.
The best iterate is then polished by a cheap stochastic hill‑climb.

All heavy lifting (convolutions, correlations) is performed by FFTs,
so each iteration is O(n log n).  The routine respects the 1000 s wall‑clock
budget, falls back to a uniform vector if nothing else succeeds, and
finally reports the score using the official `evaluate_sequence`.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Linear‑algebra helpers (FFT based)
# --------------------------------------------------------------
def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of two real vectors via FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·len(a)-1)."""
    return fft_convolve(a, a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Exact objective (2·n·max(conv(a,a))) for a normalised vector (Σa=1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Simple parametric families – cheap global sweep
# --------------------------------------------------------------
def best_initial_vector(n: int, time_deadline: float) -> np.ndarray | None:
    """
    Scan a few low‑dimensional families (power‑law, geometric,
    plateau‑plus‑tail) and return the best normalised vector seen.
    """
    best_a = None
    best_val = float('inf')

    # ---- power‑law -------------------------------------------------
    for p in np.linspace(0.30, 1.00, 71):          # ≈0.01 step
        if time.time() >= time_deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ---- geometric decay -------------------------------------------
    for r in np.linspace(0.50, 0.99, 81):
        if time.time() >= time_deadline:
            break
        a = (r ** np.arange(n, dtype=np.float64))
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ---- plateau + power‑law tail ----------------------------------
    for L_frac in np.linspace(0.05, 0.30, 6):
        if time.time() >= time_deadline:
            break
        L = int(L_frac * n)
        if L == 0 or L >= n:
            continue
        for p in np.linspace(0.4, 0.8, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head = np.full(L, tail.max())
            a = np.concatenate([head, tail])
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    return best_a


# --------------------------------------------------------------
# Smoothed‑max projected gradient descent
# --------------------------------------------------------------
def smooth_gradient_descent(a0: np.ndarray,
                           time_deadline: float,
                           beta_start: float = 2.0,
                           beta_max: float = 200.0,
                           step0: float = 2.0) -> np.ndarray:
    """
    Minimise the smoothed max ‑ objective
        fβ(a) = (1/β)·log Σ exp(β·conv(a,a))
    by projected gradient descent with an annealing schedule for β.
    The function returns the best vector (lowest max‑convolution) found
    before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # current best (w.r.t. the true objective)
    best_a = a.copy()
    best_M = full_convolution(a).max()

    it = 0
    beta = beta_start

    # occasional progress reports
    last_report = time.time()

    while time.time() < time_deadline:
        it += 1

        # --- current convolution and soft‑max weights ----------------
        conv = full_convolution(a)                     # length 2n‑1
        M = conv.max()
        # soft‑max (subtract M for numerical stability)
        scaled = beta * (conv - M)
        w = np.exp(scaled)
        w /= w.sum()                                   # probabilities over shifts

        # --- gradient = 2·corr(w, a) (cross‑correlation) ----------
        # correlation can be realised as convolution with the reversed weight vector
        w_rev = w[::-1]
        grad_full = fft_convolve(w_rev, a)             # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]        # central slice = correlation

        # --- simple back‑tracking line‑search ------------------------
        step = step0 / np.sqrt(it)
        cand_a = a - step * grad
        cand_a = project_to_simplex(cand_a)
        cand_M = full_convolution(cand_a).max()

        # try a few smaller steps if the first one did not improve
        if cand_M >= M:
            for factor in (0.5, 0.2, 0.1):
                cand = a - step * factor * grad
                cand = project_to_simplex(cand)
                cand_M2 = full_convolution(cand).max()
                if cand_M2 < cand_M:
                    cand_a, cand_M = cand, cand_M2
                if cand_M2 < M:
                    break

        # --- accept if we made progress ------------------------------
        if cand_M < M:
            a = cand_a
            M = cand_M

        # store the best true objective seen so far
        if M < best_M:
            best_M, best_a = M, a.copy()

        # --- anneal β (increase slowly) -------------------------------
        if it % 500 == 0:
            beta = min(beta * 1.05, beta_max)

        # --- occasional jitter to escape tiny plateaus ---------------
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.02 / np.sqrt(it))
            a = project_to_simplex(a + jitter)

        # --- periodic logging -----------------------------------------
        if time.time() - last_report > 30.0:
            print(f"[smooth‑GD] it={it:,} β={beta:.1f} best‑M={best_M:.6e}")
            last_report = time.time()

    return best_a


# --------------------------------------------------------------
# Stochastic hill‑climb (fine‑tuning)
# --------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               time_deadline: float,
               max_iter: int = 10_000_000) -> np.ndarray:
    """
    Tiny greedy moves: transfer a fraction of mass from one index to another.
    Only moves that strictly improve the true max‑convolution are kept.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M, best_a = M_cand, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1

        if no_improve > 200_000:
            # random jitter to hopefully escape a flat region
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

    return best_a

    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. optional fallback (previously discovered good vector)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. decide problem size (use fallback length if known)
    # -----------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # -----------------------------------------------------------------
    # 2. quick global parametric sweep for a decent start
    # -----------------------------------------------------------------
    pll_deadline = deadline - 400.0
    a_initial = best_initial_vector(n, pll_deadline)
    if a_initial is not None:
        val = compute_score(a_initial)
        print(f"[global sweep] best score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 3. smoothed‑max projected gradient descent
    # -----------------------------------------------------------------
    if best_seq is not None:
        a0 = np.array(best_seq, dtype=float)
    else:
        a0 = np.full(n, 1.0 / n, dtype=float)

    gd_deadline = deadline - 200.0
    print("[smooth‑GD] start")
    a_refined = smooth_gradient_descent(a0, gd_deadline)
    val = compute_score(a_refined)
    print(f"[smooth‑GD] refined score = {val:.6f}")
    if val < best_val:
        best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. final stochastic hill‑climb polishing
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[hill‑climb] start")
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final evaluation (official)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
