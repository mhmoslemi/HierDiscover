"""
Multi‑stage non‑symmetric optimisation for the self‑convolution peak.

The search proceeds as follows (all steps respect the wall‑clock budget):
  1. Scan a collection of analytic families (monotone power, exponential‑decay,
     symmetric product, U‑shapes, mixtures).  The best candidate for each
     length is kept.
  2. Refine the best candidate with a smooth‑max Adam optimiser – **without**
     any forced symmetry (the earlier code symmetrised every few thousand
     steps, which limited the attainable bound).
  3. Perform a long projected sub‑gradient descent on the true max‑objective.
  4. Optionally repeat Adam with a higher smoothness temperature.
  5. Finish with a simulated‑annealing hill‑climber that works directly on
     the simplex.
  6. Keep the overall best vector (any length) and return it as a plain list.

All vectors are kept on the probability simplex (∑a = 1) and are clipped only
by the helper ``project_to_simplex``.  The routine is deliberately
parameter‑rich so that the search can continue for the full 1000 s budget.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    (where * denotes full convolution).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric product family: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_decay_family(n: int, gamma: float) -> np.ndarray:
    """Strictly decreasing exponential: a_i ∝ exp(-γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_decay(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑decay."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    decay = exp_decay_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * decay
    w /= w.sum()
    return w


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (no forced symmetry)
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 800_000,
    beta_start: float = 10.0,
    beta_end: float = 5000.0,
    lr: float = 0.015,
    sym_interval: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the convolution max.
    Parameters are the same as before; setting ``sym_interval=0`` disables any
    explicit symmetry enforcement.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for the smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # optional symmetry step – disabled by default
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derived from smooth‑max surrogate)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true evaluation
        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection and evaluation
    a_fin = softmax(z)
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 3_000_000,
    step0: float = 0.05,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent for the exact max‑objective.
    The iterate stays on the simplex via Euclidean projection.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution term
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (no symmetry)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 5_000_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple random walk on the simplex with a tiny temperature schedule.
    Moves are swaps or small mass transfers.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section refinement (single‑parameter families)
# ----------------------------------------------------------------------


def golden_section_optimize(
    func, a: float, b: float, max_iter: int = 12
) -> Tuple[float, float]:
    """Minimise ``func`` on the interval [a,b] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Main driver – the required ``propose_candidate`` function
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns a plain Python list of floats (the best sequence found before the
    deadline).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Try to use the previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence(height_sequence_1)  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = list(height_sequence_1)  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to explore (larger n → finer control but higher factor 2n)
    # ------------------------------------------------------------------
    length_candidates = [2048, 4096]

    for n in length_candidates:
        if time.time() >= deadline - 30.0:
            break
        print(f"[{time.time():.1f}] --- Exploring length n = {n} ---")

        # --------------------------------------------------------------
        # 2. Coarse scan of analytic families (no symmetry enforcement)
        # --------------------------------------------------------------
        def try_candidate(vec: np.ndarray, source: str) -> None:
            nonlocal best_seq, best_score
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] NEW BEST ({source}) = {best_score:.6f}")

        # product‑family (symmetric; still useful as a starter)
        for alpha in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 10.0:
                break
            try_candidate(symmetric_product_family(n, alpha), f"prod‑α={alpha:.2f}")

        # left‑heavy power‑law (monotone decreasing)
        for beta in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 10.0:
                break
            try_candidate(left_heavy_power_family(n, beta), f"pow‑β={beta:.2f}")

        # exponential decay (strictly decreasing)
        for gamma in np.linspace(0.0, 6.0, 31):
            if time.time() >= deadline - 10.0:
                break
            try_candidate(exp_decay_family(n, gamma), f"exp‑γ={gamma:.2f}")

        # U‑shape (symmetric) – included for completeness
        for gamma in np.linspace(0.0, 6.0, 31):
            if time.time() >= deadline - 10.0:
                break
            try_candidate(exp_u_family(n, gamma), f"U‑γ={gamma:.2f}")

        # mixture uniform + decay (adds a "flat floor")
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for gamma in np.linspace(0.0, 6.0, 13):
                if time.time() >= deadline - 10.0:
                    break
                try_candidate(mixed_uniform_decay(n, lam, gamma),
                               f"mix λ={lam:.2f},γ={gamma:.2f}")

        # random decreasing start (used later for extra restarts)
        for _ in range(5):
            if time.time() >= deadline - 10.0:
                break
            try_candidate(random_monotone_start(n), "rand‑dec")

        # --------------------------------------------------------------
        # 3. Fine golden‑section refinement of the most promising product‑α
        # --------------------------------------------------------------
        # (Only if we have spotted a promising α during the coarse scan)
        # We'll locate the α that gave the current best (if it was a product)
        if best_seq is not None and time.time() < deadline - 5.0:
            # Heuristic: try to refine around the best product α seen so far
            # (store the last product α that gave a best – we approximate by re‑scanning)
            # For simplicity we just run a short golden‑section on the product family
            # over the whole interval.
            def prod_score(alpha):
                vec = symmetric_product_family(n, alpha)
                return compute_score(vec)

            lo, hi = 0.0, 12.0
            best_alpha, best_alpha_score = golden_section_optimize(
                prod_score, lo, hi, max_iter=12
            )
            vec = symmetric_product_family(n, best_alpha)
            try_candidate(vec, f"prod‑α*={best_alpha:.4f}")

        # --------------------------------------------------------------
        # 4. Adam refinement (smooth‑max) – **no symmetrisation**
        # --------------------------------------------------------------
        if best_seq is not None and time.time() < deadline - 250.0:
            print(f"[{time.time():.1f}] Adam refinement (no sym) on n={n}")
            a_start = np.array(best_seq, dtype=np.float64)
            a_opt, sc_opt = adam_smooth_optimize(
                a_start,
                deadline - 150.0,
                max_iters=600_000,
                beta_start=10.0,
                beta_end=5000.0,
                lr=0.015,
                sym_interval=0,  # keep asymmetry
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5. Projected sub‑gradient descent (exact max)
        # --------------------------------------------------------------
        if best_seq is not None and time.time() < deadline - 150.0:
            print(f"[{time.time():.1f}] Sub‑gradient descent (exact max)")
            a_sub, sc_sub = subgrad_optimize(
                np.array(best_seq, dtype=np.float64),
                deadline - 50.0,
                max_iters=2_500_000,
                step0=0.045,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 6. Second Adam pass (higher temperature, still unsymmetrised)
        # --------------------------------------------------------------
        if best_seq is not None and time.time() < deadline - 80.0:
            print(f"[{time.time():.1f}] Second Adam pass (high β, no sym)")
            a_start = np.array(best_seq, dtype=np.float64)
            a_opt2, sc_opt2 = adam_smooth_optimize(
                a_start,
                deadline - 30.0,
                max_iters=300_000,
                beta_start=50.0,
                beta_end=8000.0,
                lr=0.012,
                sym_interval=0,
            )
            if sc_opt2 < best_score:
                best_score = sc_opt2
                best_seq = a_opt2.tolist()
                print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 7. Final hill‑climb (simulated annealing) – keep asymmetry
        # --------------------------------------------------------------
        if best_seq is not None and time.time() < deadline - 5.0:
            print(f"[{time.time():.1f}] Final hill‑climb (SA) on n={n}")
            a_final, sc_final = hill_climb_sa(
                np.array(best_seq, dtype=np.float64),
                deadline,
                max_steps=5_000_000,
            )
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.tolist()
                print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. If nothing succeeded, fall back to a safe uniform distribution.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = compute_score(np.array(best_seq, dtype=np.float64))
        print("[fallback] using uniform vector")

    # ------------------------------------------------------------------
    # 9. Final official evaluation (includes all security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
