"""
Conditional‑gradient (LP) + Adam (smooth‑max) + Simulated‑Annealing search.

The routine works on a probability vector (sum = 1).  It first applies a
few Frank‑Wolfe / conditional‑gradient steps in which a linear program
maximises the L1‑mass of a partner vector g under the current peak‑height
constraint ‖f∗g‖∞ ≤ M.  After a short CG phase we switch to the Adam
optimizer on a log‑sum‑exp surrogate of the max‑convolution, finally
polishing the result with a cheap simulated‑annealing hill‑climber.
All phases respect the global 1000 s wall‑clock budget and the best
sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, score, simplex projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a vector with ∑a = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used only for a fallback start)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Conditional‑gradient (LP) optimisation
# ----------------------------------------------------------------------


def _build_sparse_pattern(n: int):
    """Pre‑compute row / column index arrays for the Toeplitz convolution matrix."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    idx = 0
    for j in range(n):
        rows[idx:idx + n] = np.arange(j, j + n)
        cols[idx:idx + n] = j
        idx += n
    return rows, cols


def conditional_gradient_optimize(a_start: np.ndarray,
                                  deadline: float,
                                  max_iters: int = 30,
                                  init_step: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style optimisation.
    Each iteration solves  max ∑g_j   s.t.  (a∗g)_i ≤ M  ,  g≥0
    and moves a a little towards the solution.
    """
    n = a_start.shape[0]
    rows, cols = _build_sparse_pattern(n)
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    step = init_step
    stagnation = 0

    it = 0
    while time.time() < deadline - 2 and it < max_iters:
        it += 1
        conv = convolution_fft(a, a)
        M = conv.max()

        # ----- linear programme -----
        data = np.tile(a, n)                     # (n*n,)  – entries of the Toeplitz matrix
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)                          # maximise ∑g  →  minimise -∑g
        b = np.full(2 * n - 1, M)

        res = opt.linprog(c, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True})

        if not res.success:
            # if the LP crashes we abort the CG phase
            print(f"[{time.time():.1f}] LP failed at iter {it} – exiting CG")
            break

        g0 = res.x
        # symmetrise & normalise (so that we stay on the simplex)
        g0 = (g0 + g0[::-1]) * 0.5
        s = g0.sum()
        if s <= 1e-12:
            break
        g = g0 / s

        # ----- convex combination -----
        a_new = (1.0 - step) * a + step * g
        a_new = (a_new + a_new[::-1]) * 0.5       # enforce symmetry
        a_new /= a_new.sum()                     # restore ∑=1 (numerical safety)

        sc_new = compute_score(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            step = min(0.5, step * 1.05)         # become a bit more aggressive
            stagnation = 0
            print(f"[{time.time():.1f}] CG iter {it}: improved to {best_score:.6f}")
        else:
            # no improvement → shrink step size
            step = max(0.01, step * 0.70)
            stagnation += 1
            if stagnation >= 8:
                # stop early if we are stuck
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (same as in the reference code)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 800_000,
                         beta_start: float = 10.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2500) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(conv)."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)                     # soft‑max parameters

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for the smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # smooth‑max weights

        # gradient of the smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑propagation through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improvement to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (same as in the reference code)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a tiny cooling schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector)
    minimising ``evaluate_sequence``.  The routine respects the wall‑clock
    budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Load a fallback construction if available
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_vec = project_to_simplex(np.array(fb_vec, dtype=np.float64))
        fb = evaluate_sequence(fb_vec.tolist())
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Choose working length (keep 2048 – the size of the known best)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
    else:
        n = 2048
        best_seq = symmetric_product_family(n, alpha=4.0).tolist()

    a = np.array(best_seq, dtype=np.float64)
    a = project_to_simplex(a)               # ensure we are on the simplex
    best_score = compute_score(a)
    best_a = a.copy()

    print(f"[{time.time():.1f}] start length = {n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Conditional‑gradient (LP) phase – a few aggressive steps
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] entering conditional‑gradient phase")
        cg_a, cg_score = conditional_gradient_optimize(
            a_start=best_a,
            deadline=deadline - 120.0,
            max_iters=25,
            init_step=0.25,
        )
        if cg_score < best_score:
            best_score = cg_score
            best_a = cg_a.copy()
            print(f"[{time.time():.1f}] after CG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimisation on the smooth surrogate (the heavy lifting)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] Adam refinement")
        adam_a, adam_score = adam_smooth_optimize(
            start_a=best_a,
            deadline=deadline - 40.0,
            max_iters=600_000,
            beta_start=12.0,
            beta_end=6000.0,
            lr=0.012,
            sym_interval=2000,
        )
        if adam_score < best_score:
            best_score = adam_score
            best_a = adam_a.copy()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Explicit symmetrisation (cheap but sometimes helpful)
    # ------------------------------------------------------------------
    a_sym = (best_a + best_a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sym_score = compute_score(a_sym)
    if sym_score < best_score:
        best_score = sym_score
        best_a = a_sym.copy()
        print(f"[{time.time():.1f}] symmetrisation improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final hill‑climb (simulated annealing) for polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA polishing")
        sa_a, sa_score = hill_climb_sa(
            start_a=best_a,
            deadline=deadline,
            max_steps=4_000_000,
        )
        if sa_score < best_score:
            best_score = sa_score
            best_a = sa_a.copy()
            print(f"[{time.time():.1f}] SA final improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Return the best sequence found (as a plain list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_report = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_report:.6f}")
    return final_seq
