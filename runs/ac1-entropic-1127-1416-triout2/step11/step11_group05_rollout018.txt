"""
Propose a step‑function sequence that minimises the evaluation function.

The algorithm consists of three stages:

1. **Large‑scale deterministic search** – we evaluate several
   analytically‑defined families (power law, geometric decay and a
   product of both) on a fine grid of parameters.  All candidates are
   normalised to a probability vector and scored with a fast FFT‑based
   convolution (exact objective).

2. **Smooth‑max L‑BFGS refinement** – starting from the best candidate we
   minimise a smooth approximation of the max‑convolution using the
   parameterisation `a = softmax(z)`.  The smoothness parameter β is
   increased gradually so that the solution approaches the true max.
   No symmetry is enforced (the previous runner forced symmetry,
   which limits the attainable bound).

3. **Short simulated‑annealing hill‑climb** – a light stochastic walk
   on the simplex (random swaps and mass‑transfer moves) removes any
   remaining asymmetries and yields a final improvement.

The routine respects a 1000 s wall‑clock budget, prints intermediate
scores and finally returns the best sequence found (as a plain Python
list of floats).
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returned vector sums to 1 and contains only non‑negative entries.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for fast seeding)
# ----------------------------------------------------------------------


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """ a_i ∝ (i+½)^{‑β}, monotone decreasing. """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def geometric_family(n: int, delta: float) -> np.ndarray:
    """
    a_i ∝ (1‑δ)^{i}, i = 0 … n‑1.
    `delta` is a small positive number; the shape is exponentially
    decreasing.  The vector is normalised to sum to 1.
    """
    r = 1.0 - delta
    i = np.arange(n, dtype=np.float64)
    w = np.exp(i * np.log(r))          # r**i
    w /= w.sum()
    return w


def mixed_power_exp_family(n: int, beta: float, delta: float) -> np.ndarray:
    """
    a_i ∝ (i+½)^{‑β}·(1‑δ)^{i}.
    This combines a power‑law decay with an exponential tail.
    """
    r = 1.0 - delta
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5) + i * np.log(r))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where

        f   = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z   (gradient w.r.t. soft‑max parameters)

    `conv` is the full self‑convolution of the soft‑max probabilities.
    The gradient is derived analytically (no finite differences).
    """
    a = softmax(z)                                  # probability vector
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # soft‑max weights over the convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta                 # smooth approximation
    w = expw / sumexp                                # normalized weights

    # d(maxc)/da + ∑_k w_k d(conv_k)/da
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]        # shape (n,)

    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation (no symmetry enforcement)
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    beta_schedule,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS minimisation of the smooth‑max surrogate.
    `beta_schedule` – increasing list of smoothing parameters.
    Returns the best probability vector found and its exact score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        # initialise L‑BFGS variables (log‑coordinates)
        z0 = np.log(a + 1e-12)

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun, z0, method="L-BFGS-B", jac=True, options=opts)
        except Exception as e:
            print(f"  L‑BFGS failure: {e}")
            continue

        a = softmax(res.x)                 # back to probability simplex

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    new best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    50 % swap two entries, 50 % transfer a fraction of mass.
    Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = max(1e-12, best_score * 1e-4)  # starting temperature
    step_scale = 0.08                     # controls size of mass transfers

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()

        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                           # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                # avoid tiny negative rounding errors
                if cand[i] < 0.0:
                    cand[j] += cand[i]   # give back the leftover
                    cand[i] = 0.0

        cand_score = compute_score(cand)

        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score /
                                                       (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase step size when improving
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # slowly shrink step size on rejections
            step_scale *= 0.97

    return best_a, best_score

            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
