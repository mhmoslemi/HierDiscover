"""
Advanced step‑function optimiser (v2).

What changed compared to the previous version?
------------------------------------------------
*   Added several classic window shapes (Hann, Hamming, Black‑man,
    Kaiser) as extra seeds – many of them are known to produce very flat
    autocorrelations.
*   Performed a quick “sub‑gradient warm‑up’’ on every seed (a few
    thousand iterations).  This dramatically increases the diversity of
    starting points before the heavy optimisation.
*   Re‑balanced the time budget:
        30 % – soft‑max (log‑sum‑exp) descent,
        25 % – variance‑flattening descent,
        25 % – long sub‑gradient refinement,
        20 % – simulated‑annealing hill‑climb.
*   Replaced the deterministic hill‑climb with a proper simulated‑annealing
    scheme that can accept occasional worsening moves, helping to escape
    shallow local minima.
*   The whole pipeline automatically picks a longer sequence (4096) when
    the time budget allows it.

The optimiser returns a plain Python list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (exact for our lengths)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators (including classic window functions)
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    return a / a.sum()


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    return w / w.sum()


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    return w / w.sum()


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    return a / a.sum()


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: choose `k` positions, assign random weights, normalise."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    return a / a.sum()


def generate_hann(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=float)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def generate_hamming(n: int) -> np.ndarray:
    """Hamming window."""
    i = np.arange(n, dtype=float)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def generate_blackman(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=float)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return w / w.sum()


def generate_kaiser(n: int, beta: float = 5.0) -> np.ndarray:
    """Kaiser window – β controls side‑lobe attenuation."""
    i = np.arange(n, dtype=float)
    alpha = (n - 1) / 2.0
    x = (i - alpha) / alpha  # runs from -1 to 1
    w = i0(beta * np.sqrt(1.0 - x ** 2))
    return w / w.sum()


# ----------------------------------------------------------------------
# Correlation helper used by the variance‑flattening phase
# ----------------------------------------------------------------------


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation (length = len(a)):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    Implemented via one FFT‑convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Phase 1 – soft‑max gradient descent (smooth surrogate of max‑norm)
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2500.0,
    step_coef: float = 0.001,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score
    it = 0
    start = time.time()

    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)  # ≤ 1
        w = exp_vals / exp_vals.sum()

        grad = 2.0 * beta * np.correlate(w, cur, mode="valid")
        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        nxt = cur - step * grad
        cur = project_to_simplex(nxt)

        if it % 5000 == 0:
            print(
                f"[soft‑max] it {it:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.1f}"
            )
    return best, best_score


# ----------------------------------------------------------------------
# Phase 2 – variance‑flattening gradient descent
# ----------------------------------------------------------------------


def variance_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)
    it = 0

    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        e = conv - avg
        grad = 4.0 * correlation_valid(e, cur)

        it += 1
        step = step0 / np.sqrt(it)

        nxt = cur - step * grad
        cur = project_to_simplex(nxt)

        if it % 2000 == 0:
            print(
                f"[var‑GD] it {it:6d} | time {time.time():6.1f}s | "
                f"best {best_score:.6f}"
            )
    return best, best_score

# Phase 3 – sub‑gradient refinement (true max‑objective)
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        step = step0 / np.sqrt(t)
        cur -= step * g
        cur = project_to_simplex(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        if t % 1000 == 0:
            # small jitter prevents stagnation
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur += jitter
            cur = project_to_simplex(cur)

    return best, best_score


# ----------------------------------------------------------------------
# Phase 4 – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a: np.ndarray, deadline: float, max_steps: int = 200_000
) -> tuple[np.ndarray, float]:
    """
    Metropolis hill‑climb that accepts occasional worsening moves.
    The temperature decays exponentially.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # initial temperature roughly 1 % of the current score
    T0 = cur_score * 0.01
    decay = 0.9995  # per step
    T = T0

    steps = 0
    while steps < max_steps and time.time() < deadline:
        # ---- propose a move -------------------------------------------------
        cand = cur.copy()
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if cand[i] > 1e-12 and i != j:
                delta = (np.random.rand() ** 2) * cand[i] * 0.5
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)  # tiny drift correction
        cand_score = compute_score(cand)

        # ---- Metropolis acceptance -----------------------------------------
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / (T + 1e-16))
            accept = np.random.rand() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                # raise temperature a bit when we find a new record
                T = min(T * 1.05, T0)

        # temperature decay
        T = max(T * decay, 1e-12)

        steps += 1
        if steps % 5000 == 0:
            print(
                f"[SA] step {steps:6d} | time {time.time():6.1f}s | "
                f"best {best_score:.6f} | T {T:.4e}"
            )
    return best, best_score

def quick_subgrad_refine(a: np.ndarray, iters: int = 3000) -> np.ndarray:
    """
    A very short sub‑gradient run – cheap enough to be applied to many seeds.
    """
    n = a.shape[0]
    cur = a.copy()
    for t in range(1, iters + 1):
        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        step = 0.05 / np.sqrt(t)
        cur -= step * g
        cur = project_to_simplex(cur)
    return cur

    now = time.time()
    remaining = deadline - now
    phase1_dead = now + remaining * 0.30
    phase2_dead = phase1_dead + remaining * 0.25
    phase3_dead = phase2_dead + remaining * 0.25
    # the rest (≈20 %) goes to simulated annealing

    # ---- Phase 1 : soft‑max gradient descent -----------------------
    print("\n[Phase 1] soft‑max gradient descent")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a,
        cur_score,
        phase1_dead,
        beta_start=5.0,
        beta_end=2500.0,
        step_coef=0.001,
    )
    best_val = min(best_val, cur_score)
    print(f"[after Phase 1] best score = {best_val:.6f}")

    # ---- Phase 2 : variance‑flattening -----------------------------
    if time.time() < phase2_dead:
        print("\n[Phase 2] variance‑flattening gradient descent")
        cur_a, cur_score = variance_gradient_descent(
            cur_a, cur_score, phase2_dead, step0=0.05
        )
        best_val = min(best_val, cur_score)
        print(f"[after Phase 2] best score = {best_val:.6f}")

    # ---- Phase 3 : long sub‑gradient refinement --------------------
    if time.time() < phase3_dead:
        print("\n[Phase 3] sub‑gradient refinement")
        cur_a, cur_score = subgradient_refine(
            cur_a,
            cur_score,
            phase3_dead,
            max_iter=800_000,   # a bit longer than the previous version
            step0=0.07,
        )
        best_val = min(best_val, cur_score)
        print(f"[after Phase 3] best score = {best_val:.6f}")

    # ---- Phase 4 : simulated‑annealing hill‑climb ------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 4] simulated‑annealing hill‑climb")
        cur_a, cur_score = annealing_hill_climb(cur_a, deadline, max_steps=300_000)
        best_val = min(best_val, cur_score)
        print(f"[after Phase 4] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion & official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
