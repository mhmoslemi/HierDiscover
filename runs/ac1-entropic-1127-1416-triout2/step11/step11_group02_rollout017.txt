- n = len(sequence)
- b_sequence = np.convolve(sequence, sequence) (full linear convolution)
- max_b = max(b_sequence)
- sum_a = np.sum(sequence)

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}= \frac{2n\,\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

Because the score is *scale‑invariant* we may restrict ourselves to
probability vectors (\(\sum_i a_i = 1\)).
The problem is then equivalent to making the discrete convolution
\(c_k=\sum_i a_i a_{k-i}\) as flat as possible – we want to reduce its
largest entry.

The previous optimiser used a coarse sweep over a few simple families
(power‑law, geometric, a short plateau + power‑law).  
The best vector found (score ≈ 1.511) is only a little better than the
uniform distribution (score = 2).  A modest refinement of the search
space already yields a noticeable improvement.

### What the new code does

1. **Extensive parametric sweep**  
   * families explored  
     * **Power‑law** \(a_i\propto (i+1)^{-p}\)  \(p\in[0.1,2.0]\)  
     * **Geometric** \(a_i\propto q^{i}\)  \(q\in[0.90,0.9999]\)  
     * **Mixed plateau + power‑law** plateau length \(L\) and exponent \(p\)  
     * **Beta‑shaped** \(a_i\propto x_i^{\alpha}(1-x_i)^{\beta}\) with
       \(x_i=(i+0.5)/n\).  
   * a **fine‑grid** around the best parameters found so far and a
     small set of **random convex combinations** of the basic families.
   * every candidate is normalised to sum = 1 and scored with the cheap
     `compute_score` (exact value of the objective for normalised vectors).

2. **Projected sub‑gradient polishing** – identical to the original
   implementation but with a larger iteration budget (the whole remaining
   time).  The sub‑gradient is the sum of the shifted copies of the
   current vector for all *active* convolution positions.

3. **Greedy stochastic hill‑climb** – the same neighbour generation as
   before, but with a slightly more aggressive step‑size schedule and a
   higher iteration limit.

4. **Fallback** – if nothing better is found the uniform vector is used.

All phases respect the global wall‑clock budget (default 1000 s).  The
final sequence is returned as a plain Python list; the official
`evaluate_sequence` function will compute the exact score.

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i} (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """
    First L = floor(L_frac * n) entries are equal to the maximum of the
    power‑law tail, the remaining entries follow the same power‑law.
    """
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ x_i^{alpha} (1-x_i)^{beta},
    where x_i = (i+0.5)/n.
    Allows a wide range of symmetric / skewed shapes.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid NaNs for extreme negative exponents by clipping x away from 0/1
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Parametric sweep – tries many families and keeps the best one.
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    """Search a wide variety of parametric families and return the best vector."""
    best_vec = None
    best_val = float('inf')

    # ----- coarse power‑law sweep -----
    for p in np.linspace(0.10, 2.00, 191):            # step ≈0.01
        if time.time() >= time_deadline:
            return best_vec
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ----- fine power‑law around best p (if we have one) -----
    if best_vec is not None:
        # approximate current p from the vector (we stored it)
        # simply repeat a finer grid around the region where improvement was seen
        for p in np.linspace(0.30, 1.20, 451):       # step ≈0.002
            if time.time() >= time_deadline:
                return best_vec
            a = power_law(n, p)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- geometric sweep extended to ratios extremely close to 1 -----
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),              # 0.90 … 0.99
            np.linspace(0.990, 0.9999, 110))):       # finer near 1
        if time.time() >= time_deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ----- mixed plateau + power‑law with finer sampling -----
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= time_deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- beta‑shape sweep (both symmetric and skewed) -----
    # range for exponents: [-0.8, 3] (step 0.2)
    alphas = np.linspace(-0.8, 3.0, 20)
    betas = np.linspace(-0.8, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= time_deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- random convex combinations of three basic shapes -----
    # basic shapes
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    power = power_law(n, 0.5)
    geom = geometric(n, 0.98)

    basic = [uniform, power, geom]

    for _ in range(2000):
        if time.time() >= time_deadline:
            break
        coeffs = np.random.rand(3)
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basic))
        a = project_to_simplex(a)                 # ensure normalisation
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec

# ----------------------------------------------------------------------
# Sub‑gradient polishing (same as before but without an artificial step‑size cap)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of each active entry: 2 * a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # tiny jitter every few thousand steps
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simple hill‑climb, more aggressive than the reference version
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    # start with a moderate step scale relative to average entry (=1/n)
    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of the source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)     # keep non‑negative
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)   # become bolder after improvement
            no_improve = 0
        else:
            step_scale *= 0.995                     # slowly shrink step if no improvement
            no_improve += 1

        it += 1
        # occasional global jitter to escape flat plateaus
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Fallback – use the previously discovered construction if present
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Very broad parametric sweep (≈ 400 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Sub‑gradient polishing (the bulk of the remaining time)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.15)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
