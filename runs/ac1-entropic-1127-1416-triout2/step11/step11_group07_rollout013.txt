def lp_update(f, t=0.5):
    n = len(f)
    M = np.max(convolution_fft(f, f))
    A = build_convolution_matrix(f)
    # b of size m: M
    b = np.full(2*n-1, M)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return f
    g = res.x
    sum_g = g.sum()
    if sum_g == 0:
        return f
    g = g / sum_g
    # mixture update
    new_f = (1 - t) * f + t * g
    new_f = new_f / new_f.sum()
    return new_f
```

"""
Advanced optimizer for the step‑function bound.

The strategy combines three complementary stages:

1. **Diverse seeding** – generate a large pool of candidate probability
   vectors (sine‑squared, beta‑pdf families, Dirichlet draws, sparse spikes)
   and keep the best one according to the exact objective
   `2·n·max(conv(a,a))`.

2. **LP‑based refinement** – following the idea from the original paper,
   we solve a linear programme that, given the current vector `f`,
   finds another non‑negative vector `g` with maximal total mass while
   respecting `conv(f,g) ≤ M`, where `M = max(conv(f,f))`.  Mixing a
   fraction of `g` into `f` strictly reduces the maximal convolution.
   Only a handful of such LP‑steps are needed (the matrix is rebuilt each
   iteration – the cost is negligible compared with the time budget).

3. **Simulated‑annealing polishing** – a long stochastic search that
   repeatedly perturbs the current distribution, accepts worsening moves
   with a temperature that linearly cools to zero, and keeps the best
   ever seen.  Moves consist of mass transfer, simple swaps and Gaussian
   jitter, always re‑normalised onto the probability simplex.

A short deterministic hill‑climb is run at the end for final polishing.
All phases respect the overall time budget (≈ 1000 s).  The returned
sequence is a plain Python list of non‑negative floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast and accurate)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (Σ a_i = 1):
        2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector)
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Helper for the LP‑based refinement
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that A @ g = conv(f, g).
    Column j contains a copy of f placed at rows j … j+n‑1.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(f: np.ndarray, mix: float = 0.3) -> np.ndarray:
    """
    One LP‑based improvement step.
    Maximise Σ g_i subject to g_i ≥ 0 and conv(f, g) ≤ M,
    where M = max(conv(f,f)).
    The solution g is re‑normalised to unit sum and mixed
    with f:   f ← (1−mix)·f + mix·g .
    """
    n = f.shape[0]
    M = np.max(convolution_fft(f, f))
    A = build_convolution_matrix(f)               # (2n‑1) × n
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)                               # maximise Σ g_i  → minimise −Σ g_i

    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if not res.success:
        # If the LP fails for any reason, simply return the original vector.
        return f

    g = res.x
    s = g.sum()
    if s == 0.0:
        return f
    g = g / s                                     # re‑normalise to sum = 1
    new_f = (1.0 - mix) * f + mix * g
    new_f = new_f / new_f.sum()
    return new_f


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    start_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic search with a linearly cooling temperature.
    Moves:
        – transfer a fraction of mass from i to j,
        – swap two entries,
        – add small Gaussian jitter.
    The best vector ever seen is returned.
    """
    cur = a.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score
    n = a.shape[0]

    start = time.time()
    init_temp = max(cur_score * 0.1, 1e-4)
    min_temp = cur_score * 1e-5

    step_scale = 0.12
    it = 0
    while time.time() < deadline:
        # temperature linearly from init_temp to min_temp
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = init_temp * (1.0 - frac) + min_temp * frac

        cand = cur.copy()
        r = np.random.rand()
        if r < 0.5:                     # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i] * np.random.rand()
                cand[i] -= delta
                cand[j] += delta
        elif r < 0.75:                 # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # Gaussian jitter
            noise = np.random.randn(n) * step_scale * 0.07
            cand += noise
            cand = np.maximum(cand, 0.0)

        # renormalise onto the simplex
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s

        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis rule
            prob = np.exp(-delta / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # occasional progress report (every few seconds)
        if it % 5000 == 0:
            elapsed = time.time() - start
            print(
                f"[SA] iter {it:6d} | time {elapsed:6.1f}s | best {best_score:.6f} | T {T:.3e}"
            )
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Final greedy hill‑climb (deterministic improvement)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill climbing that only keeps strictly improving moves.
    Used as a final polishing step.
    """
    cur = a.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    step_scale = 0.1
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = cur.copy()
        if np.random.rand() < 0.6:        # transfer move
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i] * np.random.rand()
                cand[i] -= delta
                cand[j] += delta
        else:                             # swap move
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        cand = cand / cand.sum()
        cand_score = compute_score(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                # grow step size a little when we succeed
                step_scale = min(step_scale * 1.05, 0.5)
        else:
            # shrink step size on failure
            step_scale *= 0.97

        steps += 1
        if steps % 50000 == 0:
            print(f"[HC] steps {steps} | best {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0 – optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        # height_sequence_1 may be defined globally by a previous run
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1 – decide length (larger n gives more freedom but costs a bit more)
    # ------------------------------------------------------------------
    if total_time > 850.0:
        n = 4096
    else:
        n = 2048
    print(f"Using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2 – generate a diverse pool of seeds and keep the best one
    # ------------------------------------------------------------------
    seed_vectors = []

    # deterministic shapes
    seed_vectors.append(generate_deterministic(n))
    for p in np.linspace(-3.0, 3.0, 13):
        seed_vectors.append(generate_beta_family(n, p))

    # beta pdf families (α,β) – include many extreme values
    alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    betas = alphas
    for a in alphas:
        for b in betas:
            seed_vectors.append(generate_beta_pdf(n, a, b))

    # random Dirichlet draws with different concentrations
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seed_vectors.append(random_dirichlet(n, conc=conc))

    # sparse spikes – often a surprisingly good start
    for k in (30, 60, 120):
        seed_vectors.append(random_spike(n, k=k))

    # include the fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seed_vectors.append(np.array(best_seq, dtype=float))

    # evaluate all seeds (exact objective) and remember the best
    for s in seed_vectors:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] new best => {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – start optimisation from the best seed
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)
    print(f"Starting score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4 – allocate time to the different phases
    # ------------------------------------------------------------------
    now = time.time()
    phase_lp_deadline = now + total_time * 0.15   # LP refinement ~15%
    phase_sa_deadline = deadline - 5.0            # leave a few seconds for final polishing

    # ------------------------------------------------------------------
    # Phase 1 – a handful of LP refinement steps (no gradients needed)
    # ------------------------------------------------------------------
    print("\n[Phase 1] LP‑based refinement")
    lp_iters = 0
    while time.time() < phase_lp_deadline and lp_iters < 12:
        cur_a = lp_refine(cur_a, mix=0.3)
        cur_score = compute_score(cur_a)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"[LP] iteration {lp_iters} → score {best_val:.6f}")
        lp_iters += 1

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < phase_sa_deadline:
        print("\n[Phase 2] simulated‑annealing polishing")
        cur_a, cur_score = simulated_annealing(
            cur_a, cur_score, phase_sa_deadline
        )
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"[SA] best after SA = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (short, deterministic)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 3] final hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=500_000)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"[HC] best after hill‑climb = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – official evaluation & return
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
