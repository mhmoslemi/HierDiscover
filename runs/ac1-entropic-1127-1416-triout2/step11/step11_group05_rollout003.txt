"""
Improved step‑function search for the “max‑convolution” objective.

Algorithm outline
-----------------
1. **Problem size** – Choose a length `n` (2048 / 4096 / 8192) according to
   the remaining wall‑clock budget.
2. **Analytic seeding** – Scan three low‑dimensional families
   (symmetric product, left‑heavy power, exponential U‑shape) on a fine grid
   and also generate a few random Dirichlet draws.  Keep the best few
   candidates.
3. **Smooth‑max L‑BFGS** – Starting from each seed,
   minimise a log‑sum‑exp surrogate of the true max using the exact gradient
   (derived analytically).  A schedule of increasingly large smoothing
   parameters `β` (up to 2·10⁴) drives the solution towards the true
   objective.  No artificial symmetry is enforced – the full search space
   is explored.
4. **Final polishing** – A short simulated‑annealing hill‑climb with
   swap, mass‑transfer and occasional global perturbations refines the
   best sequence found.
5. **Return** – The best sequence (as a plain Python list) is returned.
   All steps respect the global deadline (default 1000 s).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_exact_score(a: np.ndarray) -> float:
    """
    Exact objective for a distribution `a` with Σ a = 1:
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (cheap seeds)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, then normalised."""
    i = np.arange(n, dtype=np.float64)
    # Avoid log(0) by shifting by 0.5
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against infinities
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad_z) where:

        f   = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad_z = ∂f/∂z   (gradient w.r.t. soft‑max parameters)

    The gradient is derived analytically, avoiding finite‑difference noise.
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # stable soft‑max over convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta                 # smooth approximation of max
    w = expw / sumexp                                # weighting over convolution positions

    # gradient of weighted sum of convolutions w.r.t. a
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]        # shape (n,)

    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS driver (smooth max)
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS runs on the smooth‑max surrogate.
    `betas` is an ordered list of smoothing parameters (increasing).
    The function respects the global deadline and returns the best a found
    (still on the simplex, no enforced symmetry).
    """
    n = a_start.shape[0]
    best_a = a_start.copy()
    best_score = compute_exact_score(best_a)

    a = a_start.copy()
    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS with β={beta}")

        # Initialise L‑BFGS on the log‑softmax parameters
        z0 = np.log(a + 1e-14)

        # Objective wrapper returning (value, gradient)
        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as e:
            print(f"  L‑BFGS failure (β={beta}): {e}")
            continue

        a = softmax(res.x)                # back to the simplex
        # tiny re‑normalisation to avoid drift
        a = a / a.sum()
        cur_score = compute_exact_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex.
    Moves:
      * swap two entries
      * transfer a fraction of mass from i → j
      * occasional global perturbation (replace a small fraction by a Dirichlet draw)
    Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_exact_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # start temperature roughly 0.1 % of the current score
    T0 = best_score * 1e-3
    step_scale = 0.05   # typical fraction moved in a mass‑transfer move

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # ----- propose a candidate -----
        cand = cur_a.copy()
        r = np.random.rand()
        if r < 0.4:                                  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.85:                               # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        else:                                        # global perturbation
            # replace a random 5 % subset by a fresh Dirichlet draw
            mask = np.random.rand(n) < 0.05
            if np.any(mask):
                # draw fresh probabilities for the masked entries
                fresh = np.random.rand(mask.sum())
                fresh = fresh / fresh.sum()
                # preserve total mass
                total_mask = cand[mask].sum()
                cand[mask] = fresh * total_mask

        # re‑project onto the simplex
        cand = project_to_simplex(cand)
        cand_score = compute_exact_score(cand)

        delta = cand_score - cur_score
        # accept if improvement or with Metropolis probability
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase step size when we improve
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # slowly contract step size on rejection
            step_scale *= 0.97

    return best_a, best_score

            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose problem size (more points ⇒ potentially lower bound)
    # -----------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 700:
        n = 8192
    elif remaining > 350:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] using length n = {n}")

    # -----------------------------------------------------------
    # 2. Analytic families + random Dirichlet seeds
    # -----------------------------------------------------------
    candidate_pool = []

    # (a) symmetric product family
    for alpha in np.linspace(0.0, 5.0, 26):
        if time.time() >= deadline - 120.0:
            break
        vec = symmetric_product_family(n, alpha)
        candidate_pool.append(vec)

    # (b) left‑heavy power family
    for beta in np.linspace(0.0, 5.0, 26):
        if time.time() >= deadline - 120.0:
            break
        vec = left_heavy_power_family(n, beta)
        candidate_pool.append(vec)

    # (c) exponential U‑shape family
    for gamma in np.linspace(0.0, 8.0, 33):
        if time.time() >= deadline - 120.0:
            break
        vec = exp_u_family(n, gamma)
        candidate_pool.append(vec)

    # (d) a few random Dirichlet draws (concentration 0.7 gives slightly
    #     heavier tails than uniform)
    for _ in range(12):
        if time.time() >= deadline - 120.0:
            break
        raw = np.random.gamma(0.7, 1.0, size=n)
        vec = raw / raw.sum()
        candidate_pool.append(vec)

    # -----------------------------------------------------------
    # 3. Keep the most promising seeds (exact score)
    # -----------------------------------------------------------
    scored = []
    for vec in candidate_pool:
        sc = compute_exact_score(vec)
        scored.append((sc, vec))
    scored.sort(key=lambda x: x[0])
    top_k = 5
    seeds = [vec for _, vec in scored[:top_k]]
    print(f"[{time.time():.1f}] selected {len(seeds)} seeds for L‑BFGS")

    # -----------------------------------------------------------
    # 4. L‑BFGS optimisation with a long β schedule
    # -----------------------------------------------------------
    # schedule grows roughly logarithmically; later β are very sharp.
    beta_schedule = [
        1, 5, 10, 20, 50, 100, 200,
        500, 1000, 2000, 5000, 10000, 20000
    ]
    # allocate a few hundred iterations per β (still cheap)
    max_iter_stage = 250

    for idx, a_start in enumerate(seeds, 1):
        if time.time() >= deadline - 150.0:
            break
        print(f"[{time.time():.1f}] L‑BFGS run #{idx}/{len(seeds)}")
        a_opt, sc_opt = l_bfgs_optimize(a_start,
                                        betas=beta_schedule,
                                        max_iter_per_stage=max_iter_stage,
                                        deadline=deadline - 100.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}]   new global best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Final simulated‑annealing polishing
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=2_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}]   SA improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Fallback (should never be needed)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # Normalise just in case (scaling does not affect the objective)
    total = sum(best_seq)
    if total > 0:
        best_seq = [x / total for x in best_seq]

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
