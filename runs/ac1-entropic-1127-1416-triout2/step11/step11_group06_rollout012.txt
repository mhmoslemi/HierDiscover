def smooth_descent(start_a, time_deadline, gamma_start=0.5, gamma_end=8, phases=4):

    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    total_phases = phases
    gamma = gamma_start
    for phase in range(phases):
        # compute time left
        remaining = time_deadline - time.time()
        if remaining < 5: break
        # allocate fraction of remaining time based on remaining phases, e.g., each phase gets equal share.
        phase_deadline = time.time() + remaining / (total_phases - phase)
        # steps for this phase
        iter = 0
        while time.time() < phase_deadline:
            # compute conv
            conv = convolution_fft(a, a)
            # compute softmax weights
            # scaling gamma
            scale = gamma
            # avoid overflow
            max_val = np.max(conv * scale)
            exp_arg = conv * scale - max_val
            w = np.exp(exp_arg)
            w_sum = w.sum()
            w = w / w_sum
            # compute gradient
            # compute cross_corr = convolution_fft(w, a[::-1])
            cross = convolution_fft(w, a[::-1])
            # slice out n entries starting at n-1
            g = 2.0 * cross[(n-1):(n-1+n)]
            # subtract average to preserve simplex invariance (mirror)
            avg = np.dot(a, g)
            # step size schedule: eta = eta0 / sqrt(iter+1)
            eta = 0.5 / np.sqrt(iter + 1)
            # exponentiated gradient update
            exp_term = -eta * (g - avg)
            exp_term = np.clip(exp_term, -50, 50)
            a = a * np.exp(exp_term)
            # renormalise
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum
            # enforce symmetry
            a = (a + a[::-1]) / 2.0
            # evaluate score every 1000 iters
            if iter % 1000 == 0:
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
            iter += 1
        # after phase, maybe increase gamma
        gamma = min(gamma * 2, gamma_end)
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithmic outline
-------------------
1. **Generate a diverse pool of seeds** – uniform, central‑binomial,
   power‑law, geometric, symmetric Beta‑shapes, a sinusoidal shape and a few
   random Dirichlet draws.  The best among them (according to the exact
   objective) becomes the initial working point.

2. **Alternating projection onto two simple sets**  
   *Set A*: sequences whose (zero‑padded) discrete Fourier transform has a
   *constant* magnitude – this makes the auto‑convolution as flat as possible.  
   *Set B*: the probability simplex (non‑negative, sum = 1).  
   Starting from the seed we repeatedly (a) enforce constant magnitude in the
   frequency domain, (b) truncate, clip to non‑negativity and project onto the
   simplex, (c) symmetrise the vector.  The loop runs until the time budget
   allocated to this phase runs out, keeping the best sequence seen.

3. **Smoothed mirror descent** – minimise a smooth approximation of
   ``max(conv(a,a))`` using a log‑sum‑exp surrogate.  The surrogate weight
   ``γ`` is increased geometrically (0.5 → 8) so the optimisation first moves
   quickly on a smooth landscape and then refines the true max‑objective.
   Updates are performed with an exponentiated‑gradient (KL‑mirror) step,
   followed by projection onto the simplex and explicit symmetrisation.

4. **Final stochastic hill‑climb** – tiny local moves (swap two entries or
   transfer a fraction of mass) with occasional step‑size adaptation.  The
   best sequence found by the previous phases is polished here.

All phases respect the global deadline (default 1000 s) and keep the best
sequence encountered.  The returned list is ready for ``evaluate_sequence``.
"""

import time

# Basic linear‑algebra utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Analytic families for initial candidates

def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Symmetric and heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


def generate_cosine_shape(n: int) -> np.ndarray:
    """
    Positive half‑sine shape:
        a_i ∝ sin(π·(i+0.5)/n)
    Symmetric and zero‑free.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    return project_to_simplex(a)


def generate_dirichlet(alpha: float, n: int) -> np.ndarray:
    """Random Dirichlet‑type draw (Gamma with shape α)."""
    x = np.random.gamma(shape=alpha, scale=1.0, size=n)
    return project_to_simplex(x)

def alternating_projection_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Alternating projection between:
      (A) constant magnitude of the zero‑padded FFT (flattens the autocorrelation)
      (B) the probability simplex (non‑negative, sum = 1).
    Returns the best sequence seen and its exact objective.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    L = next_pow2(2 * n - 1)          # FFT length for linear convolution
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < time_deadline and it < max_iter:
        # ----- (A) impose constant magnitude in Fourier domain -----
        a_pad = np.zeros(L, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)                # length L//2+1, complex
        energy = np.sum(a ** 2)                # L2‑norm², stays invariant under FFT
        target_mag = np.sqrt(energy)           # constant magnitude to preserve energy
        A_new = target_mag * np.exp(1j * np.angle(A))

        # ----- (B) back to time domain, truncate, enforce feasibility -----
        a_full = np.fft.irfft(A_new, n=L)      # real output, length L
        a = a_full[:n]                         # keep the first n entries
        a = np.maximum(a, 0.0)                 # non‑negativity
        a = project_to_simplex(a)               # simplex constraint

        # optional: enforce symmetry (helps the convolution structure)
        a = (a + a[::-1]) / 2.0

        # periodic progress check
        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        it += 1

    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score

def smooth_mirror_descent_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    gamma_start: float = 0.5,
    gamma_end: float = 8.0,
    phases: int = 6,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) descent on a log‑sum‑exp
    surrogate of max(conv(a,a)).  The surrogate temperature γ is increased
    geometrically from ``gamma_start`` to ``gamma_end`` to gradually tighten
    the approximation.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # geometric schedule for γ
    gammas = np.geomspace(gamma_start, gamma_end, num=phases)
    for phase, gamma in enumerate(gammas):
        # allocate a fair share of the remaining budget to this phase
        remaining = time_deadline - time.time()
        if remaining <= 0:
            break
        phase_time = remaining / (len(gammas) - phase)
        phase_deadline = time.time() + phase_time

        it = 0
        while time.time() < phase_deadline:
            # --- 1) current convolution ---
            conv = convolution_fft(a, a)                # length 2n‑1

            # --- 2) soft‑max weights ---
            # w_k ∝ exp(γ·conv_k)
            scaled = conv * gamma
            max_scaled = np.max(scaled)                 # for numerical stability
            exp_vals = np.exp(scaled - max_scaled)
            w = exp_vals / np.sum(exp_vals)             # shape (2n‑1,)

            # --- 3) gradient of the surrogate ---
            # g = ∂f/∂a = 2 * cross‑correlation(w , a)
            cross = convolution_fft(w, a[::-1])         # length 3n‑2
            g = 2.0 * cross[(n - 1):(n - 1 + n)]        # slice n entries

            # --- 4) KL‑mirror step (exponentiated gradient) ---
            avg = np.dot(a, g)
            it += 1
            eta = 0.6 / np.sqrt(it)                    # diminishing stepsize
            exp_term = -eta * (g - avg)
            exp_term = np.clip(exp_term, -50.0, 50.0)   # keep numbers sane
            a = a * np.exp(exp_term)

            # --- 5) projection onto simplex & symmetrisation ---
            s = np.sum(a)
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s
            a = (a + a[::-1]) / 2.0                     # enforce symmetry

            # --- 6) occasional exact evaluation ---
            if it % 800 == 0:
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

    # final exact evaluation
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score

def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07  # initial fraction of mass to move each step

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i → j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score

    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        print(f"Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        print(f"No external sequence provided – using length {n}.")

    # ------------------------------------------------------------------
    # 0. Build a rich collection of starter candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # classic central binomial
    init_candidates.append(generate_central_binomial(n))

    # power‑law family
    for p in np.linspace(0.30, 1.00, 9):
        init_candidates.append(generate_powerlaw(n, float(p)))

    # geometric family (r close to 1)
    for r in [0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:
        init_candidates.append(generate_geometric(n, float(r)))

    # symmetric Beta‑shapes (α from 0.1 to 0.55)
    for alpha in np.linspace(0.10, 0.55, 10):
        init_candidates.append(generate_beta_shape(n, float(alpha)))

    # sinusoidal shape (half‑sine)
    init_candidates.append(generate_cosine_shape(n))

    # a few random Dirichlet draws – they often discover unexpected niches
    for a in [0.15, 0.30, 0.60, 1.0]:
        init_candidates.append(generate_dirichlet(alpha=a, n=n))

    # possibly include the previously found best construction
    try:
        init_candidates.append(np.array(height_sequence_1, dtype=np.float64))
        print("Added provided height_sequence_1 as a starter.")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose the best starter by exact evaluation
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Best starter score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Alternating projection phase (constant‑magnitude DFT)
    # ------------------------------------------------------------------
    proj_deadline = deadline - 200.0            # keep time for later phases
    if time.time() < proj_deadline:
        print("Starting alternating projection phase …")
        a_proj, score_proj = alternating_projection_optimize(
            best_seq,
            time_deadline=proj_deadline,
        )
        if score_proj < best_val:
            best_val, best_seq = score_proj, a_proj.copy()
            print(f"Alternating projection improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed mirror‑descent phase
    # ------------------------------------------------------------------
    mirror_deadline = deadline - 60.0
    if time.time() < mirror_deadline:
        print("Starting smoothed mirror‑descent phase …")
        a_mirror, score_mirror = smooth_mirror_descent_optimize(
            best_seq,
            time_deadline=mirror_deadline,
        )
        if score_mirror < best_val:
            best_val, best_seq = score_mirror, a_mirror.copy()
            print(f"Smoothed mirror‑descent improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Random‑restart bursts (cheap, keep diversity)
    # ------------------------------------------------------------------
    restart_deadline = deadline - 30.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(3):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.012 * np.random.randn(n)
            pert = project_to_simplex(pert)
            sub_dead = time.time() + 12.0
            a_tmp, s_tmp = smooth_mirror_descent_optimize(
                pert,
                time_deadline=sub_dead,
                gamma_start=0.4,
                gamma_end=6.0,
                phases=4,
            )
            if s_tmp < best_val:
                best_val, best_seq = s_tmp, a_tmp.copy()
                print(f"Random restart {r} yielded improvement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        print("Final hill‑climb polishing …")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=250_000,
        )
        if hill_score < best_val:
            best_val, best_seq = hill_score, np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard – fall back to a uniform sequence if nothing found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 7. Final reporting using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq.tolist()
