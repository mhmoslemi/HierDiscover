"""
Non‑negative step‑function optimiser.

The optimiser works in three stages:

1. **Broad global sweep** – explores three parametric families
   (power‑law, exponential, beta‑shaped) and many random Dirichlet draws.
   The best few candidates (according to the exact objective
   2·n·max(conv(a,a))) are retained.

2. **Smooth‐max gradient descent** – replaces the non‑differentiable
   max by a log‑sum‑exp (smooth‑max) with a continuation on the
   temperature β.  The gradient of the smooth‑max is computed via FFT,
   then a projected (Euclidean) gradient step on the simplex is taken.
   The schedule starts with a gentle β and gradually sharpens it,
   effectively mimicking the sub‑gradient method while giving far
   better convergence.

3. **Stochastic annealing polishing** – a final Metropolis walk with
   decreasing temperature transfers small masses between coordinates,
   always projecting back onto the simplex.

The whole procedure respects the 1000 s wall‑clock budget and returns
the best sequence found (as a plain Python list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the global baseline)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global search (baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Explore several families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.3, 1.2, 151):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(0.00005, 0.03, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep (coarse grid)
    al_vals = np.linspace(0.2, 5.0, 23)
    be_vals = np.linspace(0.2, 5.0, 23)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Random Dirichlet draws – several concentrations
    concs = [0.2, 0.5, 1.0, 2.0, 5.0]
    samples_per = 120
    for conc in concs:
        for _ in range(samples_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def smooth_descent(a0: np.ndarray, deadline: float,
                   beta_schedule, step0: float = 0.2) -> np.ndarray:
    """Continuation‑based gradient descent on a smooth approximation of
    the max‑convolution.  Uses FFT‑based convolution for an O(n log n)
    gradient.  Returns the best vector seen before the deadline."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    beta_idx = 0
    beta = beta_schedule[beta_idx]
    # how many iterations we allow before moving to the next beta
    it_per_beta = max(2000, n // 2)

    while time.time() < deadline:
        # ----- possibly increase beta -----
        if it > 0 and it % it_per_beta == 0 and beta_idx + 1 < len(beta_schedule):
            beta_idx += 1
            beta = beta_schedule[beta_idx]

        # ----- smooth max weight -----
        conv = full_convolution(a)          # length 2n‑1
        max_c = conv.max()
        # stable exponentiation
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()       # length 2n‑1, sum to 1

        # ----- gradient of smooth max (via FFT) -----
        rev_a = a[::-1]
        N = 1 << ((rev_a.size + w.size - 2).bit_length())
        A = np.fft.rfft(rev_a, n=N)
        W = np.fft.rfft(w, n=N)
        G = A * W
        conv_rev = np.fft.irfft(G, n=N)[:rev_a.size + w.size - 1]
        term = conv_rev[n - 1:n - 1 + n]      # sliding dot‑product
        grad = 2.0 * term                     # ∇ smooth‑max

        # ----- projected gradient step -----
        step = step0 / np.sqrt(it + 1)
        a = a - step * grad
        a = project_to_simplex(a)

        # ----- keep the best around -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if it % 4000 == 0 and it > 0:
            jitter = np.random.randn(n) * (step * 0.01)
            a = a + jitter
            a = project_to_simplex(a)

        it += 1
        if time.time() >= deadline:
            break

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb with temperature)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            # accept
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        # temperature schedule
        T = max(final_temp, T * cooling_factor)

        # periodic global jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

        for _ in range(3):
            seeds.append(np.random.dirichlet(np.ones(n) * 0.5))
        beta_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]
        for a0 in seeds:
            if time.time() >= smooth_deadline:
                break
            a_refined = smooth_descent(a0, smooth_deadline, beta_sched, step0=0.25)
            val = compute_score(a_refined)
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()
                print(f"[smooth] improved score = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Simulated annealing for final polishing
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()
            print(f"[anneal] final improvement = {val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fallback to uniform if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback uniform] score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
