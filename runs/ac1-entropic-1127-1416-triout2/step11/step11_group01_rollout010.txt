def fast_sa_optimize(start_a, deadline, max_steps=2_000_000):
    n = len(start_a)
    a = start_a.copy()
    # Ensure sum 1.
    a = a / a.sum()
    # Compute initial convolution via FFT (or np.convolve)
    conv = convolution_fft(a, a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best_a = a.copy()
    best_score = cur_score

    T0 = cur_score * 1e-4
    T = T0
    step_scale = 0.1   # fraction of a[i] to move

    for step in range(1, max_steps+1):
        if time.time() >= deadline:
            break
        # Temperature schedule linear decay to 0 by max_steps
        T = T0 * (1.0 - step / max_steps)
        # Choose random i,j
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1  # ensure j != i
        # Amount to move
        if a[i] <= 0:
            continue
        delta = step_scale * a[i]
        # Optionally maybe also randomize delta (uniform)
        # ensure delta not exceed a[i]
        # Could also randomise direction (increase and decrease).
        # Compute new a_i, a_j
        ai_old = a[i]
        aj_old = a[j]
        delta = min(delta, ai_old)  # ensure nonnegative
        # Save old conv entries
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j

        old2i = conv[idx2i]
        old2j = conv[idx2j]
        oldij = conv[idxij]

        # Compute new conv entries
        new2i = old2i - 2.0 * delta * ai_old + delta * delta
        new2j = old2j + 2.0 * delta * aj_old + delta * delta
        newij = oldij + 2.0 * delta * (ai_old - aj_old) - 2.0 * delta * delta

        # Apply updates
        conv[idx2i] = new2i
        conv[idx2j] = new2j
        conv[idxij] = newij

        # Update a
        a[i] = ai_old - delta
        a[j] = aj_old + delta

        # compute new max
        new_max = conv.max()
        new_score = 2.0 * n * new_max

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability
            if T > 0 and np.random.rand() < np.exp(-(new_score - cur_score) / (T + 1e-18)):
                accept = True

        if accept:
            cur_score = new_score
            cur_max = new_max
            # adapt step_scale slowly
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            # revert
            conv[idx2i] = old2i
            conv[idx2j] = old2j
            conv[idxij] = oldij
            a[i] = ai_old
            a[j] = aj_old
            # degrade step_scale
            step_scale *= 0.99
    return best_a, best_score

"""
Search for a non‑negative step‑function minimizing

        2·n·max(conv(a,a)) / (sum(a)²)

The program follows several stages, each respecting the global
wall‑clock budget (default 1000 s):

0.  Load a fallback candidate if the harness provides one.
1.  Scan a variety of analytic families (product‑shape,
    exponential U‑shape, mixed uniform/U) over a wide range of
    parameters together with many random starts.
2.  Refine the best vector with exponentiated‑gradient (mirror‑descent)
    on the true max‑convolution objective.
3.  Further refine with Adam on a smooth surrogate
    f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k) using a large final β
    (β≈3000) to approach the true max.
4.  Final fine‑tuning by a fast simulated‑annealing hill‑climb that
    updates the full convolution incrementally – only three
    convolution entries change for each elementary mass‑transfer move.
    This makes it possible to perform millions of cheap moves.

All intermediate vectors are projected onto the probability simplex
(∑a_i = 1, a_i ≥ 0); the objective is invariant under scaling, so normalising
to a simplex does not affect the final score.
The routine returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # kept for compatibility; not used actively


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}
    Symmetric, heavy at the two ends for α>0.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """
    w_i ∝ exp(-γ·dist(i)),
    where dist(i)=min(i,n−1−i) is the distance to the nearest end.
    Heavy at the ends for γ>0.
    """
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Convex combination of the uniform distribution and the
    exponential‑U family.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """
    Random decreasing vector (monotone) generated by sorting a beta‑like draw.
    """
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 30.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Mirror descent (exponentiated‑gradient) on the true max
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]          # diagonal contributes only once

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)

        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 4.0,
                         beta_end: float = 3000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimises the smooth surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The final β is large (≈3000) to approximate the true max.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable softmax over β·conv entries
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a : 2 * (a * w) (convolution with reversed w)
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[len(a) - 1:len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Fast simulated‑annealing hill‑climb (incremental convolution update)
# ----------------------------------------------------------------------


def fast_sa_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Plain SA with mass‑transfer moves.
    Only three entries of the full convolution change per move, which
    makes each iteration O(1) plus an O(n) max‑scan.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()  # ensure simplex

    conv = convolution_fft(a, a)          # full convolution, length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best_a = a.copy()
    best_score = cur_score

    # temperature schedule – start small, decay to 0
    T0 = cur_score * 1e-4
    step_scale = 0.12
    recompute_interval = max(1, n // 8)   # periodic full recompute to fight drift

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # linear temperature decay
        T = T0 * (1.0 - step / max_steps)

        # pick distinct indices i, j
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        # amount to transfer from i towards j
        delta = step_scale * a[i]
        delta = min(delta, a[i])  # keep non‑negative

        # store old values for revert
        ai_old = a[i]
        aj_old = a[j]

        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j

        old2i = conv[idx2i]
        old2j = conv[idx2j]
        oldij = conv[idxij]

        # exact update formulas (derived from (a+Δ)*(a+Δ))
        new2i = old2i - 2.0 * delta * ai_old + delta * delta
        new2j = old2j + 2.0 * delta * aj_old + delta * delta
        newij = oldij + 2.0 * delta * (ai_old - aj_old) - 2.0 * delta * delta

        # apply
        conv[idx2i] = new2i
        conv[idx2j] = new2j
        conv[idxij] = newij
        a[i] = ai_old - delta
        a[j] = aj_old + delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            if T > 0.0 and np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-18)):
                accept = True

        if accept:
            cur_score = new_score
            cur_max = new_max
            # adapt step size slowly
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            # revert changes
            conv[idx2i] = old2i
            conv[idx2j] = old2j
            conv[idxij] = oldij
            a[i] = ai_old
            a[j] = aj_old
            step_scale = max(step_scale * 0.98, 1e-4)

        # periodic full recompute to avoid drift from rounding errors
        if step % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

    return best_a, best_score
