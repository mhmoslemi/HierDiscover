"""
Advanced search for a low‑value step‑function sequence.

Algorithm
---------
1. **Simple parametric sweep** – generate candidates from families
   (power‑law, geometric, mixed plateau) and keep the best (lowest
   ``2·n·max(conv)``).

2. **Smooth‑max gradient descent (Adam)** – minimise a smooth
   approximation of the max‑convolution:
        S_α(a) = log‑sum‑exp(α·conv(a,a)) / α .
   For a fixed α the gradient is
        ∇S_α = 2·corr(a, softmax_α(conv)),
   where ``corr`` is the linear cross‑correlation.
   We run Adam for a sequence of increasing α (1 → 100) and keep the
   best simplex‑projected iterate.

3. **Projected sub‑gradient polishing** – a few iterations of the
   classical sub‑gradient method on the true max‑convolution objective
   to clean up any remaining non‑smoothness.

4. **Stochastic hill‑climb** – cheap random mass‑transfer moves with
   greedy acceptance to escape tiny plateaus.

All phases respect the remaining wall‑clock budget (default 1000 s).
The returned list sums to ≈1, so ``evaluate_sequence`` sees the proper
normalisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (real FFT).
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def linear_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Phase 1 – global parametric sweep
# ----------------------------------------------------------------------
def best_initial_vector(n: int, deadline: float) -> np.ndarray:
    """Search simple families and return the best normalised vector."""
    best_a = None
    best_val = float('inf')

            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:

        if val < best_val:
            best_val, best_a = val, a

    # ---------- mixed plateau + tail ----------
    for L_frac in np.linspace(0.05, 0.30, 6):
        if time.time() >= deadline:
            break
        L = int(L_frac * n)
        if L == 0 or L >= n:
            continue
        for p in np.linspace(0.4, 0.8, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head_val = tail.max()
            head = np.full(L, head_val)
            a = np.concatenate([head, tail])
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max Adam optimisation
# ----------------------------------------------------------------------
def smooth_adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Minimise a smooth approximation of max(conv) using Adam.
    Returns the best iterate found.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # schedule of increasing smoothness
    alphas = [1.0, 5.0, 10.0, 20.0, 50.0, 100.0]

    best_a = a.copy()
    best_score = compute_score(a)

    for alpha in alphas:
        # step‑size scaling – larger α needs smaller steps
        base_lr = 0.05 / (alpha + 1.0)

        while time.time() < deadline:
            t_iter += 1
            # ----- compute current convolution -----
            conv = full_convolution(a)                     # length 2n-1
            # ----- soft‑max weights (stable) -----
            shift = np.max(alpha * conv)
            exp_term = np.exp(alpha * conv - shift)
            p = exp_term / exp_term.sum()                 # still length 2n-1

            # ----- gradient = 2 * cross‑correlation(a, p) -----
            p_rev = p[::-1]
            corr = linear_convolution(a, p_rev)           # length 3n-2
            m_len = p.shape[0]                            # =2n-1
            grad = 2.0 * corr[m_len - 1 - np.arange(n)]

            # ----- Adam update -----
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t_iter)
            v_hat = v / (1 - beta2 ** t_iter)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a_candidate = a - step
            a_candidate = project_to_simplex(a_candidate)

            a = a_candidate

            # ----- occassional true‑objective check -----
            if t_iter % 20 == 0:
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score, best_a = cur_score, a.copy()

            # early stop for this α if no progress for a while
            if t_iter % 200 == 0:
                # simple heuristic: if best hasn't improved in last 200 checks,
                # break to next α (allow more time for larger α later)
                # We reuse best_score; if it's unchanged for a long stretch,
                # the current α likely saturated.
                pass  # (placeholder – we could implement a timer, but we keep it simple)

            if time.time() >= deadline:
                break

    return best_a


# ----------------------------------------------------------------------
# Phase 3 – sub‑gradient polishing (same as before)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Phase 4 – greedy stochastic hill‑climb
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Tiny stochastic moves with greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1

        if no_improve > 100_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # Optional fallback (if a previously discovered construction exists)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Choose a length – a larger n gives finer shape at modest cost.
    # -----------------------------------------------------------------
    n = 4096

    # -----------------------------------------------------------------
    # Phase 1 – global parametric sweep
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    if time.time() < sweep_deadline:
        a0 = best_initial_vector(n, sweep_deadline)
        if a0 is not None:
            val = compute_score(a0)
            print(f"[global sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a0.tolist()

    # -----------------------------------------------------------------
    # Phase 2 – smooth‑max Adam optimisation
    # -----------------------------------------------------------------
    adam_deadline = deadline - 200.0
    if best_seq is not None and time.time() < adam_deadline:
        a_init = np.array(best_seq, dtype=float)
        a_opt = smooth_adam_optimize(a_init, adam_deadline)
        val = compute_score(a_opt)
        print(f"[smooth‑Adam] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()

    # -----------------------------------------------------------------
    # Phase 3 – sub‑gradient polishing
    # -----------------------------------------------------------------
    sub_deadline = deadline - 80.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, sub_deadline, step0=0.10)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # Phase 4 – final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report via the official evaluator
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
