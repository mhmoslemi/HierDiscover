"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

This version expands the global exploration, adds an exponentiated‑gradient
polishing stage (a mirror‑descent on the simplex) and runs a longer,
more aggressive simulated‑annealing finish.  The search is timed to stay
well within the 1000 s wall‑clock limit and continually prints progress
and the best objective value seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}  (discrete Beta‑shaped density)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_mixed_uniform(n: int, ratio: float) -> np.ndarray:
    """
    Two‑block uniform distribution.
    ``ratio`` seconds of mass is placed uniformly on the first block,
    the remainder on the second block.
    """
    if not (0.0 < ratio < 1.0):
        ratio = 0.5
    split = int(n * ratio)
    if split == 0 or split == n:
        return np.full(n, 1.0 / n)
    a = np.concatenate(
        [np.full(split, 1.0 / split * ratio), np.full(n - split, 1.0 / (n - split) * (1 - ratio))]
    )
    return a / a.sum()


# ----------------------------------------------------------------------
# Global search (broader)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 15):
    """
    Explore a variety of families and many random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays).
    """
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep (much broader range)
    for p in np.linspace(0.05, 2.5, 201):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep (wide range)
    for lam in np.linspace(0.0002, 0.4, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep (coarse random grid)
    al_vals = np.linspace(0.2, 5.0, 13)
    be_vals = np.linspace(0.2, 5.0, 13)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Two‑block uniform mixes
    for ratio in np.linspace(0.1, 0.9, 9):
        if time.time() >= deadline:
            break
        try_candidate(generate_mixed_uniform(n, ratio))

    # 5) Random Dirichlet draws – many concetrations
    concs = [0.2, 0.5, 1.0, 2.0, 5.0]
    samples_per = 200   # ~1000 draws total
    for conc in concs:
        for _ in range(samples_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (mirror descent on simplex)
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(a0: np.ndarray, deadline: float, eta0: float = 1.0) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    The gradient is taken from all active (nearly‑maximal) convolution
    indices.  Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # gradient for each active index k:
        #   ∂conv[k]/∂a_i = a_i + a_{k-i}   (a_{k-i}=0 if out of range)
        #   we use the sum over all active k.
        len_active = len(active)
        g = a * len_active                     # the a_i term from each active k
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += a[j[mask]]               # the a_{k-i} contribution

        # step size (decreasing)
        step = eta0 / np.sqrt(t)

        # exponentiated‑gradient update
        factor = np.exp(-step * g)
        a = a * factor
        # avoid the all‑zero vector (possible with extreme steps)
        if np.all(a == 0):
            a = np.full_like(a, 1.0 / n)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # tiny jitter every few thousand iterations to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step * 0.002)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb with slower cooling)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-2,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9999,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """
    Metropolis‑style hill‑climb that accepts occasional worsening moves.
    The temperature decays slowly to allow thorough exploration.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            # accept
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            # reject – shrink step scale a bit
            step_scale = max(step_scale * 0.98, step_scale_min)

        # temperature schedule
        T = max(final_temp, T * cooling_factor)

        # occasional global jitter to keep motion alive
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = np.maximum(a, 0.0)
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (broader parameter sweep)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0  # reserve time for polishing & anneal
    if time.time() < baseline_deadline:
        candidates = baseline_search(n, baseline_deadline, max_keep=15)
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Exponentiated‑gradient polishing on the top few candidates
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # take the best few vectors from the baseline pool for polishing
        # (re‑evaluate to sort)
        scores = [(compute_score(v), v) for v in candidates]
        scores.sort(key=lambda x: x[0])
        top_k = min(5, len(scores))
        per_candidate_time = (polish_deadline - time.time()) / top_k
        for idx in range(top_k):
            a0 = scores[idx][1]
            sub_deadline = time.time() + per_candidate_time
            a_refined = exponentiated_gradient_polish(a0, sub_deadline, eta0=1.0)
            val = compute_score(a_refined)
            print(f"[polish #{idx + 1}] refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 3. Simulated annealing / stochastic hill‑climb (final refinement)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline,
                                      init_temp=1e-2,
                                      final_temp=1e-6,
                                      cooling_factor=0.9999,
                                      step_scale_init=0.07,
                                      step_scale_min=1e-5)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
