"""
Propose a non‑negative step‑function that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

i.e. minimise the L∞‑norm of the self‑convolution under the simplex.
The search consists of:

1.  Load a fallback solution if present.
2.  Very fast analytic / random scan over a few parametric families.
3.  Several warm‑starts for a *smooth‑max* optimisation.
    The smooth objective  fβ(a)=maxβ(a*a) is minimised by L‑BFGS‑B
    on the soft‑max parametrisation a=softmax(z).  A large β≈2000 makes
    the surrogate virtually the true max‑objective while keeping the
    gradient well‑behaved.  Multiple restarts are performed and the best
    result is kept.
4.  An aggressive simulated‑annealing hill‑climb that uses swaps and
    mass‑transfer moves, with a linear cooling schedule, to polish the
    best vector found by the previous stage.
5.  Return the best sequence as a plain list of floats.

All stages respect the global time budget (default 1000 s) and keep
track of the best objective value seen so far.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    zm = np.max(z)
    e = np.exp(z - zm)
    return e / e.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def beta_two_param_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i ** (alpha - 1.0)) * ((n + 0.5 - i) ** (beta - 1.0))
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Monotonically decreasing random start."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_dirichlet_start(n: int) -> np.ndarray:
    """Draw from Dirichlet(1,…,1) (i.e. uniform on the simplex)."""
    a = np.random.dirichlet(np.ones(n))
    return a


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (f, grad_a) where

        f  = max_β (a*a) = max(a*a) + (1/β)·log Σ exp(β·(a*a - max))
        grad_a = ∇_a f
    """
    conv = convolution_fft(a, a)
    maxc = conv.max()
    # weights for the smooth‑max (log‑sum‑exp trick)
    exp_w = np.exp(beta * (conv - maxc))
    w = exp_w / exp_w.sum()

    # gradient wrt a (cf. smooth‑max derivation)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)
    n = a.shape[0]
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

    # smooth objective value
    f = maxc + np.log(exp_w.sum()) / beta
    return f, grad_a


def smooth_obj_and_grad_z(z: np.ndarray, beta: float):
    """
    Objective and gradient w.r.t. the soft‑max parameters z.
    a = softmax(z) is automatically normalised.
    """
    a = softmax(z)
    f, grad_a = smooth_max_and_grad(a, beta)
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------


def lbfgs_optimize(start_a: np.ndarray,
                   beta: float,
                   deadline: float,
                   max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max surrogate using L‑BFGS‑B on the soft‑max
    parametrisation.  Returns the best probability vector and its
    exact (non‑smoothed) score.
    """
    eps = 1e-12
    z0 = np.log(start_a + eps)

    def fun(z):
        f, g = smooth_obj_and_grad_z(z, beta)
        return f, g

    options = {"maxiter": max_iter, "ftol": 1e-12}
    res = minimize(fun, z0, method="L-BFGS-B", jac=True, options=options)

    if not res.success:
        # fallback – just return the start vector
        a_opt = start_a.copy()
    else:
        a_opt = softmax(res.x)

    # make sure we are on the simplex (possible tiny numerical drift)
    a_opt = project_to_simplex(a_opt)
    sc = compute_score(a_opt)
    return a_opt, sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex: swaps or a small mass transfer between two
    randomly chosen indices.  Temperature is linearly cooled to zero.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 5e-5
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # linear cooling
        T = T0 * (1.0 - step / max_steps)
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score
