"""
Step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Global parametric sweep** – quickly scans simple families
   (power‑law, geometric, mixed plateau) to obtain a decent
   initial vector.

2. **Differential‑Evolution (global) search** – treats the vector as
   a piece‑wise constant function with a small number of segments
   (≈64).  A black‑box optimiser (SciPy) explores this low‑dimensional
   space much faster than a full‑dimensional search and can discover
   exotic shapes that the sweep cannot.

3. **Projected sub‑gradient polishing** – starting from the best
   candidate so far, repeatedly moves the point in the opposite
   direction of a sub‑gradient of ``max(conv(a,a))`` and projects back
   onto the probability simplex.

4. **Stochastic hill‑climb** – a cheap greedy neighbourhood search
   (mass transfer between two positions) is run repeatedly from the
   current best point, with an adaptive step‑size and occasional
   jitter to escape tiny plateaus.

The routine respects a wall‑clock budget (default 1000 s) and returns
the best sequence it found.  All helper functions are defined at the
top level (no closures) and only standard scientific libraries are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        # Uniform vector is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised vector
    (Σa = 1):
        2·n·max(conv(a,a))
    The score is invariant to uniform scaling of ``a``.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def try_fallback(best_seq, best_val):
    """If a global ``height_sequence_1`` exists, evaluate it and keep it."""
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass
    return best_seq, best_val

# Phase 1 – simple parametric sweep
# ----------------------------------------------------------------------
def best_initial_vector(n: int, time_deadline: float) -> np.ndarray:
    """Search several simple families and return the best (lowest score)."""
    best_a = None
    best_val = float('inf')

    # ----- power‑law sweep -----
    for p in np.linspace(0.30, 1.00, 71):   # step ≈0.01
        if time.time() >= time_deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ----- geometric (exponential) decay -----
    for r in np.linspace(0.50, 0.99, 81):   # step ≈0.006
        if time.time() >= time_deadline:
            break
        a = (r ** np.arange(n, dtype=np.float64))
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ----- mixed (plateau + power‑law tail) -----
    for L_frac in np.linspace(0.05, 0.30, 6):
        if time.time() >= time_deadline:
            break
        L = int(L_frac * n)
        if L == 0 or L >= n:
            continue
        for p in np.linspace(0.4, 0.8, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head_val = tail.max()
            head = np.full(L, head_val)
            a = np.concatenate([head, tail])
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    # ----- a few random Dirichlet samples (cheap) -----
    for _ in range(15):
        if time.time() >= time_deadline:
            break
        a = np.random.rand(n)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – global low‑dimensional search (differential evolution)
# ----------------------------------------------------------------------
def de_search(n: int, time_deadline: float) -> np.ndarray | None:
    """Low‑dimensional optimisation using SciPy's DE (≈K=64 segments)."""
    try:
        from scipy.optimize import differential_evolution
    except Exception:
        # SciPy not available – gracefully skip this phase
        return None

    # Number of constant segments (the higher the more expressive but slower)
    K = min(64, n)
    # Mapping from full indices to segment index (uniform partition)
    seg_idx = np.floor(np.arange(n) * K / n).astype(int)

    bounds = [(0.0, 5.0)] * K

    # Objective for DE – always works on a normalised vector (sum=1)
    def de_obj(params):
        seg = np.asarray(params, dtype=np.float64)
        a = seg[seg_idx]                     # expand to length n
        s = a.sum()
        if s == 0.0:
            return 1e6
        a = a / s
        return compute_score(a)

    # Conservatively limit the number of generations so that we stay well
    # within the overall time budget.
    remaining = time_deadline - time.time()
    if remaining < 5.0:
        return None
    # Rough estimate: each DE evaluation costs ≈1 ms; allocate up to 500 ms.
    max_iter = 200
    # Run the optimiser
    result = differential_evolution(
        de_obj,
        bounds,
        maxiter=max_iter,
        popsize=15,
        seed=12345,
        polish=False,
        disp=False,
        workers=1,
        updating='deferred',
    )
    # Build the final vector from the best parameters found.
    seg_opt = np.asarray(result.x, dtype=np.float64)
    a_opt = seg_opt[seg_idx]
    if a_opt.sum() == 0.0:
        a_opt = np.full(n, 1.0 / n)
    else:
        a_opt = a_opt / a_opt.sum()
    return a_opt


# ----------------------------------------------------------------------
# Phase 3 – sub‑gradient polishing (projected sub‑gradient descent)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.12) -> np.ndarray:
    """
    Projected sub‑gradient descent on ``max(conv(a,a))``.
    The step size decays like 1/√t.
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # Active offsets (within a tiny tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        # Sub‑gradient contribution for every active offset
        for k in active:
            if k == 0:
                g += 2.0 * a          # derivative of sum_i a_i^2
            else:
                # a_i appears in conv[k] with a_{i+k} and in conv[-k] with a_{i-k}
                # Both contributions are equal because conv is symmetric.
                # Hence we add the shifted vectors twice.
                g[:n - k] += 2.0 * a[k:]      # a_i from pair (i, i+k)
                g[k:] += 2.0 * a[:n - k]      # a_i from pair (i-k, i)

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# Phase 4 – greedy stochastic hill‑climb (time‑bounded)
# ----------------------------------------------------------------------
def hill_climb_time_limited(a_start: np.ndarray, deadline: float,
                            max_iter: int = 500_000) -> np.ndarray:
    """
    Simple greedy hill‑climb that moves a small fraction of mass from one
    index to another if it strictly reduces ``max(conv)``.
    The routine stops as soon as the deadline is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        if delta == 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        # long plateaus → inject a tiny random jitter
        if no_improve > 200_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # Phase 0 – optional fallback from a known construction
    best_seq, best_val = try_fallback(best_seq, best_val)

    # -----------------------------------------------------------------
    # Phase 1 – global parametric sweep
    # -----------------------------------------------------------------
    n = 2048  # length of the step function
    sweep_deadline = deadline - 600.0   # reserve ~400 s for later phases
    if time.time() < sweep_deadline:
        a_initial = best_initial_vector(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[global sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # Phase 2 – low‑dimensional global search (DE)
    # -----------------------------------------------------------------
    de_deadline = deadline - 400.0
    if time.time() < de_deadline:

        if seq_de is not None:
            val = compute_score(seq_de)
            print(f"[DE] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq_de.tolist()

    # -----------------------------------------------------------------
    # Phase 3 – sub‑gradient polishing
    # -----------------------------------------------------------------
    sub_deadline = deadline - 150.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, sub_deadline, step0=0.12)
        val = compute_score(a_polished)
