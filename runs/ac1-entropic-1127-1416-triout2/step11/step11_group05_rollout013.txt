"""
Propose a step‑function (non‑negative simplex vector) that minimises
    2·n·max(a∗a) / (sum a)²
using a combination of:
  * Fast analytic families for cheap seeding.
  * Several random Dirichlet restarts.
  * Multi‑stage L‑BFGS on the smooth‑max (log‑sum‑exp) surrogate.
  * Projected sub‑gradient descent on the true max‑convolution.
  * A final simulated‑annealing hill‑climb.

The routine respects a wall‑clock budget (default 1000 s) and always
returns the best sequence found so far.  No symmetry is forced during
optimisation – only the simplex constraint (non‑negativity + unit sum)
is maintained.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities (full aperiodic convolution via real FFT)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (float64)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # fallback to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max utilities (parameterisation that guarantees simplex)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Objective helpers (exact objective under sum‑one normalisation)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (cheap ways to obtain decent seeds)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow for α=0 (logprod may contain -inf)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def sinusoidal_family(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin^{power}(π·(i+½)/n), symmetric."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    w = np.maximum(w, 0.0)
    return w / w.sum()


def parabola_family(n: int, exp: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{exp}, symmetric."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** exp
    w = np.maximum(w, 0.0)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its exact gradient
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad_z) where
        f   = max(conv) + log‑sum‑exp(beta·(conv‑max)) / beta
        grad_z = ∂f/∂z using the soft‑max over convolution positions.
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # stable weights
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta
    w = expw / sumexp                                 # size 2n‑1, sums to 1

    # gradient wrt a
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)             # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]         # shape (n,)

    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# True max‑convolution and sub‑gradient (used for a final descent)
# ----------------------------------------------------------------------


def maxconv_and_grad(a: np.ndarray):
    """
    Returns (max_value, grad_a) for the true max‑convolution.
    The sub‑gradient corresponds to a uniform distribution over all
    indices attaining the maximum.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    maxc = conv.max()
    mask = conv == maxc
    cnt = mask.sum()
    if cnt == 0:
        # numerically should not happen
        w = np.zeros_like(conv)
    else:
        w = np.where(mask, 1.0 / cnt, 0.0)

    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]
    return maxc, grad_a


def subgradient_descent(a_start: np.ndarray,
                        max_iters: int,
                        deadline: float):
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    Operates directly on the probability simplex.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # initial step size – a reasonable heuristic
    step = 0.5

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        maxc, grad_a = maxconv_and_grad(a)

        # gradient step in the ambient space, then project back onto the simplex
        a = a - step * grad_a
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # success → gently increase step size
            step = min(step * 1.05, 1.0)
        else:
            # no improvement → shrink step size
            step *= 0.92

        # occasional aggressive decay to guarantee convergence
        if it % 1000 == 0:
            step *= 0.7

    return best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smooth surrogate
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS on the smooth‑max surrogate.
    `betas` – increasing schedule of smoothing parameters.
    Returns the best sequence seen (still on the simplex).
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        # initialise soft‑max parameters
        z0 = np.log(a + 1e-12)

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": max_iter_per_stage,
                                    "disp": False})
        except Exception as e:
            print(f"    L‑BFGS failed (β={beta}): {e}")
            continue

        a = softmax(res.x)
        a = project_to_simplex(a)          # keep exact sum‑one invariant
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved → {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Final simulated‑annealing hill‑climb (lightweight but useful)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Symmetric random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score
