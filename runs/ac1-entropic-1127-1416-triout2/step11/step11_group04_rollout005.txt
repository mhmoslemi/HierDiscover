"""
propose_candidate
-----------------
Hybrid search for a probability vector (non‑negative, sum=1) that
minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

i.e. 2·n·max(conv) for sum(a)=1.
The method combines

* a dense analytic scan of several parametric families,
* massive random Dirichlet starts,
* many short Adam‑optimisations of the smooth‑max surrogate,
* a cheap sub‑gradient refinement,
* an LP‑based “mix‑with‑better‑direction’’ step,
* a final simulated‑annealing hill‑climb.

All phases respect the global 1000 s wall‑clock budget and keep the
vector on the probability simplex after every modification.
The implementation works for any length ≤ 3000 (LP step uses a (2n‑1)×n matrix).

"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_dirichlet_start(n: int, alpha: float) -> np.ndarray:
    """Dirichlet draw with scalar concentration alpha."""
    return np.random.dirichlet([alpha] * n)


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.014) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate

          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    with Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derived from the smooth‑max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1: len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheap iterations)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for the true max‑objective.
    Step size = step0 / sqrt(t), projection onto the simplex each iteration.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (shift k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement step (with line‑search on mixing weight)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weights: list[float]) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M  (all k),   g ≥ 0.
    Afterwards return the convex mixture f_new = (1‑t)·f + t·g_norm
    where t is chosen from *mix_weights* to minimise the true score.
    """
    n = f.shape[0]
    m_len = 2 * n - 1

    # Build convolution matrix A where A @ g = f * g
    A = np.zeros((m_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ↔ minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(m_len, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f, compute_score(f)

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, compute_score(f)

    g_norm = g / sum_g

    # line‑search for the best mixing weight
    best_f = f
    best_sc = compute_score(f)
    for t in mix_weights:
        cand = (1.0 - t) * f + t * g_norm
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_f = cand
    return best_f, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 4_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    The temperature is linearly cooled to zero.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 5e-5            # small initial temperature
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)          # linear cooling
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The function respects the global time budget and returns a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback (if present)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working length (fallback length or a safe default)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
    else:
        n = 2048                    # default; keep ≤3000 for LP safety
    if n > 3000:
        n = 3000

    # ------------------------------------------------------------------
    # 2. Dense analytic scan (≈25 s)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] analytic scan, n={n}")
    # finer grids for the three families
    alphas = np.linspace(0.1, 5.0, 50)       # symmetric product
    betas = np.linspace(0.1, 5.0, 50)        # left‑heavy power
    gammas = np.linspace(0.0, 6.0, 31)       # exponential u‑shape

    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym α={alpha:.2f}) = {best_score:.6f}")

    for beta in betas:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left β={beta:.2f}) = {best_score:.6f}")

    for gamma in gammas:
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp γ={gamma:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Massive random Dirichlet start (≈30 s)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] random Dirichlet scan")
    rand_deadline = deadline - 30.0
    while time.time() < rand_deadline:
        # pick a concentration parameter uniformly in log‑space
        log_alpha = np.random.uniform(-2.0, 1.5)   # approx 0.01 … 30
        alpha = 10 ** log_alpha
        vec = random_dirichlet_start(n, alpha)
        # occasional symmetrisation (helps the objective)
        if np.random.rand() < 0.5:
            vec = (vec + vec[::-1]) * 0.5
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand α={alpha:.3f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise the current best to a probability vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = project_to_simplex(a)
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4. Repeated short Adam + sub‑gradient cycles (≈ 300 s)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] start multi‑restart Adam/sub‑grad loop")
    while True:
        remaining = deadline - time.time()
        # need at least 15 s for a meaningful pair of phases
        if remaining < 15.0:
            break

        # ---- random (or perturbed) start for this run ----
        if np.random.rand() < 0.3 and best_seq is not None:
            # perturb the current best slightly
            start = np.array(best_seq, dtype=np.float64)
            noise = np.random.normal(scale=0.02, size=n)
            start = start + noise
            start = np.clip(start, 0.0, None)
        else:
            # fresh Dirichlet start
            alpha = 10 ** np.random.uniform(-2.0, 1.0)   # 0.01 … 10
            start = random_dirichlet_start(n, alpha)

        if np.random.rand() < 0.5:
            start = (start + start[::-1]) * 0.5

        start = project_to_simplex(start)

        # ---- Adam optimisation (smooth max) ----
        adam_time = min(20.0, remaining * 0.4)   # ~40 % of remaining budget
        adam_deadline = time.time() + adam_time
        a_adam, sc_adam = adam_smooth_optimize(
            start,
            adam_deadline,
            max_iters=250_000,
            beta_start=4.0,
            beta_end=5000.0,
            lr=0.015,
        )

        # ---- short sub‑gradient polishing ----
        subgrad_time = min(8.0, deadline - time.time())
        sub_deadline = time.time() + subgrad_time
        a_sub, sc_sub = subgrad_optimize(
            a_adam,
            sub_deadline,
            max_iters=600_000,
            step0=0.12,
        )

        # ---- record improvement ----
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑based refinement (optional, cheap)
    # ------------------------------------------------------------------
    if time.time() + 10.0 < deadline:
        print(f"[{time.time():.1f}] LP refinement phase")
        a = np.array(best_seq, dtype=np.float64)
        mix_weights = [0.2, 0.1, 0.05, 0.02, 0.01]
        for it in range(12):
            if time.time() + 5.0 > deadline:
                break
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new, sc = lp_refine_step(a, M, mix_weights)
            if sc < best_score - 1e-9:
                best_score = sc
                a = a_new
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] LP step {it} improved to {best_score:.6f}")
            else:
                mix_weights = [max(w * 0.5, 0.001) for w in mix_weights]

    # ------------------------------------------------------------------
    # 6. Symmetrisation test (often beneficial)
    # ------------------------------------------------------------------
    a = np.array(best_seq, dtype=np.float64)
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_seq = a_sym.tolist()
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final aggressive hill‑climb (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a = np.array(best_seq, dtype=np.float64)
        a, sc = hill_climb_sa(a, deadline, max_steps=7_000_000)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
