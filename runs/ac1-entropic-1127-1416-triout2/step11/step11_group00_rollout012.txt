"""
Non‑negative step‑function optimiser.

Algorithm
---------
1. **Global exploration** – generate candidates from several parametric families
   (power‑law, exponential, Beta, left‑triangular) and random Dirichlet draws.
   Keep the best few (by the exact objective 2·n·max(conv(a,a))).

2. **Smooth‑max Adam optimisation** – minimise a smooth approximation of the
   maximum convolution using Adam with simplex projection.
   The smoothness is controlled by a temperature‑like parameter α that is
   increased during optimisation, turning the smooth objective into the true
   max as the run proceeds.

3. **Final stochastic hill‑climb** – a short simulated‑annealing phase
   starting from the best vector found.

All phases respect the 1000 s wall‑clock budget; the programme prints progress
and finally returns the best coefficient list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees v >= 0 and sum(v) == 1.
    """
    if np.all(v <= 0):
        # zero vector → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n‑1) using a real FFT.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())        # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}  (discrete Beta‑shaped density)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_left_triangular(n: int, width_frac: float) -> np.ndarray:
    """
    Left‑hand linear decay: a_i ∝ max(0, 1 - i / (width_frac·(n‑1))).
    The support ends at floor(width_frac·(n‑1)).
    """
    if width_frac <= 0:
        return np.full(n, 0.0)
    max_idx = int(np.floor(width_frac * (n - 1)))
    idx = np.arange(n, dtype=float)
    a = np.maximum(0.0, 1.0 - idx / (width_frac * (n - 1)))
    a[max_idx + 1 :] = 0.0
    total = a.sum()
    if total == 0:
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global baseline exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """
    Explore several families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays).
    """
    best = []                     # list of (score, vector)

    def consider(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.4, 0.9, 101):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.02, 200):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # Beta sweep (coarse grid)
    al_vals = np.linspace(0.2, 4.0, 19)
    be_vals = np.linspace(0.2, 4.0, 19)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    # left‑triangular sweep
    for width in np.linspace(0.1, 1.0, 19):
        if time.time() >= deadline:
            break
        consider(generate_left_triangular(n, width))

    # Random Dirichlet draws
    concs = [0.2, 0.5, 1.0, 2.0]
    samples_per = 80
    for conc in concs:
        for _ in range(samples_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            consider(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# Smooth‑max Adam optimisation
# ----------------------------------------------------------------------


def smooth_max_optimize(a0: np.ndarray, deadline: float,
                        alpha0: float = 5.0,
                        alpha_max: float = 200.0,
                        alpha_factor: float = 1.005,
                        lr: float = 0.02,
                        beta1: float = 0.9,
                        beta2: float = 0.999,
                        eps: float = 1e-8,
                        max_iters: int = 20000) -> np.ndarray:
    """
    Minimises the true objective 2·n·max(conv) by smoothing the max
    with a soft‑max (temperature α) and using Adam updates.
    The iterate is projected onto the probability simplex after every step.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    best_a = a.copy()
    best_val = compute_score(a)

    # Adam moments
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)

    t = 0
    alpha = alpha0

    # pre‑allocate FFT length for gradient convolution
    grad_len = n + m - 1
    n_fft_grad = 1 << ((grad_len - 1).bit_length())

    while time.time() < deadline and t < max_iters:
        # ---- evaluate current point ----
        conv = full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # ---- smooth‑max weights (soft‑max over conv) ----
        conv_max = conv.max()
        # shift for numeric stability
        exp_term = np.exp(alpha * (conv - conv_max))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp               # shape (m,)

        # ---- gradient of the smooth max (2·sum_k w_k a_{k−i}) ----
        w_rev = w[::-1]                       # reversed order
        A2 = np.fft.rfft(a, n=n_fft_grad)
        W2 = np.fft.rfft(w_rev, n=n_fft_grad)
        conv_rev = np.fft.irfft(A2 * W2, n=n_fft_grad)[:grad_len]
        # slice indices m‑n … m‑1 and reverse
        grad_slice = conv_rev[m - n: m][::-1]
        grad = 2.0 * grad_slice               # shape (n,)

        # ---- Adam update ----
        t += 1
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad ** 2)
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = project_to_simplex(a)

        # ---- temperature schedule ----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (lightweight final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """
    Metropolis hill‑climb that accepts occasional worsening moves.
    Small mass‑transfer steps are proposed and the vector is projected
    onto the simplex after each proposal.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            # accept
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            # reject
            step_scale = max(step_scale * 0.98, step_scale_min)

        # temperature schedule
        T = max(final_temp, T * cooling_factor)

        # occasional global jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1.  Global baseline exploration
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0           # reserve time for refinements
    if time.time() < baseline_deadline:
        candidates = baseline_search(n, baseline_deadline, max_keep=8)
        for vec in candidates:
            # compute *exact* objective (sum is 1 after projection)
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2.  Smooth‑max Adam optimisation from several starts
    # -----------------------------------------------------------------
    # gather a few starting points: best found so far + random Dirichlet seeds
    start_vectors = []
    if best_seq is not None:
        start_vectors.append(np.array(best_seq, dtype=float))
    # add a handful of random Dirichlet starts
    for _ in range(12):
        if time.time() >= deadline:
            break
        conc = np.random.uniform(0.1, 3.0)
        a = np.random.dirichlet([conc] * n)
        start_vectors.append(a)

    # optimise each start, sharing the remaining budget
    while start_vectors and time.time() < deadline:
        a0 = start_vectors.pop(0)
        remain = deadline - time.time()
        # give each optimisation a fair slice of the remaining time
        per_deadline = time.time() + max(20.0, remain / max(1, len(start_vectors) + 1))
        refined = smooth_max_optimize(a0, per_deadline)
        cur_score = compute_score(refined)
        if cur_score < best_val:
            best_val, best_seq = cur_score, refined.tolist()
            print(f"[smooth‑max] improved to {best_val:.6f}")
        else:
            print(f"[smooth‑max] score = {cur_score:.6f}")

    # -----------------------------------------------------------------
    # 3.  Final simulated‑annealing polishing (if time permits)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, deadline - 5.0)
        final_score = compute_score(a_final)
        if final_score < best_val:
            best_val, best_seq = final_score, a_final.tolist()
            print(f"[anneal] improved to {best_val:.6f}")
        else:
            print(f"[anneal] score = {final_score:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
