"""
Advanced step‑function optimiser (new version).

Approach
--------
1. Generate a diverse pool of seed shapes (deterministic, beta‑family,
   beta‑pdf, random Dirichlet, sparse spikes, cosine and arcsine shapes).
2. Pick the best seed (lowest exact objective on the simplex).
3. Optimise the chosen seed through four phases:
   * Phase 1 – Smooth max surrogate minimisation using Adam on the
     log‑sum‑exp (soft‑max) of the auto‑convolution.
   * Phase 2 – Variance‑flattening gradient descent (same idea as the
     reference code, but with a slightly larger step and enforced symmetry).
   * Phase 3 – Multi‑peak sub‑gradient descent: the gradient is the sum of
     sub‑gradients of *all* convolution entries that are (almost) maximal.
   * Phase 4 – Stochastic hill‑climb polishing (swap / transfer moves).
4. Throughout the optimisation the vector is kept on the probability simplex
   (non‑negative, sum = 1) and symmetrised (a == a[::-1]) because the problem
   is mirror‑symmetric.
5. When the time budget (≈1000 s) is exhausted the current best sequence is
   returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid cross‑correlation of length len(a):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m]   for i=0,…,len(a)-1.
    Implemented via one FFT‑convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1))."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: pick k positions and give them random weight."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def generate_cosine(n: int) -> np.ndarray:
    """Cosine‑square shape a_i = cos²(π·i/(2·(n‑1)))."""
    i = np.arange(n, dtype=float)
    a = np.cos(np.pi * i / (2.0 * (n - 1))) ** 2
    a /= a.sum()
    return a


def generate_arcsine(n: int) -> np.ndarray:
    """Arcsine‑like shape a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float) + 0.5
    w = 1.0 / np.sqrt(i * (n - i))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – soft‑max (log‑sum‑exp) minimisation with Adam
# ----------------------------------------------------------------------


def softmax_adam_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    lr: float = 0.01,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Minimise the smooth log‑sum‑exp surrogate using Adam."""
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    it = 0
    start = time.time()
    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # anneal beta linearly in log‑scale for stability
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        max_conv = conv.max()
        shifted = conv - max_conv                   # max → 0
        exp_vals = np.exp(beta * shifted)          # ≤1
        w = exp_vals / exp_vals.sum()              # soft‑max weights

        # gradient of Lβ
        grad = 2.0 * beta * correlation_valid(w, cur)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        nxt = cur - step
        # enforce symmetry, then project onto simplex
        nxt = (nxt + nxt[::-1]) * 0.5
        cur = project_to_simplex(nxt)

        if it % 5000 == 0:
            print(
                f"[soft‑max‑Adam] it {it:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.1f}"
            )
    return best, best_score


# ----------------------------------------------------------------------
# Phase 2 – variance‑flattening descent (same as before, with symmetry)
# ----------------------------------------------------------------------


def variance_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """Minimise variance of auto‑convolution while keeping symmetry."""
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        e = conv - avg
        grad = 4.0 * correlation_valid(e, cur)

        it += 1
        step = step0 / np.sqrt(it)

        nxt = cur - step * grad
        nxt = (nxt + nxt[::-1]) * 0.5
        cur = project_to_simplex(nxt)

        if it % 3000 == 0:
            print(
                f"[var‑GD] it {it:6d} | time {time.time():6.1f}s | best {best_score:.6f}"
            )
    return best, best_score


# ----------------------------------------------------------------------
# Phase 3 – multi‑peak sub‑gradient descent
# ----------------------------------------------------------------------


def multi_max_subgradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.09,
    tol_rel: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent where the sub‑gradient is the sum of the
    gradients of *all* (near‑maximal) convolution entries.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        max_val = conv.max()
        # tolerance to collect a few near‑max entries
        eps = max(tol_rel * max_val, 1e-12)
        idxs = np.where(conv >= max_val - eps)[0]

        # weight vector for the selected indices
        w = np.zeros_like(cur)
        w[idxs] = 1.0

        grad = 2.0 * correlation_valid(w, cur)

        it += 1
        step = step0 / np.sqrt(it)

        nxt = cur - step * grad
        nxt = (nxt + nxt[::-1]) * 0.5
        cur = project_to_simplex(nxt)

        # occasional jitter to escape flat plateaus
        if it % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(it))
            cur = cur + jitter
            cur = (cur + cur[::-1]) * 0.5
            cur = project_to_simplex(cur)

        if it % 4000 == 0:
            print(
                f"[multi‑sub‑grad] it {it:6d} | time {time.time():6.1f}s | best {best_score:.6f}"
            )
    return best, best_score


# ----------------------------------------------------------------------
# Phase 4 – stochastic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climb (swap / transfer) with symmetry enforcement."""
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()

        if np.random.rand() < 0.5:                # element swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑normalise and enforce symmetry
        cand = cand / cand.sum()
        cand = (cand + cand[::-1]) * 0.5
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a Python list of floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Choose problem size (larger size gives more freedom)
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 800.0:
        n = 4096
    print(f"[setup] Sequence length = {n}")

    # ------------------------------------------------------------------
    # Generate a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    for alpha, beta in [(0.2, 0.2), (0.5, 0.5), (1.0, 1.0),
                        (2.0, 2.0), (5.0, 5.0), (0.5, 2.0), (2.0, 0.5)]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    for k in (30, 50, 80):
        seeds.append(random_spike(n, k=k))

    # extra deterministic shapes
    seeds.append(generate_cosine(n))
    seeds.append(generate_arcsine(n))

    # fallback – if the global variable exists and matches length
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        if len(fallback) == n:
            seeds.append(np.array(fallback, dtype=float) / np.sum(fallback))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Evaluate seeds (exact objective) and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] new best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optimisation phases
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)
    cur = cur / cur.sum()
    cur_score = compute_score(cur)

    # allocate time slices
    t0 = time.time()
    phase1_deadline = t0 + total_time * 0.30
    phase2_deadline = phase1_deadline + total_time * 0.20
    phase3_deadline = phase2_deadline + total_time * 0.30
    # remaining time → hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – soft‑max Adam descent
    # --------------------------------------------------------------
    print("\n[Phase 1] soft‑max Adam descent")
    cur, cur_score = softmax_adam_descent(
        cur,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=2500.0,
        lr=0.015,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after Phase 1] best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – variance‑flattening (with symmetry)
    # --------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("\n[Phase 2] variance‑flattening gradient descent")
        cur, cur_score = variance_gradient_descent(
            cur,
            cur_score,
            phase2_deadline,
            step0=0.06,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – multi‑peak sub‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < phase3_deadline:
        print("\n[Phase 3] multi‑peak sub‑gradient refinement")
        cur, cur_score = multi_max_subgradient_descent(
            cur,
            cur_score,
            phase3_deadline,
            step0=0.09,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 3] best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – hill‑climb polishing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 4] hill‑climb polishing")
        cur, cur_score = hill_climb(cur, deadline, max_steps=2_000_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after hill‑climb] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
