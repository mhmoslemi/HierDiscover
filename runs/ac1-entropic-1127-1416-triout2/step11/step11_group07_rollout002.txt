    n = len(f)

"""
LP‑based iterative refinement for the step‑function minimisation problem.

The algorithm works with a normalised sequence (sum = 1).  The
evaluation function is homogeneous, so the normalisation does not
affect the final score – it only removes the scaling ambiguity.

1. Generate a diverse set of seed shapes and keep the best (lowest
   2·n·max(conv)).
2. Repeatedly solve a linear programme:

      maximise   Σ g_i
      subject to  f * g ≤ M          (element‑wise)
                  g_i ≥ 0

   where M = max(conv(f,f)).  The LP increases the total mass that
   can be placed without raising the maximal auto‑convolution.
   After solving the LP, the vector g is normalised to sum = 1 and
   mixed with the current f (convex combination).  This monotonically
   reduces the maximal convolution.

3. A short sub‑gradient descent on the true max‑objective refines the
   result.
4. A stochastic hill‑climb (swap / transfer moves) finishes the
   polishing.

All heavy work is done with FFT‑based convolutions.  The linear
programme is solved with ``scipy.optimize.linprog`` (HiGHS) using a
sparse Toeplitz matrix that implements the convolution constraints.
The whole search respects the time budget passed in ``budget_s``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators (same as in the previous programme)
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement step
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   Σ g_i
        subject to  conv(f,g) ≤ M   (element‑wise)
                    g_i ≥ 0
    Returns a normalised vector g (sum = 1).  If the LP fails, returns f.
    """
    n = f.shape[0]
    L = 2 * n - 1

    # Build the convolution constraint matrix A (sparse)
    # Row i corresponds to (f * g)[i] = Σ_j f[i‑j] g_j
    A = lil_matrix((L, n), dtype=np.float64)
    for j in range(n):
        # rows j … j+n‑1 receive the whole f vector
        A.rows[j:j + n] = [list(A.rows[r]) + [j] for r in range(j, j + n)]
        A.data[j:j + n] = [list(A.data[r]) + [f_val] for r, f_val in zip(range(j, j + n), f)]

    # Convert to CSR for the optimiser
    A = A.tocsr()

    c = -np.ones(n, dtype=np.float64)       # maximise Σ g_i  → minimise -Σ g_i
    b = np.full(L, M, dtype=np.float64)

    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        # LP failed – fall back to the original vector
        return f

    g = res.x
    s = g.sum()
    if s <= 1e-12:
        return f
    g_norm = g / s
    return g_norm

# Sub‑gradient descent on the true max‑objective (same as before)
# ----------------------------------------------------------------------


def subgradient_refine(a: np.ndarray, start_score: float, deadline: float,
                       max_iter: int = 400_000, step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        step = step0 / np.sqrt(t)
        cur -= step * g
        cur = project_to_simplex(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        if t % 1000 == 0:
            # small random jitter to escape shallow plateaux
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur += jitter
            cur = project_to_simplex(cur)

    return best, best_score

# Simple stochastic hill‑climb (same as before)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, deadline: float, max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Stochastic moves that are kept only when they improve the exact score."""
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.08
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = best.copy()

        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand /= cand.sum()      # renormalise to avoid drift
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best, best_score

        mix_factor = 0.5
        while time.time() < phase_lp_deadline and iteration < 25:
            iteration += 1
            conv = convolution_fft(cur, cur)
            M = conv.max()
            g = lp_refine_step(cur, M)
            if np.allclose(g, cur):
                print("LP step produced no change – exiting LP phase.")
                break
            # convex combination
            cur_new = (1.0 - mix_factor) * cur + mix_factor * g
            cur_new = project_to_simplex(cur_new)

            new_score = compute_score(cur_new)
            if new_score < cur_score:
                cur = cur_new
                cur_score = new_score
                mix_factor = min(mix_factor * 1.1, 0.8)   # be a bit bolder after success
                if cur_score < best_val:
                    best_val = cur_score
                    best_seq = cur.tolist()
                    print(f"[LP] it {iteration:2d} | improved score = {best_val:.6f}")
            else:
                # no improvement – shrink the mixing factor
                mix_factor *= 0.6
                if mix_factor < 1e-3:
                    print("Mixing factor too small – exiting LP phase.")
                    break

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement (same as previous programme)
    # ------------------------------------------------------------------
    if time.time() < phase_sub_deadline:
        print("\n[Phase 2] sub‑gradient refinement")
        cur, cur_score = subgradient_refine(
            cur, cur_score, phase_sub_deadline,
            max_iter=600_000,
            step0=0.07,
        )
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur.tolist()
            print(f"[after sub‑gradient] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – hill‑climb polishing (uses the remaining time)
    # ------------------------------------------------------------------
    if time.time() < phase_hc_deadline:
        print("\n[Phase 3] hill‑climb polishing")
        cur, cur_score = hill_climb(cur, phase_hc_deadline, max_steps=2_000_000)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur.tolist()
            print(f"[after hill‑climb] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
