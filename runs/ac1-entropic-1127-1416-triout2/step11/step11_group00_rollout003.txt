"""
Non‑negative step‑function optimiser.

The optimiser performs three stages within the 1000 s budget:

1. *Global exploration* – sweeps several parametric families
   (power‑law, exponential, Beta, U‑shaped, sinusoidal,
   double‑exponential) and a pool of random Dirichlet samples.
   The best few candidates (by the exact objective
   2·n·max(conv(a,a))/(sum a)²) are retained.

2. *Smoothed gradient descent* – starting from the best candidates,
   a soft‑max approximation of the max‑convolution is used:
   w_i ∝ exp(α·(c_i−max c)).  The gradient
   ∂ (soft‑max)/∂a = 2·conv(w, a_rev) (restricted to n entries)
   is computed with a single FFT‑based convolution.
   α is increased gradually, the step size decays, and after each
   update the vector is projected onto the probability simplex.

3. *Stochastic annealing* – a short Metropolis hill‑climb with
   mass‑transfer moves refines the result further.

All vectors are kept non‑negative and normalised (sum = 1), which
does not affect the official score.  The final sequence is returned
as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        n = v.shape[0]
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper: a * a (full linear convolution)."""
    return fft_convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector on the simplex)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 1.0 / n)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_Ushape(n: int, p: float) -> np.ndarray:
    """U‑shaped: high at both ends, low in the middle."""
    left = (np.arange(1, n + 1, dtype=float)) ** (-p)
    right = (np.arange(n, 0, -1, dtype=float)) ** (-p)
    a = left + right
    total = a.sum()
    return a / total


def generate_sinusoidal(n: int, power: float) -> np.ndarray:
    """High at the edges, low in the centre via a shifted sine."""
    x = np.arange(n, dtype=float) + 0.5
    a = (1.0 - np.sin(np.pi * x / n)) ** power
    total = a.sum()
    return a / total


def generate_double_exponential(n: int, lam1: float, lam2: float) -> np.ndarray:
    """Sum of two exponentials, one anchored at each end."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam1 * idx) + np.exp(-lam2 * (n - 1 - idx))
    total = a.sum()
    return a / total


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 6,
):
    """Explore several families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []  # list of (score, vector)

    def consider(vec: np.ndarray):
        """Project, evaluate and possibly keep."""
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep (broader range)
    for p in np.linspace(0.4, 2.0, 81):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(0.0001, 0.05, 151):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 3) Beta sweep (coarse)
    al_vals = np.linspace(0.2, 5.0, 15)
    be_vals = np.linspace(0.2, 5.0, 15)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    # 4) U‑shaped family
    for p in np.linspace(0.1, 3.0, 30):
        if time.time() >= deadline:
            break
        consider(generate_Ushape(n, p))

    # 5) Sinusoidal family
    for pw in np.linspace(0.1, 4.0, 30):
        if time.time() >= deadline:
            break
        consider(generate_sinusoidal(n, pw))

    # 6) Double‑exponential family
    for lam1 in np.linspace(0.0001, 0.05, 20):
        for lam2 in np.linspace(0.0001, 0.05, 20):
            if time.time() >= deadline:
                break
            consider(generate_double_exponential(n, lam1, lam2))

    # 7) Random Dirichlet draws with varied concentration
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    for conc in concs:
        for _ in range(80):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            consider(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed gradient descent (soft‑max approximation of max)
# ----------------------------------------------------------------------


def smooth_descent(
    a0: np.ndarray,
    deadline: float,
    init_alpha: float = 5.0,
    max_alpha: float = 200.0,
    init_step: float = 0.30,
):
    """Projected gradient descent on a smoothed version of the max‑convolution.
    Returns the best vector seen before the deadline."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step = init_step
    alpha = init_alpha
    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1

        # forward convolution
        conv = full_convolution(a)
        max_c = conv.max()

        # soft‑max weights
        # shift by max_c to keep numbers ≤ 0 → stable exponentials
        exp_w = np.exp(alpha * (conv - max_c))
        w_sum = exp_w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_w / w_sum

        # gradient via convolution with reversed a
        a_rev = a[::-1]
        g_full = fft_convolve(w, a_rev)  # length = (2n‑1)+n‑1 = 3n‑2
        g = 2.0 * g_full[(n - 1) : (n - 1 + n)]

        # adaptive step (simple 1/√t decay)
        step = init_step / np.sqrt(iteration)

        # propose new point and project
        cand = a - step * g
        cand = project_to_simplex(cand)

        cand_val = compute_score(cand)

        # simple acceptance rule (always accept improvements)
        if cand_val < best_val:
            best_val = cand_val
            best_a = cand.copy()
            a = cand
            last_improve_iter = iteration
        else:
            # occasional jitter to escape shallow plateaus
            if iteration - last_improve_iter > 2000:
                jitter = np.random.randn(n) * (init_step * 0.005 / np.sqrt(iteration))
                a = project_to_simplex(a + jitter)
                last_improve_iter = iteration

        # slowly increase the temperature parameter α
        if iteration % 150 == 0 and alpha < max_alpha:
            alpha = min(max_alpha, alpha * 1.07)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (fine‑tuning)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 1e-3,
    final_temp: float = 1e-6,
    cooling_factor: float = 0.9995,
    step_scale_init: float = 0.05,
    step_scale_min: float = 1e-5,
):
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        # temperature schedule
        T = max(final_temp, T * cooling_factor)

        # occasional global jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = project_to_simplex(a + jitter)
            cur_val = compute_score(a)

    return best_a
