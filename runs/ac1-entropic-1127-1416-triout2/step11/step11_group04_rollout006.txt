"""
PROPOSE_CANDIDATE – Fast multi‑phase search for a low‑max‑convolution step
sequence.

The objective (after normalising the sum to 1) is

        score = 2·n·max(conv(a,a))

where `conv(a,a)` is the aperiodic convolution.  A small score means that the
maximum entry of the (non‑reversed) autocorrelation is unusually low.

The algorithm works in three stages:

1.  **Quick analytic / random scan** – a dozen parametric families (uniform,
    left‑heavy power, symmetric product, exponential‑U, and a new “double‑ends”
    construction) are sampled densely.  The best‑scoring candidate is kept.

2.  **Projected sub‑gradient descent** – the true nonsmooth objective is
    minimised directly.  The sub‑gradient at the worst convolution index `k`
    is `g_i = 2·a_{k-i}` (or a single `a_i` on the diagonal).  A decreasing
    step size `α / sqrt(t)` and Euclidean projection onto the probability
    simplex drive the iterates.  Up to several million cheap iterations are
    performed while time permits.

3.  **Directed hill‑climb (simulated‑annealing style)** – each iteration
    focuses on the currently worst convolution offset.  A small amount of
    mass is transferred from the pair of indices that contribute most to that
    peak to a random “quiet’’ index.  The move is accepted if it improves the
    score or with a small Boltzmann probability that decays linearly to zero.
    This aggressive, problem‑specific neighbourhood quickly squeezes out the
    remaining few‑hundredths of a point.

All phases respect a global wall‑clock budget (default 1000 s) and the final
sequence is returned as a plain list of floats.  The routine works for any
length ≤ 3000 (the FFT‑based convolution is O(n log n)).  The best sequence
found is finally evaluated with the supplied `evaluate_sequence` function.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, full aperiodic)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via a real FFT (both vectors are 1‑D)."""
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


def double_end_family(n: int, decay: float, frac: float) -> np.ndarray:
    """
    Put mass at both ends of the interval.
    `decay` – exponential decay rate.
    `frac`  – fraction of positions (on each side) that receive mass.
    """
    m = max(1, int(frac * n))
    left = np.exp(-decay * np.arange(m, dtype=np.float64))
    right = left[::-1].copy()
    a = np.zeros(n, dtype=np.float64)
    a[:m] = left
    a[-m:] = right
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     step0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent for 2·n·max(conv(a,a)).
    Step size = step0 / sqrt(t), projection onto the simplex each iteration.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    cur_score = best_score

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # off‑diagonal terms appear twice in the derivative of a_i a_j
        g[mask] = 2.0 * a[j[mask]]
        # diagonal correction (i == j)
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        # step size schedule
        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        # occasional reporting
        if t % 250_000 == 0 or t == max_iters:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad iter {t} – best score {best_score:.6f}")

    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Directed hill‑climb (simulated‑annealing on the worst offset)
# ----------------------------------------------------------------------


def directed_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Aggressive local search that repeatedly diminishes the contribution of the
    pair of indices responsible for the current worst convolution peak.
    """
    n = start_a.size
    a = start_a.copy()
    cur_score = compute_score(a)

    best_a = a.copy()
    best_score = cur_score

    # linear cooling schedule for the acceptance temperature
    start_T = cur_score * 3e-5 + 1e-9  # small but >0
    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # --- identify the worst convolution offset and its dominant pair ---
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        i_max = int(np.argmax(contrib))
        j_max = k - i_max

        # choose a random “quiet’’ index (preferentially far from i_max/j_max)
        p = np.random.randint(n)
        while p == i_max or p == j_max:
            p = np.random.randint(n)

        # transfer a modest fraction of mass from the offending indices
        # to the quiet index.  Transfer from both ends if they differ.
        eps = 0.018   # about 2 % of the source mass
        delta_i = min(eps * a[i_max], a[i_max])
        a[i_max] -= delta_i
        a[p] += delta_i

        if j_max != i_max:
            delta_j = min(eps * a[j_max], a[j_max])
            a[j_max] -= delta_j
            a[p] += delta_j

        # renormalise (sum is preserved analytically, but small round‑off)
        a = a / a.sum()

        cand_score = compute_score(a)
        delta = cand_score - cur_score

        # linear cooling temperature
        T = start_T * (1.0 - step / max_steps)

        # accept if better, otherwise with Metropolis probability
        if delta < 0.0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            # reject – roll back the move (cheaper than recomputing projection)
            # Undo the transfer (exact reverse) to stay on the same point.
            a[p] -= (delta_i + (delta_j if j_max != i_max else 0.0))
            a[i_max] += delta_i
            if j_max != i_max:
                a[j_max] += delta_j
            # renormalise again to erase rounding error
            a = a / a.sum()

        # occasional status output
        if step % 400_000 == 0:
            print(f"[{time.time():.1f}] hill‑climb step {step} – best {best_score:.6f}")

    return best_a, best_score
