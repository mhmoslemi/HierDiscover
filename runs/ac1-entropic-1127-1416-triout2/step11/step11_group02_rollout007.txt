def solve_direction_lp(f, active_idx):
    n = f.shape[0]
    m = len(active_idx)
    # build A_ub matrix dense (m, n+1)
    A_ub = np.zeros((m, n+1), dtype=np.float64)
    for row_i, k in enumerate(active_idx):
        start = max(0, k-(n-1))
        end = min(k, n-1)
        # f segment for indices start..end inclusive
        A_ub[row_i, start:end+1] = f[start:end+1]
        A_ub[row_i, -1] = 1.0   # coefficient for c
    b_ub = np.zeros(m, dtype=np.float64)
    # equality constraint sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    # bounds
    bounds = [( -f[i], None) for i in range(n)] + [(0.0, None)]  # d_i >= -f_i, c>=0
    # objective: minimize -c => c_obj = [0,...,0, -1]
    c_obj = np.zeros(n+1)
    c_obj[-1] = -1.0
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None, None
    x = res.x
    d = x[:n]
    c_val = x[-1]
    return d, c_val

    n = a.shape[0]

    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    return conv

"""
Step‑function optimiser for the scoring routine ``evaluate_sequence``.
--------------------------------------------------------------------------------
The objective is   score = 2·n·max(conv(a,a))   with  a_i ≥ 0 , Σa_i = 1 .
It is scale‑invariant, therefore we always keep the simplex constraint
(sum‑to‑one) during optimisation.

The optimiser works in four phases:

1. **Broad stochastic sweep** – many cheap families (power‑law,
   geometric, Dirichlet draws and sparse random supports) are sampled.
   The best candidate is kept.

2. **Direction‑LP refinement** – using only the indices that contribute
   to the current maximal convolution we solve a tiny linear programme
   (≈ 20 rows, n variables) that yields a feasible direction `d` which
   *guarantees* a reduction of the peak for sufficiently small step
   size.  After a short line‑search the new point `a←a+ε d` is taken.
   This is repeated until no improvement is possible or the time budget
   expires.

3. **Projected sub‑gradient polishing** – a classic sub‑gradient method
   on `max(conv)` with Euclidean projection onto the probability simplex.

4. **Greedy stochastic hill‑climb** – tiny mass‑transfer moves are tried,
   accepted if they improve the peak (or with a tiny probability to
   escape plateaus).  This cleans the final local basin.

All phases respect the overall 1000 s budget.  The algorithm prefers a
larger problem size (n = 4096) because more degrees of freedom give a
lower theoretical constant, but falls back to n = 2048 if the time left
is small.

The code requires `numpy` and optionally `scipy` (for the linear
programme).  If SciPy is unavailable a CVXPY fallback is used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming back‑ends
# ----------------------------------------------------------------------
try:                                    # SciPy is the fastest for our LPs
    from scipy.optimize import linprog
    _HAVE_SCI = True
except Exception:                       # pragma: no cover
    _HAVE_SCI = False

try:
    import cvxpy as cp
    _HAVE_CVX = True
except Exception:                       # pragma: no cover
    _HAVE_CVX = False

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1)."""
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())

# ----------------------------------------------------------------------
# Phase‑0: Fallback – a previously found construction (if present)
# ----------------------------------------------------------------------
def _fallback_candidate():
    try:
        # ``height_sequence_1`` is injected by the competition harness.
        seq = [float(x) for x in height_sequence_1]  # type: ignore
        return seq
    except Exception:
        return None

# ----------------------------------------------------------------------
# Phase‑1 – broad stochastic sweep over simple families
# ----------------------------------------------------------------------
def best_initial_vector(n: int, deadline: float) -> np.ndarray:
    """Return the best vector found by sweeping a few parametric families."""
    best_a = None
    best_val = float('inf')
    # ---------------- power‑law family -----------------
    for p in np.linspace(0.20, 1.30, 71):          # 0.02 steps
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ---------------- geometric (exponential) family -----------------
    for r in np.linspace(0.30, 0.99, 71):
        if time.time() >= deadline:
            break
        a = r ** np.arange(n, dtype=np.float64)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ---------------- Dirichlet draws (different concentrations) -----------------
    rng = np.random.default_rng(0)
    alphas = [0.1, 0.2, 0.4, 0.7, 1.0]
    for alpha in alphas:
        for _ in range(15):
            if time.time() >= deadline:
                break
            a = rng.gamma(alpha, 1.0, size=n)
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a /= a_sum
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    # ---------------- Sparse random support -----------------
    for k_frac in [0.02, 0.05, 0.10]:          # support size = k_frac·n
        k = max(2, int(k_frac * n))
        for _ in range(20):
            if time.time() >= deadline:
                break
            idx = rng.choice(n, size=k, replace=False)
            weights = rng.random(k)
            a = np.zeros(n, dtype=np.float64)
            a[idx] = weights
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    # ---------------- Upsampled previous fallback (if any) -----------------
    fb = _fallback_candidate()
    if fb is not None:
        fb_arr = np.array(fb, dtype=np.float64)
        if fb_arr.size != n:
            # simple repeat‑until‑size (nearest neighbour up‑sampling)
            repeat = int(np.ceil(n / fb_arr.size))
            fb_arr = np.repeat(fb_arr, repeat)[:n]
        fb_arr /= fb_arr.sum()
        val = compute_score(fb_arr)
        if val < best_val:
            best_val, best_a = val, fb_arr

    return best_a if best_a is not None else np.full(n, 1.0 / n)

# ----------------------------------------------------------------------
# Phase‑2 – Directional LP refinement (Frank‑Wolfe style but linear)
# ----------------------------------------------------------------------
def _solve_direction_lp(f: np.ndarray, active_idx: np.ndarray):
    """
    Solve a tiny linear programme that yields a direction d (∑d = 0,
    d_i ≥ –f_i) such that conv(f,d) ≤ –c on the chosen active rows.
    Maximising c gives the largest guaranteed reduction of the peak.
    Returns (d, c) or (None, None) on failure.
    """
    n = f.shape[0]
    m = active_idx.size

    # Build the (dense) constraint matrix A_ub * [d;c] ≤ 0
    A_ub = np.zeros((m, n + 1), dtype=np.float64)
    for row, k in enumerate(active_idx):
        # Conv row k : coeffs are a segment of f
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1) + 1               # Python slice end (exclusive)
        A_ub[row, lo:hi] = f[lo:hi]
        A_ub[row, -1] = 1.0                  # coefficient of the scalar c
    b_ub = np.zeros(m, dtype=np.float64)

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Variable bounds: d_i ≥ –f_i , c ≥ 0
    bounds = [(-f_i, None) for f_i in f] + [(0.0, None)]

    # Objective: maximise c  →  minimise –c
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = -1.0

    # --------------------------------------------------------------
    # Try SciPy first (fast for our size), fall back to CVXPY if needed
    # --------------------------------------------------------------
    if _HAVE_SCI:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
            if not res.success:
                return None, None
            x = res.x
            return x[:n], x[-1]
        except Exception:               # pragma: no cover
            pass

    if _HAVE_CVX:
        d = cp.Variable(n)
        c = cp.Variable(nonneg=True)
        constraints = [
            A_ub[:, :n] @ d + c <= 0,
            cp.sum(d) == 0,
            d >= -f
        ]
        prob = cp.Problem(cp.Maximize(c), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=2000, abstol=1e-9, reltol=1e-9)
            if prob.status not in ("optimal", "optimal_inaccurate"):
                return None, None
            return np.asarray(d.value).flatten(), float(c.value)
        except Exception:               # pragma: no cover
            return None, None
    return None, None

def directional_refine(f0: np.ndarray, deadline: float,
                       max_iter: int = 200) -> np.ndarray:
    """
    Iteratively improve ``f`` by moving in feasible directions obtained
    from the LP above.  The procedure terminates when no improvement
    can be found or the time budget runs out.
    """
    f = f0.copy()
    best_f = f.copy()
    best_M = full_convolution(f).max()
    n = f.shape[0]
    it = 0
    max_active = 20               # rows used in the LP

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = full_convolution(f)
        M = conv.max()

        # Choose the indices that contribute most to the peak.
        # Use a small random subset if the peak plateau is large.
        if conv.size <= max_active:
            active = np.arange(conv.size)
        else:
            # take the top‑max_active entries (unsorted, cheap)
            active = np.argpartition(-conv, max_active - 1)[:max_active]

        d, c_val = _solve_direction_lp(f, active)
        if d is None or c_val is None or c_val <= 0.0:
            break

        # --------------------------------------------------------------
        # Simple line‑search over a few step‑sizes (including ε = 1)
        # --------------------------------------------------------------
        eps_candidates = [0.02, 0.05, 0.10, 0.20, 0.40, 0.80, 1.0]
        best_eps = 0.0
        best_new_M = M
        for eps in eps_candidates:
            if eps > 1.0:
                continue
            f_new = f + eps * d
            # Non‑negativity is guaranteed because d ≥ –f and eps ≤ 1
            M_new = full_convolution(f_new).max()
            if M_new < best_new_M:
                best_new_M = M_new
                best_eps = eps

        if best_eps == 0.0:          # no improvement found for any step‑size
            break

        # Apply the chosen step
        f = f + best_eps * d
        # Numerical cleanup
        f = np.maximum(f, 0.0)
        f_sum = f.sum()
        if not np.isclose(f_sum, 1.0):
            f /= f_sum

        if best_new_M < best_M - 1e-14:
            best_M = best_new_M
            best_f = f.copy()

    return best_f

# ----------------------------------------------------------------------
# Phase‑3 – Projected sub‑gradient polishing (as in the reference)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # Indices where the max is (almost) attained
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:
            # gradient of conv_k w.r.t. a is 2·a shifted
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1) + 1
            g[lo:hi] += 2.0 * a[lo:hi]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to break tiny plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# ----------------------------------------------------------------------
# Phase‑4 – tiny stochastic hill‑climb (greedy acceptance)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Greedy local search: move a small amount of mass between two coordinates."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of the source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        if no_improve > 100_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # 0. Quick fallback using the previously known construction (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    fb = _fallback_candidate()
    if fb is not None:
        try:
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more flexibility)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8192
    else:
        n = 4096

    print(f"[info] problem size = {n}")

    # ------------------------------------------------------------------
    # 1. Broad stochastic sweep
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    if time.time() < sweep_deadline:
        a0 = best_initial_vector(n, sweep_deadline)
        val = compute_score(a0)
        print(f"[sweep] best score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a0.tolist()

    # ------------------------------------------------------------------
    # 2. Directional LP refinement
    # ------------------------------------------------------------------
    refine_deadline = deadline - 200.0
    if best_seq is not None and time.time() < refine_deadline:
        f0 = np.array(best_seq, dtype=float)
        f_refined = directional_refine(f0, refine_deadline, max_iter=250)
        val = compute_score(f_refined)
        print(f"[directional LP] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, f_refined.tolist()

    # ------------------------------------------------------------------
    # 3. Sub‑gradient polishing
    # ------------------------------------------------------------------
    sub_deadline = deadline - 80.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, sub_deadline, step0=0.10)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline, max_iter=600_000)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # Fallback to a uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print(f"[uniform fallback] score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
