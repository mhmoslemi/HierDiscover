"""
Improved step‑function optimiser.

The optimiser works with three main ideas:

1. **Diverse seeds** – deterministic shapes, Dirichlet draws,
   sparse “spike” vectors and a greedy low‑collision construction that
   tries to keep the number of overlapping sums small.

2. **Adam on a smooth max‑surrogate** – minimise the log‑sum‑exp of the
   auto‑convolution (a differentiable upper‑bound of the true max) while
   staying on the probability simplex.  Adam’s adaptive steps give a far
   more aggressive search than the vanilla GD used previously.

3. **Refinement** – a momentum‑sub‑gradient phase focusses on the true
   max‑convolution followed by a simulated‑annealing hill‑climber that
   makes stochastic moves which are accepted with a Metropolis rule.

All updates are projected onto the simplex, therefore the objective
computed by ``evaluate_sequence`` is directly comparable to the intermediate
scores (2·n·max(conv) for a normalised vector).  The algorithm adapts the
sequence length (2048 or 4096) to the available time budget and returns the
best sequence it has found before the timeout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast, exact for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # uniform vector if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the “valid” correlation (length = len(a)):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    Implemented via one FFT convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1: n - 1 + n]


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign random positive
    weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def greedy_low_collision(n: int, k_target: int, attempts: int = 2000) -> np.ndarray:
    """
    Build a set of `k_target` positions such that the number of collisions
    of pairwise sums is kept low.  The algorithm is greedy:
    at each step we test a few random candidates and pick the one that
    minimises the current maximal sum‑multiplicity.
    The resulting vector uses equal weight on the selected positions.
    """
    sum_counts = np.zeros(2 * n - 1, dtype=int)
    chosen = []

    while len(chosen) < k_target:
        best_idx = None
        best_max = None
        best_counts = None

        # try a handful of random candidates
        for _ in range(30):
            cand = np.random.randint(0, n)
            if cand in chosen:
                continue

            # simulate the effect of inserting `cand`
            new_counts = sum_counts.copy()
            # contributions with existing points
            for pt in chosen:
                new_counts[pt + cand] += 1
            # self‑sum 2*cand
            new_counts[2 * cand] += 1

            cur_max = new_counts.max()
            if (best_max is None) or (cur_max < best_max):
                best_max = cur_max
                best_idx = cand
                best_counts = new_counts

        if best_idx is None:   # could not find a new point (highly unlikely)
            break
        chosen.append(best_idx)
        sum_counts = best_counts

        # early stop if we already have a perfect Sidon‑like behaviour
        if sum_counts.max() <= 2:
            # continue – we still want the full size but we already have a very
            # flat distribution of sums.
            pass

    a = np.zeros(n, dtype=float)
    if chosen:
        a[chosen] = 1.0 / len(chosen)
    else:
        a = random_dirichlet(n)
    return a

def adam_softmax_gd(
    a0: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
          Lβ(a) = log Σ exp(β·conv(a,a))

    The surrogate approaches the true max‑convolution as β → ∞.
    """
    n = a0.shape[0]
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = start_score
    it = 0
    start = time.time()
    eps = 1e-12

    while time.time() < deadline:
        # exact score for bookkeeping
        conv = convolution_fft(a, a)
        max_c = conv.max()
        cur_score = 2.0 * n * float(max_c)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # annealed β
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights (stable)
        shifted = conv - max_c          # max → 0
        exp_vals = np.exp(beta * shifted)   # ≤ 1
        w = exp_vals / exp_vals.sum()

        # gradient of Lβ
        grad = 2.0 * beta * correlation_valid(w, a)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = project_to_simplex(a)

        # occasional progress report
        if it % 2000 == 0:
            print(
                f"[Adam GD] iter {it:6d} | best {best_score:.6f} | β {beta:.0f}"
            )
    return best_a, best_score


def momentum_subgradient(
    a0: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.07,
    mu: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution
    with a simple momentum term.
    """
    n = a0.shape[0]
    a = a0.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = start_score

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift `k`
        g = np.zeros_like(a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        v = mu * v - step * g
        a = a + v
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny jitter to escape flats
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_score


def simulated_hill_climb(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 4_000_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a Metropolis acceptance rule.
    Moves: (i) swap two entries, (ii) move a fraction of mass,
    (iii) small Gaussian perturbation.
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # temperature schedule – start relatively high, cool slowly
    T0 = 0.05 * cur_score
    decay = 0.99999

    steps = 0
    while steps < max_steps and time.time() < deadline:
        T = max(T0 * (decay ** steps), 1e-8)

        cand = best.copy()

        # three possible move types
        r = np.random.rand()
        if r < 0.4:                  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.8:                # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = 0.1 * cand[i] * np.random.rand()
                cand[i] -= delta
                cand[j] += delta
        else:                        # Gaussian jitter on a few coordinates
            idx = np.random.choice(n, size=min(5, n), replace=False)
            cand[idx] += np.random.randn(len(idx)) * 0.02 * cand[idx]
            cand = np.maximum(cand, 0.0)

        # re‑normalise to stay on the simplex
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        delta = cand_score - best_score

        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best = cand
            best_score = cand_score
            # slowly increase step size when we improve
            T0 *= 1.02

        steps += 1
        if steps % 250_000 == 0:
            print(f"[hill‑climb] step {steps:,d} | best {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0 – optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1 – choose a length (larger length → better theoretical bound)
    # ------------------------------------------------------------------
    if total_time > 850.0:
        n = 4096
    else:
        n = 2048
    print(f"[setup] using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2 – generate a diverse set of seeds and keep the best one
    # ------------------------------------------------------------------
    seeds = []

    # deterministic shapes
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # beta‑pdf families
    for alpha, beta in [(0.2, 0.2), (0.5, 0.5), (1.0, 1.0),
                        (2.0, 2.0), (5.0, 5.0), (0.5, 2.0), (2.0, 0.5)]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    # random Dirichlet draws (various concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(random_dirichlet(n, conc=conc))

    # sparse spikes with a range of cardinalities
    for k in (30, 50, 70, 90, 110):
        seeds.append(random_spike(n, k=k))

    # low‑collision greedy constructions – target a size that roughly matches
    # the theoretical k ≈ sqrt(2·n / 1.5)
    k_target = max(30, int(np.sqrt(2.0 * n / 1.5)) - 5)
    for _ in range(3):
        seeds.append(greedy_low_collision(n, k_target))

    # include fallback if compatible
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.asarray(best_seq, dtype=float))

    # evaluate all seeds (exact score) and keep the best
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)          # safeguard against drift
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] new best → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – start optimisation from the best seed
    # ------------------------------------------------------------------
    cur_a = np.asarray(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------------------------------------------------------
    # 4 – allocate time to the different phases
    # ------------------------------------------------------------------
    t0 = time.time()
    phase1_deadline = t0 + total_time * 0.35
    phase2_deadline = phase1_deadline + total_time * 0.30
    phase3_deadline = phase2_deadline + total_time * 0.25
    # Remaining time is for the hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    print("\n[Phase 1] Adam soft‑max optimisation")
    cur_a, cur_score = adam_softmax_gd(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=5000.0,
        lr=0.05,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after Phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – momentum sub‑gradient on the true max‑objective
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("\n[Phase 2] Momentum sub‑gradient refinement")
        cur_a, cur_score = momentum_subgradient(
            cur_a,
            cur_score,
            phase2_deadline,
            step0=0.07,
            mu=0.9,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic hill‑climb with simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        print("\n[Phase 3] Simulated hill‑climb polishing")
        cur_a, cur_score = simulated_hill_climb(
            cur_a,
            phase3_deadline,
            max_steps=4_000_000,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 3] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
