def solve_lp_max_sum(f, M):
    n = f.shape[0]
    # objective: minimize -sum(g) => c = -np.ones(n)
    c = -np.ones(n)
    # bounds: (0, None) for each variable
    bounds = [(0, None)] * n

    # build A_ub (2n-1 by n) and b_ub vector M repeated.
    # Build using sparse matrix.
    from scipy.sparse import csr_matrix

    rows = []
    cols = []
    data = []
    # iterate over columns j
    for j in range(n):
        # f entries placed at rows j .. j + n - 1
        # row index = j + i where i from 0..n-1, coefficient = f[i]
        start = j
        end = j + n
        # for each i, row = start + i, col = j, data = f[i]
        rows.extend(range(start, end))
        cols.extend([j] * n)
        data.extend(f.tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)

    # Use linprog with method='highs' and sparse matrix
    # SciPy's linprog may accept A_ub as dense or sparse (since v1.6).
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5})  # maybe limit time.

    n = f.shape[0]
    # data = np.tile(f, n)
    data = np.tile(f, n)
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # call linprog
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception as e:
        print(f"LP solver error: {e}")
    return None

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1. **Broad parametric + random search** – generate a large pool of
   candidates from several analytic families (power‑law, exponential,
   beta, Hann, Blackman, sine, Gaussian, triangular) together with
   random Dirichlet draws.  Keep the best few (by the true objective).

2. **Linear‑programming driven update** – for a promising candidate
   `f` we solve the LP  

        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f * g)_k ≤ M   ∀k ,   M = max(conv(f,f))

   where the convolution constraints are linear in `g`.  The solution
   `g` concentrates mass where `f` is small without increasing the
   current worst convolution value `M`.  We mix `f ← (1‑t) f + t g`
   (with a modest step `t`) and repeat while time permits.  This
   Frank–Wolfe style step quickly reduces the L∞‑norm of the
   autocorrelation.

3. **Polishing** – a projected sub‑gradient descent (active‑set
   gradient) further flattens the convolution.

4. **Stochastic hill‑climb** – a final simulated‑annealing phase
   explores the neighbourhood of the best solution.

All phases respect the 1000 s wall‑clock budget; the programme always
returns a feasible non‑negative list of floats.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.zeros(n)
    return a / s


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.zeros(n)
    return a / s


def generate_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta_ - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.zeros(n)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.zeros(n)
    return a / s


def generate_hann(n: int) -> np.ndarray:
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    return w / w.sum()


def generate_blackman(n: int) -> np.ndarray:
    a = np.arange(n)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * a / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * a / (n - 1))
    )
    return w / w.sum()


def generate_sine(n: int) -> np.ndarray:
    w = np.sin(np.pi * (np.arange(1, n + 1)) / (n + 1))
    return w / w.sum()


def generate_sine_sq(n: int) -> np.ndarray:
    w = np.sin(np.pi * (np.arange(1, n + 1)) / (n + 1)) ** 2
    return w / w.sum()


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = w.sum()
    if s == 0 or not np.isfinite(s):
        return np.zeros(n)
    return w / s


def generate_triangle(n: int) -> np.ndarray:
    if n % 2 == 0:
        half = n // 2
        left = np.arange(1, half + 1)
        right = np.arange(half, 0, -1)
    else:
        half = n // 2
        left = np.arange(1, half + 2)
        right = np.arange(half + 1, 0, -1)
    w = np.concatenate([left, right])
    return w / w.sum()


# ----------------------------------------------------------------------
# Sub‑gradient polishing
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # gradient from all active indices
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-2,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9999,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb with decreasing temperature."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming helper
# ----------------------------------------------------------------------


def build_conv_matrix_structure(n: int):
    """Pre‑compute row/col indices for the Toeplitz convolution matrix."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    ptr = 0
    for j in range(n):
        rows[ptr:ptr + n] = np.arange(j, j + n, dtype=np.int32)
        cols[ptr:ptr + n] = j
        ptr += n
    return rows, cols


def solve_lp_max_sum(f: np.ndarray, M: float,
                     rows: np.ndarray, cols: np.ndarray) -> np.ndarray | None:
    """Solve  maximise Σ g_i   s.t.   f·g ≤ M (component‑wise), g≥0."""
    n = f.shape[0]
    data = np.tile(f, n)               # f repeated for each column
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n)                    # maximise Σ g_i  → minimise –Σ g_i
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception as e:
        print(f"[LP] solver error: {e}")
    return None


def lp_improvement(f0: np.ndarray, rows: np.ndarray, cols: np.ndarray,
                   deadline: float, mix_step: float = 0.20) -> np.ndarray:
    """Repeated LP → mixing steps while time permits."""
    f = f0.copy()
    best_f = f.copy()
    best_val = compute_score(f)

    while time.time() < deadline:
        conv = full_convolution(f)
        M = conv.max()
        g = solve_lp_max_sum(f, M, rows, cols)
        if g is None:
            break
        s = g.sum()
        if s <= 0:
            break
        g /= s                     # renormalise to sum‑1
        f = (1.0 - mix_step) * f + mix_step * g
        val = compute_score(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP] improved score = {best_val:.6f}")
    return best_f


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global exploration of parametric families + random draws
    # -----------------------------------------------------------------
    max_keep = 10
    best_candidates = []                     # list of (score, vector)

    def keep_candidate(vec: np.ndarray):
        """Maintain a bounded list of the best candidates seen so far."""
        v = np.maximum(vec, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best_candidates) < max_keep:
            best_candidates.append((sc, v.copy()))
            best_candidates.sort(key=lambda x: x[0])
        elif sc < best_candidates[-1][0]:
            best_candidates[-1] = (sc, v.copy())
            best_candidates.sort(key=lambda x: x[0])

        # also keep global best
        nonlocal best_seq, best_val
        if sc < best_val:
            best_val, best_seq = sc, v.tolist()
            print(f"[global] new best score = {best_val:.6f}")

    baseline_deadline = deadline - (deadline - start_time) * 0.20   # ~= first 20 %

    # ----- deterministic families -------------------------------------------------
    # power‑law
    for p in np.linspace(0.3, 1.5, 61):
        if time.time() >= baseline_deadline:
            break
        keep_candidate(generate_powerlaw(n, p))

    # exponential
    for lam in np.linspace(0.0001, 0.06, 200):
        if time.time() >= baseline_deadline:
            break
        keep_candidate(generate_exponential(n, lam))

    # beta (coarse grid, extended range)
    al_vals = np.linspace(0.05, 8.0, 27)   # step ≈0.3
    be_vals = np.linspace(0.05, 8.0, 27)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= baseline_deadline:
                break
            keep_candidate(generate_beta(n, alpha, beta))

    # specialised windows
    keep_candidate(generate_hann(n))
    keep_candidate(generate_blackman(n))
    keep_candidate(generate_sine(n))
    keep_candidate(generate_sine_sq(n))
    keep_candidate(generate_triangle(n))

    # ----- Gaussian shapes --------------------------------------------------------
    sigmas = [n * r for r in (0.05, 0.10, 0.15, 0.20, 0.30, 0.40)]
    for sigma in sigmas:
        if time.time() >= baseline_deadline:
            break
        keep_candidate(generate_gaussian(n, sigma))

    # ----- random Dirichlet draws --------------------------------------------------
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = 80
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= baseline_deadline:
                break
            a = np.random.dirichlet([conc] * n)
            keep_candidate(a)

    # -----------------------------------------------------------------
    # 2. Linear‑programming driven improvement (Frank–Wolfe style)
    # -----------------------------------------------------------------
    rows, cols = build_conv_matrix_structure(n)

    lp_deadline = deadline - (deadline - start_time) * 0.15   # allocate ~15 %
    # work on the currently best few candidates
    for sc, cand in best_candidates:
        if time.time() >= lp_deadline:
            break
        # give each candidate a slice of the remaining LP budget
        per_deadline = min(lp_deadline, time.time() + (lp_deadline - time.time()) / 4)
        refined = lp_improvement(cand, rows, cols, per_deadline, mix_step=0.20)
        # update global best if improved
        if refined is not None:
            keep_candidate(refined)

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (focused local optimisation)
    # -----------------------------------------------------------------
    polish_deadline = deadline - (deadline - start_time) * 0.07   # ~7 %
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline, step0=0.35)
        keep_candidate(a_refined)

    # -----------------------------------------------------------------
    # 4. Simulated annealing / stochastic hill‑climb
    # -----------------------------------------------------------------
    anneal_deadline = deadline
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline,
                                      init_temp=1e-2,
                                      final_temp=1e-7,
                                      cooling_factor=0.9999,
                                      step_scale_init=0.03,
                                      step_scale_min=1e-6)
        keep_candidate(a_final)

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
