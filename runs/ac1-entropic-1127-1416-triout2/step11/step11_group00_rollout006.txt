"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Broad global sampling** – generate many candidates from
   power‑law, exponential, beta families and random Dirichlet draws
   (different concentration parameters).  Keep the best few (by the
   exact objective 2·n·max(conv(a,a))).

2. **Projected Adam** – from each of the best candidates run a
   projected Adam optimiser on the true objective.
   The sub‑gradient of  max(conv(a,a))  is the sum of the shifted
   copies of ``a`` belonging to all active convolution indices.
   After each Adam step the vector is projected onto the probability
   simplex (non‑negative and sum = 1).  A small amount of random
   jitter is added from time to time to avoid stagnation.

3. **Fine‑grained simulated annealing** – a Metropolis hill‑climb
   (mass‑transfer moves) is finally applied to the best vector found
   by Adam, with a slowly decreasing temperature.

All phases respect the 1000 s wall‑clock budget.  The final sequence
(a plain Python list of floats) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Adam optimiser (projected sub‑gradient)
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  step0: float = 0.10,
                  max_iters: int = 10_000_000) -> np.ndarray:
    """Projected Adam optimisation of 2·n·max(conv(a,a)).
    Returns the best vector encountered before deadline."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active indices (within tiny relative tolerance)
        rel_eps = max_val * 1e-9
        active = np.where(conv >= max_val - rel_eps)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size slowly decays with sqrt(t)
        step = step0 / (np.sqrt(t) + 1.0)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to shake out of plateaux
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (fine‑grained)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 5e-8,
                        cooling: float = 0.9996,
                        scale_init: float = 0.03,
                        scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb with temperature decay."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # propose a small mass‑transfer move
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]               # move a fraction of a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            # accept
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, scale_min)

        # temperature schedule
        T = max(final_temp, T * cooling)

        # occasional global jitter
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Generate a diverse starting pool and keep the best ``max_keep``."""
    best = []  # list of (score, vector)

    def maybe_add(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep (broader than before)
    for p in np.linspace(0.1, 3.0, 61):          # 61 points
        if time.time() >= deadline:
            break
        maybe_add(generate_powerlaw(n, p))

    # exponential sweep (extended range)
    for lam in np.linspace(0.001, 0.2, 150):
        if time.time() >= deadline:
            break
        maybe_add(generate_exponential(n, lam))

    # beta sweep (coarser but extended)
    al_vals = np.linspace(0.2, 5.0, 16)
    be_vals = np.linspace(0.2, 5.0, 16)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            maybe_add(generate_beta(n, alpha, beta))

    # Random Dirichlet draws – varied concentration
    concs = [0.1, 0.3, 0.6, 1.2, 2.5, 5.0]
    per_conc = 300  # ≈ 1800 draws total
    for conc in concs:
        for _ in range(per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            maybe_add(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        # run Adam on a few of the top candidates (including the current best)
        # we give each run at most ~30 s
        runs = 3
        for run in range(runs):
            sub_deadline = min(polish_deadline,
                               time.time() + 30.0)
            a_refined = adam_optimize(a0, sub_deadline, step0=0.08)
            sc = compute_score(a_refined)
            print(f"[adam run {run+1}] refined score = {sc:.6f}")
            if sc < best_val:
                best_val, best_seq = sc, a_refined.tolist()
                a0 = a_refined  # use the improved vector as start for the next run
        print(f"[adam] best after polishing = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Simulated annealing (fine‑grained stochastic search)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(a_final)
        print(f"[anneal] final score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Final safety net – fall back to uniform if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
