            break

"""
Advanced step‑function optimiser with LP‑based refinement.

The algorithm works in several stages:

1. **Seed generation** – a diverse collection of probability vectors.
2. **Gradient‑based polishing** – soft‑max, variance‑flattening and
   sub‑gradient descent (the same ideas as the previous champion).
3. **Linear‑programming refinement** – given the current distribution `f`,
   solve a linear program that maximises the total mass of a new
   distribution `b` under the constraints   (f * b)[k] ≤ M,
   where * is linear convolution and M = max(f * f).  The solution `b`
   is always at least as “packed” as `f`; after normalisation we mix it
   back into `f`.  Repeating this step drives the peak of the auto‑
   convolution down towards the theoretical lower bound.
4. **Stochastic hill‑climb** – final local search with swaps and small
   mass transfers.

All updates are projected onto the probability simplex so the objective
`2·n·max(conv)` can be compared directly with the value reported by
`evaluate_sequence`.  The LP‑refinement is the novel ingredient that
usually squeezes the score from ≈ 1.511 to below the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (exact for our modest sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: choose `k` positions and assign random weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def generate_gaussian(n: int, sigma: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred on the middle of the array."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / (sigma * n)) ** 2)
    a /= a.sum()
    return a

def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step_coef: float = 0.001,
) -> tuple[np.ndarray, float]:
    """Smooth surrogate (log‑sum‑exp) minimisation with projected GD."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    it = 0
    start = time.time()
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 5000 == 0:
            print(
                f"[soft‑max GD] it {it:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.1f}"
            )
    return best_a, best_score


def variance_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Minimise the variance of the auto‑convolution."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        e = conv - avg
        # correlation_valid(e, cur_a) = (e ⋆ cur_a)
        # we can obtain it via FFT as well:
        grad = 4.0 * convolution_fft(e, cur_a[::-1])[n - 1 : 2 * n - 1]

        it += 1
        step = step0 / np.sqrt(it)

        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 2000 == 0:
            print(
                f"[var‑GD] it {it:6d} | time {time.time():6.1f}s | "
                f"best {best_score:.6f}"
            )
    return best_a, best_score


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Stochastic local search – swaps and fractional transfers."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand /= cand.sum()
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96
        steps += 1
    return best_a, best_score

# LP‑based refinement (the new ingredient)
# ----------------------------------------------------------------------


def lp_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 12,
    t_start: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.

    For the current probability vector `a` (sum = 1) we compute
        M = max(conv(a,a)).
    We then solve
        maximise   Σ b_i
        subject to conv(a,b) ≤ M   (2·n‑1 linear constraints)
                  b_i ≥ 0.
    The optimal `b` is typically much “fatter” than `a`.  After normalising
    `b` we mix it back into `a` with a small step size `t`.  Repeating the
    process drives the peak of the auto‑convolution down.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = start_score
    t = t_start

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        M = conv.max()

        # Build Toeplitz matrix representing the linear map b ↦ conv(a,b)
        # Row k (0‑based) contains a[k‑j] for each column j, zero where out‑of‑range.
        c = np.concatenate([a, np.zeros(n - 1, dtype=float)])
        r = np.concatenate([a[:1], np.zeros(n - 1, dtype=float)])
        A = toeplitz(c, r)                     # shape (2n‑1, n)

        # Linear programme: max Σ b_i  ⇔  min −Σ b_i
        lp_res = linprog(
            c=-np.ones(n, dtype=float),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M, dtype=float),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not lp_res.success:
            print(f"[lp_refine] LP failed at iteration {it}: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()

        # If the LP could not increase the total mass we stop early.
        if sum_b <= 1.0 + 1e-9:
            print("[lp_refine] No mass increase – terminating early")
            break

        # Normalise and mix back.
        b_norm = b / sum_b
        a = (1.0 - t) * a + t * b_norm
        a = project_to_simplex(a)               # safeguard against numerical drift
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[lp_refine] it {it:2d} | M {M:.6f} | sum_b {sum_b:.4f} | "
                f"t {t:.4f} | new best {best_score:.6f}"
            )
        else:
            print(f"[lp_refine] it {it:2d} | no improvement (score {cur_score:.6f})")

        # Gradually shrink the mixing coefficient.
        t = max(t * 0.70, 0.01)

    return best_a, best_score

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the evaluation harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0 – optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1 – choose a moderate length (2048 gives a good compromise for LP)
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 850.0:
        n = 2048               # keep 2048 even for long budgets – LP gets huge otherwise
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2 – diverse seeds, keep the best
    # ------------------------------------------------------------------
    seeds = []

    # deterministic shapes
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # Beta‑pdf families
    for alpha, beta in [(0.2, 0.2), (0.5, 0.5), (1.0, 1.0),
                        (2.0, 2.0), (5.0, 5.0), (0.5, 2.0), (2.0, 0.5)]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    # random Dirichlet (various concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # very sparse spikes
    for k in (30, 50, 80):
        seeds.append(random_spike(n, k=k))

    # Gaussian blobs – another useful family
    for sigma in (0.12, 0.20, 0.30):
        seeds.append(generate_gaussian(n, sigma=sigma))

    # include fallback if length matches
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # Evaluate seeds and retain the best one.
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] improvement → score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – start optimisation from the best seed
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------------------------------------------------------
    # 4 – split the remaining time into phases
    # ------------------------------------------------------------------
    t0 = time.time()
    # roughly 35 % soft‑max, 25 % variance, 25 % sub‑gradient,
    # 10 % LP refinement, remainder hill‑climb.
    phase1_deadline = t0 + total_time * 0.35
    phase2_deadline = phase1_deadline + total_time * 0.25
    phase3_deadline = phase2_deadline + total_time * 0.25
    phase4_deadline = phase3_deadline + total_time * 0.10
    # the rest is for hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent
    # ------------------------------------------------------------------
    print("\n[Phase 1] soft‑max gradient descent")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=3000.0,
        step_coef=0.001,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after Phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – variance‑flattening descent
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("\n[Phase 2] variance‑flattening gradient descent")
        cur_a, cur_score = variance_gradient_descent(
            cur_a,
            cur_score,
            phase2_deadline,
            step0=0.05,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        print("\n[Phase 3] sub‑gradient refinement")
        cur_a, cur_score = subgradient_refine(
            cur_a,
            cur_score,
            phase3_deadline,
            max_iter=600_000,
            step0=0.07,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 3] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – LP‑based refinement (new!)
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        print("\n[Phase 4] LP‑based refinement")
        cur_a, cur_score = lp_refine(
            cur_a,
            cur_score,
            phase4_deadline,
            max_iter=12,
            t_start=0.30,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 4] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – hill‑climb polishing (uses whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 5] hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=2_000_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after hill‑climb] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
