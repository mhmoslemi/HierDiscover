def exp_gradient_descent(a, start_score, deadline, eta_start=0.5, eta_end=0.02):
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    it = 0
    start = time.time()
    # compute exponent for each iteration maybe adapt
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
        # find k of max conv for gradient
        k = int(np.argmax(conv))
        # compute gradient g_i = 2 * cur_a[k - i] (if index valid) else 0
        n = cur_a.shape[0]
        g = np.zeros_like(cur_a)
        i_vals = np.arange(n)
        j = k - i_vals
        mask = (j >=0) & (j < n)
        g[mask] = 2.0 * cur_a[j[mask]]
        # adapt eta
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        eta = eta_start + fraction * (eta_end - eta_start)
        # exponentiated update: a_i <- a_i * exp(-eta * g_i)
        cur_a = cur_a * np.exp(-eta * g)
        cur_a = cur_a / cur_a.sum()
        it+=1
        if it % 2000 == 0:
            print(f"[exp‑GD] it {it} | time {elapsed:.1f}s | best {best_score:.6f}")
    return best_a, best_score

def targeted_hill_climb(a, deadline, max_steps=500_000):
    cur_a = a.copy()
    n = cur_a.shape[0]
    best_a = cur_a.copy()
    best_score = compute_score(cur_a)
    step_scale = 0.1
    steps = 0
    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))
        # compute contributions
        i_vals = np.arange(n)
        j = k - i_vals
        mask = (j>=0) & (j<n)
        contrib = np.zeros_like(cur_a)

"""
Advanced step‑function optimiser (2024‑revision).

Goal
----
Find a non‑negative sequence a_0,…,a_{n‑1} (sum = 1) that minimises

    2·n·max(conv(a,a))

where conv is the full linear convolution.
The optimiser works in several complementary phases:

0️⃣  Seed generation – deterministic, analytic (linear / exponential decay), 
    beta‑family, Dirichlet draws and sparse spikes.  The best seed is kept.

1️⃣  Smooth surrogate minimisation – Adam optimiser on the log‑sum‑exp
    surrogate L_β(a)=log∑exp(β·conv).  β is annealed from 5 to 2000.

2️⃣  Variance‑flattening – projected gradient descent on the variance of the
    auto‑convolution (helps to flatten the whole profile).

3️⃣  Sub‑gradient refinement – classic projected sub‑gradient descent on the
    true max‑objective.

4️⃣  Mirror‑descent (exponentiated‑gradient) – works directly on the
    probability simplex using the sub‑gradient of the max‑convolution.

5️⃣  Targeted hill‑climb – moves mass from the index that contributes most
    to the current peak towards a low‑contributing index.

All phases respect the probability simplex via Euclidean projection
(ℓ₂‑projection) or the natural KL‑projection of the exponentiated‑gradient
step.  The total wall‑clock budget (≈ 1000 s) is split automatically.

The final list of floats is returned; the surrounding harness will call
`evaluate_sequence` on it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a=1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n  # points in (0,1)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_linear_decay(n: int) -> np.ndarray:
    """Linearly decreasing weights a_i ∝ (n‑i)."""
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    return a


def generate_exponential_decay(n: int, r: float = 0.99) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    i = np.arange(n, dtype=float)
    a = r ** i
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: choose `k` distinct positions and assign random weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Correlation helper used by the variance‑flattening phase
# ----------------------------------------------------------------------


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation (length = len(a)):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    Implemented via one FFT‑convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]

# Phase 1 – Adam on the log‑sum‑exp surrogate

def adam_softmax_gradient_descent(
    a0: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    lr: float = 0.02,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimises L_β(a)=log∑exp(β·conv(a,a)) using Adam on the log‑softmax logits.
    The schedule for β is linear from beta_start to beta_end.
    """
    n = a0.shape[0]
    # initialise logits
    x = np.log(a0 + 1e-30)  # avoid log(0)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    t = 0
    best_a = a0.copy()
    best_score = start_score
    start = time.time()

    while time.time() < deadline:
        # current probability vector
        # (softmax, numerically stable)
        shift = x.max()
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()

        # exact score for bookkeeping
        conv = convolution_fft(a, a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # annealed β
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max weights over convolution entries
        max_conv = conv.max()
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t a of the surrogate
        grad_a = 2.0 * beta * np.correlate(w, a, mode="valid")

        # gradient w.r.t logits x (softmax Jacobian)
        avg = np.dot(grad_a, a)  # scalar
        grad_x = a * (grad_a - avg)

        # Adam updates
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        if t % 4000 == 0:
            print(
                f"[Adam GD] it {t:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.1f}"
            )

    return best_a, best_score

def variance_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Minimise the variance of the auto‑convolution."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        e = conv - avg
        grad = 4.0 * correlation_valid(e, cur_a)

        it += 1
        step = step0 / np.sqrt(it)

        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 2000 == 0:
            print(f"[var‑GD] it {it:6d} | best {best_score:.6f}")

    return best_a, best_score

def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score

def exp_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    eta_start: float = 0.5,
    eta_end: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent (a_i ← a_i·exp(-η·g_i)) where g_i is the sub‑gradient
    of max(conv).  η anneals linearly.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    start = time.time()
    it = 0

    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        k = int(np.argmax(conv))

        # sub‑gradient of max‑convolution:
        # g_i = 2·a_{k‑i} (if indices in range) else 0
        g = np.zeros_like(cur_a)
        i_vals = np.arange(n)
        j = k - i_vals
        mask = (j >= 0) & (j < n)
        g[mask] = 2.0 * cur_a[j[mask]]

        # anneal step size
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        eta = eta_start + frac * (eta_end - eta_start)

        # exponentiated‑gradient update (KL‑projection)
        cur_a = cur_a * np.exp(-eta * g)
        cur_a = cur_a / cur_a.sum()

        it += 1
        if it % 2000 == 0:
            print(f"[exp‑GD] it {it:6d} | best {best_score:.6f}")

    return best_a, best_score

def targeted_hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the index that contributes most
    to the current peak towards an index with a tiny contribution.
    Accept only improving moves.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = compute_score(cur_a)

    step_scale = 0.03
    steps = 0
    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # contribution of each component to the peak at shift k
        i_vals = np.arange(n)
        j = k - i_vals
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(cur_a)
        contrib[mask] = cur_a[i_vals[mask]] * cur_a[j[mask]]

        # pick the element with largest contribution
        if contrib.max() > 0:
            i_max = int(np.argmax(contrib))
        else:
            i_max = np.random.randint(n)

        # pick a destination with a small (or zero) contribution
        # we purposely bias towards indices with low contribution
        low_candidates = np.where(contrib < contrib.mean())[0]
        if low_candidates.size == 0:
            j_dest = np.random.randint(n)
        else:
            j_dest = int(np.random.choice(low_candidates))

        if i_max == j_dest:
            # fallback random move if both coincide
            i_max, j_dest = np.random.choice(n, size=2, replace=False)

        delta = step_scale * cur_a[i_max]
        if delta <= 0.0:
            # nothing to move – break early
            break

        cand = cur_a.copy()
        cand[i_max] -= delta
        cand[j_dest] += delta
        # re‑normalise (numerical drift)
        cand = cand / cand.sum()

        cand_score = compute_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            cur_a = cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1
        if steps % 50000 == 0:
            print(f"[targeted hill] step {steps:6d} | best {best_score:.6f}")

    return best_a, best_score

    # 1 – choose sequence length (larger length gives more freedom)

    n = 2048

    seeds = []

    phase4_deadline = phase3_deadline + total_time * 0.20
    # the remainder goes to the final hill‑climb

    # ------------------------------------------------------------------

        phase1_deadline,
        beta_start=5.0,
        beta_end=2500.0,
        lr=0.02,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after Phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – variance‑flattening (optional but cheap)
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("\n[Phase 2] variance‑flattening gradient descent")
        cur_a, cur_score = variance_gradient_descent(
            cur_a,
            cur_score,
            phase2_deadline,
            step0=0.05,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        print("\n[Phase 3] sub‑gradient refinement")

            cur_a,
            cur_score,
            phase3_deadline,
            max_iter=800_000,
            step0=0.07,
        )

            cur_score,
            phase4_deadline,
            eta_start=0.5,
            eta_end=0.02,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 4] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – targeted hill‑climb polishing (remainder of time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 5] targeted hill‑climb polishing")
        cur_a, cur_score = targeted_hill_climb(
            cur_a,
            deadline,
            max_steps=1_200_000,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 5] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
