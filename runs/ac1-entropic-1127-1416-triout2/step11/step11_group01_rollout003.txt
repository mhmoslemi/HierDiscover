"""
Propose a non‑negative step‑function that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

using a multi‑phase search:
  1.  Quick scan of several analytic families (including a sine‑power family).
  2.  Several random starts followed by a smooth‑max Adam optimisation
      (continuation on the β‑parameter) with a long schedule.
  3.  Projected sub‑gradient descent for a cheap long tail.
  4.  Aggressive LP‑based refinement (state‑of‑the‑art iteration).
  5.  Symmetrisation and a final simulated‑annealing hill‑climb.
All phases respect a global wall‑clock budget (default 1000 s) and
keep the best vector seen so far.  The returned list is ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution via real FFT.
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a∗a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def sin_power_family(n: int, p: float) -> np.ndarray:
    """Sinusoid raised to a power: a_i ∝ sin^p(π(i+1)/(n+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = np.sin(np.pi * i / (n + 1)) ** p
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


def random_sparse_dirichlet(n: int) -> np.ndarray:
    """Dirichlet with concentration < 1 gives a sparse simplex point."""
    conc = np.random.uniform(0.05, 0.4)
    a = np.random.dirichlet(np.full(n, conc))
    return a


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max (continuation on β)
# ----------------------------------------------------------------------


def adam_continuation_optimize(start_a: np.ndarray,
                               deadline: float,
                               max_iters: int = 250_000,
                               beta_start: float = 2.0,
                               beta_end: float = 3000.0,
                               lr: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate of max‑convolution:
        f_β(a) = (1/β)·log Σ exp(β·(a∗a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The schedule for β increases linearly in the exponent of the
    continuation (i.e. β = β_start·(β_end/β_start)^{t/T}).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # continuation on β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable soft‑max over β·conv
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth objective w.r.t a:
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam cont. improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheap, many iterations)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient method for the true max‑convolution objective.
    Uses Euclidean projection onto the simplex after each step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]          # diagonal term has factor 1

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (the classic iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then return the convex mixture
        f_new = (1−t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (m×n) such that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)   # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 25})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {sc:.6f}")

        # left‑heavy power family
        for beta in betas:
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {sc:.6f}")

        # exponential‑U family
        for gamma in gammas:
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {sc:.6f}")

        # mixed uniform‑U sweeps
        for _ in range(30):
            if time.time() >= deadline - 30.0:
                break
            lam = np.random.rand()
            gamma = np.random.uniform(0.0, 15.0)
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix) = {sc:.6f}")

        # sine‑power family (new addition)
        for _ in range(20):
            if time.time() >= deadline - 30.0:
                break
            p = np.random.uniform(0.5, 5.0)
            vec = sin_power_family(n, p)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sin‑p={p:.2f}) = {sc:.6f}")

        # random draws (monotone / U‑shape / sparse Dirichlet)
        for _ in range(300):
            if time.time() >= deadline - 30.0:
                break
            r = np.random.rand()
            if r < 0.4:
                vec = random_monotone_start(n)
            elif r < 0.8:
                vec = random_u_start(n)
            else:
                vec = random_sparse_dirichlet(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {sc:.6f}")

    # ------------------------------------------------------------------
    # Normalise the working vector (fallback to uniform if needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] after initial scan score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Multi‑start Adam continuation optimisation
    # ------------------------------------------------------------------
    restarts = 3
    for r in range(restarts):
        if time.time() >= deadline - 30.0:
            break
        # Decide start point: sometimes reuse current best, sometimes fresh random
        if best_seq is not None and np.random.rand() < 0.5:
            start = np.array(best_seq, dtype=np.float64)
        else:
            start = random_monotone_start(n)
        start = np.maximum(start, 0.0)
        start = start / start.sum()

        # allocate a modest share of remaining time for this run
        remaining = deadline - time.time()
        # aim for at most 130 k Adam iterations (fast enough)
        max_iters = min(150_000, int(remaining * 200))  # very rough heuristic
        max_iters = max(30_000, max_iters)

        a_opt, sc_opt = adam_continuation_optimize(
            start,
            deadline,
            max_iters=max_iters,
            beta_start=2.0,
            beta_end=3500.0,
            lr=0.015,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] restart {r} Adam improved to {sc_opt:.6f}")

        # Very cheap sub‑gradient tail after Adam (if time still left)
        if time.time() < deadline - 20.0:
            a_opt2, sc_opt2 = subgrad_optimize(
                a,
                deadline,
                max_iters=2_000_000,
                step0=0.07,
            )
            if sc_opt2 < best_score:
                best_score = sc_opt2
                a = a_opt2
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] restart {r} sub‑grad improved to {sc_opt2:.6f}")

    # ------------------------------------------------------------------
    # 4. Aggressive LP‑based refinement loop
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] starting LP refinement loop")
        mix_weight = 0.2
        lp_iter = 0
        while time.time() < deadline - 5.0 and lp_iter < 60:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc_new = compute_score(a_new)
            if sc_new < best_score - 1e-8:
                best_score = sc_new
                a = a_new
                print(f"[{time.time():.1f}] LP step {lp_iter} improved to {sc_new:.6f}")
            else:
                mix_weight = max(0.05, mix_weight * 0.85)
            lp_iter += 1

    # ------------------------------------------------------------------
    # 5. Symmetrisation test
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {sc_sym:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best = {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 7. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
