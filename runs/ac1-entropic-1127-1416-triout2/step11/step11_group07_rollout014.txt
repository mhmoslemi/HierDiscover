"""
Advanced optimisation for the step‑function problem.

The algorithm works as follows:

1. Generate a large diversity of seed vectors (deterministic shapes,
   Beta‑pdfs, Dirichlet draws, sparse spikes, optional fallback).
   All seeds are projected onto the probability simplex (sum = 1).

2. Pick the best seed according to the exact objective
   2·n·max(conv(a,a)).

3. Refine the seed with several optimisation phases, each respecting the
   simplex constraint:
   • Adam optimiser on a smooth log‑sum‑exp surrogate of the max‑norm.
   • Gradient descent that flattens the whole auto‑convolution (variance
     minimisation).
   • Sub‑gradient descent on the true L∞‑objective.
   • Simulated‑annealing hill‑climb that accepts occasional uphill moves.

The whole routine is time‑bound (``budget_s`` seconds) and always returns
the best vector found.  The returned list is a non‑negative sequence;
its scale is irrelevant because ``evaluate_sequence`` is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast, accurate)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n  # points in (0,1)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: choose `k` distinct positions and assign random weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def adam_softmax_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 8000.0,
    base_lr: float = 0.15,
    eps: float = 1e-8,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of the max‑norm.
    The surrogate is
        Lβ(a) = log Σ_k exp(β·conv_k(a,a)) .
    Its gradient is 2·β·correlation(w, a) where w is the soft‑max weight.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    m = np.zeros_like(cur_a)
    v = np.zeros_like(cur_a)

    it = 0
    start = time.time()
    while time.time() < deadline:
        # exact objective for bookkeeping
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # annealed temperature for the surrogate
        elapsed = time.time() - start
        frac = elapsed / max((deadline - start), 1e-12)
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max weights (stable version)
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of Lβ
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # Adam update (learning‑rate scaled by 1/(β+1) to keep steps moderate)
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = (base_lr / (beta + 1.0)) * m_hat / (np.sqrt(v_hat) + eps)
        cur_a = cur_a - step
        cur_a = project_to_simplex(cur_a)

        if it % 3000 == 0:
            print(
                f"[Adam] it {it:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.0f}"
            )
    return best_a, best_score


def variance_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the auto‑convolution.
    The gradient of V = Σ (conv - avg)² is   ∇V = 4·((conv‑avg)⋆a) .
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        e = conv - avg
        # correlation valid = (e ⋆ cur_a) of length n
        grad = 4.0 * convolution_fft(e, cur_a[::-1])[n - 1 : n - 1 + n]

        it += 1
        step = step0 / np.sqrt(it)
        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 2500 == 0:
            print(
                f"[Var‑GD] it {it:6d} | time {time.time():6.1f}s | best {best_score:.6f}"
            )
    return best_a, best_score


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 800_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) : g_i = 2·a_{k‑i} for valid indices
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            # add a tiny random jitter to avoid stagnation
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


def hill_climb_sa(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
    init_temp: float = 0.02,
    cooling: float = 0.999975,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with simulated annealing.
    Moves that worsen the score are occasionally accepted
    according to a decreasing temperature schedule.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = init_temp * cur_score
    step_scale = 0.08
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = cur_a.copy()

        # move type : swap or mass transfer
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)  # re‑normalise exactly
        new_score = compute_score(cand)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and np.random.rand() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_a = cand
            cur_score = new_score
            # adapt step size slightly when a move is accepted
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            # shrink step scale on rejection to encourage finer moves
            step_scale *= 0.96

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
            # if we improve, slightly re‑heat the temperature to keep exploring
            T = max(T, init_temp * best_score)

        # geometric cooling
        T *= cooling
        steps += 1

    return best_a, best_score
