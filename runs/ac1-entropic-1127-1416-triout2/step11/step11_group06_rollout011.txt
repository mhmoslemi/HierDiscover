def mirror_subgrad_optimize(start_a, time_deadline, max_iters=..., eta0=0.5, topk=5, weight_mode='softmax', temp=0.1):
    a = start_a.copy()
    best = a; best_score = compute_score(a)
    n= len(a)
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline: break
        conv = convolution_fft(a, a)
        # identify top k
        if topk >= conv.shape[0]:
            idxs = np.arange(conv.shape[0])
        else:
            idxs = np.argpartition(conv, -topk)[-topk:]
        # compute weights
        if weight_mode == 'softmax':
            w = np.exp(conv[idxs] / temp)
            w = w / w.sum()
        else:
            w = np.full(len(idxs), 1.0/len(idxs))
        # compute gradient
        g = np.zeros_like(a)
        idx_arr = np.arange(n)
        for w_i, k in zip(w, idxs):
            j = k - idx_arr
            mask = (j >= 0) & (j < n)
            # for positions where i=j (i.e., k even and i=j=k/2) we should add a[j] not 2*a[j]? We'll just add 2*a[j[mask]]
            g[mask] += 2.0 * a[j[mask]] * w_i
        # adjust for diagonal? ignore small effect.

        avg = np.dot(a, g)  # expected value of g under a

        eta = eta0 / np.sqrt(t)

        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50, 50)
        a = a * np.exp(exp_arg)
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        if t % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()

    # final check
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best = a.copy()
    return best, best_score

Non‑negative step‑function optimiser using multi‑start random Dirichlet,
active‑set mirror descent (softmax smoothing) and stochastic hill‑climbing.
"""
import time
import numpy as np

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def convolution_fft(a, b):
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a):
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())

def project_to_simplex(v):
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / v.shape[0])
    return w / w_sum

def generate_random_sparse(n, m):
    """Uniform mass on m distinct indices."""
    idx = np.random.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a

def generate_random_dirichlet(n, alpha):
    """Sample Dirichlet(alpha,…,alpha)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        x = np.ones_like(x)
    return x / x.sum()

def mirror_subgrad_optimize(
    start_a,
    time_deadline,
    max_iters=500_000,
    eta0=0.9,
    topk=5,
    temp=0.05,
):
    """Exponentiated‑gradient descent on a smoothed max‑convolution.
    The sub‑gradient is built from the `topk` largest convolution entries,
    weighted by a soft‑max with temperature `temp`.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    t = 1
    while t <= max_iters and time.time() < time_deadline:
        # full convolution
        conv = convolution_fft(a, a)

        # pick active indices
        if topk >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            # np.argpartition is O(n) and gives unsorted topk indices
            part = np.argpartition(conv, -topk)
            active_idx = part[-topk:]

        # soft‑max weights over the selected entries
        vals = conv[active_idx]
        # avoid overflow
        max_val = np.max(vals)
        w_exp = np.exp((vals - max_val) / temp)
        w = w_exp / w_exp.sum()

        # build sub‑gradient (weighted sum of the per‑peak gradients)
        idx_arr = np.arange(n)
        g = np.zeros_like(a)

        for k, weight in zip(active_idx, w):
            j = k - idx_arr
            mask = (j >= 0) & (j < n)

            # diagonal contribution (i == j) gets coefficient 1, off‑diagonal gets 2
            diag = mask & (j == idx_arr)
            off = mask & (~diag)

            if off.any():
                g[off] += 2.0 * a[j[off]] * weight
            if diag.any():
                g[diag] += a[j[diag]] * weight

        # expected value of the gradient under the current distribution
        avg = np.dot(a, g)

        # diminishing step size
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update (KL‑mirror)
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # renormalise onto the simplex
        sum_a = a.sum()
        if sum_a == 0.0 or np.isnan(sum_a):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # occasional progress check
        if (t % 500) == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        t += 1

    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()

    return best_a, best_score

def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """Very local stochastic hill‑climb on the simplex."""
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

```assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}= \frac{2\,n\;\max\bigl(\text{conv}(a,a)\bigr)}{(\sum a_i)^2},
\qquad a_i\ge 0 .
\]

Because the score is invariant under a global scaling of `a`, we may
constrain the vector to the probability simplex  
\(\sum a_i = 1\) and minimise `2·n·max(conv(a,a))`.

The programme below follows three ideas:

1. **Broad initialisation** – analytic families (uniform, binomial,
   power‑law, …) plus many *random* seeds:
   * sparse uniform vectors on a randomly chosen support,
   * Dirichlet draws with various concentration parameters.
   The best of all these starters is kept.

2. **Active‑set mirror descent** – an exponentiated‑gradient (KL‑mirror)
   optimiser that works on the *smoothed* max‑convolution.
   Instead of using only the single maximal entry, the top‑`k`
   convolution values are taken, weighted with a soft‑max (temperature
   `temp`).  The resulting sub‑gradient gives a much smoother search
   direction while still respecting the simplex constraint.

3. **Stochastic hill‑climb** – a cheap local search that swaps entries
   or moves a small fraction of mass between two positions.  It
   finishes the optimisation with a final polish.

All phases respect a wall‑clock deadline (default 1000 s).  Progress
is printed so the run can be monitored.  The final sequence is returned
as a plain Python list, ready for `evaluate_sequence`.

```python
"""
Non‑negative step‑function optimiser.

Algorithm
---------
1. Build a large pool of starter vectors:
   * analytic families (uniform, central‑binomial, power‑law, geometric,
     Beta‑shaped);
   * random sparse vectors (uniform mass on a random support);
   * Dirichlet samples with several concentration parameters.
   Keep the one with the smallest exact score.

2. Active‑set mirror‑descent on the smoothed max‑convolution.
   In each iteration the convolution is computed, the `topk`
   largest entries are selected and weighted by a soft‑max (temperature
   `temp`).  The weighted sum of the corresponding sub‑gradients is used
   in an exponentiated‑gradient update and the iterate is projected back
   onto the probability simplex.

3. Final stochastic hill‑climb (swap / fractional mass transfer) to
   squeeze out the remaining improvement.

The whole search stops a few seconds before the supplied time budget,
otherwise the harness would penalise the run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return w / s

    if x.sum() == 0.0:

    return x / x.sum()

# Mirror‑descent optimiser with an active set

def mirror_subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.9,
    topk: int = 5,
    temp: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) optimisation on a smoothed
    max‑convolution objective.

    Parameters
    ----------
    start_a : np.ndarray
        Starting point (already on the simplex).
    time_deadline : float
        Absolute Unix time after which the optimiser must stop.
    max_iters : int
        Upper bound on the number of iterations.
    eta0 : float
        Initial step‑size multiplier.
    topk : int
        How many of the largest convolution entries are used to build
        the sub‑gradient.
    temp : float
        Temperature for the soft‑max weighting over the selected entries.

    Returns
    -------
    (best_a, best_score) : tuple
        The best vector found and its exact score.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    idx_arr = np.arange(n)

    t = 1
    while t <= max_iters and time.time() < time_deadline:
        # full convolution and the top‑k entries
        conv = convolution_fft(a, a)

        if topk >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            part = np.argpartition(conv, -topk)
            active_idx = part[-topk:]

        # soft‑max weights on the selected entries
        vals = conv[active_idx]
        vmax = np.max(vals)
        w_exp = np.exp((vals - vmax) / temp)
        w = w_exp / w_exp.sum()

        # weighted sub‑gradient (KL‑mirror)
        g = np.zeros_like(a)

        for k, weight in zip(active_idx, w):
            j = k - idx_arr
            mask = (j >= 0) & (j < n)

            diag = mask & (j == idx_arr)          # i == j
            off = mask & (~diag)                 # i != j

            if off.any():
                g[off] += 2.0 * a[j[off]] * weight
            if diag.any():
                g[diag] += a[j[diag]] * weight

        # Expected value of the sub‑gradient under the current distribution
        avg = np.dot(a, g)

        # Diminishing step‑size
        eta = eta0 / np.sqrt(t)

        # Exponentiated‑gradient update
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)   # avoid overflow/underflow
        a = a * np.exp(exp_arg)

        # Re‑project onto the simplex
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # Periodic progress check
        if (t % 500) == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

        t += 1

    # Final evaluation
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()

    return best_a, best_score

    seq,
    time_deadline,

    Simple stochastic local search (swap / fractional mass transfer) that

    n = a.shape[0]
    best_a = a.copy()

        init_candidates.append(generate_beta_shape(n, float(alpha)))

    # possibly add the previously supplied construction
    try:
        init_candidates.append(np.array(height_sequence_1, dtype=np.float64))
        print("Added provided height_sequence_1 as a starter.")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Random (sparse / Dirichlet) starters
    # ------------------------------------------------------------------
    rnd_starters = []

    # sparse supports of different sizes
    for m in [30, 50, 80, 120, 180, 250, 350]:
        for _ in range(3):
            if time.time() >= deadline:
                break
            rnd_starters.append(generate_random_sparse(n, m))

    # Dirichlet draws with a spectrum of concentration parameters
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(3):
            if time.time() >= deadline:
                break
            rnd_starters.append(generate_random_dirichlet(n, alpha))

    init_candidates.extend(rnd_starters)

    # ------------------------------------------------------------------
    # Pick the best starter
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
        if (idx % 10) == 0:
            print(f"Starter {idx:04d}: score = {val:.6f}   current best = {best_val:.6f}")
        if time.time() >= deadline:
            break

    print(f"\nBest starter after exhaustive scan: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Repeated short mirror‑descent runs (exploration + exploitation)
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline - 60.0:          # reserve ~60 s for polishing
            break

        # choose a start point – either the current best or a fresh random seed
        if np.random.rand() < 0.6:
            start = best_seq.copy()
        else:
            # random Dirichlet with a log‑uniform α in [0.01, 1]
            rand_alpha = 10 ** np.random.uniform(-2, 0)
            start = generate_random_dirichlet(n, rand_alpha)

        # allocate a short time slice for this run
        slice_deadline = now + min(12.0, deadline - now - 30.0)
        # estimate a reasonable iteration budget (≈900 iters per second)
        iters = int(min(300_000, (slice_deadline - now) * 900))

        a_opt, s_opt = mirror_subgrad_optimize(
            start,
            time_deadline=slice_deadline,
            max_iters=iters,
            eta0=0.9,
            topk=7,
            temp=0.04,
        )

        if s_opt < best_val:
            best_val, best_seq = s_opt, a_opt.copy()
            print(f"Improvement from mirror run: new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        steps = 400_000
        print(f"\nStarting final hill‑climb (≤ {steps} steps).")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=deadline - 2.0,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val, best_seq = hill_score, np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved the score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final reporting (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. evaluate_sequence reports {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
```

The programme combines a diversified initial pool, a smoother active‑set mirror descent, and a polishing hill‑climb while staying well within the 1000 s wall‑clock budget.  It has a good chance of beating the previous bound 1.511041 and approaching the target 1.503.
