"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The search consists of three stages:

1. **Broad parametric & random block search** – many simple families
   (power‑law, offset‑power‑law, geometric, beta‑shape, cosine‑power)
   plus a coarse piece‑wise constant representation with a few blocks.
   The best few candidates (lowest 2·n·max(conv)/sum²) are kept.

2. **Projected sub‑gradient polishing** – starting from the best
   candidate we run a projected sub‑gradient descent on the non‑smooth
   objective `max(conv(a,a))` (equivalently on `2·n·max(conv)`);
   the gradient is the sum of the shifted copies of `a` belonging to all
   active (maximal) convolution entries.  A short line‑search over a
   handful of step sizes and a simple momentum term accelerate convergence.

3. **Greedy stochastic hill‑climb** – final fine‑tuning by moving a small
   amount of mass between two random positions and accepting only
   improvements.  The move size is adapted empirically.

All phases respect the overall wall‑clock budget (default 1000 s).  The
returned list is normalised to sum = 1, which is scale‑invariant for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length = 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised vector
    (Σa == 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant; we therefore keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = (i+offset)^(-p)   (offset>0 to avoid division by zero)."""
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    """a_i = r^i  (geometric decay)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    """a_i = (1 - i/(n-1))^beta   (monotonically decreasing)."""
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    a /= a.sum()
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    """a_i = cos(π i / (2 n))^expo  (decreasing cosine shape)."""
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    """
    Coarse piecewise‑constant shape: generate K positive values, sort them
    decreasingly to enforce monotonicity, repeat each value to fill the
    n entries (last block may be slightly larger).  Normalise to sum = 1.
    """
    # Dirichlet(1) → uniform on simplex, then sort decreasingly.
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]                     # decreasing
    # repeat each block to obtain length n
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)
    a = a[:n]                                          # safety trim
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
def sweep_candidates(n: int, deadline: float, top_k: int = 12) -> list[tuple[float, np.ndarray]]:
    """Explore families + random blocks; return list of (score, vector)."""
    best = []  # list of (score, vector), sorted ascending

    def consider(vec: np.ndarray):
        if vec is None or np.isnan(vec).any():
            return
        sc = compute_score(vec)
        if len(best) < top_k:
            best.append((sc, vec))
            best.sort(key=lambda x: x[0])
        else:
            if sc < best[-1][0]:
                best[-1] = (sc, vec)
                best.sort(key=lambda x: x[0])

    rng = np.random.default_rng(123456)

    # ----- power‑law with offset -----
    for p in np.linspace(0.35, 1.20, 44):
        if time.time() >= deadline:
            break
        for off in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]:
            if off >= n:
                continue
            vec = powerlaw_offset(n, p, off)
            consider(vec)

    # ----- geometric decay -----
    for r in np.linspace(0.55, 0.995, 88):
        if time.time() >= deadline:
            break
        vec = geometric(n, r)
        consider(vec)

    # ----- beta‑shape (polynomial tail) -----
    for beta in np.linspace(0.5, 20.0, 38):
        if time.time() >= deadline:
            break
        vec = beta_shape(n, beta)
        consider(vec)

    # ----- cosine‑power -----
    for e in np.linspace(0.3, 6.0, 30):
        if time.time() >= deadline:
            break
        vec = cosine_power(n, e)
        consider(vec)

    # ----- mixed plateau + power‑law (smooth version) -----
    for Lfrac in np.linspace(0.05, 0.30, 6):
        if time.time() >= deadline:
            break
        L = max(1, int(Lfrac * n))
        for p in np.linspace(0.4, 0.9, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head_val = tail[0]   # continuity at the joint
            head = np.full(L, head_val)
            vec = np.concatenate([head, tail])
            vec /= vec.sum()
            consider(vec)

    # ----- random block shapes (coarse search) -----
    K = 30                         # number of blocks
    trials = 4000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        vec = block_shape(n, K, rng)
        consider(vec)

    return best

# Sub‑gradient polishing with a tiny line‑search
# ----------------------------------------------------------------------
def subgrad_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv) with simple line‑search."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    t = 0
    # momentum term
    vel = np.zeros_like(a)

    # a few step‑size candidates (geometric)
    step_candidates = np.array([0.02, 0.05, 0.10, 0.18, 0.30])

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # active indices (allow a tiny tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient = sum_{k∈active} 2 * a shifted by k
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # simple momentum update
        vel = 0.85 * vel - (0.1 / np.sqrt(t)) * g

        # try a few step lengths (including a pure momentum step)
        improved = False
        for step in step_candidates:
            cand = a + step * vel
            cand = project_to_simplex(cand)
            sc = compute_score(cand)
            if sc < best_sc:
                best_sc, best_a = sc, cand.copy()
                a = cand
                improved = True
                break          # accept first improving step

        if not improved:
            # a small random jitter occasionally helps escape plateaus
            if t % 250 == 0:
                jitter = np.random.randn(n) * (0.02 / np.sqrt(t))
                a = project_to_simplex(a + jitter)
            else:
                # fallback: pure projection of a - small gradient step
                cand = a - (0.05 / np.sqrt(t)) * g
                a = project_to_simplex(cand)

    return best_a


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, max_iter: int = 600_000) -> np.ndarray:
    """Tiny random mass‑transfer moves, greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]               # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        if no_improve > 120_000:
            # shake the current point to escape a flat plateau
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Optional simple fallback (if a previously good construction is
    #    available in the global namespace)
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global parametric + random block sweep
    # -----------------------------------------------------------------
    n = 2048                     # length of the step function
    sweep_deadline = deadline - 500.0   # leave ~500 s for refinements
    if time.time() < sweep_deadline:
        top_candidates = sweep_candidates(n, sweep_deadline, top_k=12)
        # keep the best among them
        best_sc, best_vec = top_candidates[0]
        print(f"[global sweep] best score = {best_sc:.6f}")
        if best_sc < best_val:
            best_val, best_seq = best_sc, best_vec.tolist()

    # -----------------------------------------------------------------
    # 2. Sub‑gradient polishing (most time consuming)
    # -----------------------------------------------------------------
    sub_deadline = deadline - 30.0   # keep a little room for final hill‑climb
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgrad_refine(a0, sub_deadline)
        val = compute_score(a_polished)
        print(f"[sub‑grad polish] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – fall back to a uniform vector if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Final official evaluation (includes the security wrappers)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported (official) score = {final_score:.6f}")
    return best_seq
