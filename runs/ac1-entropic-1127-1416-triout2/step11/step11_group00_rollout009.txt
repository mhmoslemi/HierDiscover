"""
Non‑negative step‑function optimiser (Adam + temperature‑smoothed max).

The objective returned by ``evaluate_sequence`` is
    2·n·max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant – we can (and do) keep ``sum(a)=1``.
The problem is to minimise the L∞‑norm of the self‑convolution.
We use a smooth approximation of the max (log‑sum‑exp) with a
temperature that is slowly reduced.  The gradient of this smooth
objective w.r.t. the coefficients can be written in closed form as a
cross‑correlation between the current vector and a weight vector
derived from the soft‑max over the convolution entries.

An Adam optimiser (first‑ and second‑moment adaptation) drives the
coefficients on the probability simplex towards a flatter autocorrelation.
Occasional random jitter helps to escape tiny plateaux.
The whole optimisation is run for the allotted wall‑clock budget
(≈ 1000 s) and the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via a real FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())   # next power of two ≥ m
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    C = X * Y
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (same‑length case)."""
    return fft_convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used for comparison (sum(a) is assumed to be 1,
    otherwise the expression is scale‑invariant).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed‑max optimiser (Adam)
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser that minimises a temperature‑smoothed max of the
    self‑convolution.  The vector is kept on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.04                     # base learning rate
    t = 0

    # temperature schedule – start relatively high, shrink gradually
    conv = full_convolution(a)
    tau = max(conv.max() * 0.15, 1e-6)   # initial temperature
    tau_min = 1e-6
    tau_decay = 0.9995

    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        t += 1

        # ----- convolution and soft‑max weights -----
        conv = full_convolution(a)                 # length 2·n‑1
        max_conv = conv.max()
        # keep temperature positive and decreasing
        tau = max(tau * tau_decay, tau_min)

        # soft‑max over convolution entries (stable formulation)
        shifted = conv - max_conv                    # max becomes 0
        exps = np.exp(shifted / tau)
        w = exps / exps.sum()                        # weights sum to 1

        # ----- gradient: 2 * cross‑correlation of a with w -----
        # cross‑correlation a ⋆ w = conv_rev where
        # conv_rev = conv(a[::-1], w) (full length = 3·n‑2)
        conv_rev = fft_convolve(a[::-1], w)
        # slice indices n‑1 … n‑1+n‑1 give the n‑dimensional result
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = project_to_simplex(a)

        # ----- keep the best exact score seen so far -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if t % 750 == 0:
            jitter = np.random.randn(n) * (lr * 0.02)
            a += jitter
            a = project_to_simplex(a)
            # reset Adam moments to keep the optimiser stable after a jump
            m.fill(0.0)
            v.fill(0.0)

    return best_a
