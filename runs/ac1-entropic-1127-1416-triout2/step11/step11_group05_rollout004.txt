"""
Propose a step‑function sequence that (approximately) minimizes

    2·n·max(convolution(a,a))   (with a on the simplex).

The algorithm is a multi‑stage stochastic search:

1. **Broad seeding** – generate many candidate shapes of length `n`
   from simple analytic families (product, power, U‑shape) with
   a wide sweep of parameters, plus random Dirichlet draws.
2. **Quick evaluation** – keep only the best few seeds (by the exact
   objective) for each length.
3. **Smooth‑max L‑BFGS** – for each retained seed run L‑BFGS on a
   smooth approximation of the max‑convolution (log‑sum‑exp) with a
   schedule of increasing smoothing parameters `β`.  The optimisation
   is performed on the soft‑max parametrisation `a = softmax(z)`,
   guaranteeing non‑negativity and the simplex constraint.
4. **Symmetrisation & projection** – after every L‑BFGS stage the
   vector is forced to be symmetric and projected onto the simplex.
5. **Final simulated‑annealing polish** – a short SA hill‑climb
   refines the best solution found.
6. **Escalating problem size** – the whole pipeline is first run for
   `n = 2048` / `4096`.  If time permits, a final refinement on a
   larger length (`n = 8192`) is performed, starting from an
   up‑sampled version of the best smaller‑`n` solution.

All steps respect a 1000‑second wall‑clock budget and continuously
track the globally best sequence.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution using real FFT (fast for lengths up to ~10⁴)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (fast seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised to sum‑1."""
    i = np.arange(n, dtype=np.float64)
    # avoid log(0) when i=0 or i=n-1 → add 0.5 inside
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # handle possible overflow for α=0 (logprod may be -inf at ends)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad_z) where

        f   = smooth max = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z  (gradient w.r.t. the soft‑max parameters)

    The gradient is derived analytically, avoiding finite differences.
    """
    a = softmax(z)                                   # a lives on the simplex
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weighting over the convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta                 # smooth approximation of max
    w = expw / sumexp                                # soft‑max weights (sum = 1)

    # gradient wrt a:  grad_a = 2 * (a_rev * w)   (central slice)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]        # shape (n,)

    # back‑prop through the soft‑max (softmax Jacobian)
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Run successive L‑BFGS optimisations on a schedule of smoothing
    parameters `betas`.  The function adheres to the global deadline
    and returns the best vector found (still on the simplex).
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        # initialise the optimiser in the soft‑max domain
        z0 = np.log(a + 1e-12)

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        # limiter to avoid overrunning the deadline
        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as e:
            print(f"  L‑BFGS failure: {e}")
            continue

        a = softmax(res.x)

        # enforce symmetry and renormalise after each stage
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    new local best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Simple SA walk on the simplex.  Moves are either a swap of two
    entries or a transfer of a fraction of mass from one index to
    another.  Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # Metropolis acceptance
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # adapt step size slowly toward more aggressive moves
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score

def upsample_to(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    up = np.interp(x_new, x_old, a)
    # re‑normalise after interpolation
    return project_to_simplex(up)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  The routine respects the wall‑clock budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # small safety buffer
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback using a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem sizes based on remaining time
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    n_list = [2048, 4096]
    if remaining > 400:
        n_list.append(8192)          # larger size if we have spare seconds

    # ------------------------------------------------------------------
    # 2. Loop over problem sizes, generate seeds, and run optimisation
    # ------------------------------------------------------------------
    for n in n_list:
        if time.time() >= deadline - 30:
            break
        print(f"[{time.time():.1f}] ── Working on length n = {n}")

        # ---- generate a fairly large pool of seeds -------------------
        seeds = []

        # 2a. Analytic families – sweep a generous range of parameters
        for alpha in np.linspace(0.0, 6.0, 25):      # product family
            seeds.append(symmetric_product_family(n, alpha))
        for beta in np.linspace(0.0, 6.0, 25):       # left‑heavy power
            seeds.append(left_heavy_power_family(n, beta))
        for gamma in np.linspace(0.0, 8.0, 25):      # exponential U‑shape
            seeds.append(exp_u_family(n, gamma))

        # 2b. Random Dirichlet draws (both dense and sparse)
        for _ in range(12):
            # dense Dirichlet (concentration = 1)
            seeds.append(np.random.dirichlet(np.ones(n)))
        for _ in range(8):
            # sparse Dirichlet (concentration < 1) – creates spikier shapes
            seeds.append(np.random.dirichlet(np.full(n, 0.3)))

        # 2c. Uniform vector (the classic baseline)
        seeds.append(np.full(n, 1.0 / n))

        # ----------------------------------------------------------------
        # 2d. Quick evaluation of all seeds, keep the best few
        # ----------------------------------------------------------------
        scores = []
        for vec in seeds:
            # tiny safeguard: keep numeric stability
            vec = np.maximum(vec, 0.0)
            vec = project_to_simplex(vec)
            scores.append(compute_score(vec))

        # keep the top K seeds for further optimisation
        K = 7
        top_idx = np.argsort(scores)[:K]
        top_seeds = [seeds[i] for i in top_idx]

        # ----------------------------------------------------------------
        # 2e. L‑BFGS refinement for each top seed
        # ----------------------------------------------------------------
        for idx, a_start in enumerate(top_seeds):
            if time.time() >= deadline - 30:
                break
            # schedule of smoothing parameters – the last values are very sharp
            betas = [1, 5, 30, 150, 800, 3000, 12000]
            # adapt number of internal L‑BFGS iterations to problem size
            max_iter_stage = max(50, int(200 * (4096 / n)))
            a_opt, sc_opt = l_bfgs_optimize(a_start,
                                            betas=betas,
                                            max_iter_per_stage=max_iter_stage,
                                            deadline=deadline - 5.0)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}] ★ global best improved to {best_score:.6f} (n={n})")

        # ----------------------------------------------------------------
        # 2f. Symmetrisation & final projection for this length
        # ----------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            a_arr = np.array(best_seq, dtype=np.float64)
            a_arr = (a_arr + a_arr[::-1]) * 0.5
            a_arr = project_to_simplex(a_arr)
            sc_sym = compute_score(a_arr)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = a_arr.tolist()
                print(f"[{time.time():.1f}] symmetry projection improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Optional upscale to a larger length (if we finished with a
    #    smaller n and still have time)
    # ------------------------------------------------------------------
    if best_seq is not None and len(best_seq) < 8192 and time.time() < deadline - 60:
        # try to exploit the larger theoretical bound by up‑sampling
        a_small = np.array(best_seq, dtype=np.float64)
        a_big = upsample_to(a_small, 8192)
        print(f"[{time.time():.1f}] up‑sampling to n=8192 for final refinement")
        betas = [30, 150, 800, 3000, 12000]
        a_refined, sc_refined = l_bfgs_optimize(a_big,
                                                betas=betas,
                                                max_iter_per_stage=100,
                                                deadline=deadline - 5.0)
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] up‑sampled refinement best = {best_score:.6f}")

        # tiny symmetry enforcement (still useful after up‑sampling)
        a_arr = np.array(best_seq, dtype=np.float64)
        a_arr = (a_arr + a_arr[::-1]) * 0.5
        a_arr = project_to_simplex(a_arr)
        sc_sym = compute_score(a_arr)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_arr.tolist()
            print(f"[{time.time():.1f}] symmetry after up‑sample improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing polish on the globally best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA polish")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=4_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA polish improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback safety net (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 512
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Final evaluation and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return best_seq
```

This implementation expands the exploration of analytic families, adds numerous random Dirichlet seeds, runs a more gradual smoothing schedule for L‑BFGS, upsamples to a larger problem size when time permits, and finishes with a longer simulated‑annealing polish. All steps respect the 1000 s wall‑clock budget and return the best sequence found.
