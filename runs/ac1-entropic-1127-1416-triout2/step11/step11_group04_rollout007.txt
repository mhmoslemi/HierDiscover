def solve_lp_for_g(f, M):
    n = f.shape[0]
    m_len = 2*n - 1
    A = np.zeros((m_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)]*n
    b_ub = np.full(m_len, M, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, False
    g = np.maximum(res.x, 0.0)
    if g.sum() <= 0:
        return None, False
    return g, True

The objective  

\[
\Phi(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

is scale‑invariant, therefore we may restrict ourselves to probability vectors  
\(\sum_i a_i = 1,\;a_i\ge 0\).  
For such vectors  

\[
\Phi(a)=2\,n\;\max (a*a)
\]

so the task is to minimise the peak of the auto‑convolution.  
The program below follows three complementary ideas

1. **Fast analytic / random scan** – generate many candidate families
   (symmetric, left‑heavy, exponential‑decay, monotone, etc.) and keep the
   best one as a seed.

2. **Iterated LP‑refinement** – given a seed \(f\) with current peak
   \(M=\max(f*f)\) we solve  

   \[
   \max\_{g\ge 0}\; \mathbf{1}^\top g
   \quad\text{s.t.}\quad f*g\le M ,
   \]

   obtain a direction \(g\) and mix \(f\) and the normalised \(g\) with
   an *optimal* mixing weight \(t\in(0,1]\).  
   The line‑search for \(t\) is performed analytically on the
   quadratic expression of the convolution, which is cheap
   (vectorised over a fine grid).  
   Repeating this step quickly drives the peak down.

3. **Aggressive simulated‑annealing hill‑climb** – a final
   random‑walk on the simplex (swaps and mass transfers) with a linear
   cooling schedule removes any remaining local irregularities.

All phases respect the global wall‑clock limit (default 1000 s).  
The code is self‑contained, uses only NumPy/SciPy and can be executed
directly by the judge.

```python
"""
Step‑function optimiser for minimise 2·n·max(conv(a,a)) / (sum(a)**2).

Main ideas
-----------
* fast analytic / random scan to obtain a good seed,
* iterative linear‑programming refinement with an exact 1‑D line‑search
  for the optimal mixing weight,
* final simulated‑annealing hill‑climb on the simplex.

The implementation works for any length up to ~3000 (the LP builds a
(2n‑1)×n matrix).  All helper functions are top‑level; no closures are
used.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric distribution – more mass near the ends."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing power law."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponentially decaying from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing shape (sorted power‑law)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random centre‑decay."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Linear‑programming direction (g) for a given bound M
# ----------------------------------------------------------------------


def solve_lp_for_g(f: np.ndarray, M: float):
    """
    Solve  max Σ g_j   s.t.   (f * g)_k ≤ M  ∀k,   g ≥ 0 .
    Returns (g, success_flag).
    """
    n = f.shape[0]
    m_len = 2 * n - 1

    # Build the Toeplitz convolution matrix A @ g = f * g
    A = np.zeros((m_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ↔ minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(m_len, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, False

    g = np.maximum(res.x, 0.0)
    if g.sum() <= 0.0:
        return None, False
    return g, True


def vectorised_line_search(C0: np.ndarray, C1: np.ndarray, C2: np.ndarray,
                          n: int, cur_score: float):
    """
    Find the mixing weight t∈(0,1] that minimises
        2·n·max( (1‑t)²·C0 + 2t(1‑t)·C1 + t²·C2 )
    using a coarse grid + local refinement.
    Returns (best_t, best_score).  If no improvement, best_t = 0.
    """
    # coarse grid
    t_grid = np.linspace(0.0, 1.0, 101)[1:]          # exclude 0
    t = t_grid[:, None]                             # (T,1)
    one_minus_t = 1.0 - t

    conv_grid = (one_minus_t**2) * C0 + 2.0 * t * one_minus_t * C1 + (t**2) * C2
    max_vals = conv_grid.max(axis=1)
    scores = 2.0 * n * max_vals

    best_idx = np.argmin(scores)
    best_t = float(t_grid[best_idx])
    best_score = float(scores[best_idx])

    # if we already improve, try a finer neighbourhood
    if best_t > 0.0 and best_score < cur_score - 1e-12:
        lo = max(0.0, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)
        t_fine = np.linspace(lo, hi, 41)[1:-1]          # drop the ends
        tf = t_fine[:, None]
        one_minus_tf = 1.0 - tf
        conv_fine = (one_minus_tf**2) * C0 + 2.0 * tf * one_minus_tf * C1 + (tf**2) * C2
        max_fine = conv_fine.max(axis=1)
        scores_fine = 2.0 * n * max_fine
        fine_idx = np.argmin(scores_fine)
        if scores_fine[fine_idx] < best_score - 1e-12:
            best_t = float(t_fine[fine_idx])
            best_score = float(scores_fine[fine_idx])

    if best_score >= cur_score - 1e-12:
        return 0.0, cur_score
    return best_t, best_score


def lp_refine_iterative(a: np.ndarray, deadline: float,
                        max_outer_iters: int = 30):
    """
    Repeatedly:
        * solve the LP for the current bound M = max(a*a)
        * mix a with the normalised LP solution using the optimal weight t
    until no improvement or the time limit is reached.
    Returns (best_a, best_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_outer_iters:
        conv = convolution_fft(best_a, best_a)
        M = float(conv.max())

        g, ok = solve_lp_for_g(best_a, M)
        if not ok:
            break

        # normalise direction to a probability vector
        g_norm = g / g.sum()

        # pre‑compute the three convolutions needed for the 1‑D search
        C0 = conv                               # a*a
        C1 = convolution_fft(best_a, g_norm)   # a*g
        C2 = convolution_fft(g_norm, g_norm)   # g*g

        # line‑search for the best mixing weight
        t_opt, new_score = vectorised_line_search(C0, C1, C2, n, best_score)

        if t_opt <= 0.0:
            # no improving mix found
            break

        # update the iterate
        best_a = (1.0 - t_opt) * best_a + t_opt * g_norm
        best_a = project_to_simplex(best_a)
        best_score = new_score
        print(f"[{time.time():.1f}] LP‑iter {it}: improved to {best_score:.6f} (t={t_opt:.4f})")
        it += 1

    return best_a, best_score

# Stochastic hill‑climb (simulated‑annealing style)

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    Temperature is linearly cooled to zero.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 5e-5          # small initial temperature
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:        # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                             # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)    # linear cooling
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score
```
