def propose_candidate(seed=12345, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    best_seq = None
    best_score = float('inf')
    
    # Helper to evaluate and update best
    def consider(vec):
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best score {best_score:.6f}")
    
    # Stage 0: try simple uniform baseline
    n0 = 1024
    uniform = np.full(n0, 1.0/n0)
    consider(uniform)

    # Stage 1: analytic families scanning
    # Choose a moderate n for scanning (e.g., 4096)
    n = 4096

    # product family scan near optimal region
    for alpha in np.linspace(0.2, 1.2, 25):
        if time.time() >= deadline - 10: break
        vec = symmetric_product_family(n, alpha)
        consider(vec)
    
    # power law left heavy scan
    for beta in np.linspace(0.0, 1.5, 20):
        if time.time() >= deadline - 10: break
        vec = left_heavy_power_family(n, beta)
        consider(vec)
    
    # exp U shape
    for gamma in np.linspace(0.0, 4.0, 20):
        if time.time() >= deadline - 10: break
        vec = exp_u_family(n, gamma)
        consider(vec)
    
    # beta family fine scan around .5
    for p in np.linspace(0.2, 0.8, 13):

"""
Propose a low‑peak step‑function for the auto‑convolution objective.

The algorithm works in several phases, always respecting a 1000 s
wall‑clock budget:

1. **Fast analytic scan** – a handful of parametrised families
   (product‑power, power‑law, exponential‑U, beta, mixtures) are
   evaluated on a moderately large length (n≈4096).  The best
   candidate is kept.

2. **Large‑scale Adam optimisation** – the best candidate is refined
   using Adam on a smooth log‑sum‑exp surrogate of the true
   max‑convolution.  The optimiser works on the log‑probabilities
   `z = log(a+ε)` and periodically enforces symmetry and the simplex
   constraint.

3. **Projected sub‑gradient descent** – the exact max‑convolution
   objective is minimised with a simple decreasing step‑size scheme.
   This helps to escape the smoothing bias of Adam.

4. **Optional short simulated‑annealing polish** – a cheap hill‑climb
   on the simplex is performed if time permits.

All intermediate vectors are projected onto the probability simplex
(`∑a_i = 1, a_i ≥ 0`).  The final sequence is returned as a plain
list of floats; `evaluate_sequence` is called once at the end so that
the official wrapper performs its safety checks.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.ifft(C, nfft).real[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective : 2·n·max(a*a)/(sum(a)^2).
    The vector does not need to be normalised; the formula is
    scale‑invariant.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    maxc = conv.max()
    return 2.0 * n * maxc / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. sum=1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric product family:
       a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against overflow for extreme α
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w[w <= 0] = 0.0
    w /= w.sum()
    return w


def mixture_product_uniform(n: int, lam: float, alpha: float) -> np.ndarray:
    """Convex mixture of the symmetric product family and a uniform vector."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    prod = symmetric_product_family(n, alpha)
    w = lam * uniform + (1.0 - lam) * prod
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Random initialisers (fallback / diversification)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """~5 % non‑zero entries, equal mass on the support."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 10.0,
                         beta_end: float = 4000.0,
                         lr: float = 0.018,
                         sym_interval: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best vector (projected on the simplex) and its exact score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise log‑probabilities
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for the smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the surrogate w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective check
        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improvement to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 800_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for  max_k (a*a)_k .
    Returns the best vector encountered and its score.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improvement to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (lightweight)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 150_000) -> Tuple[np.ndarray, float]:
    """Very cheap SA on the simplex – mainly a polishing phase."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 5e-5  # tiny initial temperature
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # simple linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the wall‑clock budget
    and returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_score = float("inf")

    def consider(vec: np.ndarray) -> None:
        """Update global best if the candidate is better."""
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0. Very small uniform baseline (just in case)
    # ------------------------------------------------------------------
    n0 = 256
    consider(np.full(n0, 1.0 / n0))

    # ------------------------------------------------------------------
    # 1. Analytic families – coarse scan (n = 4096)
    # ------------------------------------------------------------------
    n = 4096
    print(f"[{time.time():.1f}] scanning analytic families (n={n})")

    # product family – dense search around the region that historically
    # gave good results (α ≈ 0.5 … 1.2)
    for alpha in np.linspace(0.25, 1.25, 21):
        if time.time() >= deadline - 20:
            break
        vec = symmetric_product_family(n, alpha)
        consider(vec)

    # left‑heavy power‑law (useful for very edge‑concentrated shapes)
    for beta in np.linspace(0.0, 1.5, 16):
        if time.time() >= deadline - 20:
            break
        vec = left_heavy_power_family(n, beta)
        consider(vec)

    # exponential U‑shape
    for gamma in np.linspace(0.0, 5.0, 21):
        if time.time() >= deadline - 20:
            break
        vec = exp_u_family(n, gamma)
        consider(vec)

    # beta family – scan a finer region around p,q≈0.5
    for p in np.linspace(0.2, 0.9, 15):
        for q in np.linspace(0.2, 0.9, 15):
            if time.time() >= deadline - 20:
                break
            vec = beta_family(n, p, q)
            consider(vec)

    # mixture of product family with uniform – extra flexibility
    for lam in np.linspace(0.0, 1.0, 6):
        for alpha in np.linspace(0.3, 1.0, 12):
            if time.time() >= deadline - 20:
                break
            vec = mixture_product_uniform(n, lam, alpha)
            consider(vec)

    # ------------------------------------------------------------------
    # 2. Adam refinement on the best candidate found so far
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 180:
        print(f"[{time.time():.1f}] long Adam refinement (phase 1)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 120,
            max_iters=350_000,
            beta_start=8.0,
            beta_end=4500.0,
            lr=0.017,
            sym_interval=2500,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam‑1 best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. A couple of short Adam restarts from random initialisers
    # ------------------------------------------------------------------
    if time.time() < deadline - 100:
        for r in range(3):
            if time.time() >= deadline - 50:
                break
            print(f"[{time.time():.1f}] Adam restart #{r+1}")
            # random start: either sparse or exponential‑U
            a0 = random_sparse_start(len(best_seq) if best_seq else n)
            if r % 2 == 1:
                a0 = random_u_start(len(best_seq) if best_seq else n)

            a_opt, sc_opt = adam_smooth_optimize(
                a0,
                deadline - 20,
                max_iters=120_000,
                beta_start=10.0,
                beta_end=5000.0,
                lr=0.015,
                sym_interval=1500,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}] Adam restart improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 60:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 10,
            max_iters=600_000,
            step0=0.04,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing polish (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2:
        print(f"[{time.time():.1f}] final SA polish")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=150_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Fallback – should never be needed
    # ------------------------------------------------------------------
    if not best_seq:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = compute_score(best_seq)

    # final official evaluation (includes the security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
