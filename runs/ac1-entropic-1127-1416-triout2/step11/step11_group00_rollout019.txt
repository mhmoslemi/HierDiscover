"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Exploratory sweep** – generate many candidates from
   several parametric families (power‑law, exponential, beta‑shaped,
   sinusoidal) together with random Dirichlet draws.
   The best few (by the exact objective 2·n·max(conv)) are kept.

2. **Smooth‑max gradient descent** – for each kept candidate a
   smooth approximation of the max‑convolution
   (log‑sum‑exp with a temperature β) is minimised.
   The gradient of the smooth max can be written as a convolution,
   which is evaluated efficiently with FFTs.
   After each step the vector is projected onto the probability simplex.
   The temperature β is increased gradually so the process ends at the
   true max‑convolution.

3. **Optional stochastic polishing** – a short simulated‑annealing
   phase (small mass transfers) can be applied to escape possible
   local minima.

The whole search respects the 1000 s wall‑clock budget; the best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (next power-of‑two length)."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    C = X * Y
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a."""
    return convolve_fft(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
       2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smoothmax(conv: np.ndarray, beta: float) -> float:
    """Smooth approximation of max(conv) = log‑sum‑exp / β."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return float(m + np.log(exps.sum()) / beta)


def gradient_of_smoothmax(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Return the gradient of the smooth‑max of the convolution
    with respect to a (the factor 2 is already included).
    """
    n = a.shape[0]
    conv = full_convolution(a)

    # stable softmax weights for the smooth max
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    Z = exps.sum()
    w = exps / Z                     # shape (2n-1,)

    # gradient = 2 * (W @ a) where W_{ij}=w_{i+j}
    w_rev = w[::-1]
    grad_full = convolve_fft(a, w_rev)        # length 3n‑2
    offset = 2 * n - 2
    grad = 2.0 * grad_full[offset - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 1.0 / n)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_sine(n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n‑1))  (zero at the ends, peak in the middle)."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * i / (n - 1.0))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """Explore families + random Dirichlet. Return up to ``max_keep`` best vectors."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.4, 0.9, 101):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential sweep (fine around promising region)
    for lam in np.linspace(0.0005, 0.0055, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta sweep (including the reversed‑triangle shape)
    al_vals = np.linspace(0.2, 4.0, 19)
    be_vals = np.linspace(0.2, 4.0, 19)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # sine family
    try_candidate(generate_sine(n))

    # random Dirichlet draws with various concentrations
    concs = [0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(80):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------


def smoothmax_descent(a0: np.ndarray, deadline: float,
                      beta_start: float = 1.0, beta_end: float = 400.0,
                      beta_mul: float = 1.018,
                      step0: float = 0.12, step_min: float = 2e-5) -> np.ndarray:
    """
    Gradient descent on the smooth approximation of max(conv).
    Temperature β is increased gradually (log‑space) until ``beta_end``.
    Projection onto the probability simplex is applied after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    cur_beta = beta_start
    step = step0
    it = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        # smooth max (for monitoring only)
        smooth_val = smoothmax(conv, cur_beta)

        # gradient of smooth max
        grad = gradient_of_smoothmax(a, cur_beta)

        # simple back‑tracking line search
        improved = False
        for _ in range(5):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = compute_score(a_new)
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                improved = True
                a = a_new
                break
            # if not improved, shrink step
            step = max(step * 0.5, step_min)

        if not improved:
            # even small step did not improve – keep current a
            a = a_new

        # slowly increase temperature
        cur_beta = min(beta_end, cur_beta * beta_mul)

        # slowly increase step size when progress is good
        if improved:
            step = min(step * 1.05, step0)

        # occasional jitter to avoid stagnation
        if it % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(it + 1))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3, final_temp: float = 1e-6,
                        cooling_factor: float = 0.9997,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    """Small mass‑transfer Metropolis hill‑climb with exponential cooling."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a
