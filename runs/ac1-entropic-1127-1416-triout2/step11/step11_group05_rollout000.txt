The algorithm is a two‑stage optimiser

1. **Coarse optimisation** on a modest size (`n₀ = 1024`).  
   – several analytical families are scanned,  
   – a handful of random Dirichlet starts are generated,  
   – each start is refined with L‑BFGS on a smooth approximation of the `max` (log‑sum‑exp) using an increasing schedule of the smoothing parameter `β`.  
   The best coarse distribution is kept.

2. **Fine optimisation** on the target size (`n = 4096`).  
   – the coarse result is up‑sampled by linear interpolation,  
   – symmetry (`a[i] = a[n‑1‑i]`) is enforced,  
   – a second L‑BFGS run with much larger `β` (up to 10⁵) drives the solution towards the true `max`,  
   – a short simulated‑annealing (SA) hill‑climb removes residual asymmetries.

All constraints (non‑negativity, simplex) are satisfied automatically by the soft‑max parametrisation; after every major step we re‑project onto the simplex and re‑symmetrise.  
The routine is written to respect a wall‑clock budget (`budget_s`, default 1000 s) and always returns the best sequence found so far.

```python
"""
Propose a step‑function sequence that minimises evaluate_sequence.

The optimiser works in two stages:
  1️⃣  coarse search on n₀=1024 (analytic families + random restarts,
      L‑BFGS on a smooth max–approximation with a β‑schedule).
  2️⃣  up‑sample to n=4096, enforce symmetry, then a stronger L‑BFGS
      pass (β up to 1e5) followed by a brief simulated‑annealing hill‑climb.

All variables are kept on the simplex via a soft‑max parametrisation;
symmetry and the simplex projection are applied after every major step.
The routine respects a 1000 s wall‑clock budget and always returns the
best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast for many calls)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(a*a) for a on the simplex (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for cheap seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalized."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against infinities caused by log(0)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad_z) where f is a smooth approximation of max(conv)
    and grad_z is the gradient w.r.t. the soft‑max parameters.
    """
    a = softmax(z)                     # a lives on the simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()

    # stable log‑sum‑exp
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta   # smooth approx. of max(conv)

    w = expw / sumexp                  # soft‑max weights over convolution indices

    # gradient of weighted sum of convolutions w.r.t. a
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)          # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]      # shape (n,)

    # back‑propagation through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS driver for a whole β‑schedule
# ----------------------------------------------------------------------


def lbfgs_schedule(a_start: np.ndarray,
                   betas,
                   max_iter_per_stage: int,
                   deadline: float,
                   name: str = "stage"):
    """
    Repeated L‑BFGS optimisation on the smooth max surrogate.
    Returns the best a (on the simplex) and its exact score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            print(f"[{time.time():.1f}] time exhausted before β={beta}")
            break
        print(f"[{time.time():.1f}] L‑BFGS {name} β={beta}")

        # initialise z = log a (soft‑max trick)
        z0 = np.log(a + 1e-15)

        def fun(z):
            return smooth_max_and_grad(z, beta, n)

        # limit internal iterations – we will stop early if time runs out
        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as exc:
            print(f"  L‑BFGS failed at β={beta}: {exc}")
            continue

        a = softmax(res.x)

        # enforce symmetry (helps numerical stability)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Very lightweight SA on the simplex.
    Moves are either a swap or a small mass transfer.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature ≈ 0.1 % of current score
    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # Metropolis criterion
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # increase step size when we are improving
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # slowly shrink step size on rejections
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the simplex constraint.

    The function respects the wall‑clock budget and returns a plain
    Python list of floats (the best sequence found).
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a couple of seconds for final housekeeping
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Fallback – previously discovered sequence (if any)
    # -----------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Coarse optimisation on a modest size
    # -----------------------------------------------------------
    n_small = 1024
    print(f"[{time.time():.1f}] coarse stage (n={n_small})")

    # a) scan analytic families (very cheap)
    for alpha in np.linspace(0.0, 3.0, 13):
        if time.time() >= deadline - 120.0:
            break
        vec = symmetric_product_family(n_small, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

    for beta in np.linspace(0.0, 3.0, 13):
        if time.time() >= deadline - 120.0:
            break
        vec = left_heavy_power_family(n_small, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

    for gamma in np.linspace(0.0, 4.0, 11):
        if time.time() >= deadline - 120.0:
            break
        vec = exp_u_family(n_small, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    # b) random Dirichlet restarts
    n_random = 5
    for rnd in range(n_random):
        if time.time() >= deadline - 120.0:
            break
        # Dirichlet(1) = uniform on simplex
        rnd_vec = np.random.rand(n_small)
        rnd_vec /= rnd_vec.sum()
        # quick L‑BFGS refinement (moderate β)
        betas_coarse = [5, 30, 150, 800]
        a_opt, sc_opt = lbfgs_schedule(rnd_vec,
                                       betas=betas_coarse,
                                       max_iter_per_stage=120,
                                       deadline=deadline - 40.0,
                                       name=f"rnd{rnd}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] random start {rnd} improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 2. Upsample to the target length
    # -----------------------------------------------------------
    n_large = 4096
    if best_seq is None:
        # very unlikely, but keep a safe fallback
        best_seq = [1.0 / n_large] * n_large
        best_score = evaluate_sequence(best_seq)

    a_small = np.array(best_seq, dtype=np.float64)
    # linear interpolation to the larger grid
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = project_to_simplex(a_large)
    a_large = (a_large + a_large[::-1]) * 0.5   # enforce symmetry
    a_large = project_to_simplex(a_large)

    # -----------------------------------------------------------
    # 3. Fine L‑BFGS passes on the full size
    # -----------------------------------------------------------
    if time.time() < deadline - 30.0:
        betas_fine = [500, 2000, 8000, 30000, 100000]
        a_opt, sc_opt = lbfgs_schedule(a_large,
                                       betas=betas_fine,
                                       max_iter_per_stage=200,
                                       deadline=deadline - 10.0,
                                       name="fine")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] fine L‑BFGS best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Final symmetry projection (numerical clean‑up)
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_arr = (a_arr + a_arr[::-1]) * 0.5
        a_arr = project_to_simplex(a_arr)
        sc_sym = compute_score(a_arr)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_arr.tolist()
            print(f"[{time.time():.1f}] symmetry projection improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Short simulated‑annealing polishing
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=2_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final safety check and return
    # -----------------------------------------------------------
    if best_seq is None:
        # last resort: uniform distribution
        best_seq = [1.0 / n_large] * n_large
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
```
