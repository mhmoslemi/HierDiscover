"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

The optimizer works in three phases:

1. **Fast analytic seeding** – dozens of simple families (product, power,
   U‑shape, Beta‑shape) are scanned on a moderately large length (2048 or
   4096) to obtain a good initial candidate.

2. **Multi‑restart L‑BFGS on a smooth‑max surrogate** – the true objective
   ``max(conv)`` is approximated by a log‑sum‑exp with an increasing
   smoothing parameter β.  Each start is optimised with L‑BFGS on the
   soft‑max parametrisation (automatically enforcing non‑negativity and the
   simplex).  A small number of aggressive restarts with random initial
   shapes are performed while respecting the global time budget.

3. **Final simulated‑annealing hill‑climb** – a long‑running SA walk on the
   simplex refines the best sequence found so far.  Symmetrisation is only
   used as a post‑processing step (the optimiser is allowed to discover
   asymmetric optima).

All steps respect the 1000 s wall‑clock limit and continuously keep the
best sequence seen.  The routine returns a plain Python list of floats
ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for fast seeding)
# ----------------------------------------------------------------------


def product_family_alpha(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}; works for negative α as well."""
    i = np.arange(n, dtype=np.float64)
    logw = -alpha * (np.log(i + 0.5) + np.log(n - 0.5 - i))
    # use log‑space to avoid overflow
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    logw = -beta * np.log(i + 0.5)
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logw = -gamma * dist
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+½)^{p‑1}·(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    logw = (p - 1.0) * np.log(i + 0.5) + (q - 1.0) * np.log(n - 0.5 - i)
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where

        f   = smooth max = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z  (gradient w.r.t. the soft‑max parameters)
    """
    a = softmax(z)                          # probability vector
    conv = convolution_fft(a, a)            # length 2n‑1
    maxc = conv.max()
    # stable soft‑max over the convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta

    w = expw / sumexp                       # weights (sum to 1)
    a_rev = a[::-1]
    # convolution of reversed a with the weights
    conv_rev = convolution_fft(a_rev, w)    # length 3n‑2
    # central slice corresponds to gradient w.r.t. a_i
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

    # back‑propagation through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation with a schedule of β values
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float,
                    symmetrize: bool = False):
    """
    Repeated L‑BFGS runs on the smooth‑max surrogate.
    ``betas`` – list of smoothing parameters (increasing).
    ``symmetrize`` – if True, enforce a = (a + a[::-1])/2 after each stage.
    Returns the best sequence (on the simplex) and its exact score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        # initialise the optimisation variables (log‑probabilities)
        z0 = np.log(a + 1e-12)

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        # limit internal L‑BFGS iterations
        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as e:
            print(f"  L‑BFGS failure (β={beta}): {e}")
            continue

        a = softmax(res.x)

        if symmetrize:
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved (β={beta}) → {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a linear temperature schedule.
    50 % swaps, 50 % mass transfers.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature roughly proportional to current best score
    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        temp = T0 * (1 - step / max_steps) + 1e-18
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score

            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (γ={gamma:.2f}) → {best_score:.6f}")

    # Beta‑type shapes (p,q)
    for _ in range(12):
        if time.time() >= deadline - 120.0:
            break
        p = np.random.uniform(0.2, 4.0)
        q = np.random.uniform(0.2, 4.0)
        vec = beta_family(n, p, q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (p={p:.2f},q={q:.2f}) → {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. First L‑BFGS refinement (moderate β, allow asymmetry)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        a_start = np.array(best_seq, dtype=np.float64)
        betas_stage1 = [5, 30, 150, 500, 2000]
        a_opt, sc_opt = l_bfgs_optimize(a_start,
                                        betas=betas_stage1,
                                        max_iter_per_stage=150,
                                        deadline=deadline - 150.0,
                                        symmetrize=False)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after first L‑BFGS best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Multi‑restart quick L‑BFGS passes (random starts)
    # -----------------------------------------------------------
    max_restarts = 18
    restarts = 0
    while time.time() < deadline - 140.0 and restarts < max_restarts:
        a0 = random_start_vector(n)
        betas_quick = [30, 150, 800, 3000]   # sharper, but only a few stages
        a_opt, sc_opt = l_bfgs_optimize(a0,
                                        betas=betas_quick,
                                        max_iter_per_stage=80,
                                        deadline=deadline - 140.0,
                                        symmetrize=False)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] restart #{restarts+1} improved → {best_score:.6f}")
        restarts += 1

    # -----------------------------------------------------------
    # 5. Aggressive L‑BFGS pass on the current best (large β)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        a_start = np.array(best_seq, dtype=np.float64)
        betas_stage2 = [500, 2000, 8000, 20000]
        a_opt, sc_opt = l_bfgs_optimize(a_start,
                                        betas=betas_stage2,
                                        max_iter_per_stage=120,
                                        deadline=deadline - 70.0,
                                        symmetrize=False)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after aggressive L‑BFGS best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Enforce perfect symmetry and re‑project (optional)
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetry projection improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=2_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Fallback in the unlikely event we found nothing
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
