    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2  # base learning rate
    t = 0
    best_a = a0.copy()
    best_score = compute_score(best_a)
    no_improve_iter = 0
    # schedule list of (alpha, steps)
    schedule = [(5, 1500), (15, 1500), (40, 1500), (80, 1500), (150, 1500)]
    for alpha, max_steps in schedule:
        for _ in range(max_steps):
            if time.time() >= deadline:
                return best_a, best_score

            a = np.exp(b - np.max(b))
            a /= a.sum()

"""
Step‑function optimizer – smoothed log‑sum‑exp gradient descent
================================================================

The code searches for a non‑negative sequence `a` (∑a = 1) that minimises

    2·n·max_k (a * a)[k]

where `*` denotes the full linear convolution.  The objective is scale‑invariant,
so we keep the probabilities on the simplex.

Algorithm
---------

1. **Seed generation** – a modest collection of candidates is built from
   simple analytical families (power‑law, geometric) and from random Dirichlet
   draws.  The fallback construction `height_sequence_1` (if present) is also
   added.

2. **Smoothed gradient descent** – we minimise the smooth approximation

        F(a) = (1/α)·log∑_k exp(α·(a*a)[k])

   (log‑sum‑exp of the convolution).  Its gradient is cheap:
   `∇F(a) = (2/α)·Conv(a, softmax(α·(a*a)))` where the softmax is taken over the
   convolution values.  The optimisation is performed on the log‑itself
   (logits) to avoid explicit simplex projection; a standard Adam update
   is used and the logits are periodically clipped to keep the probabilities
   well‑behaved.  A schedule of increasing smoothing parameters α (5 → 150)
   guides the iterates from a very smooth landscape to a near‑exact max‑conv.

3. **Evaluation & best‑track** – every few hundred Adam steps the exact
   score `2·n·max(conv)` is computed; the best vector across all seeds and
   all iterations is stored.

4. **Final stochastic hill‑climb** – the best candidate is polished by a
   cheap greedy mass‑transfer neighbourhood (the same hill‑climber used in the
   reference implementation).  This removes remaining tiny irregularities.

The whole search respects the supplied time budget (default 1000 s).  If no
candidate is found a uniform vector is returned as a safe fallback.

All helper functions are defined at module level – no closures or lambdas.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Convolution utilities (real FFT, full linear convolution)
# -------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2·n‑1
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[: 2 * n - 1]
    return conv


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (length len(a)+len(b)-1)."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 2).bit_length())          # next power of two ≥ n+m‑1
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)[: n + m - 1]
    return conv


# -------------------------------------------------------------
# Exact objective (used for tracking and final reporting)
# -------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact score 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# -------------------------------------------------------------
# Smoothed log‑sum‑exp objective and its gradient w.r.t. a
# -------------------------------------------------------------
def smoothed_obj_and_grad(a: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv) and its gradient.
    Approximation:   F(a) = (1/α)·log(∑ exp(α·conv_k))
    Gradient:       ∇F = (2/α)·Conv(a, softmax(α·conv))
    """
    n = a.shape[0]
    conv = full_convolution(a)                     # length 2·n‑1
    maxc = conv.max()
    # stable softmax over α·conv
    shifted = conv - maxc                           # ≤ 0
    exp_alpha = np.exp(alpha * shifted)           # under‑flow is fine
    Z = exp_alpha.sum()
    softmax = exp_alpha / Z                         # shape (2·n‑1,)

    # convolution a * reversed(softmax)
    w_rev = softmax[::-1]                           # length 2·n‑1
    conv_rev = fft_convolve(a, w_rev)               # length 3·n‑2
    idx = (2 * n - 2) - np.arange(n)                # indices of interest
    grad_a = (2.0 / alpha) * conv_rev[idx]

    # smooth objective value (log‑sum‑exp)
    obj = maxc + np.log(Z) / alpha
    return obj, grad_a


# -------------------------------------------------------------
# Log‑itself representation (logits) → probabilities on the simplex
# -------------------------------------------------------------
def logits_to_probs(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – always yields a vector on the simplex."""
    shifted = logits - np.max(logits)
    exp_shift = np.exp(shifted)
    return exp_shift / exp_shift.sum()


# -------------------------------------------------------------
# Smoothed Adam optimiser starting from a given probability vector
# -------------------------------------------------------------
def gradient_descent_from(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Runs a smoothed log‑sum‑exp Adam optimisation.
    Returns the best probability vector (∑a = 1) found before deadline
    together with its exact score.
    """
    n = a0.shape[0]
    eps = 1e-15
    b = np.log(a0 + eps)               # logits, guarantee positivity
    m = np.zeros_like(b)               # Adam first moment
    v = np.zeros_like(b)               # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.25                           # base learning rate
    t = 0

    best_a = a0.copy()
    best_score = compute_score(best_a)
    no_improve = 0                       # early‑stop counter

    # (α, steps) schedule – start smooth and sharpen progressively
    schedule = [(5, 1500), (15, 1500), (40, 1500), (80, 1500), (150, 1500)]

    for alpha, max_steps in schedule:
        for _ in range(max_steps):
            # time guard
            if time.time() >= deadline:
                return best_a, best_score

            t += 1
            a = logits_to_probs(b)

            # smooth objective & gradient
            _obj, grad_a = smoothed_obj_and_grad(a, alpha)

            # chain rule: ∇_b F = a ⊙ (∇_a F – (a·∇_a F))
            dot = np.dot(a, grad_a)
            grad_b = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_b
            v = beta2 * v + (1.0 - beta2) * (grad_b * grad_b)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr / (np.sqrt(v_hat) + eps_adam)
            b -= step * m_hat

            # keep logits in a reasonable range (prevents overflow)
            b = np.clip(b, -25.0, 25.0)

            # exact evaluation every few hundred steps
            if t % 250 == 0:
                a = logits_to_probs(b)
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                # simple early stop if stagnant
                if no_improve > 4000:
                    return best_a, best_score

    # final projection (already on simplex)
    a = logits_to_probs(b)
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# -------------------------------------------------------------
# Simple greedy hill‑climb (mass transfer neighbourhood)
# -------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """
    Tiny stochastic moves with greedy acceptance (same spirit as the reference).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        if no_improve > 100_000:
            # occasional jitter to escape tiny plateaus
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point required by the harness
# -------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Finds a step‑function (non‑negative vector) with a low value of
    evaluate_sequence().  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for the harness itself
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (if the global variable exists)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Build a modest set of seeds
    # -----------------------------------------------------------------
    n = 2048

    seeds = []

    # power‑law family
    for p in np.linspace(0.30, 0.90, 5):
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        seeds.append(a)

    # geometric decay
    for r in np.linspace(0.60, 0.95, 5):
        a = (r ** np.arange(n, dtype=np.float64))
        a /= a.sum()
        seeds.append(a)

    # random Dirichlet draws (different concentrations)
    for conc in [0.05, 0.2, 0.5, 1.0]:
        a = np.random.dirichlet(np.full(n, conc))
        seeds.append(a)

    # include the best vector found so far (if any)
    if best_seq is not None:
        seeds.append(np.array(best_seq, dtype=float))

    # -----------------------------------------------------------------
    # 2. Optimise each seed with smoothed Adam, track the best
    # -----------------------------------------------------------------
    for idx, a0 in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"[seed {idx}] start smoothed Adam optimisation")
        a_opt, score_opt = gradient_descent_from(a0, deadline)
        print(f"[seed {idx}] smoothed Adam best score = {score_opt:.6f}")
        if score_opt < best_score:
            best_score, best_seq = score_opt, a_opt.tolist()

    # -----------------------------------------------------------------
    # 3. Final polishing with a greedy hill‑climb (if time permits)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[hill‑climb] start final polishing")
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, deadline - 5.0, max_iter=500_000)
        final_score = compute_score(a_final)
        print(f"[hill‑climb] final polished score = {final_score:.6f}")
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – uniform vector if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report the official score and return the list
    # -----------------------------------------------------------------
    reported = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {reported:.6f}")
    return best_seq
