"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

by a combination of fast random initialisation, long Adam optimisation on a
smooth‑max surrogate, cheap sub‑gradient descent, several LP‑refinements and a
final aggressive simulated‑annealing hill‑climb with a richer move set.
The routine respects a global wall‑clock limit (default 1000 s) and always
projects the vector onto the probability simplex, so the objective is
scale‑invariant.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1): 2·n·max(a*a)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the fast initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Monotonically decreasing normalised vector."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Symmetric exponential‑decay shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


def random_sparse_start(n: int, avg_nonzeros: int = 50) -> np.ndarray:
    """Pick a random set of positions and put random mass on them."""
    k = np.random.randint(max(2, avg_nonzeros // 2), avg_nonzeros * 2)
    k = min(k, n)
    idx = np.random.choice(n, size=k, replace=False)
    w = np.random.rand(k)
    a = np.zeros(n)
    a[idx] = w
    return a / a.sum()


def dirichlet_start(n: int, alpha: float = 0.5) -> np.ndarray:
    """Draw from a Dirichlet distribution (Gamma(α) i.i.d.)."""
    w = np.random.gamma(alpha, 1.0, size=n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.014) -> tuple[np.ndarray, float]:
    """Minimise the log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1: len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 1000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheap iterations)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 8_000_000,
                     step0: float = 0.1) -> tuple[np.ndarray, float]:
    """Sub‑gradient method for the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (shift k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diagonal case (i=j) counts only once
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 10_000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement step (with line‑search on mixing weight)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weights: list[float]) -> tuple[np.ndarray, float]:
    """Solve max Σ g_j  s.t. (f * g)_k ≤ M,  g ≥ 0, then mix."""
    n = f.shape[0]
    m_len = 2 * n - 1

    # build convolution matrix A where (A @ g) = f * g
    A = np.zeros((m_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ↔ minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(m_len, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f, compute_score(f)

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, compute_score(f)

    g_norm = g / sum_g

    best_f = f
    best_sc = compute_score(f)
    for t in mix_weights:
        cand = (1.0 - t) * f + t * g_norm
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_f = cand
    return best_f, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style) with richer moves
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a linearly cooled temperature."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 5e-5
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        move_type = np.random.rand()

        if move_type < 0.33:                   # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        elif move_type < 0.66:                  # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        else:                                   # three‑way redistribution
            i, j, k = np.random.randint(n, size=3)
            if i != j and i != k:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                # split delta randomly between j and k
                split = np.random.rand()
                cand[j] += delta * split
                cand[k] += delta * (1.0 - split)
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)   # linear cooling
        if delta < 0.0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The function respects the global time budget and returns a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Load fallback (if present)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working length
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2. Fast analytic / random scan (≈30 s)
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print(f"[{time.time():.1f}] quick scan (n={n})")
        alphas = np.linspace(0.0, 12.0, 13)
        betas = np.linspace(0.0, 12.0, 13)
        gammas = np.linspace(0.0, 6.0, 13)

        for alpha in alphas:
            if time.time() >= deadline - 5.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (sym α={alpha:.2f}) = {best_score:.6f}")

        for beta in betas:
            if time.time() >= deadline - 5.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (left β={beta:.2f}) = {best_score:.6f}")

        for gamma in gammas:
            if time.time() >= deadline - 5.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (exp γ={gamma:.2f}) = {best_score:.6f}")

        # a handful of random starts (including sparse and Dirichlet)
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            r = np.random.rand()
            if r < 0.25:
                vec = random_monotone_start(n)
            elif r < 0.5:
                vec = random_u_start(n)
            elif r < 0.75:
                vec = random_sparse_start(n, avg_nonzeros=60)
            else:
                vec = dirichlet_start(n, alpha=0.4)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise starter to a probability vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. Adam (smooth‑max) optimisation – up to ~25 % of remaining budget
    # ------------------------------------------------------------------
    if time.time() + 70.0 < deadline:
        print(f"[{time.time():.1f}] Adam optimisation start")
        a, sc = adam_smooth_optimize(
            a,
            deadline - 10.0,
            max_iters=600_000,
            beta_start=4.0,
            beta_end=5000.0,
            lr=0.014,
        )
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy sub‑gradient descent – bulk of the remaining budget
    # ------------------------------------------------------------------
    if time.time() + 140.0 < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient descent start")
        a, sc = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=8_000_000,
            step0=0.1,
        )
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑refinement loop (more iterations, finer mixing)
    # ------------------------------------------------------------------
    if time.time() + 40.0 < deadline:
        print(f"[{time.time():.1f}] LP refinement loop start")
        mix_weights = [0.2, 0.1, 0.05, 0.02, 0.01]
        for it in range(20):
            if time.time() >= deadline - 5.0:
                break
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new, sc = lp_refine_step(a, M, mix_weights)
            if sc < best_score - 1e-9:
                best_score, best_seq = sc, a_new.tolist()
                a = a_new
                print(f"[{time.time():.1f}] LP step {it} improved to {best_score:.6f}")
            else:
                # shrink mixing budget when stagnating
                mix_weights = [max(w * 0.6, 0.001) for w in mix_weights]

    # ------------------------------------------------------------------
    # 6. Symmetrisation (often beneficial)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score, best_seq = sc_sym, a_sym.tolist()
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final aggressive hill‑climb (SA) – use all remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a, sc = hill_climb_sa(a, deadline, max_steps=10_000_000)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
