"""
Search for a non‑negative step‑function (list of heights) that minimizes the
provided ``evaluate_sequence`` objective.

The algorithm works on a relatively large grid (default ``n = 8192``) to
approximate the continuous extremal problem.  It proceeds through the
following stages, always respecting the wall‑clock budget (default 1000 s):

1. **Coarse scan of analytic families** – product‑power, left‑heavy power,
   exponential‑U, beta‑type and a uniform+U mixture.  The best vector is kept.
2. **Random exploration of the beta family** – many random (p,q) pairs are
   tried to locate a promising region.
3. **Golden‑section refinement** of the product‑power exponent α around the
   best value found in step 1.
4. **Two Adam‑optimisation passes** on a smooth log‑sum‑exp surrogate of the
   true maximum‐convolution objective.  The iterates are projected onto the
   probability simplex after each batch.
5. **Projected sub‑gradient descent** on the exact max‑objective to polish the
   solution.
6. **Final simulated‑annealing hill‑climb** for a last local improvement.

All vectors are kept on the simplex (∑a = 1) and are symmetrised after each
major phase – a property that is known to be optimal for the autocorrelation
problem.  The routine returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector on the simplex (∑a = 1).
    The value is invariant under scaling, so we can safely assume ∑a = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric product family:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the grid: (i+½)^{p‑1}(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Random initialisers (fallback / exploration)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start: ≈5 % non‑zero entries, uniform mass on them."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sorted powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth log‑sum‑exp surrogate of the max‑objective
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.008,
                         sym_interval: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of ``max(conv)``.

    Parameters
    ----------
    start_a : np.ndarray
        Initial probability vector (∑a = 1, a≥0).
    deadline : float
        Unix timestamp after which the optimisation must stop.
    max_iters : int
        Upper bound on the number of Adam iterations.
    beta_start / beta_end : float
        Exponential schedule for the smooth‑max temperature.
    lr : float
        Learning rate.
    sym_interval : int
        Every ``sym_interval`` steps the iterate is symmetrised and re‑projected.

    Returns
    -------
    best_a, best_score
        Best vector found (as a numpy array) and its exact (non‑smoothed) score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise the log‑soft‑max parameters
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (exponential increase)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            # enforce symmetry and simplex feasibility
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 3000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.04) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for the exact max‑objective.

    Parameters
    ----------
    start_a : np.ndarray
        Starting probability vector.
    deadline : float
        Unix timestamp after which the method must stop.
    max_iters : int
        Upper bound on the number of updates.
    step0 : float
        Initial step size (scaled by 1/√t).

    Returns
    -------
    best_a, best_score
        Best vector and its exact score.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the simplex
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> Tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk on the simplex.

    Moves:
      – 50 % swap two entries,
      – 50 % transfer a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section optimisation for a single scalar parameter
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 15) -> Tuple[float, float]:
    """
    Minimises ``func`` on the interval [a,b] using golden‑section search.
    Returns the best parameter and its function value.
    """
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:          # we minimise
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. optional fallback (previous run)
    # -----------------------------------------------------------
    try:
        fb_vec = globals().get('height_sequence_1')
        if fb_vec:
            fb = evaluate_sequence([float(x) for x in fb_vec])
            print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
            if fb < best_score:
                best_score = fb
                best_seq = [float(x) for x in fb_vec]
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose working length (larger n gives a finer approximation)
    # -----------------------------------------------------------
    n = 8192  # up to ≈2 × previous length while still fast for FFTs

    # -----------------------------------------------------------
    # 2. Coarse scan of analytic families
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] coarse scan (n={n})")
    best_alpha = None
    # product family
    for alpha in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best prod‑α={alpha:.3f} → {best_score:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best left‑β={beta:.3f} → {best_score:.6f}")

    # exponential‑U shape
    for gamma in np.linspace(0.0, 5.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best exp‑γ={gamma:.3f} → {best_score:.6f}")

    # beta family (grid exploration)
    for p in np.linspace(0.1, 3.0, 13):
        if time.time() >= deadline - 30.0:
            break
        for q in np.linspace(0.1, 3.0, 13):
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best beta p={p:.2f},q={q:.2f} → {best_score:.6f}")

    # mixture uniform + U shape
    for _ in range(800):
        if time.time() >= deadline - 30.0:
            break
        lam = np.random.rand()
        gamma = np.random.uniform(0.0, 5.0)
        vec = mixed_uniform_u_family(n, lam, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best mix λ={lam:.3f},γ={gamma:.3f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Random exploration of the beta family (more samples)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] random beta exploration")
    for _ in range(2500):
        if time.time() >= deadline - 30.0:
            break
        p = np.random.uniform(0.05, 5.0)
        q = np.random.uniform(0.05, 5.0)
        vec = beta_family(n, p, q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best random beta p={p:.3f},q={q:.3f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Fine golden‑section optimisation of the product family
    # -----------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product‑α around {best_alpha:.3f}")

        def score_of_alpha(alpha_val: float) -> float:
            return compute_score(symmetric_product_family(n, alpha_val))

        lo = max(0.0, best_alpha - 0.6)
        hi = min(4.0, best_alpha + 0.6)
        best_alpha, best_alpha_score = golden_section_optimize(score_of_alpha,
                                                               lo, hi,
                                                               max_iter=15)
        vec = symmetric_product_family(n, best_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] refined product‑α={best_alpha:.6f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. First Adam (smooth‑max) refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] first Adam refinement (smooth max)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 80.0,
            max_iters=300_000,
            beta_start=10.0,
            beta_end=8000.0,
            lr=0.008,
            sym_interval=2000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Projected sub‑gradient descent (exact max)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 50.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=2_000_000,
            step0=0.04,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Second Adam pass (higher temperature, finer tuning)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] second Adam pass (high β)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 15.0,
            max_iters=200_000,
            beta_start=50.0,
            beta_end=12000.0,
            lr=0.006,
            sym_interval=1500,
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Explicit symmetrisation test (might improve a little)
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 9. Final hill‑climb (simulated annealing)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=5_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 10. Fallback to a safe uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] fallback to uniform, score = {best_score:.6f}")

    # final evaluation via the official API (includes safety checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
