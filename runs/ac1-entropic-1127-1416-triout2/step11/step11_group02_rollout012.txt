def smooth_gradient_descent(a0, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t=0
    beta = beta_start
    while time.time() < deadline:
        # compute conv
        conv = full_convolution(a)
        M = conv.max()
        # compute weights
        shifted = beta*(conv - M)  # <= 0
        exp_shifted = np.exp(shifted)  # stable as <=0
        sum_exp = exp_shifted.sum()
        weights = exp_shifted / sum_exp
        # gradient
        grad = 2.0 * correlation(a, weights)
        # step size schedule
        step = step0 / np.sqrt(t+1)
        # update
        a_new = a - step * grad
        # project
        a_new = project_to_simplex(a_new)
        # evaluate
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        t += 1
        # increase beta gradually
        if beta < beta_end:
            beta = min(beta_end, beta * beta_factor)
    return best_a

Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence(sequence).

Algorithm
---------
1. **Broad parametric sweep** – explore several simple families:
   * power‑law, geometric decay,
   * a flat head followed by a power‑law tail,
   * sparse “spike” vectors (few equal‑weight non‑zeros at random positions).
   The best candidate (lowest 2·n·max(conv)) is kept.

2. **Smooth‑max gradient descent** – minimise a smooth approximation of the
   max‑convolution:
       f₍β₎(a)=max(conv)+ (1/β)·log Σ exp(β·(conv–max(conv))).
   The gradient of the inner weighted sum is obtained by a single FFT‑based
   correlation.  β is increased gradually so that the approximation tightens.
   After each step the iterate is projected onto the probability simplex.

3. **Fine stochastic hill‑climb** – starting from the refined vector,
   repeatedly move a small fraction of mass from one index to another.
   The fraction is reduced progressively (0.02 → 0.001).  Only moves that
   improve the true objective are accepted, giving a deterministic local
   search that can shave off the last few 10⁻⁶ of the score.

All phases respect the overall 1000 s wall‑clock budget (a small safety
margin is kept).  The final sequence is returned as a plain Python list.

import time
import numpy as np

# -------------------------------------------------------------
# Linear‑algebra helpers
# -------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # If all entries are non‑positive the projection is the uniform vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of size a power of two."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector with Σa = 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    return float(2.0 * n * np.max(full_convolution(a)))


def correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Linear correlation (a ⋆ w) for non‑negative vectors.
       Returns c[i] = Σ_j a[j]·w[i+j] for i = 0 … len(a)-1.
    """
    n = a.shape[0]
    m = w.shape[0]
    L = 1 << ((n + m - 2).bit_length())  # ≥ n+m-1
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    R = A * np.conj(W)
    corr = np.fft.irfft(R, L)[:n]
    return corr

        if time.time() >= deadline:
            return best_a or np.full(n, 1.0 / n)
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ---- geometric decay --------------------------------------------
    for r in np.linspace(0.20, 0.99, 180):
        if time.time() >= deadline:
            break
        a = (r ** np.arange(n, dtype=np.float64))
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ---- flat head + power‑law tail ---------------------------------
    for L_frac in np.linspace(0.02, 0.20, 5):
        if time.time() >= deadline:
            break
        L = max(1, int(L_frac * n))
        for p in np.linspace(0.3, 1.2, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head_val = tail.max()
            head = np.full(L, head_val)
            a = np.concatenate([head, tail])
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    # ---- random spike vectors (few equal‑weight atoms) ---------------
    # max theoretical size of a Sidon‑like set is ≈√n; we try up to 256 spikes.
    for m in [2, 4, 8, 16, 32, 64, 128, 256]:
        for _ in range(20):
            if time.time() >= deadline:
                break
            positions = rng.choice(n, size=m, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[positions] = 1.0 / m
            # optional small random perturbation of the equal weights
            perturb = rng.normal(scale=0.05, size=m)
            a[positions] *= np.maximum(0.0, 1.0 + perturb)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    return best_a or np.full(n, 1.0 / n)


# -------------------------------------------------------------
# Phase 2 – smooth‑max gradient descent (log‑sum‑exp)
# -------------------------------------------------------------
def smooth_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Minimise a smooth approximation of max(conv(a,a)) using projected
    gradient descent.  The smoothing parameter β is increased gradually
    from β₀ to βₘₐₓ.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    beta = 5.0          # start with modest smoothing
    beta_max = 200.0
    beta_factor = 1.02
    step0 = 0.6

    n = a.shape[0]

    while time.time() < deadline:
        # ---------------------------------------------------------
        # current convolution and smooth weights
        # ---------------------------------------------------------
        conv = full_convolution(a)
        M = conv.max()

        # weights = softmax of β·(conv – M)  (all ≤ 0)
        shifted = beta * (conv - M)          # ≤ 0, safe exp
        exp_shifted = np.exp(shifted)
        sum_exp = exp_shifted.sum()
        weights = exp_shifted / sum_exp      # length 2n‑1

        # ---------------------------------------------------------
        # gradient of Σ weights·conv = 2·(a ⋆ weights)
        # ---------------------------------------------------------
        grad = 2.0 * correlation(a, weights)

        # ---------------------------------------------------------
        # step size (standard 1/√t schedule) and update
        # ---------------------------------------------------------
        t += 1
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        # ---------------------------------------------------------
        # keep the best iterate seen so far
        # ---------------------------------------------------------
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # increase smoothing parameter slowly
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

    return best_a


# -------------------------------------------------------------
# Phase 3 – fine deterministic hill‑climb (tiny moves)
# -------------------------------------------------------------
def fine_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Deterministic greedy improvement: repeatedly transfer a tiny fraction
    of mass from one coordinate to another.  The fraction is decreased
    step‑wise to allow ever finer adjustments.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    # fractions of the source mass that will be moved; start coarse, finish very fine
    fractions = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    rng = np.random.default_rng(987654)

    for frac in fractions:
        # allocate a generous number of trials for each precision level
        max_trials = 200_000
        trial = 0
        while time.time() < deadline and trial < max_trials:
            i = rng.integers(0, n)
            if a[i] <= 1e-15:
                trial += 1
                continue
            delta = frac * a[i]
            # ensure we have a non‑trivial move
            if delta <= 0.0:
                trial += 1
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1    # guarantee i != j

            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            # due to rounding, enforce non‑negativity explicitly
            if cand[i] < 0:
                cand[i] = 0.0

            cur_score = compute_score(cand)
            if cur_score < best_score:
                best_score, best_a = cur_score, cand.copy()
                a = cand   # move the current point to the improved one
            trial += 1

    return best_a

# Main entry point required by the harness
# -------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety buffer for the wrapper that calls us
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Optional fallback (use a previously discovered good vector)
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    try:
        fallback = list(globals().get('height_sequence_1', []))
        if fallback:
            val = evaluate_sequence(fallback)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global parametric sweep (≈ 200 s)
    # -----------------------------------------------------------------
    phase1_end = min(deadline, start_time + 200.0)
    n = 2048
    if time.time() < phase1_end:
        a_initial = best_initial_vector(n, phase1_end)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[global sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Smooth‑max gradient descent (≈ 400 s)
    # -----------------------------------------------------------------
    phase2_end = min(deadline, time.time() + 400.0)
    if best_seq is not None and time.time() < phase2_end:
        a0 = np.array(best_seq, dtype=float)
        a_refined = smooth_gradient_descent(a0, phase2_end)
        val = compute_score(a_refined)
        print(f"[smooth‑grad] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 3. Fine deterministic hill‑climb (use the remaining budget)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = fine_hill_climb(a_start, deadline)
        val = compute_score(a_final)
        print(f"[fine hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – if nothing succeeded, fall back to uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
