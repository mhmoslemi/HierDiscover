"""
search for a non‑negative step‑function sequence that minimises
evaluate_sequence().  The algorithm combines:

  • coarse scanning of several analytic families (product‑α,
    left‑heavy β, exponential‑U, mixed uniform‑U, asymmetric product p,q);
  • a fine 2‑D grid search over the asymmetric product family (p,q);
  • Adam optimisation of a smooth log‑sum‑exp surrogate of the max‑convolution,
    without any enforced symmetry;
  • projected sub‑gradient descent on the true max‑objective;
  • a short simulated‑annealing hill‑climb.

All vectors are kept on the probability simplex (∑a=1) and the routine
stops before the 1000 s wall‑clock deadline, returning the best
sequence found as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all already normalised to sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and exponential‑U."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, p: float, q: float) -> np.ndarray:
    """Generalised product family a_i ∝ (i+½)^{‑p}·(n‑½‑i)^{‑q}."""
    i = np.arange(n, dtype=np.float64)
    logw = -p * np.log(i + 0.5) - q * np.log(n - 0.5 - i)
    w = np.exp(logw)
    # guard against under‑/overflow
    w[np.isnan(w)] = 0.0
    w /= w.sum()
    return w

# Optimisers
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 5.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.015,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of max(conv).
    No forced symmetry (sym_interval = 0) by default.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derived from smooth‑max surrogate)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 2500 == 0:
            a_tmp = softmax(z)
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> Tuple[np.ndarray, float]:
    """Simple simulated‑annealing walk on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:           # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function sequence minimising
    evaluate_sequence() within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq: List[float] = []
    best_score = float("inf")
    best_params = None  # for asymmetric product family

    # ------------------------------------------------------------------
    # optional fallback using a previously known construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")

            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 2048

    # ------------------------------------------------------------------
    # 1. coarse scan of simple analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse family scan (n={n})")
    # symmetric product family
    for alpha in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_params = ('sym', alpha)
            print(f"[{time.time():.1f}] new best sym‑α={alpha:.3f} → {best_score:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_params = ('left', beta)
            print(f"[{time.time():.1f}] new best left‑β={beta:.3f} → {best_score:.6f}")

    # exponential U shape
    for gamma in np.linspace(0.0, 6.0, 21):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_params = ('expU', gamma)
            print(f"[{time.time():.1f}] new best exp‑γ={gamma:.3f} → {best_score:.6f}")

    # mixture uniform + U
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                best_params = ('mix', lam, gamma)
                print(f"[{time.time():.1f}] new best mix λ={lam:.2f},γ={gamma:.2f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. asymmetric product family (p,q) – coarse 2‑D grid
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] asymmetric product coarse grid")
        p_vals = np.linspace(0.0, 6.0, 61)
        q_vals = np.linspace(0.0, 6.0, 61)
        for p in p_vals:
            for q in q_vals:
                if time.time() >= deadline - 5.0:
                    break
                vec = asymmetric_product_family(n, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    best_params = (p, q)
                    print(f"[{time.time():.1f}] new best asym p={p:.3f} q={q:.3f} → {best_score:.6f}")
            else:
                continue
            break

    # ------------------------------------------------------------------
    # 3. finer grid around the best (p,q) if we have one
    # ------------------------------------------------------------------
    if best_params and isinstance(best_params, tuple) and len(best_params) == 2:
        if time.time() < deadline - 20.0:
            p0, q0 = best_params
            print(f"[{time.time():.1f}] fine grid around p={p0:.3f}, q={q0:.3f}")
            p_fine = np.linspace(max(0.0, p0 - 0.5), p0 + 0.5, 51)
            q_fine = np.linspace(max(0.0, q0 - 0.5), q0 + 0.5, 51)
            for p in p_fine:
                for q in q_fine:
                    if time.time() >= deadline - 5.0:
                        break
                    vec = asymmetric_product_family(n, p, q)
                    sc = compute_score(vec)
                    if sc < best_score:
                        best_score = sc
                        best_seq = vec.tolist()
                        best_params = (p, q)
                        print(f"[{time.time():.1f}] refined asym p={p:.4f} q={q:.4f} → {best_score:.6f}")
                else:
                    continue
                break

    # ------------------------------------------------------------------
    # 4. Adam optimisation (smooth‑max) – no forced symmetry
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] Adam refinement (no symmetry)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 100.0,
            max_iters=600_000,
            beta_start=5.0,
            beta_end=5000.0,
            lr=0.014,
            sym_interval=0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent on true max
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=1_200_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=3_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. fallback to a safe uniform vector if nothing found
    # ------------------------------------------------------------------
    if not best_seq:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # final official evaluation
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
