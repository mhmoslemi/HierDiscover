"""
Multi‑phase optimiser for the step‑function attack.

The search proceeds as follows

1️⃣  Produce a large, diverse set of seed vectors on the probability simplex
    (deterministic windows, beta‑pdfs, Dirichlet draws, sparse spikes, etc.).
2️⃣  Run a soft‑max (log‑sum‑exp) projected gradient descent that minimises a
    smooth surrogate of `max(conv(a,a))`.  The temperature β is annealed from
    5 to a few thousand.
3️⃣  Apply a mirror‑descent / exponentiated‑gradient sub‑gradient method.
    The sub‑gradient of the true max‑objective is `g_i = 2·a_{k‑i}` where `k`
    is the index of the current maximal convolution entry.  The update
    `a ← a·exp(-η·g)` stays in the positive orthant; a final renormalisation
    projects onto the probability simplex.
4️⃣  Finish with a simple stochastic hill‑climb / simulated‑annealing
    exploration that repeatedly transfers a random fraction of mass between
    two coordinates (or swaps two entries).  Only improving moves are kept,
    but occasional uphill moves help escape shallow basins.
5️⃣  The best vector found is returned as a plain Python list; the harness
    will then call `evaluate_sequence` on it.

All phases respect a global time budget (default 1000 s) and stop as soon
as the deadline is reached.  The implementation uses only NumPy and the
standard library, and works with any sequence length ≤ 4096 (the default
is 2048, switched to 4096 if the allocated time is large enough).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1))."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta density evaluated on the centred mesh of {0,…,n‑1}."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration `conc`."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Sparse vector with `k` random non‑zero entries."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def generate_gaussian(n: int, sigma_rel: float = 0.15) -> np.ndarray:
    """Gaussian‑shaped window centred in the middle."""
    i = np.arange(n, dtype=float)
    sigma = sigma_rel * (n - 1)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = w / w.sum()
    return a


def generate_even_spikes(n: int, m: int = 30) -> np.ndarray:
    """`m` equally spaced spikes with equal weight."""
    if m >= n:
        return np.ones(n, dtype=float) / n
    idx = np.linspace(0, n - 1, m, dtype=int)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – soft‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 4000.0,
    step_coef: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        Lβ(a)=log Σ exp(β·conv(a,a))
    using projected gradient descent with annealed temperature β.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score
    it = 0
    start = time.time()
    while time.time() < deadline:
        # evaluate exact score for bookkeeping
        conv = convolution_fft(cur, cur)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # annealed temperature
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights
        shifted = conv - conv.max()          # max(conv) → 0
        exp_vals = np.exp(beta * shifted)   # all ≤ 1, numerically stable
        w = exp_vals / exp_vals.sum()

        # gradient of surrogate
        grad = 2.0 * beta * np.correlate(w, cur, mode="valid")

        # step size (decays with √iteration and with β)
        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        nxt = cur - step * grad
        cur = project_to_simplex(nxt)

        if it % 5000 == 0:
            print(
                f"[soft‑max GD] it {it:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.1f}"
            )
    return best, best_score


# ----------------------------------------------------------------------
# Phase 2 – mirror / exponentiated‑gradient sub‑gradient descent
# ----------------------------------------------------------------------


def mirror_subgradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    eta_start: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Mirror (exponentiated‑gradient) descent on the true max‑objective.
    Update rule: a ← a * exp(‑η·g)  and then renormalise onto the simplex.
    """
    cur = a.copy()
    best = cur.copy()
    best_score = start_score
    it = 0
    n = a.shape[0]

    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        k = int(np.argmax(conv))
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        it += 1
        eta = eta_start / np.sqrt(it)

        # multiplicative update (always stays non‑negative)
        cur = cur * np.exp(-eta * g)

        s = cur.sum()
        if s == 0.0 or not np.isfinite(s):
            cur = random_dirichlet(n, 1.0)
        else:
            cur /= s

        if it % 5000 == 0:
            print(
                f"[mirror sub‑grad] it {it:6d} | best {best_score:.6f}"
            )
    return best, best_score


# ----------------------------------------------------------------------
# Phase 3 – stochastic hill‑climb / simulated‑annealing polishing
# ----------------------------------------------------------------------


def stochastic_hill_climb(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    init_step: float = 0.12,
    min_step: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Very simple stochastic local search.
    Two move types are used:
        * swap two coordinates,
        * transfer a random fraction of mass from i → j.
    Only improving moves are kept; the step magnitude decays with time.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = start_score

    it = 0
    while time.time() < deadline:
        frac = (deadline - time.time()) / (deadline - (deadline - (deadline - time.time())))  # just 1…
        # linear decay of allowed transfer magnitude
        step_scale = max(min_step, init_step * (1.0 - (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - (deadline - time.time())))))

        cand = cur.copy()
        if np.random.rand() < 0.5:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                       # transfer a random fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0 and i != j:
                delta = step_scale * cand[i] * np.random.rand()
                cand[i] -= delta
                cand[j] += delta
                # guard against tiny negatives due to rounding
                cand[i] = max(cand[i], 0.0)

        # renormalise (numeric drift correction)
        total = cand.sum()
        if total == 0.0 or not np.isfinite(total):
            cand = random_dirichlet(n, 1.0)
        else:
            cand /= total

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        # occasional logging
        it += 1
        if it % 20000 == 0:
            print(f"[hill‑climb] it {it:6d} | best {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Choose a reasonable length (larger if we have plenty of time)
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 850.0:
        n = 4096
    print(f"[info] Using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 0 – (optional) load a fallback sequence if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in globals()["height_sequence_1"]]  # type: ignore
        if len(fallback) == n:
            fb = evaluate_sequence(fallback)
            print(f"[fallback] score = {fb:.6f}")
            if fb < best_val:
                best_val = fb
                best_seq = fallback
    except Exception:
        pass  # no fallback available

    # ------------------------------------------------------------------
    # 1 – generate a diverse collection of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic shapes
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # beta pdf families
    for alpha, beta in [
        (0.2, 0.2),
        (0.5, 0.5),
        (1.0, 1.0),
        (2.0, 2.0),
        (5.0, 5.0),
        (0.5, 2.0),
        (2.0, 0.5),
    ]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    # Gaussian windows and evenly‑spaced spikes
    seeds.append(generate_gaussian(n, sigma_rel=0.15))
    seeds.append(generate_gaussian(n, sigma_rel=0.25))
    seeds.append(generate_even_spikes(n, m=30))
    seeds.append(generate_even_spikes(n, m=45))

    # random Dirichlet draws (various concentration)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # very sparse vectors
    for k in (30, 50, 80, 120):
        seeds.append(random_spike(n, k=k))

    # include fallback if it matches length and we have one
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float))

    # ------------------------------------------------------------------
    # 2 – evaluate all seeds, keep the current best
    # ------------------------------------------------------------------
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)          # guard against tiny drift
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] new best → score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – start optimisation from the best seed
    # ------------------------------------------------------------------
    cur_a = np.asarray(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)
    print(f"[init] best seed score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4 – split the remaining budget into phases
    # ------------------------------------------------------------------
    t0 = time.time()
    phase1_deadline = t0 + total_time * 0.30
    phase2_deadline = phase1_deadline + total_time * 0.35
    phase3_deadline = deadline - 0.5   # leave a tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent
    # ------------------------------------------------------------------
    print("\n[Phase 1] soft‑max gradient descent")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=4000.0,
        step_coef=0.001,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after Phase 1] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror (exponentiated‑gradient) sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("\n[Phase 2] mirror sub‑gradient descent")
        cur_a, cur_score = mirror_subgradient_descent(
            cur_a,
            cur_score,
            phase2_deadline,
            eta_start=0.8,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic hill‑climb / simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        print("\n[Phase 3] stochastic hill‑climb polishing")
        cur_a, cur_score = stochastic_hill_climb(
            cur_a,
            cur_score,
            phase3_deadline,
            init_step=0.12,
            min_step=0.001,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 3] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\n[final] evaluate_sequence reports {final_val:.6f}")

    return final_seq
