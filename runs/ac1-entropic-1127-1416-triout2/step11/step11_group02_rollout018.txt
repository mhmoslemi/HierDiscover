def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

"""
Simulated‑annealing optimiser for `evaluate_sequence`.

The objective (2·n·max(conv(a,a)) / (sum(a)²)) is scale‑invariant,
so we work on the probability simplex (Σa = 1, a_i ≥ 0).  The algorithm
keeps a current candidate `a` and repeatedly proposes a tiny mass‑transfer
move a[i] → a[j].  Moves are accepted if they improve the score, otherwise
with a Metropolis probability that depends on a slowly cooling temperature.
Occasional random restarts and a simple projection onto the simplex keep the
search lively.  The routine stops when the wall‑clock budget (default
1000 s) is exhausted and returns the best sequence found.

No external optimisation packages are required – only NumPy and the
provided `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (project onto probability simplex, FFT‑based full conv)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (Σv = 1, v ≥ 0)."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear full convolution a*a using an FFT.  Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())           # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used during the search for a vector whose sum is 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                      budget_s: int = 1000,
                      **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises the
    official ``evaluate_sequence`` score.  A simple simulated‑annealing
    scheme with mass‑transfer moves is used; the best sequence discovered
    before the wall‑clock deadline is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # problem size – the reference runs used 2048 entries
    # -----------------------------------------------------------------
    n = 2048

    # -----------------------------------------------------------------
    # initialise a feasible point (uniform distribution)
    # -----------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    cur_val = compute_score(a)

    # keep the best vector seen so far
    best = a.copy()
    best_val = cur_val

    # -----------------------------------------------------------------
    # optional fallback to a previously discovered good construction
    # -----------------------------------------------------------------
    try:
        fallback = np.array(height_sequence_1, dtype=float)   # type: ignore
        val_f = compute_score(fallback)
        if val_f < best_val:
            best = fallback.copy()
            best_val = val_f
            a = best.copy()
            cur_val = best_val
            print(f"[fallback] used with score = {best_val:.6f}")
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Simulated‑annealing parameters
    # -----------------------------------------------------------------
    T = 0.12               # initial temperature (controls move size)
    T_min = 1e-6
    decay = 0.99995        # geometric decay per iteration
    it = 0

    # -----------------------------------------------------------------
    # Main optimisation loop
    # -----------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- propose a neighbour by moving a small fraction of mass -----
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # the transferable amount scales with the current temperature
        frac = T * rng.random()
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta                     # Σ stays equal to 1, non‑negativity holds

        cand_val = compute_score(cand)

        # ----- acceptance test (Metropolis) -----

            a = cand
            cur_val = cand_val

                best = cand.copy()

    print(f"Reported score (evaluate_sequence) = {final_score:.6f}")

    return final_seq
