def smoothed_gradient_optimize(start_a, deadline, max_iters=1_000_000, eta0=0.8, alpha_start=30.0, alpha_end=300.0, alpha_steps=5):
   a = start_a.copy()
   best_a = a.copy()
   best_score = compute_score(a)
   n = len(a)
   # schedule alpha: we can increase linearly over iterations or use geometric; maybe after each epoch of T_iter = max_iters//alpha_steps increase.
   # Compute iteration count per alpha.
   # We'll compute alpha based on progress: fraction = t / max_iters; alpha = alpha_start + (alpha_end-alpha_start)*fraction**0.5 (square root schedule).
   for t in range(1, max_iters+1):
       if time.time() >= deadline:
           break

       # conv a*a
       conv = convolution_fft(a, a)
       # compute smoothed objective
       # normally we don't need exact value for gradient update, but maybe compute for monitoring.
       if t % 500 == 0:
           exact = compute_score(a)
           if exact < best_score:
               best_score = exact
               best_a = a.copy()
               print(f"Iter {t}: exact_score={exact:.6f}")

       # compute alpha schedule:
       frac = t / max_iters
       alpha = alpha_start + (alpha_end - alpha_start) * (frac ** 0.5)   # gradually increase
       # compute softmax p
       # scale conv for numeric stability
       scaled = alpha * conv
       max_scaled = np.max(scaled)
       exp_scaled = np.exp(scaled - max_scaled)
       p = exp_scaled / exp_scaled.sum()
       # compute correlation(p, a)
       corr = correlation_fft(p, a)   # returns length n
       # gradient
       grad = 4.0 * n * corr
       # step size
       eta = eta0 / np.sqrt(t)
       # EG update
       a = a * np.exp(-eta * grad)
       # renormalize onto simplex
       sum_a = a.sum()
       if sum_a == 0:
           a = np.full_like(a, 1.0 / n)
       else:
           a = a / sum_a

   # final exact evaluation
   final_score = compute_score(a)
   if final_score < best_score:
       best_score = final_score
       best_a = a.copy()
   return best_a, best_score

"""
Improved optimiser for ``evaluate_sequence``.

Main ideas
----------
1. Build a richer family of analytic seeds (uniform, central‑binomial,
   power‑law, geometric, Beta‑shaped, Gaussian, sine‑power) and pick the
   best by an exact evaluation.

2. Optimise the *smooth* approximation of the max‑convolution objective:
      F_α(a) = (2·n/α)·log( Σ_k exp(α·(a∗a)_k) )
   The gradient of F_α is
       ∇F_α = 4·n·corr(p , a)   with  p_k = softmax(α·(a∗a)_k) .
   We perform exponentiated‑gradient (KL‑mirror) descent on the
   probability simplex, slowly annealing the smoothing parameter α
   from a low value (≈30) up to a larger one (≈300).  This gives a
   much more informative descent direction than using the raw sub‑gradient
   of the max.

3. A few cheap random‑restart bursts are added to avoid being trapped in
   a local basin.

4. Finally a light stochastic hill‑climb (swap / fractional mass transfer)
   polishes the solution.

All phases respect a wall‑clock deadline (default 1000 s).  The function
``propose_candidate`` is the entry point required by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1:
        2·n·max(a∗a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v ≥ 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete normal centred at the middle, normalised to sum 1."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return project_to_simplex(a)


def generate_sine_power(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^power, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** power
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Correlation (used for the smooth gradient)
# ----------------------------------------------------------------------


def correlation_fft(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation of p (length 2·n‑1) with a (length n).
    Returns an array of length n, where
        corr[i] = Σ_j p[i+j]·a[j]   (indices out of range give 0).
    """
    a_rev = a[::-1]
    conv = convolution_fft(p, a_rev)                # length 3·n‑2
    # correlation_i = conv[n‑1‑i] for i = 0..n‑1
    return conv[n - 1::-1][:n]


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient optimiser
# ----------------------------------------------------------------------


def smoothed_exp_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.5,
    alpha_start: float = 30.0,
    alpha_end: float = 300.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) descent on the smooth
    approximation
        F_α(a) = (2·n/α)·log Σ_k exp(α·(a∗a)_k).
    ``α`` is annealed from ``alpha_start`` to ``alpha_end``.
    The function returns the best vector encountered and its exact score.
    """
    a = np.asarray(start_a, dtype=np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # --- linear convolution a∗a ---------------------------------
        conv = convolution_fft(a, a)                     # length 2·n‑1

        # --- schedule for smoothing parameter α --------------------
        frac = t / max_iters
        # square‑root schedule gives a gentle early increase
        alpha = alpha_start + (alpha_end - alpha_start) * (frac ** 0.5)

        # --- soft‑max weights p_k ∝ exp(α·conv_k) ------------------
        scaled = alpha * conv
        max_scaled = np.max(scaled)                     # for numerical stability
        exp_scaled = np.exp(scaled - max_scaled)
        p = exp_scaled / exp_scaled.sum()               # p is a probability vector

        # --- gradient:  ∇F = 4·n·corr(p, a) -----------------------
        corr = correlation_fft(p, a)                    # length n
        grad = 4.0 * n * corr

        # --- diminishing stepsise ---------------------------------
        eta = eta0 / np.sqrt(t)

        # --- exponentiated‑gradient update (KL‑mirror) ------------
        # clamp the exponent to avoid overflow/underflow
        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # --- projection back onto the simplex --------------------
        a = np.maximum(a, 1e-15)        # keep strictly positive for numerical stability
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # --- occasional exact evaluation --------------------------
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Iter {t:06d}: exact score = {cur_score:.6f}")

    # final exact check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best sequence encountered.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
