"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex ∑a_i = 1.  The search consists of

*  a quick analytic scan of several parametric families,
*  scalar optimisation of the family parameters (bounded minimise_scalar),
*  a low‑dimensional differential‑evolution search on a spline‑parameterised
   shape (to discover patterns not covered by the analytic families),
*  a smooth‑max Adam refinement (log‑sum‑exp surrogate) with a growing β,
*  a final aggressive simulated‑annealing hill‑climb.

All phases respect the global wall‑clock budget (default 1000 s) and keep the
vector on the probability simplex after each modification.  The best sequence
found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize_scalar, differential_evolution

# ----------------------------------------------------------------------
# Convolution / score utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper families (used for the quick analytic scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric shape ≈ 1/√((i+½)(n‑½‑i))^α."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law a_i ∝ (i+½)^‑β."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential decay from the centre (distance‑based)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max / Adam optimisation of the smoothed surrogate
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.014) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    with Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for β in log‑space
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derivation from the smooth‑max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1: len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (simple but aggressive)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    Temperature is linearly cooled to zero.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 5e-5            # small initial temperature
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)   # linear cooling
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score

    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2. Quick analytic scan (≈20 s)
    # ------------------------------------------------------------------
    if time.time() + 20.0 < deadline:
        print(f"[{time.time():.1f}] quick analytic scan (n={n})")
        for alpha in np.linspace(0.0, 12.0, 13):
            if time.time() >= deadline - 5.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym α={alpha:.2f}) = {best_score:.6f}")

        for beta in np.linspace(0.0, 12.0, 13):
            if time.time() >= deadline - 5.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left β={beta:.2f}) = {best_score:.6f}")

        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 5.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp γ={gamma:.2f}) = {best_score:.6f}")

        # random monotone starts
        for _ in range(40):
            if time.time() >= deadline - 5.0:
                break
            if np.random.rand() < 0.5:
                vec = np.sort(np.random.rand(n) ** np.random.uniform(0.5, 5.0))[::-1]
            else:
                vec = np.exp(-np.random.uniform(0.0, 6.0) * np.minimum(
                    np.arange(n), n - 1 - np.arange(n)))
            vec /= vec.sum()
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine parameters of the three families with scalar optimisation
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print(f"[{time.time():.1f}] scalar optimisation of families")
        # symmetric product family
        def sym_obj(alpha):
            return compute_score(symmetric_product_family(n, alpha))

        res = minimize_scalar(sym_obj, bounds=(0.0, 20.0), method='bounded',
                              options={'xatol': 0.02, 'maxiter': 40})
        if res.success:
            vec = symmetric_product_family(n, res.x)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] sym‑opt α={res.x:.4f} → {best_score:.6f}")

        # left‑heavy power family
        def left_obj(beta):
            return compute_score(left_heavy_power_family(n, beta))

        res = minimize_scalar(left_obj, bounds=(0.0, 12.0), method='bounded',
                              options={'xatol': 0.02, 'maxiter': 40})
        if res.success:
            vec = left_heavy_power_family(n, res.x)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] left‑opt β={res.x:.4f} → {best_score:.6f}")

        # exponential‑U family
        def exp_obj(gamma):
            return compute_score(exp_u_family(n, gamma))

        res = minimize_scalar(exp_obj, bounds=(0.0, 8.0), method='bounded',
                              options={'xatol': 0.02, 'maxiter': 40})
        if res.success:
            vec = exp_u_family(n, res.x)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] exp‑opt γ={res.x:.4f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Normalise starter to a probability vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 5. Low‑dimensional differential‑evolution (≈30 s)
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print(f"[{time.time():.1f}] DE on spline‑parameterised shape")
        # number of control points (≈ n/32, capped)
        m = max(32, min(128, n // 32))
        ctrl_pos = np.linspace(0, n - 1, m)

        bounds = [(-8.0, 2.0)] * m  # log‑weights

        def de_obj(x):
            w = np.exp(x)                     # enforce positivity
            seq = np.interp(np.arange(n), ctrl_pos, w)
            seq = np.maximum(seq, 0.0)
            ss = seq.sum()
            if ss == 0.0:
                seq = np.full(n, 1.0 / n)
            else:
                seq = seq / ss
            return compute_score(seq)

        try:
            result = differential_evolution(
                de_obj,
                bounds,
                maxiter=30,
                popsize=5,
                polish=False,
                callback=lambda x, convergence: time.time() >= deadline,
            )
            de_seq = np.interp(np.arange(n), ctrl_pos, np.exp(result.x))
            de_seq = np.maximum(de_seq, 0.0)
            de_seq /= de_seq.sum()
            de_score = compute_score(de_seq)
            if de_score < best_score:
                best_score = de_score
                best_seq = de_seq.tolist()
                print(f"[{time.time():.1f}] DE improved to {best_score:.6f}")
        except Exception as e:
            print(f"[{time.time():.1f}] DE failed: {e}")

    # ------------------------------------------------------------------
    # 6. Adam optimisation on the smooth‑max surrogate
    # ------------------------------------------------------------------
    if time.time() + 80.0 < deadline:
        print(f"[{time.time():.1f}] Adam refinement start")
        a = np.array(best_seq, dtype=np.float64)
        a = a / a.sum()
        a, sc = adam_smooth_optimize(
            a,
            deadline - 5.0,
            max_iters=400_000,
            beta_start=4.0,
            beta_end=5000.0,
            lr=0.012,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Symmetrisation (often helpful)
    # ------------------------------------------------------------------
    a = np.array(best_seq, dtype=np.float64)
    a = project_to_simplex((a + a[::-1]) * 0.5)
    sc_sym = compute_score(a)
    if sc_sym < best_score:
        best_score = sc_sym
        best_seq = a.tolist()
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final aggressive hill‑climb (SA)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a = np.array(best_seq, dtype=np.float64)
        a, sc = hill_climb_sa(a, deadline, max_steps=5_000_000)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Official evaluation & return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
