"""
Advanced step‑function optimiser for ``evaluate_sequence``.

Algorithm – staged search
-------------------------
1. **Parametric families sweep** – a richer catalogue (power‑law,
   geometric, sinusoidal, cosine window, flat‑head+tail) is explored.
   The best vector (Σ = 1) is kept as a starting point.

2. **Smooth‑max gradient descent** – the non‑smooth objective
   ``max(conv(a,a))`` is replaced by a log‑sum‑exp surrogate.
   Its gradient is obtained analytically via a single FFT‑based
   convolution.  An exponentiated‑gradient (mirror‑descent) step
   on the probability simplex drives the value down.  The smoothing
   parameter β is increased gradually so that the optimisation
   focuses more on the true maximum.

3. **Active‑set mass‑transfer** – at each iteration the set of
   shifts attaining the current maximal convolution is identified.
   The contribution of every coefficient to these active shifts is
   computed (again by FFT).  Mass is moved from the most‑responsible
   index to the least‑responsible one; a simple line‑search on the
   transferred amount is performed.  This dramatically reduces the
   L∞‑norm of the convolution while keeping Σ = 1.

4. **Final stochastic hill‑climb** – a cheap greedy neighbour search
   cleans the remaining local basin.

All phases respect the remaining wall‑clock budget (default 1000 s).
The returned Python list contains non‑negative floats; the harness
will invoke ``evaluate_sequence`` on it.
"""

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, v.shape[0] + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())


# ----------------------------------------------------------------------
# Phase 1 – expanded parametric sweep
# ----------------------------------------------------------------------
def best_initial_vector(n: int, deadline: float) -> np.ndarray:
    """Explore several simple families and return the best one."""
    best_a = None
    best_val = float("inf")
    # helper to try a candidate
    def try_vec(a):
        nonlocal best_a, best_val
        if time.time() >= deadline:
            return
        if a.sum() == 0:
            return
        a = a / a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ----- power‑law ---------------------------------------------------
    for p in np.linspace(0.30, 1.00, 71):
        if time.time() >= deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        try_vec(a)

    # ----- geometric ---------------------------------------------------
    for r in np.linspace(0.50, 0.99, 81):
        if time.time() >= deadline:
            break
        a = (r ** np.arange(n, dtype=np.float64))
        try_vec(a)

    # ----- sinusoidal (sin^p) -----------------------------------------
    x = np.arange(n, dtype=np.float64) + 0.5
    base = np.sin(np.pi * x / n)
    for p in np.linspace(0.5, 5.0, 10):
        if time.time() >= deadline:
            break
        a = base ** p
        try_vec(a)

    # ----- cosine window (1 - cos) ^ p ---------------------------------
    base = 1.0 - np.cos(2.0 * np.pi * np.arange(n, dtype=np.float64) / (n - 1))
    for p in np.linspace(0.1, 3.0, 15):
        if time.time() >= deadline:
            break
        a = base ** p
        try_vec(a)

    # ----- flat head + power‑law tail ---------------------------------
    for L_frac in np.linspace(0.05, 0.30, 6):
        if time.time() >= deadline:
            break
        L = max(1, int(L_frac * n))
        for p in np.linspace(0.4, 0.9, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head = np.full(L, tail.max())
            a = np.concatenate([head, tail])
            try_vec(a)

    # ----- triangular shape (single peak) ------------------------------
    peak = np.arange(1, n // 2 + 2)
    if n % 2 == 0:
        tri = np.concatenate([peak, peak[-2::-1]])
    else:
        tri = np.concatenate([peak, peak[-1::-1]])
    try_vec(tri.astype(float))

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max gradient descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Compute a smooth approximation of max(conv) and its gradient.
    Returns (smooth_val, grad) where grad has the same shape as a.
    """
    conv = full_convolution(a)               # length 2n-1
    max_c = conv.max()
    # log‑sum‑exp (stable)
    exps = np.exp(beta * (conv - max_c))     # shift by max_c for stability
    sum_exps = exps.sum()
    weights = exps / sum_exps                # softmax over shifts
    smooth_val = max_c + np.log(sum_exps) / beta

    # gradient w.r.t a: 2 * (a_rev * weights)[n-1 : n-1+n]
    a_rev = a[::-1]
    # convolution length = n + (2n-1) - 1 = 3n-2
    nfft = next_pow2(3 * a.shape[0] - 2)
    A_rev_fft = np.fft.rfft(a_rev, n=nfft)
    W_fft = np.fft.rfft(weights, n=nfft)
    conv_rev = np.fft.irfft(A_rev_fft * W_fft, n=nfft)
    start = a.shape[0] - 1
    grad = 2.0 * conv_rev[start:start + a.shape[0]]
    return smooth_val, grad


def gradient_descent_smooth(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the smooth‑max surrogate.
    The iterate is always kept on the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    beta = 1.0
    it = 0
    while time.time() < deadline:
        it += 1
        # compute smooth max and gradient
        smooth_val, grad = smooth_max_and_grad(a, beta)

        # adapt stepsize (simple diminishing schedule)
        eta = 0.5 / np.sqrt(it)

        # exponentiated‑gradient update, protect against overflow
        upd = np.exp(np.clip(-eta * grad, -20.0, 20.0))
        a = a * upd
        a = project_to_simplex(a)

        # evaluate true objective
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # slowly increase the smoothing parameter
        if it % 30 == 0:
            beta = min(beta * 1.5, 120.0)

    return best_a


# ----------------------------------------------------------------------
# Phase 3 – active‑set mass‑transfer refinement
# ----------------------------------------------------------------------
def active_move_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly move a small amount of mass from the coefficient that
    contributes most to the current active constraints to the one that
    contributes least.  Each candidate move is accepted only if it
    reduces the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    step_scale = 0.2
    it = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # build mask of active shifts
        mask = np.zeros_like(conv)
        mask[active] = 1.0

        # contributions = 2 * (a_rev * mask)[n-1 : n-1+n]
        a_rev = a[::-1]
        nfft = next_pow2(3 * n - 2)
        A_rev_fft = np.fft.rfft(a_rev, n=nfft)
        M_fft = np.fft.rfft(mask, n=nfft)
        conv_rev = np.fft.irfft(A_rev_fft * M_fft, n=nfft)
        contrib = 2.0 * conv_rev[n - 1:n - 1 + n]

        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))

        # no useful direction?
        if contrib[i_max] <= contrib[i_min] + 1e-15:
            break

        # propose a transfer
        delta = min(step_scale * a[i_max], a[i_max] * 0.5)
        if delta <= 1e-15:
            break
        cand = a.copy()
        cand[i_max] -= delta
        cand[i_min] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.93  # shrink step on failure

        # occasional jitter to escape tiny plateaus
        if it % 500 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Phase 4 – tiny stochastic hill‑climb (greedy acceptance)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Greedy local search that moves a fraction of mass between two indices."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        if no_improve > 100_000:
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Choose a reasonable length – larger n gives finer granularity.
    # Empirically 2048 is a good compromise for the 1000 s budget.
    # -----------------------------------------------------------------
    n = 2048

    # -----------------------------------------------------------------
    # 1. Global parametric sweep (≈200 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 550.0
    if time.time() < sweep_deadline:
        a_initial = best_initial_vector(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[global sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Smooth‑max gradient descent (≈250 s)
    # -----------------------------------------------------------------
    grad_deadline = deadline - 250.0
    if best_seq is not None and time.time() < grad_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_grad = gradient_descent_smooth(a0, grad_deadline)
        val = compute_score(a_grad)
        print(f"[smooth‑grad] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_grad.tolist()

    # -----------------------------------------------------------------
    # 3. Active‑set mass‑transfer refinement (≈150 s)
    # -----------------------------------------------------------------
    active_deadline = deadline - 80.0
    if best_seq is not None and time.time() < active_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_active = active_move_refine(a0, active_deadline)
        val = compute_score(a_active)
        print(f"[active‑move] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_active.tolist()

    # -----------------------------------------------------------------
    # 4. Final stochastic hill‑climb (remaining budget)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
