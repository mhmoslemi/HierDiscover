def smooth_exponential_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    eta0: float = 0.8,
    beta_start: float = 1.0,
    beta_end: float = 80.0,
) -> tuple[np.ndarray, float]:

    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    while True:
        if time.time() >= time_deadline:
            break
        t += 1
        # exponent schedule
        # linear increase: beta = beta_start + (beta_end - beta_start) * min(1.0, t / max_iters_est)
        # we need estimate of max_iters_est ~ (time_left * it_per_sec). But unknown. Use an approximate constant, maybe 200000.
        # Simpler: beta = beta_start + (beta_end - beta_start) * (t / (t + 10000))  # gradually increase.
        # Could use tanh function: beta = beta_end * (1 - np.exp(-t/5000)) maybe.
        # We'll use simple schedule: beta = min(beta_end, beta_start + t / 2000)   # increase by 1 every 2000 iterations.
        beta = beta_start + t / 2000.0
        if beta > beta_end:
            beta = beta_end

        # conv
        conv = convolution_fft(a, a)
        # compute weights w = softmax(beta*conv)
        # for numerical stability subtract max
        max_beta_conv = np.max(beta * conv)
        w = np.exp(beta * conv - max_beta_conv)
        w_sum = w.sum()
        w /= w_sum

        # compute correlation g = Σ_j w_{i+j} a_j via convolution with reversed w
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        # get g entries
        idx_arr = np.arange(n)
        # conv_rev length = n + (2n-1) - 1 = 3n-2
        # shift = len(w_rev) - 1 = 2n-2
        shift = len(w_rev) - 1
        g = conv_rev[shift - idx_arr]  # shape n

        # grad a
        grad_a = 4.0 * n * g

        # avg
        avg = np.dot(a, grad_a)

        # step size
        eta = eta0 / np.sqrt(t)

        exp_arg = -eta * (grad_a - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # renorm, avoid zero
        sum_a = a.sum()
        if sum_a <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # occasional progress check
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

"""
non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. Build a small portfolio of analytic seeds (uniform, geometric,
   power‑law, binomial, beta‑shaped).  Pick the best few by the exact
   objective ``2·n·max(conv(a,a))``.

2. Refine the best seed with a *smooth* KL‑mirror (exponentiated‑gradient)
   descent.  The non‑smooth max‑operator is replaced by a soft‑max
   (log‑sum‑exp) with a temperature β that is slowly increased from 1 to
   ≈80.  This yields a dense sub‑gradient that guides the optimiser past
   plateaus encountered by the hard‑max version.

3. Finish with a cheap stochastic hill‑climb (swap / fractional mass
   moves) to squeeze out the last bits of improvement.

All phases respect a wall‑clock deadline (default 1000 s) and keep the
best sequence seen.  The returned list already satisfies the simplex
constraint (∑a_i = 1, a_i ≥ 0) and can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Heavy‑tailed (U‑shaped) for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth KL‑mirror optimiser (exponentiated‑gradient with soft‑max)
# ----------------------------------------------------------------------


def smooth_exponential_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    eta0: float = 0.8,
    beta_start: float = 1.0,
    beta_end: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on the smoothed objective

        F(a) = 2·n·log‑sum‑exp(β·conv(a,a)) / β ,

    with β gradually increased from ``beta_start`` to ``beta_end``.
    The update is

        a_i ← a_i·exp[ -η (∂F/∂a_i – ⟨a,∂F/∂a⟩) ]

    where the gradient ∂F/∂a_i = 4·n·∑_j w_{i+j}·a_j and
    w = softmax(β·conv).
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    # we will increase β by roughly 1 every 2000 iterations
    beta_step = 2000.0

    while time.time() < time_deadline:
        t += 1

        # ---- temperature schedule -------------------------------------------------
        beta = beta_start + t / beta_step
        if beta > beta_end:
            beta = beta_end

        # ---- convolution and soft‑max weights --------------------------------------
        conv = convolution_fft(a, a)                     # length 2n‑1
        # soft‑max (log‑sum‑exp) – numerically stable
        weighted = beta * conv
        max_w = np.max(weighted)
        w = np.exp(weighted - max_w)
        w_sum = w.sum()
        w /= w_sum                                        # probability vector, length 2n‑1

        # ---- gradient of the smoothed maximum ------------------------------------
        # g_i = Σ_j w_{i+j} a_j  (cross‑correlation)
        w_rev = w[::-1]                                   # reverse for correlation
        conv_rev = convolution_fft(a, w_rev)              # length 3n‑2
        shift = len(w_rev) - 1                            # = 2n‑2
        idx = np.arange(n)
        g = conv_rev[shift - idx]                         # shape (n,)

        grad_a = 4.0 * n * g                               # ∂F/∂a

        # ---- KL‑mirror (exponentiated‑gradient) update ----------------------------
        avg = np.dot(a, grad_a)                            # ⟨a,∂F/∂a⟩
        eta = eta0 / np.sqrt(t)
        exp_arg = -eta * (grad_a - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)            # keep exponent in safe range
        a = a * np.exp(exp_arg)

        # ---- projection back to the simplex ---------------------------------------
        sum_a = a.sum()
        if sum_a <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # ---- occasional logging ---------------------------------------------------
        if t % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[exp‑grad] iter {t:07d} – β={beta:.2f} – score={cur_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
