"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm
---------
0. Load a fallback sequence if the global ``height_sequence_1`` exists.
1. Quick stochastic scan – generate many candidates from analytic families,
   random Dirichlet draws and monotone starts; keep the best.
2. Exponentiated‑gradient descent on the true max‑convolution objective.
   Gradient is taken from all (or the top) convolution entries that attain
   the current maximum; the update is a multiplicative mirror‑descent
   step followed by normalisation (intrinsically stays on the simplex).
3. A handful of LP‑based refinements (same as in the reference code) –
   they solve ``max Σ g_j`` s.t. ``(a*g)_k ≤ M`` and mix the optimum
   with the current vector via a line‑search.
4. Aggressive simulated‑annealing hill‑climb that moves mass between two
   random coordinates (or swaps them) and accepts uphill moves with a
   linearly decreasing temperature.
5. Optional symmetrisation of the final vector.
6. Return the best sequence found (plain Python list of floats).

All phases respect the global wall‑clock budget (default 1000 s) and
project onto the probability simplex whenever necessary.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast for many calls)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic / random families (used for the quick scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-0.5-i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+0.5)^{-β} (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing shape (sorted powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_dirichlet_start(n: int, concentration: float) -> np.ndarray:
    """Sample from Dirichlet(α,…,α) with shared concentration."""
    # Gamma(α,1) for each coordinate
    g = np.random.gamma(concentration, 1.0, size=n)
    a = g / g.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the L∞‑convolution objective
# ----------------------------------------------------------------------


def eg_descent(start_a: np.ndarray,
               deadline: float,
               max_iters: int = 500_000,
               eta0: float = 0.8,
               tol_rel: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on
        f(a) = max_k (a * a)_k
    The update is a_i ← a_i·exp(‑η·g_i) followed by normalisation.
    """
    n = start_a.size
    a = start_a.copy()
    # ensure strictly positive to avoid zero‑mass drift
    a = np.maximum(a, 1e-20)
    a /= a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()

        # indices that are (near) maximal – a tiny relative tolerance
        idxs = np.where(conv >= max_val * (1.0 - tol_rel))[0]
        if idxs.size == 0:
            idxs = np.array([int(np.argmax(conv))])

        # gradient of each selected max entry
        g = np.zeros_like(a)
        for k in idxs:
            # g_i += 2·a_{k‑i} whenever the partner index lies in [0,n)
            i = np.arange(n)
            j = k - i
            mask = (j >= 0) & (j < n)
            if np.any(mask):
                g[mask] += 2.0 * a[j[mask]]
        # average over the active set
        g /= idxs.size

        # exponentiated‑gradient update
        eta = eta0 / math.sqrt(t)
        # protect from overflow/underflow inside the exponential
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        a = a * np.exp(delta)

        # renormalise (simple L1‑projection onto the simplex)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f} (iter {t})")

    # final check
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (same as reference)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weights: list[float]) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M  ∀k,   g ≥ 0.
    Return the best convex mixture (1‑t)·f + t·g_norm over the supplied
    mixing weights.
    """
    n = f.shape[0]
    m_len = 2 * n - 1
    # Build convolution matrix A (A @ g = f * g)
    A = np.zeros((m_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ↔ minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(m_len, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f, compute_score(f)

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, compute_score(f)
    g_norm = g / sum_g

    best_f = f
    best_sc = compute_score(f)
    for t in mix_weights:
        cand = (1.0 - t) * f + t * g_norm
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_f = cand
    return best_f, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (aggressive)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 4_000_000) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex.  Moves:
        * swap two entries,
        * transfer a fraction of mass between two positions,
        * add tiny Gaussian jitter to all entries (occasionally).
    Temperature is linearly cooled to zero.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature: a small fraction of the score
    T0 = cur_score * 4e-5
    step_scale = 0.13

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # occasionally (every 1000 steps) perform a tiny Gaussian jitter
        if step % 1000 == 0 and np.random.rand() < 0.05:
            cand = cur_a + np.random.normal(scale=step_scale * 0.05, size=n)
            cand = np.maximum(cand, 0.0)
        else:
            cand = cur_a.copy()
            if np.random.rand() < 0.55:                     # transfer mass
                i, j = np.random.randint(n, size=2)
                if i != j and cand[i] > 1e-12:
                    delta = step_scale * cand[i] * np.random.rand()
                    cand[i] -= delta
                    cand[j] += delta
            else:                                            # swap
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]

        # re‑normalise onto the simplex
        s = cand.sum()
        if s == 0.0:
            cand = np.full(n, 1.0 / n)
        else:
            cand = cand / s

        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)   # linear cooling
        if delta < 0.0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase the typical step size when we improve
                step_scale = min(step_scale * 1.08, 0.5)
        else:
            # slowly shrink step size on rejection
            step_scale *= 0.97

    return best_a, best_score
