def adam_smoothing(a, start_score, deadline):
    n = a.size
    cur = a.copy()
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0
    best = cur.copy()
    best_score = start_score
    beta_start = 5.0
    beta_end = 2000.0
    lr = 0.01

    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0*n*float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # compute beta
        elapsed = time.time() - start
        fraction = elapsed/(deadline-start)
        beta = beta_start + fraction*(beta_end-beta_start)

        # shifted conv for numerics
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        grad = 2.0 * beta * np.correlate(w, cur, mode='valid')
        t += 1
        m = 0.9 * m + (1 - 0.9) * grad
        v = 0.999 * v + (1 - 0.999) * (grad ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        nxt = cur - step
        cur = project_to_simplex(nxt)
    return best, best_score

"""
Advanced optimiser for the step‑function problem.

The algorithm works in four stages:
  1.  A large pool of seeds (deterministic, Beta‑pdf, polynomial,
      random Dirichlet, sparse spikes) is generated; the best seed
      (according to the exact objective) is kept.
  2.  Adam optimisation on a smooth “soft‑max” surrogate of the max‑norm
      (log‑sum‑exp of the auto‑convolution).  The temperature β is
      annealed from 5 to 2000 while the Adam learning‑rate is kept
      modest.  After each update the vector is projected onto the
      probability simplex.
  3.  Variance‑flattening descent (gradient of the L2 variance of the
      auto‑convolution) using Adam – this flattens the whole convolution
      profile.
  4.  A classic sub‑gradient descent on the true max‑objective with a
      momentum term, followed by a stochastic hill‑climb polishing.

All phases are time‑boxed; the remaining time after phase 4 is spent
in hill‑climb moves that are kept only when they improve the exact
score.  The final sequence is returned as a plain Python list and is
evaluated with the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# -------------------------------------------------------------
# FFT‑based convolution utilities
# -------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation of vectors ``e`` (length 2n‑1) and ``a`` (length n):
    (e ⋆ a)[i] = Σ_m e[i+m]·a[m]   for i = 0,…,n‑1 .
    Implemented via a single FFT convolution with a reversed copy of ``a``.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]


# -------------------------------------------------------------
# Simplex projection
# -------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v <= 0):
        # Edge case: all non‑positive → return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# -------------------------------------------------------------
# Seed generators
# -------------------------------------------------------------

def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    return a / a.sum()


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    if w.sum() == 0:
        w = np.ones_like(w)
    return w / w.sum()


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on the discrete set {0,…,n‑1}."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    if w.sum() == 0:
        w = np.ones_like(w)
    return w / w.sum()


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    return a / a.sum()


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: random weights on `k` distinct indices."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    return a / a.sum()


# -------------------------------------------------------------
# Objective utilities
# -------------------------------------------------------------

def exact_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (Σa = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# -------------------------------------------------------------
# Optimisation phases
# -------------------------------------------------------------

def adam_softmax(a: np.ndarray, start_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        Lβ(a)=log Σ exp(β·conv(a,a)).
    The temperature β is linearly annealed from 5 to 2000.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta_start, beta_end = 5.0, 2000.0
    lr = 0.015
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    start = time.time()

    while time.time() < deadline:
        # current exact score – keep track of the best ever seen
        cur_score = exact_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # annealed temperature
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # compute soft‑max weights
        conv = convolution_fft(cur, cur)
        max_c = conv.max()
        shifted = conv - max_c            # prevents overflow
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of Lβ  (see analysis)
        grad = 2.0 * beta * correlation_valid(w, cur)

        # Adam update (minimise → move opposite to gradient)
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        nxt = cur - step
        cur = project_to_simplex(nxt)

        # occasional progress report
        if it % 4000 == 0:
            print(f"[adam‑softmax] it {it:6d} | time {time.time() - start:6.1f}s | best {best_score:.6f}")

    return best, best_score


def adam_variance(a: np.ndarray, start_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the variance of the auto‑convolution:
        V(a)=∑_k (conv(k)-avg)² ,   avg = 1/(2n‑1).
    This flattens the whole convolution profile.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    lr = 0.03
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    start = time.time()

    while time.time() < deadline:
        cur_score = exact_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        conv = convolution_fft(cur, cur)
        e = conv - avg
        grad = 4.0 * correlation_valid(e, cur)

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        nxt = cur - step
        cur = project_to_simplex(nxt)

        if it % 3000 == 0:
            print(f"[adam‑var ] it {it:6d} | time {time.time() - start:6.1f}s | best {best_score:.6f}")

    return best, best_score


def subgradient_momentum(a: np.ndarray, start_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective,
    enhanced with a simple momentum term.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    velocity = np.zeros_like(cur)
    mu = 0.85            # momentum coefficient
    step0 = 0.12         # initial step scale

    it = 0
    start = time.time()

    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))

        # sub‑gradient at the worst index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        it += 1
        step = step0 / np.sqrt(it)

        # momentum update
        velocity = mu * velocity + step * g
        cur -= velocity
        cur = project_to_simplex(cur)

        cur_score = exact_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        if it % 2000 == 0:
            # tiny random jitter to escape plateaus
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(it))
            cur += jitter
            cur = project_to_simplex(cur)

    return best, best_score


def hill_climb(a: np.ndarray, deadline: float, max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that only accepts moves improving the exact
    max‑convolution score.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = exact_score(best)
    step_scale = 0.09

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()

        # two elementary move types
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()          # correct any drift
        cur_score = exact_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)   # slowly enlarge steps on success
        else:
            step_scale *= 0.95                         # shrink on failure

        steps += 1

    return best, best_score
