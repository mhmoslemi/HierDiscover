def hill_climb_targeted(start_a, deadline, max_steps=2_000_000):
    n = start_a.size
    cur_a = start_a.copy()
    # compute initial conv and score
    conv = convolution_fft(cur_a, cur_a)
    cur_score = 2.0 * n * float(conv.max())
    best_a = cur_a.copy()
    best_score = cur_score
    step_scale = 0.1
    for step in range(1, max_steps+1):
        if time.time() >= deadline:
            break
        # recompute conv?
        conv = convolution_fft(cur_a, cur_a)
        max_idx = int(np.argmax(conv))
        # compute contributions
        # j = max_idx - i
        i_arr = np.arange(n)
        j_arr = max_idx - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = cur_a[i_arr[mask]] * cur_a[j_arr[mask]]
        # choose i according to contribution distribution (if all zero fallback uniform)
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i = np.random.choice(n, p=prob)
        else:
            i = np.random.randint(n)
        # choose j uniformly from all not equal to i
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # transfer delta
        delta = step_scale * cur_a[i]
        if delta <= 0:
            continue
        # adjust
        cur_a[i] -= delta
        cur_a[j] += delta
        # ensure nonnegativity
        if cur_a[i] < 0:
            cur_a[i] = 0.0
        # sum unchanged approx; reproject onto simplex to correct any rounding
        cur_a = project_to_simplex(cur_a)
        # compute new conv and score
        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        # occasional progress print
        if step % 50000 == 0:
            print(f"[{time.time():.1f}] hill climb step {step}, best {best_score:.6f}")
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

The search combines:
   • A fast scan over several analytic families (including a flexible
     asymmetric beta‑shape, Gaussian‑like, cosine‑power and polynomial‑power).
   • Multiple short Adam runs on a smooth‑max surrogate, followed by a
     longer Adam refinement on the best start.
   • A projected sub‑gradient descent (true max‑objective) for robust fine
     tuning.
   • A directed simulated‑annealing hill‑climb that preferentially moves
     mass away from indices that contribute most to the current maximal
     convolution value.
   • Symmetrisation after the main phases (the optimal shape is
     symmetric, and averaging with its reverse never harms the objective).

All phases respect a global wall‑clock budget (default 1000 s).  The code
works for any length ≤ ~5000 (no LP is used, so larger n is fine); the
default length is 3000 unless a fallback sequence is available.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # only needed for the fallback import


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (real‑valued inputs)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-0.5-i))^{-α} (symmetric beta‑like shape)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def asymmetric_beta_family(n: int, a: float, b: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-a} * (n-0.5-i)^{-b}, a,b>0."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-a * np.log(i + 0.5) - b * np.log(n - 0.5 - i))
    a_vec = w / w.sum()
    return a_vec


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-β} (mass concentrated on the left)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """w_i ∝ exp(-γ·dist(i, centre)) with dist = min(i,n‑1‑i)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = w / w.sum()
    return a


def cosine_pow_family(n: int, power: float) -> np.ndarray:
    """w_i ∝ (cos(π·i/(n‑1)) + 1)^{power} (non‑negative, symmetric)."""
    i = np.arange(n, dtype=np.float64)
    w = (np.cos(np.pi * i / (n - 1)) + 1.0) ** power
    a = w / w.sum()
    return a


def polynomial_pow_family(n: int, power: float) -> np.ndarray:
    """w_i ∝ max(0, 1 - ((i‑centre)/(n/2))**2)^{power}."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    radius = n / 2.0
    x = (i - centre) / radius
    base = 1.0 - x * x
    base = np.clip(base, 0.0, None)
    w = base ** power
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """A random decreasing (non‑increasing) vector, normalised."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Randomised exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1: len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the genuine max‑objective.
    The step size decays as step0 / sqrt(t) and we project onto the simplex.
    """
    n = start_a.shape[0]
    a = np.clip(start_a, 0.0, 1.0).astype(np.float64).copy()
    a = a / a.sum()                        # enforce the simplex

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the max‑entry (shift k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal case where i==j should have factor 1
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb_targeted(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex that preferentially shifts mass away from
    indices contributing most to the current maximal convolution entry.
    """
    n = start_a.size
    cur_a = start_a.copy()
    conv = convolution_fft(cur_a, cur_a)
    cur_score = 2.0 * n * float(conv.max())
    best_a = cur_a.copy()
    best_score = cur_score

    step_scale = 0.12   # fraction of the source entry that is moved

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # locate the index giving the current max
        k_max = int(np.argmax(conv))

        # contributions of each i to that max (a_i * a_{k_max-i})
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = cur_a[i_arr[mask]] * cur_a[j_arr[mask]]

        # pick a source index weighted by its contribution (fallback uniform)
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            src = np.random.choice(n, p=prob)
        else:
            src = np.random.randint(n)

        # pick a destination (any other index)
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # move a fraction of mass from src to dst
        delta = step_scale * cur_a[src]
        if delta > 0:
            cur_a[src] -= delta
            cur_a[dst] += delta
            # projection onto simplex (keeps sum exactly 1)
            cur_a = project_to_simplex(cur_a)

        # recompute convolution and score for the new vector
        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
            # slowly increase the move size when we are improving
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # otherwise shrink it a bit
            step_scale *= 0.97

        if step % 50000 == 0:
            print(f"[{time.time():.1f}] hill‑climb step {step}, best = {best_score:.6f}")

    return best_a, best_score

    a = project_to_simplex(a)
    sc_sym2 = compute_score(a)
    if sc_sym2 < best_score:
        best_score = sc_sym2
        best_seq = a.tolist()
        print(f"[{time.time():.1f}] second symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final directed hill‑climb (use up remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (targeted SA) start")
        a, sc = hill_climb_targeted(a, deadline, max_steps=6_000_000)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
