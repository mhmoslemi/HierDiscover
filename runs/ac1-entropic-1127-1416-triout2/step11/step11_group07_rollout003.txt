"""
Advanced step‑function optimiser (new version).

Goal
----
Minimise `evaluate_sequence` by searching over non‑negative vectors that sum to a
constant (the objective is scale‑invariant).  The optimisation proceeds in four
phases:

1.  Diverse seeds → keep the best.
2.  Smooth surrogate minimisation (soft‑max of the autocorrelation).
3.  Variance‑flattening gradient descent (flatten the whole convolution).
4.  Polyak step sub‑gradient descent on the true max‑convolution.
5.  Simulated‑annealing / hill‑climb polishing.

All updates are projected onto the probability simplex so the sum stays equal
to 1, which makes the objective simply `2·n·max(conv)`.  The code respects the
global time budget (default 1000 s) and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑ a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # Uniform distribution is the projection of a non‑positive vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape a_i ∝ min(i+1, n-i)."""
    i = np.arange(n, dtype=float)
    w = np.minimum(i + 1.0, n - i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Correlation helper (used in sub‑gradient phases)
# ----------------------------------------------------------------------


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation (length = len(a)):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    Implemented via one FFT‑convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Phase 1 – soft‑max gradient descent (smooth surrogate of max)
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step_coef: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate Lβ(a)=log Σ exp(β·conv(a,a))
    using projected gradient descent with an annealed temperature β.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    it = 0
    start = time.time()
    while time.time() < deadline:
        # exact score for bookkeeping
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # annealed temperature
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights (stable formulation)
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)  # all ≤ 1
        w = exp_vals / exp_vals.sum()

        # gradient of the surrogate (see derivation in the original code)
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # step size (decays with √iteration and with β)
        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        # projected update
        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 5000 == 0:
            print(
                f"[soft‑max GD] it {it:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.1f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – variance‑flattening gradient descent
# ----------------------------------------------------------------------


def variance_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Minimise variance of the autocorrelation (flattens the whole profile).
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        e = conv - avg
        grad = 4.0 * correlation_valid(e, cur_a)

        it += 1
        step = step0 / np.sqrt(it)

        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 2000 == 0:
            print(
                f"[var‑GD] it {it:6d} | time {time.time():6.1f}s | best {best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Polyak step sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def polyak_subgradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective using a
    Polyak step size: step = (f(x) - f★) / ||g||²,
    where f★ is the trivial lower bound 1/(2n‑1).
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    f_star = 1.0 / (2 * n - 1)          # lower bound on max convolution (average)

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        M = conv.max()
        cur_score = 2.0 * n * float(M)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # mask of maximal indices (use a small tolerance)
        mask = np.isclose(conv, M, atol=1e-12)
        e = mask.astype(float)

        # sub‑gradient of max(conv) = 2 * (e ⋆ a)
        grad = 2.0 * correlation_valid(e, cur_a)

        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq == 0.0:
            break

        # Polyak step (clamped to avoid overshooting)
        step = (M - f_star) / g_norm_sq
        step = max(step, 0.0)          # safety
        step = min(step, 0.2)          # cap for stability

        cur_a = cur_a - step * grad
        cur_a = project_to_simplex(cur_a)

        # occasional jitter to escape flat regions
        it += 1
        if it % 5000 == 0:
            cur_a += np.random.randn(n) * (step * 0.05)
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – simulated‑annealing / hill‑climb polishing
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a: np.ndarray,
    deadline: float,
    init_temp: float = 0.02,
    cooling: float = 0.99995,
    min_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a Metropolis acceptance rule.
    Moves consist of swaps or small mass transfers.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    cur_a = best_a.copy()
    cur_score = best_score

    temp = init_temp
    it = 0
    while time.time() < deadline:
        cand = cur_a.copy()

        if np.random.rand() < 0.5:                       # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = cand[i] * np.random.rand() * 0.1
                cand[i] -= delta
                cand[j] += delta

        # re‑project to the simplex to eliminate numerical drift
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        # acceptance
        if cand_score < cur_score:
            cur_a, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
        else:
            # Metropolis criterion
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                cur_a, cur_score = cand, cand_score

        # temperature decay
        temp = max(temp * cooling, min_temp)

        it += 1
        if it % 20000 == 0:
            print(
                f"[anneal] iter {it:7d} | best {best_score:.6f} | temp {temp:.2e}"
            )
    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Reserve a small safety margin for the harness.
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Choose a working length (larger gives more expressive power).
    # ------------------------------------------------------------------
    n = 4096 if total_time > 800.0 else 2048
    print(f"[setup] sequence length = {n}")

    # ------------------------------------------------------------------
    # 0 – optionally start from a known good construction.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        if len(fallback) != n:
            # linear interpolation to the requested length
            idx_src = np.arange(len(fallback))
            idx_tgt = np.linspace(0, len(fallback) - 1, n)
            fb = np.interp(idx_tgt, idx_src, fallback)
            fb = np.maximum(fb, 0.0)
            fb /= fb.sum()
            fallback = fb.tolist()
        fb_val = evaluate_sequence(fallback)
        print(f"[fallback] initial score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1 – generate a diverse set of seeds and keep the best one.
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))
    for alpha, beta in [
        (0.2, 0.2),
        (0.5, 0.5),
        (1.0, 1.0),
        (2.0, 2.0),
        (5.0, 5.0),
        (0.5, 2.0),
        (2.0, 0.5),
    ]:
        seeds.append(generate_beta_pdf(n, alpha, beta))
    # random Dirichlet draws
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))
    # very sparse spikes – often a good starting point
    for k in (30, 50, 80, 120):
        seeds.append(random_spike(n, k=k))
    # symmetric triangular shape
    seeds.append(generate_triangular(n))

    # possibly include the fallback (already normalised)
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float))

    # evaluate all seeds (exact objective) and keep the best
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] improvement → score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Initialise optimisation from the best seed.
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)
    print(f"[init] start score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Allocate time for the four optimisation phases.
    # ------------------------------------------------------------------
    t0 = time.time()
    # Rough 25 % each, the last phase gets the remainder.
    phase1_dead = t0 + total_time * 0.25
    phase2_dead = phase1_dead + total_time * 0.25
    phase3_dead = phase2_dead + total_time * 0.25
    # phase 4 runs until the overall deadline (minus a tiny safety margin).

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent (smooth surrogate).
    # ------------------------------------------------------------------
    if time.time() < phase1_dead:
        print("\n[Phase 1] soft‑max gradient descent")
        cur_a, cur_score = softmax_gradient_descent(
            cur_a,
            cur_score,
            phase1_dead,
            beta_start=5.0,
            beta_end=3000.0,
            step_coef=0.0012,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – variance‑flattening gradient descent.
    # ------------------------------------------------------------------
    if time.time() < phase2_dead:
        print("\n[Phase 2] variance‑flattening gradient descent")
        cur_a, cur_score = variance_gradient_descent(
            cur_a,
            cur_score,
            phase2_dead,
            step0=0.045,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Polyak step sub‑gradient descent on the true max.
    # ------------------------------------------------------------------
    if time.time() < phase3_dead:
        print("\n[Phase 3] Polyak sub‑gradient descent")
        cur_a, cur_score = polyak_subgradient_descent(
            cur_a,
            cur_score,
            phase3_dead,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 3] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing / hill‑climb polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 4] annealing hill‑climb polishing")
        # initialise temperature based on the current gap to the trivial lower bound
        init_temp = max((cur_score - 1.0) * 0.02, 0.01)
        cur_a, cur_score = annealing_hill_climb(
            cur_a,
            deadline,
            init_temp=init_temp,
            cooling=0.99994,
            min_temp=1e-7,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 4] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final conversion and official evaluation.
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
