"""
Non‑negative step‑function optimiser.

Core idea
---------
* Initialise many analytic and random simplexes (uniform, power‑law,
  geometric, Beta‑shaped, Dirichlet samples).
* Refine each candidate with a *soft‑max* smoothed version of the max‑
  convolution objective.  The smoothed objective
      F_λ(a) = (1/λ)·log Σ exp(λ·(a*a)_k)
  is differentiable; we use an exponentiated‑gradient (KL‑mirror) update
  a_i ← a_i·exp(‑η·∇F_λ(a)_i) and re‑normalise onto the probability
  simplex.  λ is increased gradually (continuation) so that the method
  first follows a smooth landscape and finally concentrates on the true
  maximum.
* Perform several random‑restart bursts, each limited by the remaining
  wall‑clock time.
* Finish with a cheap stochastic hill‑climb (swap / mass‑move) to polish
  the best solution.
* Return the final sequence as a plain list (already on the simplex);
  ``evaluate_sequence`` will compute the official score.

The optimiser stays fully on the simplex, respects the 0‑≤‑value
constraints and never produces numbers >1000 (the sum is 1, so each entry
≤1).  All heavy work is done with FFT‑based convolutions (O(n log n))
so that many thousands of iterations fit easily into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]          # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed mirror‑descent optimiser (soft‑max of the max‑convolution)
# ----------------------------------------------------------------------


def smooth_mirror_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.6,
    lambda0: float = 12.0,
    lambda_factor: float = 1.018,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth approximation

        F_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)

    The method runs until ``time_deadline`` or ``max_iters`` is hit.
    Returns the best vector (on the simplex) and its exact score.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    lam = lambda0

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # 1) self‑convolution
        conv = convolution_fft(a, a)            # length 2n‑1

        # 2) soft‑max probabilities p_k ∝ exp(λ·conv_k)
        w = lam * conv
        w_max = np.max(w)
        exp_w = np.exp(w - w_max)               # stable
        p = exp_w / exp_w.sum()                 # shape (2n‑1,)

        # 3) gradient ∇F_λ(a) = 2·Σ_k p_k·a_{k‑i}  (i‑wise)
        #    realized as convolution of a with reversed p
        p_rev = p[::-1]
        grad_full = convolution_fft(a, p_rev)   # length 3n‑2
        grad = 2.0 * grad_full[::-1][:n]        # keep first n entries

        # 4) exponentiated‑gradient (KL‑mirror) step
        eta = eta0 / np.sqrt(t)
        # clip gradient to avoid overflow in the exponent
        grad_clip = np.clip(grad, -100.0, 100.0)
        a = a * np.exp(-eta * grad_clip)

        # 5) renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # periodic progress check
        if t % 800 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # increase λ (continuation)
        lam = min(lam * lambda_factor, 5_000.0)

    # one final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps: int = 300_000) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

    # Beta‑shaped family (α in (0.05,0.6])
    for alpha in np.linspace(0.05, 0.60, 12):
        init_candidates.append(generate_beta_shape(n, float(alpha)))

    # possibly include the previously found best construction
    try:
        init_candidates.append(np.array(height_sequence_1, dtype=np.float64))
        print("Added provided height_sequence_1 as a starter.")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Choose the best starter by exact evaluation
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Best starter score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Random‑restart smoothed mirror descent
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    descent_deadline = time.time() + 0.70 * time_left   # ~70 % of remaining budget
    n_restarts = 6

    for r in range(n_restarts):
        if time.time() >= descent_deadline:
            break

        # Dirichlet‑like random start with a random concentration
        alpha = 10 ** np.random.uniform(-1.0, 0.0)   # α ∈ [0.1, 1.0]
        g = np.random.gamma(alpha, 1.0, size=n)
        rand_vec = g / g.sum()

        # Blend a bit with the current best to stay in a promising region
        mixed = 0.7 * best_seq + 0.3 * rand_vec
        mixed = project_to_simplex(mixed)

        # Determine how many iterations we can afford for this burst
        remaining = descent_deadline - time.time()
        max_iters = int(min(150_000, remaining * 1300))

        print(f"Restart {r}: α={alpha:.3f}, iterations={max_iters}")

        a_opt, opt_score = smooth_mirror_optimize(
            mixed,
            time_deadline=descent_deadline,
            max_iters=max_iters,
            eta0=0.6,
            lambda0=12.0,
            lambda_factor=1.018,
        )
        if opt_score < best_val:
            best_val, best_seq = opt_score, a_opt.copy()
            print(f"   Improved to {best_val:.6f} after restart {r}")

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb (polishing)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 300_000
        print(f"Final hill‑climb: up to {steps} steps")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val, best_seq = hill_score, np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Safeguard – fall back to uniform if something went wrong
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 6. Final reporting using the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
