"""
Optimiser for the “step‑function” challenge.

The algorithm combines several complementary phases:

0.  Build a diverse pool of seed vectors (deterministic sin² shape,
    Beta‑family shapes, Dirichlet draws, sparse spikes, and the fallback
    sequence if available).  All seeds are normalised to a probability
    vector (sum = 1) because the evaluation is scale‑invariant.

1.  Exponential‑gradient descent on a smooth log‑sum‑exp surrogate of the
    max‑convolution.  The surrogate temperature β is annealed from a modest
    value (≈ 5) up to a very large one (≈ 2·10⁴).  An exponentiated‑gradient
    update keeps the iterate on the probability simplex without the need
    for an explicit Euclidean projection.

2.  Classic sub‑gradient descent on the true max‑objective.  This refines
    the solution obtained by phase 1 and quickly eliminates residual
    peaks.

3.  Targeted stochastic hill‑climbing.  At each step the index k with the
    current maximal convolution value is located; the contribution of
    each coefficient a_i to that peak (g_i = 2·a_{k‑i}) is used to move a
    small fraction of mass from “high‑impact’’ entries to “low‑impact’’
    entries.  Random swaps and jitter are added to escape shallow local
    minima.  The routine runs until the time budget is exhausted.

Time is split adaptively (≈45 % / 30 % / remaining) and the sequence length
is chosen as 2048 or 4096 depending on the total time available.  The
final vector is returned as a plain Python list; it is automatically
valid for `evaluate_sequence` because all entries are non‑negative and
their sum is ≥ 0.01.

The goal is to push the score below the previous best (≈ 1.511) and
approach the target 1.503 within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (Σa = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a

# Phase 1 – exponential‑gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 20000.0,
    base_eta: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        Lβ(a) = log Σ exp(β·conv(a,a))
    using an exponentiated‑gradient update that stays on the simplex.
    β is annealed linearly from `beta_start` to `beta_end`.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = compute_score(cur)

    start = time.time()
    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur, cur)
        cur_max = conv.max()
        cur_score = 2.0 * n * float(cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # annealed temperature
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max weights (stable formulation)
        shifted = conv - cur_max               # max → 0
        exp_args = beta * shifted
        # avoid overflow/underflow
        exp_args = np.clip(exp_args, -700, 0)
        w = np.exp(exp_args)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.exp(np.clip(exp_args, -700, 0))
            w_sum = w.sum()
        w /= w_sum

        # gradient of the surrogate (valid correlation)
        grad = 2.0 * beta * np.correlate(w, cur, mode="valid")

        # exponentiated‑gradient step
        it += 1
        eta = base_eta / np.sqrt(it)
        log_adj = -eta * grad
        # prevent overflow in exp
        log_adj -= log_adj.max()
        cur = cur * np.exp(log_adj)

        # renormalise to the simplex
        s = cur.sum()
        if s <= 0 or np.isnan(s):
            cur = np.full_like(cur, 1.0 / n)
        else:
            cur = cur / s

        if it % 2000 == 0:
            print(
                f"[exp‑GD] it {it:5d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.0f}"
            )
    return best, best_score


# ----------------------------------------------------------------------
# Phase 2 – sub‑gradient refinement on the true max‑objective
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            # occasional jitter to escape flat regions
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – targeted stochastic hill‑climbing
# ----------------------------------------------------------------------


def hill_climb_targeted(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic moves that explicitly try to reduce the current maximal
    convolution value.  At each step the peak index is located, the
    contribution of each coefficient to that peak is estimated,
    and a small fraction of mass is transferred from a high‑impact
    coefficient to a low‑impact one.  Random swaps and jitter are
    interleaved to maintain exploration.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.02
    steps = 0
    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(np.argmax(conv))

        # contribution to the peak (gradient of the peak w.r.t. a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k_max - idx_i
        contrib = np.zeros(n)
        contrib[idx_i] = 2.0 * best_a[idx_j]

        # pick a source index weighted by its contribution (high → more likely)
        src_candidates = np.where(best_a > 1e-12)[0]
        if src_candidates.size == 0:
            src = np.random.randint(n)
        else:
            src_weights = contrib[src_candidates] + 1e-16
            src_weights = src_weights / src_weights.sum()
            src = np.random.choice(src_candidates, p=src_weights)

        # pick a target index weighted opposite to contribution (low → more likely)
        tgt_weights = (contrib.max() - contrib) + 1e-16
        tgt_weights = tgt_weights / tgt_weights.sum()
        tgt = np.random.choice(n, p=tgt_weights)

        if src == tgt:
            steps += 1
            continue

        delta = step_scale * best_a[src]
        delta = min(delta, best_a[src])          # never make source negative
        cand = best_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # re‑normalise (protect against numerical drift)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0 or not np.isfinite(cand_sum):
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / cand_sum

        cand_score = compute_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.3)   # be a bit bolder
        else:
            # occasional random swap regardless of improvement
            if np.random.rand() < 0.08:
                i, j = np.random.randint(n, size=2)
                cand2 = best_a.copy()
                cand2[i], cand2[j] = cand2[j], cand2[i]
                cand2 = cand2 / cand2.sum()
                sc2 = compute_score(cand2)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = cand2
                    step_scale = min(step_scale * 1.07, 0.3)

            step_scale *= 0.96

        # occasional jitter to inject noise
        if steps % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            jitter_cand = best_a + jitter
            jitter_cand = np.maximum(jitter_cand, 0.0)
            jitter_cand = jitter_cand / jitter_cand.sum()
            j_score = compute_score(jitter_cand)
            if j_score < best_score:
                best_score = j_score
                best_a = jitter_cand
                step_scale = min(step_scale * 1.05, 0.3)

        steps += 1
        if steps % 20000 == 0:
            print(f"[hill‑climb] step {steps} best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0       # safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Choose a modest length (2048 or 4096)
    # ------------------------------------------------------------------
    n = 4096 if total_time > 800.0 else 2048
    print(f"[info] Using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 1 – generate a diverse set of seed vectors
    # ------------------------------------------------------------------
    seeds = []

    # deterministic sin² shape
    seeds.append(generate_deterministic(n))

    # symmetric polynomial family (wide p range)
    for p in np.linspace(-4.5, 4.5, 13):
        seeds.append(generate_beta_family(n, p))

    # extra edge‑heavy shapes
    for p in (-6.0, -8.0):
        seeds.append(generate_beta_family(n, p))

    # Beta‑pdf extremes
    for α, β in [(0.05, 0.05), (0.1, 0.1), (0.2, 0.8), (0.8, 0.2),
                 (0.1, 5.0), (5.0, 0.1), (0.5, 0.5), (2.0, 2.0)]:
        seeds.append(generate_beta_pdf(n, α, β))

    # Dirichlet draws with various concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc))

    # Very sparse spikes – often a good starting point
    for k in (20, 40, 80, 120):
        seeds.append(random_spike(n, k=k))

    # ------------------------------------------------------------------
    # 0 – optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        if len(fallback) == n:
            fb_vec = np.asarray(fallback, dtype=float)
            fb_vec = fb_vec / fb_vec.sum()
            seeds.append(fb_vec)
            print("[info] Fallback sequence added to seed pool")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Evaluate all seeds and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = s / s.sum()
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.copy()
            print(f"[seed] improvement → score = {best_val:.6f}")

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    cur_a = best_seq.copy()
    cur_score = compute_score(cur_a)
    print(f"[info] Initial best score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – allocate time for the three optimisation phases
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now

    # Phase 1 – exponential‑gradient descent (≈45 % of remaining)
    phase1_end = now + remaining * 0.45
    print("\n[Phase 1] Exponential‑gradient descent")
    cur_a, cur_score = exp_gradient_descent(
        cur_a,
        phase1_end,
        beta_start=5.0,
        beta_end=20000.0,
        base_eta=0.3,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after Phase 1] best score = {best_val:.6f}")

    # Phase 2 – sub‑gradient refinement (≈30 % of remaining)
    now = time.time()
    if now < deadline:
        phase2_end = now + (deadline - now) * 0.30
        print("\n[Phase 2] Sub‑gradient refinement")
        cur_a, cur_score = subgradient_refine(
            cur_a,
            cur_score,
            phase2_end,
            max_iter=600_000,
            step0=0.07,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # Phase 3 – targeted hill‑climb (use whatever time is left)
    now = time.time()
    if now < deadline - 0.5:
        print("\n[Phase 3] Targeted hill‑climb")
        cur_a, cur_score = hill_climb_targeted(cur_a, deadline, max_steps=5_000_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after hill‑climb] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\n[final] reported score = {final_val:.6f}")

    return final_seq
