def optimise_smooth_max(initial_a, deadline, gamma_start=5.0, gamma_end=200.0, total_iters=None):
    a = initial_a.copy()
    # ensure positivity and normalization
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.time()
    it = 0
    while time.time() < deadline and (total_iters is None or it < total_iters):
        it+=1
        # compute conv and softmax weights
        conv = convolution_fft(a, a)
        # gradually increase gamma
        if total_iters is not None:
            gamma = gamma_start + (gamma_end - gamma_start) * (it-1)/total_iters
        else:
            gamma = gamma_start
        # stable softmax
        max_c = conv.max()
        exp_vals = np.exp(gamma * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n-1:n-1+len(a)]

        # Adam updates
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

"""
Hybrid optimisation with a smooth‑max surrogate and Adam updates.

The algorithm improves on the previous exponentiated‑gradient approach
by replacing the hard max in the objective with a temperature‑controlled
soft‑max (log‑sum‑exp).  The resulting objective is smooth, allowing the
use of an Adam‑style adaptive step‑size.  The temperature (γ) is increased
gradually (continuation) so the optimisation first sees a smoothed
landscape and finally converges to the true max‑convolution objective.

For each considered length n (powers of two) a small library of diverse
initial shapes (arcsine, beta‑like, Dirichlet draws, perturbed arcsine) is
generated.  The best few are refined with the Adam optimiser; the globally
best solution is polished with a longer run and a higher final temperature.
All vectors are kept on the probability simplex (non‑negative and sum = 1),
which respects the scale‑invariance of the official score.

The routine respects a wall‑clock budget (default 1000 s) and always
returns a feasible list of floats before the time expires.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    Returns the full ``len(a)+len(b)-1`` result.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    a is assumed non‑negative; works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (onto {x≥0, sum x = 1})
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the algorithm of [Wang & Carreira-Perpinán 2013].
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the largest rho satisfying the KKT condition
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical clean‑up
    w = w / max(w.sum(), 1e-12)
    return w


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws, normalised)."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny random perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Stable soft‑max of γ·conv, returns a probability vector of same length."""
    max_c = conv.max()
    # subtract max for numerical stability
    exp_vals = np.exp(gamma * (conv - max_c))
    total = exp_vals.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approximated max‑convolution.
    Using the identity:
        ∂/∂a_t Σ_k w_k (a∗a)_k = 2 Σ_j w_{t+j} a_j .
    Implemented via a single convolution (a_rev * w) and slicing.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reverse for the correlation trick
    conv_rev = convolution_fft(a_rev, w)   # length = n + (2n‑1) − 1 = 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]  # slice of length n
    return grad


def optimise_adam_smooth_max(
    initial_a: np.ndarray,
    deadline: float,
    *,
    gamma_start: float = 5.0,
    gamma_end: float = 200.0,
    max_iter: int = 150_000,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector seen (projected onto the simplex) and its
    official ``compute_score`` value.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()                     # initialise on the simplex
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution and soft‑max weights
        conv = convolution_fft(a, a)
        # linear schedule for the temperature γ
        if max_iter > 1:
            gamma = gamma_start + (gamma_end - gamma_start) * (it - 1) / (max_iter - 1)
        else:
            gamma = gamma_start
        w = _softmax_weights(conv, gamma)

        # gradient of the smooth objective
        grad = _gradient_smooth_max(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # parameter update (gradient descent)
        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project back onto the probability simplex
        a = project_simplex(a)

        # track best according to the true (non‑smooth) score
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional logging
        if it % 2000 == 0:
            print(
                f"    Adam iter={it:,d} γ={gamma:.1f} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc
