"""
Improved search for a low‑max‑convolution step sequence.

The algorithm proceeds in several stages:

1.  **Seed generation** – a diverse pool of candidate vectors is created
    for a few lengths (512, 1024, 2048).  Shapes include the classic arcsine,
    generalized‑power (β‑family), random Beta‑distributions, block‑mass,
    sparse‑random spikes and mixtures with the uniform vector.  Every
    candidate is projected onto the probability simplex (∑a = 1) and scored
    with a fast surrogate 2·n·max(conv(a,a)).

2.  **Linear‑programming refinement** –  given a seed `f` we compute
    `M = max(conv(f,f))`.  The LP

        max   Σ b_j
        s.t.  conv(f,b) ≤ M    (component‑wise)
              b ≥ 0

    (the convolution term is linear in `b`) returns a direction `b`
    that can increase the total mass while respecting the current worst‑case
    convolution bound.  After normalising `b` we mix it with `f` using the
    analytically optimal convex combination parameter `t∈[0,1]` that minimises
    the resulting maximum convolution.  The procedure is repeated a few
    times (typically 3–4) until no further progress is observed.

3.  **Exact sub‑gradient descent** – with the refined vector we run a
    diminishing‑step sub‑gradient scheme on the true objective
    `2·n·max(conv(a,a))` (the sum is fixed to 1).  This quickly irons out
    residual peaks.

4.  **Optional Adam polishing** – a short Adam optimisation on a
    temperature‑controlled soft‑max surrogate is performed if time permits.

5.  **Official evaluation** – the best vector seen so far is evaluated with
    the provided `evaluate_sequence`.  After all lengths have been processed
    a final global polishing (sub‑gradient + Adam) is applied to the current
    best vector.

All steps respect the global time budget (default 1000 s) and guarantee a
non‑empty non‑negative result.  The use of SciPy’s sparse LP solver keeps
the linear‑programming stage memory‑efficient even for `n = 2048`.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be on the probability simplex,
    therefore the denominator disappears and the objective is
        2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators (used for the seed pool)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classical arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_block_shape(n: int, frac: float) -> np.ndarray:
    """
    Two symmetric blocks at the ends, each carrying half the total mass.
    ``frac`` is the fraction of positions occupied by each block.
    """
    block_len = max(1, int(frac * n))
    seq = np.zeros(n, dtype=float)
    val = 0.5 / block_len
    seq[:block_len] = val
    seq[-block_len:] = val
    return seq


def generate_sparse_random_shape(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """
    ``k`` random non‑zero entries, then normalised to sum = 1.
    """
    seq = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=min(k, n), replace=False)
    weights = rng.random(idx.size)
    seq[idx] = weights
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)
    return seq


# ----------------------------------------------------------------------
# Linear‑programming refinement utilities
# ----------------------------------------------------------------------
def build_convolution_matrix(f: np.ndarray):
    """
    Build a sparse matrix A such that A @ b = conv(f, b) (linear convolution).
    The matrix has shape (2*n‑1, n) and only non‑negative entries.
    """
    n = f.shape[0]
    L = 2 * n - 1
    offsets = np.arange(n, dtype=int)                # row – col = d, d = 0..n‑1
    # each diagonal d contains the constant f[d] on (L‑d) rows
    data = [np.full(L - d, f[d], dtype=float) for d in offsets]
    A = _sp.diags(data, offsets, shape=(L, n), format='csr')
    return A


def solve_lp_for_b(f: np.ndarray, M: float, deadline: float):
    """
    Solve   max Σ b_j   s.t.   conv(f, b) ≤ M,   b ≥ 0 .
    Returns the optimal vector b (un‑scaled) or ``None`` on failure.
    """
    if _opt is None or _sp is None:                     # pragma: no cover
        return None
    if time.time() > deadline - 5.0:
        return None
    A = build_convolution_matrix(f)                     # sparse matrix
    b_ub = np.full(A.shape[0], M, dtype=float)
    c = -np.ones(f.shape[0], dtype=float)              # minimise –Σb  → maximise Σb
    bounds = [(0, None)] * f.shape[0]
    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:                              # pragma: no cover
        print(f"LP exception: {e}")
        return None
    if not res.success:
        # a failure is not fatal – we simply skip the LP step
        return None
    return res.x


def optimal_mix(f: np.ndarray, b: np.ndarray):
    """
    Find the convex combination t∈[0,1] that minimises
        max_k [(1‑t)²·conv(f,f) + 2t(1‑t)·conv(f,b) + t²·conv(b,b)] .
    Returns the mixed vector and the achieved maximal convolution.
    """
    conv_ff = convolution_fft(f, f)
    conv_fb = convolution_fft(f, b)
    conv_bb = convolution_fft(b, b)

    # coarse grid + local refinement
    best_t = 0.0
    best_max = np.inf
    for t in np.linspace(0.0, 1.0, 21):
        cur = (1 - t) ** 2 * conv_ff + 2 * t * (1 - t) * conv_fb + t ** 2 * conv_bb
        m = cur.max()
        if m < best_max:
            best_max = m
            best_t = t
    # finer search around the best coarse point
    lo, hi = max(0.0, best_t - 0.05), min(1.0, best_t + 0.05)
    for t in np.linspace(lo, hi, 21):
        cur = (1 - t) ** 2 * conv_ff + 2 * t * (1 - t) * conv_fb + t ** 2 * conv_bb
        m = cur.max()
        if m < best_max:
            best_max = m
            best_t = t

    new_f = (1 - best_t) * f + best_t * b
    new_f = project_to_simplex(new_f)
    return new_f, best_max


def lp_refine(initial: np.ndarray, max_iters: int, deadline: float) -> np.ndarray:
    """
    Apply the LP refinement repeatedly (up to ``max_iters``) until
    the maximum self‑convolution no longer decreases.
    """
    f = project_to_simplex(initial.copy())
    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break
        cur_conv = convolution_fft(f, f)
        M = float(cur_conv.max())
        b = solve_lp_for_b(f, M, deadline)
        if b is None:
            break
        # normalise b to the simplex
        s = b.sum()
        if s <= 0:
            break
        b = b / s
        f_new, new_M = optimal_mix(f, b)
        if new_M >= M - 1e-12:               # no genuine improvement
            break
        f = f_new
    return f


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (fast, works on the true objective)
# ----------------------------------------------------------------------
def subgradient_descent_exact(initial: np.ndarray,
                              deadline: float,
                              max_iter: int = 40000,
                              step0: float = 0.04) -> np.ndarray:
    """
    Sub‑gradient descent on
        f(a) = 2·n·max(conv(a,a)),   Σa = 1,   a ≥ 0 .
    The step size η_t = step0 / √t.
    """
    a = project_to_simplex(initial.copy())
    n = a.size
    L = 2 * n - 1
    best = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() > deadline - 1.0:
            break
        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        # indices (with tolerance) where the maximum is attained
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient = 2· Σ_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        grad_conv = convolution_fft(a, mask_rev)          # length n+L‑1
        grad = 2.0 * grad_conv[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
        # occasional progress report
        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d} best={best_sc:.6f}")

    return best


# ----------------------------------------------------------------------
# Adam optimisation on a smooth surrogate (optional polishing step)
# ----------------------------------------------------------------------
def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    The result is the Euclidean gradient of 2·n·softmax_max.
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_c = float(conv.max())
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)                     # soft‑max weights
    # weighted sum of shifted copies of a
    w_rev = w[::-1]
    grad_conv = convolution_fft(a, w_rev)               # length n+L‑1
    L = 2 * n - 1
    grad = 2.0 * n * grad_conv[L - 1 - np.arange(n)]
    return grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 20000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2500,
                         noise_scale: float = 2e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() > deadline - 1.0:
            break
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

        if it % 5000 == 0:
            print(f"    Adam it={it:,d} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def generate_seed_pool(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """
    Produce a diverse collection of candidate vectors for length ``n``.
    All vectors are already normalised to sum = 1.
    """
    pool: List[np.ndarray] = []

    # deterministic starters
    pool.append(np.full(n, 1.0 / n))
    pool.append(arcsine_initial(n))

    # power‑shape sweep
    for p in np.linspace(0.3, 1.5, 7):
        pool.append(generate_power_shape(n, p, eps=0.0))

    # random Beta family
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.5)
        beta = rng.uniform(0.1, 2.5)
        eps = rng.uniform(0.001, max(1.0, 0.02 * n))
        pool.append(generate_beta_shape(n, alpha, beta, eps))

    # block shapes
    for _ in range(10):
        frac = rng.uniform(0.02, 0.15)
        pool.append(generate_block_shape(n, frac))

    # sparse random spikes
    for _ in range(10):
        k = rng.integers(max(2, n // 200), max(5, n // 50))
        pool.append(generate_sparse_random_shape(n, k, rng))

    # arcsine–uniform mixtures
    for _ in range(10):
        w = rng.random()
        mix = w * arcsine_initial(n) + (1.0 - w) * np.full(n, 1.0 / n)
        pool.append(mix / mix.sum())

    return pool


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence`` value.
    The routine combines random seed generation, a lightweight LP‐based
    refinement, exact sub‑gradient descent and a short Adam polishing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Exploration over several lengths
    # --------------------------------------------------------------
    lengths = [512, 1024, 2048]  # larger lengths are more expensive for the LP

    for n in lengths:
        if time.time() >= deadline - 15.0:
            break
        print(f"\n=== Length n = {n} ===")

        # ----------- 1a. build a pool of seeds ----------
        seed_pool = generate_seed_pool(n, rng)

        # evaluate fast scores and keep the most promising
        scored = [(compute_score(vec), vec) for vec in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_vectors = [vec for _, vec in scored[:5]]   # keep 5 best seeds

        # ----------- 1b. refine each seed ----------
        for vec in top_vectors:
            if time.time() >= deadline - 8.0:
                break

            # LP based refinement (up to 3 inner iterations)
            refined = lp_refine(vec, max_iters=4, deadline=deadline)

            # Exact sub‑gradient descent (fast, gives a solid improvement)
            if time.time() < deadline - 5.0:
                refined = subgradient_descent_exact(refined, deadline,
                                                    max_iter=20000,
                                                    step0=0.04)

            # Optional Adam polishing (only if enough time remains)
            if time.time() < deadline - 4.0 and _opt is not None:
                refined, _ = adam_optimize_smooth(
                    refined,
                    deadline,
                    max_iter=12000,
                    lr0=0.1,
                    lam_start=2.0,
                    lam_end=80.0,
                    noise_every=2000,
                    noise_scale=1e-5,
                )

            # Official evaluation
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"  Candidate (n={n}) official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception:
                pass

    # --------------------------------------------------------------
    # 2.  Global polishing of the best vector found so far
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)
        print("\nFinal global polishing (sub‑gradient + Adam)...")
        a = subgradient_descent_exact(a, deadline, max_iter=50000, step0=0.02)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=30000,
            lr0=0.08,
            lam_start=2.0,
            lam_end=80.0,
            noise_every=1500,
            noise_scale=5e-6,
        )
        try:
            val = evaluate_sequence(a.tolist())
            print(f"Final polished official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print("*** Updated global best after final polishing ***")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3.  Guard – guarantee a return value
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform vector.")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
