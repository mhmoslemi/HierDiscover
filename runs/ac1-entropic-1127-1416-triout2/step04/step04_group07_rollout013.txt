assistantfinal**Solution Overview**

The goal is to minimise  

\[
E(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `*` is the a‑periodic convolution.  
The quantity is scale‑invariant, therefore we restrict the search to the
simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge 0 .
\]

`E(a)=2n·\max(a*a)` on the simplex, so we have to make the maximum entry of
the self‑convolution as small as possible.

The algorithm consists of three ideas  

1. **A good deterministic start** – the coefficients  

   \[
   c_i = \frac{\binom{2i}{i}}{4^i}
   \qquad(i=0,\dots ,n-1)
   \]

   are the first `n` terms of the power series of \((1-x)^{-1/2}\).  
   For the infinite series the self‑convolution is perfectly constant,
   and after truncation the sequence already yields a value
   `≈ π/2 ≈ 1.57`, noticeably better than the uniform start.

2. **Smooth surrogate optimisation** – we minimise a log‑sum‑exp
   approximation of the maximum

   \[
   M_\alpha(a)=\frac 1\alpha\log\!\Bigl(\sum_k e^{\alpha (a*a)_k}\Bigr)
   \]

   with projected gradient descent.  
   The temperature `α` is slowly increased (annealing) so that the
   surrogate converges to the true max.

3. **Exact sub‑gradient polishing** – once the surrogate stage is finished
   we switch to a projected Adam optimiser that works with the true
   sub‑gradient of `max(a*a)`.  
   The Adam update is performed in log‑space (so the iterates stay
   strictly positive) and after each step the vector is projected back
   onto the simplex.

Multiple restarts are performed until the time budget is exhausted.
Besides the Catalan start we also try a few random initialisations
(`uniform`, `Dirichlet`, `geometric`).  
The best sequence seen over the whole run is returned.

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection needs a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution using FFT (length 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Returns (max_conv, sub‑gradient of max_conv wrt a).
    The sub‑gradient is the average over all active (i.e. maximal) shifts.
    """
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    # tolerance to collect all ties
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ------------------------------------------------------------
# Stage 1 – smooth optimisation (log‑sum‑exp surrogate)
# ------------------------------------------------------------
def _smooth_optimize(
    init_a: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex for the surrogate
        M_α(a) = (1/α)·log Σ exp(α·(a*a))
    α is annealed (multiplied by `alpha_mul` every `inc_every`
    iterations).  Returns the best vector encountered and its *exact*
    objective value 2·n·max(a*a).
    """
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())   # next power of two
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()     # exact score (sum(a)=1)

    alpha = 0.5
    alpha_mul = 1.12
    inc_every = 250
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1

        # ----- convolution -------------------------------------------------
        conv = _full_conv(a, L)                # length 2·n‑1

        # ----- soft‑max weights (log‑sum‑exp) ------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)   # safe
        w = exp_scaled / exp_scaled.sum()          # Σ w = 1

        # ----- gradient of the surrogate (4·n·correlation) ----------------
        # correlation of w with a (length n, shift 0…n‑1)
        fft_w = np.fft.rfft(w, n=L)
        fft_a = np.fft.rfft(a, n=L)
        corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
        corr = corr[n - 1 : n - 1 + n]           # keep n entries
        grad = 4.0 * n * corr

        # ----- step size --------------------------------------------------
        lr = 0.8 / (1.0 + it * 0.0008)
        a -= lr * grad

        # ----- simplex projection -----------------------------------------
        a = _project_simplex(a)

        # ----- exact objective --------------------------------------------
        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            # periodic progress report
            if it % 500 == 0:
                print(f"Smooth  iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # ----- occasional jitter to escape plateaus -----------------------
        if stagn > 3000:
            a += 5e-4 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        # ----- anneal temperature -----------------------------------------
        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 3000.0)

    return best_a, best_score


# ------------------------------------------------------------
# Stage 2 – Adam optimiser on the true max‑convolution
# ------------------------------------------------------------
def _adam_optimize(
    init_a: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *true* max‑convolution.
    The variables are kept in log‑space (log‑softmax) to guarantee
    positivity; after each step the vector is projected onto the simplex.
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # initialise logits (small epsilon prevents log(0))
    eps = 1e-12
    b = np.log(init_a + eps)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # exact objective of the initial point
    a = np.exp(b - b.max())
    a /= a.sum()
    max_val, _ = _maxconv_and_subgrad(a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a.copy()
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1

        # ----- probabilities from logits -----------------------------------
        b_max = b.max()
        a = np.exp(b - b_max)
        a /= a.sum()

        # ----- exact max‑convolution and its sub‑gradient -------------------
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        # keep globally best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            if it % 2000 == 0:
                print(f"Adam iter {it:7d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # ----- early exit -------------------------------------------------
        if best_score < 1.503:
            break
        if stagn > 6000:
            # random jitter in log‑space
            b += 0.01 * rng.standard_normal(n)
            stagn = 0

        # ----- Adam update (gradient of the objective) --------------------
        grad_obj = 2.0 * n * grad_a                     # ∇ (2n·max)
        avg = np.dot(a, grad_obj)                       # ⟨a,∇⟩
        grad_b = a * (grad_obj - avg)                   # natural gradient

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.45 / np.sqrt(it + 1.0)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_a, best_score


# ------------------------------------------------------------
# Catalan (central‑binomial) seed generator
# ------------------------------------------------------------
def _catalan_seed(n: int) -> np.ndarray:
    """
    Returns the first `n` coefficients of (1‑x)^{-1/2},
    i.e. c_i = binom(2i,i) / 4**i  (recurrence c_{i+1}=c_i·(2i+1)/(2(i+1))).
    The vector is normalised to sum to 1.
    """
    c = np.empty(n, dtype=float)
    c[0] = 1.0
    for i in range(1, n):
        c[i] = c[i - 1] * (2 * i - 1) / (2.0 * i)
    c /= c.sum()
    return c

    """
    rng = np.random.default_rng(123456)            # deterministic RNG
    start_time = time.time()

    best_a = None

            a0 = sample / sample.sum()
        elif kind == "geometric":
            r = 0.97 + 0.03 * rng.random()
            a0 = r ** np.arange(n, dtype=float)
            a0 = a0 / a0.sum()
        else:

    if best_a is None:
