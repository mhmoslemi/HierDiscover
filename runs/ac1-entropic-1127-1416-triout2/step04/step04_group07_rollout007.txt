"""
Optimising a non‑negative step‑function for the bound

    2·n·max(conv(a,a)) / (sum(a)**2)

The programme works in three stages:

1.  **Coarse parameter sweep** – we generate a few simple families of
    sequences (geometric, mixed uniform/edge‑heavy) and keep the best
    candidate.

2.  **Smooth surrogate optimisation** – a log‑sum‑exp (soft‑max)
    surrogate of the max‑convolution is minimised by projected gradient
    descent on the probability simplex.  The temperature (α) is raised
    quickly so that the surrogate converges to the true max.

3.  **Fine Adam polishing** – the exact non‑smooth objective is
    optimised from the best point found in step 2 using Adam on the
    sub‑gradient of the max‑convolution.  Random jitter is injected
    periodically to escape plateaus.

All helper routines are top‑level, use only NumPy and SciPy, and respect
the 1000 s time budget.  The returned list contains only non‑negative
floats and is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution of a, length 2·n‑1, using FFT length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max_conv, sub‑gradient) for the aperiodic self‑convolution."""
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    # tolerance to decide which entries are active
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Stage 1 – smooth optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_optimize(
    init_a: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex for the smoothed objective

        2·n·M_α(a) ,   M_α = (1/α)·log Σ exp(α·conv)

    The temperature α starts small (very smooth) and is multiplied by
    ``alpha_mul`` every ``inc_every`` iterations.  The routine returns the
    best vector seen (according to the *exact* non‑smoothed objective).
    """
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())  # next power of two
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()

    # annealing schedule
    alpha = 0.2
    alpha_mul = 1.25
    inc_every = 200  # increase temperature more frequently

    it = 0
    stagn = 0
    while time.time() < deadline:
        it += 1

        # convolution
        conv = _full_conv(a, L)

        # soft‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)  # safe
        w = exp_scaled / exp_scaled.sum()        # Σ w = 1

        # gradient (4·n·correlation)
        # correlation of w with a (see previous implementation)
        fft_w = np.fft.rfft(w, n=L)
        fft_a = np.fft.rfft(a, n=L)
        corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
        n = a.shape[0]
        corr = corr[n - 1 : n - 1 + n]  # keep the segment of length n
        grad = 4.0 * n * corr

        # step size – a slower decay than the original code
        lr = 0.9 / (1.0 + it * 0.0005)
        a -= lr * grad
        a = _project_simplex(a)

        # exact objective
        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"Smooth iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # random jitter after long stagnation
        if stagn > 2500:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        # anneal α
        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 3000.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Adam optimiser on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(
    init_a: np.ndarray, deadline: float, restart_id: int, target: float = 1.4
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the exact objective.
    The optimisation runs until ``deadline`` or the objective falls below
    ``target``.  The routine returns the best vector encountered.
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # logits representation – prevents hitting the simplex boundary directly
    eps = 1e-12
    b = np.log(init_a + eps)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initial score
    a0 = np.exp(b - b.max())
    a0 = a0 / a0.sum()
    max_val, _ = _maxconv_and_subgrad(a0, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # probabilities from logits
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        # exact sub‑gradient
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # early termination
        if best_score < target:
            break
        if stagn > 6000:
            b += 0.01 * np.random.randn(n)     # random jitter in log‑space
            stagn = 0

        # Adam update on logits
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)               # ⟨a,∇⟩
        grad_b = a * (grad_obj - avg)          # chain‑rule for soft‑max

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper – cheap parametric families for the coarse sweep
# ----------------------------------------------------------------------
def _geometric_family(n: int, r: float) -> np.ndarray:
    """Symmetric geometric shape a_i ∝ r^i + r^{n‑1‑i}."""
    left = r ** np.arange(n)
    right = r ** np.arange(n - 1, -1, -1)
    a = left + right
    a /= a.sum()
    return a


def _mixed_uniform_geometric(n: int, r: float, mix: float) -> np.ndarray:
    """Blend of uniform and geometric families."""
    geo = _geometric_family(n, r)
    uni = np.full(n, 1.0 / n)
    a = mix * uni + (1.0 - mix) * geo
    a /= a.sum()
    return a

            best_a = a.copy()
            print(f"  Random Dirichlet → score {score:.6f}")

    # try to reuse the previous best if present (may be already good)
    if init_from_prev and previous is not None:
        a = previous.copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  Previous sequence → score {score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – several restarts of the two‑stage method
    # ------------------------------------------------------------------
    restart_id = 0
    strategies = ["prev", "uniform", "dirichlet", "geometric"]
    while time.time() < deadline - 4.0:
        kind = strategies[restart_id % len(strategies)]

        # ----- generate a fresh feasible start ---------------------------------
        if kind == "prev" and best_a is not None:
            a0 = best_a.copy()
        elif kind == "uniform":
            a0 = np.full(n, 1.0 / n)
        elif kind == "dirichlet":
            a0 = rng.gamma(0.5, 1.0, size=n)
            a0 = a0 / a0.sum()
        elif kind == "geometric":
            r = 0.93 + 0.06 * rng.random()
            a0 = _geometric_family(n, r)
        else:
            a0 = np.full(n, 1.0 / n)

        # allocate a time slice for this restart
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(30.0, 0.44 * remaining)

        # ----- Stage 1: smooth optimisation ------------------------------------
        a_smooth, score_smooth = _smooth_optimize(a0, slice_deadline, rng)

        # ----- Stage 2: Adam polishing -----------------------------------------
        if time.time() < deadline - 2.0:
            a_fine, score_fine = _adam_optimize(
                a_smooth, deadline, restart_id, target=1.4
            )
        else:
            a_fine, score_fine = a_smooth, score_smooth

        # ----- keep global best -------------------------------------------------
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            print(
                f"*** Global best after restart {restart_id:02d}: {best_score:.6f}"
            )
            if best_score < 1.503:
                print(
                    f"Target reached (score={best_score:.6f}) after {restart_id+1} restarts."
                )
                break

        restart_id += 1

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return value
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # Convert to plain Python list as required by the harness
    return [float(v) for v in best_a.tolist()]
