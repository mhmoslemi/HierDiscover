def adam_optimize(a0, time_deadline, max_iter=200000, lr=0.05,
                  beta1=0.9, beta2=0.999, eps=1e-8,
                  beta_start=0.1, beta_mul=1.001, beta_max=50.0):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    beta = beta_start
    t = 0
    while t < max_iter and time.time() < time_deadline:
        t += 1
        # compute convolution and softmax weights
        conv = convolution_fft(a, a)   # len 2n-1
        max_conv = np.max(conv)
        # compute softmax weights
        shifted = conv - max_conv   # <=0
        exp = np.exp(beta * shifted)
        w = exp / np.sum(exp)        # length 2n-1
        # compute gradient via convolution of reversed a and w
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)  # length 3n-2
        grad = 2.0 * grad_full[n-1:n-1+n]      # length n
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_to_simplex(a_new)
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        # increase beta gradually
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)
        # occasional jitter
        if t % 2000 == 0:
            a += np.random.randn(n) * (lr * 0.01 / np.sqrt(t+1))
            a = project_to_simplex(a)
    return best_a, best_score

def hill_climb_sa(seq, time_deadline,
                  init_temp=0.01, final_temp=0.0001,
                  max_steps=500_000):
    a = np.array(seq, dtype=float)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    start_time = time.time()
    steps = 0
    while time.time() < time_deadline and steps < max_steps:
        # temperature schedule based on elapsed proportion
        elapsed = time.time() - start_time
        total = time_deadline - start_time
        T = init_temp * (final_temp / init_temp) ** (elapsed / total) if total > 0 else final_temp
        
        # propose move
        cand = best_a.copy()
        move_type = np.random.rand()
        if move_type < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # transfer fraction of mass from i to j
            frac = np.random.rand() * 0.5  # up to 50% of i's mass
            delta = frac * cand[i]
            cand[i] -= delta
            cand[j] += delta
        
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        delta_score = cur_score - best_score
        
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / (T + 1e-12)):
            # accept
            best_a = cand
            best_score = cur_score if cur_score < best_score else best_score
        steps += 1
    return best_a.tolist(), best_score

"""
Propose a low‑score step function.

Algorithm Overview
-------------------
1. **Random seeding** – generate many candidates quickly.
   * Two families are used:
     – Uniform weight on a random sparse subset (as in the previous work).
     – Dirichlet‑distributed vectors (controllable sparsity via the
       concentration parameter).
   * The best candidate from this fast phase seeds the subsequent refinements.

2. **Soft‑max Adam optimisation** – minimise a smooth proxy of the
   objective  `2·n·max(a*a)`.  
   * The max is smoothed with a temperature‑controlled soft‑max:
     `g_s ∝ exp(β·conv_s)`.  For β→∞ the weights concentrate on the
     maximal shift, while small β yields a gentler landscape.
   * The gradient of the smoothed objective is obtained analytically
     via a single convolution, and an Adam update (first‑ and second‑
     moment estimates) with Euclidean projection onto the probability
     simplex keeps the vector feasible.
   * β is increased slowly (geometric schedule) so the optimisation
     starts smooth and ends focused on the true max.

3. **Simulated‑annealing hill‑climber** – a stochastic local search that
   swaps entries or transfers a fraction of mass while accepting
   worsening moves with a temperature schedule.  This stage polishes
   the Adam result.

All work respects the 1000 s budget.  The final list of floats is
returned directly to the harness – `evaluate_sequence` will be called
by the evaluator.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution a * b using a real FFT (fast).
    Returns a NumPy array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a) == 1.
    Returns 2·n·max(a*a) which is exactly the value returned by
    evaluate_sequence for a normalised vector.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random candidate generators
# ----------------------------------------------------------------------


def random_sparse_candidate(n: int, density: float) -> np.ndarray:
    """Uniform weight on a random subset of size k ≈ density·n."""
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_candidate(n: int, concentration: float) -> np.ndarray:
    """Sample a Dirichlet(alpha) vector (sum == 1).  Small concentration → sparse."""
    g = np.random.gamma(concentration, 1.0, size=n)
    a = g / g.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation with soft‑max smoothing of the max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(init_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 300_000,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  beta_start: float = 0.1,
                  beta_mul: float = 1.001,
                  beta_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of 2·n·max(a*a) using a soft‑max
    approximation of the max.  The vector is kept on the simplex.
    Returns the best vector seen and its (exact) score.
    """
    a = init_a.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    beta = beta_start
    t = 0

    while t < max_iter and time.time() < time_deadline:
        t += 1

        # ----- compute convolution and soft‑max weights -----
        conv = convolution_fft(a, a)                 # length 2n‑1
        max_conv = float(np.max(conv))
        # soft‑max on the shifted values (max is subtracted for stability)
        shifted = conv - max_conv                     # ≤0
        weights = np.exp(beta * shifted)
        weights /= np.sum(weights)                    # size 2n‑1, sums to 1

        # ----- gradient of the smoothed objective -----
        # grad_j = 2 * sum_i a_i * weights_{i+j}
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, weights)  # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]       # slice of length n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # ----- temperature schedule for the soft‑max -----
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        # occasional jitter to escape flat plateaus
        if t % 2000 == 0:
            a += np.random.randn(n) * (lr * 0.01 / np.sqrt(t + 1))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(seq: list[float],
                  time_deadline: float,
                  init_temp: float = 0.02,
                  final_temp: float = 0.0002,
                  max_steps: int = 500_000) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb with temperature controlled acceptance.
    Moves consist of swaps or fractional mass transfers.
    Returns the best vector encountered and its exact score.
    """
    cur = np.array(seq, dtype=float)
    n = cur.shape[0]

    cur_score = compute_score(cur)
    best_vec = cur.copy()
    best_score = cur_score

    start = time.time()
    steps = 0

    while time.time() < time_deadline and steps < max_steps:
        # ----- temperature schedule -----
        elapsed = time.time() - start
        total = max(time_deadline - start, 1e-12)
        T = init_temp * (final_temp / init_temp) ** (elapsed / total)

        # ----- propose a move -----
        cand = cur.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                frac = np.random.rand() * 0.5          # up to 50 % of mass
                delta = frac * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        # ----- Metropolis acceptance -----
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()

        steps += 1

    return best_vec.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The algorithm combines a fast random search, a soft‑max Adam optimiser,
    and a final simulated‑annealing hill‑climber, all within the allotted
    time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback (previous best known solution)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Random search (sparse subsets + Dirichlet)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    while time.time() < deadline - 30.0:   # reserve time for refinements
        n = int(np.random.choice(lengths))
        # decide which family to use
        if np.random.rand() < 0.6:
            # sparse uniform subset
            density = np.random.uniform(0.20, 0.60)   # avoid very dense extremes
            cand = random_sparse_candidate(n, density)
        else:
            # Dirichlet vector
            concentration = np.random.uniform(0.05, 2.0)
            cand = random_dirichlet_candidate(n, concentration)

        cur_val = compute_score(cand)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cand.tolist()
            print(f"New random best: score={best_val:.6f} (n={n}, mode={'sparse' if density else 'dirichlet'})")

        # occasional early‑exit if we are very close to deadline
        if time.time() > deadline - 25.0:
            break

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        print("\nStarting Adam soft‑max optimisation")
        init_a = np.array(best_seq, dtype=float)
        refined_a, refined_score = adam_optimize(
            init_a,
            time_deadline=deadline - 5.0,
            max_iter=300_000,
            lr=0.06,
            beta_start=0.1,
            beta_mul=1.002,
            beta_max=40.0,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Adam improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final polishing with simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nPolishing with simulated‑annealing hill‑climber")
        polished_seq, polished_val = hill_climb_sa(
            best_seq,
            time_deadline=deadline,
            init_temp=0.02,
            final_temp=0.0002,
            max_steps=500_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: fall back to a uniform vector if nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (the harness will also call it, but we print it)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
