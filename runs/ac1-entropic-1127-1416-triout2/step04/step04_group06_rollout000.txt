"""
Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence()`.

Algorithm Overview
------------------
* **Re‑parameterisation** – the vector a lives on the probability simplex.
  We optimise logits `z` with `a = softmax(z)`.  This automatically enforces
  non‑negativity and Σa = 1.

* **Smoothed objective** – to minimise the L∞‑norm of the self‑convolution
  we use the log‑sum‑exp smooth approximation

        F_α(a) = (1/α)·log Σ_i exp(α·(a∗a)_i)

  for a gradually increasing smoothing parameter `α`.  When `α` is large,
  `F_α` approaches the true maximum.

* **Adam on the logits** – a fast first‑order optimiser (Adam) is applied
  to the gradient of `F_α` w.r.t. the logits.  The gradient of `F_α`
  w.r.t. the probabilities `a` is
        ∇_a F_α = 2·corr(a, p) ,  p_i = softmax_i(α·(a∗a)_i)

  and the chain‑rule gives the gradient w.r.t. `z`:
        ∇_z = a ⊙ (∇_a – (∇_a·a)·1).

* **Multiple restarts** – a handful of simple random initialisers
  (Beta, Dirichlet, edge‑biased, sparse) are tried.  The best solution
  across all restarts is kept.

* **Local hill‑climbing** – after the Adam phase a cheap deterministic
  mass‑transfer move is performed repeatedly.  Only moves that improve
  the *exact* objective (the true max‑convolution) are accepted.

The whole search respects the 1000 s budget and prints progress
messages.  The returned list is already normalised to sum = 1.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full_like(x, 1.0 / x.size)
    return e / s


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1."""
    s = v.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` for a vector that already satisfies
    sum(a) == 1.  Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution length = 2*n‑1
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def autocorr(a: np.ndarray) -> np.ndarray:
    """Compute a ∗ a (linear convolution, length 2·n‑1)."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    return np.fft.irfft(fft_a * fft_a, n=2 * n - 1)


def grad_smoothed_max(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) w.r.t. the probabilities `a`.
    Returns a vector of length n.
    """
    n = a.shape[0]
    conv = autocorr(a)                     # length 2n‑1
    maxc = conv.max()
    # softmax over α·(conv - maxc) for numerical stability
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0 or not np.isfinite(sum_exp):
        p = np.full_like(conv, 1.0 / conv.size)
    else:
        p = exp_vals / sum_exp               # shape (2n‑1,)

    # correlation a ⋆ p  (non‑circular)
    L = 1 << ((3 * n - 2).bit_length())     # next power of two ≥ 3n‑2
    fft_a = np.fft.rfft(a, n=L)
    fft_p = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_p), n=L)
    offset = len(p) - 1                     # lag 0 appears at this index
    grad = 2.0 * corr[offset:offset + n]    # gradient ∇_a F_α
    return grad


# ----------------------------------------------------------------------
# Random initialisers (all return a vector with sum = 1)
# ----------------------------------------------------------------------


def init_beta(n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points."""
    a = np.random.uniform(0.2, 5.0)
    b = np.random.uniform(0.2, 5.0)
    x = (np.arange(n) + 0.5) / n
    # log‑density to avoid overflow/underflow
    logw = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return simplex_normalise(w)


def init_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet(1,…,1) vector."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """Linear edge‑heavy shape (larger values near the two ends)."""
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = dist_to_center.astype(float) + 1.0
    return simplex_normalise(a)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Put mass on k random positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within ``budget_s`` seconds.
    Returns the best sequence found (as a list of floats, sum == 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size and start from a known solution if available
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be injected by the caller.
        init_vec = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_vec.size
        best_a = simplex_normalise(init_vec)
        best_score = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1 (n={n}), score={best_score:.6f}")
    except NameError:
        # Fallback: use a moderate size (≈4000) uniform start
        n = 4096
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_a)
        print(f"[Init] No previous sequence – using uniform start, n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to evaluate exact objective (fast, because sum==1)
    # ------------------------------------------------------------------
    def exact_score(v: np.ndarray) -> float:
        return compute_score(v)

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0:   # reserve a few seconds for finalisation
        # ----- pick a random initialiser -----
        init_choice = np.random.randint(4)
        if init_choice == 0:
            a = init_beta(n)
        elif init_choice == 1:
            a = init_dirichlet(n)
        elif init_choice == 2:
            a = init_edge_concentrated(n)
        else:
            k = np.random.randint(5, 30)
            a = init_sparse(n, k)

        # Occasionally blend with the incumbent best solution
        if np.random.rand() < 0.2:
            a = 0.5 * a + 0.5 * best_a
            a = simplex_normalise(a)

        # ---------- Adam on logits ----------
        eps = 1e-12
        z = np.log(a + eps)                 # logits (no constraints)
        m = np.zeros_like(z)                # first moment
        v = np.zeros_like(z)                # second moment
        beta1 = 0.9
        beta2 = 0.999
        eta = 0.15                          # base learning rate
        alpha_start = 0.5
        alpha_end = 200.0
        max_iters = 3000
        for it in range(1, max_iters + 1):
            # time check
            if time.time() > deadline - 4.0:
                break

            # reconstruct probability vector
            a = softmax(z)

            # schedule for the smoothing parameter α
            alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

            # gradient of smoothed max w.r.t. a
            grad_a = grad_smoothed_max(a, alpha)          # shape (n,)

            # chain rule: gradient w.r.t. logits z
            dot = np.dot(grad_a, a)
            grad_z = a * (grad_a - dot)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            z -= eta * m_hat / (np.sqrt(v_hat) + 1e-8)

            # Occasionally evaluate the true objective and keep the best
            if (it & 31) == 0:  # every 32 iterations
                cur_score = exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Restart {restart}] it {it:4d} → new best {best_score:.6f}")

        # ---------- Local hill‑climbing refinement ----------
        cur_a = best_a.copy()
        cur_score = exact_score(cur_a)

        # Simple temperature schedule proportional to current score
        T = cur_score * 0.02
        T_end = cur_score * 1e-5
        while time.time() < deadline - 2.0 and T > T_end:
            # pick source index proportional to its mass
            src = np.random.choice(n, p=cur_a)
            # pick a distinct target index uniformly
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1
            if cur_a[src] <= 0.0:
                continue
            # transfer a random fraction of mass from src to tgt
            delta = cur_a[src] * np.random.rand() * 0.5  # at most half of src mass
            cand = cur_a.copy()
            cand[src] -= delta
            cand[tgt] += delta

            cand_score = exact_score(cand)
            if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[SA  R{restart}] new best {best_score:.6f}")

            # geometric cooling
            T *= 0.9993

        restart += 1

    # ------------------------------------------------------------------
    # Return the best vector found (already normalised)
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {restart} restarts: {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
