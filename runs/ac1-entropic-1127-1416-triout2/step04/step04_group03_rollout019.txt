def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2

    # define lengths progression
    lengths = [256, 512, 1024, 2048, 4096, 8192]  # but may stop earlier based on time

    best_seq = None
    best_score = float('inf')

    # Starting candidate: arcsine shape for smallest length
    a = arcsine_initial(lengths[0])
    # optionally also try uniform and a few beta shapes, keep the best
    # we can keep a pool of few best (maybe 3) to upsample

    for idx, n in enumerate(lengths):
        if time.time() >= deadline-5:
            break
        # ensure we have a vector of proper length: produce via upsampling if not first length
        if a.shape[0] != n:
            a = upsample_vector(a, n)
        # run random sampling among beta shapes (maybe 200 samples)
        num_samples = min(400, int((deadline - time.time()) * 0.5))  # allocate time
        for _ in range(num_samples):
            if time.time() >= deadline - 4:
                break
            # random parameters
            alpha = rng.uniform(0.02, 1.5)
            beta = rng.uniform(0.02, 1.5)
            eps = rng.uniform(1e-6, 0.001 * n)  # small epsilon
            cand = generate_beta_shape(n, alpha, beta, eps)
            sc = compute_score(cand)
            if sc < compute_score(a):
                a = cand
        # gradient based refinement
        if time.time() < deadline - 3:
            max_iter = int(min(150_000, (deadline - time.time()) * 30))
            a, _ = adam_optimize(a, deadline, max_iter=max_iter, lr0=0.2)
        if time.time() < deadline - 3:
            max_iter = int(min(80_000, (deadline - time.time()) * 25))
            a, _ = subgradient_optimize(a, deadline, max_iter=max_iter)
        if time.time() < deadline - 3:
            max_iter = int(min(100_000, (deadline - time.time()) * 25))
            a, _ = random_local_search(a, deadline, max_iter=max_iter)

        # final evaluation at current length
        cur_eval = evaluate_sequence(a.tolist())
        print(f"Length {n} candidate score = {cur_eval:.6f}")
        if cur_eval < best_score:
            best_score = cur_eval
            best_seq = a.tolist()
            print(f"*** New best at length {n}: {best_score:.6f} ***")

    # final polishing on best_seq with full time remaining
    if best_seq and time.time() < deadline - 2:
        a = np.array(best_seq, dtype=float)
        # final longer Adam maybe 200k iterations
        if time.time() < deadline - 2:
            max_iter = int(min(300_000, (deadline - time.time()) * 35))
            a, _ = adam_optimize(a, deadline, max_iter=max_iter, lr0=0.1)
        if time.time() < deadline - 2:
            max_iter = int(min(150_000, (deadline - time.time()) * 30))
            a, _ = subgradient_optimize(a, deadline, max_iter=max_iter)
        if time.time() < deadline - 2:
            max_iter = int(min(200_000, (deadline - time.time()) * 30))
            a, _ = random_local_search(a, deadline, max_iter=max_iter)
        final_score = evaluate_sequence(a.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = a.tolist()
            print(f"*** Updated best after final polish: {best_score:.6f} ***")

    # fallback
    if not best_seq:
        best_seq = [1.0] * 256
        best_score = evaluate_sequence(best_seq)
    print(f"Search finished. Best final score {best_score:.6f}")
    return best_seq

"""
Propose a low‑discrepancy step‑function (vector) that minimizes

    2·n·max(conv(a,a)) / (Σa)²

The algorithm consists of three phases:

1. **Coarse exploration** – for a few candidate lengths we evaluate
   symmetric “block‑shapes” (two constant edge blocks and a uniform
   interior) on a dense grid of parameters.  The best candidate
   together with a few standard starters (uniform, arcsine, random
   beta‑families) is kept.

2. **Refinement** – the retained vectors are refined by three
   optimisation stages:
      • L‑BFGS‑B on a smooth log‑sum‑exp surrogate (variables are the
        log‑soft‑max of the vector, guaranteeing non‑negativity and the
        simplex constraint);
      • Adam optimiser on the same surrogate (helps escape shallow
        basins);
      • Stochastic local search with Gaussian perturbations.

3. **Final polish** – the globally best vector found so far is
   polished again with the three stages using the remaining time.

All heavy work respects the overall time budget (default 1000 s) and the
routine always returns a non‑empty, non‑negative list that is accepted
by ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Optional SciPy import
# ----------------------------------------------------------------------
try:
    from scipy.optimize import minimize
    _SCIPY = True
except Exception:          # pragma: no cover
    _SCIPY = False

# ----------------------------------------------------------------------
# Global for the L‑BFGS callback (used to enforce the time limit)
# ----------------------------------------------------------------------
_GLOBAL_DEADLINE = None


class _TimeOutError(RuntimeError):
    """Raised inside the L‑BFGS callback when the time budget is exhausted."""
    pass


def _lbfgs_callback(_xk):
    """SciPy L‑BFGS‑B callback – abort if we passed the deadline."""
    if _GLOBAL_DEADLINE is not None and time.time() > _GLOBAL_DEADLINE:
        raise _TimeOutError()


# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate used throughout the search:
        2·n·max(conv(a,a))
    The vector is assumed to be normalised (Σa = 1), hence this equals
    ``evaluate_sequence``'s return value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}
    (Duchi et al., 2008)."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetric, diverging at the ends)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def block_shape(n: int, edge_frac: float, p: float) -> np.ndarray:
    """
    Two constant edge blocks of relative weight `p` times the interior weight.
    ``edge_frac`` ∈ [0,0.5] is the fraction of entries taken from each side.
    ``p`` > 0 determines how heavy the edges are (p=1 ⇒ uniform).

    The returned vector is already normalised to Σa = 1.
    """
    if edge_frac <= 0.0:
        return np.full(n, 1.0 / n)

    m = int(round(edge_frac * n))
    m = max(1, min(m, n // 2))                # at least one entry, at most half
    interior = n - 2 * m
    # w = p·v, and 2·m·w + interior·v = 1  ⇒  v = 1/(interior + 2·m·p)
    v = 1.0 / (interior + 2.0 * m * p)
    w = p * v
    a = np.empty(n, dtype=float)
    a[:m] = w
    a[-m:] = w
    if interior > 0:
        a[m:n - m] = v
    return a


def upsample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation to a larger size.
    The total mass is preserved (the result is re‑normalised to Σ=1).
    """
    old_n = v.shape[0]
    if new_n == old_n:
        return v.copy()
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False) + 0.5 / old_n
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False) + 0.5 / new_n
    new_vals = np.interp(new_x, old_x, v)
    new_vals /= new_vals.sum()
    return new_vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp approximation to max(conv(a,a)).
    Returns (gradient w.r.t. a, surrogate value B).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # softmax weights
    B = max_c + np.log(sum_exp) / lam     # smooth approx. of max(conv)

    # weighted sub‑gradient:
    # ∂ (Σ_k w_k (a*a)_k) / ∂a = 2 * (a ∗ w_rev)   (∗ = linear conv)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B

# Optimisers

def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    noise_every: int = 1500,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature (smoothness)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    The step size decays like step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term (the indices that achieve the max)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Stochastic local walk on the simplex.
    Tiny probability to accept uphill moves helps escape local minima.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

    return best_a, best_sc


def lbfgs_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 30_000,
    lam: float = 80.0,
) -> Tuple[np.ndarray, float]:
    """
    L‑BFGS‑B optimiser on the smooth log‑sum‑exp surrogate.
    Variables are the log‑softmax parameters, guaranteeing non‑negativity
    and exact simplex projection.
    """
    if not _SCIPY:                 # pragma: no cover
        # SciPy missing – fall back to Adam
        return adam_optimize(initial_a, time_deadline, max_iter=max_iter)

    global _GLOBAL_DEADLINE
    _GLOBAL_DEADLINE = time_deadline

    n = len(initial_a)
    x0 = np.log(initial_a + 1e-12)       # safe initial log‑weights

    def objective(x, n_dim, lam_param):
        """Returns (value, gradient) for scipy.minimize."""
        a = np.exp(x)
        s = a.sum()
        a = a / s                         # simplex projection (softmax)
        conv = convolution_fft(a, a)
        max_c = np.max(conv)
        # objective = 2·n·max_c (the denominator is 1 after normalisation)
        val = 2.0 * n_dim * max_c

        # gradient of the smooth surrogate (using the same λ for consistency)
        grad_a, _ = smooth_gradient(a, lam_param)

        # chain rule through the softmax:
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        return val, grad_x

    try:
        res = minimize(
            objective,
            x0,
            args=(n, lam),
            method='L-BFGS-B',
            jac=True,
            bounds=None,
            callback=_lbfgs_callback,
            options={'maxiter': max_iter, 'disp': False},
        )
        best_x = res.x
    except _TimeOutError:          # pragma: no cover
        # Time limit hit – keep the current point
        best_x = x0

    a_opt = np.exp(best_x)
    a_opt = a_opt / a_opt.sum()
    best_sc = compute_score(a_opt)
    return a_opt, best_sc

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine combines:
      • a dense grid of two‑block shapes,
      • random beta‑family sampling,
      • L‑BFGS‑B → Adam → sub‑gradient → stochastic local search,
      • a final polishing stage.
    All work respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Try a simple fallback (uniform 256‑vector) – guaranteed feasible
    # ------------------------------------------------------------------
    try:
        fallback = [1.0] * 256
        val = evaluate_sequence(fallback)
        print(f"Fallback uniform (256) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Exploration over a handful of lengths (doubling schedule)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 6.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ----------------------------------------------------------------
        # 1a. Deterministic starters
        # ----------------------------------------------------------------
        local_best_a = None
        local_best_score = float("inf")
        starters = [
            np.full(n, 1.0 / n),           # uniform
            arcsine_initial(n),           # classic arcsine
        ]
        for st in starters:
            sc = compute_score(st)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = st.copy()

        # ----------------------------------------------------------------
        # 1b. Grid search over block shapes (two edge blocks + interior)
        # ----------------------------------------------------------------
        edge_fractions = np.linspace(0.0, 0.5, 11)      # 0, 0.05,…,0.5
        p_values = np.logspace(-1, 2, 30)              # 0.1 … 100
        for ef in edge_fractions:
            for p in p_values:
                cand = block_shape(n, ef, p)
                sc = compute_score(cand)
                if sc < local_best_score:
                    local_best_score = sc
                    local_best_a = cand.copy()

        # ----------------------------------------------------------------
        # 1c. Random beta‑family sampling (wide parameter ranges)
        # ----------------------------------------------------------------
        max_samples = 500
        for _ in range(max_samples):
            if time.time() >= deadline - 6.0:
                break
            alpha = rng.uniform(0.02, 2.5)
            beta = rng.uniform(0.02, 2.5)
            eps = rng.uniform(1e-8, max(0.01, 0.02 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)
            sc = compute_score(cand)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = cand.copy()

        # ----------------------------------------------------------------
        # 2.  Refinement of the locally best candidate
        # ----------------------------------------------------------------
        if local_best_a is None or time.time() >= deadline - 4.0:
            continue

        # L‑BFGS‑B (smooth surrogate)
        refined_a, _ = lbfgs_optimize(
            initial_a=local_best_a,
            time_deadline=deadline,
            max_iter=int(min(30_000, (deadline - time.time()) * 25)),
            lam=80.0,
        )

        # Adam (fine‑grained descent)
        if time.time() < deadline - 3.0:
            refined_a, _ = adam_optimize(
                refined_a,
                time_deadline=deadline,
                max_iter=int(min(100_000, (deadline - time.time()) * 30)),
                lr0=0.18,
                lam_start=1.0,
                lam_end=80.0,
                noise_every=1200,
                noise_scale=5e-5,
            )

        # Sub‑gradient polishing
        if time.time() < deadline - 2.5:
            refined_a, _ = subgradient_optimize(
                refined_a,
                time_deadline=deadline,
                step0=0.25,
                max_iter=int(min(80_000, (deadline - time.time()) * 25)),
            )

        # Random local search (last escape stage)
        if time.time() < deadline - 2.0:
            refined_a, _ = random_local_search(
                refined_a,
                time_deadline=deadline,
                init_step=0.018,
                max_iter=int(min(120_000, (deadline - time.time()) * 30)),
            )

        # Official evaluation (once per length)
        val = evaluate_sequence(refined_a.tolist())
        print(f"Length {n} refined official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined_a.tolist()
            print(f"*** New global best after length {n}: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Final polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\n=== Final polishing of the global best ===")
        a = np.array(best_seq, dtype=float)

        # L‑BFGS‑B long run
        if time.time() < deadline - 2.5:
            a, _ = lbfgs_optimize(
                a,
                time_deadline=deadline,
                max_iter=int(min(80_000, (deadline - time.time()) * 30)),
                lam=100.0,
            )

        # Adam polish
        if time.time() < deadline - 2.0:
            a, _ = adam_optimize(
                a,
                time_deadline=deadline,
                max_iter=int(min(200_000, (deadline - time.time()) * 35)),
                lr0=0.12,
                lam_start=1.0,
                lam_end=120.0,
                noise_every=1500,
                noise_scale=3e-5,
            )

        # Sub‑gradient polish
        if time.time() < deadline - 1.5:
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.2,
                max_iter=int(min(150_000, (deadline - time.time()) * 30)),
            )

        # Random local polish
        if time.time() < deadline - 1.0:
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.012,
                max_iter=int(min(250_000, (deadline - time.time()) * 40)),
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
