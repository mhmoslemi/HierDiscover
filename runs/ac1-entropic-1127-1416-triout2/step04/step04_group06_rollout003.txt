"""
step‑function optimizer – primal‑dual mirror‑descent + guided SA
----------------------------------------------------------------
* Works on the probability simplex (sum = 1, a_i ≥ 0).
* Uses a smooth log‑sum‑exp approximation of the L∞‑norm of
  the self‑convolution (the quantity we finally want to minimise).
* At each iteration we:
    – compute the self‑convolution via FFT,
    – form a soft‑max distribution p over convolution indices,
    – evaluate the gradient g_i = 2·∑_k p_k·a_{k‑i} (cross‑correlation),
    – apply an exponentiated‑gradient (mirror‑descent) step with a
      diminishing step‑size and a momentum term,
    – slowly increase the smoothing parameter α until it is large
      enough to approximate the true max.
* Multiple random restarts are performed; each restart starts from the
  current best vector plus a small random perturbation.
* After the mirror‑descent phase a short simulated‑annealing (SA)
  refinement moves mass from indices with large gradient (high contribution
  to the peak) toward indices with small gradient, accepting uphill moves
  with a geometrically cooling temperature.
* The routine respects the supplied time‑budget (default 1000 s) and
  returns the best vector found (as a plain Python list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale v to have sum = 1 (preserves non‑negativity)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector already normalised
    to sum = 1.  Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution via FFT (length 2*n‑1)
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised symmetric Beta(α,β) density on n points, normalised."""
    x = (np.arange(n) + 0.5) / n
    # log‑density (avoid overflow/underflow)
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return simplex_normalise(w)


def init_random_beta(n: int) -> np.ndarray:
    """Random centred Beta distribution (α ≈ β)."""
    a = np.random.uniform(0.2, 8.0)
    b = np.random.uniform(0.2, 8.0)
    return beta_sequence(a, b, n)


def init_dirichlet(n: int) -> np.ndarray:
    """Dirichlet(1,…,1) – i.i.d. Gamma(1)."""
    v = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(v)


def init_edge_concentrated(n: int) -> np.ndarray:
    """Linear edge‑heavy shape (larger values near the two ends)."""
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    v = dist_to_center.astype(float) + 1.0
    return simplex_normalise(v)


def init_sparse(n: int, k: int = 12) -> np.ndarray:
    """Put mass on k randomly chosen positions."""
    v = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    v[chosen] = vals
    return simplex_normalise(v)


def init_hann(n: int) -> np.ndarray:
    """Classical Hann window (zero at the ends)."""
    x = np.arange(n, dtype=float)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
    return simplex_normalise(w)


def init_hamming(n: int) -> np.ndarray:
    """Hamming window (small side‑lobe)."""
    x = np.arange(n, dtype=float)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
    return simplex_normalise(w)


def init_blackman(n: int) -> np.ndarray:
    """Blackman window (stronger side‑lobe suppression)."""
    x = np.arange(n, dtype=float)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    w = (a0
         - a1 * np.cos(2.0 * np.pi * x / (n - 1))
         + a2 * np.cos(4.0 * np.pi * x / (n - 1)))
    return simplex_normalise(w)


def resample_to(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear resampling of a vector to a new length (preserves shape)."""
    old_n = v.shape[0]
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    new_v = np.interp(x_new, x_old, v)
    return simplex_normalise(new_v)

        g_avg = np.zeros_like(a)

        # ----- mirror‑descent iterations -----
        for it in range(iters_per_restart):
            now = time.time()
            if now > deadline - 6.0:
                break

            # 1) self‑convolution (FFT)
            fft_a = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

            # 2) soft‑max distribution over convolution entries
            maxc = conv.max()
            # shift for numerical stability
            exp_vals = np.exp(alpha_start * (conv - maxc))
            sum_exp = exp_vals.sum()
            if not np.isfinite(sum_exp) or sum_exp == 0.0:
                p = np.full_like(conv, 1.0 / conv.size)
            else:
                p = exp_vals / sum_exp

            # 3) gradient g_i = 2·∑_k p_k·a_{k‑i}
            # Use cross‑correlation via FFT (length ≥ 3n‑2)
            L = 1 << ((3 * n - 2).bit_length())
            fft_a_full = np.fft.rfft(a, n=L)
            fft_p = np.fft.rfft(p, n=L)
            corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L)
            offset = len(p) - 1
            g = 2.0 * corr[offset:offset + n]

            # 4) momentum smoothing
            g_avg = momentum_beta * g_avg + (1.0 - momentum_beta) * g

            # 5) exponentiated‑gradient update on the simplex
            eta = eta0 / math.sqrt(it + 1.0)
            # clip the exponent to avoid overflow/underflow
            exp_arg = -eta * g_avg
            exp_arg = np.clip(exp_arg, -50.0, 50.0)
            a = a * np.exp(exp_arg)
            a = simplex_normalise(a)

            # 6) increase smoothing parameter α
            # we accelerate the increase after half the iterations
            if it < iters_per_restart // 2:
                alpha_start = alpha_start + (min_alpha - alpha_start) / (iters_per_restart // 2)
            else:
                alpha_start = alpha_start + (alpha_end - min_alpha) / (iters_per_restart // 2)

            # ----- occasional evaluation -----
            if (it & 31) == 0:  # every 32 iterations
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Restart {restart}] it {it:4d} → new best {best_score:.6f}")

        # ----- short simulated‑annealing refinement on the best so far -----
        # (only if we still have a few seconds left)
        cur_a = best_a.copy()
        cur_score = best_score
        T = cur_score * 0.03      # start temperature (few percent of score)
        T_end = cur_score * 1e-5
        sa_iters = 0
        # pre‑compute gradient for guided moves
        while time.time() < deadline - 3.0 and T > T_end:
            # pick source – heavy‑gradient index (probability ∝ a)
            src = np.random.choice(n, p=cur_a)
            # deterministic target: index with smallest gradient (computed on‑fly)
            # we recompute a cheap gradient approximation:
            fft_cur = np.fft.rfft(cur_a, n=2 * n - 1)
            conv_cur = np.fft.irfft(fft_cur * fft_cur, n=2 * n - 1)
            maxc_cur = conv_cur.max()
            exp_vals_cur = np.exp(alpha_end * (conv_cur - maxc_cur))
            p_cur = exp_vals_cur / exp_vals_cur.sum()
            L = 1 << ((3 * n - 2).bit_length())
            fft_a_cur = np.fft.rfft(cur_a, n=L)
            fft_p_cur = np.fft.rfft(p_cur, n=L)
            corr_cur = np.fft.irfft(fft_a_cur * np.conj(fft_p_cur), n=L)
            offset = len(p_cur) - 1
            g_cur = 2.0 * corr_cur[offset:offset + n]

            tgt = np.argmin(g_cur)

            if src == tgt or cur_a[src] <= 0.0:
                continue

            # move a random fraction of the source mass to the target
            delta = cur_a[src] * np.random.rand() * 0.6
            cand = cur_a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = simplex_normalise(cand)

            cand_score = compute_score(cand)
            if cand_score < cur_score:
                accept = True
            else:
                accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

            if accept:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[SA  R{restart}] iter {sa_iters:6d} → new best {best_score:.6f}")

            # geometric cooling
            T *= 0.9994
            sa_iters += 1

        restart += 1

    # ------------------------------------------------------------------
    # 3. Return the best vector (already normalised to sum = 1)
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {restart} restarts: {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
