    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0
    best_seq = None
    best_val = float('inf')
    
    # fallback if exist
    try:
        fallback = [float(x) for x in height_sequence_1]
        val = evaluate_sequence(fallback)
        if val < best_val:
            best_seq = fallback
            best_val = val
    except NameError:
        pass
    
    # choose sequence length
    n = 2048
    # initial shape arcsine
    a = arcsine_initial(n)  # returns sum=1; we may scale but not needed
    a = a / a.sum()  # ensure sum=1
    cur_val = evaluate_sequence(a.tolist())
    print(f"Initial arcsine val = {cur_val}")
    if cur_val < best_val:
        best_val, best_seq = cur_val, a.tolist()
    
    # main loop
    iteration = 0
    while time.time() < deadline - 5.0:
        iteration += 1
        # compute M
        conv = convolution_fft(a, a)
        M = conv.max()
        # Build Toeplitz matrix for convolution constraints
        # Use efficient construction: each row corresponds to shifted version of a
        # We'll use scipy.linalg.toeplitz
        from scipy.linalg import toeplitz
        
        # Column c = [a[0], a[1],..., a[-1], 0,0,...] length 2n-1
        c = np.r_[a, np.zeros(n-1)]
        # Row r = [a[0], zeros...] length n
        r = np.r_[a[0], np.zeros(n-1)]
        A = toeplitz(c, r)  # shape (2n-1, n)
        # Upper bounds vector
        b_ub = np.full(A.shape[0], M, dtype=float)
        # Linear program maximize sum(b). Equivalent to minimize -sum(b)
        c_obj = -np.ones(n, dtype=float)
        # Solve
        try:
            res = linprog(c=c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        except Exception as e:
            print("LP solve error:", e)
            break
        if not res.success:
            print("LP failed:", res.message)
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:

        # mix

        g = b / sum_b

        a = (1 - t) * a + t * g
        # renorm to sum 1
        a = a / a.sum()
        cur_val = evaluate_sequence(a.tolist())

        if cur_val < best_val:
            best_val = cur_val

    B = np.fft.rfft(b, n=n_fft)

    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals

    """Sub‑gradient of max_k (a*a)_k."""
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)

        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(200_000, remaining * 300))
        print(f"\nPolishing stage: {max_iter} sub‑gradient iterations remaining")
        a_refined, _ = optimise_subgradient(
            initial_a=a0,
            deadline=deadline,
            eta0=0.3,
            max_iter=max_iter,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a_refined.tolist()
            print("*** Updated global best after polishing ***")

    # ------------------------------------------------------------------
    # 4. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
