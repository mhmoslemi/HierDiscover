The program searches a non‑negative step‑function `a = (a₀,…,a_{n‑1})` that minimises  

\[
E(a)=\frac{2n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}.
\]

Because the objective is homogeneous we may normalise `∑a_i = 1`; the
evaluation then reduces to `2 n · max(conv)`.  
The search consists of two complementary phases and many cheap restarts:

1. **Smooth surrogate optimisation** – we minimise a log‑sum‑exp
   approximation of the `max` using projected gradient descent on the
   probability simplex.  The surrogate becomes the true `max` as the
   temperature `α` grows.  After every gradient step the vector is
   *symmetrised* (`a ← (a + a[::-1])/2`) and projected onto the simplex.

2. **Multiplicative sub‑gradient refinement** – a true sub‑gradient of the
   `max` is obtained from the active convolution entries.
   An exponential‑gradient (mirror‑descent) update `a_i ← a_i·e^{‑η·g_i}`
   keeps the iterates on the simplex.  Again the vector is forced to be
   symmetric.

Both phases are very cheap (FFT‑based convolutions) and can be repeated
with different random initialisations (Dirichlet, geometric, uniform,
previous best).  The best sequence found before the time budget expires
is returned.

The code below follows exactly the required API (`propose_candidate`);
all helper functions are top‑level, there are no closures and only
standard scientific libraries are used.

```python
"""
Two‑phase optimiser for 2·n·max(conv(a,a)) / (sum(a)**2).

Phase 1 – smooth optimisation of a log‑sum‑exp surrogate using projected
gradient descent on the probability simplex.  The temperature α is
annealed, the iterate is kept symmetric (a = (a + a[::-1])/2) and
projected onto the simplex after each step.

Phase 2 – multiplicative (exponential) sub‑gradient descent on the
exact maximum of the convolution, again with symmetry enforcement.
Multiple random restarts (uniform, Dirichlet, geometric, previous best)
are performed until the time budget runs out.  The best sequence
encountered is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog  # not used directly, kept for possible extensions


# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution of a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    # only the first 2·n‑1 entries are meaningful
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of the weight vector w (length 2·n‑1) with a (length n).
    Returns the segment that aligns with the n coordinates of a.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Phase 1 – smooth optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_optimize(
    init_a: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the simplex for the smooth surrogate

        M_α(a) = (1/α)·log Σ exp(α·conv(a,a)) .

    The temperature α is slowly increased (annealed).  The iterate is
    kept symmetric and projected onto the simplex after each step.
    Returns the best vector seen and its exact (non‑smooth) score.
    """
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())  # smallest power of two ≥ 2n‑1
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()

    alpha = 0.5               # start with a very smooth surrogate
    alpha_mul = 1.12          # multiplicative increase
    inc_every = 250           # iterations between temperature updates
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1

        # ----- convolution -------------------------------------------------
        conv = _full_conv(a, L)               # length 2·n‑1

        # ----- log‑sum‑exp weights ----------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)   # safe soft‑max
        w = exp_scaled / exp_scaled.sum()          # Σ w = 1

        # ----- gradient (4·n·correlation) ---------------------------------
        corr = _corr_w_a(w, a, L)               # length n
        grad = 4.0 * n * corr

        # ----- step size (simple diminishing schedule) --------------------
        lr = 0.8 / (1.0 + it * 0.001)           # a few tenths initially
        a -= lr * grad

        # ----- enforce symmetry -------------------------------------------
        a = (a + a[::-1]) / 2.0

        # ----- simplex projection -----------------------------------------
        a = _project_simplex(a)

        # ----- exact objective --------------------------------------------
        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"[smooth] iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # ----- occasional jitter -------------------------------------------
        if stagn > 3500:
            a += 1e-3 * rng.standard_normal(n)
            a = (a + a[::-1]) / 2.0
            a = _project_simplex(a)
            stagn = 0

        # ----- temperature annealing ---------------------------------------
        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 5000.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – exponential (mirror‑descent) sub‑gradient optimisation
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    init_a: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent on the true maximum of the
    aperiodic self‑convolution.  The iterate stays on the simplex
    (a = soft‑max of logits) and is forced symmetric after each update.
    """
    a = init_a.copy()
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # initialise best record
    max_val, _ = _maxconv_and_subgrad(a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    it = 0
    stagn = 0
    while time.time() < deadline:
        it += 1

        max_val, grad = _maxconv_and_subgrad(a, fft_len)

        # multiplicative update (mirror‑descent on the simplex)
        lr = 0.6 / np.sqrt(it)          # diminishing step size
        a = a * np.exp(-lr * grad)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # enforce symmetry
        a = (a + a[::-1]) / 2.0

        # re‑evaluate exact objective
        max_val, _ = _maxconv_and_subgrad(a, fft_len)
        score = 2.0 * n * max_val

        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"[exp‑grad] iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # occasional random jitter to escape plateaus
        if stagn > 5000:
            a += 5e-4 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            a = (a + a[::-1]) / 2.0
            stagn = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (used by both phases)
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Returns (max_val, subgrad) for the aperiodic self‑convolution of a.
    The sub‑gradient corresponds to the average of the gradients of all
    active (max‑attaining) convolution entries.
    """
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    # tolerant active set (numerical noise)
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        # contribution 2·a_{k-i} (the i‑th component of the gradient)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad
```
