"""
Improved optimiser for the step‑function bound

The code searches for a non‑negative sequence a (∑a = 1) that minimises
    2·n·max(conv(a,a))
where conv is the aperiodic self‑convolution.
It combines three ideas:

*  A fast smooth surrogate (log‑sum‑exp) minimised by projected gradient
   descent with an aggressive annealing schedule (Stage 1).

*  Adam on the true max‑convolution (sub‑gradient) to polish the result
   (Stage 2).

*  A tiny stochastic local‑search that moves a random amount of mass from
   one coordinate to another and keeps any improvement (Stage 3).

Several initialisations (uniform, geometric, Dirichlet, piecewise constant)
are tried on a few problem sizes; the best sequence found before the
time‑budget expires is returned as a plain ``list`` of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution of a, length 2·n‑1 (FFT length L)."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """Correlation of w (length 2·n‑1) with a (length n)."""
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Stage 1 – smooth optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_optimize(init_a: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the simplex minimising the soft‑max surrogate of
    the max‑convolution.  The temperature α is increased slowly.
    Returns the best vector (sum = 1) seen and its exact (non‑smooth) score.
    """
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())            # next power of two
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()

    # annealing schedule
    alpha = 0.5                # start very smooth
    inc_every = 200            # iterations between α updates
    alpha_mul = 1.12
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1

        # convolution
        conv = _full_conv(a, L)

        # soft‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()     # Σ w = 1

        # gradient of the surrogate (4·n·correlation)
        corr = _corr_w_a(w, a, L)
        grad = 4.0 * n * corr

        # step size – a simple decreasing schedule with a tiny momentum term
        lr = 0.6 / (1.0 + it * 0.001)
        a -= lr * grad

        # projection back to simplex
        a = _project_simplex(a)

        # exact objective
        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"Smooth  it {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # occasional jitter to escape plateaus
        if stagn > 3000:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        # anneal α
        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 2500.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Return (max_conv, sub‑gradient) of the aperiodic self‑convolution.
    The sub‑gradient is the average of the gradients contributed by all
    active (max‑attaining) shift positions.
    """
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Stage 2 – Adam optimiser on the true max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   deadline: float,
                   restart_id: int,
                   target: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the sub‑gradient of the exact objective.
    Stops when the global target is reached or the deadline expires.
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # work with logits – this guarantees positivity and handles the simplex
    eps = 1e-12
    b = np.log(init_a + eps)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initial evaluation
    a0 = np.exp(b - b.max())
    a0 /= a0.sum()
    max_val, _ = _maxconv_and_subgrad(a0, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # probabilities from logits
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        # sub‑gradient of the max‑convolution
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} it {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # early exit if we beat the target
        if best_score < target:
            break

        # occasional random jitter when stuck
        if stagn > 6000:
            b += 0.01 * np.random.randn(n)
            stagn = 0

        # Adam update (gradient w.r.t. logits)
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)                # ⟨a,∇f⟩
        grad_b = a * (grad_obj - avg)            # chain rule for soft‑max

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.35 / np.sqrt(it)                  # decaying learning rate
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – tiny stochastic local refinement
# ----------------------------------------------------------------------
def _refine_local(a: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a small amount of mass between two coordinates,
    keeping any improvement.  Very cheap, but often squeezes a few extra
    1e‑4 from the objective.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())

    best_a = a.copy()
    conv = _full_conv(best_a, L)
    best_score = 2.0 * n * conv.max()

    step_factor = 0.2                # initial maximal transfer as a fraction of source mass

    while time.time() < deadline:
        src, tgt = rng.integers(0, n, size=2, endpoint=False)
        if src == tgt:
            continue

        max_delta = step_factor * best_a[src]
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = best_a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        if a_new[src] < -1e-15:                 # guard against numerical drift
            continue

        conv = _full_conv(a_new, L)
        score = 2.0 * n * conv.max()

        if score < best_score:
            best_score = score
            best_a = a_new
            # shrink the step – we are approaching a plateau
            step_factor = max(0.01, step_factor * 0.98)
        else:
            # occasionally enlarge the step to jump out of flat regions
            if rng.random() < 0.01:
                step_factor = min(0.5, step_factor * 1.04)

    return best_a, best_score
