"""
Propose a low‑discrepancy step function via a multi‑stage global optimiser.

The search proceeds as follows (all time‑limited by a global deadline):

1.  For a set of lengths (256 … 4096) a *differential evolution* (DE)
    population is initialised with several deterministic seeds
    (uniform, arcsine, triangular, a few Beta‑shapes) plus random
    Dirichlet samples.  Each DE generation creates a mutant,
    recombines it with the target, projects the trial onto the
    probability simplex (Σa = 1, a_i ≥ 0) and keeps it if it improves the
    fast surrogate score 2·n·max(conv(a,a)).  The best individual of the
    run is retained.

2.  The DE winner is refined by (i) exact sub‑gradient descent on the
    true non‑smooth objective 2·n·max(conv) and (ii) Adam optimisation on
    a temperature‑controlled soft‑max approximation of the max.  Both
    steps respect the simplex constraint.

3.  The refined vector is finally evaluated with the official
    ``evaluate_sequence`` function; the best result over all lengths is
    stored.

4.  A short global polishing on the overall best candidate (sub‑gradient
    + Adam) is performed before returning.

All helper functions are top‑level, there are no closures or lambda
expressions, and the routine respects the 1 CPU / 1000 s budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score when Σa = 1:
    f(a) = 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1 / √((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) before simplex projection.
    """
    n = a.size
    conv = convolution_fft(a, a)               # length 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)            # sum to 1

    # weighted shift of a
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length n + (2·n‑1) - 1 = 3·n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected) and its surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule for the soft‑max
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the true objective 2·n·max(conv).
    Uses a diminishing step size η_t = step0 / sqrt(t) and simplex projection.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def hill_climb_local(a: np.ndarray,
                     deadline: float,
                     max_iter: int = 30_000,
                     step_frac: float = 0.15) -> np.ndarray:
    """
    Deterministic hill‑climbing: repeatedly move a small fraction of mass
    from a random index to another if it strictly reduces the surrogate.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    n = cur.size
    rng = np.random.default_rng()

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * step_frac * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = compute_score(cand)

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc

    return cur


def differential_evolution(initial_pop: List[np.ndarray],
                           deadline: float,
                           max_gens: int = 2000,
                           F: float = 0.8,
                           Cr: float = 0.7) -> Tuple[np.ndarray, float]:
    """
    Simple DE on the simplex:
    - Each individual is projected onto the simplex after mutation/crossover.
    - The surrogate score 2·n·max(conv) is used for selection.
    Returns the best individual and its score.
    """
    pop = [project_to_simplex(v) for v in initial_pop]
    scores = [compute_score(v) for v in pop]
    popsize = len(pop)
    n = pop[0].size
    rng = np.random.default_rng()

    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    for gen in range(1, max_gens + 1):
        if time.time() >= deadline:
            break
        for i in range(popsize):
            # sample three distinct individuals
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, 3, replace=False)
            a = pop[a_idx]
            b = pop[b_idx]
            c = pop[c_idx]

            # mutation
            mutant = a + F * (b - c)

            # crossover
            cross = rng.random(n) < Cr
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])

            # projection onto simplex
            trial = project_to_simplex(trial)

            trial_score = compute_score(trial)

            # selection
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score

                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()

        # occasional reporting
        if gen % 200 == 0:
            print(f"    DE gen={gen} best={best_score:.6f}")

    return best_vec, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine combines differential evolution, sub‑gradient descent,
    Adam on a smooth surrogate, and a final hill‑climbing polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a fallback if a previous solution is available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Main search over several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Length n = {n} ===")

        # ---------- 1a. Build initial DE population ----------
        popsize = max(30, int(0.01 * n))
        initial_pop: List[np.ndarray] = []

        # deterministic seeds
        seeds = [
            np.full(n, 1.0 / n),
            arcsine_initial(n),
            generate_triangular_shape(n, n // 2),
        ]
        # a few random Beta / power shapes
        for _ in range(6):
            alpha = rng.uniform(0.1, 5.0)
            beta = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, alpha, beta, eps))

        for _ in range(4):
            p = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # add the deterministic seeds to the population
        for sv in seeds:
            initial_pop.append(sv)

        # fill the rest with random Dirichlet vectors
        while len(initial_pop) < popsize:
            vec = rng.dirichlet(np.ones(n))
            initial_pop.append(vec)

        # ---------- 1b. Differential Evolution ----------
        remaining = deadline - time.time()
        max_de_gens = int(min(3000, remaining * 30))
        print(f"  DE: popsize={popsize}, max_gens={max_de_gens}")
        de_best_vec, de_best_score = differential_evolution(
            initial_pop,
            deadline,
            max_gens=max_de_gens,
            F=0.8,
            Cr=0.7,
        )
        print(f"  → DE best surrogate score = {de_best_score:.6f}")

        # ---------- 1c. Sub‑gradient refinement ----------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_sub_iters = int(min(80_000, remaining * 25))
            print(f"  Sub‑gradient descent (max_iters={max_sub_iters}) …")
            de_best_vec = subgradient_descent_exact(
                de_best_vec,
                deadline,
                max_iter=max_sub_iters,
                step0=0.045,
            )

        # ---------- 1d. Adam on smooth surrogate ----------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_adam_iters = int(min(100_000, remaining * 25))
            print(f"  Adam smooth (max_iters={max_adam_iters}) …")
            de_best_vec, _ = adam_optimize_smooth(
                de_best_vec,
                deadline,
                max_iter=max_adam_iters,
                lr0=0.12,
                lam_start=2.0,
                lam_end=80.0,
                noise_every=2500,
                noise_scale=3e-5,
            )

        # ---------- 1e. Local hill‑climbing ----------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_hc_iters = int(min(30_000, remaining * 20))
            print(f"  Hill‑climbing (max_iters={max_hc_iters}) …")
            de_best_vec = hill_climb_local(
                de_best_vec,
                deadline,
                max_iter=max_hc_iters,
                step_frac=0.12,
            )

        # ---------- 1f. Official evaluation ----------
        try:
            val = evaluate_sequence(de_best_vec.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = de_best_vec.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2.  Global polishing of the overall best
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing (sub‑gradient + Adam + hill‑climb)…")
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        # allocate roughly half the remaining budget to each phase
        it_sub = int(min(120_000, remaining * 12))
        it_adam = int(min(120_000, remaining * 12))
        it_hc = int(min(40_000, remaining * 4))

        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.03)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=it_adam,
            lr0=0.09,
            lam_start=2.0,
            lam_end=80.0,
            noise_every=2000,
            noise_scale=2e-5,
        )
        a = hill_climb_local(a, deadline, max_iter=it_hc, step_frac=0.1)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
