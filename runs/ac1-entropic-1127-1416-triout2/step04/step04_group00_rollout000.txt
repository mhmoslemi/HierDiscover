"""
Propose a low‑score step function.

Algorithm
---------
1. **Large random sparse subsets** –  
   Pick a large length `n` (up to 32768) and a density `p` (≈ 0.15‑0.5).
   Choose `k = round(p·n)` distinct positions and put equal mass `1/k`
   on those positions (the rest are zero).  The vector sums to 1, so
   the evaluation score is exactly `2·n·max(a*a)`.

   For large `n` the distribution of pair‑sums becomes very close to
   uniform, giving a score only a few percent above the theoretical
   optimum `≈ 1`.  Fast FFT based convolution (`compute_score`) makes it
   possible to test thousands of random candidates within the 1000 s
   budget.

2. **Fallback & final evaluation** –  
   If a previous solution (`height_sequence_1`) is available we start
   from it.  After the random search the best candidate is evaluated
   with the official `evaluate_sequence` function and returned.

The method needs only the public `evaluate_sequence` and runs entirely
on the CPU without any file or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    if x <= 0:
        return 1
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of two real vectors using a real FFT.
    Returns a NumPy array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum(a)=1.
    Returns 2·n·max(a∗a) where * denotes the linear convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------

    FFT‑based score; the best candidate found is finally returned after
    a call to the official `evaluate_sequence` for verification.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a tiny safety margin for the final evaluation
    deadline = start + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a previously known candidate if it exists (fallback)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Random sparse‑subset search on large lengths
    # ------------------------------------------------------------------
    # Lengths we are willing to try – larger lengths give a tighter
    # distribution of sums but make the final exact convolution cheaper
    # than > 40000 elements.
    lengths = [16384, 32768]

    # Density interval – lower bound avoids tiny k (high variance),
    # upper bound keeps k well below n so that the convolution is not
    # dominated by the triangular shape of the dense case.
    min_density = 0.15

    it = 0
    while time.time() < deadline - 5.0:
        # --------------------------------------------------------------
        # Sample a problem size and a density

        n = int(rng.choice(lengths))

        if k > n:
