def gradient_descent(a, deadline, max_iter=100000, lr0=0.1, beta_start=5.0, beta_inc=1.01, beta_max=500.0):
    best_a = a.copy()
    best_score = compute_score(a)
    a_rev = a[::-1].copy()
    beta = beta_start
    for t in range(1, max_iter+1):
        if time.time() > deadline: break
        conv = convolution_fft(a, a)
        # compute softmax weights
        # to prevent overflow:
        z = beta * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        w = exp_z / np.sum(exp_z)

        # compute gradient via conv of w and a_rev
        grad = 2.0 * convolution_fft(w, a[::-1])
        # take slice n-1: n-1+n = 2n-1? Actually convolution length = len(w)+len(a)-1 = (2n-1)+n-1 = 3n-2.
        # Use slice
        n = len(a)
        grad = grad[n-1 : n-1 + n]  # size n

        step = lr0 / np.sqrt(t)
        a_new = a - step * grad
        # enforce non-neg via projection
        a_new = np.maximum(a_new, 0)
        a_new = project_to_simplex(a_new)

        # evaluate
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        # optionally increase beta gradually
        if t % 2000 == 0:
            beta = min(beta * beta_inc, beta_max)

    return best_a, best_score

"""
Improved step‑function search.

The algorithm works in three stages:

1. **Random sparse start** – pick a random support of size k≈p·n
   (p∈[0.25,0.75]), put unit mass on that support (uniform or
   Dirichlet‑distributed) and normalise to sum = 1.

2. **Soft‑max gradient descent** – minimise a smooth approximation of the
   objective 2 n·max(a∗a).  For a temperature β the loss is
   Lβ(a)=log‑sum‑exp(β·conv(a,a)).  Its gradient can be written as a
   convolution, so each iteration needs only two FFTs (a∗a and
   w∗a_rev).  β is increased during optimisation so the procedure first
   flattens the whole spectrum and then concentrates on reducing the true
   maximum.  After each update the vector is projected onto the simplex
   (non‑negative, sum = 1).

3. **Fast hill‑climb** – a final stochastic hill‑climber (swap or transfer
   a small fraction of mass) polishes the best vector found.

All work respects the 1000 s budget.  The routine returns a plain Python
list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` when the sum of ``a`` equals 1.
    Returns 2·n·max(a∗a) – exactly the value returned by
    ``evaluate_sequence`` for a normalised vector.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Soft‑max based gradient descent on the max‑convolution objective
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
    lr0: float = 0.15,
    beta_start: float = 1.0,
    beta_factor: float = 1.02,
    beta_max: float = 400.0,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed objective
        Lβ(a) = log‑sum‑exp(β·conv(a,a))

    The gradient is 2·(w * a_rev) where
        w_k = exp(β·conv_k) / Σexp(β·conv)
    and a_rev is a reversed copy of ``a``.
    After each step the vector is projected onto the probability simplex.
    """
    a = a_init.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    beta = beta_start
    a_rev = a[::-1]          # will be refreshed each iteration

    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        # --- auto‑convolution ------------------------------------------------
        conv = convolution_fft(a, a)               # length 2n‑1

        # --- soft‑max weights ------------------------------------------------
        z = beta * conv
        max_z = np.max(z)
        w = np.exp(z - max_z)
        w /= np.sum(w)                              # shape 2n‑1

        # --- gradient (two FFTs total) --------------------------------------
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)      # length 3n‑2
        grad = grad_full[n - 1 : n - 1 + n]         # slice of length n

        # --- step (diminishing) --------------------------------------------
        step = lr0 / np.sqrt(t)
        a_new = a - step * grad

        # --- keep feasible ---------------------------------------------------
        a_new = np.maximum(a_new, 0.0)
        a_new = project_to_simplex(a_new)

        # --- keep the best vector seen --------------------------------------
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # --- annealing of β -------------------------------------------------
        if t % 1000 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # occasional tiny jitter to escape flat plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (lr0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps: int = 200_000):
    """
    Stochastic hill‑climb that either swaps two entries or transfers a
    fraction of mass from one position to another.  The true objective
    ``compute_score`` is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:               # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.6)   # increase step on success
        else:
            step_scale *= 0.96                         # decrease on failure

        max_steps -= 1

    return best_a.tolist(), best_score
