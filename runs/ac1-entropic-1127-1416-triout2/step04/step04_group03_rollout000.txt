def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0
    best_seq, best_val = None, float('inf')
    # if height_sequence_1 present
    try:
        fallback = [float(x) for x in height_sequence_1]
    except Exception:
        fallback = None
    if fallback:
        val = evaluate_sequence(fallback)
        if val < best_val:
            best_seq, best_val = fallback, val

    # exploration candidate lengths
    lengths = [1024, 2048, 4096, 8192] # maybe adjust based on time
    for n in lengths:
        if time.time() >= deadline - 5: break
        # starters
        candidates = []
        candidates.append(np.full(n, 1.0/n))  # uniform
        candidates.append(arcsine_initial(n))
        candidates.append(gaussian_shape(n, sigma=n/8.0)) # maybe
        # random beta shapes
        max_samples = 1200
        for _ in range(max_samples):
            if time.time() >= deadline - 5: break
            alpha = rng.uniform(0.2, 2.0)  # expand range
            beta = rng.uniform(0.2, 2.0)
            eps = rng.uniform(1e-10, 0.01)
            cand = generate_beta_shape(n, alpha, beta, eps)
            candidates.append(cand)
        # optionally Gaussian variations
        for sigma_scale in [0.12, 0.15, 0.2]:
            if time.time() >= deadline - 5: break
            cand = gaussian_shape(n, sigma=int(sigma_scale*n))
            candidates.append(cand)

        # find best by fast surrogate
        best_local = None; best_local_score = float('inf')
        for cand in candidates:
            sc = compute_score(cand)
            if sc < best_local_score:
                best_local_score = sc
                best_local = cand

        # refinement
        if best_local is not None and time.time() < deadline - 2:
            # allocate for Adam
            remaining = deadline - time.time()
            max_iter_adam = int(min(400_000, remaining * 45))
            refined_a, _ = adam_optimize(best_local, time_deadline=deadline,

                                         lam_start=1.0, lam_end=120.0,

                                                   step0=0.22, max_iter=max_iter_sub)

            # random local search
            if time.time() < deadline - 2:
                max_iter_loc = int(min(250_000, (deadline - time.time()) * 35))
                refined_a, _ = random_local_search(refined_a, time_deadline=deadline,
                                                   init_step=0.015, max_iter=max_iter_loc)

            # evaluate official
            val = evaluate_sequence(refined_a.tolist())
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"New global best: {best_val:.6f} at n={n}")

    # final polishing on best_seq if any time left
    if best_seq and time.time() < deadline - 5:
        a = np.array(best_seq, dtype=float)
        # run extra Adam polish
        remaining = deadline - time.time()
        max_iter_adam = int(min(500_000, remaining * 50))
        a, _ = adam_optimize(a, time_deadline=deadline,

        # subgradient

"""
Propose a non‑negative step sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a))²

The algorithm improves on the previous baseline by

* searching a richer family of symmetric shapes (beta, Gaussian, sine,
  arcsine) on several lengths,
* enforcing exact symmetry after every update (the optimal solution is
  symmetric),
* using longer optimisation runs (Adam → sub‑gradient → stochastic local
  search) with aggressive schedules,
* allocating the remaining time to a final polishing phase.

All heavy work uses fast FFT based convolutions and a cheap surrogate
`2·n·max(conv)`; the official `evaluate_sequence` is called only for the
few final candidates.
"""

import time
import math
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def enforce_symmetry(v: np.ndarray) -> np.ndarray:
    """Make a vector symmetric by averaging with its reversal."""
    return (v + v[::-1]) * 0.5


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluation (a is already normalised
    Σa = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape factories
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine density (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    The returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """
    Symmetric discrete Gaussian shape centred at (n‑1)/2.
    sigma is the standard deviation in index units.
    """
    i = np.arange(n, dtype=float)
    centre = (n - 1) * 0.5
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def sin_shape(n: int) -> np.ndarray:
    """Simple sine‑shaped density (symmetric, zero at the ends)."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n)
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)           # length 2·n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                 # soft‑max weights, sum to 1
    B = max_c + np.log(sum_exp) / lam      # smooth approximation of max

    # Weighted gradient of the quadratic form:
    # ∂/∂a_i Σ_k w_k (a*a)_k = 2·Σ_k w_k a_{k-i}
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)   # length 2·n‑1
    # extract the n relevant entries (centered)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.18,
    lam_start: float = 1.0,
    lam_end: float = 120.0,
    noise_every: int = 1200,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth surrogate.  Returns the best vector
    (projected onto the simplex and symmetrised) and its fast surrogate
    score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a = enforce_symmetry(a)
    a = project_to_simplex(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic schedule for the temperature λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)
        a_new = enforce_symmetry(a_new)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.22,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    The step size decays like step0 / sqrt(t).
    """
    a = a0.copy()
    a = enforce_symmetry(a)
    a = project_to_simplex(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term (a_i * a_{idx-i})
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)
        a_new = enforce_symmetry(a_new)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex with Gaussian perturbations.
    Accepts improving moves and occasional uphill moves for escape.
    """
    a = a0.copy()
    a = enforce_symmetry(a)
    a = project_to_simplex(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # decaying step size
        step = init_step / np.sqrt(it)

        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)
        a_prop = enforce_symmetry(a_prop)
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        # Accept improvements, otherwise a very small uphill probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(
                f"    Random local iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine explores several lengths, samples a rich family of
    symmetric shapes, refines the best candidates with Adam → sub‑gradient →
    stochastic local search and finally performs a long polishing run.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a small safety margin so we never overrun the time limit
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback to a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths
    # ------------------------------------------------------------------
    # lengths are chosen to balance discretisation error and speed
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ---- 1a. Deterministic starter shapes (uniform, arcsine, Gaussian,
        #      sine) -------------------------------------------------
        starters = [
            np.full(n, 1.0 / n),           # uniform
            arcsine_initial(n),            # classic arcsine
            gaussian_shape(n, sigma=n / 8.0),  # moderately wide Gaussian
            sin_shape(n),                  # sine shape
        ]

        local_best_a = None
        local_best_score = float("inf")
        for st in starters:
            sc = compute_score(st)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = st.copy()

        # ---- 1b. Random sampling of symmetric beta shapes -------------
        max_samples = 1200
        for _ in range(max_samples):
            if time.time() >= deadline - 8.0:
                break
            # draw parameters from a wide (log‑uniform) range
            alpha = rng.uniform(0.15, 2.2)
            beta = rng.uniform(0.15, 2.2)
            eps = rng.uniform(1e-10, 0.02 * n)
            cand = generate_beta_shape(n, alpha, beta, eps)
            sc = compute_score(cand)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = cand

        # ---- 1c. Additional Gaussian variations ------------------------
        for sigma_frac in (0.09, 0.12, 0.16):
            if time.time() >= deadline - 8.0:
                break
            sigma = sigma_frac * n
            cand = gaussian_shape(n, sigma)
            sc = compute_score(cand)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = cand

        # --------------------------------------------------------------
        # 2.  Refinement of the locally best candidate
        # --------------------------------------------------------------
        if local_best_a is not None and time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            # --- Adam -------------------------------------------------
            max_iter_adam = int(min(350_000, remaining * 45))
            print(f"  Refining with Adam (max_iter={max_iter_adam}) …")
            refined_a, _ = adam_optimize(
                initial_a=local_best_a,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.18,
                lam_start=1.0,
                lam_end=120.0,
                noise_every=1200,
                noise_scale=4e-5,
            )

            # --- Sub‑gradient -----------------------------------------
            if time.time() < deadline - 2.0:
                max_iter_sub = int(min(180_000, (deadline - time.time()) * 30))
                refined_a, _ = subgradient_optimize(
                    refined_a,
                    time_deadline=deadline,
                    step0=0.22,
                    max_iter=max_iter_sub,
                )

            # --- Random local search ----------------------------------
            if time.time() < deadline - 2.0:
                max_iter_loc = int(min(250_000, (deadline - time.time()) * 35))
                refined_a, _ = random_local_search(
                    refined_a,
                    time_deadline=deadline,
                    init_step=0.015,
                    max_iter=max_iter_loc,
                )

            # --- Official evaluation ----------------------------------
            val = evaluate_sequence(refined_a.tolist())
            print(f"  → refined official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:
            # fallback to raw best candidate if refinement never entered
            if local_best_a is not None:
                val = evaluate_sequence(local_best_a.tolist())
                print(f"  Raw candidate official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = local_best_a.tolist()
                    print("*** Updated global best (raw) ***")

    # ------------------------------------------------------------------
    # 3.  Final polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Final polishing of the global best ===")
        a = np.array(best_seq, dtype=float)

        # Stage A – long Adam run
        remaining = deadline - time.time()
        max_iter_adam = int(min(500_000, remaining * 50))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.14,
            lam_start=1.0,
            lam_end=150.0,
            noise_every=1000,
            noise_scale=2e-5,
        )

        # Stage B – sub‑gradient polishing
        if time.time() < deadline - 3.0:
            max_iter_sub = int(min(250_000, (deadline - time.time()) * 35))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.18,
                max_iter=max_iter_sub,
            )

        # Stage C – stochastic local polishing
        if time.time() < deadline - 2.0:
            max_iter_loc = int(min(300_000, (deadline - time.time()) * 45))
            print(f"  Random local polishing (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.01,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # Simple safe fallback (uniform 256‑vector)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
