```python
"""
Hybrid optimiser for the step‑function bound.

We start from a feasible vector (usually the best known
`height_sequence_1`), normalise it to the probability simplex and run a
short smooth gradient‑descent phase.  Afterwards a linear‑programming
refinement is applied: for the current vector f we solve

    max  ∑ g_i
  s.t.  f * g ≤ M,   g_i ≥ 0

where `*` denotes aperiodic convolution and `M = max(f*f)`.  The resulting
g has a larger ℓ₁‑norm while respecting the same peak‑convolution bound.
A convex combination of f and the normalised g (or g itself) is kept only
if it improves the true objective `evaluate_sequence`.  The whole process
is repeated with several random restarts until the time budget (≈1000 s)
is exhausted.

The algorithm is purely deterministic except for the choice of random
initial vectors; it uses only NumPy, SciPy and the provided
`evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based aperiodic self‑convolution (full length 2·n‑1)
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Return full convolution a*a using an FFT of length L (power of two)."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Smooth optimisation – log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _smooth_optimize(init_a: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Gradient descent on the simplex minimising the true objective
    (2·n·max(conv)) but using a soft‑max surrogate that is annealed.
    """
    a = init_a.copy()
    n = a.shape[0]
    # make a a probability vector (sum = 1)
    a = _project_simplex(a)
    L = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2·n‑2
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()
    alpha = 0.5
    inc_every = 300
    alpha_mul = 1.15
    it = 0
    stagn = 0
    while time.time() < deadline:
        it += 1
        conv = _full_conv(a, L)
        scaled = alpha * conv
        # safe soft‑max
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()
        # correlation of w with a (length‑n segment)
        fft_w = np.fft.rfft(w, n=L)
        fft_a = np.fft.rfft(a, n=L)
        corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
        n_a = n
        corr = corr[n_a - 1 : n_a - 1 + n_a]  # segment of length n
        grad = 4.0 * n * corr
        lr = 0.7 / (1.0 + it * 0.001)
        a -= lr * grad
        a = _project_simplex(a)
        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"Smooth iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1
        if stagn > 4000:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0
        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 2000.0)
    return best_a


# ----------------------------------------------------------------------
# Helper that builds the constraint matrix for the LP
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, active_shifts: np.ndarray) -> lil_matrix:
    """
    Return a sparse matrix A such that (A @ g)[r] = (f * g)[active_shifts[r]].
    Each row contains the coefficients f_i that multiply the corresponding
    entry of g.
    """
    n = f.shape[0]
    rows = active_shifts.shape[0]
    A = lil_matrix((rows, n), dtype=np.float64)
    for r, s in enumerate(active_shifts):
        i_min = max(0, s - (n - 1))
        i_max = min(s, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = s - i_vals
        A[r, j_vals] = f[i_vals]
    return A


# ----------------------------------------------------------------------
# Linear‑programming refinement step
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform a sequence of LP refinements starting from f (which is assumed
    to be a probability vector).  The loop stops when no improvement is
    observed or when the time budget expires.
    """
    n = f.shape[0]
    best_a = f.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"LP start score {best_score:.6f}")

    while time.time() < deadline - 0.5:
        # compute current convolution and active set
        conv_full = _full_conv(best_a, 1 << ((2 * n - 2).bit_length()))
        M = conv_full.max()
        # active constraints – those within numerical tolerance of the max
        tol = max(M * 1e-12, 1e-14)
        active = np.where(conv_full >= M - tol)[0]

        # If the active set is huge we truncate it (the most critical rows)
        if active.size > 200:
            # keep the 200 shifts with largest values (closest to M)
            order = np.argsort(-(conv_full[active]))
            active = active[order[:200]]

        # Build a sparse matrix for the selected rows
        A = _build_constraint_matrix(best_a, active).tocsr()

        # Linear programme: maximise sum(g)  ⇔ minimise –sum(g)
        c = -np.ones(n)
        b_ub = np.full(A.shape[0], M)
        bounds = [(0, None)] * n

        # Solve with HiGHS (fast, works with sparse matrices)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print("LP solver failed – terminating LP refinement.")
            break

        g = res.x
        if not np.isfinite(g).all():
            print("LP produced non‑finite vector – terminating.")
            break

        sum_g = g.sum()
        if sum_g <= 1e-12:
            print("LP returned near‑zero vector – terminating.")
            break

        g_norm = g / sum_g

        improved = False
        # try a few convex combinations of the old and the new vector
        for t in (0.5, 0.2, 0.1, 0.05):
            candidate = (1.0 - t) * best_a + t * g_norm
            candidate = candidate / candidate.sum()
            cand_score = evaluate_sequence(candidate.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_a = candidate
                improved = True
                print(f"LP blend t={t:.2f} improved score to {best_score:.6f}")
                break
        if not improved:
            # try the normalised LP solution itself
            cand_score = evaluate_sequence(g_norm.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_a = g_norm
                improved = True
                print(f"LP direct update improved score to {best_score:.6f}")

        if not improved:
            # nothing helped – we are at a (local) optimum for this
            # refinement scheme
            print("LP step could not improve – stopping refinement.")
            break

    return best_a

# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list:

    Returns a plain list of non‑negative floats whose evaluation score
    is as low as possible within the allocated time budget.

    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Determine problem size (number of steps)
    # --------------------------------------------------------------
    try:

        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = prev.shape[0]
        init_from_prev = True
    except Exception:

        n = 4000
        prev = None
        init_from_prev = False

    # Helper to produce a fresh feasible start (probability simplex)
    # --------------------------------------------------------------

            a = prev.copy()

                else:
                    a = np.concatenate([a, np.zeros(n - a.shape[0])])
            a = np.maximum(a, 0.0)
            s = a.sum()

                a = a / s
            else:
                a = np.full(n, 1.0 / n)
            return a
        if kind == "uniform":
            return np.full(n, 1.0 / n)
        if kind == "dirichlet":
            # sparse Dirichlet, α=0.5 gives many small entries
            x = rng.gamma(0.5, 1.0, size=n)
            return x / x.sum()
        if kind == "geometric":

            left = r ** np.arange(n)
            right = r ** np.arange(n - 1, -1, -1)
            a = left + right
            return a / a.sum()

        return np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Global bookkeeping
    # --------------------------------------------------------------
    best_overall = None
    best_score = float("inf")
    restart_id = 0
    start_strategies = ["prev", "uniform", "dirichlet", "geometric"]

    if not init_from_prev:
        start_strategies = [s for s in start_strategies if s != "prev"]

    # Main optimisation loop – several restarts until time runs out
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        kind = start_strategies[restart_id % len(start_strategies)]

        # ----- generate a feasible start point -----------------------
        a0 = _make_start(kind)

        # ----- allocate a slice of time for this restart -------------
        remaining = deadline - time.time()
        # give at most half of the remaining budget to the smooth phase,
        # but not less than 10 seconds (to keep the LP step sensible)
        smooth_deadline = time.time() + max(10.0, min(30.0, 0.45 * remaining))

        # ----- Stage 1: smooth optimisation (log‑sum‑exp) ------------
        a_smooth = _smooth_optimize(a0, smooth_deadline, rng)

        # ----- Stage 2: LP refinement ---------------------------------
        # give the rest of the allocated slice to the LP part
        lp_deadline = deadline - 0.5   # keep a tiny margin for the final return
        a_refined = _lp_refine(a_smooth, lp_deadline)

        # ----- evaluate the candidate --------------------------------
        cand_score = evaluate_sequence(a_refined.tolist())
        print(f"Restart {restart_id:02d} ({kind}) final score {cand_score:.6f}")

            best_score = cand_score
            best_overall = a_refined.copy()
            print(f"*** New global best {best_score:.6f} after restart {restart_id}")

            # early exit if we beat the target
            if best_score < 1.503:
                print(f"Target reached (score={best_score:.6f}) after {restart_id+1} restarts.")
                break

        restart_id += 1

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return value
    # ------------------------------------------------------------------
    if best_overall is None:
        # Should never happen, but keep a safe default
        best_overall = np.full(n, 1.0 / n)

    # Ensure plain Python list of floats (the evaluator expects this)
    return [float(v) for v in best_overall.tolist()]
```assistantfinal**Solution Overview**

The evaluator computes  

\[
\text{score}(a)=\frac{2\,n\;\max\_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

The score is *scale‑invariant*, therefore we keep the vector on the
probability simplex (\(\sum a_i = 1\)).  
Two complementary optimisation ideas are combined:

1. **Smooth gradient descent** – a log‑sum‑exp surrogate of the max‑convolution is
   minimised on the simplex.  This quickly yields a good starting vector.  

2. **Linear‑programming refinement** – for the current vector \(f\) we compute  

   \[
   M=\max_k (f*f)_k .
   \]

   The linear programme  

   \[
   \max\limits_{g\ge0}\;\sum_i g_i\quad
   \text{s.t.}\;\;(f*g)_k\le M\;\;\forall k
   \]

   finds a feasible vector \(g\) that can carry *more* ℓ₁‑mass while still
   respecting the same peak‑convolution bound.  
   A convex combination of \(f\) and the normalised \(g\) (or \(g\) itself) is kept
   only if it lowers the true score.  Only the *active* constraints – the shifts
   where \((f*f)_k\) equals \(M\) – are used, which makes the LP very small
   (a few hundred rows) even for several thousand variables.

The whole procedure is wrapped in a time‑bounded loop with several random
restarts (uniform, Dirichlet, geometric, and the previously found best
sequence).  The best sequence seen before the deadline is returned.

All helper functions are top‑level, no closures are used, and only
`numpy`, `scipy` and the supplied `evaluate_sequence` are required.

```python
"""
Hybrid optimiser for the step‑function bound.

* smooth gradient descent on a log‑sum‑exp surrogate (simplex‑projected GD)
* LP refinement: maximise ℓ₁‑norm of a vector g under the linear constraints
  f * g ≤ M, where M = max(f * f).  Only the active constraints are kept.
* several random restarts; the best sequence found before the 1000 s budget
  is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based full aperiodic self‑convolution (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using an FFT of length L (a power of two)."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Smooth optimisation – log‑sum‑exp surrogate (annealed)
# ----------------------------------------------------------------------
def _smooth_optimize(init_a: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> np.ndarray:
    """Gradient descent on the simplex using a soft‑max surrogate of max(conv)."""
    a = _project_simplex(init_a.copy())
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2·n‑2
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()

    alpha = 0.5
    inc_every = 300
    alpha_mul = 1.15
    it = stagn = 0

    while time.time() < deadline:
        it += 1

        # ----- convolution -------------------------------------------------
        conv = _full_conv(a, L)                # length 2·n‑1

        # ----- soft‑max weights -------------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)   # safe
        w = exp_scaled / exp_scaled.sum()          # Σ w = 1

        # ----- gradient (4·n·correlation) ---------------------------------
        fft_w = np.fft.rfft(w, n=L)
        fft_a = np.fft.rfft(a, n=L)
        corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
        corr = corr[n - 1 : n - 1 + n]            # the n‑length segment
        grad = 4.0 * n * corr

        # ----- step & projection -------------------------------------------
        lr = 0.7 / (1.0 + it * 0.001)
        a -= lr * grad
        a = _project_simplex(a)

        # ----- exact objective ---------------------------------------------
        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"Smooth iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # ----- occasional jitter -------------------------------------------
        if stagn > 4000:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        # ----- anneal temperature -------------------------------------------
        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 2000.0)

    return best_a


# ----------------------------------------------------------------------
# Build the sparse constraint matrix for the LP
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray,
                             active_shifts: np.ndarray) -> lil_matrix:
    """
    Return a sparse matrix A (|active| × n) such that
        (A @ g)[r] = (f * g)[active_shifts[r]] .
    Each row contains the coefficients f_i that multiply the appropriate
    entry of g.
    """
    n = f.shape[0]
    rows = active_shifts.shape[0]
    A = lil_matrix((rows, n), dtype=np.float64)

    for r, s in enumerate(active_shifts):
        i_min = max(0, s - (n - 1))
        i_max = min(s, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = s - i_vals                  # indices of g that appear
        A[r, j_vals] = f[i_vals]

    return A


# ----------------------------------------------------------------------
# Linear‑programming refinement step
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly solve the LP

        max  Σ g_i
        s.t. f * g ≤ M ,  g ≥ 0

    where M = max(f * f).  The LP is built only for the *active*
    shifts (those attaining the maximum).  After each successful LP,
    a convex combination of the old and the new vector (or the new
    vector alone) is kept if it strictly improves the true score.
    """
    n = f.shape[0]
    best_a = f.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"LP start score {best_score:.6f}")

    while time.time() < deadline - 0.5:
        # ---- current convolution and active constraints -----------------
        conv_full = _full_conv(best_a,
                               1 << ((2 * n - 2).bit_length()))
        M = conv_full.max()
        tol = max(M * 1e-12, 1e-14)
        active = np.where(conv_full >= M - tol)[0]

        # limit the number of rows to keep the LP tiny
        if active.size > 200:
            order = np.argsort(-(conv_full[active]))
            active = active[order[:200]]

        # ---- build sparse matrix A --------------------------------------
        A = _build_constraint_matrix(best_a, active).tocsr()

        # ---- solve the LP (max Σ g_i  ⇔  min –Σ g_i) --------------------
        c = -np.ones(n)                       # objective
        b_ub = np.full(A.shape[0], M)         # RHS: M for each active shift
        bounds = [(0, None)] * n               # g_i ≥ 0

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True})

        if not res.success:
            print("LP solver failed – stopping refinement.")
            break

        g = res.x
        if not np.isfinite(g).all():
            print("LP gave non‑finite solution – stopping.")
            break

        sum_g = g.sum()
        if sum_g <= 1e-12:
            print("LP returned near‑zero vector – stopping.")
            break

        g_norm = g / sum_g

        improved = False
        # ---- try a few convex blends of old and new ----------------------
        for t in (0.5, 0.2, 0.1, 0.05):
            cand = (1.0 - t) * best_a + t * g_norm
            cand = cand / cand.sum()
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand
                improved = True
                print(f"LP blend t={t:.2f} improved to {best_score:.6f}")
                break

        if not improved:
            # ---- maybe the normalised LP solution itself is better --------
            cand_score = evaluate_sequence(g_norm.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_a = g_norm
                improved = True
                print(f"LP direct update improved to {best_score:.6f}")

        if not improved:
            # nothing helped – we reached a local optimum for this scheme
            print("LP step produced no improvement – terminating LP loop.")
            break

    return best_a
