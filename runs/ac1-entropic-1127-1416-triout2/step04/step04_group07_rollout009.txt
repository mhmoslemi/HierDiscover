    best_score = evaluate_sequence(a.tolist())
    while time.time() < deadline:
        # random pair
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose move: fraction of a[i] transferred to a[j]
        delta = step * a[i] * rng.random()

            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalize sum slight different? sum unchanged.
        # ensure nonneg boolean check
        # compute score
        score = evaluate_sequence(a_new.tolist())

            a = a_new

"""
Improved step‑function optimiser
================================

The goal is to minimise

    2 * n * max(conv(a,a)) / (sum(a)**2)

for a non‑negative vector `a`.  The code follows the successful
two‑stage scheme from the reference implementation but adds a few
simple but effective ideas:

* a larger and more diverse set of initialisations,
* a longer annealing schedule for the smooth (log‑sum‑exp) phase,
* a more aggressive learning‑rate schedule,
* a final “hill‑climbing’’ refinement that moves a tiny amount of mass
  between two coordinates if it improves the exact objective,
* a two‑phase time allocation: many short restarts to explore the
  space, then a long deterministic refinement of the best candidate
  found.

All helper functions are top‑level, the optimiser works on the
probability simplex (the vector is always normalised to sum = 1,
hence the objective reduces to `2·n·max(conv)`), and the returned
list can be fed directly to the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Full (linear) self‑convolution of a, length 2·n‑1, using FFT length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of the weight vector w (length 2·n‑1) with a (length n).
    Returns the segment of length n that corresponds to shifts k=0…n‑1.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Stage 1 – smooth optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_optimize(
    init_a: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the simplex minimising the surrogate
        2·n·M_α(a),   M_α = (1/α)·log Σ exp(α·conv(a,a))

    The temperature α is increased gradually (annealing) until a large
    value (≈1e4) so that the surrogate converges to the true max.
    The function returns the best vector found (still normalised to sum = 1)
    together with its exact (non‑smoothed) score.
    """
    a = init_a.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    # best so far (exact objective, not the surrogate)
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    # annealing schedule – start very smooth and finish very sharp
    alpha = 0.5
    alpha_mul = 1.18
    max_alpha = 15000.0
    inc_every = 250          # increase α every this many iterations

    it = 0
    stagn = 0                # number of iterations without improvement

    while time.time() < deadline:
        it += 1

        # ----- convolution -------------------------------------------------
        conv = _full_conv(a, L)                # length 2·n‑1

        # ----- soft‑max weights -------------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)   # safe numerics
        w = exp_scaled / exp_scaled.sum()          # Σ w = 1

        # ----- gradient (4·n·correlation) ---------------------------------
        corr = _corr_w_a(w, a, L)               # length n
        grad = 4.0 * n * corr

        # ----- step size – 1/√t schedule (more aggressive early on) -----
        lr = 0.6 / np.sqrt(it)

        a -= lr * grad
        a = _project_simplex(a)                 # keeps sum(a)=1

        # ----- exact objective (no smoothing) ----------------------------
        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"[Smooth] it={it:6d}   best={best_score:.6f}")
        else:
            stagn += 1

        # ----- occasional jitter to escape flat plateaus ---------------
        if stagn > 3000:
            a += 1e-4 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        # ----- anneal α ---------------------------------------------------
        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, max_alpha)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Adam optimiser on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max_conv, sub‑gradient) for the aperiodic self‑convolution."""
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


def _adam_optimize(
    init_a: np.ndarray, deadline: float, restart_id: int, target: float = 1.50
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the exact max‑convolution.
    Returns the best vector seen before the deadline.
    """
    n = init_a.shape[0]
    fft_len = _next_pow2(2 * n - 1)

    # logits (log‑softmax) – a tiny epsilon avoids log(0)
    eps = 1e-12
    b = np.log(init_a + eps)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # current best (exact objective)
    a_cur = np.exp(b - b.max())
    a_cur /= a_cur.sum()
    max_val, _ = _maxconv_and_subgrad(a_cur, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a_cur.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- probabilities from logits ---------------------------------
        b_max = b.max()
        a = np.exp(b - b_max)
        a /= a.sum()

        # ----- exact objective & true sub‑gradient -----------------------
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"[Adam {restart_id:02d}] it={it:6d}   best={best_score:.6f}")
        else:
            stagn += 1

        if best_score < target:
            break

        if stagn > 6000:
            # gentle random jitter in log‑space
            b += 0.005 * np.random.randn(n)
            stagn = 0

        # ----- Adam update (gradient of original objective) -------------
        grad_obj = 2.0 * n * grad_a
        # projected gradient on the simplex (the usual trick for softmax)
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.45 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_a, best_score


# ----------------------------------------------------------------------
# Local “hill‑climbing’’ refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _local_search_optimize(
    init_a: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic coordinate descent: repeatedly move a tiny
    amount of probability mass from a random source index to a random
    destination index.  The move is kept only if the exact objective
    improves.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_score = evaluate_sequence(a.tolist())
    best_a = a.copy()
    step_factor = 0.12            # typical relative size of a move
    no_improve = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] == 0.0:
            continue

        # propose moving a random fraction of a[i] to a[j]
        delta = step_factor * a[i] * rng.random()
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # numerical safety
        if a_new[i] < 0:
            continue

        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new
            a = a_new
            no_improve = 0
            # when we find an improvement we tighten the step a bit
            step_factor = max(0.02, step_factor * 0.96)
        else:
            no_improve += 1
            if no_improve > 2000:
                # slowly increase the step size again to explore
                step_factor = min(0.3, step_factor * 1.03)
                no_improve = 0

    return best_a, best_score

        # ---- generate a feasible start point ---------------------------------
        if kind == "prev" and previous is not None:
            a0 = previous.copy()
            if a0.shape[0] != n:
                if a0.shape[0] > n:

        elif kind == "uniform":
            a0 = np.full(n, 1.0 / n)
        elif kind == "random_dirichlet":
            # α = 0.5 gives a fairly sparse start

            a0 = sample / sample.sum()
        elif kind == "geometric":
            r = 0.97 + 0.03 * rng.random()
            left = r ** np.arange(n)
            right = r ** np.arange(n - 1, -1, -1)
            a0 = left + right
            a0 = a0 / a0.sum()
        elif kind == "sinusoid":
            a0 = np.sin(np.pi * (np.arange(n) + 0.5) / n)

            best_overall_a = a_fine.copy()
            print(
