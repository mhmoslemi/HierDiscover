def full_correlation(w, a):
    # w length m, a length n, returns length m+n-1 correlation r where
    # r[l] = sum_{j=0}^{m-1} w[j] * a[j+l] (zero pad a).
    L = int(2**np.ceil(np.log2(len(w) + len(a) - 1)))
    # compute FFT of w and a
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a, n=L)
    # Compute correlation: ifft(fw * conj(fa))
    r = np.fft.irfft(fw * np.conj(fa), n=L)
    # crop to length len(w)+len(a)-1
    return r[:len(w)+len(a)-1]

Thus gradient = 2 * r[:n] because we need first n entries.

"""
search algorithm for ``evaluate_sequence``

The objective for a vector a (sum(a)=1) is

    score = 2·n·max(conv) ,   conv = a * a   (linear convolution)

We minimise the maximum of the self‑convolution.  The code below
optimises a smooth approximation of the max (log‑sum‑exp) with
Adam on the probability simplex.  The simplex is enforced by the
standard “softmax’’ re‑parameterisation  a_i = exp(p_i) / Σ_j exp(p_j).

A softmax temperature α is increased during the run: when α is small
the objective is smooth and the gradient points towards reducing the
overall mass of the convolution; when α becomes large the weighted
average concentrates on the peak and the method approaches a
sub‑gradient descent for the true max.  All convolutions and
correlations are computed with FFTs (O(n log n)) so the optimiser can
run many thousands of iterations within the 1000 s budget.

Occasional random perturbations of the log‑variables p help to escape
local minima.  The best vector seen (still normalised to sum‑1) is
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of a and b using FFT with zero‑padding to ``length``.
    ``length`` must be a power of two ≥ len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, n=length)
    fb = np.fft.rfft(b, n=length)
    conv = np.fft.irfft(fa * fb, n=length)
    return conv


def _max_conv_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector whose sum is 1.
    Returns 2·n·max(conv) where conv = a * a (linear convolution).
    """
    n = a.shape[0]
    length = _next_pow2(2 * n - 1)
    conv = _full_convolution_fft(a, a, length)[: 2 * n - 1]
    return 2.0 * n * conv.max()

# Main optimiser
# ----------------------------------------------------------------------

        # reuse the same FFT length L_corr (it is ≥ 2n‑1)
        fft_a = np.fft.rfft(a, n=L_corr)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L_corr)[: 2 * n - 1]

        # --------------------------------------------------------------
        # smooth‑max weights w_k  (softmax over conv values)
        # --------------------------------------------------------------
        # shift by the maximum to avoid overflow
        shift = conv_full.max()
        exp_arg = alpha * (conv_full - shift)
        # clip to avoid numerical under‑flow / overflow
        exp_arg = np.clip(exp_arg, -700.0, 700.0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                     # extreme case – fall back to uniform
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # --------------------------------------------------------------
        # gradient of the smooth‑max objective w.r.t. a
        # --------------------------------------------------------------
        # we need  grad_a[i] = 2 Σ_k w_k a_{k-i}
        # this can be written as 2 * (a ⋆ w)[i]  where ⋆ denotes the
        # correlation that can be obtained via a convolution with w reversed.
        w_rev = w[::-1]                       # length 2n‑1
        # convolution a * w_rev (full length n + (2n‑1) - 1 = 3n‑2)
        fft_w_rev = np.fft.rfft(w_rev, n=L_corr)
        corr_full = np.fft.irfft(fft_a * fft_w_rev, n=L_corr)
        # indices for grad_a:   grad_i = 2 * corr_full[(2n‑2) - i]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * corr_full[idx]

        # --------------------------------------------------------------
        # back‑propagation through the softmax re‑parameterisation:
        #   p  -> a  -> objective
        # --------------------------------------------------------------
        # derivative w.r.t. a is grad_a.
        # derivative w.r.t. p (logits) under the simplex constraint:
        #   ∂f/∂p_i = a_i * (grad_a_i - Σ_j a_j grad_a_j)
        dot = np.dot(a, grad_a)
        grad_p = a * (grad_a - dot)

        # --------------------------------------------------------------
        # Adam update on p
        # --------------------------------------------------------------
        it += 1
        lr = base_lr / math.sqrt(it)         # a gentle decay

        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        p -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # schedule for α (smoothness) – slowly become the true max
        # --------------------------------------------------------------
        if it % 200 == 0 and alpha < max_alpha:
            alpha = min(alpha * alpha_mul, max_alpha)

        # --------------------------------------------------------------
        # occasional random perturbation to promote exploration
        # --------------------------------------------------------------
        if it % 5000 == 0:
            # add a small Gaussian jitter to the logits
            p += np.random.randn(n) * 0.05

        # --------------------------------------------------------------
        # keep the best (exact) score seen so far
        # --------------------------------------------------------------
        if it % 100 == 0:
            cur_score = 2.0 * n * conv_full.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Iter {it:6d}: new best score {best_score:.6f} (α={alpha:.1f})")

    # ------------------------------------------------------------------
    # 3 – final reporting
    # ------------------------------------------------------------------
    print(f"Optimisation finished after {it} iterations; best score = {best_score:.6f}")

    # Normalise once more before returning (the evaluation function is
    # scale‑invariant, but we keep sum == 1 for safety).
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
