def softmax_weights(conv, beta):
    # conv: np.ndarray
    # returns weight vector w of same shape summing to 1
    c = conv * beta
    cmax = np.max(c)
    exp_c = np.exp(c - cmax)
    sum_exp = np.sum(exp_c)
    return exp_c / sum_exp
```

def annealed_descent(a, deadline, max_iters=20000, eta0=0.5, beta0=1.0, beta_factor=1.02):
    a = normalize(a)
    best_a = a.copy()
    best_sc = compute_score(a)

            best_a = a.copy()
    return best_a, best_sc

"""
Hybrid annealed exponential‑gradient + sub‑gradient search.

The algorithm works in two stages:
  1.  Fast exploration on a moderate length (n≈1024).  Many low‑dimensional
      parametric families are sampled, the best few are refined by an
      annealed “soft‑max” exponential‑gradient descent that uses a smooth
      approximation of the max‑convolution (softmax with increasing temperature).
  2.  The best solution is up‑sampled to a larger length (n≈2048) and refined
      again with the same annealed routine (more iterations) and finally with
      a classic sub‑gradient mirror‑descent (as in the reference code) for a
      last polishing step.

All updates keep the vector on the probability simplex (sum = 1) and are
strictly non‑negative, so the returned sequence is always feasible.
The routine respects the supplied time budget (default 1000 s) and prints
intermediate scores.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def generate_initial_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Create a modest library of diverse start vectors."""
    cand: List[np.ndarray] = []

    # deterministic families
    cand.append(uniform_shape(n))
    cand.append(arcsine_initial(n))

    # random power‑law shapes
    for _ in range(8):
        p = rng.uniform(0.3, 3.0)                     # allow heavier tails
        eps = rng.uniform(0.0, max(5.0, 0.01 * n))
        cand.append(powerlaw_shape(n, p, eps))

    # symmetric beta‑like shapes
    for _ in range(5):
        a = rng.uniform(0.2, 6.0)
        cand.append(beta_shape(n, a))

    # mixtures of arcsine & uniform
    for _ in range(5):
        w = rng.uniform(0.0, 0.9)                     # weight of uniform part
        mix = (1.0 - w) * arcsine_initial(n) + w * uniform_shape(n)
        cand.append(normalize(mix))

    # mixtures of arcsine & a random power‑law
    for _ in range(5):
        w = rng.uniform(0.0, 0.9)
        p = rng.uniform(0.3, 3.0)
        eps = rng.uniform(0.0, max(5.0, 0.01 * n))
        power = powerlaw_shape(n, p, eps)
        mix = (1.0 - w) * arcsine_initial(n) + w * power
        cand.append(normalize(mix))

    # random Dirichlet draws
    for _ in range(5):
        cand.append(random_dirichlet(n, rng))

    return cand


# ----------------------------------------------------------------------
# Soft‑max based annealed exponential‑gradient descent
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute a soft‑max weight vector w_i ∝ exp(beta * conv_i).
    Returns a normalized vector (sum = 1).
    """
    if beta <= 0.0:
        # uniform weights when temperature is infinite
        return np.full_like(conv, 1.0 / len(conv))

    log_w = beta * conv
    # subtract max for numerical stability
    log_w -= np.max(log_w)
    exp_w = np.exp(log_w)
    sum_exp = np.sum(exp_w)
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / len(conv))
    return exp_w / sum_exp


def annealed_exponential_descent(
    a0: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    beta0: float = 1.0,
    beta_factor: float = 1.02,
    beta_max: float = 200.0,
) -> Tuple[np.ndarray, float]:
    """
    Annealed exponential‑gradient (mirror‑descent) optimisation.
    Starts with a low temperature (beta0) and gradually increases it,
    turning the soft‑max into an approximation of the true max.
    Returns the best iterate (according to the fast score) and its score.
    """
    a = normalize(a0.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1
    beta = beta0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # convolution a * a
        conv = convolution_fft(a, a)

        # soft‑max weight vector for the current temperature
        w = softmax_weights(conv, beta)

        # gradient of Σ w_i·conv_i
        grad = gradient_from_weights(a, w)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # multiplicative (exponentiated) update
        factor = np.exp(-eta * grad)
        # safety clipping
        factor = np.clip(factor, 1e-30, 1e30)
        a = a * factor
        a = normalize(a)

        # increase temperature (tighten the max approximation)
        beta = min(beta * beta_factor, beta_max)

        # occasional evaluation of the true objective
        if it % 1000 == 0 or it == max_iter:
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient (mirror‑descent) refinement – same as the reference code
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution objective.
    Multiplicative updates keep the vector on the simplex.
    Returns the best iterate and its (exact) score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # binary weights for all maximising indices (ties)
        w = np.zeros(conv_len, dtype=float)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Utility: up‑sample a probability vector to a larger length
# ----------------------------------------------------------------------


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to length ``new_n``.
    The returned vector is re‑normalised to sum to 1.
    """
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    interpolated = np.interp(x_new, x_old, seq)
    return normalize(interpolated)
