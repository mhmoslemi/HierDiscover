def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5

    best_seq = None
    best_score = float('inf')

    # optional: fallback
    if 'height_sequence_1' in globals():
        try:
            fallback = [float(x) for x in globals()['height_sequence_1']]
            val = evaluate_sequence(fallback)
            print("Fallback score:", val)
            if val < best_score:
                best_seq = fallback
                best_score = val
        except Exception:
            pass

    length_options = [2048, 4096, 8192]

    for n in length_options:
        if time.time() >= deadline - 10:
            break
        print(f"--- Searching for length {n} ---")
        # Stage 0: generate candidate list
        candidates = []

        # 0a: arcsine
        a_arcsine = arcsine_initial(n)
        candidates.append(a_arcsine)

        # 0b: uniform
        candidates.append(np.full(n, 1.0/n))

        # 0c: symmetric beta grid
        alphas = np.linspace(0.2, 2.0, 30)  # 30 values
        for alpha in alphas:
            beta = alpha  # symmetric
            for eps in [1e-6, 0.01, 0.1]:
                cand = generate_beta_shape(n, alpha, beta, eps)
                candidates.append(cand)

        # 0d: random beta sampling (few hundred)
        for _ in range(400):
            alpha = rng.uniform(0.05, 2.0)
            beta = rng.uniform(0.05, 2.0)
            eps = rng.uniform(1e-6, 0.02*n)
            candidates.append(generate_beta_shape(n, alpha, beta, eps))

        # deduplicate maybe.

        # Evaluate fast scores and keep top few
        top_k = 10
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        best_candidates = [c for _,c in scored[:top_k]]
        best_candidate_scores = [s for s,_ in scored[:top_k]]

"""
Propose a step‑function (non‑negative vector) that minimizes
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2).

The algorithm works as follows:

1. **Exploratory parametric search** – for each candidate length we
   generate a modest set of symmetric beta‑family shapes
   a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}  (including the classic arcsine
   α=β=0.5) and keep the one with the smallest *fast* score
   `2·n·max(conv(a,a))`.  The search is cheap because only a few hundred
   candidates are evaluated.

2. **Intensive refinement** – the best candidate for a length is refined
   by three stages:
      * Adam optimisation on a smooth log‑sum‑exp surrogate of the
        max‑convolution (the temperature λ is increased from 1 to a
        large value, e.g. 200).  All iterates are projected onto the
        probability simplex.
      * Projected sub‑gradient descent on the *exact* max‑convolution.
      * Stochastic local search with Gaussian perturbations.

   The number of optimisation steps is controlled by the remaining
   time budget, so the routine always finishes before the 1000 s limit.

3. **Final polishing** – the globally best vector found across all
   lengths is polished once more with the three optimisation stages,
   using a larger iteration budget.

All helper functions are top‑level, use only NumPy and the standard
library, and respect the time budget.  The returned list is guaranteed
to be non‑empty, non‑negative and normalised (∑a = 1), therefore
`evaluate_sequence` never yields ``inf``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official evaluation: 2·n·max(conv(a,a)).
    The vector a is assumed to be already normalised (Σa = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)            # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # softmax weights (sum to 1)
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of max(conv)

    # Weighted sub‑gradient: ∂ Σ_k w_k (a*a)_k / ∂a = 2·(a * w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)  # should be 1 after projection
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 200.0,
    noise_every: int = 1200,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # λ increases from lam_start to lam_end (quadratic schedule)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)
        cur_sc = compute_score(a_new)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 120_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    The step size decays like step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.size

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # Gradient of the active quadratic term (active index idx)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 30000 == 0:
            print(f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex: add Gaussian noise, project back,
    keep improvements (tiny probability of uphill moves for escape).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # Accept improvements; allow occasional uphill moves for escape
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(f"    Random local iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
