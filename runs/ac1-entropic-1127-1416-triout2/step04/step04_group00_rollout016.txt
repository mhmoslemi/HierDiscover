"""
Propose a low‑score step function.

Algorithm outline
-----------------
1. **Random sparse‑subset init** – generate many candidates where a
   fraction `p` of the `n` positions get equal mass `1/k`.  Keep the best
   (lowest `evaluate_sequence` score).

2. **LP‑based alternating optimisation** – for the current best vector
   `a` compute `M = max(a*a)`.  Solve the linear program

       maximise   sum(b)
       subject to b ≥ 0,
                  (a ⊛ b)[ℓ] ≤ M  for all ℓ

   where `⊛` is the ordinary (full) convolution.  The LP is linear because
   `a` is constant.  The optimum `b` spreads the mass into positions where
   `a` is small, keeping the convolution bounded.  Mix the new `b` into
   `a` with a small step `t` and re‑project onto the probability simplex.
   Re‑iterate while time permits – this monotonically reduces the peak
   `max(a*a)` and therefore the evaluation score.

3. **Smooth sub‑gradient refinement** – a few hundred thousand sub‑gradient
   steps (the gradient of the maximal convolution entry) further lower the
   peak.

4. **Hill‑climb polishing** – stochastic swaps / mass‑transfer moves
   accepted only when they improve the score.

All steps respect the 1000 s budget and use only NumPy and CVXPY
(ECOS solver).  The returned list is a plain Python list of non‑negative
floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the exact max‑convolution
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the true objective
    2 n·max(a⊛a).  The sub‑gradient at the current maximal shift k
    is g[i] = 2·a[k‑i] (if the index is valid, otherwise 0).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming step (a ⊛ b ≤ M)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def solve_lp_for_b(a: np.ndarray, M: float):
    """
    Solve   maximise   1ᵀb   subject to   b ≥ 0,
                (a ⊛ b) ≤ M   (element‑wise)

    Returns the optimal b as a NumPy array, or None on failure.
    """
    if cp is None:
        return None
    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)
    # a is a constant NumPy array – `cp.conv` treats it as a constant term.
    conv_ab = cp.conv(a, b)          # length 2n‑1
    constraints = [conv_ab <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    except Exception as e:            # pragma: no cover
        print(f"LP solve exception: {e}")
        return None
    if b.value is None:
        return None
    return np.asarray(b.value, dtype=float)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

        p = np.random.uniform(0.20, 0.90)            # density of the support

        a = np.zeros(n, dtype=float)

    # ------------------------------------------------------------------
