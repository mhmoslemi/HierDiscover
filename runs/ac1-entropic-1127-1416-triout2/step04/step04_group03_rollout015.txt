def lp_refine(f, sqrt2n, rows_idx, cols_idx, deadline, max_iters=30):
    best_f = f.copy()
    best_M = max_self_conv(f)
    for it in range(max_iters):
        if time.time() >= deadline - 0.1: break
        M = max_self_conv(f)
        # Build constraint matrix
        data = np.tile(f, len(f))
        A = csr_matrix((data, (rows_idx, cols_idx)), shape=(2*len(f)-1, len(f)))
        # Solve LP
        c = -np.ones(len(f))
        b_ub = np.full(2*len(f)-1, M, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print(f"LP failed at iter {it}, status: {res.message}")
            break
        b = res.x
        sum_b = np.sum(b)
        if sum_b <= 0:
            break
        g = b * (sqrt2n / sum_b)
        f = 0.5 * (f + g)  # mixing t=0.5
        # Compute new M
        new_M = max_self_conv(f)
        if new_M < best_M - 1e-9:
            best_M = new_M
            best_f = f.copy()
        # Print progress:
        print(f"LP iter {it+1}: M={new_M:.6f}, best={best_M:.6f}")
    return best_f, best_M

"""
propose_candidate
=================

Search for a non‑negative step vector that minimizes

    2·n·max(a∗a) / (Σa)²

(the same value returned by ``evaluate_sequence``).  The algorithm
consists of three phases:

1. **Initialisation** – pick a good starting shape (arcsine density or
   a previously discovered best vector) and normalise it to
   Σa = √(2n).

2. **Alternating linear‑program refinement** – given the current vector
   *f* compute M = max(f∗f).  Solve the linear program

        maximise   Σ b_i
        subject to   (f ∗ b)_k ≤ M    for all k
                     b_i ≥ 0

   (the convolution in the constraints is linear in *b*).  After the LP
   we obtain *b*, rescale it to the same ℓ₁‑norm as *f*,
   and replace *f* by the average (f+g)/2 (with g the rescaled *b*).
   Each iteration provably cannot increase the objective and usually
   yields a noticeable drop.  The LP is solved with ``scipy.optimize.linprog``
   using a sparse convolution matrix, which makes an iteration cheap even
   for n≈2000.

3. **Fine‑grained stochastic polishing** – a short random‑walk on the
   simplex (Gaussian perturbations followed by Euclidean projection)
   refines the result further.

All work respects the overall time budget (default 1000 s).  The
function returns a regular Python list of floats that is guaranteed to be
non‑empty, non‑negative and normalised, so ``evaluate_sequence`` never
produces ``inf``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for the LP phase
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_self_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a∗a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Symmetric arcsine shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Linear‑program helpers (construction of the convolution matrix)
# ----------------------------------------------------------------------


def build_conv_matrix_indices(n: int):
    """
    Pre‑compute row/column index arrays for the *n*×*n* convolution matrix
    A such that (A @ b) = f ∗ b   (f is supplied later).
    """
    # i = row‑index inside f, j = column index (= position of b)
    i = np.arange(n, dtype=np.int32)[:, None]    # shape (n,1)
    j = np.arange(n, dtype=np.int32)[None, :]    # shape (1,n)

    rows = (i + j).ravel()                      # length n*n
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    return rows, cols


def lp_refine(
    f: np.ndarray,
    rows_idx: np.ndarray,
    cols_idx: np.ndarray,
    deadline: float,
    max_iters: int = 15,
    mix: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Alternating LP refinement described in the problem statement.
    Returns the best vector seen during the refinement and its
    max‑convolution value (which equals the official score when the
    ℓ₁‑norm is √(2n)).
    """
    n = f.size
    sqrt2n = math.sqrt(2.0 * n)

    best_f = f.copy()
    best_M = max_self_conv(f)

    for it in range(1, max_iters + 1):
        # time‑check
        if time.time() >= deadline - 0.5:
            break

        M = max_self_conv(f)

        # Build the sparse convolution matrix A_f for the *current* f
        data = np.tile(f, n)                     # repeats f for each column
        A = csr_matrix((data, (rows_idx, cols_idx)),
                       shape=(2 * n - 1, n))

        # Linear programme: maximise sum(b)  →  minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Linprog with the HiGHS interior‑point solver (fast for sparse data)
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=(0, None),          # b_i ≥ 0
            method="highs",
            options={"presolve": True},
        )

        if not res.success:            # pragma: no cover
            print(f"    LP failed at iter {it}: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Rescale to keep the ℓ₁‑norm equal to √(2n)
        scale = sqrt2n / sum_b
        g = b * scale

        # Convex combination (mix parameter can be tuned)
        f = (1 - mix) * f + mix * g

        # Update best record
        M_new = max_self_conv(f)
        if M_new < best_M:
            best_M = M_new
            best_f = f.copy()

        print(f"    LP iter {it:2d}:  max‑conv = {M_new:.6f}   best = {best_M:.6f}")

    return best_f, best_M


# ----------------------------------------------------------------------
# Simple stochastic local‑search (Gaussian perturbations on the simplex)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    init_step: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex: Gaussian noise + Euclidean projection.
    Keeps the best vector seen.  Very cheap and works well as a final
    polish after the LP phase.
    """
    n = a0.size
    sqrt2n = math.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_sc = max_self_conv(a)

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline - 0.1:
        it += 1
        step = init_step / math.sqrt(it)
        proposal = a + rng.normal(scale=step, size=n)
        # project onto the simplex with the required ℓ₁‑norm
        proposal = project_to_simplex(proposal)
        proposal *= sqrt2n     # enforce Σa = √(2n)

        sc = max_self_conv(proposal)
        if sc < best_sc:
            best_sc = sc
            best_a = proposal.copy()
            a = proposal
        else:
            # occasionally accept a non‑improving move to escape plateaus
            if rng.random() < 0.001:
                a = proposal

        if it % 5000 == 0:
            print(f"    Random walk iter {it:,d}:  cur = {sc:.6f}  best = {best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Determine the problem size (use the length of a previously known
    # good construction if it exists, otherwise fall back to 2048).
    # ------------------------------------------------------------------
    try:
        fallback = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        n = fallback.size
        if n == 0:
            raise ValueError
        print(f"Using length n = {n} from height_sequence_1")
    except Exception:
        n = 2048
        fallback = None
        print(f"No prior sequence – using default length n = {n}")

    sqrt2n = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build a set of initial candidates and pick the best (lowest max‑conv)
    # ------------------------------------------------------------------
    candidates = []

    # 1) arcsine shape (scaled to Σa = √(2n))
    arcsine = arcsine_initial(n) * (sqrt2n / np.sum(arcsine_initial(n)))
    candidates.append(arcsine)

    # 2) fallback sequence if available (also rescaled)
    if fallback is not None:
        fb = fallback.astype(float)
        fb = fb * (sqrt2n / fb.sum())
        candidates.append(fb)

    # 3) a few random Beta‑family samples (cheap, give diversity)
    for _ in range(5):
        alpha = rng.uniform(0.05, 1.5)
        beta = rng.uniform(0.05, 1.5)
        eps = rng.uniform(1e-6, 0.01)
        i = np.arange(n, dtype=float)
        left = (i + eps) ** (alpha - 1.0)
        right = (n - 1 - i + eps) ** (beta - 1.0)
        shape = left * right
        if shape.sum() == 0:
            continue
        shape = shape / shape.sum() * sqrt2n
        candidates.append(shape)

    best_f = None
    best_M = float("inf")
    for cand in candidates:
        M = max_self_conv(cand)
        if M < best_M:
            best_M = M
            best_f = cand.copy()
    assert best_f is not None

    print(f"Initial best max‑conv = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – alternating LP refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        rows_idx, cols_idx = build_conv_matrix_indices(n)
        # allocate a reasonable number of LP iterations; stop early if time runs out
        max_lp_iters = 12
        best_f, best_M = lp_refine(
            best_f,
            rows_idx,
            cols_idx,
            deadline,
            max_iters=max_lp_iters,
            mix=0.5,
        )
        print(f"After LP phase best max‑conv = {best_M:.6f}")
    else:
        print("SciPy not available – skipping LP phase")

    # ------------------------------------------------------------------
    # Phase 2 – brief random local search for polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        best_f, best_M = random_local_search(best_f, deadline, init_step=0.02)
        print(f"After random walk best max‑conv = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Convert to the format required by the official checker.
    # (The official checker normalises the vector internally, but we keep
    # the ℓ₁‑norm equal to √(2n) for consistency with the theory.)
    # ------------------------------------------------------------------
    final_seq = best_f.tolist()
    # final sanity check – the harness will reject empty lists
    if not final_seq:
        final_seq = [1.0] * n

    # Optional: one final call to the official evaluator to ensure we
    # don't accidentally produce an invalid vector.  (Never raises, just
    # prints the score for logging.)
    try:
        val = evaluate_sequence(final_seq)
        print(f"Final official score = {val:.6f}")
    except Exception:  # pragma: no cover
        pass

    return final_seq
