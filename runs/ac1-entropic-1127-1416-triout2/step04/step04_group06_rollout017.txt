"""
Propose a non‑negative step‑function (list of heights) that minimizes
evaluate_sequence().

Algorithm
---------
1. **Initialisation** – load the previous best sequence (if available),
   otherwise start from a uniform vector of length ≈5000.

2. **Exploratory families** – generate a wide variety of simple shapes
   (Beta‑densities, Hann / Blackman windows, triangular, edge‑heavy,
   sparse Dirichlet).  Each candidate is normalised to sum = 1 and its
   exact score (2·n·max(conv)) is evaluated.  The best candidate is kept.

3. **Phase A – smoothed‑max mirror descent** – starting from the current
   best vector run several restarts of exponentiated‑gradient descent on
   the smooth approximation
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
   The smoothing parameter α is increased from 1 up to 300 while the
   step‑size η decays like 1/√t.  All iterates stay on the probability
   simplex (multiplicative update + normalisation).  The best vector seen
   during the whole run is retained.

4. **Phase B – L₂‑flattening descent** – minimise the squared‑L₂ norm of
   the self‑convolution ( Σ_k (a∗a)_k² ).  Its gradient is
        g = 4·corr( a∗a , a )
   where `corr` denotes the (non‑circular) cross‑correlation.
   An additive gradient step with a decreasing step‑size is taken,
   followed by Euclidean projection onto the simplex.

5. **Phase C – hard‑max sub‑gradient descent** – repeatedly identify the
   index k where the convolution is maximal, treat this entry as a
   Dirac weight and perform a single exponentiated‑gradient step against
   the corresponding sub‑gradient.

6. **Phase D – guided simulated‑annealing** – local mass‑transfer moves
   that respect the simplex are performed with an exponential cooling
   schedule.  Moves that reduce the exact score are always accepted;
   otherwise they are accepted with the usual Metropolis probability.

The whole procedure respects the 1000 s budget; the best vector (already
normalised to sum = 1) is returned as a plain Python list.

# Basic linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (preserves zeros)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution / correlation utilities (FFT based)
# ----------------------------------------------------------------------


def self_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a∗a (length 2·n−1) using FFT.
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return conv


def correlation(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Non‑circular cross‑correlation (p ⋆ a) of lengths
    len(p)=2·n−1, len(a)=n.  Result has length n.
    """
    n = a.shape[0]
    L = 1 << ((len(p) + n - 2).bit_length())   # next power of two ≥ len(p)+n-1
    fft_p = np.fft.rfft(p, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_p * np.conj(fft_a), n=L)
    offset = len(p) - 1
    return corr[offset:offset + n]

# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector already normalised
    to sum = 1.  Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = self_convolution(a)
    return 2.0 * n * conv.max()

def init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=float)


def init_triangular(n: int) -> np.ndarray:
    """Symmetric linear “tent” shape."""
    idx = np.arange(n)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(idx - mid) / (mid + 1.0)
    a = np.maximum(a, 0.0)
    return simplex_normalise(a)


def init_hann(n: int) -> np.ndarray:
    """Classic Hann window – zero at both ends."""
    i = np.arange(n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return simplex_normalise(a)


def init_blackman(n: int) -> np.ndarray:
    """Blackman window – stronger edge tapering."""
    i = np.arange(n)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """Linear increase towards the two ends (used in the reference code)."""
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = dist_to_center.astype(float) + 1.0
    return simplex_normalise(a)


def init_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet(1,…,1) – i.i.d. Γ(1) = Exp(1)."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Put all mass on `k` randomly chosen positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on `n` points, normalised to sum = 1.
    Mid‑point rule is used to avoid the singularities at the boundaries.
    """
    x = (np.arange(n) + 0.5) / n
    # log‑density (stable for extreme α,β)
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow/underflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return simplex_normalise(w)


# ----------------------------------------------------------------------
# Smoothed‑max mirror‑descent (Phase A)
# ----------------------------------------------------------------------


def mirror_descent(
    a0: np.ndarray,
    max_iters: int,
    eta0: float,
    alpha_start: float,
    alpha_end: float,
    deadline: float,
    best_score_ref: list,
    best_vec_ref: list,
) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the soft‑max approximation of
    max(conv).  Updates stay on the simplex by multiplicative weights.
    Whenever a better value is found the external best references are updated.
    """
    n = a0.shape[0]
    a = a0.copy()
    L = 1 << ((3 * n - 2).bit_length())   # FFT length for cross‑correlation
    for it in range(max_iters):
        if time.time() > deadline - 2.0:       # leave a tiny margin
            break

        # 1) self‑convolution
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

        # 2) soft‑max weights over convolution entries
        maxc = conv.max()
        exp_vals = np.exp(alpha_start * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp

        # 3) gradient = 2·corr(p, a)
        fft_a_full = np.fft.rfft(a, n=L)
        fft_p = np.fft.rfft(p, n=L)
        corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L)
        offset = conv.size - 1
        g = 2.0 * corr[offset:offset + n]

        # 4) exponentiated‑gradient update
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)

        # 5) linear increase of smoothing parameter α
        alpha_start = alpha_start + (alpha_end - alpha_start) / max_iters

        # 6) occasional reporting
        if (it & 31) == 0:
            cur_score = compute_score(a)
            if cur_score < best_score_ref[0]:
                best_score_ref[0] = cur_score
                best_vec_ref[0] = a.copy()
                print(f"[Mirror] it {it:5d} → new best {cur_score:.6f}")

    return a


# ----------------------------------------------------------------------
# L₂‑flattening descent (Phase B)
# ----------------------------------------------------------------------


def l2_flatten_descent(
    a0: np.ndarray,
    max_iters: int,
    eta0: float,
    deadline: float,
    best_score_ref: list,
    best_vec_ref: list,
) -> np.ndarray:
    """
    Gradient descent on  Σ_k (a∗a)_k² .
    Gradient = 4·corr( a∗a , a ).
    An additive step followed by Euclidean projection onto the simplex is used.
    """
    n = a0.shape[0]
    a = a0.copy()
    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break

        # convolution and loss (loss itself is not needed, only gradient)
        conv = self_convolution(a)

        # gradient = 4·corr(conv, a)
        g = 4.0 * correlation(conv, a)

        # simple additive step with a decaying stepsize
        eta = eta0 / math.sqrt(it + 1.0)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)          # enforce non‑negativity
        a = project_to_simplex(a_new)           # keep Σ a = 1

        # occasional evaluation
        if (it & 31) == 0:
            cur_score = compute_score(a)
            if cur_score < best_score_ref[0]:
                best_score_ref[0] = cur_score
                best_vec_ref[0] = a.copy()
                print(f"[L2]   it {it:5d} → new best {cur_score:.6f}")

    return a


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient descent (Phase C)
# ----------------------------------------------------------------------


def hard_max_descent(
    a0: np.ndarray,
    max_iters: int,
    eta0: float,
    deadline: float,
    best_score_ref: list,
    best_vec_ref: list,
) -> np.ndarray:
    """
    At each iteration the index k of the maximal convolution entry is
    identified, a Dirac weight `p` is placed there and the resulting
    sub‑gradient (2·corr(p,a)) is used for an exponentiated‑gradient update.
    """
    n = a0.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    a = a0.copy()
    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break

        conv = self_convolution(a)
        k = int(conv.argmax())

        # Dirac weight at the maximal index
        p = np.zeros_like(conv)
        p[k] = 1.0

        # gradient = 2·corr(p, a)
        fft_a_full = np.fft.rfft(a, n=L)
        fft_p = np.fft.rfft(p, n=L)
        corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L)
        offset = conv.size - 1
        g = 2.0 * corr[offset:offset + n]

        # exponentiated‑gradient step
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)

        if (it & 31) == 0:
            cur_score = compute_score(a)
            if cur_score < best_score_ref[0]:
                best_score_ref[0] = cur_score
                best_vec_ref[0] = a.copy()
                print(f"[Hard]  it {it:5d} → new best {cur_score:.6f}")

    return a


# ----------------------------------------------------------------------
# Guided simulated‑annealing (Phase D)
# ----------------------------------------------------------------------


def simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    best_score_ref: list,
    best_vec_ref: list,
) -> np.ndarray:
    """Small, temperature‑controlled mass‑transfer walk."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_score = compute_score(cur_a)

    # temperature schedule
    T = cur_score * 0.02
    T_end = cur_score * 1e-5
    it = 0
    while time.time() < deadline - 0.5 and T > T_end:
        # pick source index proportionally to its current mass
        src = np.random.choice(n, p=cur_a)
        # pick a distinct target uniformly at random
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] <= 0.0:
            continue
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score_ref[0]:
                best_score_ref[0] = cand_score
                best_vec_ref[0] = cand.copy()
                print(f"[SA]   iter {it:6d} → new best {cand_score:.6f}")

        T *= 0.9995
        it += 1

    return cur_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (sum = 1) that (hopefully) minimises
    evaluate_sequence() within the allotted time.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # tiny safety margin

    # ------------------------------------------------------------------
    # 0 – load previous solution or fall back to a uniform start
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_vec.size
        best_a = simplex_normalise(init_vec)
        best_score = compute_score(best_a)
        print(f"[Init] Loaded previous vector (n={n}), score={best_score:.6f}")
    except NameError:
        n = 5146                       # default length used in earlier runs
        best_a = init_uniform(n)
        best_score = compute_score(best_a)
        print(f"[Init] No previous vector – using uniform start (n={n}), score={best_score:.6f}")

    # keep best values in mutable containers for inner functions
    best_score_ref = [best_score]
    best_vec_ref = [best_a.copy()]

    # ------------------------------------------------------------------
    # 1 – explore a rich collection of simple families
    # ------------------------------------------------------------------
    print("[Exploration] evaluating simple families ...")
    # Uniform
    cand = init_uniform(n)
    sc = compute_score(cand)
    if sc < best_score_ref[0]:
        best_score_ref[0] = sc
        best_vec_ref[0] = cand.copy()
        print(f"  uniform → new best {sc:.6f}")

    # Triangular
    cand = init_triangular(n)
    sc = compute_score(cand)
    if sc < best_score_ref[0]:
        best_score_ref[0] = sc
        best_vec_ref[0] = cand.copy()
        print(f"  triangular → new best {sc:.6f}")

    # Hann window
    cand = init_hann(n)
    sc = compute_score(cand)
    if sc < best_score_ref[0]:
        best_score_ref[0] = sc
        best_vec_ref[0] = cand.copy()
        print(f"  hann → new best {sc:.6f}")

    # Blackman window
    cand = init_blackman(n)
    sc = compute_score(cand)
    if sc < best_score_ref[0]:
        best_score_ref[0] = sc
        best_vec_ref[0] = cand.copy()
        print(f"  blackman → new best {sc:.6f}")

    # Edge‑concentrated linear
    cand = init_edge_concentrated(n)
    sc = compute_score(cand)
    if sc < best_score_ref[0]:
        best_score_ref[0] = sc
        best_vec_ref[0] = cand.copy()
        print(f"  edge‑linear → new best {sc:.6f}")

    # Random Dirichlet
    for _ in range(8):
        cand = init_dirichlet(n)
        sc = compute_score(cand)
        if sc < best_score_ref[0]:
            best_score_ref[0] = sc
            best_vec_ref[0] = cand.copy()
            print(f"  dirichlet → new best {sc:.6f}")

    # Sparse random (few non‑zeros)
    for k in (5, 10, 20, 30):
        cand = init_sparse(n, k=k)
        sc = compute_score(cand)
        if sc < best_score_ref[0]:
            best_score_ref[0] = sc
            best_vec_ref[0] = cand.copy()
            print(f"  sparse(k={k}) → new best {sc:.6f}")

    # Grid over symmetric Beta(α,α) and asymmetric Beta(α,β)
    alphas = np.arange(0.2, 6.01, 0.4)   # coarser grid for speed
    betas = np.arange(0.2, 6.01, 0.4)
    for a_ in alphas:
        for b_ in betas:
            if time.time() > deadline - 10.0:
                break
            cand = beta_sequence(a_, b_, n)
            sc = compute_score(cand)
            if sc < best_score_ref[0]:
                best_score_ref[0] = sc
                best_vec_ref[0] = cand.copy()
                print(f"  beta({a_:.1f},{b_:.1f}) → new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2 – Phase A: smoothed‑max mirror descent with several restarts
    # ------------------------------------------------------------------
    print("[Phase A] smoothed‑max mirror descent")
    restart = 0
    while time.time() < deadline - 30.0:
        # initialise a start vector – blend a random family with the current best
        choice = np.random.rand()
        if choice < 0.25:
            a_start = init_dirichlet(n)
        elif choice < 0.5:
            a_start = init_hann(n)
        elif choice < 0.75:
            a_start = init_blackman(n)
        else:
            a_start = init_edge_concentrated(n)

        # occasional mixing with the global best to keep good mass in the neighbourhood
        if np.random.rand() < 0.2:
            a_start = simplex_normalise(0.5 * a_start + 0.5 * best_vec_ref[0])

        a_start = simplex_normalise(a_start)

        a_start = mirror_descent(
            a0=a_start,
            max_iters=3000,
            eta0=0.9,
            alpha_start=1.0,
            alpha_end=300.0,
            deadline=deadline,
            best_score_ref=best_score_ref,
            best_vec_ref=best_vec_ref,
        )
        restart += 1
        print(f"[Phase A] restart {restart} finished, best score so far {best_score_ref[0]:.6f}")

    # ------------------------------------------------------------------
    # 3 – Phase B: L₂‑flattening descent (more direct flattening of conv)
    # ------------------------------------------------------------------
    print("[Phase B] L₂‑flattening descent")
    a_start = best_vec_ref[0].copy()
    a_start = l2_flatten_descent(
        a0=a_start,
        max_iters=4000,
        eta0=0.2,
        deadline=deadline,
        best_score_ref=best_score_ref,
        best_vec_ref=best_vec_ref,
    )

    # ------------------------------------------------------------------
    # 4 – Phase C: hard‑max sub‑gradient descent (focus on the current peak)
    # ------------------------------------------------------------------
    print("[Phase C] hard‑max sub‑gradient descent")
    a_start = hard_max_descent(
        a0=a_start,
        max_iters=2000,
        eta0=0.5,
        deadline=deadline,
        best_score_ref=best_score_ref,
        best_vec_ref=best_vec_ref,
    )

    # ------------------------------------------------------------------
    # 5 – Phase D: final guided simulated‑annealing refinement
    # ------------------------------------------------------------------
    print("[Phase D] simulated annealing")
    _ = simulated_annealing(
        a0=a_start,
        deadline=deadline,
        best_score_ref=best_score_ref,
        best_vec_ref=best_vec_ref,
    )

    # ------------------------------------------------------------------
    # 6 – Return the best vector found (as a plain Python list)
    # ------------------------------------------------------------------
    final_score = best_score_ref[0]
    print(f"[Done] best score after all phases: {final_score:.6f}")
    return [float(v) for v in best_vec_ref[0].tolist()]
