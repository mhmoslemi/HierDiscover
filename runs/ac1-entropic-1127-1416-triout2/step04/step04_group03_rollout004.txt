def solve_lp_max_sum(f, M):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f[i]
    b = cp.Variable(n)
    objective = cp.Maximize(cp.sum(b))
    constraints = [b >= 0,
                   A @ b <= M * np.ones(2*n-1)]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)  # adjust iteration limit
    if b.value is None:
        # fallback: return uniform
        return np.full(n, 1.0/n)
    return np.array(b.value).flatten()

"""
Propose a low‑discrepancy step sequence.

The algorithm mixes three complementary ideas:

1. **Smooth gradient descent (Adam)** on a log‑sum‑exp surrogate of the
   max self‑convolution.  This quickly drives a random beta‑family or the
   classic arcsine shape towards a good local optimum.

2. **Linear‑programming refinement** – given the current vector `f`
   we solve
  
        max   Σ b_j
        s.t.  b ≥ 0,
              (f ★ b)_k ≤ M   for all k,

   where `M = max(f ★ f)`.  The solution `b` is a “dense” partner that
   fills the convolution bound.  Mixing `f` with the normalised `b`
   (controlled by a decreasing step size) provably never increases the
   objective and often yields a noticeable drop in the max‑convolution.

   The LP is solved with SciPy’s HiGHS interior‑point method; the
   convolution matrix is built once per length and reused.

3. **Final polishing** – a short Adam run followed by a tiny stochastic
   walk finishes the search.

The routine explores a few lengths (1024 and 2048), keeps the best
candidate and returns a non‑empty list of non‑negative numbers.  The
whole search respects a 1000 s budget.
"""

import time
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)) for a already normalised
    (Σa = 1).  This equals the official `evaluate_sequence` value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """Symmetrised arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric beta‑family density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and Adam optimiser
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # softmax weights, sum to 1
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of max

    # weighted sum of sub‑gradients:
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    noise_every: int = 1500,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature (square‑root schedule → gentle ramp‑up)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

# Linear programming refinement
# ----------------------------------------------------------------------


# cache for the (row, col) pattern used to build the Toeplitz‑like matrix
_conv_template_cache = {}


def _conv_template(n: int) -> Tuple[np.ndarray, np.ndarray]:
    """
    Return (row_idx, col_idx) describing the linear convolution matrix
    for a fixed length n.  Each column i contains n copies of the same row
    index block (i,…,i+n‑1); the data will be `f[i]` repeated n times.
    """
    if n not in _conv_template_cache:
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        for i in range(n):
            start = i * n
            rows[start:start + n] = np.arange(i, i + n, dtype=np.int32)
            cols[start:start + n] = i
        _conv_template_cache[n] = (rows, cols)
    return _conv_template_cache[n]


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP
        max Σ b_j
        s.t.   b ≥ 0,
               (f ★ b)_k ≤ M  ∀k.
    Returns the optimal b (may be dense).  If the LP fails, returns a
    uniform distribution.
    """
    n = len(f)
    rows, cols = _conv_template(n)
    data = np.repeat(f, n)                     # length n²
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Upper‑bound vector (all entries equal to M)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Maximise Σ b → minimise -Σ b
    c = -np.ones(n, dtype=float)

    # Use HiGHS (fast interior‑point).  We turn off printing.
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=(0, None),
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )

    if res.success and res.x is not None:
        return np.array(res.x, copy=True)
    else:
        # fallback – uniform vector
        return np.full(n, 1.0 / n)


def lp_refine(
    a: np.ndarray,
    steps: int,
    t0: float,
    time_deadline: float,
) -> np.ndarray:
    """
    Apply `steps` LP refinements to vector `a`.
    In each step we:
        * compute M = max(a★a),
        * solve the LP for b,
        * mix a ← (1‑t) a + t (b / Σb),
        * re‑project onto the simplex.
    The mixing coefficient t decays linearly from `t0`.
    """
    n = len(a)
    for it in range(steps):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        b = solve_lp_max_sum(a, M)
        sb = b.sum()
        if sb <= 0 or not np.isfinite(sb):
            b = np.full(n, 1.0 / n)
            sb = 1.0
        g = b / sb

        # decreasing mixing weight
        t = t0 / (1.0 + 0.1 * it)
        a = (1.0 - t) * a + t * g
        a = project_to_simplex(a)

        if (it + 1) % 5 == 0:
            cur = compute_score(a)
            print(f"    LP step {it + 1:3d}  cur_score={cur:.6f}")

    return a

        print(f"  Official score for n={n}: {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a.tolist()
            print(f"  *** New global best: {best_val:.6f} (n={n}) ***")

    # --------------------------------------------------------------
    # 2.  Guard – guarantee a non‑empty return value
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
