def convolution_fft(a,b):
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]

"""
search_step_function_v2.py

A faster and more exhaustive search for a non‑negative step function that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

(the same quantity returned by ``evaluate_sequence``).

The algorithm consists of three stages:

1. **Broad parametric sweep** – generate many candidate shapes from a few
   low‑dimensional families (arcsine, symmetric Beta, power‑law, mixtures
   with the uniform distribution and random Dirichlet draws).  All
   candidates are normalised to the simplex and a cheap FFT‑based score is
   used to keep the most promising ones.

2. **Mirror‑descent refinement** – each retained candidate is improved with
   an exponentiated‑gradient (mirror‑descent) method on the simplex.
   The sub‑gradient of the ``max`` term is taken by assigning equal weight
   to all maximising convolution entries.  The update uses a log‑space
   formulation (to avoid overflow) and a classic 1/√t step‑size schedule.
   The best candidate of each length receives a longer run.

3. **Final polishing** – the overall best sequence is refined once more
   with a long run of the same sub‑gradient method and a few stochastic
   perturbations are tried to escape possible local minima.

All work is time‑budget aware (default 1000 s) and always returns a feasible
list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    a is assumed to be non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    # reverse w because conv(a, rev(w)) yields the required cross‑correlation
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = len(a)+len(w)-1
    idx = len(a) - 1                                 # position that aligns with a_j
    grads = 2.0 * conv[idx:idx + len(a)]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/sqrt(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Mirror‑descent (sub‑gradient) refinement
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent (exponentiated‑gradient) on the exact objective.
    The iterate stays on the simplex; the update is performed in log‑space
    to keep the numbers stable.  Returns the best vector seen and its score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1
    eps = 1e-30                # for safe log

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # binary weight for the (sub)gradient of the max‑function
        w = np.zeros(conv_len, dtype=float)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)
        # Clip gradients to avoid overflow in the exponent
        grad = np.clip(grad, -1000.0, 1000.0)

        eta = eta0 / np.sqrt(it)          # classic 1/√t schedule
        log_a = np.log(a + eps) - eta * grad
        a = np.exp(log_a)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
