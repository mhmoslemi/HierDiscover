"""
A fast mixed optimiser for the step‑function bound.

The objective
    2·n·max(conv(a,a)) / (sum(a)²)

is invariant under scaling, therefore we work on the probability simplex
(sum = 1, aᵢ ≥ 0).  The optimiser consists of three phases

1.  A smooth surrogate (log‑sum‑exp) minimised by a limited‑memory
    BFGS algorithm on the simplex (soft‑max re‑parameterisation).
2.  A short Adam run on the true max‑convolution (sub‑gradient) – a
    refined version of the original “Stage 2”.
3.  A very cheap stochastic local‑search that repeatedly transfers a
    tiny amount of mass between two random coordinates and accepts the
    move if the exact objective improves (simulated‑annealing style).

All phases respect the 1000 s budget, keep the best sequence seen and
return it as a plain ``list`` of floats.  The routine works for any
length (default ≈ 5 000, the size of the previous record) and does not
use any file or network I/O.
"""

import time
import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

# ----------------------------------------------------------------------
#  Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_conv(a: np.ndarray) -> np.ndarray:
    """Aperiodic self‑convolution of a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())  # next power of two
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact evaluation used by the harness (sum is 1 → formula simplifies)."""
    n = a.shape[0]
    conv = _full_conv(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
#  Phase 1 – smooth (log‑sum‑exp) optimisation with L‑BFGS‑B
# ----------------------------------------------------------------------


def _smooth_lbfgs(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Minimise a log‑sum‑exp surrogate of the max‑convolution."""
    n = a0.shape[0]
    # work in the “logits’’ space – soft‑max guarantees the simplex
    eps = 1e-12
    x0 = np.log(a0 + eps)

    # surrogate:   M_α(a) = (1/α)·log Σ exp(α·conv(a))
    # we will start with a modest α and let the optimiser increase it.
    # The optimiser works on the logits; gradients are obtained by the
    # chain rule (soft‑max → a → conv → surrogate).
    def fun_and_grad(logits: np.ndarray, α: float):
        # probabilities on the simplex
        logits_max = np.max(logits)
        probs = np.exp(logits - logits_max)
        probs /= probs.sum()

        # convolution of the probabilities
        conv = _full_conv(probs)

        # log‑sum‑exp of α·conv
        scaled = α * conv
        m = np.max(scaled)
        ex = np.exp(scaled - m)
        Z = np.sum(ex)
        loss = (m + np.log(Z)) / α

        # gradient w.r.t. a (soft‑max space)
        w = ex / Z                      # soft‑max weights over shifts
        # correlation of the weights with the probabilities (exact gradient)
        # we use the formula from the original code:
        L = 1 << ((2 * n - 2).bit_length())
        fft_w = np.fft.rfft(w, n=L)
        fft_a = np.fft.rfft(probs, n=L)
        corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
        grad_a = 4.0 * n * corr[:n]     # d (2·n·max) / d a   (ignoring α‑dependence)

        # chain rule: d loss / d logits = (∂loss/∂a) * ∂a/∂logits
        # for soft‑max a_i = exp(z_i)/Σexp(z),  ∂a_i/∂z_j = a_i(δ_{ij}‑a_j)
        avg = np.dot(probs, grad_a)
        grad_logits = probs * (grad_a - avg)

        # scale because we are minimising, not maximising
        return loss, grad_logits

    # schedule for the smoothing parameter α
    α = 0.5
    α_mul = 1.2
    α_max = 2000.0

    best_a = a0.copy()
    best_val = _score(best_a)

    while time.time() < deadline:
        # L‑BFGS‑B works on unrestricted variables – we keep the box bounds
        # and rely on the soft‑max to enforce the simplex.
        bounds = Bounds(-np.inf, np.inf)

        # minimise the current surrogate
        res = minimize(
            lambda z: fun_and_grad(z, α)[0],
            x0,
            jac=lambda z: fun_and_grad(z, α)[1],
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 200, "ftol": 1e-12, "gtol": 1e-12},
        )
        x0 = res.x  # warm‑start for the next α

        # evaluate the true objective
        a = np.exp(x0 - np.max(x0))
        a /= a.sum()
        cur_val = _score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"Smooth LBFGS  α={α:7.3f} → new best = {best_val:.6f}")

        # anneal α
        α = min(α * α_mul, α_max)

    return best_a, best_val


# ----------------------------------------------------------------------
#  Phase 2 – Adam on the true max‑convolution (same as the original code)
# ----------------------------------------------------------------------


def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max(conv), sub‑gradient) for aperiodic self‑convolution."""
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


def _adam_optimize(init_a: np.ndarray, deadline: float, restart_id: int,
                   target: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser on the true max‑convolution."""
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    # logits for the soft‑max representation
    eps = 1e-12
    b = np.log(init_a + eps)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    ada_eps = 1e-8

    # initial evaluation
    a = np.exp(b - b.max())
    a /= a.sum()
    max_val, _ = _maxconv_and_subgrad(a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1
        a = np.exp(b - b.max())
        a /= a.sum()

        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_a, stagn = cur_score, a.copy(), 0
            print(
                f"Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}"
            )
        else:
            stagn += 1

        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * np.random.randn(n)
            stagn = 0

        # gradient w.r.t. logits (soft‑max trick)
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + ada_eps)

    return best_a, best_score


# ----------------------------------------------------------------------
#  Phase 3 – cheap stochastic refinement (mass‑transfer hill‑climbing)
# ----------------------------------------------------------------------


def _local_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfers, accepting strict improvements."""
    n = a.shape[0]
    cur_score = _score(a)
    best_a, best_val = a.copy(), cur_score
    rng = np.random.default_rng()
    it = 0
    # Simple linear cooling schedule for the acceptance probability
    start_temp = 5e-3
    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        if a[i] == 0.0:
            continue

        # transfer a small fraction of the mass at i to j
        frac = rng.uniform(5e-4, 2e-3)            # step size adaption
        delta = frac * a[i]                      # never make i negative
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # exact evaluation (FFT based, cheap for our size)
        new_val = _score(a_new)

        # strict improvement only (no simulated‑annealing in this short phase)
        if new_val < best_val - 1e-12:
            a, cur_score = a_new, new_val
            best_a, best_val = a_new.copy(), new_val
            print(f"Refine iter {it:6d}: best = {best_val:.6f}")

        # optional very light “temperature’’ to escape tiny plateaus
        # (not a strict requirement, kept for safety)
        elif rng.random() < np.exp((cur_score - new_val) / max(start_temp * (1 - it / 1e6), 1e-8)):
            a, cur_score = a_new, new_val

    return best_a, best_val
