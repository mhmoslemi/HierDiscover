            best_a = a.copy()
            last_improve = time.time()
            print(f"Iter {iteration:7d} – new best {best_score:.6f}")

        # occasional jitter to escape very flat plateaus
        if iteration % 10000 == 0 and iteration > 0:
            jitter = 0.02 * np.random.rand(n) * a
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # if we have not improved for a while, perform a mild stochastic restart
        if time.time() - last_improve > 120.0 and iteration % 5000 == 0:
            a = 0.7 * best_a + 0.3 * np.random.dirichlet(np.ones(n))
            a /= a.sum()
            last_improve = time.time()
            print(f"Iter {iteration:7d} – stochastic restart")

        if iteration % 20000 == 0 and iteration > 0:
            print(f"Iter {iteration:7d}: current best {best_score:.6f}")

        iteration += 1

    print(f"Finished after {iteration} iterations; best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4 – return the best sequence (already normalised to sum == 1)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
