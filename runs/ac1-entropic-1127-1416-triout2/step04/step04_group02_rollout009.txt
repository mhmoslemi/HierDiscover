def optimize_sequence(a0, deadline, max_iter=5000, lr=0.1):
    a = a0.copy()
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    n = len(a)
    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)  # length 2n-1
        # smooth max weights
        alpha = 5.0 + 0.05*it  # increase gradually
        # compute w: w_k = exp(alpha*(conv_k - max_conv))
        max_val = conv.max()
        # subtract max for numerical stability
        w = np.exp(alpha * (conv - max_val))
        Z = w.sum()
        w /= Z
        # gradient = 2 * convolve(w, a)[0:n]  (taking first n)
        grad = 2.0 * convolution_fft(w, a)[:n]
        # use gradient of max (not of actual objective)
        # update moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
        if it % 500 == 0:
            # compute actual objective
            val = evaluate_sequence(a.tolist())
            print(f"    iter={it} evaluated={val}")
            # update best if needed
    return a

def upsample_to(a, new_n):
    old_n = len(a)
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    # Ensure non-negatives and normalize
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new
```

"""
Hybrid smooth‑max gradient descent for the min‑max self‑convolution problem.

The algorithm proceeds in stages:

1. **Generate diverse seeds** (arcsine, Dirichlet draws, perturbed arcsine) for a
   modest length (n≈512) and run a short Adam‑style optimisation on the
   *smooth* approximation of the max‑convolution.
2. **Upsample** the best seed to larger lengths (1024, 2048, 4096, 8192) – each
   step simply interpolates the current shape and renormalises it.
3. **Refine** the upsampled shape at the new length with a longer Adam run.
   The loss is a soft‑max of the convolution (parameter α grows with the
   iteration), which yields a dense gradient that points away from all large
   convolution entries, not only the single active one.
4. **Final polishing** – one extra optimisation pass on the globally best
   shape with a very small step‑size.

All vectors are kept non‑negative and are symmetrised after each update
(`a ← (a+a[::-1])/2`) because the optimal step function is known to be
symmetric.  After each optimisation window the true objective
`evaluate_sequence` is queried only periodically (every few hundred steps) to
keep the overhead low.

The routine respects the supplied wall‑clock budget (default 1000 s) and
always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.  Returns the full (len(a)+len(b)-1)
    convolution without any circular wrap‑around.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Shape generators (used for the initial seeds)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser to avoid division by zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector and normalise to sum = 1."""
    if concentration <= 0:
        # heavy‑tailed draws – raise exponent to obtain sparse vectors
        raw = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        raw = rng.gamma(concentration, 1.0, size=n)
    raw = np.maximum(raw, 1e-12)
    return raw / raw.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation, clip, and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Helper: upscale a vector while keeping the total mass = 1
# ----------------------------------------------------------------------


def upsample_to(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to length new_n, followed by normalisation."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    # work on a unit interval – this preserves the shape under refinement
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Adam‑style optimiser on the smooth max of the convolution
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, alpha: float) -> Tuple[np.ndarray, float]:
    """
    Compute the gradient of the smooth‑max surrogate
        φ_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    and return the gradient together with the (non‑smooth) max value
    max_k (a∗a)_k (needed for the official objective).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_val = conv.max()
    # softmax weights (stable – subtract max_val)
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient = 2 * (w * a) convolution, first n entries only
    grad_full = 2.0 * convolution_fft(w, a)
    grad = grad_full[:n]                             # shape matches a
    return grad, max_val


def optimise_smooth_max(
    a0: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 20_000,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_every: int = 500,
    rng: np.random.Generator,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smooth‑max of the self‑convolution.
    Returns the best vector encountered (as ndarray) and its *true* score
    from ``evaluate_sequence``.
    """
    a = np.maximum(a0, 1e-30)
    a /= a.sum()                     # work on the probability simplex
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # best solution found so far
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # α grows slowly – this makes the surrogate tighter as optimisation proceeds
        alpha = 5.0 + 0.02 * it

        grad, max_conv = _smooth_gradient(a, alpha)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Gradient descent step (we *minimise* the smooth‑max)
        a = a - step

        # Enforce feasibility: non‑negativity, symmetry, and renormalisation
        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) * 0.5                     # symmetry
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # Occasionally inject a little noise to shake out plateaus
        if it % 2000 == 0:
            a += rng.normal(0.0, 1e-5, size=n)
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) * 0.5
            a /= a.sum()

        # Periodic evaluation with the *true* objective
        if it % eval_every == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"    iter={it:,d}  new best score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine builds several seeds, runs a smooth‑max Adam optimiser,
    progressively upsamples to larger lengths and finally returns the best
    candidate found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin – keep a couple of seconds for the final conversion/return
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback – use a previously discovered sequence if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback sequence score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 0.  Seed generation at a modest length (512)
    # ------------------------------------------------------------------
    init_n = 512
    seeds: List[np.ndarray] = [
        arcsine_initial(init_n),                     # classic arcsine shape
        random_dirichlet(init_n, 0.5, rng),          # sparse Dirichlet
        random_dirichlet(init_n, 1.0, rng),          # standard Dirichlet
        random_dirichlet(init_n, 2.0, rng),          # more uniform
        perturb(arcsine_initial(init_n), rng, scale=0.03),
    ]

    # Run a short optimiser on each seed and keep the best one
    print("\n=== Stage 1: Optimising short seeds (n=512) ===")
    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline - 5.0:
            break
        print(f"  seed #{idx + 1}")
        # Rough iteration budget: a few thousand steps should be enough at this size
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(6_000, max(1_500, remaining * 30)))
        refined, score = optimise_smooth_max(
            a0=seed_vec,
            deadline=deadline,
            max_iter=max_iter,
            lr=0.07,
            eval_every=500,
            rng=rng,
        )
        print(f"    score after short run = {score:.6f}")
        if score < best_val:
            best_val = score
            best_seq = refined.tolist()
            print(f"    *** New global best (short seed) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1.  Progressive up‑sampling and refinement
    # ------------------------------------------------------------------
    # Target lengths – powers of two are cheap for the FFT‑based convolutions
    lengths = [1024, 2048, 4096, 8192]

    # start from the (possibly scaled) best sequence we have so far
    current = np.array(best_seq, dtype=float) if best_seq else arcsine_initial(init_n)

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Stage 2: Upsample & optimise (n={n}) ===")
        # Upsample to the new length, preserving the total mass = 1
        current = upsample_to(current, n)

        # Allocate a dynamic iteration budget depending on the remaining time.
        # Rough rule: ~25 iterations per second for the larger sizes.
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(30_000, max(3_000, remaining * 25)))
        refined, score = optimise_smooth_max(
            a0=current,
            deadline=deadline,
            max_iter=max_iter,
            lr=0.04,
            eval_every=700,
            rng=rng,
        )
        print(f"    evaluated score = {score:.6f}")

        if score < best_val:
            best_val = score
            best_seq = refined.tolist()
            print(f"    *** New global best (n={n}) = {best_val:.6f}")

        # keep refined vector for the next up‑sampling step
        current = refined

    # ------------------------------------------------------------------
    # 2.  Final polishing on the globally best sequence (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Final polishing ===")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(40_000, remaining * 25))
        a_refined, polished_score = optimise_smooth_max(
            a0=a,
            deadline=deadline,
            max_iter=max_iter,
            lr=0.02,
            eval_every=800,
            rng=rng,
        )
        print(f"Polished score = {polished_score:.6f}")
        if polished_score < best_val:
            best_val = polished_score
            best_seq = a_refined.tolist()
            print("*** Updated global best after polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible (non‑empty) sequence
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
