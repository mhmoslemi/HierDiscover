"""Step‑function optimiser.

This module searches for a non‑negative vector `a` (interpreted as a step
function) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

by

1.   generating a diverse set of low‑dimensional candidates,
2.   improving each candidate with a *soft‑max* mirror‑descent on the
     probability simplex (log‑parameterised, temperature‑annealed),
3.   polishing the best result with the original exponentiated‑gradient
     sub‑gradient method.

The optimisation runs inside the given time budget (default 1000 s) and
always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length, zero‑padded)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale invariant).
    The vector is assumed non‑negative; the function works for any shape.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}
    (via one FFT‑based convolution).
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine shape ≈ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex by exponential sampling."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Soft‑max mirror‑descent optimiser
# ----------------------------------------------------------------------


def improve_candidate(
    a0: np.ndarray,
    deadline: float,
    *,
    max_iter: int = None,
    beta_start: float = 0.5,
    beta_max: float = 12.0,
    eta0: float = 0.08,
) -> Tuple[np.ndarray, float]:
    """
    Refine a probability vector using a temperature‑annealed soft‑max
    weight on the convolution entries.  The optimisation works in the
    log‑parameter space (logits) to keep the iterate on the simplex
    without an explicit projection.

    Parameters
    ----------
    a0 : np.ndarray
        Initial non‑negative vector (does not need to sum to 1).
    deadline : float
        Unix timestamp after which the routine must stop.
    max_iter : int | None
        Hard iteration limit (if provided).
    beta_start, beta_max : float
        Temperature schedule for the soft‑max (β multiplies the conv values).
    eta0 : float
        Base learning‑rate for the logits (scaled as η₀/√t).

    Returns
    -------
    best_a : np.ndarray
        The best vector seen (on the simplex).
    best_score : float
        Its fast score (2·n·max(conv)).
    """
    # initialise logits safely (avoid log(0))
    eps = 1e-12
    u = np.log(np.maximum(a0, eps))
    n = len(u)

    best_a = normalize(np.exp(u - np.max(u)))
    best_score = compute_score(best_a)

    it = 0
    beta = beta_start
    while time.time() < deadline:
        it += 1
        if max_iter is not None and it > max_iter:
            break

        # current distribution (soft‑max)
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        a = exp_u / np.sum(exp_u)

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)                      # length = 2n‑1
        max_c = np.max(conv)
        w = np.exp(beta * (conv - max_c))
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = gradient_from_weights(a, w)

        # gradient w.r.t. logits (chain rule for soft‑max)
        avg = np.dot(grad_a, a)
        grad_u = a * (grad_a - avg)

        # learning‑rate schedule ηₜ = η₀ / √t
        eta = eta0 / np.sqrt(it)
        u -= eta * grad_u

        # slowly raise the temperature (more emphasis on the true max)
        if beta < beta_max:
            beta *= 1.001

        # occasional bookkeeping
        if it % 100 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # early exit if we have already beaten the target
                if best_score < 1.503:
                    # keep searching a bit to confirm, but we have a good candidate
                    pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Classic exponentiated‑gradient sub‑gradient (used for final polish)
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) sub‑gradient descent on the
    true non‑smooth objective ``max(conv(a,a))``.  The iterate stays on the
    simplex after every step.  Returns the best vector seen.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        w = np.zeros(conv_len, dtype=float)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc
