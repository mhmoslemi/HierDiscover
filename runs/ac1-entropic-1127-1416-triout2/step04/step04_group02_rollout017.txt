def project_to_simplex(v):
    # ensure sum(v)=1 and v>=0
    # using algorithm from https: ... O(n log n)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Improved search for a low‑max‑convolution step function.

Algorithm
---------
1. **Exploration** – several lengths are tried (4096, 2048, 1024).  For each
   length a modest library of diverse shapes (uniform, arcsine, power‑law,
   beta‑like, Dirichlet, perturbed arcsine and an up‑sample of the current
   best) is generated and ranked by a fast FFT based score.

2. **Smooth exponential‑gradient descent** –  Starting from the few best
   candidates a mirror‑descent on a *smoothed* version of the objective
   ``max_k (a∗a)_k`` is performed.  The smooth max is approximated by a
   soft‑max with an annealed temperature `beta`.  The gradient of the
   smoothed objective is obtained analytically via a single extra FFT
   (cross‑correlation).  An exponentiated update guarantees non‑negativity
   and a simple renormalisation keeps the vector on the probability simplex.

3. **Final polishing** – the globally best sequence is refined for the
   remaining time with a longer run, a higher final temperature and a
   smaller step‑size.

The routine respects the supplied wall‑clock budget (default 1000 s) and
always returns a feasible non‑negative list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale invariant):
        2·n·max(a*a) / (sum(a)²)
    a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (exponential/gamma draws, then normalise)."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny random perturbation and re‑normalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smooth exponential‑gradient optimiser
# ----------------------------------------------------------------------


def optimise_smooth_expgrad(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int,
    *,
    eta0: float = 0.8,
    beta_final: float = 80.0,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *smoothed* max‑convolution.
    Uses a soft‑max with temperature `beta` that is increased linearly
    from 1 to `beta_final`.  The update is multiplicative (exponentiated
    gradient) and the vector is re‑normalised onto the simplex after each step.
    Returns the best vector seen and its (non‑smooth) score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # annealed temperature
        beta = 1.0 + (beta_final - 1.0) * (it / max_iter)

        # convolution a*a
        conv = convolution_fft(a, a)
        max_val = conv.max()

        # soft‑max weights
        w = np.exp(beta * (conv - max_val))
        p = w / w.sum()                     # shape (2n‑1,)

        # gradient = 2 * cross‑correlation(p, a)
        # performed via one extra FFT convolution
        rev = convolution_fft(a, p[::-1])
        grad = 2.0 * rev[L - 1 - np.arange(n)]

        # step size schedule η_t = η0 / sqrt(t)
        eta = eta0 / np.sqrt(it)

        # exponentiated (multiplicative) update
        a = a * np.exp(-eta * grad)

        # optional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # renormalise onto the simplex (sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional reporting and best‑track update
        if it % 2000 == 0:
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
            print(
                f"      smooth iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} beta={beta:.1f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine explores a few lengths, builds a small library of
    parameterised candidates, refines the most promising ones with a smooth
    exponential‑gradient optimiser and finally polishes the overall best
    sequence.  A feasible list of floats is always returned before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin for the very final evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Lengths to explore – start with the largest to give it most time.
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a small but diverse library of start vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # (i) trivial uniform
        candidates.append(np.full(n, 1.0 / n))

        # (ii) classic arcsine
        candidates.append(arcsine_initial(n))

        # (iii) random power‑law shapes
        for _ in range(4):
            p = rng.uniform(0.05, 1.5)
            eps = rng.uniform(0.001, 2.0)
            candidates.append(generate_powerlaw(n, p, eps))

        # (iv) beta‑like shapes
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            candidates.append(generate_beta_like(n, alpha, beta))

        # (v) Dirichlet / exponential draws (dense and sparse)
        for conc in [0.3, 0.8, 1.5]:
            candidates.append(random_dirichlet(n, conc, rng))

        # (vi) small perturbations of arcsine
        for _ in range(2):
            candidates.append(perturb(arcsine_initial(n), rng, scale=0.03))

        # (vii) up‑sample of the current global best (if any and smaller)
        if best_seq and len(best_seq) < n:
            small = np.asarray(best_seq, dtype=float)
            x_small = np.linspace(0.0, 1.0, len(small))
            x_big = np.linspace(0.0, 1.0, n)
            up = np.interp(x_big, x_small, small)
            up = up / up.sum()
            candidates.append(up)

        # --------------------------------------------------------------
        # 1b. Quick sweep with the cheap internal score – keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:4]]   # keep four most promising

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate with smooth exponentiated‑gradient.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # allocate iterations based on remaining time
            remaining = max(0.0, deadline - time.time())
            # we expect ~300 iterations per second for n≈4000
            max_iter = int(min(250_000, max(5_000, remaining * 300)))

            refined, _ = optimise_smooth_expgrad(
                initial_a=cand,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.8,
                beta_final=80.0,
                noise_every=500,
                noise_scale=5e-6,
            )

            # final evaluation with the official evaluator
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2. Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(400_000, remaining * 300))

        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_smooth_expgrad(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
            beta_final=120.0,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
