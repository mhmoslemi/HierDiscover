```
def solve_lp_max_sum_g(f, M):
    n = len(f)
    L = 2*n - 1
    # Build A
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    # Objective is maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(L, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return uniform vector
        return np.full(n, 1.0/n)
    return res.x
```

"""
Hybrid optimizer for the step‑function challenge.

Approach
--------
1️⃣  Initialise a non‑negative vector on the probability simplex.
    Several starters are tried (arcsine, uniform, Beta‑family, random).

2️⃣  **Linear‑programming refinement** – for the current vector `f`
    compute `M = max(conv(f,f))`.  Solve the LP

        maximise    Σ g_j
        subject to  f * g ≤ M      (element‑wise)
                    g_j ≥ 0

    The solution `g₀` is the most massive vector that does not increase the
    worst autocorrelation.  It is normalised to sum = 1 and the current
    iterate is moved towards it by a diminishing step size
    `τ = τ₀ / sqrt(iter)` (convex combination, projection onto the simplex).

    This simple “LP‑push” iteration rapidly drives down the bound.

3️⃣  After a few LP steps the vector is polished with the already‑proven
    sub‑gradient descent, Adam on a soft‑max surrogate and a short simulated
    annealing run (the same utilities as in the earlier baseline).

4️⃣  The best vector found over several lengths (256, 512, 1024, 2048) is
    returned.  All time‑budget checks respect the 1000 s limit, and a
    guaranteed fallback is provided.

The implementation uses only NumPy and SciPy (if available).  It is fully
self‑contained and respects the required function signature `propose_candidate`.
"""

import time

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    After normalising Σa = 1 the denominator disappears, therefore the
    score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
    (Duchi et al., 2008).
    """
    if np.all(v <= 0):
        # fallback – should never happen in practice
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1)
    right = np.arange(n - peak - 1)[::-1]
    seq = np.concatenate([left, right]).astype(float)
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    The sum‑constraint is handled later by projection onto the simplex.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the smooth surrogate."""
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Exact sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Projection onto the simplex keeps Σa = 1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """Tiny Metropolis‑style local search using the fast surrogate."""
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta          # sum preserved, non‑negativity kept

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def solve_lp_max_sum_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   s.t.   conv(f,g) ≤ M , g ≥ 0 .
    Returns the optimal g (may be dense).  If the LP fails, a copy of f is
    returned as a safe fallback.
    """
    n = f.shape[0]
    L = 2 * n - 1

    # Build the constraint matrix A where A @ g = conv(f,g)
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n)                     # maximise Σ g  → minimise -Σ g
    bounds = [(0.0, None)] * n
    b_ub = np.full(L, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})

    if res.success and res.x is not None:
        return res.x
    # fallback – return the current iterate unchanged
    return f.copy()


def lp_push_iteration(f: np.ndarray,
                      deadline: float,
                      max_iter: int = 80,
                      tau0: float = 0.3) -> Tuple[np.ndarray, float]:
    """
    Perform a few LP‑push steps starting from ``f``.
    Returns the best vector seen (still on the simplex) together with its fast
    surrogate score.
    """
    best_f = f.copy()
    best_sc = compute_score(f)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current fast score (used for early logging)
        cur_sc = compute_score(f)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_f = f.copy()
            print(f"      LP‑push new best fast score = {best_sc:.6f} (it={it})")

        # bound M from the current vector
        M = np.max(convolution_fft(f, f))

        # solve LP (if SciPy is present)
        if _HAVE_SCIPY:
            g0 = solve_lp_max_sum_g(f, M)
        else:
            g0 = f.copy()      # degenerate fallback

        sum_g0 = g0.sum()
        if sum_g0 > 1e-12:
            g = g0 / sum_g0
        else:
            g = f.copy()

        # diminishing convex step size
        tau = tau0 / np.sqrt(it)

        # try a small set of candidate step sizes and keep the best
        best_local = f
        best_local_sc = cur_sc
        for factor in (tau, tau * 0.5, tau * 2.0):
            cand = (1.0 - factor) * f + factor * g
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cand_sc = compute_score(cand)
            if cand_sc < best_local_sc:
                best_local_sc = cand_sc
                best_local = cand

        f = best_local

    return best_f, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluate_sequence value.
    The routine mixes random exploration, a dedicated LP‑push phase and the
    gradient‑based polishing pipeline from the baseline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Working on length n = {n} ===")

        # ----------- 1a. generate a modest pool of diverse starters ----------
        pool: List[np.ndarray] = []

        # deterministic baselines
        pool.append(arcsine_initial(n))
        pool.append(np.full(n, 1.0 / n))

        # beta‑family random draws
        for _ in range(3):
            a = rng.uniform(0.2, 3.0)
            b = rng.uniform(0.2, 3.0)
            eps = rng.uniform(0.001, 0.5)
            pool.append(generate_beta_shape(n, a, b, eps))

        # inverse‑polynomial arcsine‑like
        for _ in range(2):
            p = rng.uniform(0.2, 2.5)
            eps = rng.uniform(0.0, 1.0)
            pool.append(generate_power_shape(n, p, eps))

        # random simplex points
        for _ in range(2):
            v = rng.random(n)
            pool.append(project_to_simplex(v))

        # ------------------------------------------------------------------
        # 2.  LP‑push refinement for each seed
        # ------------------------------------------------------------------
        for seed_vec in pool:
            if time.time() >= deadline - 5.0:
                break

            f = project_to_simplex(seed_vec.astype(float).copy())

            # A few LP iterations (fast) – time‑adaptive budget
            remaining = deadline - time.time()
            max_lp_iters = max(10, int(min(80, remaining * 30)))
            f, _ = lp_push_iteration(f, deadline, max_iter=max_lp_iters, tau0=0.3)

            # ------------------------------------------------------------------
            # 3.  Fine‑grained polishing (sub‑gradient → Adam → SA)
            # ------------------------------------------------------------------
            if time.time() < deadline - 5.0:
                # sub‑gradient descent
                f = subgradient_descent_exact(f, deadline, max_iter=30000, step0=0.04)

            if time.time() < deadline - 5.0:
                # Adam on the smooth surrogate
                f, _ = adam_optimize_smooth(
                    initial_a=f,
                    deadline=deadline,
                    max_iter=8000,
                    lr0=0.12,
                    lam_start=2.0,
                    lam_end=80.0,
                    noise_every=2500,
                    noise_scale=3e-5,
                )

            if time.time() < deadline - 5.0:
                # short simulated annealing polishing
                f = simulated_annealing(f, rng, deadline, max_iters=12000,
                                        init_temp=0.025, decay=0.9998)

            # ------------------------------------------------------------------
            # 4.  Official evaluation
            # ------------------------------------------------------------------
            try:
                val = evaluate_sequence(f.tolist())
                print(f"    Official score = {val:.6f} (n={n})")
                if val < best_val:
                    best_val = val
                    best_seq = f.tolist()
                    print(f"    *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception as e:
                print("    Evaluation error:", e)

    # ------------------------------------------------------------------
    # 5.  Global polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        # allocate the rest of the budget roughly equally
        sub_iters = int(min(200_000, remaining * 30 // 2))
        adam_iters = int(min(200_000, remaining * 30 // 2))

        print("\nFinal global polishing (sub‑gradient + Adam)…")
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.02)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=adam_iters,
            lr0=0.1,
            lam_start=2.0,
            lam_end=80.0,
            noise_every=2000,
            noise_scale=2e-5,
        )
        try:
            final_val = evaluate_sequence(a.tolist())
            print(f"Final polished official score = {final_val:.6f}")
            if final_val < best_val:
                best_val = final_val
                best_seq = a.tolist()
                print("*** Updated global best after final polish ***")
        except Exception as e:
            print("Final polishing evaluation error:", e)

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = np.full(256, 1.0 / 256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
