"""
Propose a low‑score step function.

The algorithm consists of three quick phases that comfortably fit into the
1000 s budget.

1. **Random dense‑subset search** – for a few large lengths (4096–16384) we
   draw many random subsets of indices of size 0.65 · n … 0.90 · n.  The
   vector has uniform mass 1/k on the chosen positions, so the sum of the
   coefficients is exactly 1.  The objective reduces to
       2·n·max(counts)/k²,
   where `counts` are the ordered‑pair sums of the index set.  For dense,
   random sets the maximum count is very close to its average, yielding a
   score only slightly above the theoretical lower bound ≈ 1.

2. **Adam‑style projected sub‑gradient optimisation** – starting from the
   best random candidate we minimise the maximum convolution value directly.
   The sub‑gradient of the max‑convolution at the current offending shift
   `s` is `g[i] = 2·a[s−i]` (zero where the shift is out of range).  An
   Adam update is applied and the iterate is projected back onto the
   probability simplex (non‑negative entries that sum to 1).

3. **Simple swap hill‑climb** – a final polish that repeatedly swaps a
   non‑zero index with a zero index (keeping the number of non‑zero entries
   constant).  Each swap is accepted only if it improves the objective.

All phases are stopped well before the deadline; the best sequence found
is returned as a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT (fast).
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when the sum of a is 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def compute_score_from_indices(n: int, idx: np.ndarray) -> float:
    """
    Evaluate the objective for a binary indicator vector of length ``n``
    having ones at positions ``idx`` and uniform weight 1/k (k = len(idx)).
    """
    k = idx.size
    if k == 0:
        return float("inf")
    # binary indicator
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    conv = convolution_fft(b, b)
    max_cnt = float(np.max(conv))
    # objective = 2·n·max_cnt / k²  (sum of coefficients = 1)
    return 2.0 * n * max_cnt / (k * k)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).  Implements the algorithm of
    Wang & Carreira‑Perpinán (2013).
    """
    if np.all(v <= 0):
        # Uniform distribution as a fallback
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient optimisation
# ----------------------------------------------------------------------


def adam_subgrad_optimize(init_a: np.ndarray,
                          time_deadline: float,
                          max_iter: int = 150_000,
                          lr: float = 0.07,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise ``max(conv)`` using an Adam update on the sub‑gradient.
    The iterate is always projected onto the probability simplex,
    guaranteeing non‑negativity and unit sum.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        s = int(np.argmax(conv))                # shift giving the max

        # sub‑gradient of max‑convolution (ordered pairs)
        idx = np.arange(n)
        j = s - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adam step with projection
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (lr * 0.005 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple swap‑based hill‑climb
# ----------------------------------------------------------------------


def hill_climb_swaps(seq: list[float],
                     time_deadline: float,
                     max_steps: int = 250_000) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that swaps a non‑zero entry with a zero entry.
    The number of non‑zero positions stays constant (k), the weight on each
    non‑zero entry stays 1/k, hence the sum is always 1.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    k = int(round(1.0 / np.max(a))) if np.max(a) > 0 else int(np.count_nonzero(a))
    best_a = a.copy()
    best_score = compute_score(best_a)

    nonzero = np.flatnonzero(best_a)
    zero = np.where(best_a == 0)[0]

    step = 0
    while step < max_steps and time.time() < time_deadline:
        # pick a random non‑zero and a random zero position
        i = np.random.choice(nonzero)
        j = np.random.choice(zero)

        cand = best_a.copy()
        cand[i] = 0.0
        cand[j] = 1.0 / k

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # update index sets
            nonzero = np.flatnonzero(best_a)
            zero = np.where(best_a == 0)[0]

        step += 1

    return best_a.tolist(), best_score
