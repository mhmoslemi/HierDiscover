def lp_improve(a, delta=0.2, deadline=None):
    n = len(a)
    a_np = np.array(a, dtype=float)
    L = 2*n - 1
    conv0 = convolution_fft(a_np, a_np)  # length L

    # Build matrix A
    A = np.zeros((L, n), dtype=float)
    for i in range(n):
        A[i:i+n, i] = a_np

    d = cp.Variable(n)
    t = cp.Variable()

    constraints = [
        conv0 + 2 * (A @ d) <= t,
        a_np + d >= 0,
        cp.sum(d) == 0,
        cp.norm(d, 1) <= delta  # optional
    ]
    # Additional constraint: t <= current_max (maybe?) we want reduction; but we just minimize.

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except cp.error.SolverError:
        return a

    if d.value is None:
        return a
    direction = d.value
    # choose step factor alpha = 1 (max) or do backtracking
    new_a = a_np + direction
    # ensure nonneg:
    new_a = np.maximum(new_a,0)
    # ensure sum=1 (project onto simplex):
    new_a = project_to_simplex(new_a)
    return new_a

def line_search_update(a, d, max_iter=10):
    alpha = 1.0
    cur_sc = compute_score(a)
    for _ in range(max_iter):
        a_new = a + alpha * d
        if np.any(a_new < -1e-12):
            alpha *= 0.5
            continue
        a_new = np.maximum(a_new, 0)
        a_new = project_to_simplex(a_new)
        new_sc = compute_score(a_new)
        if new_sc < cur_sc:
            return a_new
        else:
            alpha *= 0.5
    return a

def lp_improve(a, deadline=None, delta=None):
    """
    Perform a single linear‑programming improvement step.
    Returns a new vector if an improvement is found, otherwise the original.
    """
    import cvxpy as cp
    n = len(a)
    a_np = np.array(a, dtype=float)
    L = 2 * n - 1
    conv0 = convolution_fft(a_np, a_np)

    # construct matrix A such that (A @ d)[k] = sum_i a[i] * d[k-i]
    # Using broadcasting: each column i adds a shifted copy of a
    A = np.zeros((L, n), dtype=float)
    for i in range(n):
        A[i:i+n, i] = a_np

    d = cp.Variable(n)
    t = cp.Variable()
    # constraints: conv0 + 2*A*d <= t
    cons = [conv0 + 2 * (A @ d) <= t,
            a_np + d >= 0,
            cp.sum(d) == 0]
    if delta is not None:
        cons.append(cp.norm(d, 1) <= delta)

    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    except Exception as e:
        # solver failed
        return a

    if d.value is None:
        return a

    direction = d.value
    # line search for improvement
    cur_sc = compute_score(a_np)
    # try full step
    cand = a_np + direction
    # clip and project for safety
    cand = np.maximum(cand, 0.0)
    cand = project_to_simplex(cand)

        cand_sc = compute_score(cand)
        if cand_sc + 1e-9 < cur_sc:
            return cand
        alpha *= 0.5
    # no improvement
    return a

"""
Propose a low‑discrepancy step function.

The search proceeds in three layers:

1. **Exploration** – generate a diverse pool of candidates of several
   lengths (256…2048) using arcsine, beta‑family, power‑law,
   mixture and triangular shapes.  Keep the best few seeds.

2. **Refinement** – for each seed run a short Adam optimisation on a
   temperature‑controlled soft‑max surrogate of the true objective,
   followed by an exact sub‑gradient descent, a linear‑programming
   improvement step (LP) that directly attacks the max‑convolution,
   and a cheap greedy hill‑climb.  The LP solves  

        minimise   t
        subject to  conv(a,a) + 2·A·d ≤ t
                    a + d ≥ 0
                    Σ d = 0

   where ``A`` is the convolution matrix of ``a``; the direction
   ``d`` is line‑searched back to the original objective.

3. **Global polishing** – after the best vector across all lengths is
   found, a final round of sub‑gradient descent and Adam smoothening is
   applied, then the official ``evaluate_sequence`` is called.

All operations respect the total time budget (default 1000 s) and
produce a non‑negative sequence that minimises the official evaluation
function.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score when Σ a = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for the initial pool)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """(i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1)
    right = np.arange(n - peak - 1)[::-1]
    seq = np.concatenate([left, right]).astype(float)
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

# Optimisers

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max); the simplex constraint is handled
    later by a Euclidean projection.
    """
    n = len(a)
    conv = convolution_fft(a, a)                # length L = 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)             # sum w = 1

    # weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length n + L - 1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 20_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 150.0,
                         noise_every: int = 2500,
                         noise_scale: float = 2e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for the temperature (λ)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # occasional jitter to escape plateaus
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 50_000,
                              step0: float = 0.03) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    Uses a diminishing step η_t = step0 / sqrt(t) and projects onto the simplex.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # indices attaining the maximum (within a tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·∑_{k∈K} shift(a,k)
        conv_rev = convolution_fft(a, mask[::-1])
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def lp_improve(a: np.ndarray,
               deadline: float,
               delta: float = None) -> np.ndarray:
    """
    Perform a single linear‑programming improvement step.
    The LP tries to minimise the maximal entry of the linearised
    convolution ``conv(a,a) + 2·conv(a,d)`` while keeping the sum fixed.
    Returns a new vector if an improvement is found, otherwise the input.
    """
    try:
        import cvxpy as cp
    except Exception:
        # cvxpy not available – fall back to the original vector
        return a

    n = len(a)
    a_np = np.array(a, dtype=float)
    L = 2 * n - 1
    conv0 = convolution_fft(a_np, a_np)

    # Build the convolution matrix A such that (A @ d)[k] = Σ_i a_i d_{k-i}
    A = np.zeros((L, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a_np

    d = cp.Variable(n)
    t = cp.Variable()
    cons = [
        cp.Constant(conv0) + 2 * (A @ d) <= t,
        a_np + d >= 0,
        cp.sum(d) == 0
    ]
    if delta is not None:
        cons.append(cp.norm(d, 1) <= delta)

    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000, verbose=False)
    except Exception:
        return a

    if d.value is None:
        return a

    direction = np.asarray(d.value).reshape(-1)

    # line‑search on the true surrogate (fast) to guarantee improvement
    cur_sc = compute_score(a_np)
    cand = a_np + direction
    cand = np.maximum(cand, 0.0)
    cand = project_to_simplex(cand)
    cand_sc = compute_score(cand)

    if cand_sc + 1e-9 < cur_sc:
        return cand

    # try scaled steps if the full step is too aggressive
    alpha = 0.5
    while alpha > 1e-4:
        cand = a_np + alpha * direction
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)
        if cand_sc + 1e-9 < cur_sc:
            return cand
        alpha *= 0.5

    return a  # no improvement found

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    Combines random shape sampling, Adam on a smooth surrogate,
    exact sub‑gradient descent, a linear‑programming refinement,
    and a final greedy hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 1536, 2048]

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ---------- 1a. generate a pool of candidates ----------
        pool: List[Tuple[float, np.ndarray]] = []

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for st in starters:
            pool.append((compute_score(st), st.copy()))

        # beta‑family
        for _ in range(150):
            alpha = rng.uniform(0.1, 2.5)
            beta = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.001, max(0.5, 0.02 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)
            pool.append((compute_score(cand), cand))

        # power‑law (generalised arcsine)
        for _ in range(150):
            p = rng.uniform(0.2, 2.5)
            eps = rng.uniform(0.0, 1.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # mixtures of arcsine & uniform
        for _ in range(100):
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # simple triangular bumps
        for _ in range(100):
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, int(peak))
            pool.append((compute_score(cand), cand))

        # keep the best few seeds for intensive refinement
        pool.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in pool[:5]]

        # ---------- 2.  Refinement of each seed ----------
        for seed_idx, seed_vec in enumerate(top_seeds, 1):
            if time.time() >= deadline - 10.0:
                break
            print(f"\n--- Seed {seed_idx}/{len(top_seeds)} (n={n}) ---")

            # Phase A – smooth Adam optimisation (short run)
            remaining = deadline - time.time()
            max_iter_adam = int(min(10_000, remaining * 5))
            refined, _ = adam_optimize_smooth(
                initial_a=seed_vec,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=2.0,
                lam_end=150.0,
                noise_every=2500,
                noise_scale=2e-5,
            )

            # Phase B – exact sub‑gradient descent
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_sub = int(min(40_000, remaining * 8))
                refined = subgradient_descent_exact(
                    refined,
                    deadline,
                    max_iter=max_iter_sub,
                    step0=0.03,
                )

            # Phase C – linear‑programming refinement (up to 4 attempts)
            if time.time() < deadline - 5.0:
                for lp_round in range(4):
                    if time.time() >= deadline - 2.0:
                        break
                    refined_lp = lp_improve(refined, deadline, delta=None)
                    if compute_score(refined_lp) + 1e-9 < compute_score(refined):
                        refined = refined_lp
                        print(f"    LP improvement {lp_round + 1} succeeded "
                              f"(fast score = {compute_score(refined):.6f})")
                    else:
                        break

            # Phase D – greedy hill‑climbing fine‑tuning
            if time.time() < deadline - 5.0:
                cur_sc = compute_score(refined)
                n_vars = len(refined)
                for it in range(20_000):
                    if time.time() >= deadline - 1.0:
                        break
                    i, j = rng.integers(0, n_vars, size=2)
                    if i == j:
                        continue
                    delta = refined[i] * 0.1 * rng.random()
                    if delta <= 1e-15:
                        continue
                    cand = refined.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand_sc = compute_score(cand)
                    if cand_sc < cur_sc:
                        refined = cand
                        cur_sc = cand_sc

                # optional short Adam polish after hill‑climb
                if time.time() < deadline - 3.0:
                    refined, _ = adam_optimize_smooth(
                        initial_a=refined,
                        deadline=deadline,
                        max_iter=5_000,
                        lr0=0.08,
                        lam_start=80.0,
                        lam_end=200.0,
                        noise_every=1000,
                        noise_scale=1e-5,
                    )

            # Final official evaluation of the candidate
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"    Official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    print(f"    *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        it_sub = int(min(150_000, remaining * 12))
        it_adam = int(min(150_000, remaining * 12))

        print("\nGlobal polishing (sub‑gradient + Adam) …")
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.02)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=it_adam,
            lr0=0.08,
            lam_start=80.0,
            lam_end=200.0,
            noise_every=2000,
            noise_scale=1e-5,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
