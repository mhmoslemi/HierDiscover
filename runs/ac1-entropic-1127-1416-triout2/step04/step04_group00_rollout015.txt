"""
Propose a low‑score step function.

Algorithm
---------
1. **Broad random search** – generate many candidates of several lengths.
   * Uniform weight on a random subset of positions (density p ∈ [0.2,0.8]).
   * Low‑discrepancy (Van‑der‑Corput) subset.
   * Quadratic‑residue subset (mod‑n squares).
   The fast FFT‑based convolution is used to evaluate 2·n·max(a*a) which,
   for a vector with sum = 1, equals the score returned by the official
   `evaluate_sequence`.

2. **Projected Adam sub‑gradient descent** – starting from the best
   candidate, minimise the non‑smooth objective 2·n·max(conv) by the
   sub‑gradient at the currently maximal shift.  Adam‑style moment
   adaptation gives faster convergence than a plain diminishing step.

3. **Linear‑program refinement** – for the current vector f (∑f=1) compute
   M = max(f * f).  Solve the LP

        maximise   ∑ b_j
        subject to  (f * b)_k ≤ M   for all k
                    b_j ≥ 0

   (`scipy.optimize.linprog` with the HiGHS solver).  The optimal b has
   a larger total mass while never increasing the cross‑convolution.
   Normalise `g = b / sum(b)` and either replace f by g (if its own
   self‑convolution improves) or mix f and g (convex combination) to
   obtain a strictly smaller maximal convolution.  A few LP iterations
   are performed while time permits.

4. **Final hill‑climber** – stochastic swaps and mass‑transfer moves
   (projected onto the simplex) accept any improvement of the true
   score.  This polishing stage catches remaining local refinements.

All work respects the 1000 s budget; intermediate progress is printed.
The function returns a plain list of non‑negative floats ready for the
evaluation harness.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple probability‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def random_sparse_seq(n: int, p: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform weights on a random subset of density p."""
    k = max(2, int(round(p * n)))
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def van_der_corput_seq(n: int, k: int) -> np.ndarray:
    """
    Low‑discrepancy set using the radical inverse in base 2.
    Returns a vector of length n with weight 1/k on the k generated indices.
    """
    def radical_inverse(i: int) -> float:
        rev = 0
        base = 1
        while i:
            rev = rev * 2 + (i & 1)
            i >>= 1
            base <<= 1
        return rev / base

    positions = set()
    i = 0
    while len(positions) < k:
        pos = int(radical_inverse(i) * n)
        if pos < n:
            positions.add(pos)
        i += 1
    a = np.zeros(n, dtype=float)
    a[list(positions)] = 1.0 / k
    return a


def quadratic_residue_seq(n: int, k: int) -> np.ndarray:
    """
    Use residues i² mod n (n need not be prime).  Keep the first k distinct
    residues and assign equal weight.
    """
    residues = set()
    i = 0
    while len(residues) < k:
        residues.add((i * i) % n)
        i += 1
    a = np.zeros(n, dtype=float)
    a[list(residues)] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Projected Adam sub‑gradient optimisation
# ----------------------------------------------------------------------


def adam_subgrad_optimize(init_a: np.ndarray,
                          time_deadline: float,
                          max_iter: int = 250_000,
                          alpha0: float = 0.08) -> tuple[np.ndarray, float]:
    """Adam‑style sub‑gradient descent on 2·n·max(conv)."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha0 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (alpha0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program refinement (LP on cross‑convolution)
# ----------------------------------------------------------------------


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  maximise ∑ b_j   subject to   (f * b)_k ≤ M   and   b ≥ 0.
    Returns the optimal b or None on failure.
    """
    n = len(f)
    rows = 2 * n - 1
    # build dense constraint matrix A where A[k, j] = f[k-j] for valid indices
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]          # contribution of f[i] to shifts i..i+n-1

    b_ub = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)    # maximise sum(b)  <=> minimise -sum(b)
    bounds = [(0, None)] * n

    # HiGHS (default) is fast for dense problems of this size.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=bounds, method='highs',
                                options={"presolve": True})
    if res.success:
        return res.x
    return None


def lp_refine(f: np.ndarray,
              time_deadline: float,
              max_outer: int = 4,
              mix_weight: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinements.
    Returns the improved vector (still on the simplex) and its score.
    """
    best_a = f.copy()
    best_score = compute_score(best_a)

    for _ in range(max_outer):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(best_a, best_a)
        M = float(np.max(conv))

        b = solve_lp_for_g(best_a, M)
        if b is None:
            break

        sum_b = float(np.sum(b))
        if sum_b <= 1e-12:
            break

        g = b / sum_b                      # normalise to the simplex
        # Try a direct replacement
        new_score = compute_score(g)
        if new_score < best_score * 0.999:   # strict improvement
            best_a = g
            best_score = new_score
            continue

        # Otherwise mix current and g
        mixed = (1.0 - mix_weight) * best_a + mix_weight * g
        mixed = project_to_simplex(mixed)
        new_score = compute_score(mixed)
        if new_score < best_score:
            best_a = mixed
            best_score = new_score
        else:
            # No progress – stop LP refinements.
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climber (swap / transfer) – uses the *true* objective
# ----------------------------------------------------------------------


def hill_climb(seq: list[float],
               time_deadline: float,
               max_steps: int = 300_000) -> tuple[list[float], float]:
    """Stochastic hill‑climber that uses the official `evaluate_sequence`."""
    a = np.array(seq, dtype=float)
    a = a / np.sum(a)                     # keep on simplex
    n = a.shape[0]

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    step_scale = 0.12  # fraction of mass moved during a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
                    # re‑project onto simplex (only i may become negative)
                    cand = project_to_simplex(cand)

        cur_score = evaluate_sequence(cand.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.08, 0.5)   # successful move → larger step
        else:
            step_scale *= 0.96                         # failure → shrink step

        max_steps -= 1

    return best_a.tolist(), best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine respects the 1000 s time budget, prints progress, and finally
    returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Keep a global fallback if the notebook provides one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – massive random / low‑discrepancy search
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192, 16384]
    while time.time() < deadline - 5.0:
        n = int(rng.choice(lengths))
        # pick a density in a wide interval – sometimes a bit dense, sometimes sparse
        p = float(rng.uniform(0.15, 0.85))
        k = max(2, int(round(p * n)))

        # Choose a generation mode
        mode = rng.integers(0, 3)
        if mode == 0:
            a = random_sparse_seq(n, p, rng)
        elif mode == 1:
            a = van_der_corput_seq(n, k)
        else:
            a = quadratic_residue_seq(n, k)

        cur_score = compute_score(a)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a.tolist()
            print(f"New random best: n={n}, p={p:.3f}, score={best_val:.6f}")

        # Keep the loop running until we get close to the deadline
        if time.time() > deadline - 200.0:
            break

    # ------------------------------------------------------------------
    # Phase 2 – Adam sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print("\nStarting Adam sub‑gradient refinement")
        a_init = np.array(best_seq, dtype=float)
        refined_a, refined_score = adam_subgrad_optimize(
            a_init,
            time_deadline=deadline - 15.0,
            max_iter=300_000,
            alpha0=0.10,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Adam refinement improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP‑based refinement (few iterations, high impact)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 25.0:
        print("\nStarting LP‑based refinement")
        a_init = np.array(best_seq, dtype=float)
        refined_a, refined_score = lp_refine(
            a_init,
            time_deadline=deadline - 10.0,
            max_outer=5,
            mix_weight=0.35,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"LP refinement improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final hill‑climbing polish
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nFinal hill‑climb polishing")
        polished_seq, polished_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=400_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain Python list – the harness will call `evaluate_sequence` itself.
    return best_seq
