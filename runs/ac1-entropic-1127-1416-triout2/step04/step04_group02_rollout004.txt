def simplex_projection(v):

   # v can be any real numbers, returns vector w >=0 sum=1

Pseudo:

   u = np.sort(v)[::-1]

   cssv = np.cumsum(u) - 1

   rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]

   theta = cssv[rho] / (rho+1)

   w = np.maximum(v - theta, 0)

Return w.

def smooth_max_conv(a, alpha):

    conv = conv_full_fft(a, a)  # length 2n-1

    max_c = conv.max()

    # compute stable softmax

    exp_terms = np.exp(alpha * (conv - max_c))

    sum_exp = exp_terms.sum()

    sm = max_c + np.log(sum_exp) / alpha

    w = exp_terms / sum_exp

    return sm, w

def convolution_fft_full(x, y):

    m = len(x) + len(y) - 1

    n_fft = next_pow2(m)

    X = np.fft.rfft(x, n=n_fft)

    Y = np.fft.rfft(y, n=n_fft)

    Z = X * Y

    conv = np.fft.irfft(Z, n=n_fft)[:m]

    return conv

"""
Improved search for a low‑max‑convolution step function.

The core idea
---------------
* Work on the probability simplex (∑a = 1).  The official objective is then
  2·n·max(a∗a) – scale‑invariant, so fixing the sum is safe.
* Approximate the non‑smooth max by a smooth log‑sum‑exp:
        fα(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
  and minimise fα with a first‑order method.
* The gradient of fα is available in closed form and can be computed in
  O(n log n) using FFTs:
        ∇fα(a) = 2·conv_rev[w] – w_{2j}·a_j,
  where w_k = exp(α·(a∗a)_k) / Σ exp(α·(a∗a)) and conv_rev denotes the
  (full) convolution of the reversed a with w.
* Optimise with Adam (exponential‑moving‑average) while projecting after
  every step onto the simplex (ensuring non‑negativity and unit sum).
* Use an increasing schedule for α (5 → 10 → 20 → 40 → 80) – early
  iterations see a smooth landscape, later ones focus on the true max.
* Explore a few sequence lengths (2048, 4096) and a handful of diverse
  initial shapes (uniform, arcsine, Dirichlet draws, and the fallback
  construction if available).  The best candidate is finally polished
  with a longer Adam run at a larger α.

The whole routine respects the supplied time budget and always returns a
feasible non‑negative list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive → return the zero vector
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective / gradient (smooth max‑convolution)
# ----------------------------------------------------------------------


def smooth_max_and_weights(conv: np.ndarray, alpha: float) -> Tuple[float, np.ndarray]:
    """
    Return a smooth approximation of max(conv) and the associated
    softmax weights w_k = exp(alpha·conv_k) / Σ exp(alpha·conv).
    The approximation is: fα = max(conv) + log Σ exp(alpha·(conv-max))/alpha.
    """
    max_c = conv.max()
    # stability: shift by max_c before exponentiation
    shifted = conv - max_c
    exp_terms = np.exp(alpha * shifted)
    sum_exp = exp_terms.sum()
    # smoothed value (not used directly in the optimiser, but handy for logging)
    smooth_val = max_c + math.log(sum_exp) / alpha
    w = exp_terms / sum_exp
    return smooth_val, w


def gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the smooth max (the one that corresponds to the
    softmax weights w).  Formula:
        g = 2·(rev(a) ∗ w)[n‑1:n‑1+n]  –  w_{2j}·a_j
    """
    n = len(a)
    a_rev = a[::-1]
    conv_rev = conv_full_fft(a_rev, w)               # length = n + (2n‑1) – 1 = 3n‑2
    start = n - 1
    g = 2.0 * conv_rev[start:start + n]

    idx = np.arange(n)
    mask = (2 * idx) < len(w)
    if mask.any():
        g[mask] -= w[2 * idx[mask]] * a[mask]
    return g


def raw_score(a: np.ndarray) -> float:
    """
    Exact objective for vectors that already satisfy Σa = 1:
        2·n·max(a∗a)
    """
    n = len(a)
    conv = conv_full_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def uniform_initial(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from a Dirichlet‑like distribution (Gamma draws)."""
    if concentration <= 0:
        # heavy‑tailed “sparse” draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()


def resize_sequence(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length, followed by renormalisation."""
    old_n = len(seq)
    if old_n == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return new_seq / total

# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------


def optimise_adam(
    init_a: np.ndarray,
    deadline: float,
    alpha_schedule: List[float],
    max_iters: int,
    learning_rate: float = 0.02,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max‑convolution.
    Returns the best vector seen (with respect to the *exact* score)
    and that exact score.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()               # stay on the simplex

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_sc = raw_score(a)

    it = 0
    alpha_idx = 0
    alpha = alpha_schedule[alpha_idx]

    while it < max_iters and time.time() < deadline:
        # switch to the next α when the allocated budget per α is exhausted
        # we simply change α after a fixed number of iterations:
        if it % (max_iters // len(alpha_schedule)) == 0 and it > 0:
            alpha_idx = min(alpha_idx + 1, len(alpha_schedule) - 1)
            alpha = alpha_schedule[alpha_idx]

        # --- forward pass -------------------------------------------------
        conv = conv_full_fft(a, a)                 # full convolution
        _, w = smooth_max_and_weights(conv, alpha)  # softmax weights

        # --- gradient ------------------------------------------------------
        grad = gradient_smooth(a, w)

        # --- Adam update ---------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = learning_rate / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # --- projection onto the simplex ------------------------------------
        a = simplex_projection(a)

        # --- keep the best exact score ---------------------------------------
        cur_sc = raw_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional logging (lightweight)
        if it % 5000 == 0:
            print(f"    Adam iter={it:,d} α={alpha:.1f} cur_score={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        # give the polishing phase ~30 % of the remaining budget
        max_iter = int(min(250_000, remaining * 300))

        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_adam(
            init_a=a,
            deadline=deadline,
            alpha_schedule=[80.0, 120.0, 200.0],
            max_iters=max_iter,
            learning_rate=0.008,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # a safe fallback (uniform 256‑vector)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
