"""
Improved search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three stages:

1.  **Initial candidates** –  a few simple families (arcsine, uniform,
    random Dirichlet) and, if available, the previously discovered best
    construction (global variable ``height_sequence_1``).  The best
    candidate according to a fast FFT based score is selected.

2.  **Smooth mirror‑descent** –  a multiplicative (exponentiated‑gradient)
    update on the simplex with a *soft* max weight:
    w_i ∝ exp(β·(conv_i – max(conv))) .  This yields a smooth gradient
    that focuses on the nearly‑maximal convolution entries while staying
    well‑behaved.  The schedule η_t = η₀/√t and a fixed temperature β
    give stable progress.  The best iterate (according to the exact
    objective) is kept.

3.  **Hard sub‑gradient polishing** –  the final candidate is refined
    with the classic sub‑gradient method that uses the true max‑weight
    (uniform on the set of maximisers).  This finishes the optimisation
    within the remaining time budget.

The routine respects the supplied time limit (default 1000 s) and always
returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_smooth(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta: float = 10.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent using a softened max‑weight.
    The update is   a ← a * exp(-η·grad)  and the iterate is projected
    back onto the simplex.  The weight vector for the gradient is
    w_i ∝ exp(β·(conv_i – max(conv))) which approximates the sub‑gradient
    of the max‑function while remaining smooth.
    Returns the best iterate seen (on the exact objective) and its score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    # initial exact score (sum is 1, so score = 2·n·max(conv))
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    best_sc = 2.0 * n * max_val

    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # softmax‑type weights on the convolution entries
        exp_vals = np.exp(beta * (conv - max_val))   # max entry gets weight 1
        w = exp_vals / np.sum(exp_vals)             # ∑ w = 1

        # gradient of the smoothed objective
        grad = gradient_from_weights(a, w)

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update and simplex projection
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val          # exact score (sum(a)=1)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    smooth iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the true max‑convolution objective.
    Uses multiplicative (exponentiated‑gradient) updates and a 1/√t step‑size.
    Returns the best vector seen and its exact score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    n = len(a)
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # binary weight: uniform on the (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    score.  The routine respects the time budget, builds a few simple
    initial candidates, refines the best one with a smooth exponentiated‑
    gradient scheme and finally polishes it with a hard sub‑gradient
    descent.  The best sequence found is returned as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin to avoid a hard timeout
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 1.  Gather initial candidates (length 2048 is a good compromise)
    # ------------------------------------------------------------------
    n = 2048
    candidates: List[np.ndarray] = []

    # a) use a previously discovered construction if present
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.ndim == 1:
            fallback = normalize(fallback)
            candidates.append(fallback)
            print("[fallback] length =", len(fallback))
    except NameError:
        pass

    # b) simple parametric families
    candidates.append(arcsine_initial(n))
    candidates.append(uniform_shape(n))

    # c) a few random Dirichlet draws for diversity
    for _ in range(3):
        candidates.append(random_dirichlet(n, rng))

    # d) random power‑law shapes (helps exploring the neighbourhood)
    for _ in range(2):
        p = rng.uniform(0.30, 0.70)
        eps = rng.uniform(0.0, max(5.0, 0.015 * n))
        candidates.append(powerlaw_shape(n, p, eps))

    # ------------------------------------------------------------------
    # 2.  Quick ranking with the fast internal score
    # ------------------------------------------------------------------
    scored = [(compute_score(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    best_candidate = scored[0][1]
    best_fast = scored[0][0]
    print(f"Initial best fast score = {best_fast:.6f}")

    # keep a copy as current best (exact evaluation will be done later)
    current = best_candidate.copy()

    # ------------------------------------------------------------------
    # 3.  Smooth mirror‑descent refinement (uses ~half remaining time)
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    smooth_deadline = time.time() + remaining * 0.5
    print(f"Starting smooth refinement (deadline in {remaining*0.5:.1f}s)")

    refined_smooth, smooth_score = refine_smooth(
        a=current,
        deadline=smooth_deadline,
        eta0=1.0,
        beta=10.0,
        report_every=0,
    )
    print(f"Smooth refinement best score (exact) = {smooth_score:.6f}")

    # evaluate with the official routine for a reliable comparison
    val = evaluate_sequence(refined_smooth.tolist())
    print(f"Smooth refined official score = {val:.6f}")
    if val < best_val:
        best_val = val
        best_seq = refined_smooth.tolist()
        print("*** New global best after smooth refinement ***")

    # ------------------------------------------------------------------
    # 4.  Hard sub‑gradient polishing (uses the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\nStarting hard sub‑gradient polish")
        polished, _ = refine_subgradient(
            a=refined_smooth,
            deadline=deadline,
            max_iter=100_000,
            eta0=0.7,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished official score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after hard polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector (very unlikely to happen)
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
