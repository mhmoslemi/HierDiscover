def annealed_optimize(a_init, deadline, max_iter, T0=None, decay=0.99, eta0=0.7, report=False):
    a = a_init.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    conv_len = 2*n-1
    # Compute initial temperature as maybe max(conv) or median?
    conv = convolution_fft(a, a)
    max_conv = np.max(conv)
    if T0 is None:
        T = max_conv  # large initial T
    else:
        T = T0

    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # compute conv
        conv = convolution_fft(a, a)
        max_conv = np.max(conv)
        # compute softmax weights
        # stable: subtract max
        w = np.exp( (conv - max_conv) / T )
        w_sum = np.sum(w)
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to uniform
            w = np.full_like(conv, 1.0/conv_len)
        else:
            w = w / w_sum
        # compute gradient
        grad = gradient_from_weights(a, w)
        # step size
        eta = eta0 / np.sqrt(it)
        # limit large grads
        step = eta * grad
        # clip
        step = np.clip(step, -50, 50)
        a = a * np.exp(-step)
        # projection to simplex
        a = normalize(a)
        # track best
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # update temperature
        T = max(T * decay, 1e-8)
        # optionally random perturbation occasionally
        if it % 500 == 0:
            # small random jitter
            jitter = rng.normal(scale=0.01, size=n)  # zero-mean
            # add jitter but maintain positivity
            a = np.clip(a + jitter, 0, None)
            a = normalize(a)
            # optionally compute new best
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # optionally report
        if report and it % 1000 == 0:
            print(...)

    return best_a, best_score

assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function `a = (a₀,…,aₙ₋₁)` that minimizes  

\[
E(a)=\frac{2\,n\;\max\!\bigl((a*a)_k\bigr)}{\bigl(\sum_i a_i\bigr)^2}\; .
\]

`E(a)` is invariant under scaling, therefore the search is carried out on the
probability simplex (`∑aᵢ = 1`).  
The algorithm consists of three stages

1. **Wide parametric sweep** – a few hundred candidate shapes are generated
   from simple families (arcsine, power‑law, β‑shapes, semicircle,
   triangular, random Dirichlet, and mixtures of arcsine + uniform).  
   The cheap fast score `compute_score` is used to keep the best `k`
   candidates.

2. **Deterministic annealing (soft‑max gradient descent)** –  
   each retained candidate is refined by a smooth approximation of the
   `max` in the objective.  For a temperature `T` the weight vector  

   \[
   w_k=\frac{\exp\bigl((c_k-\max c)/T\bigr)}{\sum_j\exp\bigl((c_j-\max c)/T\bigr)}
   \]

   (where `c = a*a`) yields a differentiable surrogate
   `∑ w_k c_k`.  Its gradient is obtained by a single FFT‑based convolution
   (`gradient_from_weights`).  An exponentiated‑gradient (mirror‑descent)
   update keeps the iterate on the simplex.  The temperature is multiplied
   by a factor `<1` each iteration, thus gradually turning the smooth
   surrogate into the true `max`.  The routine runs until the global time
   budget expires.

3. **Final polishing** – a short run of the original sub‑gradient (hard‑max)
   mirror‑descent is applied to the current global best to squeeze the last
   bit of improvement.

All heavy work is performed with FFT‑based convolutions (`O(n log n)`);
for `n = 2048` a full iteration costs only a few milliseconds, so thousands
of iterations fit comfortably inside the 1000 s budget.

The code respects the required signature `propose_candidate(seed, budget_s,
**kwargs) → List[float]`, prints progress, and **always** returns a feasible
(non‑negative, sum > 0) sequence.

```python
"""
search_step_function_v2.py

A two‑stage optimizer for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)²)

The method first sweeps a collection of low‑dimensional families
(arcsine, power‑law, β‑shape, semicircle, triangular, mixtures, random
Dirichlet) and keeps the most promising candidates.  Each retained
candidate is then refined by a deterministic‑annealing gradient descent
that smooths the ``max`` via a soft‑max weighting; the temperature is
gradually reduced, turning the surrogate into the true objective.
A final short mirror‑descent (hard‑max) polishing step is applied to the
overall best vector.  All convolutions use FFT for speed, the iterate is
kept on the simplex with a multiplicative (exponentiated‑gradient) update,
and the routine respects a 1000 s wall‑time budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities (FFT‑based convolution, simplex projection,
# gradient of a weighted sum of convolutions)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * b using a real FFT.
    The result has length len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the scale‑invariant objective:
        2·n·max(conv(a,a)) / (sum(a)²)
    `a` is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m·w_{m+j}.
    Computed via a single FFT‑based convolution.
    """
    # w is defined on indices 0 … 2n‑2.  For the gradient we need the
    # reversed weight vector.
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                           # position of the central term
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial broad sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution ≈ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def semicircle_shape(n: int) -> np.ndarray:
    """Discrete semicircle density ≈ √(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    vals = np.sqrt((i + 0.5) * (n - 0.5 - i))
    return normalize(vals)


def triangular_shape(n: int) -> np.ndarray:
    """Symmetric linear ramp up to the centre, then down."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    vals = np.minimum(i, n - 1 - i) + 1.0   # +1 avoids zeros at the ends
    return normalize(vals)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Deterministic‑annealing gradient descent (soft‑max surrogate)
# ----------------------------------------------------------------------


def annealed_optimize(
    a_init: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 50_000,
    eta0: float = 0.8,
    decay: float = 0.995,
) -> Tuple[np.ndarray, float]:
    """
    Refines `a_init` using a smooth approximation of the max‑convolution.
    The temperature is multiplied by `decay` each iteration; as it cools,
    the algorithm behaves like the hard‑max sub‑gradient method.
    Returns the best vector seen and its fast score.
    """
    a = normalize(a_init.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    # Initial temperature based on the spread of the convolution values.
    conv = convolution_fft(a, a)
    spread = np.max(conv) - np.min(conv)
    T = max(spread * 0.5, 1e-12)          # avoid T = 0 at the start

    for it in range(1, max_iter + 1):
        if time.time() >= deadline - 0.01:
            break

        # current convolution
        conv = convolution_fft(a, a)
        max_c = np.max(conv)

        # soft‑max weights – stable version
        exp_vals = np.exp((conv - max_c) / T)
        w = exp_vals / np.sum(exp_vals)

        # gradient of the weighted sum (soft max)
        grad = gradient_from_weights(a, w)

        # step size schedule  η_t = η₀ / √t
        eta = eta0 / np.sqrt(it)

        # multiplicative (exponentiated‑gradient) update, clipped for safety
        upd = -eta * grad
        upd = np.clip(upd, -40.0, 40.0)     # prevents overflow in the exponent
        a = a * np.exp(upd)
        a = normalize(a)

        # track the best solution (according to the exact fast score)
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # temperature cooling
        T = max(T * decay, 1e-12)

        # occasional jitter to help escape shallow plateaus
        if (it % 500) == 0:
            jitter = rng.normal(scale=0.005, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = normalize(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient refinement (hard‑max) – used for final polishing
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
) -> Tuple[np.ndarray, float]:
    """
    Classic mirror‑descent on the true ``max`` objective.
    Returns the best vector encountered and its exact fast score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline - 0.01:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # binary weight vector selecting all maximising indices
        w = np.zeros(conv_len, dtype=float)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc

def generate_initial_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """
    Build a diverse collection of candidate vectors (all normalised to sum‑1).
    The pool contains deterministic shapes, a grid over a few parametric families,
    several mixtures of arcsine+uniform, and a handful of random Dirichlet draws.
    """
    cand: List[np.ndarray] = []

    # deterministic base families
    cand.append(uniform_shape(n))
    cand.append(arcsine_initial(n))
    cand.append(semicircle_shape(n))
    cand.append(triangular_shape(n))

    # power‑law grid (p in [0.3,0.8], eps relative to n)
    for p in np.linspace(0.30, 0.80, 9):
        for eps_rel in [0.0, 0.001 * n, 0.01 * n]:
            cand.append(powerlaw_shape(n, p, eps_rel))

    # beta‑shape grid (a parameter)
    for a in np.linspace(0.2, 5.0, 9):
        cand.append(beta_shape(n, a))

    # arcsine + uniform mixtures
    arcs = arcsine_initial(n)
    uni = uniform_shape(n)
    for w in np.linspace(0.0, 0.5, 7):      # weight of the uniform component
        mix = (1.0 - w) * arcs + w * uni
        cand.append(normalize(mix))

    # arcsine + beta mixtures (adds a second degree of freedom)
    for w in np.linspace(0.05, 0.3, 4):
        beta = beta_shape(n, rng.uniform(0.5, 3.0))
        mix = (1.0 - w) * arcs + w * beta
        cand.append(normalize(mix))

    # random Dirichlet samples
    for _ in range(8):
        cand.append(random_dirichlet(n, rng))

    return cand

        # 2. Quick ranking with the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 8                     # a modest number keeps the total run time small
        top_candidates = [c for _, c in scored[:top_k]]
        print("  top initial scores:",
              " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]))

        # --------------------------------------------------------------
        # 3. Refine each retained candidate with annealed gradient descent.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            cur_score = compute_score(cand)
            print(f"  candidate {idx + 1}/{len(top_candidates)} start_score = {cur_score:.6f}")

            # Estimate a reasonable iteration budget based on the remaining time.
            remaining = max(0.0, deadline - time.time())
            # roughly 15 iterations per second for n≈2000 → 0.07 s per 1 k it.
            max_iter = int(min(40_000, max(2_000, remaining * 15)))

            refined, _ = annealed_optimize(
                a_init=cand,
                deadline=deadline,
                rng=rng,
                max_iter=max_iter,
                eta0=0.8,
                decay=0.995,
            )

            # official evaluation (includes the safety‑clamping performed by
            # ``evaluate_sequence``)
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 4. Final polishing of the overall best sequence (hard‑max mirror‑descent).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best (hard‑max mirror‑descent)")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(80_000, max(5_000, remaining * 15)))

        polished, _ = refine_subgradient(
            a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
```
