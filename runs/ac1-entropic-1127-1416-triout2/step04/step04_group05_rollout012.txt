def project_simplex(v):
    """Project onto {x≥0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

                      lam_start=10.0, lam_factor=1.001, lam_max=1e5,
                      lr0=0.5, max_iters=10**7):

    conv = conv_self(a)
    best_score = 2.0 * n * conv.max()

    v = np.zeros_like(a)

    beta2 = 0.999
    eps = 1e-8
    lam = lam_start

    it = 0

        # compute conv
        conv = conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score

"""
Hybrid optimisation for the “max self‑convolution” problem.

Algorithm
---------
1. **Initialisation** – start from the best known sequence
   (global ``height_sequence_1`` if present) and normalise it to a
   probability vector (sum = 1).  If it does not exist a uniform vector of
   length 5146 is used.

2. **Smooth‑gradient Adam phase** – minimise a smooth approximation of
   ``max(conv)`` where `conv = a * a`.
   * The smooth max is the log‑sum‑exp with a temperature parameter λ.
   * λ is increased slowly (homotopy) so the optimiser first sees a
     well‑behaved landscape and ends on the true ∞‑norm.
   * Adam updates are performed on the simplex: after each step the
     vector is projected back onto the probability simplex.
   * The best vector (according to the *exact* objective
     ``2·n·max(conv)``) is retained.

3. **Targeted hill‑climbing** – from the best point of phase 2 the
   algorithm performs a large number of greedy mass‑transfer moves:
   mass is moved from an index with large sub‑gradient (i.e. contributing
   heavily to the current maximal convolution entry) to an index with
   small sub‑gradient.  Always‑accept improving moves and occasionally
   accept worsening moves (a tiny Metropolis temperature) to escape
   shallow plateaux.

4. **Random‑restart shortcuts** – while time permits the algorithm
   launches a few short Adam runs from fresh Dirichlet draws; any
   improvement replaces the current best.

The whole routine respects the 1000 s budget, returns the best
non‑negative sequence it has found, and works entirely on the simplex,
so ``evaluate_sequence`` never rejects the candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:        # all entries are ≤ 0
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise (tiny drift may appear)
    w_sum = w.sum()
    if w_sum == 0.0:
        # fall back to uniform
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------


def self_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) convolution a * a using FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return conv


def smooth_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute the (soft‑max) weight vector for a given λ.
    The weights sum to one and concentrate on the largest convolution
    entries as λ grows.
    """
    max_val = conv.max()
    # shift to avoid overflow (the exponent is always ≤ 0)
    shifted = lam * (conv - max_val)
    # numerical safety: clip extreme negatives before exp
    # (underflow to 0 is fine, overflow cannot happen because shifted ≤ 0)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    return w


def smooth_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) w.r.t. the vector a.
    Formula: ∂L/∂a = 2 * (w * a) where * denotes linear convolution.
    The result is sliced to length n (the original size of a).
    """
    n = a.shape[0]
    # full convolution length = len(w)+len(a)-1 = (2n‑1)+n‑1 = 3n‑2
    L = 1 << ((len(w) + n - 2).bit_length())   # next power of two
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    full = np.fft.irfft(W * A, n=L)
    start = len(w) - 1            # index of the first coefficient that
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient for the exact max (used in the hill‑climbing phase)
# ----------------------------------------------------------------------


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at point a.
    Averages the exact gradients of all (approximately) active convolution
    entries (those attaining the current maximum up to a tiny tolerance).
    """
    n = a.shape[0]
    max_b = conv.max()
    eps = max_b * 1e-9
    active = np.where(conv >= max_b - eps)[0]

    if active.size == 0:                     # safety fallback
        return 2.0 * a

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


# ----------------------------------------------------------------------
# Gradient‑descent (Adam) on the smooth objective
# ----------------------------------------------------------------------


def adam_optimize(a_init: np.ndarray,
                  n: int,
                  deadline: float,
                  lam_start: float = 10.0,
                  lam_factor: float = 1.001,
                  lam_max: float = 1e5,
                  base_lr: float = 0.5,
                  max_iters: int = 10_000_000) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam optimiser on the smoothed max‑convex objective.
    Returns (best_a, best_score, last_a).
    """
    a = a_init.copy()
    best_a = a.copy()
    conv = self_convolution(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # exact objective (for bookkeeping)
        conv = self_convolution(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] iter {it:6d}: improved best {best_score:.6f}")

        # smooth weights and gradient
        w = smooth_weights(conv, lam)
        grad = smooth_gradient(a, w)

        # Adam update (no bias‑correction for speed)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # step size decays like 1/√t
        lr = base_lr / math.sqrt(it + 1.0)

        a = a - lr * m / (np.sqrt(v) + eps)
        a = project_simplex(a)

        # increase λ (homotopy)
        lam = min(lam_max, lam * lam_factor)

        it += 1

    return best_a, best_score, a


# ----------------------------------------------------------------------
# Local hill‑climbing (mass‑transfer) refinement
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray,
               n: int,
               deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill climbing guided by the sub‑gradient.
    Improves the solution found by the Adam phase.
    """
    a = a_start.copy()
    conv = self_convolution(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    it = 0
    # a tiny temperature for occasional uphill moves
    T = cur_score * 0.10
    T_end = cur_score * 1e-5

    while time.time() < deadline:
        # obtain a sub‑gradient of the true max‑convolution
        grad = subgradient_of_max_conv(a, conv)

        # choose source (large gradient) and target (small gradient)
        src_probs = grad / grad.sum()
        tgt_probs = (1.0 / (grad + 1e-12))
        tgt_probs /= tgt_probs.sum()

        src = np.random.choice(n, p=src_probs)
        tgt = np.random.choice(n, p=tgt_probs)

        if src == tgt or a[src] <= 0.0:
            it += 1
            continue

        # transfer a random fraction (up to 30 %) of the source mass
        delta = a[src] * np.random.rand() * 0.30
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        conv_c = self_convolution(cand)
        cand_score = 2.0 * n * conv_c.max()

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-12))

        if accept:
            a = cand
            conv = conv_c
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[Hill] iter {it:7d}: new best {best_score:.6f}")

        # geometric cooling
        T = max(T_end, T * 0.9995)

        # occasional diversification (hard restart) – cheap but powerful
        if it % 100_000 == 0 and it > 0:
            a = np.random.dirichlet(np.ones(n))
            conv = self_convolution(a)
            cur_score = 2.0 * n * conv.max()
            print(f"[Hill] iter {it}: random restart performed")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point – the function asked for by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑height sequence that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered (normalised to sum = 1, which
    is safe for the evaluator).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # 1 – problem size & starting point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"[Main] Using existing height_sequence_1 of length {n}.")
    except NameError:
        n = 5146
        init = np.ones(n, dtype=float) / n
        print(f"[Main] No existing sequence – using uniform start, length {n}.")

    # Normalise to a probability vector (sum == 1) and guard against NaNs
    init = np.nan_to_num(init, nan=0.0, posinf=0.0, neginf=0.0)
    init = np.maximum(0.0, init)
    if init.sum() == 0.0:
        init = np.ones_like(init) / init.shape[0]
    a0 = init / init.sum()

    # --------------------------------------------------------------
    # 2 – Adam optimisation on a smoothed max‑convolution
    # --------------------------------------------------------------
    phase1_deadline = deadline - 200.0           # keep ~200 s for polishing
    best_a, best_score, cur_a = adam_optimize(
        a0, n, phase1_deadline,
        lam_start=10.0,
        lam_factor=1.001,
        lam_max=1e5,
        base_lr=0.5
    )
    print(f"[Main] Adam phase finished. Best score so far {best_score:.6f}")

    # --------------------------------------------------------------
    # 3 – Targeted hill‑climbing refinement
    # --------------------------------------------------------------
    best_a, best_score = hill_climb(best_a, n, deadline)
    print(f"[Main] Hill‑climbing finished. Final best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4 – Optional quick random‑restart Adam bursts (time permitting)
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:  # leave a tiny slack for return
        # short burst from a fresh Dirichlet start
        a_rand = np.random.dirichlet(np.ones(n))
        burst_deadline = min(deadline, time.time() + 15.0)  # 15 s per burst
        rand_best, rand_score, _ = adam_optimize(
            a_rand, n, burst_deadline,
            lam_start=10.0,
            lam_factor=1.002,
            lam_max=5e4,
            base_lr=0.4,
            max_iters=3000
        )
        if rand_score < best_score:
            best_score = rand_score
            best_a = rand_best
            print(f"[Main] Random‑restart burst improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5 – Return the best sequence (still normalised to sum == 1)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
