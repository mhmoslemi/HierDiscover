A fast stochastic optimizer for the “peak autocorrelation” problem.
It combines:

*  A broad parametric sweep (uniform, arcsine, symmetric beta,
   power‑law, random Dirichlet, and mixtures).
*  A smooth sub‑gradient descent on the simplex using exponentiated‑gradient
   updates.  The sub‑gradient is taken w.r.t. a softened Lp‑approximation
   of the max‑convolution (soft‑max with a temperature that is increased
   during the run, i.e. simulated‑annealing of the objective).
*  A final polishing stage on the globally best shape with a long run.

All heavy evaluation is performed with an FFT‑based fast score
(identical to ``evaluate_sequence`` up to the clipping performed inside
the official function).  The official function is called only for the
few final candidates, keeping the wall‑time well below the 1000 s
budget.

The routine returns a feasible list of non‑negative floats and prints
progress information and the best score obtained.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A * B
    conv = np.fft.irfft(C, n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective used by evaluate_sequence.
    a is assumed non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m·w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    # convolution of a with reversed weight vector
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)   # length = len(a) + len(w) - 1 = 3n-2
    idx = 2 * len(a) - 2                # position of the central element
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    # a < 1 gives singularities at the ends (useful for our problem)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Smooth sub‑gradient descent (exponentiated‑gradient on the simplex)
# ----------------------------------------------------------------------


def refine_smooth(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 100_000,
    eta0: float = 0.6,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient refinement using a soft‑max weighting of the
    convolution entries.  The temperature parameter ``beta`` is increased
    linearly from ``beta_start`` to ``beta_end`` so the method first sees
    a smoothed version of the max‑function and then focuses on the true
    peak.  The iterate is always projected onto the probability simplex.

    Returns the best vector encountered and its internal score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution and soft‑max weights
        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # linear schedule for beta
        beta = beta_start + (beta_end - beta_start) * (it - 1) / max_iter

        # soft‑max: w_i ∝ exp(beta * (conv_i - max_val))
        # this keeps the maximum entry weight = 1, others ≤ 1
        exp_arg = beta * (conv - max_val)
        # avoid overflow/underflow
        np.clip(exp_arg, -700, 700, out=exp_arg)
        w = np.exp(exp_arg)
        w_sum = np.sum(w)
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.zeros_like(conv)

        # gradient of the weighted sum of convolutions
        grad = gradient_from_weights(a, w)

        # exponentiated‑gradient update
        eta = eta0 / math.sqrt(it)
        # limit gradient magnitude to avoid numerical blow‑up
        np.clip(grad, -200, 200, out=grad)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    iter={it:,d} beta={beta:.1f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    score.  The routine respects the time budget, explores several
    parametric families, keeps the top few shapes, refines them with a
    smooth sub‑gradient method, and finally returns the best sequence
    found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Slight safety margin before the hard timeout
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Use any previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore.  1024 and 2048 give a good trade‑off; 4096 is
    # attempted only if time permits.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    # Global list of (score, vector) for the best candidates across lengths
    global_best_candidates: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n>>> Exploring length n = {n}")

        # --------------------------------------------------------------
        # 1.  Generate a fairly large pool of random shapes.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # deterministic seeds
        candidates.append(uniform_shape(n))
        candidates.append(arcsine_initial(n))

        # random power‑law shapes
        for _ in range(12):
            p = rng.uniform(0.2, 0.9)                 # exponent controlling tail heaviness
            eps = rng.uniform(0.0, max(5.0, 0.01 * n))
            candidates.append(powerlaw_shape(n, p, eps))

        # symmetric beta‑like shapes (including a < 0.5)
        for _ in range(12):
            a = rng.uniform(0.1, 1.2)                 # a=0.5 gives arcsine; smaller = sharper ends
            candidates.append(beta_shape(n, a))

        # mixtures of arcsine with uniform (helps smoothing)
        for _ in range(8):
            w = rng.uniform(0.0, 0.5)
            mix = (1.0 - w) * arcsine_initial(n) + w * uniform_shape(n)
            candidates.append(normalize(mix))

        # random Dirichlet draws
        for _ in range(10):
            candidates.append(random_dirichlet(n, rng))

        # --------------------------------------------------------------
        # 2.  Quick ranking with the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 20
        top_scored = scored[:top_k]
        print("  top initial scores:",
              " ".join(f"{sc:.6f}" for sc, _ in top_scored[:5]))

        # Store globally for later refinement
        for sc, vec in top_scored:
            global_best_candidates.append((sc, vec))

    # Keep only the globally best 30 candidates (to limit refinement work)
    global_best_candidates.sort(key=lambda x: x[0])
    global_best_candidates = global_best_candidates[:30]

    # ------------------------------------------------------------------
    # 3.  Refine each retained candidate with smooth sub‑gradient descent.
    # ------------------------------------------------------------------
    for idx, (init_sc, cand) in enumerate(global_best_candidates):
        if time.time() >= deadline - 5.0:
            break
        print(f"\nRefining candidate {idx + 1}/{len(global_best_candidates)} (init score {init_sc:.6f})")

        # Allocate iteration budget proportional to remaining time.
        time_left = max(0.0, deadline - time.time())
        # Rough estimate: ~0.00015 s per iteration for n≈2000 → ~6600 it/s.
        # Use a conservative factor of 3000 it/s.
        max_iter = int(min(120_000, max(5_000, time_left * 3000)))

        refined, refined_sc = refine_smooth(
            a=cand,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.6,
            beta_start=5.0,
            beta_end=200.0,
            report_every=0,
        )
        # Official evaluation (includes the clipping performed inside evaluate_sequence)
        val = evaluate_sequence(refined.tolist())
        print(f"    official score = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print(f"    *** New global best: {best_val:.6f} (n={len(refined)}) ***")

    # ------------------------------------------------------------------
    # 4.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)
        time_left = max(0.0, deadline - time.time())
        max_iter = int(min(250_000, max(20_000, time_left * 3000)))

        polished, _ = refine_smooth(
            a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
            beta_start=20.0,
            beta_end=400.0,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector of moderate size
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
```
