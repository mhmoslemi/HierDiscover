def simulated_annealing(a0, deadline, rng, max_iter=2000, temp0=1e-3, cooling=0.999):
   a = a0.copy()
   cur_score = compute_score(a)
   best_a = a.copy()

   for iter in range(max_iter):
       if time.time() >= deadline:
           break
       # propose
       b = propose_neighbor(a, rng, max_step=0.02)
       new_score = compute_score(b)
       delta = new_score - cur_score
       if delta < 0 or rng.random() < np.exp(-delta / temp):
           a = b
           cur_score = new_score
           if cur_score < best_score:
               best_score = cur_score
               best_a = a.copy()
       temp *= cooling
   return best_a, best_score

def softmax_gradient_descent(initial_a, deadline, betas, iter_per_beta=2000, eta0=0.5):
   a = np.maximum(initial_a, 1e-30)
   a = a / a.sum()
   best_a = a.copy()
   best_sc = compute_score(a)
   n = len(a)
   for beta in betas:
       for it in range(1, iter_per_beta+1):
           if time.time() >= deadline:
               return best_a, best_sc
           conv = convolution_fft(a, a)  # length m
           # compute softmax weights
           # to avoid overflow, subtract max before exponentiate.
           max_c = conv.max()
           exp_vals = np.exp(beta * (conv - max_c))
           sum_exp = exp_vals.sum()

           # length N for linear correlation: 2n-1 (m) + n -1 = 3n-2
           N = next_pow2(len(p) + len(a) - 1)
           P = np.fft.fft(p, N)

           # cross-correlation = ifft(P * conj(A))

           grad = 4.0 * n * cross  # shape n
           eta = eta0 / np.sqrt(it)
           # exponentiated update
           a = a * np.exp(-eta * grad)
           # renormalize
           a = a / a.sum()
           cur_sc = compute_score(a)
           if cur_sc < best_sc:
               best_sc = cur_sc
               best_a = a.copy()
   return best_a, best_sc

"""
Hybrid search for a low‑max‑convolution step function.

The routine combines three ideas:

1. **Diverse seeds** – families (arcsine, power‑law, beta‑like, Dirichlet,
   mixtures) are sampled for a few lengths (powers of two).  The cheap
   “compute_score” (the official objective without the security‐checks)
   ranks the seeds.

2. **Local stochastic improvement** – a short simulated‑annealing run moves
   tiny mass between random bins while always respecting the sum‑to‑1
   simplex.  This quickly discovers better basins that pure sub‑gradient
   descent often misses.

3. **Smooth max‑gradient descent** – the non‑smooth max is replaced by a
   soft‑max (log‑sum‑exp) with a schedule of increasing β.  The exact
   gradient of the smooth surrogate is a cross‑correlation that can be
   evaluated with a single FFT, so each iteration is O(n log n).  An
   exponentiated‑gradient step (multiplicative update) maintains non‑negativity
   and is projected onto the simplex after each step.

The best vector found is finally polished with the original exponentiated
sub‑gradient method that directly minimises the true max‑convolution.
The whole search respects the supplied time budget (default 1000 s) and
always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative; the function works for any shape.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Simulated annealing (local stochastic improvement)
# ----------------------------------------------------------------------


def propose_neighbor(a: np.ndarray, rng: np.random.Generator, max_step: float = 0.02) -> np.ndarray:
    """Move a small amount of mass from one random bin to another."""
    n = len(a)
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a  # nothing to move
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000,
                        start_temp: float = 0.005,
                        cooling: float = 0.9995,
                        max_step: float = 0.02) -> Tuple[np.ndarray, float]:
    """
    Very light SA that preserves the simplex.
    Returns the best vector seen and its fast score.
    """
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    temp = start_temp

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        cand = propose_neighbor(a, rng, max_step)
        cand_sc = compute_score(cand)
        delta = cand_sc - cur_sc
        if delta < 0.0 or rng.random() < np.exp(-delta / max(1e-12, temp)):
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        temp *= cooling
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 500,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution:
        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k).

    An exponentiated‑gradient update is performed, followed by a projection
    onto the probability simplex (simple renormalisation).
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            # convolution a∗a
            conv = convolution_fft(a, a)                     # length 2n‑1
            # soft‑max weights (stable version)
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()                    # Σ p = 1

            # cross‑correlation p ⋆ a  (linear, length n)
            N = next_pow2(len(p) + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]     # = Σ_k p_k·a_{k‑j}
            cross = np.maximum(cross, 0.0)                   # numerical safety

            # gradient of the smooth objective
            grad = 4.0 * n * cross                           # non‑negative

            # exponentiated‑gradient step with 1/√t schedule
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # renormalise onto the simplex
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient on the true max‑convolution (final polish)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k, the gradient contribution is 2·a_{k‑j}.
    When several offsets share the maximal value they are averaged.
    """
    n = len(a)
    max_val = conv.max()
    # active offsets (exact equality; tolerance is tiny because values are exact floats)
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,
                         max_iter: int = 200_000,
                         noise_every: int = 0,
                         noise_scale: float = 0.0) -> Tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent directly on Φ(a)=max_k (a∗a)_k.
    Returns the best vector seen and its (true) fast score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine respects a wall‑clock budget (default 1000 s) and
    always returns a feasible list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Explore several sequence lengths (powers of two are FFT‑friendly).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a diverse library of starting vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # (i) uniform
        candidates.append(np.full(n, 1.0 / n))

        # (ii) classic arcsine
        candidates.append(arcsine_initial(n))

        # (iii) random power‑law shapes
        for _ in range(5):
            p = rng.uniform(0.2, 1.2)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            candidates.append(generate_powerlaw(n, p, eps))

        # (iv) beta‑like shapes
        for _ in range(3):
            alpha = rng.uniform(0.5, 5.0)
            beta = rng.uniform(0.5, 5.0)
            candidates.append(generate_beta_like(n, alpha, beta))

        # (v) Dirichlet / exponential draws
        for conc in [0.3, 1.0, 2.0]:
            candidates.append(random_dirichlet(n, conc, rng))

        # (vi) perturbed arcsine (helps to leave the immediate basin)
        for _ in range(2):
            candidates.append(perturb(arcsine_initial(n), rng, scale=0.02))

        # (vii) simple mixtures of arcsine and uniform
        for _ in range(2):
            w = rng.random()
            mix = w * arcsine_initial(n) + (1.0 - w) * np.full(n, 1.0 / n)
            candidates.append(mix / mix.sum())

        # --------------------------------------------------------------
        # 1b.  Quick sweep with the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:5]]   # keep the five most promising

        # --------------------------------------------------------------
        # 1c.  Refine each retained candidate.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # ---- Simulated annealing -------------------------------------------------
            max_iter_sa = int(min(3000, (deadline - time.time()) * 250))
            start_temp = max(0.005, 0.03 * compute_score(cand))
            sa_a, _ = simulated_annealing(cand, deadline, rng,
                                          max_iter=max_iter_sa,
                                          start_temp=start_temp,
                                          cooling=0.9995,
                                          max_step=0.02)

            val_sa = evaluate_sequence(sa_a.tolist())
            print(f"    SA evaluated score = {val_sa:.6f}")
            if val_sa < best_val:
                best_val = val_sa
                best_seq = sa_a.tolist()
                print(f"    *** New global best (SA) {best_val:.6f} ***")

            # ---- Smooth‑max gradient descent ----------------------------------------
            # a modest β schedule; later stages tighten the approximation.
            betas = [5, 15, 30, 60, 120]
            # number of inner iterations per β – adapt to remaining time
            remaining_time = max(0.0, deadline - time.time())
            it_per_beta = max(150, int(min(1500, remaining_time * 180) / len(betas)))
            sg_a, _ = softmax_gradient_descent(sa_a, deadline,
                                               betas=betas,
                                               iter_per_beta=it_per_beta,
                                               eta0=0.4)

            val_sg = evaluate_sequence(sg_a.tolist())
            print(f"    smooth‑gd evaluated score = {val_sg:.6f}")
            if val_sg < best_val:
                best_val = val_sg
                best_seq = sg_a.tolist()
                print(f"    *** New global best (smooth‑gd) {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing with max‑convolution sub‑gradient")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(200_000, remaining * 300))
        a_refined, _ = optimise_subgradient(initial_a=a,
                                            deadline=deadline,
                                            eta0=0.3,
                                            max_iter=max_iter,
                                            noise_every=0,
                                            noise_scale=0.0)
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
