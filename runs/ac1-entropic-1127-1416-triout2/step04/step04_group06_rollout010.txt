   For a given smoothing parameter ``α`` the gradient w.r.t. the
   probability vector ``a`` is
        g = 2·corr(p , a)            where  p = softmax(α·conv)
   and ``corr`` is the (non‑circular) cross–correlation.  We work in
   the log‑domain ``w = log a`` and apply Adam updates; the
   probabilities are recovered by a softmax (which automatically enforces
   positivity and unit sum).

3. The smoothing parameter is increased linearly from ``α_start`` to
   ``α_end`` (typically 1 → 500).  The learning rate decays as
   ``lr / sqrt(t)``.

4. After each Adam run a short simulated‑annealing (mass‑transfer)
   refinement is performed to escape possible local minima.

5. Multiple random restarts (Dirichlet, Beta‑U‑shapes, edge‑heavy,
   sparse) are executed until the time budget (≈1000 s) is exhausted.
   The best vector seen over all restarts is returned.

The whole routine uses only NumPy (FFT based convolutions → O(n log n))
and fits comfortably into the allotted runtime.
"""

import time
import math
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def compute_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a vector already on the simplex."""
    n = a.shape[0]
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def softmax_alpha(x: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max of ``alpha·x`` with numerical stability."""
    m = np.max(x)
    ex = np.exp(alpha * (x - m))
    s = ex.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full_like(x, 1.0 / x.size)
    return ex / s


def cross_corr(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ p) of length ``len(a)``.
    Result = Σ_k p_k·a_{k‑i} (zero outside the support).
    """
    n = a.shape[0]
    L = next_pow2(n + p.size - 1)          # ≥ 3·n‑2
    fa = np.fft.rfft(a, n=L)
    fp = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(fa * np.conj(fp), n=L)
    offset = p.size - 1                     # lag 0 aligns here
    return corr[offset : offset + n]


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discrete Beta(α,β) density on ``n`` points, normalised to unit sum."""
    x = (np.arange(n) + 0.5) / n
    # log‑density; clip to avoid overflow/underflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return simplex_normalise(w)


def dirichlet_init(n: int) -> np.ndarray:
    """Random Dirichlet(1,…,1) (i.e. i.i.d. Gamma(1))."""
    a = np.random.gamma(1.0, 1.0, size=n)
    return simplex_normalise(a)


def edge_concentrated_init(n: int) -> np.ndarray:
    """U‑shaped: larger values at the two ends, smaller in the middle."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)          # distance to the nearer edge
    a = (dist.astype(float) + 1.0)               # edges get larger weight
    return simplex_normalise(a)


def sparse_init(n: int, k: int = 12) -> np.ndarray:
    """Place mass on ``k`` random positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


def upsample_to_target(arr: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of ``arr`` to length ``target_n`` (preserves mass)."""
    if arr.size == target_n:
        return arr.copy()
    src_x = np.linspace(0.0, 1.0, arr.size)
    dst_x = np.linspace(0.0, 1.0, target_n)
    up = np.interp(dst_x, src_x, arr)
    up = np.maximum(0.0, up)
    return simplex_normalise(up)


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found (as a Python list, sum == 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    # ------------------------------------------------------------------
    # 0. Choose working length ------------------------------------------------
    # ------------------------------------------------------------------
    # Target length – a power of two gives fast FFTs.
    DEFAULT_N = 8192
    try:
        # ``height_sequence_1`` may be present in the global scope.
        prev = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = max(prev.size, DEFAULT_N)
        n = next_pow2(n)          # enforce power of two for speed
        best_a = upsample_to_target(prev, n)
        best_score = compute_score(best_a)
        print(f"[Init] Loaded previous solution, up‑sampled to n={n}, score={best_score:.6f}")
    except NameError:
        n = DEFAULT_N
        best_a = np.full(n, 1.0 / n, dtype=float)   # uniform start
        best_score = compute_score(best_a)
        print(f"[Init] No previous solution – using uniform start, n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Pre‑compute FFT size (same for every iteration) -------------------------
    # ------------------------------------------------------------------
    L_fft = next_pow2(3 * n - 2)                 # >= n + (2n‑1) – 1

    # ------------------------------------------------------------------
    # 2. Helper to evaluate a vector quickly (already normalised)
    # ------------------------------------------------------------------
    def eval_fast(v: np.ndarray) -> float:
        return compute_score(v)

    # ------------------------------------------------------------------
    # 3. Collection of simple initialisers (all produce sum = 1)
    # ------------------------------------------------------------------
    init_functions = [
        lambda: beta_sequence(np.random.uniform(0.1, 0.9),
                              np.random.uniform(0.1, 0.9), n),  # heavy edges (U‑shape)
        lambda: beta_sequence(np.random.uniform(0.5, 3.0),
                              np.random.uniform(0.5, 3.0), n),  # bell‑shaped
        lambda: dirichlet_init(n),
        lambda: edge_concentrated_init(n),
        lambda: sparse_init(n, k=np.random.randint(6, 25)),
    ]

    # ------------------------------------------------------------------
    # 4. Optimisation primitives
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    restart = 0
    while time.time() < deadline - 4.0:       # leave a few seconds for finalisation
        # ----- draw a fresh start vector ---------------------------------
        init_fn = np.random.choice(init_functions)
        a = init_fn()
        # occasional blending with the current best solution
        if np.random.rand() < 0.15:
            a = simplex_normalise(0.5 * a + 0.5 * best_a)

        # ----- Adam variables -------------------------------------------------
        w = np.log(a + eps)                     # work in log‑domain
        m = np.zeros(n, dtype=float)
        v = np.zeros(n, dtype=float)

        # ----- schedule parameters -------------------------------------------
        max_iters = 6000
        lr0 = 0.30
        alpha_start = 1.0
        alpha_end = 500.0

        # ----- main Adam‑mirrored‑descent loop -------------------------------
        t = 0
        for it in range(max_iters):
            # time‑out guard
            if time.time() > deadline - 2.0:
                break

            # increasing smoothing parameter
            alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

            # ---- forward pass ------------------------------------------------
            # FFT of current a (length L_fft)
            fa = np.fft.rfft(a, n=L_fft)

            # self‑convolution (linear, length 2n‑1)
            conv_full = np.fft.irfft(fa * fa, n=L_fft)
            conv = conv_full[: 2 * n - 1]

            # soft‑max over convolution entries (smooth max)
            p = softmax_alpha(conv, alpha)

            # ---- gradient of the smooth max ---------------------------------
            # cross‑correlation a ⋆ p  (length n)
            g = 2.0 * cross_corr(a, p)          # factor 2 from the exact derivative

            # gradient w.r.t. log‑probabilities w
            grad_w = g * a                      # element‑wise

            # ---- Adam update -------------------------------------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_w
            v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = lr0 / math.sqrt(it + 1.0)      # decaying learning rate
            w -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- map back to the simplex ------------------------------------
            # softmax ensures positivity and unit sum
            w_shift = w - np.max(w)             # improve numerical stability
            a = np.exp(w_shift)
            a /= a.sum()

            # ---- occasional progress check ----------------------------------
            if (it & 63) == 0:                  # every 64 iterations
                cur_score = eval_fast(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Restart {restart}] it {it:4d} → new best {best_score:.6f}")

        # ----- short simulated‑annealing refinement -------------------------
        cur_a = best_a.copy()
        cur_score = eval_fast(cur_a)
        T = cur_score * 0.02               # start temperature (fraction of score)
        T_end = cur_score * 1e-5
        while time.time() < deadline - 1.0 and T > T_end:
            # pick a source index proportionally to its mass
            src = np.random.choice(n, p=cur_a)
            # pick a different target uniformly
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1
            if cur_a[src] <= 0.0:
                continue
            delta = cur_a[src] * np.random.rand() * 0.5   # move at most 50 % of src
            cand = cur_a.copy()
            cand[src] -= delta
            cand[tgt] += delta

            cand_score = eval_fast(cand)
            if cand_score < cur_score:
                accept = True
            else:
                accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

            if accept:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[SA   R{restart}] iter → new best {best_score:.6f}")

            T *= 0.9993   # slow geometric cooling

        restart += 1

    # ------------------------------------------------------------------
    # 5. Final polishing (one longer Adam run on the current best)
    # ------------------------------------------------------------------
    final_a = best_a.copy()
    w = np.log(final_a + eps)
    m = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)
    lr0 = 0.25
    alpha_start = 1.0
    alpha_end = 800.0
    max_iters = 8000
    for it in range(max_iters):
        if time.time() > deadline - 0.5:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        fa = np.fft.rfft(final_a, n=L_fft)
        conv_full = np.fft.irfft(fa * fa, n=L_fft)
        conv = conv_full[: 2 * n - 1]

        p = softmax_alpha(conv, alpha)
        g = 2.0 * cross_corr(final_a, p)
        grad_w = g * final_a

        t = it + 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / math.sqrt(it + 1.0)
        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        w_shift = w - np.max(w)
        final_a = np.exp(w_shift)
        final_a /= final_a.sum()

        if (it & 127) == 0:
            cur = eval_fast(final_a)
            if cur < best_score:
                best_score = cur
                best_a = final_a.copy()
                print(f"[Final] it {it:4d} → new best {best_score:.6f}")

    print(f"[Done] Best score after {restart} restarts: {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
```
