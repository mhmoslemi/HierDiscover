"""
search_step_function.py

A time‑bounded optimiser for the non‑negative step‑function problem
defined by ``evaluate_sequence``.  The program proceeds in three
phases:

1.  **Broad random sampling** –  millions of candidate density
    functions are generated by a two‑parameter asymmetric Beta family
    (left and right shape exponents) together with a small additive
    epsilon.  The cheap surrogate ``2·n·max(conv)`` is used to keep the
    best few candidates for each length.

2.  **Multi‑stage local refinement** – every kept candidate is
    projected onto the simplex and then improved by
        * Adam on a log‑sum‑exp (soft‑max) surrogate of the max‑convolution,
        * a projected sub‑gradient descent on the true max,
        * a brief stochastic local search (Gaussian perturbations).
    All steps respect the remaining time budget.

3.  **Global polishing** – the overall best vector obtained from all
    lengths is polished again with a long Adam run followed by the same
    sub‑gradient and random walk stages.

The implementation uses only NumPy and standard Python, works on a
single CPU and guarantees a non‑empty return value.  It is deliberately
written without closures or lambda expressions so that the competition
harness can import the ``propose_candidate`` function directly.
"""

import time
import numpy as np
from typing import List, Tuple

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(a,a)) (sum(a) must be 1)."""
    return 2.0 * a.size * float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1} (Duchi et al. 2008)."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Shape generators
# --------------------------------------------------------------

def generate_beta_general(
    n: int,
    left_alpha: float,
    right_alpha: float,
    eps: float,
) -> np.ndarray:
    """
    Asymmetric beta‑family density:
        a_i ∝ (i+ε)^{αL‑1}·(n‑1‑i+ε)^{αR‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (left_alpha - 1.0)
    right = (n - 1 - i + eps) ** (right_alpha - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) starter."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


# --------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# --------------------------------------------------------------

def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of  log‑sum‑exp  approximation of  max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # Stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # softmax weights, sum 1
    B = max_c + np.log(sum_exp) / lam            # smooth max

    # Weighted quadratic term ∂/∂a Σ_k w_k (a*a)_k
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 150.0,
    noise_every: int = 1000,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser of the smooth surrogate.
    Returns best vector found and its (fast) surrogate score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = max_convolution(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # Quadratic temperature schedule for lambda (smoothness)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = max_convolution(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 10_000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# --------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# --------------------------------------------------------------

def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 80_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true maximum of the convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = max_convolution(a)

    n = a.size

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        idx = int(np.argmax(conv))

        # Gradient of the active term (linearity in a)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = max_convolution(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 20_000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

# Stochastic local search
# --------------------------------------------------------------

def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 120_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple random walk on the simplex with occasional acceptance of
    non‑improving moves (tiny probability) to escape shallow minima.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = max_convolution(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = max_convolution(a_prop)

        # Accept improvements; occasionally accept worse ones
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 30_000 == 0:
            print(
                f"    Random local iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

                )

            # Stochastic local walk (tiny step size)
            if time.time() < deadline - 2.0:
                max_iter_loc = int(min(100_000, (deadline - time.time()) * 30))
                refined_a, _ = random_local_search(
                    refined_a,
                    time_deadline=deadline,
                    init_step=0.018,
                    max_iter=max_iter_loc,
                )

            # Official (slow) evaluation – once per length
            val = evaluate_sequence(refined_a.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:
            # No refinement performed – fall back to raw best
            val = evaluate_sequence(local_best_a.tolist())
            print(f"  Raw candidate official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, local_best_a.tolist()
                print("*** Updated global best (raw) ***")

    # ----------------------------------------------------------
    # 3.  Final polishing of the overall best vector
    # ----------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Final global polishing ===")
        a = np.array(best_seq, dtype=float)

        # Stage A – long Adam run with a very high λ to mimic true max
        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 35))
        print(f"  Adam polishing – max_iter={max_iter_adam}")
        a, _ = adam_optimize(
            init_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.15,
            lam_start=1.0,
            lam_end=200.0,
            noise_every=1000,
            noise_scale=4e-5,
        )

        # Stage B – sub‑gradient polishing
        if time.time() < deadline - 3.0:
            max_iter_sub = int(min(120_000, (deadline - time.time()) * 30))
            print(f"  Sub‑gradient polishing – max_iter={max_iter_sub}")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.20,
                max_iter=max_iter_sub,
            )

        # Stage C – stochastic local walk
        if time.time() < deadline - 2.0:
            max_iter_loc = int(min(200_000, (deadline - time.time()) * 40))
            print(f"  Random walk polishing – max_iter={max_iter_loc}")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.015,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ----------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ----------------------------------------------------------
    if not best_seq:
        # Simple safe fallback – uniform 256‑vector
        best_seq = [1.0 / 256] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
