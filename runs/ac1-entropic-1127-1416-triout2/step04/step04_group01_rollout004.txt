"""
Propose a low‑discrepancy step sequence with a very low
evaluate_sequence() value.

The algorithm works in stages:

1.  Start with a modest size (n = 1024) and generate a few
    diverse seeds: the classic arcsine shape, the uniform
    vector, an up‑sampled copy of the best vector found so far,
    and a handful of random Beta‑family shapes.

2.  For each seed run a short but aggressive optimisation
    pipeline:
        * exact sub‑gradient descent on   f(a)=2·n·max(conv(a,a))
        * Adam optimisation on a smooth soft‑max surrogate
          of the max‑convolution
        * very cheap simulated‑annealing polishing

    All updates keep the vector on the probability simplex
    (sum = 1, non‑negative), therefore the official objective
    (which is homogeneous) is unchanged.

3.  The best vector from step 2 is up‑sampled to the next larger
    length (2048 → 4096 → 8192 → 16384) by linear interpolation
    and refined again with the same pipeline.  Larger lengths
    approximate the continuous limit better, which lowers the
    asymptotic constant 2·max(f∗f).

4.  After the length‑sweep a final polishing (sub‑gradient + Adam)
    is performed on the overall best vector.

All heavy work uses FFT‑based convolutions (O(n log n)) and the
total wall‑clock time is limited to the supplied budget
(≤ 1000 s).  The function returns a non‑empty list of non‑negative
floats; the search guarantees a result even if no improvement is
found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    After projection onto the simplex Σa = 1 the denominator disappears,
    therefore the score is simply 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
    (Duchi et al., 2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def upsample_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length, then re‑project onto simplex."""
    old_len = len(seq)
    if old_len == new_len:
        return seq.copy()
    x_old = np.arange(old_len, dtype=float)
    x_new = np.linspace(0, old_len - 1, new_len)
    up = np.interp(x_new, x_old, seq)
    return project_to_simplex(up)

# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) where the sum constraint
    is handled later by projection onto the simplex.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)            # sum to 1

    # weighted shift of a: we need 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length n + (2n‑1) -1 = 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    lr0: float = 0.12,
    lam_start: float = 3.0,
    lam_end: float = 70.0,
    noise_every: int = 2500,
    noise_scale: float = 3e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """
    Sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t).  Projection onto the simplex.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # indices attaining the maximum (tolerance for floating error)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 120_000,
    init_temp: float = 0.02,
    decay: float = 0.9999,
) -> np.ndarray:
    """
    Simple Metropolis‑style local search, using the fast surrogate
    2·n·max(conv) for acceptance.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluate_sequence()
    value.  The routine works on a cascade of lengths (1024 → 2048 → 4096 …
    ) and combines random Beta seeds, FFT‑based sub‑gradient descent,
    Adam on a smooth surrogate and a brief simulated‑annealing polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a small safety margin before the hard deadline
    deadline = start_time + budget_s - 1.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Length cascade – start small, then up‑sample
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    # keep the best vector found so far (will be up‑sampled)
    current_best_vec = None

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} (budget left {deadline - time.time():.1f}s) ===")

        # ---------- generate seeds ----------
        seeds: List[np.ndarray] = []

        # deterministic baselines
        seeds.append(arcsine_initial(n))
        seeds.append(np.full(n, 1.0 / n))

        # up‑sample previous best if we have one
        if current_best_vec is not None:
            seeds.append(upsample_sequence(np.asarray(current_best_vec, dtype=float), n))

        # random Beta family (few of them – they are cheap)
        for _ in range(30):
            alpha = rng.uniform(0.05, 2.5)
            beta = rng.uniform(0.05, 2.5)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            seeds.append(generate_beta_shape(n, alpha, beta, eps))

        # power‑law (generalised arcsine) shapes
        for _ in range(30):
            p = rng.uniform(0.3, 5.0)       # heavier edge bias for larger p
            eps = rng.uniform(0.0, 1.0)
            seeds.append(generate_power_shape(n, p, eps))

        # tiny mixture of arcsine and uniform
        for _ in range(10):
            w = rng.random()
            arc = arcsine_initial(n)
            uni = np.full(n, 1.0 / n)
            mix = w * arc + (1.0 - w) * uni
            seeds.append(project_to_simplex(mix))

        # project all seeds onto the simplex (guard against numerical drift)
        seeds = [project_to_simplex(s) for s in seeds]

        # ---------- quick evaluation & keep top seeds ----------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:4]]   # keep a few promising ones

        # ---------- optimisation of each seed ----------
        for seed_vec in top_seeds:
            if time.time() >= deadline - 3.0:
                break

            # Phase 1 – exact sub‑gradient descent (fast, robust)
            remaining = deadline - time.time()
            max_iter_sub = int(min(40_000, remaining * 30))
            print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
            refined = subgradient_descent_exact(seed_vec, deadline, max_iter=max_iter_sub, step0=0.04)

            # Phase 2 – Adam on smooth surrogate
            if time.time() < deadline - 2.0:
                remaining = deadline - time.time()
                max_iter_adam = int(min(40_000, remaining * 30))
                print(f"  Adam smooth optimisation (max_iter={max_iter_adam}) …")
                refined, _ = adam_optimize_smooth(
                    refined,
                    deadline,
                    max_iter=max_iter_adam,
                    lr0=0.12,
                    lam_start=3.0,
                    lam_end=70.0,
                    noise_every=2500,
                    noise_scale=2e-5,
                )

            # Phase 3 – very light simulated‑annealing polish
            if time.time() < deadline - 1.0:
                remaining = deadline - time.time()
                max_iter_sa = int(min(20_000, remaining * 30))
                print(f"  Simulated annealing (max_iter={max_iter_sa}) …")
                refined = simulated_annealing(refined, rng, deadline, max_iters=max_iter_sa)

            # ----- Official evaluation -----
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"  → official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    current_best_vec = refined.copy()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception as e:
                print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Global polishing of the best vector (if we have one)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter_total = int(min(200_000, remaining * 30))

        # split remaining budget between sub‑gradient and Adam
        it_sub = max_iter_total // 2
        it_adam = max_iter_total - it_sub

        print("\nFinal global polishing (sub‑gradient + Adam) …")
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.02)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=it_adam,
            lr0=0.09,
            lam_start=3.0,
            lam_end=70.0,
            noise_every=2000,
            noise_scale=1e-5,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 1024
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
