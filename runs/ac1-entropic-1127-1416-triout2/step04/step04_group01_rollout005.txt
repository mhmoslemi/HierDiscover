"""
Propose a step‑function sequence with a low evaluate_sequence score.

The algorithm consists of three stages:

1. **Broad exploration** – for several lengths (256 … 8192) a large
   pool of candidate shapes is generated.  Families include:
   * arcsine / beta‑family (asymmetric peaks at the ends),
   * monotone power‑law and exponential decays,
   * mixtures with the uniform vector,
   * sparse random vectors,
   * simple triangular bumps.
   All candidates are normalised to Σa = 1 and ranked by the
   fast surrogate 2·n·max(conv(a,a)), which equals the official
   objective for sum‑one vectors.

2. **Local refinement** – the best few seeds from each length are
   processed by a projected sub‑gradient descent on the exact
   objective (max‑convolution) followed by a short simulated‑annealing
   polishing phase.  Both phases respect the simplex constraint
   (non‑negative entries, Σa = 1).

3. **Global polishing** – the overall best vector is further refined
   with a longer sub‑gradient run and a final annealing pass.

All work respects the 1000 s time budget; time checks are performed
after every major loop.  The returned list contains non‑negative
floats and is guaranteed to be non‑empty.
"""

import time
import numpy as np
from typing import List, Tuple


# ----------------------------------------------------------------------
# Low‑level helpers (FFT convolution, projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a and b using real FFT.
    Length of the result is len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is already normalised to Σa = 1,
    therefore the denominator of evaluate_sequence disappears.
    """
    n = a.size
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (Duchi et al.)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_exponential_shape(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-gamma * i)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_monotone_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Monotone power‑law a_i ∝ (i+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    vals = (i + eps) ** (-p)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at `peak`."""
    left = np.arange(peak + 1)
    right = np.arange(n - peak - 1)[::-1]
    seq = np.concatenate([left, right]).astype(float)
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_sparse_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a few random locations and assign random positive mass."""
    k = int(rng.integers(2, min(10, n // 2)))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vec = np.zeros(n, dtype=float)
    vec[idx] = vals
    vec /= vec.sum()
    return vec

# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The sub‑gradient is 2·∑_{k∈K} shift(a,k) where K are the
    indices attaining the current maximum.
    """
    a = project_to_simplex(init_a.copy())
    best = a.copy()
    best_sc = compute_score(a)
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # indices attaining the maximum (tolerance for floating errors)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # diminishing step size
        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best


def simulated_annealing(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 30_000,
    init_temp: float = 0.02,
    decay: float = 0.9999,
) -> np.ndarray:
    """
    Simple Metropolis‑style mass‑transfer annealing.
    A small fraction of the mass of a random coordinate is moved to another.
    Acceptance uses the exact fast surrogate 2·n·max(conv).
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # transfer up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

            refined = subgradient_descent_exact(seed_vec, deadline, max_iter=it_sub, step0=0.08)

            # Simulated annealing polish
            if time.time() < deadline - 2.0:
                remaining = deadline - time.time()
                it_sa = int(min(30_000, remaining * 30))
                refined = simulated_annealing(refined, rng, deadline,
                                             max_iters=it_sa,
                                             init_temp=0.02,
                                             decay=0.9999)

            # Evaluate (fast surrogate == official for sum = 1)
            val = compute_score(refined)
            if val < best_score:
                best_score = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_score:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 3.  Global polishing of the best vector found so far
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing …")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        it_sub = int(min(60_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.04)

        remaining = deadline - time.time()
        it_sa = int(min(70_000, remaining * 30))
        a = simulated_annealing(a, rng, deadline,
                                max_iters=it_sa,
                                init_temp=0.015,
                                decay=0.99995)

        final_val = compute_score(a)
        if final_val < best_score:
            best_score = final_val
            best_seq = a.tolist()
            print(f"*** Updated global best after final polish: {best_score:.6f} ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_score = compute_score(np.full(256, 1.0 / 256))

    print(f"\nFinished search. Best reported surrogate score = {best_score:.6f}")
    return best_seq
