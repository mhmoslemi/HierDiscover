def project_simplex(v):
    """Euclidean projection of v onto simplex {x >=0, sum=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Heuristic search for a low‑max self‑convolution step function.

The algorithm works in three stages:

1. **Broad parameter sweep** – generate many candidate shapes on the

3. **Final polishing** – the best vector found so far is refined once
   more with a longer run of the same Adam optimiser.

All work respects the supplied time budget (default 1000 s) and always
returns a feasible non‑negative list of floats.

from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of real vectors a and b via FFT.
    The output length is len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        score = 2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}.
    Implementation follows the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Families producing initial candidates (all vectors are normalised)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) with a tiny epsilon."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with parameters (a,a)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    base = (i + 0.5 + eps) ** (a - 1) * (n - 1 - i + 0.5 + eps) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Adam‑style smooth optimisation on the simplex
# ----------------------------------------------------------------------


def refine_adam(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 50_000,
    gamma_start: float = 8.0,
    gamma_end: float = 200.0,
) -> Tuple[np.ndarray, float]:
    """
    Refine vector a0 by minimising a smoothed version of
    ``max(conv(a,a))`` using an Adam optimiser.
    The temperature (soft‑max parameter) is increased linearly from
    ``gamma_start`` to ``gamma_end`` (continuation method).
    Returns the best vector seen and its exact score.
    """
    a = normalize(a0.astype(float).copy())
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    for t in range(1, max_iter + 1):
        # time check
        if time.time() >= deadline:
            break

        # linearly increase gamma
        gamma = gamma_start + (gamma_end - gamma_start) * (t / max_iter)

        conv = convolution_fft(a, a)                     # length 2n‑1
        cmax = np.max(conv)
        # soft‑max weights (stable)
        w_exp = np.exp(gamma * (conv - cmax))
        w = w_exp / np.sum(w_exp)

        # gradient of the smoothed max
        grad = gradient_from_weights(a, w)

        # Adam step size schedule (decreasing)
        lr = 0.07 / np.sqrt(t)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)

        # keep track of the best exact score every few iterations
        if t % 3000 == 0:
            val = evaluate_sequence(a.tolist())
            if val < best_score:
                best_score = val
                best = a.copy()
                print(f"    [refine] iter {t:,d} new best {best_score:.6f}")

    # final evaluation
    final_val = evaluate_sequence(a.tolist())
    if final_val < best_score:
        best_score = final_val
        best = a.copy()
        print(f"    [refine] final improvement {best_score:.6f}")

    return best, best_score

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    score.  The routine respects the time budget, explores several
    parametric families, refines the most promising candidates with a
    smooth Adam optimiser and finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a small safety margin before the hard timeout
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore.  2048 gave the previous best; we also try a
    # smaller (1024) and a larger (4096) length when time permits.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n>>> Exploring length n = {n}")

        # ------------------------------
        # 1. Build a library of candidates
        # ------------------------------
        candidates: List[np.ndarray] = []

        # deterministic families
        candidates.append(arcsine_initial(n))
        candidates.append(uniform_shape(n))

        # symmetric beta (log‑spaced a)
        for a in np.geomspace(0.05, 5.0, 18):
            candidates.append(beta_shape(n, a))

        # power‑law grid
        p_vals = np.linspace(0.20, 0.80, 13)
        eps_vals = np.linspace(0.0, 0.07 * n, 9)   # up to ≈7 % of n
        for p in p_vals:
            for eps in eps_vals:
                candidates.append(powerlaw_shape(n, p, eps))

        # arcsine + uniform mixtures
        for w in np.linspace(0.0, 0.5, 9):
            mix = (1.0 - w) * arcsine_initial(n) + w * uniform_shape(n)
            candidates.append(normalize(mix))

        # random Dirichlet draws
        for _ in range(25):
            candidates.append(random_dirichlet(n, rng))

        # ------------------------------
        # 2. Quick fast scoring, keep the best few
        # ------------------------------
        scored: List[Tuple[float, np.ndarray]] = []
        for vec in candidates:
            sc = compute_score(vec)
            scored.append((sc, vec))
        scored.sort(key=lambda x: x[0])
        top_k = 7
        top_candidates = [c for _, c in scored[:top_k]]
        print("  top initial scores:",
              " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]))

        # ------------------------------
        # 3. Refine each retained candidate
        # ------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            cur_sc = compute_score(cand)
            print(f"  refinement {idx + 1}/{len(top_candidates)} start_score = {cur_sc:.6f}")

            # allocate iterations according to remaining time
            secs_left = max(0.0, deadline - time.time())
            # empirically ~3000 iterations per second for n≈2000
            max_iter = int(min(80_000, max(5_000, secs_left * 3000)))

            refined, refined_val = refine_adam(
                a0=cand,
                deadline=deadline,
                max_iter=max_iter,
                gamma_start=8.0,
                gamma_end=200.0,
            )
            # final official evaluation
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 4. Final polishing of the global best (time permitting)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)
        secs_left = max(0.0, deadline - time.time())
        max_iter = int(min(150_000, max(20_000, secs_left * 3000)))
        polished, _ = refine_adam(
            a0=a,
            deadline=deadline,
            max_iter=max_iter,
            gamma_start=30.0,
            gamma_end=300.0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # safe fallback: uniform vector of modest length
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
