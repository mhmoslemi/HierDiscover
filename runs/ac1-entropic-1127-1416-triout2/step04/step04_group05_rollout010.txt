"""
Propose a step‑height sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from the previously known ``height_sequence_1``
   (if available) or from a uniform vector.  The vector is normalised to a
   probability distribution (sum = 1) and represented by log‑weights
   ``x`` (soft‑max parametrisation).

2. **Smooth‑max Adam optimisation** – minimise a smooth approximation of
   the ∞‑norm of the self‑convolution while staying on the simplex.
   The smooth max is the log‑sum‑exp of the convolution entries with a
   temperature ``α`` that is slowly increased.  Gradients are obtained
   analytically (correlation of the soft‑max weights with the current
   vector) and an Adam update is performed on the log‑weights.  The best
   exact (non‑smooth) score seen during this phase is stored.

3. **Guided simulated‑annealing refinement** – starting from the best
   point of phase 2, perform a classic Metropolis walk that moves a random
   fraction of mass from an index with large gradient to an index with
   small gradient.  A geometric cooling schedule is used and the current
   best solution is kept.

The whole routine respects the 1000 s time budget and always returns a
non‑negative sequence (the normalised probability vector).  All
operations are performed with FFTs, giving an O(n log n) cost per
iteration (n≈5 000 in the default setting).
"""

import time
import math

# Helper utilities

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_len(n: int) -> int:
    """
    Length of the zero‑padded FFT used for both the convolution
    (size 2 n‑1) and the correlation needed for the gradient
    (size 3 n‑2).  The same length is used for both, chosen as the next
    power of two for speed.
    """
    return _next_pow2(3 * n - 2)


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself, using an FFT of length ``L``.
    Returns the first 2 n‑1 entries (the true linear convolution).
    """
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact evaluation (sum of a is 1) from a convolution vector."""
    return 2.0 * n * conv.max()


def _smooth_max_and_grad(a: np.ndarray,
                         fft_a: np.ndarray,
                         conv: np.ndarray,
                         alpha: float,
                         L: int,
                         n: int) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the max‑convolution and its gradient w.r.t.
    the probability vector ``a``.

    Returns
    -------
    smooth_obj : float
        2 n · ( max(conv) + log‑sum‑exp(alpha·(conv‑max))/α )
    grad_a : np.ndarray
        ∂smooth_obj / ∂a   (shape (n,))
    """
    # --- smooth max (log‑sum‑exp) ------------------------------------
    max_c = conv.max()
    # shift to avoid overflow – all deltas ≤ 0
    delta = conv - max_c
    exp_delta = np.exp(alpha * delta)          # safe: max entry is 1.0
    sum_exp = exp_delta.sum()
    # smooth objective value
    smooth_obj = 2.0 * n * (max_c + np.log(sum_exp) / alpha)

    # --- gradient w.r.t. a -----------------------------------------
    # weights for the linear combination of convolution entries
    w = exp_delta / sum_exp                     # shape (2n‑1,)

    # correlation w * a (linear, not circular)
    fft_w = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(fft_w * np.conj(fft_a), n=L)

    start = n - 1                               # slice giving indices 0…n‑1
    corr = corr_full[start:start + n]           # length n
    # grad_a = 4·n·corr   (see derivation in analysis)
    grad_a = 4.0 * n * corr
    return smooth_obj, grad_a


def _subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max‑convolution.  The gradient is the average
    of the exact gradients of all active entries (those attaining the
    current maximum up to a tiny tolerance).
    """
    n = a.shape[0]
    max_val = conv.max()
    eps = max_val * 1e-8
    active = np.where(conv >= max_val - eps)[0]

    if active.size == 0:
        return 2.0 * a

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to one).  Implementation of
    Duchi‑Shalev‑Shwartz‑Singer‑Chandra (2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    # --------------------------------------------------------------
    best_a /= best_a.sum()
    return [float(v) for v in best_a.tolist()]
