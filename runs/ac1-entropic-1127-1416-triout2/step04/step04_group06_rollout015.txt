Algorithm – aggressive smoothing + max‑subgradient mirror descent
-----------------------------------------------------------------
1.  Load the previously found sequence (if available) and keep it as the
    current best solution.

2.  Repeatedly restart from a variety of simple distributions

    restart is occasionally blended with the current best vector to keep

    smooth approximation

    of the sup‑norm of the self‑convolution.
    – The smoothing parameter α is increased exponentially from 1 to 2000,

4.  After the descent a short simulated‑annealing (mass‑transfer) phase

5.  The best vector seen over the whole budget is returned (already

def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector that already satisfies
    sum(a) == 1.  Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow/underflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_random_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_random_beta(n: int) -> np.ndarray:
    """Draw a Beta‑shaped density on the simplex."""
    a = np.random.uniform(0.2, 5.0)
    b = np.random.uniform(0.2, 5.0)
    return beta_sequence(a, b, n)


def init_random_edge_concentrated(n: int) -> np.ndarray:

    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = dist_to_center.astype(float) + 1.0

            a = init_random_exponential(n)
        else:
            a = init_random_powerlaw(n)

        # Occasionally blend with the global best (helps preserve good mass)
        if np.random.rand() < 0.12:
            a = 0.5 * a + 0.5 * best_a
            a = simplex_normalise(a)

        # ----- mirror‑descent on the smoothed max of the self‑convolution -----
        max_iters = 3500
        eta0 = 1.2               # base step size for exponentiated‑gradient
        alpha_start = 1.0
        alpha_end = 2000.0       # final smoothing – almost the true max
        for it in range(max_iters):
            if time.time() > deadline - 4.0:
                break

            # 1) self‑convolution (FFT, length 2n‑1)
            fft_a = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

            # 2) soft‑max weights over the convolution entries
            maxc = conv.max()
            # exponential stability trick (subtract max)
            exp_vals = np.exp(alpha_start * (conv - maxc))
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                p = np.full_like(conv, 1.0 / conv.size)
            else:
                p = exp_vals / sum_exp

            # 3) exact gradient of f_α (2·corr(a,p))
            fft_a_full = np.fft.rfft(a, n=L_corr)
            fft_p_full = np.fft.rfft(p, n=L_corr)
            corr = np.fft.irfft(fft_a_full * np.conj(fft_p_full), n=L_corr)
            offset = len(p) - 1                     # lag 0 appears at this index
            g = 2.0 * corr[offset:offset + n]       # gradient on the simplex

            # 4) exponentiated‑gradient update (mirror descent on the simplex)
            eta = eta0 / math.sqrt(it + 1.0)
            a = a * np.exp(-eta * g)
            a = simplex_normalise(a)

            # 5) increase smoothing parameter α (exponential schedule)
            alpha_start = alpha_start * (alpha_end / alpha_start) ** (1.0 / max_iters)

            # 6) occasional logging and best‑solution update
            if (it & 31) == 0:          # every 32 iterations
                cur_score = eval_fast(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Restart {restart_counter}] it {it:4d} → new best {best_score:.6f}")

        # ----- short simulated‑annealing refinement on the current best -----
        cur_a = best_a.copy()
        cur_score = eval_fast(cur_a)
        T = cur_score * 0.02          # start temperature (relative)
        T_end = cur_score * 1e-5
        sa_iters = 0
        while time.time() < deadline - 2.0 and T > T_end:
            # simple mass‑transfer move (keeps sum exactly 1)
            src = np.random.choice(n, p=cur_a)       # pick source proportional to its mass
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1
            if cur_a[src] <= 0.0:
                continue
            # transfer at most 10 % of the source mass (helps stability)
            delta = cur_a[src] * np.random.rand() * 0.1
            cand = cur_a.copy()
            cand[src] -= delta
            cand[tgt] += delta

            cand_score = eval_fast(cand)
            if cand_score < cur_score:
                accept = True
            else:
                accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

            if accept:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[SA  R{restart_counter}] iter {sa_iters:5d} → new best {best_score:.6f}")

            # geometric cooling
            T *= 0.9995
            sa_iters += 1

        restart_counter += 1

    # ------------------------------------------------------------------
    # 3. Return the best vector (already normalised to sum = 1)
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {restart_counter} restarts: {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
