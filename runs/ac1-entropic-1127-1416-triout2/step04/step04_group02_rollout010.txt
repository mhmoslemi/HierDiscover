def smooth_objective_and_grad(a, beta):
    # a: probability vector (sum=1)
    # Compute conv=a*a.
    conv = convolution_fft(a, a)
    # compute exponentiated weights:
    # to avoid overflow, subtract max before exponentiate:
    max_c = conv.max()
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    # compute gradient via convolution of w and reversed a
    a_rev = a[::-1]
    grad_full = convolution_fft(w, a_rev)
    # gradient_i = 2 * grad_full[n-1 + i]
    grad = 2.0 * grad_full[len(a)-1: len(a)-1 + len(a)]
    # Return smooth approximation of max: (max_c + log sum exp(beta*(conv-max_c))/beta)
    # Actually smooth max = max_c + log(sum exp(beta*(conv-max_c))) / beta

"""
Propose a low‑max‑convolution step function.

The algorithm proceeds in three stages:

1. **Candidate generation** – for a few sequence lengths (1024, 2048, 4096)
   we create a diverse pool of starting vectors (arcsine, Dirichlet,
   power‑law, uniform, perturbed arcsine).  The cheap spectral score
   ``compute_score`` is used to keep the most promising few.

2. **Smooth‑max gradient descent** – each retained start is refined by
   minimizing a smooth approximation of the max‑convolution
   ``Φ(a)=max_k (a∗a)_k``.  For a temperature β the smooth objective is

          φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

   whose gradient can be computed via two FFT‑based convolutions.
   We perform projected gradient descent (with an explicit Euclidean
   projection onto the probability simplex) while slowly raising β,
   thus turning the smooth objective into the true max.  An
   Adam‑style step‑size schedule (lr ∝ 1/√t) is used; the algorithm
   stops when the allotted time expires.

3. **Final polishing** – the best sequence found overall is run again
   with a higher final β and a smaller learning rate to obtain the
   final answer.

All numbers are kept non‑negative, normalized to sum to one (the
objective is scale‑invariant), and the routine always returns a feasible
list before the 1000 s budget expires.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official (scale‑invariant) score:
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape factories (used to generate diverse initial vectors)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float,
                     rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like draw (Gamma or heavy‑tailed exponential)."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add tiny Gaussian noise and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smooth‑max optimisation (projected gradient descent)
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, beta: float) -> Tuple[float, np.ndarray]:
    """
    Returns the smoothed max value φ_β(a) and its gradient.
    Gradient = 2·(w * a_rev)_{n-1 : n-1+n}, where w are soft‑max weights.
    """
    # convolution a∗a
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    shifted = conv - max_c
    exp_beta = np.exp(beta * shifted)
    w = exp_beta / exp_beta.sum()                    # still length 2n‑1

    # gradient via convolution w * reversed a
    a_rev = a[::-1]
    grad_full = convolution_fft(w, a_rev)           # length 3n‑2
    n = len(a)
    grad = 2.0 * grad_full[n - 1: n - 1 + n]

    # smooth max value (log‑sum‑exp)
    smooth_val = max_c + np.log(exp_beta.sum()) / beta
    return smooth_val, grad


def optimise_smooth(initial_a: np.ndarray,
                    deadline: float,
                    *,
                    beta_start: float = 0.1,
                    beta_end: float = 300.0,
                    lr_start: float = 0.2,
                    max_iter: int = 200_000) -> Tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth approximation of the max‑convolution.
    Returns the best vector encountered and its *official* score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()                     # work on the probability simplex
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())

    t = 0
    beta = beta_start
    lr = lr_start

    while time.time() < deadline and t < max_iter:
        t += 1

        # compute smooth objective and gradient
        _, grad = _smooth_max_and_grad(a, beta)

        # gradient step and projection
        a = a - lr * grad
        a = simplex_projection(a)

        # learning‑rate decay (Adam‑style could be used, this simple schedule works well)
        lr = lr_start / np.sqrt(t + 1.0)

        # gradually increase β (geometric schedule)
        if beta < beta_end:
            beta = min(beta_end, beta * 1.0015)

        # occasional evaluation of the real objective
        if t % 500 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"    smooth iter={t:,d}  β={beta:.2f}  best={best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    value.  The routine respects the given time budget (default 1000 s) and
    always returns a feasible list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # a tiny safety margin for the final call(s)
    deadline = start + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Explore a few lengths (powers of two are FFT‑friendly)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Length n = {n} ===")

        # ---- generate a rich library of initial vectors -----------------
        candidates: List[np.ndarray] = []

        # (i) arcsine (very strong baseline)
        candidates.append(arcsine_initial(n))

        # (ii) uniform
        candidates.append(np.full(n, 1.0 / n))

        # (iii) Dirichlet / exponential draws (various concentrations)
        for conc in [0.3, 0.7, 1.5, 3.0]:
            candidates.append(random_dirichlet(n, conc, rng))

        # (iv) random power‑law shapes
        for _ in range(4):
            p = rng.uniform(0.1, 1.2)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            candidates.append(generate_powerlaw(n, p, eps))

        # (v) small perturbations of the arcsine shape
        for _ in range(3):
            candidates.append(perturb(arcsine_initial(n), rng, scale=0.03))

        # ---- cheap internal ranking, keep only the most promising --------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]   # keep three

        # ---- optimise each retained candidate ---------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # allocate a slice of the remaining budget – aim for a few‑seconds
            # of effective optimisation per candidate
            cand_deadline = min(deadline,
                                time.time() + max(1.0, (deadline - time.time()) * 0.25))
            refined, _ = optimise_smooth(cand,
                                         cand_deadline,
                                         beta_start=0.1,
                                         beta_end=500.0,
                                         lr_start=0.25,
                                         max_iter=500_000)

            # final evaluation with the official evaluator
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_smooth(a,
                                       deadline,
                                       beta_start=0.5,
                                       beta_end=800.0,
                                       lr_start=0.05,
                                       max_iter=400_000)
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
