def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2

    best_seq = None
    best_val = float('inf')

    # define evaluation only compute_score (fast) and evaluate_sequence (official)
    
    # Choose lengths to try - powers of two up to maybe 8192.
    lengths = [1024, 2048, 4096, 8192]  # but only those fitting in time.
    # We'll adapt: start with smaller n and if time left, go larger.

    for n in lengths:
        if time.time() >= deadline - 5:
            break
        # Parameter families:
        # 1) Beta-shaped (α,β)
        # 2) Power-law shape (p)
        # 3) Symmetric polynomial shape (p)
        # 4) Mixed: combination of arcsine and uniform (weighted convex combination)
        # We'll produce vector a = normalized shape (sum=1) for each param.
        candidates = []
        # uniform
        candidates.append(np.full(n, 1/n))

        # arcsine
        candidates.append(arcsine_initial(n))

        # random draws of beta parameters
        for _ in range(30):
            alpha = rng.uniform(0.1, 5.0)
            beta = rng.uniform(0.1, 5.0)
            cand = generate_beta_like(n, alpha, beta)
            candidates.append(cand)

        # random power law exponents
        for _ in range(20):
            p = rng.uniform(0.1, 2.0)
            eps = rng.uniform(0.001, 1.0)
            cand = generate_powerlaw(n, p, eps)
            candidates.append(cand)

        # random polynomial: a_i ∝ (i+0.5)*(n-0.5-i) ^ p (peak at centre)
        for _ in range(20):
            p = rng.uniform(-0.5, 2.0)  # negative would give spike at ends
            i = np.arange(n, dtype=float)
            base = ((i + 0.5) * (n - 0.5 - i)) ** p
            base = np.maximum(base, 0)
            cand = base / base.sum() if base.sum()>0 else np.full(n, 1/n)
            candidates.append(cand)

        # Dirichlet draws
        for conc in [0.3, 0.8, 2.0]:
            for _ in range(5):
                cand = random_dirichlet(n, conc, rng)
                candidates.append(cand)

        # maybe mixing arcsine and uniform
        for _ in range(10):
            w = rng.uniform(0,1)
            cand = w * arcsine_initial(n) + (1-w) * np.full(n, 1/n)
            cand = cand / cand.sum()
            candidates.append(cand)

        # compute scores
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top = [c for _,c in scored[:6]]

        # refine each with subgradient descent
        for idx,cand in enumerate(top):
            if time.time() >= deadline - 5:
                break
            # Determine iteration budget based on remaining time and n
            rem = deadline - time.time()
            # allocate proportionally: maybe 30% time for refinement overall
            max_iter = int(min(200_000, max(5_000, rem * 150 / (n/1024))))
            # run subgradient with custom schedule
            refined, refined_sc = optimise_subgradient(
                initial_a=cand,
                deadline=deadline,
                eta0=0.9,
                max_iter=max_iter,
                noise_every=300,
                noise_scale=1e-5,
            )
            val = evaluate_sequence(refined.tolist())
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"New best after refinement n={n} idx={idx}: {val:.6f}")

    # final polishing if time remains
    if best_seq and time.time() < deadline - 2:
        a = np.array(best_seq)
        remain = deadline - time.time()
        max_iter = int(min(300_000, max(10_000, remain * 200 / (len(a)/1024))))
        a_ref, sc = optimise_subgradient(
            initial_a=a,
            deadline=deadline,
            eta0=0.4,
            max_iter=max_iter,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_ref.tolist())
        if final_val < best_val:
            best_val = final_val
            best_seq = a_ref.tolist()
    if not best_seq:
        best_seq = [1.0]*256

    return best_seq

def optimise_smoothed(initial_a, deadline, *, eta0=0.5, gamma0=5.0, gamma_factor=1.001, max_iter=200000, noise_every=0, noise_scale=0.0):
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # compute conv
        conv = convolution_fft(a, a)
        # compute gradient with smoothing
        gamma = gamma0 * (gamma_factor ** (it-1))
        # softmax weights
        maxc = conv.max()
        # compute stable exponentials
        exp_term = np.exp(gamma * (conv - maxc))
        w = exp_term / exp_term.sum()
        # compute gradient = 2 * conv(w, a_rev) slice
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)
        start = n-1
        grad = 2.0 * grad_full[start:start+n]
        # update
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)
        # projection onto simplex
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        # occasionally print status
        if it % 5000 == 0:
            print(f"    smoothed iter={it} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc
```

    arcsine and mixtures) over several sequence lengths.  The cheap
    `compute_score` (which evaluates the official objective on a vector whose
    sum equals 1) is used to keep only the most promising candidates.

2.  A *smoothed* mirror‑descent optimizer.  The non‑smooth objective
    ``max_k (a*a)_k`` is approximated by a soft‑max
    ``g_γ(a)= 1/γ log Σ exp(γ·conv_k)``.
    The gradient of this smooth proxy can be written as a convolution,
    allowing an O(n log n) update per iteration.  The update is
    multiplicative (exponentiated‑gradient) and the iterates are projected
    onto the probability simplex, guaranteeing non‑negativity.

    The smoothing parameter ``γ`` is increased slowly during optimisation,
    so the method first follows a smooth landscape and later concentrates on
    the true maximum.

3.  A final polishing phase (again with the smoothed optimiser but with a
    smaller step‑size) is performed on the globally best sequence before
    returning it.

All work is timed against the supplied budget (default 1000 s); the routine
always returns a feasible list of floats before the deadline.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    For vectors normalised to sum = 1 this reduces to 2·n·max(conv).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        # heavy‑tailed “sparse” draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        # standard Gamma draws
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny random perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smoothed mirror‑descent optimiser (max‑convolution proxy)
# ----------------------------------------------------------------------


def optimise_smoothed(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.8,
    gamma0: float = 4.0,
    gamma_factor: float = 1.001,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Minimise  max_k (a*a)_k  via a smoothed soft‑max proxy.
    The iterate stays on the simplex (sum = 1) using exponentiated gradients.
    Returns the best vector encountered and its (true) score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()                         # probability simplex
    n = len(a)
    best_a = a.copy()
    # because sum = 1, the true score is simply 2*n*max(conv)
    cur_conv = convolution_fft(a, a)
    best_sc = 2.0 * n * np.max(cur_conv)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution (full length 2n‑1)
        conv = convolution_fft(a, a)

        # soft‑max weights
        gamma = gamma0 * (gamma_factor ** (it - 1))
        maxc = conv.max()
        # stabilize the exponentials
        exp_term = np.exp(gamma * (conv - maxc))
        w = exp_term / exp_term.sum()          # length 2n‑1, sums to 1

        # gradient = 2 * (w * a_rev) sliced to length n
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)   # length 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # step‑size (standard 1/√t schedule)
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # optional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # re‑project onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # true score (no extra convolution needed – we already have conv)
        cur_sc = 2.0 * n * np.max(conv)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    smoothed iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine explores several sequence lengths, samples a rich
    collection of parametric families, retains the best few candidates and
    refines them with a smoothed mirror‑descent optimiser.  A final polishing
    run is performed on the overall best sequence before returning.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a two‑second safety margin for the final official evaluation
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to explore – powers of two give cheap FFTs.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]   # 8192 would be too expensive in the time budget

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a large yet cheap library of initial shapes.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # (i) Uniform
        candidates.append(np.full(n, 1.0 / n))

        # (ii) Classic arcsine
        candidates.append(arcsine_initial(n))

        # (iii) Random beta‑like shapes (log‑uniform α,β)
        for _ in range(45):
            loga = rng.uniform(-1.5, 1.2)   # α in roughly [0.03, 16]
            logb = rng.uniform(-1.5, 1.2)   # β in same range
            alpha = 10 ** loga
            beta = 10 ** logb
            candidates.append(generate_beta_like(n, alpha, beta))

        # (iv) Random power‑law shapes
        for _ in range(30):
            p = rng.uniform(0.05, 2.0)
            eps = rng.uniform(0.001, max(1.0, 0.03 * n))
            candidates.append(generate_powerlaw(n, p, eps))

        # (v) Symmetric polynomial shapes (i·(n‑1‑i))^p
        for _ in range(30):
            p = rng.uniform(-0.5, 2.5)
            i = np.arange(n, dtype=float)
            base = ((i + 0.5) * (n - 0.5 - i)) ** p
            base = np.maximum(base, 0.0)
            cand = base / base.sum() if base.sum() > 0 else np.full(n, 1.0 / n)
            candidates.append(cand)

        # (vi) Dirichlet / exponential draws (dense & sparse)
        for conc in [0.3, 0.7, 1.5, 3.0]:
            for _ in range(3):
                candidates.append(random_dirichlet(n, conc, rng))

        # (vii) Small random perturbations of arcsine (helps escape local minima)
        for _ in range(8):
            candidates.append(perturb(arcsine_initial(n), rng, scale=0.015))

        # --------------------------------------------------------------
        # 1b. Quick sweep using the cheap ``compute_score``; keep the best.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:6]]   # six most promising

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate with the smoothed optimiser.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # Allocate a slice of the remaining budget.
            remaining = max(0.0, deadline - time.time())
            # Rough estimate: ~300 iterations per second for n≈2000,
            # scaled linearly with n.
            est_iter_per_sec = 300.0 * (2048.0 / n)
            max_iter = int(min(180_000, max(5_000, remaining * est_iter_per_sec)))

            refined, _ = optimise_smoothed(
                initial_a=cand,
                deadline=deadline,
                eta0=0.9,
                gamma0=5.0,
                gamma_factor=1.001,
                max_iter=max_iter,
                noise_every=400,
                noise_scale=5e-6,
            )

            # Official evaluation (scale‑invariant, so we may keep the vector as‑is)
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        est_iter_per_sec = 300.0 * (2048.0 / len(a))
        max_iter = int(min(300_000, max(15_000, remaining * est_iter_per_sec)))

        print("\nFinal polishing of the global best")
        a_ref, _ = optimise_smoothed(
            initial_a=a,
            deadline=deadline,
            eta0=0.4,                # finer steps
            gamma0=20.0,
            gamma_factor=1.0015,
            max_iter=max_iter,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_ref.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_ref.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
