def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    After projection onto the simplex Σa = 1 the denominator disappears,
    therefore the score is simply 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Shape generators (all return a normalised probability vector)
# ------------------------------------------------------------

def arcsine_initial(n: int) -> np.ndarray:
    """Classic symmetric arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_exponential_shape(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) (monotonically decreasing)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-rate * i)
    return vals / vals.sum()


def generate_power_law_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=float)
    vals = i ** (-p)
    return vals / vals.sum()


def generate_two_step_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Simple two‑level step: a fraction of the entries get a high value,
    the remainder get a low value.  The split point and the ratio are
    sampled at random.
    """
    split = rng.integers(1, n)            # number of “high” entries
    high = rng.uniform(0.5, 5.0)           # relative magnitude of the high part
    low = rng.uniform(0.0, high)          # low part (can be zero)
    high_vec = np.full(split, high, dtype=float)
    low_vec = np.full(n - split, low, dtype=float)
    seq = np.concatenate([high_vec, low_vec])
    rng.shuffle(seq)                      # random order of the two levels
    return seq / seq.sum()

# Optimisers
# ------------------------------------------------------------

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) (the sum‑constraint is handled later).
    """
    n = len(a)
    conv = convolution_fft(a, a)                 # length L = 2·n‑1
    max_c = np.max(conv)

    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)              # soft‑max weights

    # weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)         # length n+L‑1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate (soft‑max of the convolution).
    Returns the best vector (projected onto the simplex) together with its
    fast surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the true objective
    f(a)=2·n·max(conv(a,a))  (the sum of a is kept equal to 1).
    Step size η_t = step0/√t.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(conv.max())

        # mask of indices attaining the maximum (numerical tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Metropolis‑style local search.  Small amounts of mass are moved from
    one coordinate to another while preserving the ℓ₁‑norm.
    The fast surrogate 2·n·max(conv) is used for the acceptance test.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta          # sum is preserved, non‑negativity kept

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

    return best


def lp_improve(a: np.ndarray,
               M: float,
               deadline: float,
               max_iter: int = 5,
               t: float = 0.2) -> np.ndarray:
    """
    One iteration of the LP‑based improvement described in the statement.
    It solves:
        max   ∑ b_j
        s.t.  b ≥ 0,
              conv(a, b) ≤ M.
    The resulting b (if ℓ₁‑norm larger than 1) is mixed back into a.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        # SciPy not present – simply skip the LP step.
        return a

    n = len(a)
    L = 2 * n - 1

    # Build the convolution matrix (A_ub @ b ≤ M)
    # Row k corresponds to coefficients a_{k‑j} for b_j.
    A = np.zeros((L, n), dtype=float)
    for k in range(L):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # i runs over the overlapping a‑indices
        for i in range(lo, hi + 1):
            j = k - i
            A[k, j] = a[i]

    # Linear program: minimise -∑b  (i.e. maximise ∑b)
    c = -np.ones(n, dtype=float)
    b_ub = np.full(L, M, dtype=float)          # RHS of the inequalities
    bounds = [(0.0, None) for _ in range(n)]

    # We run only a few iterations (max_iter) – each call does one LP.
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        b_vec = res.x
        sum_b = b_vec.sum()
        if sum_b <= 1.0 + 1e-9:
            # No improvement in ℓ₁‑norm – stop.
            break
        g = b_vec / sum_b                     # normalise to sum = 1
        a = (1.0 - t) * a + t * g
        a = project_to_simplex(a)             # re‑project to avoid drift

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluate_sequence()
    value.  The routine combines diverse random sampling, sub‑gradient /
    Adam optimisation, a linear‑program improvement step and simulated
    annealing.  The best vector found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        pool: List[Tuple[float, np.ndarray]] = []

        # ---- deterministic starters -------------------------------------------------
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for st in starters:
            pool.append((compute_score(st), st.copy()))

        # ---- random families --------------------------------------------------------
        # beta family (asymmetric)
        for _ in range(180):
            alpha = rng.uniform(0.05, 5.0)
            beta = rng.uniform(0.05, 5.0)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)
            pool.append((compute_score(cand), cand))

        # inverse‑polynomial (generalised arcsine)
        for _ in range(180):
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # mixture arcsine / uniform
        for _ in range(80):
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # simple triangular bumps
        for _ in range(80):
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

        # exponential decay shapes
        for _ in range(80):
            rate = rng.uniform(0.0005, 0.03)  # slower decay = flatter shape
            cand = generate_exponential_shape(n, rate)
            pool.append((compute_score(cand), cand))

        # power‑law decay shapes
        for _ in range(80):
            p = rng.uniform(0.4, 3.0)
            cand = generate_power_law_shape(n, p)
            pool.append((compute_score(cand), cand))

        # two‑step (high/low) shapes
        for _ in range(80):
            cand = generate_two_step_shape(n, rng)
            pool.append((compute_score(cand), cand))

        # random permutation of a few good bases (adds diversity)
        for _ in range(40):
            base = arcsine_initial(n)
            cand = base[rng.permutation(n)]
            pool.append((compute_score(cand), cand))

        # keep the most promising seeds
        pool.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in pool[:6]]

        # ------------------------------------------------------------------
        # 2.  Refinement of each seed
        # ------------------------------------------------------------------
        for seed_vec in top_seeds:
            if time.time() >= deadline - 5.0:
                break

            # Phase A – exact sub‑gradient descent (longer than before)
            remaining = deadline - time.time()
            max_iter_sub = int(min(150_000, remaining * 30))
            print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
            refined = subgradient_descent_exact(
                seed_vec,
                deadline,
                max_iter=max_iter_sub,
                step0=0.04,
            )

            # Phase B – simulated‑annealing polishing
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_sa = int(min(150_000, remaining * 30))
                print(f"  Simulated annealing (max_iter={max_iter_sa}) …")
                refined = simulated_annealing(
                    refined,
                    rng,
                    deadline,
                    max_iters=max_iter_sa,
                    init_temp=0.03,
                    decay=0.9998,
                )

            # Phase C – LP improvement (few quick iterations)
            if time.time() < deadline - 5.0:
                M_cur = float(convolution_fft(refined, refined).max())
                print(f"  LP improvement (current M={M_cur:.6e}) …")
                refined = lp_improve(refined, M_cur, deadline, max_iter=4, t=0.25)

            # Final official evaluation
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"  → official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        # Split remaining budget between sub‑gradient and Adam (smooth)
        it_sub = int(min(200_000, remaining * 25))
        it_adam = int(min(150_000, remaining * 25))

        print("\nFinal global polishing (sub‑gradient + Adam) …")
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.02)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=it_adam,
            lr0=0.12,
            lam_start=2.0,
            lam_end=100.0,
            noise_every=1500,
            noise_scale=2e-5,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
