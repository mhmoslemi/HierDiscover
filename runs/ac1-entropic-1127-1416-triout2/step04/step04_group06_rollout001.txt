Standard algorithm:

def simplex_project(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().

Algorithm
---------

* Load the previously found best vector (if available) and keep it as a
  global best.
* Perform many independent restarts.  Each restart begins from a random
  distribution drawn from a handful of simple families (Beta density,
  Dirichlet, edge‑heavy, sparse) and optionally mixed with the current
  best.
* For each restart run a **sub‑gradient descent** on the exact
  L∞‑norm of the self‑convolution:
    - Compute the convolution a∗a (FFT, O(n log n)).
    - Find the index set I where the convolution attains its maximum.
    - The sub‑gradient of max_k (a∗a)_k w.r.t. a is
          g_i = Σ_{k∈I} 2·a_{k−i}   (zero if k−i is out of range).
      This can be built by slicing the vector a for each maximal index.
    - Take an additive step a ← a – η·g and **project back onto the
      probability simplex** (non‑negative entries summing to 1).
    - Step size η decays as η₀/√t.
    - Track the best vector seen during the descent.
* After the sub‑gradient phase run a short **simulated‑annealing**
  refinement using simple mass‑transfer moves, again keeping the best
  vector.
* Continue restarts until the time budget (≈1000 s) is exhausted and
  return the overall best vector (list of floats, sum ≈ 1).

The method directly minimises the true max‑convolution rather than a
smooth approximation, and the simplex projection guarantees feasibility.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_project(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0                # cumulative sum minus 1
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of evaluate_sequence() for a vector that already
    satisfies sum(a) == 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution length = 2*n‑1
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * np.max(conv)


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    # log‑density to avoid overflow/underflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector (i.e. i.i.d. Gamma(1))."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_project(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """
    Linear “edge‑heavy” shape: larger values near both ends,
    smaller in the centre.
    """
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = dist_to_center.astype(float) + 1.0
    return simplex_project(a)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Place mass on k randomly chosen positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_project(a)
