def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (preserves non‑negativity)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector a that already satisfies
    sum(a) == 1.  Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution length = 2*n‑1
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n          # mid‑point rule
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)   # avoid overflow/underflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector (i.e. i.i.d. Gamma(1))."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """Linear “edge‑heavy” shape: larger values near the two ends, smaller in centre."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)
    a = dist.astype(float) + 1.0
    return simplex_normalise(a)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Place mass on *k* randomly chosen positions (the rest are zero)."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


def init_exponential(n: int, decay: float = 0.001) -> np.ndarray:
    """a_i ∝ exp(‑decay · i)  (monotonically decreasing)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-decay * i)
    return simplex_normalise(a)

    and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # linearly increase smoothing parameter α
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # convolution a∗a
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

        # soft‑max weights over convolution entries
        maxc = conv.max()
        cur_alpha = alpha_start + (alpha_end - alpha_start) * (it + 1) / max_iters
        # stable exp
        exp_vals = np.exp(cur_alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp

        # gradient g = cross‑correlation of a with p
        L = 1 << ((3 * n - 2).bit_length())    # next power of two ≥ 3n‑2
        fft_a_full = np.fft.rfft(a, n=L)
        fft_p = np.fft.rfft(p, n=L)
        corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L)
        offset = len(p) - 1
        g = corr[offset:offset + n]

        # exponentiated‑gradient update on the simplex
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)

        # occasional exact evaluation to monitor progress
        if (it & 31) == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[MD] it {it:5d} → new best {best_score:.6f}")

    return best_a, best_score


def max_gradient_descent(
    a0: np.ndarray,
    max_iters: int,
    eta0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Mirrors the exact max‑gradient (sub‑gradient of ℓ∞) descent.
    At each iteration only the (one‑hot) index achieving the current
    maximum of the self‑convolution is used to form the gradient.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # convolution a∗a
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

        # indices attaining the maximum
        max_val = conv.max()
        max_idxs = np.flatnonzero(np.abs(conv - max_val) < 1e-14)

        # build gradient: sum over all active indices
        g = np.zeros(n, dtype=float)
        idx_array = np.arange(n)
        for k in max_idxs:
            src = k - idx_array
            mask = (src >= 0) & (src < n)
            g[mask] += a[src[mask]]

        # update with multiplicative step (mirror descent on simplex)
        eta = eta0 / math.sqrt(it + 1.0)
        # protect against overflow/underflow
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)

        if (it & 31) == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[MGD] it {it:5d} → new best {best_score:.6f}")

    return best_a, best_score


def simulated_annealing(
    a0: np.ndarray,
    init_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Classic SA with mass‑transfer moves.  The temperature starts at 2 % of
    the current score and cools geometrically to 1e‑5 % of the start.
    """
    a = a0.copy()
    cur_score = init_score
    best_a = a.copy()
    best_score = init_score

    T = cur_score * 0.02
    T_end = cur_score * 1e-5
    n = a.shape[0]
    iteration = 0

    while time.time() < deadline and T > T_end:
        # pick a source weighted by current mass
        src = np.random.choice(n, p=a)
        # pick a different target uniformly
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1
        if a[src] <= 0.0:
            continue
        # transfer a random fraction of src's mass
        delta = a[src] * np.random.rand()
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] iter {iteration:6d} → new best {best_score:.6f}")

        T *= 0.9995          # geometric cooling
        iteration += 1

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence() under a 1000‑second budget.
    Returns the best sequence found (as a list of floats, sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # 0. Load previous solution (if supplied) – otherwise start uniform.
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        init_vec = simplex_normalise(init_vec)
        n = init_vec.size
        best_a = init_vec.copy()
        best_score = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1 (n={n}), score={best_score:.6f}")
    except NameError:
        # No previous vector – use a uniform vector of length 5146 (typical size)
        n = 5146
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_a)
        print(f"[Init] No previous sequence – using uniform start, n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Family of simple initialisers (all already normalised to sum = 1)
    # ------------------------------------------------------------------
    init_fns = [
        lambda: beta_sequence(np.random.uniform(0.2, 5.0),
                              np.random.uniform(0.2, 5.0), n),
        lambda: init_dirichlet(n),
        lambda: init_edge_concentrated(n),
        lambda: init_sparse(n, k=np.random.randint(5, 30)),
        lambda: init_exponential(n, decay=np.random.uniform(0.0005, 0.005))
    ]

    # ------------------------------------------------------------------
    # 2. Phase 1 – many random restarts with smoothed mirror descent.
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * budget_s   # about 55 % of total time
    restart_counter = 0
    while time.time() < phase1_deadline - 5.0:   # keep a few seconds for later phases
        # pick an initial vector
        a0 = np.random.choice(init_fns)()
        # occasional blend with the current best to keep good mass pattern
        if np.random.rand() < 0.2:
            a0 = 0.5 * a0 + 0.5 * best_a
            a0 = simplex_normalise(a0)

        # mirror descent parameters (tuned for speed vs precision)
        max_iters = 5000
        eta0 = 0.9
        alpha_start = 1.0
        alpha_end = 3000.0
        a_opt, score_opt = mirror_descent(
            a0, max_iters, eta0, alpha_start, alpha_end, deadline=phase1_deadline
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Restart {restart_counter}] improved to {best_score:.6f}")

        restart_counter += 1

    print(f"[Phase1] Completed {restart_counter} restarts, best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Phase 2 – precise max‑gradient descent from the current best.
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.90 * budget_s   # allocate up to 90 % of time
    max_iters = int(1e5)   # very large; the loop will stop by the time limit
    a_opt, score_opt = max_gradient_descent(
        best_a, max_iters, eta0=0.6, deadline=phase2_deadline
    )
    if score_opt < best_score:
        best_score = score_opt
        best_a = a_opt.copy()
        print(f"[Phase2] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Phase 3 – final simulated‑annealing refinement.
    # ------------------------------------------------------------------
    phase3_deadline = deadline
    a_opt, score_opt = simulated_annealing(best_a, best_score, deadline=phase3_deadline)
    if score_opt < best_score:
        best_score = score_opt
        best_a = a_opt.copy()
        print(f"[Phase3] final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best vector (already normalised to sum ≈ 1).
    # ------------------------------------------------------------------
    print(f"[Done] Best score after all phases: {best_score:.6f} (n={n})")
    return [float(v) for v in best_a.tolist()]
