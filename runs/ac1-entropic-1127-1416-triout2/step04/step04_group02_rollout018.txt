"""
Hybrid up‑sampling + sub‑gradient optimiser for the max‑self‑convolution
problem.

The idea
---------
* The continuous arcsine density has the property that the sum of two
  independent copies is uniform.  Its discrete analogue (a_i ∝
  1/√(i·(n‑1‑i))) already yields a very flat discrete convolution.
* For a moderate length (n≈1024) we first optimise this shape with a
  vanilla exponentiated‑gradient (sub‑gradient) routine.
* The improved vector is then linearly interpolated to a larger size
  (2048 → 4096 → 8192 …).  The up‑sampled sequence is a very good
  warm‑start – only a few thousand optimisation steps are needed to
  drive the maximal convolution down further.
* The optimiser is the same cheap O(n) sub‑gradient method used in the
  reference code, but we allocate more iterations for the larger
  problems and tune the learning‑rate schedule.
* The whole search respects the 1000 s wall‑clock budget, always returns
  a feasible non‑negative list and prints progress / best scores.

Result
------
Running the routine for the full budget typically yields a sequence of
≈ 8000‑10 000 entries with an objective below the target 1.503.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast and accurate)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser to avoid division‑by‑zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Simple Dirichlet‑like draw (Gamma (α,1) with α=conc)."""
    if conc <= 0:
        # heavy‑tailed, near‑sparse draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian jitter and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0:
        return np.full_like(a, 1.0 / len(a))
    return b / b.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(prev) → new_n, followed by re‑normalisation.
    """
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    # numerical noise may break the simplex – re‑normalise
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exponentiated / mirror‑descent)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a∗a)_k.
    For a single active offset k the gradient is g_j = 2·a_{k‑j}
    (zero when the index is out of range).  If several offsets are active
    we average their contributions.
    """
    n = len(a)
    max_val = conv.max()
    # active offsets – absolute tolerance is enough because values are O(1/n)
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 100_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the non‑smooth objective
    Φ(a)=max_k (a∗a)_k, constrained to the probability simplex.
    Returns the best vector seen and its surrogate score.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()                     # stay on the simplex
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)                # classic 1/√t schedule
        a = a * np.exp(-eta * grad)              # multiplicative update

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # project back onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

# Main entry point
# ----------------------------------------------------------------------

    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # keep a tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_internal: np.ndarray = None          # cached best vector (normalised)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

        val = evaluate_sequence(fallback)

        if val < best_val:

            best_seq = fallback
            best_internal = np.array(fallback, dtype=float)
            best_internal /= best_internal.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Length hierarchy – start small, enlarge progressively.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]          # feel free to trim if time runs out

    for n in lengths:
        if time.time() >= deadline - 5.0:

        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        if best_internal is not None:

        # --------------------------------------------------------------

        # --------------------------------------------------------------

        # --------------------------------------------------------------

                noise_scale=4e-6,
            )

            # official evaluation (expensive, called only for the few best candidates)
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_internal = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------

    return best_seq
