"""
Propose a low‑score step function.

The algorithm is a multi‑stage stochastic optimiser:

1. **Random initialisation** – generate many candidate vectors.
   *Sparse* candidates: pick a random subset of indices and put equal
   mass on them (the classic “random set” approach).
   *Dense* candidates: sample an exponential (Dirichlet) vector and
   project onto the probability simplex.

2. **Deterministic annealing (soft‑max gradient descent)** – starting from
   each promising candidate we minimise the smoothed maximum of the
   auto‑convolution
        f(a) = (1/β)·log Σ_s exp(β·(a⋆a)_s) .
   The gradient of the soft‑max can be written as a convolution, so each
   iteration requires two fast FFT‑based convolutions and an Adam‑style
   update.  The temperature β is increased linearly from a small value
   (smooth objective) to a larger value (focussing on the true maximum).

   The iterate is projected onto the probability simplex after each step,
   keeping the sum of the entries equal to 1 and all entries non‑negative.

3. **Hill‑climbing polishing** – a fast stochastic hill‑climber (swap /
   mass‑transfer moves) is applied to the best vector found by the
   gradient phase.  It accepts only improving moves, but the step size
   is adapted automatically.

All stages respect the 1000 s time budget.  The best sequence discovered
is returned as a plain list of floats, ready for the external
`evaluate_sequence` harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of real vectors using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when the sum of a equals 1.
    Returns 2·n·max(a⋆a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Gradient descent with soft‑max smoothing of the max‑convolution
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a_init: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr: float = 0.07,
    beta_start: float = 0.5,
    beta_end: float = 6.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective
        f_β(a) = (1/β)·log Σ_s exp(β·(a⋆a)_s).

    Parameters
    ----------
    a_init : np.ndarray
        Starting vector (already normalized to sum = 1).
    time_deadline : float
        Absolute time (seconds since epoch) at which to stop.
    max_iter : int
        Upper bound on the number of iterations.
    lr : float
        Base learning rate for Adam.
    beta_start, beta_end : float
        Temperature schedule; β grows linearly from start to end.

    Returns
    -------
    best_a : np.ndarray
        The best vector (lowest exact score) encountered.
    best_score : float
        Exact score 2·n·max(a⋆a) of `best_a`.
    """
    a = a_init.astype(float).copy()
    n = len(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    a_rev = a[::-1]          # will be updated each iteration

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # ---------- compute (a⋆a) ----------
        conv = convolution_fft(a, a)               # length 2n‑1
        conv_max = np.max(conv)

        # ---------- soft‑max weights ----------
        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * (t / max_iter)

        # stability: subtract max before exponentiation
        exp_vals = np.exp(beta * (conv - conv_max))
        w = exp_vals / np.sum(exp_vals)            # sum_s w_s = 1

        # ---------- gradient ----------
        # g_i = 2 Σ_s w_s a_{s‑i}
        # This is 2·(w * a_rev) evaluated at indices (n‑1 … n‑1+n‑1)
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)      # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # keep only n entries

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ---------- projection ----------
        a = project_to_simplex(a)

        # ---------- exact evaluation ----------
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to shake out of plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_score

# Simple hill‑climber (same spirit as the original code)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber that swaps entries or moves a fraction of mass.
    Returns the best sequence found (as a plain Python list) and its exact score.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # step size as a fraction of the mass at the source index
    step_scale = 0.12

    for _ in range(max_steps):
        if time.time() >= time_deadline:
            break

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # increase after success
        else:
            step_scale *= 0.97                         # decrease after failure

    return best_a.tolist(), best_score

def random_sparse_vector(n: int, density: float) -> np.ndarray:
    """
    Return a length‑n vector with `k = max(2, round(density·n))`
    uniformly weighted non‑zero entries (sum = 1).
    """
    k = max(2, int(round(density * n)))
    if k > n:
        k = n
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dense_dirichlet(n: int) -> np.ndarray:
    """Sample a dense positive vector from an exponential distribution and project to the simplex."""
    a = np.random.exponential(scale=1.0, size=n)
    return project_to_simplex(a)

            best_val = cur_score
            best_seq = a0.tolist()
            print(
                f"New random start best: score={best_val:.6f} (n={n}, type={init_type})"
            )

        # Heuristic: only launch the expensive gradient phase if the
        # candidate is at least *some* percent better than the current best.
        # This prevents wasting time on clearly inferior starts.
        if cur_score < best_val * 1.25:
            # --------------------------------------------------------------
            # 2.1️⃣  Soft‑max gradient descent (deterministic annealing)
            # --------------------------------------------------------------
            time_for_gd = min(15.0, (deadline - time.time() - polishing_reserve) / 2)
            gd_deadline = time.time() + time_for_gd

            refined_a, refined_score = softmax_gradient_descent(
                a0,
                time_deadline=gd_deadline,
                max_iter=120_000,
                lr=0.07,
                beta_start=0.5,
                beta_end=6.0,
            )

            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_a.tolist()
                print(
                    f"  Gradient descent improved best to {best_val:.6f} (n={n})"
                )

            # --------------------------------------------------------------
            # 2.2️⃣  Brief hill‑climb polish (if time permits)
            # --------------------------------------------------------------
            if time.time() < deadline - polishing_reserve:
                hill_deadline = deadline - polishing_reserve
                polished_seq, polished_score = hill_climb_permutation(
                    refined_a.tolist(),
                    time_deadline=hill_deadline,
                    max_steps=200_000,
                )
                if polished_score < best_val:
                    best_val = polished_score
                    best_seq = polished_seq
                    print(
                        f"  Hill‑climb polished best to {best_val:.6f} (n={n})"
                    )

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing on the overall best sequence (if any)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 2.0:
            print("\nFinal hill‑climb polishing")
            final_seq, final_score = hill_climb_permutation(
                best_seq,
                time_deadline=deadline,
                max_steps=int(1.0e5),
            )
            if final_score < best_val:
                best_val = final_score
                best_seq = final_seq
                print(f"Final polishing improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Safeguard – fall back to a tiny uniform vector if nothing found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5️⃣  Report the final score using the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return a plain Python list of floats (the harness expects this)
    return best_seq
