assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=2\,n\;\max\bigl((a*a)\bigr) ,
\qquad a_i\ge 0 ,\;\;\sum_i a_i=1 .
\]

`score` is invariant under a global scaling of `a`, therefore we can work on the **probability simplex** (`∑a=1`).  

The algorithm consists of two phases:

1. **Continuation‑Adam optimisation** –  
   *Parameters* are the unconstrained vector `w`.  
   `a = softmax(w)` guarantees the simplex constraints.  
   A smooth surrogate of the ℓ∞‑norm is used  

   \[
   \Phi_\beta(a)=\log\!\Bigl(\sum_k e^{\beta\, (a*a)_k}\Bigr) ,
   \]

   whose gradient w.r.t. `a` can be written as a convolution
   (computed by one FFT).  
   By slowly increasing the smoothing factor `β` (continuation) the
   optimiser first sees a very smooth landscape and later concentrates
   on the true maximum.  Adam (exponential‑moving‑average of gradients)
   is applied on `w`.

2. **Fine‑grained simulated annealing** –  
   Starting from the best point of phase 1, mass is moved from indices
   that contribute most to the current maximal convolution entry to
   indices that contribute the least.  A geometric cooling schedule
   accepts worsening moves with the Metropolis probability.  This
   polishing step is cheap (few 10 000 moves) and typically pushes the
   score a little further.

Both phases use only NumPy and FFTs, run completely on the simplex,
and respect the 1000 s time budget.

```python
"""
Propose a step‑height sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise a probability vector (uniform or the previously found
   ``height_sequence_1``).  Work with the unconstrained log‑probabilities
   ``w`` and recover the simplex vector with a softmax.

2. Continuation – Adam optimisation on a smooth approximation of the
   ℓ∞‑norm of the self‑convolution:
      * β (smoothness) starts small (≈1) and is multiplied by 1.2 every
        few thousand iterations until a maximum β≈200.
      * The gradient of  Φ_β(a)=log Σ exp(β·conv) is
          grad_a = 2·(a * p)      (convolution),
        where   p = softmax(β·conv).
      * The chain rule for the softmax yields the gradient w.r.t. `w`,
        which is fed to Adam (β₁=0.9, β₂=0.999).

3. Simulated annealing – from the best point of phase 2:
   * pick a source index proportional to the sub‑gradient of the
     currently maximal convolution entry,
   * pick a target index inversely proportional to that sub‑gradient,
   * move a random fraction of the source mass to the target,
   * accept the move with the Metropolis rule (temperature decays
     geometrically).

The best vector seen during the whole run is returned (still normalised
to sum = 1).  All entries are non‑negative and the evaluation function
will never reject the output.

"""

import time
import math
import numpy as np


def _softmax_from_w(w: np.ndarray) -> np.ndarray:
    """Return a simplex vector a = softmax(w)."""
    w_shift = w - np.max(w)          # numeric stability
    exp_w = np.exp(w_shift)
    a = exp_w / exp_w.sum()
    return a


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a, length 2·n‑1, using FFT."""
    n = a.shape[0]
    L = 2 * n - 1
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact value of ``evaluate_sequence`` for a vector with sum = 1."""
    return float(2.0 * n * conv.max())


def _smooth_gradient_w(a: np.ndarray, beta: float) -> tuple:
    """
    Gradient of the smoothed max‑convolution w.r.t. the log‑weights w.

    Returns (grad_w, conv, p) where
        grad_w – ∂Φ_β/∂w,
        conv   – self‑convolution a*a,
        p      – softmax(β·conv) used in the computation.
    """
    n = a.shape[0]
    L = 2 * n - 1

    # self‑convolution
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)

    # softmax over the convolution entries (smooth max)
    maxc = conv.max()
    # exp(β·(conv − maxc)) stays ≤ 1, avoids overflow
    exp_term = np.exp(beta * (conv - maxc))
    p = exp_term / exp_term.sum()          # shape (2n‑1)

    # gradient w.r.t. a : 2·(a * p)  (convolution, first n entries)
    fft_p = np.fft.rfft(p, n=L)
    grad_a_full = np.fft.irfft(fft_a * fft_p, n=L)
    grad_a = 2.0 * grad_a_full[:n]

    # chain rule for the softmax a = softmax(w)
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return grad_w, conv, p


def _subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at point ``a``.
    Averaged over all active (near‑maximal) convolution indices.
    """
    n = a.shape[0]
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]

    if active.size == 0:
        return 2.0 * a

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The algorithm runs for at most ``budget_s``
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 1 – initialise a on the simplex
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        a = init / init.sum()
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        n = 5146
        a = np.ones(n, dtype=float) / n
        print(f"No existing sequence – starting uniform, length {n}.")

    # log‑weights w (softmax parametrisation)
    w = np.log(a + 1e-12) - np.max(np.log(a + 1e-12))

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5                     # initial learning‑rate

    # best solution tracking
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(best_a), n)
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2 – continuation + Adam optimisation
    # --------------------------------------------------------------
    beta = 1.0          # start with a smooth approximation
    beta_max = 200.0
    beta_update_every = 5000          # iterations
    it = 0

    while time.time() < deadline - 20.0:          # leave ~20 s for SA
        it += 1

        # obtain current simplex vector
        a = _softmax_from_w(w)

        # gradient of the smooth max‑convolution and the raw convolution
        grad_w, conv, _ = _smooth_gradient_w(a, beta)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decreasing step size
        lr = lr0 / math.sqrt(it)

        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional clipping to keep w numerically stable
        if np.max(w) - np.min(w) > 50.0:
            w = np.clip(w, np.max(w) - 50.0, np.max(w))

        # evaluate the true objective (no smoothing)
        cur_score = _score_from_conv(conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"Iter {it:6d}: new best {best_score:.6f}")

        # increase smoothness parameter β periodically
        if it % beta_update_every == 0:
            beta = min(beta_max, beta * 1.2)

    print(f"Adam phase finished after {it} iterations, best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 3 – guided simulated annealing (fine polishing)
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    conv_cur = _conv_self(cur_a)
    cur_score = _score_from_conv(conv_cur, n)

    T_start = cur_score * 0.05           # start at 5 % of the current score
    T_end = cur_score * 1e-5
    sa_iter = 0

    while time.time() < deadline:
        # geometric cooling
        T = max(T_end, T_start * (0.9995 ** sa_iter))

        # sub‑gradient of the max‑convolution entry (active set)
        grad = _subgradient_of_max_conv(cur_a, conv_cur)

        # source – proportional to the sub‑gradient (high contribution)
        src_probs = grad / grad.sum()
        src = np.random.choice(n, p=src_probs)

        # target – inversely proportional (low contribution)
        inv = 1.0 / (grad + 1e-12)
        tgt_probs = inv / inv.sum()
        tgt = np.random.choice(n, p=tgt_probs)

        if src == tgt or cur_a[src] <= 0.0:
            sa_iter += 1
            continue

        # move a random fraction of the source mass
        delta = cur_a[src] * np.random.rand() * 0.5   # ≤ 50 % of source
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # evaluate candidate
        conv_cand = _conv_self(cand)
        cand_score = _score_from_conv(conv_cand, n)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-12))

        if accept:
            cur_a = cand
            conv_cur = conv_cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {sa_iter:7d}: new best {best_score:.6f}")

        # periodic random restart to avoid dead‑ends
        if sa_iter % 50000 == 0 and sa_iter > 0:
            cur_a = np.random.dirichlet(np.ones(n))
            conv_cur = _conv_self(cur_a)
            cur_score = _score_from_conv(conv_cur, n)
            print(f"SA iter {sa_iter}: random restart performed")

        if sa_iter % 20000 == 0 and sa_iter > 0:
            print(f"SA iter {sa_iter:7d}: best so far {best_score:.6f}")

        sa_iter += 1

    print(f"Simulated‑annealing finished after {sa_iter} iterations, final best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4 – return best sequence (already normalised to sum = 1)
    # --------------------------------------------------------------
    best_a /= best_a.sum()
    return [float(v) for v in best_a.tolist()]
```
