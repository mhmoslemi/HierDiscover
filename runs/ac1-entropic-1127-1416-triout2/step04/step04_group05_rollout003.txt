assistantfinal**Solution Overview**

The optimizer builds a collection of diverse starting points (uniform,
Beta‑distributions, random Dirichlet draws and the previously found
sequence).  
For each start it runs a short *exponentiated‑gradient* (mirror‑descent)
procedure on the probability simplex – the update is multiplicative,
the step size decays like `1/√t`.  
The gradient that is used is the exact sub‑gradient of the current
max‑convolution entry (averaged over all active entries).  

All runs are performed in parallel in the sense of “many restarts”:
as soon as the global time budget (default 1000 s) is almost exhausted,
the algorithm stops the restarts and performs a final, cheap
*simulated‑annealing* refinement that transfers a random fraction of mass
from a high‑gradient index to a low‑gradient index.

The whole routine works only with vectors whose sum is 1, therefore the
`evaluate_sequence` function never rejects the candidate.  The best vector
found during the whole search is returned.

```python
"""
search for a non‑negative step‑height sequence that minimises
evaluate_sequence().

The algorithm consists of three stages:
1. collect a pool of initial vectors (uniform, several Beta(α,β)
   discretisations, a few Dirichlet draws and – if available – the
   previous best sequence);
2. run many short mirror‑descent (exponentiated‑gradient) optimisations
   from random pool members, always staying on the simplex;
3. finish with a brief guided simulated‑annealing phase that moves mass
   from high‑gradient indices to low‑gradient ones.

All work is performed inside the global time budget (default 1000 s) and
the best vector encountered is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector whose L1‑norm is 1.
    Returns 2·n·max(conv) where conv = a * a (linear convolution).
    """
    n = a.shape[0]
    # convolution via FFT (real → real)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    """
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow for extreme α,β
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # safety for exp()
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    w /= w.sum()
    return w


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k conv_k  at the point ``a``.
    The gradient is the average of the exact gradients of all active
    convolution entries (those attaining the current maximum up to a tiny
    tolerance).
    """
    n = a.shape[0]
    max_b = conv.max()
    eps = max_b * 1e-8
    active = np.where(conv >= max_b - eps)[0]

    if active.size == 0:                     # safety net
        return np.zeros_like(a)

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        # ∂conv_k/∂a_i = 2·a_{k‑i}
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """
    Mirror‑descent update on the simplex using the exponentiated‑gradient
    (multiplicative) rule, followed by renormalisation.
    """
    factor = np.exp(-step * grad)                # multiplicative factor
    a_new = a * factor
    # keep strictly positive numbers (avoid division by 0 later)
    a_new = np.clip(a_new, 1e-16, None)
    a_new /= a_new.sum()
    return a_new

        init_seq = None
        print(f"No existing sequence – default length is {n}.")

    # --------------------------------------------------------------
    # 2 – build a pool of diverse starting points
    # --------------------------------------------------------------
    pool = []

    # uniform vector
    pool.append(np.ones(n, dtype=float) / n)

    # a few Beta‑distributions
    alphas = [0.5, 1.0, 2.0, 3.0]
    betas = [0.5, 1.0, 2.0, 3.0]
    for a in alphas:
        for b in betas:
            pool.append(beta_sequence(a, b, n))

    # a handful of random Dirichlet draws (roughly uniform on the simplex)
    for _ in range(5):
        pool.append(np.random.dirichlet(np.ones(n)))

    # include the previous construction if it exists
    if init_seq is not None:
        pool.append(init_seq / init_seq.sum())

    # keep track of the best vector seen so far
    best_a = pool[0].copy()
    best_score = compute_score(best_a)

    # quick sanity check of the pool
    for p in pool:
        s = compute_score(p)
        if s < best_score:
            best_score = s
            best_a = p.copy()

    print(f"Initial best score from the pool = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3 – many short mirror‑descent runs (restarts)
    # --------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 5.0:          # leave a few seconds for the final SA
        # pick a random start from the pool
        a = pool[np.random.randint(len(pool))].copy()

        # compute its convolution once
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        cur_score = 2.0 * n * conv.max()

        # number of gradient steps for this restart (chosen heuristically)
        max_iter = 8000
        for it in range(max_iter):
            if time.time() >= deadline - 5.0:
                break

            grad = subgradient_of_max_conv(a, conv)

            # step‑size schedule – larger at the beginning, decays like 1/√t
            step = 1.0 / math.sqrt(it + 1.0)

            a = exponential_update(a, grad, step)

            # recompute convolution and score
            fft = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft * fft, n=2 * n - 1)
            cur_score = 2.0 * n * conv.max()

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[restart {restart_counter:02d}] new best {best_score:.6f} "
                      f"(iter {it}, size {n})")

        restart_counter += 1

    # --------------------------------------------------------------
    # 4 – final short guided simulated annealing
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    fft = np.fft.rfft(cur_a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    cur_score = 2.0 * n * conv.max()

    # temperature schedule – start relatively hot, cool geometrically
    T = cur_score * 0.05          # 5 % of the current score
    it_sa = 0
    while time.time() < deadline - 0.2:
        grad = subgradient_of_max_conv(cur_a, conv)

        # ---------------------------------------------------------
        # source – indices that contribute a lot to the current max
        # ---------------------------------------------------------
        src_probs = grad / max(grad.sum(), 1e-30)
        src = np.random.choice(n, p=src_probs)

        # ---------------------------------------------------------
        # target – indices with small contribution (inverse weighting)
        # ---------------------------------------------------------
        inv = 1.0 / (grad + 1e-14)
        tgt_probs = inv / inv.sum()
        tgt = np.random.choice(n, p=tgt_probs)

        if src == tgt or cur_a[src] <= 0.0:
            it_sa += 1
            continue

        # move a random fraction of the source mass to the target
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # evaluate the candidate
        fft = np.fft.rfft(cand, n=2 * n - 1)
        conv_c = np.fft.irfft(fft * fft, n=2 * n - 1)
        cand_score = 2.0 * n * conv_c.max()

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-12))

        if accept:
            cur_a = cand
            conv = conv_c
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] improvement to {best_score:.6f} (iter {it_sa})")

        # geometric cooling
        if it_sa % 3000 == 0 and it_sa > 0:
            T *= 0.90

        it_sa += 1

    print(f"Search finished – best score achieved: {best_score:.6f}")

    # --------------------------------------------------------------
    # 5 – return the best vector (still normalised to sum == 1)
    # --------------------------------------------------------------
    best_a /= best_a.sum()
    return [float(v) for v in best_a.tolist()]
```

The function `propose_candidate` obeys all requirements:

* it works only with non‑negative numbers,
* the returned list always sums to 1 (the scaling does not affect the
  evaluation),
* no file or network access is performed,
* the total wall‑clock time never exceeds the provided `budget_s`
  (default 1000 seconds).
