```
def compute_convolution(a):
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2*n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2*n - 1)
    return conv

def compute_gradient(a, p):
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())    # next power of two >= 3n-2
    fft_a = np.fft.rfft(a, n=L)
    fft_p = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_p), n=L)
    offset = len(p) - 1
    return corr[offset:offset+n]
```

```
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    alpha = alpha_start
    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break
        # compute convolution
        conv = compute_convolution(a)
        # compute softmax p
        maxc = conv.max()
        # stable exponentiation
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0/conv.size)
        else:
            p = exp_vals / sum_exp
```

The goal is to find a non‑negative vector `a` (the heights of a step function) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `*` is the linear convolution and `n = len(a)`.  
If `a` is normalised to `sum(a)=1` the objective reduces to  

\[
F(a)=2\,n\;\max (a*a).
\]

Hence we minimise the peak of the self‑convolution while keeping the vector on the probability simplex.

The programme below implements a **two‑stage optimisation** that fits comfortably into the 1000 s budget:

1. **Entropic mirror descent on a smooth approximation of the max‑norm**  
   The smooth surrogate  

   \[
   \phi_\alpha(a)=\frac1\alpha\log\sum_k \exp\bigl(\alpha\,(a*a)_k\bigr)
   \]

   has gradient  

   \[
   \nabla\phi_\alpha(a)_i = \sum_k p_k\,a_{k-i},
   \qquad  
   p_k=\frac{\exp(\alpha\,(a*a)_k)}{\sum_j\exp(\alpha\,(a*a)_j)} .
   \]

   An exponentiated‑gradient (mirror descent on the simplex) step  

   \[
   a_i \leftarrow a_i\;\exp(-\eta\,\nabla\phi_\alpha(a)_i),\qquad  
   a\gets a/\sum a
   \]

   is performed with a diminishing step‑size `η` and a **slowly increasing smoothing
   parameter** `α`.  
   FFT‑based convolutions give each iteration `O(n log n)` time.

2. **Guided local refinement**  
   After the long mirror‑descent phase we use the *current* gradient to move a small
   amount of probability mass from indices with a **large gradient** (they tend to
   increase the peak) to indices with a **small gradient**.  Each move is accepted
   only if the true objective improves; otherwise it is discarded.  This cheap
   hill‑climbing removes the remaining local artefacts and typically squeezes a
   few extra digits out of the objective.

The algorithm repeatedly restarts from a handful of simple families
(`Beta`, `Dirichlet`, edge‑concentrated and sparse vectors),
occasionally blending the restart with the best point found so far.
All restarts share the same time budget, and the best vector encountered
throughout the run is finally returned.

The implementation respects the sandbox restrictions (no lambdas, no I/O,
single‑CPU) and can be used directly by the evaluator.

```python
"""
Propose a step‑function (non‑negative vector) minimising
evaluate_sequence().

Algorithm
---------
1. Initialise from several simple families (Beta, Dirichlet,
   edge‑heavy, sparse) and keep the best vector seen.
2. Long entropic mirror‑descent on the smooth log‑sum‑exp surrogate
   of max(conv(a,a)).  The smoothing parameter α is linearly increased
   from 1 to a large value (≈600) while the step size η decays as
   1/√t.  All convolutions are performed with FFTs (O(n log n)).
3. Guided local refinement: repeatedly move a small amount of mass
   from the index with the largest gradient to the index with the
   smallest gradient; accept only if the true objective improves.
4. Return the best normalised vector found within the time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector that already satisfies
    sum(a) == 1.
    Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution length = 2*n‑1
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def compute_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2*n-1)."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    return np.fft.irfft(fft_a * fft_a, n=2 * n - 1)


def compute_gradient(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max‑smoothed max‑norm:
        g_i = Σ_k p_k * a_{k-i}
    computed via a non‑circular cross‑correlation (FFT, O(n log n)).
    """
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())          # next power of two ≥ 3n‑2
    fft_a = np.fft.rfft(a, n=L)
    fft_p = np.fft.rfft(p, n=L)
    # cross‑correlation a ⋆ p  (linear, non‑circular)
    corr = np.fft.irfft(fft_a * np.conj(fft_p), n=L)
    offset = len(p) - 1               # index where lag 0 appears
    return corr[offset:offset + n]


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over the convolution entries (numerically stable)."""
    maxc = conv.max()
    # avoid overflow / under‑flow
    shifted = alpha * (conv - maxc)
    np.clip(shifted, -700.0, 0.0, out=shifted)   # exp(≤‑700) ≈ 0
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


# ----------------------------------------------------------------------
# Random initialisers (all produce a simplex‑normalised vector)
# ----------------------------------------------------------------------


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    np.clip(logw, -700.0, 700.0, out=logw)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return simplex_normalise(w)


def init_beta(n: int) -> np.ndarray:
    """Beta family with parameters drawn from a wide range (including <1)."""
    a = np.random.uniform(0.05, 5.0)
    b = np.random.uniform(0.05, 5.0)
    return beta_sequence(a, b, n)


def init_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet(1,…,1) – i.i.d. Gamma(1)."""
    v = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(v)


def init_edge_heavy(n: int) -> np.ndarray:
    """Linear edge‑heavy shape (larger values near the two ends)."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)
    v = dist.astype(float) + 1.0
    return simplex_normalise(v)


def init_sparse(n: int) -> np.ndarray:
    """Very sparse vector: mass on a handful of random indices."""
    k = np.random.randint(5, min(200, n // 10 + 2))
    v = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=k, replace=False)
    v[chosen] = np.random.rand(k) + 1e-6
    return simplex_normalise(v)


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def mirror_descent(
    a_start: np.ndarray,
    max_iters: int,
    eta0: float,
    alpha_start: float,
    alpha_end: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent on the log‑sum‑exp surrogate.
    Returns the best vector (and its score) obtained during the run.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(best_a)

    # linear schedule for α
    for it in range(max_iters):
        if time.time() > deadline - 4.0:
            break

        # current smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it + 1) / max_iters

        # 1) convolution a*a
        conv = compute_convolution(a)

        # 2) soft‑max probabilities
        p = softmax_weights(conv, alpha)

        # 3) gradient of the smooth objective
        g = compute_gradient(a, p)

        # 4) exponentiated‑gradient update on the simplex
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)

        # 5) occasional evaluation
        if (it & 31) == 0:   # every 32 iterations
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[MD] iter {it:5d} – new best {best_score:.6f}"
                )
    return best_a, best_score


def local_refine(
    a_start: np.ndarray,
    cur_score: float,
    deadline: float,
    max_moves: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Guided hill‑climbing: move a small amount of probability from the index
    with the largest gradient to the index with the smallest gradient.
    Each move is accepted only if the true objective improves.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best = cur_score

    # Initialise the gradient (high α focuses on the true max)
    alpha = 600.0
    conv = compute_convolution(a)
    p = softmax_weights(conv, alpha)
    grad = compute_gradient(a, p)

    move = 0
    while move < max_moves and time.time() < deadline - 2.0:
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if a[src] <= 1e-12:
            # choose a random source with non‑zero mass
            src = np.random.choice(n, p=a)
        # transfer up to 10 % of the source mass
        delta = a[src] * 0.1 * np.random.rand()
        if delta <= 0.0:
            move += 1
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)
        if cand_score < best:
            a = cand
            best = cand_score
            best_a = cand.copy()
            # recompute gradient at the new point
            conv = compute_convolution(a)
            p = softmax_weights(conv, alpha)
            grad = compute_gradient(a, p)
            print(f"[Refine] move {move:5d} → new best {best:.6f}")
        else:
            # stale gradient may mis‑lead; refresh occasionally
            if move % 200 == 0:
                conv = compute_convolution(a)
                p = softmax_weights(conv, alpha)
                grad = compute_gradient(a, p)

        move += 1

    return best_a, best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence() within ~budget_s seconds.
    Returns the best vector found (as a list of floats that sum to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2            # tiny safety margin

    # ------------------------------------------------------------------
    # Load previous solution if available; otherwise start from uniform.
    # ------------------------------------------------------------------
    try:
        init_best = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        init_best = simplex_normalise(init_best)
        n = init_best.size
        best_a = init_best.copy()
        best_score = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1 (n={n}), score={best_score:.6f}")
    except NameError:
        n = 5146                         # same size as the previous run
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_a)
        print(f"[Init] No previous solution – uniform start, n={n}, score={best_score:.6f}")

    init_fns = [init_beta, init_dirichlet, init_edge_heavy, init_sparse]

    restart = 0
    # ---------------------------------------------------------------
    # Main restart loop – each restart performs a long MD run plus
    # a short guided refinement.
    # ---------------------------------------------------------------
    while time.time() < deadline - 8.0:          # keep a few seconds for final return
        # ---- pick a random initialiser --------------------------------
        a0 = init_fns[np.random.randint(len(init_fns))](n)

        # occasional blend with the current best (helps preserve good mass)
        if np.random.rand() < 0.15:
            a0 = (a0 + best_a) * 0.5
            a0 = simplex_normalise(a0)

        # ---- long mirror‑descent --------------------------------------
        md_iters = 12000                       # many iterations – still cheap
        a_md, sc_md = mirror_descent(
            a_start=a0,
            max_iters=md_iters,
            eta0=0.8,
            alpha_start=1.0,
            alpha_end=600.0,
            deadline=deadline,
        )
        if sc_md < best_score:
            best_score = sc_md
            best_a = a_md.copy()
            print(f"[Restart {restart}] after MD new best {best_score:.6f}")

        # ---- guided local refinement -----------------------------------
        a_ref, sc_ref = local_refine(best_a, best_score, deadline)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()
            print(f"[Restart {restart}] after refine new best {best_score:.6f}")

        restart += 1

    # ---------------------------------------------------------------
    # Final short run starting from the overall best point
    # ---------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_final, sc_final = mirror_descent(
            a_start=best_a,
            max_iters=8000,
            eta0=0.5,
            alpha_start=1.0,
            alpha_end=1000.0,
            deadline=deadline,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"[Final] improvement to {best_score:.6f}")

    print(f"[Done] Best score after {restart} restarts: {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
```
