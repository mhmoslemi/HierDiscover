"""
Improved search for a non‑negative step sequence that minimizes the official
evaluation function.  The algorithm works in three stages:

1. **Wide‑range random exploration** of several parametric families
   (Beta‑shapes, generalised arcsine, mixtures, triangular bumps and
   Dirichlet‑samples).  Only the best few candidates are kept
   (memory‑efficient “top‑K” heap).

2. **Heavy local refinement** of each selected seed:
   * a smooth Adam optimisation on a soft‑max surrogate,
   * exact sub‑gradient descent on the true objective,
   * a short simulated‑annealing polishing phase.
   The optimisation budget is split dynamically according to the remaining
   time.

3. **Final global polishing** of the overall best vector with a longer
   sub‑gradient + Adam run.

All operations stay within the 1000 s time limit, use only NumPy and the
standard library, and finally return a list of non‑negative heights ready for
``evaluate_sequence``.  The code is deliberately self‑contained – all helper
functions are defined at module level.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    After the projection onto the simplex Σa = 1 the denominator disappears,
    therefore the score is simply ``2·n·max(conv(a,a))``.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    A tiny epsilon (≥1e‑6) is required when α or β < 1 to avoid division‑by‑zero.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},   p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) where the sum constraint is handled
    by the subsequent projection onto the simplex.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length L = 2·n‑1
    max_c = np.max(conv)

    # Numerically stable soft‑max (weights sum to 1)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # Weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # Objective factor 2·n
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    The step size is η_t = step0 / sqrt(t).  Projection onto the simplex
    keeps the sum fixed to 1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search.
    Moves a small fraction of mass from one coordinate to another.
    The fast surrogate (2·n·max) is used for the acceptance test.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Up to 20 % of the mass at i is moved
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine combines a large random sampling of parametrised
    families with a heavy local optimisation (Adam → sub‑gradient → SA).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Wide random exploration (memory‑efficient top‑K)
    # ------------------------------------------------------------------
    n = 2048                     # length that gave the previous best
    top_k = 8                    # keep this many promising seeds
    best_pool: List[Tuple[float, np.ndarray]] = []  # (score, vector)

    def maybe_insert(score: float, vec: np.ndarray):
        """Insert (score,vec) into best_pool keeping its size ≤ top_k."""
        if len(best_pool) < top_k:
            best_pool.append((score, vec))
        else:
            # index of worst element
            worst_idx = max(range(len(best_pool)),
                            key=lambda i: best_pool[i][0])
            if score < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (score, vec)

    # deterministic easy starters
    uniform = np.full(n, 1.0 / n)
    maybe_insert(compute_score(uniform), uniform)

    arc = arcsine_initial(n)
    maybe_insert(compute_score(arc), arc)

    # random sampling loop
    max_candidates = 180_000
    cnt = 0
    while time.time() < deadline - 30.0 and cnt < max_candidates:
        cnt += 1
        r = rng.random()
        if r < 0.35:  # beta family
            alpha = rng.uniform(0.02, 5.0)
            beta = rng.uniform(0.02, 5.0)
            eps = rng.uniform(1e-6, 3.0)          # avoid zero when α<1
            cand = generate_beta_shape(n, alpha, beta, eps)
        elif r < 0.65:  # power / generalized arcsine
            p = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, 3.0)
            cand = generate_power_shape(n, p, eps)
        elif r < 0.80:  # mixture arcsine / uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)
        elif r < 0.90:  # triangular bump
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
        else:          # full Dirichlet sample (many small entries)
            dir_alphas = rng.exponential(scale=1.0, size=n) + 0.01
            cand = dir_alphas / dir_alphas.sum()

        sc = compute_score(cand)
        maybe_insert(sc, cand)

    # Extract the seed vectors (sorted by score, best first)
    best_pool.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in best_pool]

    # ------------------------------------------------------------------
    # 2.  Refinement of each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seed_vectors, 1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Refining seed {idx}/{len(seed_vectors)} ---")

        # A – smooth Adam optimisation (long run)
        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 45))
        print(f"    Adam smooth, max_iter={max_iter_adam}")
        refined, _ = adam_optimize_smooth(
            initial_a=seed_vec,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.12,
            lam_start=2.0,
            lam_end=100.0,
            noise_every=2000,
            noise_scale=2e-5,
        )

        # B – exact sub‑gradient descent
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(250_000, remaining * 40))
            print(f"    Sub‑gradient descent, max_iter={max_iter_sub}")
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=max_iter_sub,
                step0=0.03,
            )

        # C – simulated‑annealing polishing
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sa = int(min(250_000, remaining * 40))
            print(f"    Simulated annealing, max_iter={max_iter_sa}")
            refined = simulated_annealing(
                refined,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.015,
                decay=0.99995,
            )

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        sub_iters = int(min(300_000, remaining * 30))
        adam_iters = int(min(300_000, remaining * 30))

        print("\n--- Final global polishing ---")
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.02)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=adam_iters,
            lr0=0.08,
            lam_start=2.0,
            lam_end=80.0,
            noise_every=2000,
            noise_scale=1e-5,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
