def random_antisymmetric_vector(n, p):
    # returns array a of length n with sum=1
    k_target = int(max(2, round(p * n)))
    # Randomly choose up to 2*k_target positions biased? We'll loop until we get at least 2 after anti-sym enforcement.
    while True:
        idx = np.random.choice(n, size=k_target, replace=False)
        # enforce antisym
        selected = set(idx)
        new_selected = set()
        while selected:
            i = selected.pop()
            c = n - 1 - i
            if c in selected:
                # both present
                if np.random.rand() < 0.5:
                    new_selected.add(i)
                else:
                    new_selected.add(c)
                selected.discard(c)
            else:
                new_selected.add(i)
        # optional: exclude middle index if n odd
        if n % 2 == 1:
            mid = (n - 1) // 2
            if mid in new_selected and np.random.rand() < 0.5:
                new_selected.remove(mid)
        if len(new_selected) >= 2:
            break
    a = np.zeros(n, dtype=float)
    k = len(new_selected)
    a[list(new_selected)] = 1.0 / k
    return a
```

def exponentiated_subgrad_optimize(a_init, time_deadline, max_iter=500_000, eta0=0.1):
    a = a_init.copy()
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute gradient: g_i = 2 * a_{k - i} if valid else 0
        # Let's compute using vectorized:
        idx = np.arange(n)
        j = k - idx
        mask = (j >=0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # step size
        eta = eta0 / np.sqrt(t)
        # update multiplicatively
        a = a * np.exp(-eta * g)
        # project onto simplex
        a = project_to_simplex(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score=cur_score
            best = a.copy()
        # occasional jitter
        if t % 1000 == 0:
            a += np.random.randn(n) * 1e-4
            a = project_to_simplex(a)
    return best, best_score
```

"""
Propose a low‑score step function.

The algorithm works in three phases:

1️⃣  A random point on the probability simplex (Dirichlet draw) is taken as
    a start vector of length *n* (we use n=1024 – large enough to give a
    theoretical lower bound of ≈1.001 while keeping each optimisation step cheap).

2️⃣  Exponentiated sub‑gradient descent on the exact objective
    `score(a)=2·n·max(a*a)`.  
    For the current maximal convolution index *k* the sub‑gradient is
    g[i]=2·a[k‑i] (where the index is inside the array, otherwise 0).
    An iteration performs a multiplicative update
        a ← a·exp(−η·g)   (η decreasing as 1/√t)
    followed by a renormalisation onto the simplex.
    The update moves mass away from the indices that currently create the
    biggest convolution value, flattening the autocorrelation.
    The loop runs as long as time permits (≈ few hundred‑thousand steps).

3️⃣  A short stochastic hill‑climber (swap / mass‑transfer moves) is applied
    to the best vector found so far, always projecting back onto the simplex.

All work respects the 1000 s budget.  The final list of non‑negative floats is
returned for the `evaluate_sequence` harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (fast, linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a) (exactly the value returned by evaluate_sequence
    for a normalised vector).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    # `rho` is the last index where the condition holds
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Simple hill‑climber that swaps entries or moves a fraction of mass.
    The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        # re‑normalise onto the simplex
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient optimisation of the exact objective
# ----------------------------------------------------------------------


def exp_subgrad_optimize(init_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 5_000_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on   f(a)=2·n·max(a*a).
    Update rule: a ← a·exp(−η·g)   with projection onto the simplex.
    g is a sub‑gradient at the current maximal convolution shift.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution (additive form)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diminishing step size: η = η0 / sqrt(t)
        eta = eta0 / np.sqrt(t)

        # multiplicative update (always stays non‑negative)
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny jitter to help escape flat plateaus
        if t % 5000 == 0:
            a += np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score

    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nFinal hill‑climb refinement")
        polished_seq, polished_val = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=200_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Safeguard – if nothing succeeded, fall back to a tiny uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
