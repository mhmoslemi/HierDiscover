"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().

Algorithm – aggressive mirror‑descent + long simulated‑annealing
-----------------------------------------------------------------
* Multiple restarts with a variety of simple random initiators
  (Beta‑densities, Dirichlet, edge‑heavy, sparse, triangular).
* On each restart a smoothed max‑convolution is minimised by
  exponentiated‑gradient (mirror descent) on the probability simplex.
  The smoothing parameter α is increased from 0.5 up to 400,
  the first half of the iterations use a soft‑max, the second half
  uses the exact hard‑max (sub‑gradient) for a sharper descent.
* After gradient descent a long simulated‑annealing (mass‑transfer)
  phase refines the best vector found.
* The whole search respects the 1000 s time budget; the best vector
  (already normalised to sum = 1) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global placeholders – will be set inside propose_candidate
# ----------------------------------------------------------------------
_GLOBAL_N = None          # length of the sequence (set at start)

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector already normalised
    to sum = 1 (i.e. 2·n·max(conv(a,a))).
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Initialiser families (all return a vector of length _GLOBAL_N that sums to 1)
# ----------------------------------------------------------------------


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # guard against overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_beta() -> np.ndarray:
    """Beta‑shape with random parameters."""
    a = np.random.uniform(0.2, 5.0)
    b = np.random.uniform(0.2, 5.0)
    return beta_sequence(a, b, _GLOBAL_N)


def init_dirichlet() -> np.ndarray:
    """Uniform Dirichlet (i.e. i.i.d. Gamma(1))."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=_GLOBAL_N)
    return simplex_normalise(a)


def init_edge_concentrated() -> np.ndarray:
    """Linear edge‑heavy shape (large values near the two ends)."""
    idx = np.arange(_GLOBAL_N)
    dist_to_center = np.minimum(idx, _GLOBAL_N - 1 - idx)
    a = dist_to_center.astype(float) + 1.0
    return simplex_normalise(a)


def init_sparse(k: int = 10) -> np.ndarray:
    """Place mass on k random positions."""
    a = np.zeros(_GLOBAL_N, dtype=float)
    chosen = np.random.choice(_GLOBAL_N, size=min(k, _GLOBAL_N), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


def init_triangular() -> np.ndarray:
    """Symmetric triangular distribution (peak in the middle)."""
    idx = np.arange(_GLOBAL_N)
    a = np.minimum(idx + 1, _GLOBAL_N - idx)          # 1,2,…,mid,…,2,1
    return simplex_normalise(a.astype(float))


def init_random_uniform() -> np.ndarray:
    """Pure uniform start (handy for occasional blending)."""
    a = np.full(_GLOBAL_N, 1.0 / _GLOBAL_N, dtype=float)
    return a


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def mirror_descent(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   restart_id: int) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the smoothed max‑convolution.
    The first half of the iterations use a soft‑max, the second half a hard
    max (sub‑gradient).  Returns the best vector found during the run and
    its score.
    """
    n = a.shape[0]

    # schedule parameters
    eta0 = 0.7
    alpha_start = 0.5
    alpha_end = 400.0

    best_local_a = a.copy()
    best_local_score = compute_score(a)

    # pre‑compute FFT length sufficient for a * p (length 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())            # next power of two

    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break

        # 1) self‑convolution
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

        max_val = conv.max()

        # 2) probability distribution over convolution entries
        if it < max_iters // 2:                     # softened stage
            exp_vals = np.exp(alpha_start * (conv - max_val))
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                p = np.full_like(conv, 1.0 / conv.size)
            else:
                p = exp_vals / sum_exp
        else:                                        # hard max (sub‑gradient)
            mask = conv >= max_val - 1e-12
            if mask.any():
                p = mask.astype(float)
                p /= p.sum()
            else:
                p = np.full_like(conv, 1.0 / conv.size)

        # 3) gradient = cross‑correlation of a with p
        fft_a_full = np.fft.rfft(a, n=L)
        fft_p = np.fft.rfft(p, n=L)
        corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L)
        offset = conv.size - 1                     # lag 0 appears at this index
        g = corr[offset:offset + n]

        # 4) exponentiated‑gradient update on the simplex
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)

        # 5) update smoothing parameter α
        alpha_start = alpha_start + (alpha_end - alpha_start) / max_iters

        # occasional logging / best‑record
        if (it & 31) == 0:                         # every 32 iterations
            cur_score = compute_score(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_a = a.copy()
                print(f"[Descent R{restart_id:02d} it {it:4d}] "
                      f"new best {best_local_score:.6f}")

    return best_local_a, best_local_score


def simulated_annealing(a: np.ndarray,
                        deadline: float,
                        restart_id: int) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer SA that keeps the vector on the probability simplex.
    Returns the best vector seen during the SA phase.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = compute_score(cur_a)

    # start temperature as a modest fraction of the current score
    T = cur_score * 0.02
    T_end = cur_score * 1e-5
    cooling = 0.9995
    iter_counter = 0

    while time.time() < deadline - 0.5 and T > T_end:
        # pick a source (biased by current mass) and a distinct target
        src = np.random.choice(n, p=cur_a)
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        if cur_a[src] <= 0.0:
            continue

        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score_global:  # will be updated in the caller
                print(f"[SA    R{restart_id:02d}] iter {iter_counter:5d} → "
                      f"new best {cand_score:.6f}")

        T *= cooling
        iter_counter += 1

    return cur_a, cur_score

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:

    Search for a non‑negative height sequence that minimises

    # 0. Global initialisation and time bookkeeping
    # ------------------------------------------------------------------
    np.random.seed(seed)

    deadline = start_time + budget_s - 0.2          # tiny safety margin

    global _GLOBAL_N, best_score_global

    # 1. Load any previously supplied solution (height_sequence_1)
    # --------------------------------------------------------------
    try:
        height_sequence_1               # noqa: F821  (may be defined externally)
        init_vec = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        init_vec = simplex_normalise(init_vec)
        _GLOBAL_N = init_vec.size
        best_a = init_vec.copy()
        best_score_global = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1 (n={_GLOBAL_N}), score={best_score_global:.6f}")
    except NameError:
        # No prior data – fall back to a modest length (≈5 k)
        _GLOBAL_N = 5146
        best_a = np.full(_GLOBAL_N, 1.0 / _GLOBAL_N, dtype=float)
        best_score_global = compute_score(best_a)
        print(f"[Init] No previous vector – using uniform start, n={_GLOBAL_N}, "
              f"score={best_score_global:.6f}")

    n = _GLOBAL_N

    # --------------------------------------------------------------
    # 2. Collection of initialisers (all produce sum==1)
    # --------------------------------------------------------------
    init_functions = [
        init_beta,
        init_dirichlet,
        init_edge_concentrated,
        lambda: init_sparse(k=np.random.randint(5, 30)),   # use a tiny lambda wrapper only for k variability
        init_triangular,
        init_random_uniform,
    ]

    # --------------------------------------------------------------
    # 3. Main restart loop
    # --------------------------------------------------------------
    restart_id = 0
    while time.time() < deadline - 5.0:
        # ----- pick a fresh start vector -----
        init_fn = np.random.choice(init_functions)
        a = init_fn()
        # occasionally blend with the current elite (helps keep promising mass)
        if np.random.rand() < 0.25:
            a = (a + best_a) * 0.5
            a = simplex_normalise(a)

        # ----- mirror‑descent phase -----
        max_iters = 4000                      # more iterations than the reference code
        a, local_score = mirror_descent(a, max_iters, deadline, restart_id)

        if local_score < best_score_global:
            best_score_global = local_score
            best_a = a.copy()
            print(f"[Restart {restart_id:02d}] post‑descent best {best_score_global:.6f}")

        # ----- simulated‑annealing refinement -----

            best_a = a.copy()
            print(f"[Restart {restart_id:02d}] after SA new best {best_score_global:.6f}")

        restart_id += 1

    # ------------------------------------------------------------------
    # 4. Return the best vector (already normalised to sum = 1)
    # ------------------------------------------------------------------
    print(f"[Done] Completed {restart_id} restarts, best score {best_score_global:.6f}")
    return [float(v) for v in best_a.tolist()]
