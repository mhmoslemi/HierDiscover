"""
Propose a low‑score step‑function (non‑negative vector).

Algorithm
---------
1.  Exploration.
    * For each candidate length (1024, 2048, 4096, 8192) generate a
      large pool of vectors:
        - the uniform vector,
        - the classic arcsine shape,
        - symmetric/asymmetric Beta families (a_i ∝ (i+ε)^{α‑1}
          (n‑1‑i+ε)^{β‑1}),
        - random mixtures of the above.
    * Keep the best candidate (according to the fast surrogate
      2·n·max(conv(a,a))) for the current length.

2.  Multi‑stage refinement of the locally best vector.
    * Adam optimisation on a smooth log‑sum‑exp surrogate of the
      maximum convolution (λ is annealed from 1 to a large value).
    * Projected sub‑gradient descent on the *true* max‑convolution.
    * Stochastic local search (Gaussian perturbations, occasional
      uphill moves) on the probability simplex.
    * Each stage runs as long as the global time budget permits (the
      optimisation functions stop automatically when the deadline is
      reached).

3.  Global polishing.
    * The overall best vector found for any length is taken and
      refined again with the three stages, using a larger number of
      iterations because the remaining time is smaller.

All vectors are kept on the probability simplex (Σa = 1).  For a
probability vector the official ``evaluate_sequence`` value equals the
fast surrogate, therefore we never call the expensive evaluator until
the very end (once per length and once for the final polished vector).

The routine is deterministic for a given ``seed`` and respects the
total time budget (default 1000 s).  If no valid vector is found a
simple uniform fallback is returned.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, arg‑max index) of the self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)) for a already normalised
    (Σa = 1).  This equals the official evaluation value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised) – already normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # softmax weights, sum to 1
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of max

    # Weighted sub‑gradient:
    # d/da Σ_k w_k (a*a)_k = 2 * (a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 120.0,
    noise_every: int = 1200,
    noise_scale: float = 6e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # λ is annealed quadratically
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5_000 == 0:
            print(
                f"    Adam it={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   max(conv(a,a)).
    Step size decays like step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 60_000 == 0:
            print(
                f"    Sub‑grad it={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 500_000,
) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex: add Gaussian noise, project back,
    keep improvements (tiny probability of uphill moves for escape).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # deterministic acceptance of improvements,
        # occasional uphill acceptance to escape local minima
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 100_000 == 0:
            print(
                f"    Random local it={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.  The routine mixes large‑scale random
    Beta‑family sampling with three progressive optimisation stages
    (Adam → sub‑gradient → stochastic local search) and a final
    polishing of the globally best vector.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to explore
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ---- 1a. Deterministic starters ---------------------------------
        starters = [
            np.full(n, 1.0 / n),   # uniform
            arcsine_initial(n),   # arcsine
        ]

        # ---- 1b. Random Beta‑family sampling ----------------------------
        # Number of random samples adapts to remaining time.
        # Rough budget: 0.2 s per 1 000 samples for n≈2048 → we can afford a few thousand.
        max_samples = int(min(5000, max(1000, (deadline - time.time()) * 3000 / len(lengths))))
        print(f"  Sampling {max_samples} random beta shapes …")
        for _ in range(max_samples):
            alpha = rng.uniform(0.05, 1.8)
            beta = rng.uniform(0.05, 1.8)
            eps = rng.uniform(1e-8, max(0.01, 0.02 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)
            starters.append(cand)

        # ---- 1c. Simple mixtures (uniform + any starter) ---------------
        mixed = []
        for base in starters:
            for w in np.linspace(0.0, 1.0, 11):  # 0,0.1,…,1
                mixed.append(w * np.full(n, 1.0 / n) + (1 - w) * base)
        candidates = starters + mixed

        # ---- 1d. Pick the best raw candidate ----------------------------
        local_best_a = None
        local_best_score = float("inf")
        for a in candidates:
            sc = compute_score(a)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = a

        print(f"  Best raw surrogate score = {local_best_score:.6f}")

        # ------------------------------------------------------------------
        # 2.  Refinement of the locally best candidate
        # ------------------------------------------------------------------
        if local_best_a is not None and time.time() < deadline - 5.0:
            # ---- Phase A – Adam (smooth surrogate) -----------------------
            remaining = deadline - time.time()
            adam_time = min(remaining * 0.35, 120.0)          # at most 2 min
            adam_deadline = time.time() + adam_time
            print(f"  Adam optimisation (up to {adam_time:.1f}s)…")
            refined_a, _ = adam_optimize(
                initial_a=local_best_a,
                time_deadline=adam_deadline,
                max_iter=400_000,
                lr0=0.22,
                lam_start=1.0,
                lam_end=130.0,
                noise_every=1500,
                noise_scale=5e-5,
            )

            # ---- Phase B – Sub‑gradient descent -------------------------
            if time.time() < deadline - 2.0:
                sub_time = min((deadline - time.time()) * 0.30, 100.0)
                sub_deadline = time.time() + sub_time
                print(f"  Sub‑gradient optimisation (up to {sub_time:.1f}s)…")
                refined_a, _ = subgradient_optimize(
                    refined_a,
                    time_deadline=sub_deadline,
                    step0=0.22,
                    max_iter=400_000,
                )

            # ---- Phase C – Stochastic local search -----------------------
            if time.time() < deadline - 2.0:
                loc_time = deadline - time.time() - 1.5
                loc_deadline = time.time() + loc_time
                print(f"  Random local search (up to {loc_time:.1f}s)…")
                refined_a, _ = random_local_search(
                    refined_a,
                    time_deadline=loc_deadline,
                    init_step=0.018,
                    max_iter=800_000,
                )

            # ---- Official evaluation ------------------------------------
            val = evaluate_sequence(refined_a.tolist())
            print(f"  → official score for n={n}: {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:
            # If refinement never started, fall back to the raw candidate.
            if local_best_a is not None:
                val = evaluate_sequence(local_best_a.tolist())
                print(f"  Raw candidate official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = local_best_a.tolist()
                    print("*** Updated global best (raw) ***")

    # ------------------------------------------------------------------
    # 3.  Final polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Final polishing of the global best ===")
        a = np.array(best_seq, dtype=float)

        # Phase A – long Adam run
        remaining = deadline - time.time()
        adam_time = min(remaining * 0.35, 180.0)
        adam_deadline = time.time() + adam_time
        print(f"  Adam polishing (up to {adam_time:.1f}s)…")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=adam_deadline,
            max_iter=600_000,
            lr0=0.15,
            lam_start=1.0,
            lam_end=150.0,
            noise_every=1000,
            noise_scale=4e-5,
        )

        # Phase B – sub‑gradient
        if time.time() < deadline - 2.5:
            sub_time = min((deadline - time.time()) * 0.25, 120.0)
            sub_deadline = time.time() + sub_time
            print(f"  Sub‑gradient polishing (up to {sub_time:.1f}s)…")
            a, _ = subgradient_optimize(
                a,
                time_deadline=sub_deadline,
                step0=0.18,
                max_iter=500_000,
            )

        # Phase C – stochastic local search
        if time.time() < deadline - 1.0:
            loc_time = deadline - time.time() - 0.5
            loc_deadline = time.time() + loc_time
            print(f"  Random local polishing (up to {loc_time:.1f}s)…")
            a, _ = random_local_search(
                a,
                time_deadline=loc_deadline,
                init_step=0.014,
                max_iter=1_000_000,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
