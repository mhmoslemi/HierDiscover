"""
Propose a step‑function that minimizes evaluate_sequence().

Algorithm
---------
* Choose a relatively large dimension (≈1.2·10⁴).  Larger n gives
  many more degrees of freedom for flattening the self‑convolution.
* Multiple random restarts are performed.  Each restart starts from
  a simple parametric family (Dirichlet, Beta, edge‑heavy, sparse,
  or an “arcsine‑like” distribution) and is blended occasionally with
  the best solution seen so far.
* On a restart a mirror‑descent (exponentiated‑gradient) loop minimises
  the smooth approximation

        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

  of the L∞‑norm of the convolution.  The gradient of f_α is a
  cross‑correlation between the current vector and the soft‑max weights
  over the convolution entries.  The update is

        a_i ← a_i·exp(−η·g_i) ,   a ← a / Σ a

  with a step‑size η that decays like 1/√t and a smoothing parameter α
  that grows geometrically from 0.5 to 5 000.
* After the smooth phase a brief simulated‑annealing (SA) pass performs
  random mass‑transfer moves to escape any remaining local minima.
* The best vector (already normalised to sum = 1) is returned.

The whole search respects the 1000 s time budget; intermediate progress
and the best objective value (2·n·max(conv)) are printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to sum = 1 (preserves non‑negativity)."""
    s = v.sum()
    if s <= 0.0 or not np.isfinite(s):
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size, dtype=float)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of evaluate_sequence() for a vector already
    normalised to sum = 1 (i.e. 2·n·max(conv(a,a))).
    """
    n = a.shape[0]
    conv = np.fft.irfft(np.fft.rfft(a, n=2 * n - 1) * np.fft.rfft(a, n=2 * n - 1),
                       n=2 * n - 1)
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)      # guard against overflow
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.ones_like(w)
    return simplex_normalise(w)


def init_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector (i.i.d. Gamma(1))."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """Linear “edge‑heavy” shape: larger values near the two ends."""
    idx = np.arange(n, dtype=float)
    dist = np.minimum(idx, n - 1 - idx) + 1.0
    return simplex_normalise(dist)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Place mass on k randomly chosen positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    a[chosen] = np.random.rand(chosen.size) + 1e-6
    return simplex_normalise(a)


def init_arcsine_like(n: int) -> np.ndarray:
    """
    Discretised arcsine‑type density ∝ 1/√(x·(1−x)).
    This places more mass near the ends, a shape that is often useful
    for flattening the self‑convolution.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = 1.0 / np.sqrt(np.maximum(x * (1.0 - x), 1e-16))
    return simplex_normalise(w)


def init_triangular(n: int) -> np.ndarray:
    """A symmetric triangular ramp, rising to the centre and falling back."""
    idx = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    w = centre - np.abs(idx - centre) + 1.0
    return simplex_normalise(w)


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence() within ``budget_s`` seconds.
    Returns the best sequence found (as a list of floats, sum == 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2           # safety margin for finalisation

    # ------------------------------------------------------------------
    # 0. Determine problem size and initialise best‑so‑far vector
    # ------------------------------------------------------------------
    try:
        prev = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # Upscale / downscale to the target size if necessary
        target_n = max(prev.size, 12000)
        if prev.size != target_n:
            # linear interpolation to the new grid
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, target_n)
            best_a = simplex_normalise(np.interp(new_x, old_x, prev))
        else:
            best_a = simplex_normalise(prev)
        n = target_n
        best_score = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1, upscaled to n={n}, score={best_score:.6f}")
    except NameError:
        n = 12000
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_a)
        print(f"[Init] No previous sequence – using uniform start, n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Simple initialisers (all produce sum = 1)
    # ------------------------------------------------------------------
    init_funs = [
        lambda: beta_sequence(np.random.uniform(0.2, 5.0),
                              np.random.uniform(0.2, 5.0), n),
        lambda: init_dirichlet(n),
        lambda: init_edge_concentrated(n),
        lambda: init_sparse(n, k=np.random.randint(5, 30)),
        lambda: init_arcsine_like(n),
        lambda: init_triangular(n),
    ]

    # ------------------------------------------------------------------
    # 2. Optimisation helpers (no closures)
    # ------------------------------------------------------------------
    def eval_fast(v: np.ndarray) -> float:
        """Fast evaluation (assumes sum(v) == 1)."""
        return compute_score(v)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    restart_id = 0
    while time.time() < deadline - 8.0:   # keep a few seconds for clean return
        # ----- 3.1 pick a start point ---------------------------------
        a = np.random.choice(init_funs)()
        # Occasionally blend with the current best (helps retain good mass)
        if np.random.rand() < 0.15:
            a = 0.5 * a + 0.5 * best_a
            a = simplex_normalise(a)

        # ----- 3.2 mirror‑descent on the smoothed max -----------------
        max_iters = 3000
        eta0 = 0.9                      # initial step size (will decay)
        alpha_start = 0.5
        alpha_end = 5000.0
        log_alpha_ratio = math.log(alpha_end / alpha_start + 1e-30)

        # pre‑compute FFT lengths once (they depend only on n)
        conv_len = 2 * n - 1
        L_corr = 1 << ((3 * n - 2).bit_length())   # power of two ≥ 3n‑2

        for it in range(max_iters):
            if time.time() > deadline - 5.0:
                break

            # ----- convolution (self‑convolution) --------------------
            fft_a = np.fft.rfft(a, n=conv_len)
            conv = np.fft.irfft(fft_a * fft_a, n=conv_len)

            # ----- soft‑max over convolution entries -----------------
            maxc = conv.max()
            # use maxc for numerical stability
            exp_vals = np.exp(alpha_start * (conv - maxc))
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                p = np.full_like(conv, 1.0 / conv.size, dtype=float)
            else:
                p = exp_vals / sum_exp

            # ----- gradient (cross‑correlation a * p) ---------------
            fft_a_full = np.fft.rfft(a, n=L_corr)
            fft_p = np.fft.rfft(p, n=L_corr)
            corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L_corr)
            offset = conv_len - 1          # index where lag 0 appears
            g = corr[offset: offset + n]

            # ----- exponentiated‑gradient (mirror descent) ----------
            eta = eta0 / math.sqrt(it + 1.0)
            # Clip the gradient to avoid overflow in the exponent
            g = np.clip(g, -50.0, 50.0)
            a = a * np.exp(-eta * g)
            a = simplex_normalise(a)

            # ----- update smoothing parameter α (geometric schedule) -
            alpha_start = alpha_start * math.exp(log_alpha_ratio / max_iters)

            # ----- occasional progress report ------------------------
            if (it & 63) == 0:   # every 64 iterations
                cur_score = eval_fast(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Restart {restart_id}] it {it:4d} → new best {best_score:.6f}")

        # ----- 3.3 short simulated‑annealing refinement --------------
        cur_a = best_a.copy()
        cur_score = eval_fast(cur_a)

        # temperature schedule (relative to current score)
        T = cur_score * 0.03
        T_end = cur_score * 1e-5
        sa_iters = 0
        while time.time() < deadline - 2.0 and T > T_end:
            # choose a source index proportionally to its weight
            src = np.random.choice(n, p=cur_a)
            # choose a distinct target index uniformly
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1
            if cur_a[src] <= 0.0:
                continue
            delta = cur_a[src] * np.random.rand() * 0.5   # move at most 50 % of source mass
            cand = cur_a.copy()
            cand[src] -= delta
            cand[tgt] += delta

            cand_score = eval_fast(cand)
            if cand_score < cur_score:
                accept = True
            else:
                accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

            if accept:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[SA  R{restart_id}] iter {sa_iters:5d} → new best {best_score:.6f}")

            # geometric cooling
            T *= 0.9993
            sa_iters += 1

        restart_id += 1

    # ------------------------------------------------------------------
    # 4. Return the best vector (already normalised to sum = 1)
    # ------------------------------------------------------------------
    print(f"[Done] Completed {restart_id} restarts, best score = {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
