def max_conv_gradient(conv, a):
    n = a.shape[0]
    k = int(conv.argmax())
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    g = np.zeros_like(a)
    if low <= high:
        idx = np.arange(low, high+1)
        g[idx] = a[k - idx] + a[idx]
    return g
```

    a = initial_a.copy()
    max_iters = 5000
    base_eta = 0.8
    for it in range(max_iters):
        if time.time() > deadline - 0.01: break
        # compute conv
        fft_len = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[:2*n-1]

        # compute gradient

        # step size
        eta = base_eta / np.sqrt(it + 1)
        # update
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)

                best_score = cur_score
                best_a = a.copy()

def simulated_annealing(a, n, best_score):
    cur = a.copy()
    cur_score = compute_score(cur)
    T = cur_score * 0.02
    T_end = cur_score * 1e-5
    while time.time() < deadline and T > T_end:
        i = np.random.choice(n, p=cur) # heavier for large mass?
        j = np.random.randint(n)
        if i == j: continue
        delta = cur[i] * np.random.rand()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = simplex_normalise(cand) # but sum unchanged already
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < np.exp((cur_score - cand_score) / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] new best {best_score:.6f}")
        T *= 0.9995
    return best_a, best_score

"""
Optimisation of a non‑negative step‑function for the evaluation
evaluate_sequence().  The algorithm consists of two stages:

1. **Hard‑max projected gradient descent** – at each iteration the linear
   convolution a∗a is computed (FFT).  The index of the maximal entry is
   identified and the sub‑gradient of that entry is used to perform a
   multiplicative (exponentiated) update.  After the update the vector is
   projected back onto the probability simplex (non‑negative and sum = 1).
   This directly minimises the true L∞‑norm rather than a smooth
   approximation, allowing many cheap iterations.

2. **Intensive simulated‑annealing refinement** – once the gradient phase
   has stagnated a mass‑transfer Metropolis walk is run for a long time
   (temperature slowly cooled) to escape residual local minima.

Multiple random restarts (Dirichlet, Beta‑shaped, edge‑heavy and sparse
initialisers) are performed while the overall time budget (``budget_s``)
is respected.  The best vector seen so far is kept and returned.
All heavy work is performed with ``np.fft`` (O(n log n)) and the code
contains no closures or lambdas, only top‑level helper functions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale ``v`` to have sum = 1 while keeping non‑negativity."""
    s = v.sum()
    if s <= 0.0 or not np.isfinite(s):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector already normalised
    to sum = 1.
    """
    n = a.shape[0]
    # FFT length – next power of two ≥ 2*n‑1 for speed
    L = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b


def max_conv_gradient(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry w.r.t. ``a``.
    Returns a vector ``g`` such that ``g_i = ∂(a∗a)_{k_max}/∂a_i``.
    """
    n = a.shape[0]
    k = int(conv.argmax())               # index of the current maximum
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    g = np.zeros_like(a)
    if low <= high:
        idx = np.arange(low, high + 1)
        g[idx] = a[k - idx] + a[idx]     # a_{k-i} + a_i
    return g


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on ``n`` points, then normalised.
    """
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid under/overflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector (i.i.d. Gamma(1))."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """
    Linear “edge‑heavy’’ shape: larger values near the two ends,
    smaller in the centre.
    """
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = dist_to_center.astype(float) + 1.0
    return simplex_normalise(a)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Place all mass on ``k`` randomly chosen positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


def _pick_initialiser(n: int) -> np.ndarray:
    """Randomly choose one of the simple families."""
    choice = np.random.randint(4)
    if choice == 0:
        return beta_sequence(
            np.random.uniform(0.2, 5.0),
            np.random.uniform(0.2, 5.0),
            n,
        )
    if choice == 1:
        return init_dirichlet(n)
    if choice == 2:
        return init_edge_concentrated(n)
    # choice == 3
    return init_sparse(n, k=np.random.randint(5, 30))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence()`` within ``budget_s`` seconds.
    Returns the best sequence found (as a list of floats, sum ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # 0.  Load a previously known solution if it exists
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is injected by the surrounding harness.
        init_best = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        init_best = simplex_normalise(init_best)
        n = init_best.size
        best_a = init_best.copy()
        best_score = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1 (n={n}), score={best_score:.6f}")
    except NameError:
        # fallback – uniform start, length ≈5 k
        n = 5146
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_a)
        print(f"[Init] No previous sequence – using uniform start, n={n}, score={best_score:.6f}")

    # Pre‑compute FFT length for this ``n``
    FFT_LEN = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1.  Main restart loop
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 8.0:      # leave a few seconds for final return
        # ---- create a fresh start -------------------------------------------------
        a = _pick_initialiser(n)

        # occasionally blend with the current best (helps retain good mass)
        if np.random.rand() < 0.2:
            a = 0.5 * a + 0.5 * best_a
            a = simplex_normalise(a)

        # ---- hard‑max projected gradient descent ----------------------------------
        max_iters = 4000                     # enough iterations for one restart
        eta0 = 0.8                           # base step size for the multiplicative update
        for it in range(max_iters):
            now = time.time()
            if now > deadline - 4.0:
                break

            # (a) convolution via FFT
            fft_a = np.fft.rfft(a, n=FFT_LEN)
            conv = np.fft.irfft(fft_a * fft_a, n=FFT_LEN)[: 2 * n - 1]

            # (b) gradient of the current maximal entry
            g = max_conv_gradient(conv, a)

            # (c) multiplicative (exponentiated) update on the simplex
            eta = eta0 / math.sqrt(it + 1.0)
            # keep the exponent in a safe range
            delta = -eta * np.clip(g, -50.0, 50.0)
            a = a * np.exp(delta)
            a = simplex_normalise(a)

            # (d) occasional progress report
            if (it & 31) == 0:          # every 32 iterations
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(
                        f"[Restart {restart}] it {it:4d} → new best {best_score:.6f}"
                    )

        # ---- short simulated‑annealing refinement ---------------------------------
        cur_a = best_a.copy()
        cur_score = compute_score(cur_a)
        T = cur_score * 0.02          # initial temperature (small fraction of score)
        T_end = cur_score * 1e-5
        while time.time() < deadline - 2.0 and T > T_end:
            # pick a source proportionally to its mass (more likely to move large mass)
            src = np.random.choice(n, p=cur_a)
            # pick a distinct target uniformly
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1
            if cur_a[src] <= 0.0:
                continue
            delta = cur_a[src] * np.random.rand()  # amount to transfer
            cand = cur_a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = simplex_normalise(cand)        # renormalise (keeps sum = 1)

            cand_score = compute_score(cand)
            if cand_score < cur_score:
                accept = True
            else:
                accept = np.random.rand() < math.exp(
                    -(cand_score - cur_score) / (T + 1e-12)
                )
            if accept:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(
                        f"[SA  R{restart}] iter {int(T*1e5):5d} → new best {best_score:.6f}"
                    )
            # geometric cooling
            T *= 0.9995

        restart += 1

    # ------------------------------------------------------------------
    # 2.  Return the best vector (already normalised to sum ≈ 1)
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {restart} restarts: {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
