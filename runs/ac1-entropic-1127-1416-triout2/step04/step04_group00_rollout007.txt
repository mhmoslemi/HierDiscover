def annealed_gradient_descent(a: np.ndarray, deadline: float):
    a = a.copy()
    n = a.shape[0]
    # start with a as probability simplex (its sum may be 1)
    # ensure sum=1
    a = a / np.sum(a) if np.sum(a) > 0 else np.full(n, 1.0/n)
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    tau = 1.0
    while time.time() < deadline:
        t += 1
        # increase tau every few iterations
        if t % 5000 == 0:
            tau = min(tau * 1.5, 200.0)
        # compute conv
        conv = convolution_fft(a, a)
        # compute softmax weights: p_i = exp(tau*conv_i)
        # subtract max for numeric stability
        maxc = np.max(conv)
        exps = np.exp(tau * (conv - maxc))
        p = exps / np.sum(exps)
        # gradient: g = 2 * correlation(p, a)
        a_rev = a[::-1]
        # compute convolution p * a_rev (full)
        conv_pg = convolution_fft(p, a_rev)  # returns length len(p)+len(a_rev)-1
        # slice appropriate indices: g_i = 2 * conv_pg[n-1 + i]
        g = 2.0 * conv_pg[n-1:n-1+len(a)]
        # step size: eta = 0.1 / sqrt(t)
        eta = 0.1 / np.sqrt(t)
        a -= eta * g
        # projection to simplex
        a = project_to_simplex(a)
        # evaluate
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a

"""
Propose a step‑function (list of non‑negative heights) with a very low
`evaluate_sequence` score.

The algorithm consists of three phases, all respecting the 1000 s budget.

1. **Large‑scale random sparse search**  
   For several powers‑of‑two lengths up to 65536 we repeatedly draw a
   random subset of indices of moderate density (≈ 0.4‑0.6) and assign
   equal mass 1/k to the selected positions.  For a vector a with
   Σa=1 the score is `2·n·max(a∗a)`.  With large n the random
   distribution of pairwise sums is almost flat, giving an expected
   score only slightly above 1.  The fast FFT‑based evaluation allows
   tens of thousands of such trials within the budget.

2. **Annealed soft‑max gradient descent** (optional)  
   Starting from the best random vector we minimise a smooth
   approximation of the objective:
       Jτ(a)= (1/τ)·log Σ exp(τ·conv[a]),   conv[a]=a∗a .
   The gradient is `g = 2·corr(p, a)` where `p` are the soft‑max
   weights of the convolution.  After each step the vector is projected
   back onto the probability simplex (non‑negative, unit sum).  The
   temperature τ is increased gradually (annealing) so the method first
   follows a smooth landscape and finally concentrates on the true max.

3. **Final hill‑climb polish**  
   A stochastic hill‑climber that swaps entries or moves a fraction of
   mass between two positions (projected onto the simplex) is run for a
   few hundred thousand cheap steps, guaranteeing a monotone improvement
   when possible.

If no candidate is generated the method falls back to a uniform 256‑size
vector, which is always valid.  The returned list is ready for the
`evaluate_sequence` harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of the official `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a∗a), i.e. the exact score used by the harness.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Hill‑climber used for the final polishing stage
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """
    Stochastic hill‑climber that swaps entries or moves a fraction of mass.
    The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap of two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0:
                        cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Annealed soft‑max gradient descent (phase 2)
# ----------------------------------------------------------------------


def softmax_gradient_descent(init_a: np.ndarray,
                             time_deadline: float,
                             tau_start: float = 1.0,
                             tau_end: float = 200.0,
                             step0: float = 0.12) -> np.ndarray:
    """
    Minimise the smoothed objective Jτ(a) = (1/τ)·log Σ exp(τ·(a∗a))
    by projected gradient descent with a slowly increasing temperature τ.
    The vector is always kept on the probability simplex.
    """
    a = init_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    tau = tau_start
    tau_inc = (tau_end / tau_start) ** (1.0 / 5000)  # geometric growth

    while time.time() < time_deadline:
        t += 1
        # increase temperature every few iterations (geometric schedule)
        if t % 200 == 0:
            tau = min(tau * tau_inc, tau_end)

        # 1) a∗a
        conv = convolution_fft(a, a)          # length 2n‑1
        # 2) soft‑max weights
        maxc = np.max(conv)
        exps = np.exp(tau * (conv - maxc))
        p = exps / np.sum(exps)               # length 2n‑1, sums to 1

        # 3) gradient = 2·corr(p, a)
        a_rev = a[::-1]
        corr = convolution_fft(p, a_rev)      # length 3n‑2
        g = 2.0 * corr[n - 1:n - 1 + n]        # slice to length n

        # 4) step (diminishing)
        step = step0 / np.sqrt(t)
        a = a - step * g

        # 5) projection onto simplex
        a = project_to_simplex(a)

        # 6) keep the best (true) score seen so far
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a tiny safety margin for the harness

    try:

    # ------------------------------------------------------------------

    lengths = [4096, 8192, 16384, 32768, 65536]

        a = np.zeros(n, dtype=float)

        cur_val = compute_score(a)
        if cur_val < best_val:

        # occasional early exit if we already beat the target comfortably

            best_val = refined_val
            print(f"Hill‑climb final best = {best_val:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:

    # Final evaluation using the official harness (for logging)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
