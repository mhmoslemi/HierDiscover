def propose_candidate(budget_s=1000):
    import numpy as np, time
    from scipy.optimize import linprog

    rng = np.random.default_rng()
    deadline = time.time() + budget_s - 0.5

    # Try to get previous best if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        n = prev.shape[0]
    except Exception:
        n = 4000  # choose moderate length
        prev = None

    # Helper functions
    def full_conv(a):
        L = 1 << ((2*n-2).bit_length())
        fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fft * fft, n=L)
        return conv[:2*n-1]

    def score_of(a):
        conv = full_conv(a)
        max_b = conv.max()
        S = a.sum()
        if S < 0.01:
            return np.inf
        return 2.0 * n * max_b / (S*S)

    # Stage: random restarts with gradient descent (Adam)
    best_a = None
    best_score = float('inf')
    # use initial distribution: random Dirichlet
    while time.time() < deadline - 30:
        # initialize
        a0 = rng.dirichlet(alpha=np.full(n, 0.5))
        # run Adam for limited time (maybe 10 seconds)
        t_end = min(deadline - 5, time.time() + 8)
        a_opt, sc_opt = _adam_optimize(a0, t_end, restart_id=0, target=1.48)  # target lower as early break, but we set >? not needed
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"Initial Adam best {best_score:.6f}")

    if best_a is None:
        if prev is not None:
            best_a = prev.copy()
        else:
            best_a = np.full(n, 1.0 / n)
        best_score = score_of(best_a)

    # Simulated annealing / permutation search
    # define temperature schedule
    T0 = (best_score - (2.0*n/(2*n-1))) * 0.5  # half of excess over lower bound
    if T0 <= 0: T0 = 1e-3
    T = T0
    decay = 0.9995

    # precompute convolution for best_a
    cur_a = best_a.copy()
    cur_score = best_score

    it = 0
    while time.time() < deadline - 0.5:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        a_new = cur_a.copy()
        a_new[i], a_new[j] = a_new[j], a_new[i]
        new_score = score_of(a_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"Iter {it:07d}: new best {best_score:.6f}")

        # occasional small mass transfer
        if rng.random() < 0.1:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_a[i] > 1e-12:
                delta_val = rng.random() * 0.01 * cur_a[i]  # move up to 1% of i's mass
                a_new = cur_a.copy()
                a_new[i] -= delta_val
                a_new[j] += delta_val
                new_score = score_of(a_new)
                delta = new_score - cur_score
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                    cur_a = a_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = cur_a.copy()
                        print(f"Iter {it:07d} transfer: new best {best_score:.6f}")

        # temperature decay
        T *= decay

    # final output: ensure all values nonnegative and clip to <=1000
    best_a = np.clip(best_a, 0.0, 1000.0)
    # renormalize to avoid near-zero sum
    if best_a.sum() < 0.01:
        best_a = np.full_like(best_a, 1.0 / best_a.size)
    return [float(v) for v in best_a.tolist()]

"""
Two‑stage optimisation with a final simulated‑annealing polishing step.

* Stage 1 – smooth optimisation of the log‑sum‑exp surrogate of the
  maximum self‑convolution, performed on the probability simplex via
  Euclidean projection.

* Stage 2 – Adam optimiser on the true (nonsmooth) objective using the
  sub‑gradient of the maximum convolution entry.

* Stage 3 – a lightweight simulated‑annealing / swap‑based local search
  that attempts to reduce the maximum convolution by re‑ordering mass
  and by tiny mass‑transfer moves.

The routine runs many short restarts (different random initialisations)
until the time budget expires and returns the best sequence found.
All numbers are non‑negative, clipped to the evaluator limits, and the
returned object is a plain ``list`` of ``float``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution of a using FFT length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of the weight vector w (length 2·n‑1) with a (length n).
    Returns the n‑length segment that corresponds to shifts k=0…n‑1.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Stage 1 – smooth optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_optimize(init_a: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Gradient descent on the simplex minimizing the smooth surrogate."""
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())   # next power of two
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()
    # annealing schedule
    alpha = 0.5
    alpha_mul = 1.12
    inc_every = 250
    it = 0
    stagn = 0
    while time.time() < deadline:
        it += 1
        conv = _full_conv(a, L)                    # 2·n‑1 values
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)   # stabilised soft‑max
        w = exp_scaled / exp_scaled.sum()          # Σ w = 1

        corr = _corr_w_a(w, a, L)                  # length n
        grad = 4.0 * n * corr

        lr = 0.7 / (1.0 + it * 0.001)              # gently decreasing
        a -= lr * grad
        a = _project_simplex(a)

        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"Smooth  iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if stagn > 3000:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 4000.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Max‑convolution and its sub‑gradient (exact, not smoothed)
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max convolution value, sub‑gradient) for aperiodic self‑conv."""
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Stage 2 – Adam optimisation on the true objective
# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   deadline: float,
                   restart_id: int,
                   target: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the logits of a soft‑max representation.
    Returns the best (a, score) seen before the deadline.
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # logits (log‑softmax) – small epsilon prevents log(0)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    a0 = np.exp(b - b.max())
    a0 /= a0.sum()
    max_val, _ = _maxconv_and_subgrad(a0, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if best_score < target:
            break
        if stagn > 6000:
            b += 0.01 * np.random.randn(n)
            stagn = 0

        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: exact evaluation (identical to the supplied evaluator but without
# the security checks).  Used by the local annealing stage.
# ----------------------------------------------------------------------
def _eval_score(a: np.ndarray, L: int) -> float:
    conv = np.fft.irfft(np.fft.rfft(a, n=L) *
                        np.fft.rfft(a, n=L), n=L)[: 2 * a.shape[0] - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * a.shape[0] * max_b / (s * s)


# ----------------------------------------------------------------------
# Stage 3 – very cheap simulated‑annealing / swap‑based polishing
# ----------------------------------------------------------------------
def _local_anneal(a: np.ndarray,
                  deadline: float,
                  init_temp: float = 0.2,
                  rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Simple annealing that swaps entries or transfers a small amount of mass.
    Returns the best sequence encountered before the deadline.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    cur = a.copy()
    cur_score = _eval_score(cur, L)
    best = cur.copy()
    best_score = cur_score
    T = init_temp
    decay = 0.9999

    while time.time() < deadline:
        # ----- swap two positions -------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i != j:
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
            cand_score = _eval_score(cand, L)
            delta = cand_score - cur_score
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best = cur.copy()
                    best_score = cur_score
                    print(f"Anneal iter swap: new best {best_score:.6f}")

        # ----- tiny mass transfer between two positions ----------------------------
        if rng.random() < 0.1:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur[i] > 1e-12:
                amt = cur[i] * rng.random() * 0.02   # move ≤ 2 % of i's mass
                cand = cur.copy()
                cand[i] -= amt
                cand[j] += amt
                cand_score = _eval_score(cand, L)
                delta = cand_score - cur_score
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                    cur = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best = cur.copy()
                        best_score = cur_score
                        print(f"Anneal iter transfer: new best {best_score:.6f}")

        T *= decay
    return best, best_score
