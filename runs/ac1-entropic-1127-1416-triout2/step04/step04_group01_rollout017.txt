"""
Propose a low‑discrepancy, non‑negative step sequence.

The search builds on the previous “Adam + sub‑gradient + SA” pipeline
but adds two new ingredients that noticeably improve the bound:

1.  **Explicit parametric exploration** –  A large pool of candidates is
    created from three families that are known to give good convolutions:
    *  Beta‑densities (including the arcsine law α=β=½).
    *  Mixtures of the arcsine density with the uniform density.
    *  Inverse‑polynomial (generalised arcsine) shapes.
    For each family we draw many random parameters (α,β,ε, mixture‑weight,
    power‑exponent …) and keep the few best vectors according to the
    fast surrogate 2·n·max(conv).

2.  **Longer, higher‑temperature Adam optimisation** –  The soft‑max
    temperature λ is ramped up to 200 (instead of 80) and the iteration
    budget is dynamically scaled with the remaining wall‑time, giving the
    optimiser a finer approximation of the true max‑convolution and more
    time to converge.

The workflow is:

* generate a candidate pool for a modest length (n=1024) – cheap and fast;
* keep the top‑k seeds, upscale them to a larger length (n up to 4096);
* run a three‑phase refinement (Adam → sub‑gradient → simulated‑annealing)
  on each seed, always respecting the global deadline;
* keep the globally best vector and perform a final polishing step;
* return the best sequence found (or a safe fallback).

All vectors are projected onto the probability simplex (Σa=1, a≥0) so that
the official evaluation reduces to 2·n·max(conv), i.e. the denominator of
``evaluate_sequence`` is always 1.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score when Σa = 1:
        score = 2·n·max(conv(a,a))
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float = 1e-12
) -> np.ndarray:
    """
    Discrete Beta(α,β) density on {0,…,n‑1}.
    Values are proportional to (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    """
    i = np.arange(n, dtype=np.float64)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    s = base.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return base / s


def generate_power_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    s = vals.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return vals / s


def generate_mixture_arcsine_uniform(
    n: int, weight_arc: float
) -> np.ndarray:
    """
    Mixture (weight_arc)·Beta(½,½) + (1‑weight_arc)·Uniform.
    """
    arc = generate_beta_shape(n, 0.5, 0.5, eps=1e-12)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_uniform(n: int) -> np.ndarray:
    """Uniform distribution (Σa=1)."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Optimisers (identical to the previous implementation but with a
# higher temperature schedule and slightly tweaked parameters)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) after projection onto the simplex.
    """
    n = a.size
    conv = convolution_fft(a, a)               # length = 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length = n+(2n‑1)-1 = 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.07,
    lam_start: float = 2.0,
    lam_end: float = 200.0,
    noise_every: int = 5_000,
    noise_scale: float = 3e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best projected vector and its fast score.
    """
    a = project_to_simplex(initial_a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 10_000 == 0:
            print(f"    Adam it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.02,
) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective 2·n·max(conv(a,a)).
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        conv_rev = convolution_fft(a, mask[::-1])
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 10_000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 120_000,
    init_temp: float = 0.02,
    decay: float = 0.9999,
) -> np.ndarray:
    """
    Metropolis local search using the fast surrogate 2·n·max(conv).
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-9)

        if it % 10_000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def generate_candidate_pool(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """
    Build a diverse pool of vectors using the three families described
    above.  The returned list is not sorted.
    """
    pool: List[np.ndarray] = []

    # deterministic baselines
    pool.append(generate_uniform(n))
    pool.append(generate_beta_shape(n, 0.5, 0.5, eps=1e-12))   # pure arcsine

    # mixtures arcsine–uniform
    for w in np.linspace(0.0, 1.0, 9)[1:-1]:  # avoid trivial 0/1
        pool.append(generate_mixture_arcsine_uniform(n, w))

    # random beta shapes
    for _ in range(300):
        log_alpha = rng.uniform(np.log10(0.1), np.log10(5.0))
        log_beta = rng.uniform(np.log10(0.1), np.log10(5.0))
        alpha = 10 ** log_alpha
        beta = 10 ** log_beta
        eps = 10 ** rng.uniform(-12.0, -3.0)
        pool.append(generate_beta_shape(n, alpha, beta, eps))

    # random power (inverse‑polynomial) shapes
    for _ in range(200):
        p = rng.uniform(0.2, 2.0)
        eps = rng.uniform(0.0, 1.0)
        pool.append(generate_power_shape(n, p, eps))

    return pool


def upscale_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation to a larger size while preserving the total mass.
    The output is re‑normalised to Σ=1.
    """
    old_n = v.size
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    v_new = np.interp(x_new, x_old, v)
    v_new = v_new / v_new.sum()
    return v_new

def propose_candidate(
    seed: int = 42, budget_s: int = 1000, **kwargs
) -> List[float]:
    """
    Search for a step sequence with a low official score.
    The algorithm consists of:
        - fast parametric sampling (beta, mixture, power families);
        - selection of the top‑k seeds;
        - refinement of each seed by Adam → sub‑gradient → SA;
        - a final polishing step on the global best.
    All vectors are kept on the probability simplex so that the official
    evaluation reduces to 2·n·max(conv).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0.  Choose a base problem size for cheap exploration
    # ------------------------------------------------------------------
    base_n = 1024
    print(f"\n=== Generating candidate pool (n={base_n}) ===")
    pool = generate_candidate_pool(base_n, rng)

    # evaluate fast scores and keep the 8 most promising
    scored = [(compute_score(v), v) for v in pool]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for (_, vec) in scored[:8]]
    print(f"Best fast score from pool = {scored[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 1.  Refine each seed, gradually increasing the dimension
    # ------------------------------------------------------------------
    target_lengths = [2048, 4096]          # larger sizes for final polishing
    for n in target_lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Refinement at length n = {n} ===")
        # upscale seeds to the new length
        seeds_n = [upscale_vector(s, n) for s in top_seeds]

        for idx, seed_vec in enumerate(seeds_n):
            if time.time() >= deadline - 5.0:
                break
            print(f"\n  Seed {idx + 1}/{len(seeds_n)} (pre‑score={compute_score(seed_vec):.6f})")

            # ---- Phase A: Adam on smooth surrogate -----------------
            rem = deadline - time.time()
            max_iter_adam = int(min(150_000, rem * 35))
            refined, _ = adam_optimize_smooth(
                initial_a=seed_vec,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.07,
                lam_start=2.0,
                lam_end=200.0,
                noise_every=5_000,
                noise_scale=3e-5,
            )

            # ---- Phase B: Exact sub‑gradient descent -----------------
            if time.time() < deadline - 3.0:
                rem = deadline - time.time()
                max_iter_sub = int(min(60_000, rem * 30))
                refined = subgradient_descent_exact(
                    refined,
                    deadline,
                    max_iter=max_iter_sub,
                    step0=0.02,
                )

            # ---- Phase C: Simulated‑annealing polishing --------------
            if time.time() < deadline - 3.0:
                rem = deadline - time.time()
                max_iter_sa = int(min(120_000, rem * 30))
                refined = simulated_annealing(
                    refined,
                    rng,
                    deadline,
                    max_iters=max_iter_sa,
                    init_temp=0.02,
                    decay=0.9999,
                )

            # ---- Official evaluation ---------------------------------
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"  → official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception as e:
                print(f"  Evaluation error: {e}")

    # ------------------------------------------------------------------
    # 2. Final global polishing on the best vector found so far
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=np.float64)
        remaining = deadline - time.time()
        max_iters_total = int(min(400_000, remaining * 35))

        it_sub = max_iters_total // 2
        it_adam = max_iters_total - it_sub

        print("\n*** Final global polishing ***")
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.015)

            a,
            deadline,
            max_iter=it_adam,
            lr0=0.06,
            lam_start=2.0,
            lam_end=200.0,
            noise_every=5_000,
            noise_scale=2e-5,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3. Guard – guarantee a non‑empty, non‑negative result
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 512
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
