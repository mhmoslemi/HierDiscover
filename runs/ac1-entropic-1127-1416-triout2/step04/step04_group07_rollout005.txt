def _objective_and_grad_logits(b, alpha):
    a = np.exp(b - np.max(b))
    a_sum = np.sum(a)
    a = a / a_sum

    # FFT length
    L = 1 << ((2*n - 2).bit_length())
    conv = _full_conv(a, L)   # length 2n-1
    # stable log-sum-exp
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    M_alpha = (max_scaled + np.log(sum_exp)) / alpha  # because M = (1/α) log Σ exp(α conv) = (1/α)*(max_scaled + log sum_exp)
    # objective
    obj = 2.0 * n * M_alpha

    w = exp_scaled / sum_exp  # weights sum to 1
    corr = _corr_w_a(w, a, L)   # length n
    grad_a = 4.0 * n * corr

    # convert to gradient w.r.t b:
    avg = np.dot(a, grad_a)    # sum_i a_i * grad_a_i
    grad_b = a * (grad_a - avg)

    return obj, grad_b
```

"""
Propose a low‑score step‑function.

The algorithm mixes three ideas:

1. **Soft‑max (log‑sum‑exp) surrogate** – a smooth approximation
   `M_α(a) = (1/α)·log Σexp(α·conv(a,a))` of the true maximum
   of the aperiodic self‑convolution.  Its gradient can be written
   as a correlation of the soft‑max weights with the current vector.

2. **Continuation L‑BFGS optimisation** – we repeatedly minimise
   the surrogate on the probability simplex using the log‑softmax
   parametrisation `a = softmax(b)`.  Starting from a low temperature
   (`α≈0.5`) and increasing it geometrically (up to `α≈2·10⁴`) drives the
   solution towards a true minimiser of the max‑convolution while keeping
   the optimisation well‑behaved.

3. **Multi‑restart & cheap post‑processing** – a few diverse initial
   points (uniform, Dirichlet, geometric, optionally the previous best)
   are tried until the time budget runs out.  After the L‑BFGS phase a
   short projected‑gradient refinement is performed.

All helper functions are top‑level; there are no closures.
The returned list contains only non‑negative floats, ready for
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution of a (length 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of weight vector w (length 2·n‑1) with a (length n).
    Returns the n‑vector needed for the gradient of the surrogate.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp) together with its gradient w.r.t. logits
# ----------------------------------------------------------------------
def _obj_and_grad_logits(
    b: np.ndarray, alpha: float, n: int, L: int
) -> tuple[float, np.ndarray]:
    """
    Given logits b (a = softmax(b)), return
        f(b) = 2·n·M_α(a)     and   ∇_b f(b)

    where M_α(a) = (1/α)·log Σ_k exp(α·conv_k(a,a)).
    """
    # ----- softmax -----
    b_max = np.max(b)
    a = np.exp(b - b_max)
    a_sum = a.sum()
    a = a / a_sum

    # ----- convolution & smooth max -----
    conv = _full_conv(a, L)                         # length 2·n‑1
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)        # stable
    sum_exp = np.sum(exp_scaled)

    # smooth maximum M_α
    M_alpha = (max_scaled + np.log(sum_exp)) / alpha
    f = 2.0 * n * M_alpha

    # ----- gradient of smooth max wrt a -----
    w = exp_scaled / sum_exp                         # soft‑max weights (Σ w = 1)
    corr = _corr_w_a(w, a, L)                        # length n
    grad_a = 4.0 * n * corr                          # ∇_a (2·n·M_α)

    # ----- chain rule through softmax -----
    avg = np.dot(a, grad_a)
    grad_b = a * (grad_a - avg)                      # ∇_b f
    return f, grad_b


# ----------------------------------------------------------------------
# Simple projected‑gradient refinement (optional)
# ----------------------------------------------------------------------
def _refine_by_projgrad(a: np.ndarray, deadline: float) -> np.ndarray:
    """A few hundred projected‑gradient steps on the true max‑convolution."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    it = 0
    while time.time() < deadline and it < 5000:
        it += 1
        conv = _full_conv(a, L)
        max_val = conv.max()
        # tolerance for active set
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient of max(conv)
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        if active.size > 1:
            grad /= float(active.size)

        # simple diminishing step size
        lr = 0.7 / (1.0 + it * 0.001)
        a -= lr * grad
        a = _project_simplex(a)
    return a

# ----------------------------------------------------------------------

    """

    deadline = time.time() + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (use the size of the previously known best
    # if available; otherwise fall back to a moderate default).
    # ------------------------------------------------------------------
    try:

        n = previous.shape[0]
        have_prev = True
    except Exception:
        previous = None
        n = 5146                                          # default length from earlier runs
        have_prev = False

    L = 1 << ((2 * n - 2).bit_length())                  # FFT length (next power of two)

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    restart_id = 0
    # a small pool of cheap initialisers
    strategies = []
    if have_prev:
        strategies.append("prev")
    strategies.extend(["uniform", "random_dirichlet", "geometric"])

    # temperature schedule for the smooth surrogate
    alphas = [0.5, 5.0, 50.0, 500.0, 5000.0, 20000.0]

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:
        kind = strategies[restart_id % len(strategies)]

        # ----- generate a feasible start point (probability vector) -----
        if kind == "prev" and have_prev:
            a0 = previous.copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        elif kind == "uniform":

        elif kind == "random_dirichlet":

            right = r ** np.arange(n - 1, -1, -1)
            a0 = left + right
            a0 = a0 / a0.sum()
        else:

        for alpha in alphas:

            res = minimize(

            conv = _full_conv(a, L)
            max_conv = conv.max()
            score = 2.0 * n * max_conv

            best_a = a.copy()
            print(f"Restart {restart_id:02d}: new best = {best_score:.6f}")

        restart_id += 1

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return value
    # ------------------------------------------------------------------

    # Return as plain Python list (the evaluator expects this)
    return [float(v) for v in best_a.tolist()]
