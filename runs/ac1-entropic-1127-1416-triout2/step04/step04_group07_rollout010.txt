"""
Proposes a step function sequence minimizing
    2·n·max(conv(a,a)) / (sum(a)**2)

The optimizer combines:

1. Warm‑start candidates (previous best, uniform, Dirichlet,
   geometric, arcsine).
2. A smooth log‑sum‑exp surrogate minimised by L‑BFGS‑B on the
   soft‑max logits (simplex handled analytically).
   The temperature α is increased geometrically.
3. A cheap sub‑gradient balance step that moves probability mass
   from indices with large sub‑gradient to those with small
   sub‑gradient, directly reducing the true max‑convolution.
4. A short Adam polishing phase (as in the reference code).

The routine runs until the given time budget expires and returns the
best sequence discovered as a plain list of floats.
"""
import time
import numpy as np
import scipy.optimize

# ---------- helper functions ----------
def _project_simplex(v):
    # Euclidean projection onto probability simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _full_conv(a, L):
    """Aperiodic self‑convolution via FFT (length 2·len(a)-1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]

def _corr_w_a(w, a, L):
    """Correlation of w (length 2·n-1) with a (length n)."""
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]

def _softmax(b):
    # stable softmax returning a vector that sums to 1
    b_max = np.max(b)
    e = np.exp(b - b_max)
    e_sum = e.sum()
    return e / e_sum

def _smooth_obj_and_grad(b, alpha, L, n):
    """
    Returns (objective, gradient) for the log‑sum‑exp surrogate.
    The variables are the logits b (no constraints).  The
    distribution a = softmax(b) lies on the simplex.
    """
    a = _softmax(b)
    conv = _full_conv(a, L)
    # compute soft‑max over α·conv
    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    M_alpha = (max_scaled + np.log(sum_exp)) / alpha   # (= (1/α) log Σ exp(α conv))
    obj = 2.0 * n * M_alpha

    # weights of the smooth max
    w = exp_scaled / sum_exp
    # correlation w ⋆ a
    corr = _corr_w_a(w, a, L)
    # gradient wrt a (see derivation in analysis)
    grad_a = 2.0 * n * (corr + a)          # shape (n,)

    # chain rule through soft‑max
    avg = np.dot(a, grad_a)
    grad_b = a * (grad_a - avg)           # shape (n,)
    return obj, grad_b.astype(np.float64)

def _l_bfgs_optimize(initial_a, alphas, deadline, rng):
    """
    Runs L‑BFGS‑B on the smooth surrogate for a sequence of temperatures.
    Returns the best feasible vector (as a probability vector) found.
    """
    n = initial_a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    # start from logits of the initial distribution
    b = np.log(initial_a + 1e-12)

    best_a = initial_a.copy()
    best_score = float('inf')
    # callback for early stopping when time runs out
    def _make_callback():
        start = time.time()
        def cb(xk):
            if time.time() > deadline:
                raise StopIteration
        return cb
    callback = _make_callback()

    # iterate over temperature schedule
    for alpha in alphas:
        if time.time() > deadline:
            break
        # wrapper required by scipy (func returns f, grad)
        def func(b_vec):
            return _smooth_obj_and_grad(b_vec, alpha, L, n)
        # limit the number of internal L‑BFGS iterations to keep within budget
        maxiter = 150
        try:
            b_opt, f_opt, d = scipy.optimize.fmin_l_bfgs_b(
                func, b, maxiter=maxiter, disp=False, callback=callback
            )
        except StopIteration:
            break
        # evaluate the exact (non‑smoothed) objective
        a_opt = _softmax(b_opt)
        # we evaluate with the provided evaluate_sequence for reliability
        try:
            score = evaluate_sequence(a_opt.tolist())
        except Exception:
            # fallback: compute directly
            conv = _full_conv(a_opt, L)
            score = 2.0 * n * conv.max() / (np.sum(a_opt) ** 2)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
    return best_a, best_score

def _subgrad_balance(a, L, deadline, max_iters=20000, step_frac=0.5):
    """
    Simple coordinate‑wise mass‑transfer scheme using the true sub‑gradient.
    Moves probability mass from the index with maximal sub‑gradient to the
    index with minimal sub‑gradient.  Stops when the sub‑gradient becomes
    (nearly) flat or when time expires.
    """
    n = a.shape[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _full_conv(a, L)
        max_val = conv.max()
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]
        if active.size == 0:
            break
        # build normalized mask w
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        # sub‑gradient = 2·(w ⋆ a)
        corr = _corr_w_a(w, a, L)
        grad = 2.0 * corr

        i_max = np.argmax(grad)
        i_min = np.argmin(grad)
        # termination when all components are almost equal
        if grad[i_max] - grad[i_min] <= 1e-12:
            break
        delta = min(a[i_max], step_frac * a[i_max])
        if delta <= 0:
            break
        a[i_max] -= delta
        a[i_min] += delta
    return a

def _adam_optimize(init_a, deadline, rng, target=1.5):
    # Re‑use the Adam routine from the reference implementation
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # logits (log‑softmax) – small epsilon prevents log(0)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # initial evaluation
    a0 = np.exp(b - b.max())
    a0 = a0 / a0.sum()
    fft_a = np.fft.rfft(a0, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # probabilities from logits
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        # full convolution
        fft_a = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
        else:
            stagn += 1

        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * rng.standard_normal(n)   # log‑space jitter
            stagn = 0

        # sub‑gradient of the true max‑convolution (averaged over active shifts)
        # reuse the simple routine from the reference
        # compute active mask
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        # sub‑gradient = 2·(w ⋆ a)
        corr = _corr_w_a(w, a, fft_len)
        subgrad_a = 2.0 * corr

        grad_obj = 2.0 * n * subgrad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score

    """
    rng = np.random.default_rng(123456)
    # keep a small safety margin for the harness
    deadline = time.time() + budget_s - 1.0

"""
Proposes a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser works in three phases:

1. **Warm‑starts** – several deterministic or random initial vectors
   (previous best, uniform, Dirichlet, geometric, arcsine).

2. **Smooth optimisation** – minimises a log‑sum‑exp surrogate of the
   max‑convolution with L‑BFGS‑B on the soft‑max logits.  The temperature
   α is increased geometrically (0.5 → 2048) so that the surrogate
   approaches the true max.

3. **Fine‑tuning** – a cheap sub‑gradient “mass‑balance” step that moves
   probability mass from the index with the largest sub‑gradient to the
   one with the smallest, directly decreasing the true max‑convolution.
   Afterwards a short Adam polish (as in the reference code) is applied.

The routine runs until the supplied time budget expires (default 1000 s)
and returns the best vector found as a plain list of floats.  All
vectors lie on the probability simplex, are non‑negative and sum to 1,
so the evaluation is scale‑invariant.
"""

import time
import numpy as np
import scipy.optimize  # type: ignore


# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """Correlation of w (length 2·n‑1) with a (length n)."""
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


def _softmax(b: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (sums to 1)."""
    b_max = np.max(b)
    e = np.exp(b - b_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) objective and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(b: np.ndarray, alpha: float, L: int, n: int):
    """
    Returns (objective, gradient) for the log‑sum‑exp surrogate.
    Variables are the logits b (unconstrained).  The simplex vector is
    a = softmax(b).  The objective is 2·n·M_α(a) where
        M_α = (1/α)·log Σ exp(α·conv(a,a)).
    """
    a = _softmax(b)
    conv = _full_conv(a, L)

    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    M_alpha = (max_scaled + np.log(sum_exp)) / alpha
    obj = 2.0 * n * M_alpha

    w = exp_scaled / sum_exp                     # soft‑max weights
    corr = _corr_w_a(w, a, L)                    # correlation w⋆a, length n
    grad_a = 2.0 * n * (corr + a)                # ∂obj/∂a

    avg = np.dot(a, grad_a)
    grad_b = a * (grad_a - avg)                  # chain rule through soft‑max
    return obj, grad_b.astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – L‑BFGS on the smooth surrogate (temperature schedule)
# ----------------------------------------------------------------------
def _l_bfgs_optimize(init_a: np.ndarray,
                     alphas: list[float],
                     deadline: float,
                     rng: np.random.Generator):
    """
    Runs L‑BFGS‑B for a sequence of temperatures (alphas).
    Returns the best probability vector found and its exact (non‑smooth)
    score as measured by ``evaluate_sequence``.
    """
    n = init_a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    b = np.log(init_a + 1e-12)   # logits of the starting distribution

    best_a = init_a.copy()
    best_score = float("inf")

    def _callback(xk):
        if time.time() > deadline:
            # raising StopIteration aborts the optimiser gracefully
            raise StopIteration

    for alpha in alphas:
        if time.time() > deadline:
            break

        def func(b_vec):
            return _smooth_obj_and_grad(b_vec, alpha, L, n)

        # a modest cap on internal iterations keeps us inside the time budget
        maxiter = 150
        try:
            b_opt, f_opt, _ = scipy.optimize.fmin_l_bfgs_b(
                func, b, maxiter=maxiter, disp=False, callback=_callback
            )
        except StopIteration:
            break

        a_opt = _softmax(b_opt)

        # exact (non‑smooth) evaluation
        try:
            score = evaluate_sequence(a_opt.tolist())
        except Exception:  # fallback if ``evaluate_sequence`` is unavailable
            conv = _full_conv(a_opt, L)
            score = 2.0 * n * conv.max() / (np.sum(a_opt) ** 2)

        if score < best_score:
            best_score = score
            best_a = a_opt.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – cheap sub‑gradient balancing (mass transfer)
# ----------------------------------------------------------------------
def _subgrad_balance(a: np.ndarray,
                     L: int,
                     deadline: float,
                     max_iters: int = 20000,
                     step_frac: float = 0.4):
    """
    Repeatedly moves a fraction of probability mass from the index with the
    largest sub‑gradient to the one with the smallest.  The sub‑gradient of
    ``max(conv)`` at a point a is 2·(w ⋆ a) where w is the normalised
    indicator of the active set of maximal convolution entries.
    """
    n = a.shape[0]
    for _ in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_conv(a, L)
        max_val = conv.max()
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]
        if active.size == 0:
            break

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size          # normalised mask
        corr = _corr_w_a(w, a, L)               # correlation w⋆a
        grad = 2.0 * corr                       # sub‑gradient

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if grad[i_max] - grad[i_min] <= 1e-12:
            break

        delta = min(a[i_max], step_frac * a[i_max])
        if delta <= 0:
            break
        a[i_max] -= delta
        a[i_min] += delta

    return a


# ----------------------------------------------------------------------
# Phase 3 – short Adam polishing (as in the reference implementation)
# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   target: float = 1.5):
    """
    Adam optimiser on the true max‑convolution (sub‑gradient).  Stops early
    if the target bound is reached.
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # initial evaluation
    a0 = np.exp(b - b.max())
    a0 = a0 / a0.sum()
    fft_a = np.fft.rfft(a0, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # logits → probabilities
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        # convolution and max
        fft_a = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
        else:
            stagn += 1

        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * rng.standard_normal(n)
            stagn = 0

        # sub‑gradient of the true max‑convolution (averaged over active shifts)
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        corr = _corr_w_a(w, a, fft_len)      # 2·(w⋆a) will be the sub‑gradient
        subgrad_a = 2.0 * corr

        grad_obj = 2.0 * n * subgrad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score
