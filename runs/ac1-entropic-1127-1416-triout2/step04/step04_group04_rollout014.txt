"""
propose_candidate.py

Heuristic optimiser for the step‑function inequality

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm follows three ideas that improve on the previous
implementation:

1. **Richer parametric sweep** –  a finer grid of arcsine / power‑law /
   symmetric‑Beta families (including a mixture with the uniform
   distribution) is explored for each candidate length.  The fast
   ``compute_score`` (scale‑invariant) is used to keep only the best
   few seeds.

2. **Soft‑max mirror descent** –  instead of a pure sub‑gradient that
   only looks at the current maximal convolution entry we use a
   temperature‑controlled soft‑max weight vector.  The temperature is
   increased during the run so the early iterations see a smooth
   landscape (large step‑sizes) while the final iterations behave like
   the classic sub‑gradient method.  The update is multiplicative
   (exponentiated‑gradient) and the iterate is projected onto the
   probability simplex after every step.  A symmetry enforcement
   (mirroring the vector) cuts the effective dimension in half.

3. **Final polishing** –  after the global best has been found we run
   a longer soft‑max descent followed by a short binary‑weight descent
   to tighten the bound.

All work is done with FFT‑based convolutions for speed, the search
honours the supplied time budget and always returns a feasible (non‑negative)
list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the scale‑invariant objective
        2·n·max(conv(a,a)) / (sum(a)²)
    a is assumed to be non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k  w.r.t. a.
    Implements g_j = 2· Σ_m a_m·w_{m+j}
    using one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    eps = max(eps, 1e-6)                     # avoid division by zero
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Mirror‑descent style refinements
# ----------------------------------------------------------------------


def refine_softmax(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 3000,
    eta0: float = 0.7,
    alpha_start: float = 0.5,
    alpha_end: float = 1200.0,
    enforce_symmetry: bool = True,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with a temperature‑controlled
    soft‑max approximation to the max‑operator.

    Parameters
    ----------
    a : np.ndarray
        Starting probability vector (already normalised).
    deadline : float
        Unix timestamp after which the routine must stop.
    max_iter : int
        Upper bound on the number of iterations (can be cut short by the deadline).
    eta0 : float
        Initial step‑size (decreases as 1/√t).
    alpha_start / alpha_end :
        Temperature schedule: ``alpha`` grows geometrically from
        ``alpha_start`` to ``alpha_end``.  Small ``alpha`` gives a smooth
        weight vector, large ``alpha`` approaches the true sub‑gradient.
    enforce_symmetry :
        If True, the iterate is forced to be symmetric after every
        projection (a_i = a_{n‑1‑i}).  The optimal solution is expected to
        be symmetric, and this reduces the effective dimensionality.

    Returns
    -------
    best_a : np.ndarray
        Best vector seen.
    best_sc : float
        Its exact (scale‑invariant) score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    # geometric temperature schedule
    if max_iter > 1:
        log_ratio = np.log(alpha_end / max_alpha_start) if (alpha_end > alpha_start) else 0.0
    else:
        log_ratio = 0.0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # convolution of a with itself
        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # temperature for this iteration (geometric progression)
        t_frac = (it - 1) / (max_iter - 1) if max_iter > 1 else 0.0
        alpha = alpha_start * (alpha_end / alpha_start) ** t_frac

        # soft‑max weights (stable formulation)
        shifted = conv - max_val                      # ≤ 0
        # avoid overflow/underflow
        exp_vals = np.exp(alpha * shifted)
        w = exp_vals / np.sum(exp_vals)              # length = 2n‑1

        # gradient w.r.t. a
        grad = gradient_from_weights(a, w)

        # learning rate schedule (classic 1/√t)
        eta = eta0 / np.sqrt(it)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad)

        # enforce non‑negativity & project onto simplex
        a = normalize(a)

        if enforce_symmetry:
            a = (a + a[::-1]) / 2.0
            a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 3000,
    eta0: float = 0.7,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent where the weight vector is the
    (uniform) indicator of the maximal convolution entry.
    Mirrors the original implementation but includes optional symmetry.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # binary weight vector: 1 on all maximal indices, 0 elsewhere
        w = np.zeros(conv_len, dtype=float)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        # enforce symmetry (optional – the optimum is symmetric)
        a = (a + a[::-1]) / 2.0
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc

def propose_candidate(
    seed: int = 123,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function whose ``evaluate_sequence``
    score is as low as possible.  The routine respects the provided
    time budget (default 1000 s) and always returns a feasible list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to explore.
    # ------------------------------------------------------------------
    lengths = [1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n>>> Exploring length n = {n}")

        # ----------------- 2. Build a rich pool of seeds -----------------
        candidates: List[np.ndarray] = []

        # a) deterministic families
        candidates.append(uniform_shape(n))
        candidates.append(arcsine_initial(n))

        # b) power‑law sweep
        for p in np.linspace(0.30, 0.80, 6):
            for eps in (0.5, max(5.0, 0.02 * n)):
                candidates.append(powerlaw_shape(n, p, eps))

        # c) symmetric beta sweep
        for a_param in np.linspace(0.2, 4.0, 6):
            candidates.append(beta_shape(n, a_param))

        # d) mixtures of arcsine and uniform
        for w in np.linspace(0.0, 0.5, 5):
            mix = (1.0 - w) * arcsine_initial(n) + w * uniform_shape(n)
            candidates.append(normalize(mix))

        # e) random Dirichlet draws
        for _ in range(12):
            candidates.append(random_dirichlet(n, rng))

        # ----------------- 3. Quick ranking (fast score) ---------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 7
        top_candidates = [c for _, c in scored[:top_k]]
        print(
            "  top initial scores:",
            " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]),
        )

        # ----------------- 4. Refine each seed -------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            start_sc = compute_score(cand)
            print(f"  candidate {idx + 1}/{len(top_candidates)} start_score = {start_sc:.6f}")

            # allocate a budget of iterations depending on remaining time
            remaining = max(0.0, deadline - time.time())
            # aim for roughly 8–10 iterations per second
            max_iter = int(min(15000, max(1500, remaining * 9)))

            # 1) soft‑max descent (smooth → sharp)
            refined, _ = refine_softmax(
                a=cand,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.7,
                alpha_start=0.3,
                alpha_end=1200.0,
                enforce_symmetry=True,
            )

            # 2) a short binary‑weight sub‑gradient pass for final polishing
            refined, _ = refine_subgradient(
                a=refined,
                deadline=deadline,
                max_iter=int(min(3000, max(300, remaining * 2))),
                eta0=0.5,
            )

            # evaluate with the official (fully‑checked) function
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 5. Final polishing of the global best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)

        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(30000, max(5000, remaining * 8)))

        # longer soft‑max phase
        polished, _ = refine_softmax(
            a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.6,
            alpha_start=0.5,
            alpha_end=3000.0,
            enforce_symmetry=True,
        )

        # short binary‑weight sub‑gradient refinement
        polished, _ = refine_subgradient(
            a=polished,
            deadline=deadline,
            max_iter=int(min(8000, max(1000, remaining * 1.5))),
            eta0=0.4,
        )

        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector of modest size
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
```

This implementation expands the parametric search, replaces the pure binary‑weight mirror descent with a temperature‑controlled soft‑max version (giving smoother gradients early on) and always enforces symmetry, which empirically yields a lower maximal convolution and consequently a smaller final score. It respects the 1000 s budget, prints progress, and guarantees a feasible return.
