"""
Propose a low‑score step function.

Algorithm (≈ 1000 s budget)

1. **Fallback** – use the previously supplied sequence if it is better.
2. **Random sparse search** – generate many random sparse vectors (different
   lengths, densities and uniform or random weights) and keep the best.
3. **Smooth‑max descent** – starting from the best vector (and a few random
   restarts) minimise a smooth approximation of the objective
   `2·n·max(a∗a)`:
        f_T(a) = T · log( Σ_k exp( (a∗a)_k / T) )
   The gradient of f_T is cheap to compute via two FFT‑based convolutions.
   Temperature T is slowly decreased (simulated‑annealing style) and a
   projected gradient step is performed (projection onto the probability
   simplex).
4. **Hill‑climb polish** – a stochastic neighbour search (swap or mass
   transfer) that accepts strictly improving moves.
5. Return the best sequence found (as a plain Python list of floats).

All arithmetic is performed on vectors normalised to have sum = 1;
`evaluate_sequence` is scale‑invariant, so the returned list can be used
directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when the vector is already
    normalised to sum = 1.
    Returns 2·n·max(a∗a) (exactly the value returned by `evaluate_sequence`).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Hill‑climber used for the final polish
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    Only strictly improving moves are kept.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # initial fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
                    cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → increase step
        else:
            step_scale *= 0.97                         # failure → decrease step

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) descent
# ----------------------------------------------------------------------


def smooth_descent(a_init: np.ndarray,
                   time_deadline: float,
                   max_iter: int = 200_000,
                   step0: float = 0.07,
                   temp_decay: float = 0.99995,
                   temp_factor: float = 5.0,
                   temp_min: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth approximation

        f_T(a) = T * log Σ_k exp( (a∗a)_k / T )

    The gradient is g = 2 * Σ_k w_k * shift(a, k) where w_k are the
    soft‑max weights.  Temperature T is geometrically decayed.
    """
    a = a_init.astype(float).copy()
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(best_a)

    # initial temperature – a multiple of the current max convolution
    cur_conv = convolution_fft(a, a)
    temp = max(cur_conv.max() * temp_factor, temp_min)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # ----- convolution and soft‑max weights -----
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        # geometric cooling
        temp = max(temp * temp_decay, temp_min)

        # soft‑max (shifted for numerical stability)
        shifted = conv - max_conv
        w = np.exp(shifted / temp)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient via convolution with reversed weights -----
        w_rev = w[::-1]
        grad_full = convolution_fft(a, w_rev)          # length 3n‑2
        L = len(w)                                      # = 2n‑1
        # slice L‑n … L‑1 and reverse to obtain g_i = 2 Σ_j w_{i+j} a_j
        grad = 2.0 * grad_full[L - n:L][::-1]

        # ----- projected gradient step -----
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape flat regions
        if t % 5000 == 0:
            a += np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score

# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    The search proceeds through three phases:
    * random sparse‑subset generation,
    * smooth‑max gradient descent (with a few restarts),
    * final hill‑climb polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 0 – try the supplied fallback if it exists
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random sparse‑subset search (very cheap, many samples)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]
    while time.time() < deadline - 250.0:   # reserve time for refinements
        n = int(np.random.choice(lengths))
        density = np.random.uniform(0.15, 0.85)
        k = max(2, int(round(density * n)))

        # random subset of positions
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)

        if np.random.rand() < 0.5:
            # uniform masses on the chosen positions
            a[idx] = 1.0 / k
        else:
            # random Dirichlet‑like masses on the chosen positions
            w = np.random.rand(k)
            w /= w.sum()
            a[idx] = w

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.tolist()
            print(f"Random subset best: score={best_val:.6f} (n={n}, k={k}, dens={density:.2f})")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max descent (several restarts)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print("\n=== Smooth‑max descent phase ===")
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline - 120.0:
                break

            if r == 0:
                a0 = np.array(best_seq, dtype=float)
            else:
                n = int(np.random.choice(lengths))
                a0 = np.random.rand(n)
                a0 = a0 / a0.sum()            # dense random start

            refined_a, refined_score = smooth_descent(
                a0,
                time_deadline=deadline - 60.0,
                max_iter=300_000,
                step0=0.07,
                temp_decay=0.99994,
                temp_factor=5.0,
                temp_min=1e-7,
            )
            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_a.tolist()
                print(f"Smooth‑descent improvement: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n=== Hill‑climb polishing ===")
        refined_seq, refined_val = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=500_000,
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq
            print(f"Hill‑climb improvement: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final evaluation (using the official harness function)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
