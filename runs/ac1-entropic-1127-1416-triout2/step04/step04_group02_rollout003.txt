def smooth_optim(a, deadline, max_iter, eta0, beta0, beta_factor):
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        beta = beta0 * (beta_factor ** (it / max_iter))  # exponential growth
        # compute conv
        conv = convolution_fft(a, a)
        max_val = conv.max()
        # compute weight via softmax
        exp_scaled = np.exp(beta * (conv - max_val))
        w = exp_scaled / exp_scaled.sum()
        # compute gradient via correlation
        grad = correlation_fft(a, w)[:n]  # linear correlation in full length, first n entries
        grad = 2.0 * grad

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # ensure nonneg
        a = np.maximum(a, 0)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # optional occasional noise
        if it % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score

"""
Propose a step‑function (non‑negative height vector) with a low
evaluate_sequence() value.

The search combines three ideas:

1.   **Rich initial library** –  For each candidate length we sample
     uniform, arcsine, Dirichlet draws and many Beta‑shaped vectors
     α,β∈(0.01,2).  All candidates are normalised to sum‑one; the
     cheap internal score ``compute_score`` (FFT‑based) is used to keep
     only the most promising shapes.

2.   **Smooth optimisation** –  A soft‑max (“log‑sum‑exp’’) surrogate for
     the non‑smooth max‑convolution is minimised on the probability
     simplex.  For a given β the objective is
        ϕβ(a)= (1/β)·log Σₖ exp(β·(a∗a)ₖ) .
     Its gradient is a simple cross‑correlation of the weight vector
     wₖ∝exp(β·(a∗a)ₖ).  An exponentiated‑gradient (mirror‑descent)
     update keeps the iterate non‑negative; β is slowly increased so
     that the surrogate approaches the true max.

3.   **Final max‑convolution polishing** –  The best vector from the
     smooth phase is refined with the original sub‑gradient method
     (exact max‑convolution gradient) to obtain the final candidate.

All work respects the 1000 s wall‑clock limit and always returns a
feasible list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Scale‑invariant objective used by the official evaluator:
        2·n·max(a∗a) / (sum(a)²)
    ``a`` is assumed non‑negative and is not required to be normalised.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Helper to obtain the cross‑correlation a ⋆ w (full linear correlation)
# ----------------------------------------------------------------------


def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation of ``a`` (length n) with ``w`` (length 2n‑1):
        (a ⋆ w)[ℓ] = Σ_j a_j·w_{j+ℓ}    for ℓ=0,…,n‑1

    Implemented via a convolution of the reversed ``a`` with ``w``.
    """
    rev_a = a[::-1]
    conv = convolution_fft(rev_a, w)          # length = n + (2n‑1) – 1 = 3n‑2
    start = len(a) - 1                         # position of lag 0
    return conv[start:start + len(a)]


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def uniform_initial(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1}·(n‑1‑i+½)^{β‑1} (normalised)."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws, normalised)."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny random perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smooth_optimize(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.8,
    beta0: float = 5.0,
    beta_mult: float = 1.001,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed objective
        φβ(a) = (1/β)·log Σₖ exp(β·(a∗a)ₖ).

    The update is multiplicative (exponentiated gradient) and the iterate
    is always projected onto the probability simplex.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    beta = beta0
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # (a∗a) and soft‑max weights
        conv = convolution_fft(a, a)
        max_val = conv.max()
        exp_scaled = np.exp(beta * (conv - max_val))
        w = exp_scaled / exp_scaled.sum()          # length = 2n‑1

        # gradient = 2·(a ⋆ w)
        grad = 2.0 * correlation_fft(a, w)

        # step size ηₜ = η₀ / √t
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # re‑project onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # keep best internal score occasionally
        if it % 500 == 0:
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

        # anneal β → stronger approximation of the max
        beta = min(500.0, beta * beta_mult)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient optimiser for the exact max‑convolution
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k (where the max is attained) the gradient is
        g_j = 2·a_{k‑j}   (if the index is valid)
    The average is taken if several offsets are active.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient descent on the true max‑convolution.
    Returns the best vector seen and its internal (scale‑invariant) score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

    # ------------------------------------------------------------------

    lengths = [1024, 2048, 4096, 8192]

        pool: List[np.ndarray] = []

                initial_a=a_smooth,

            # ------------------------------------------------------------------

            # ------------------------------------------------------------------

    return best_seq
