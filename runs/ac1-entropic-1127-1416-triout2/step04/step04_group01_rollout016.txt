def golden_section_minimize(func, a, b, tol=1e-4, max_iter=50):
    phi = (1 + np.sqrt(5))/2
    resphi = 2 - phi
    c = b - resphi * (b - a)
    d = a + resphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - resphi * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + resphi * (b - a)
            fd = func(d)
        if b - a < tol:
            break
    return (a + b) / 2, min(fc, fd)
```

def linear_program_update(f, M, deadline):
    # Determine active constraints
    conv_f = convolution_fft(f, f)
    max_val = conv_f.max()
    # active indices within 0.001 of max?
    tol = max_val * 0.01  # 1% tolerance
    active = np.where(conv_f >= max_val - tol)[0]
    if len(active) == 0:
        active = [np.argmax(conv_f)]
    # Build constraint matrix A_ub (len(active) x n)
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active):
        # for j in 0..n-1:
        # f_i where i = k - j
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # for j = k - i
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))
    # b_ub = np.full(len(active), max_val)
    c = -np.ones(n)  # maximize sum(g) -> minimize -sum(g)
    # bounds (0, None)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=np.full(len(active), max_val), bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return f
    g = res.x
    if g.sum() == 0:
        return f
    g = g / g.sum()
    # combine
    t = 0.15
    new_f = (1 - t) * f + t * g
    return new_f

"""
Propose a low‑discrepancy step sequence.

Main ideas
----------
*   A richer set of seed generators (Beta, power‑law, asymmetric
    power‑law, mixture, triangular) is sampled for each length.
*   For each length the best few seeds are refined by a long Adam
    optimisation on a soft‑max surrogate of the true objective.
*   The refined vector is then polished by exact sub‑gradient descent.
*   A cheap Frank–Wolfe style Linear‑Program (LP) step is applied:
    maximise the total mass of a new vector under the current
    convolution bound; the resulting direction is mixed with the old
    vector and projected onto the probability simplex.
*   If time permits the best vector is up‑sampled to a larger length
    and polished again.
*   All steps respect the global time budget (default 1000 s) and all
    vectors are projected onto the simplex, i.e. Σa = 1, a ≥ 0.
    The official score `evaluate_sequence` is scale‑invariant, therefore
    fixing the sum to one does not affect optimality.
"""

import time
import numpy as np
from typing import List, Tuple

# -----------------------------------------------------------------------------
# Basic utilities
# -----------------------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.  Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    With Σa = 1 the denominator vanishes and the objective reduces to
    `2·n·max(conv(a,a))`.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# -----------------------------------------------------------------------------
# Seed generators
# -----------------------------------------------------------------------------

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """(i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},   p>0."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_asym_power_shape(n: int,
                              p_left: float,
                              p_right: float,
                              eps: float) -> np.ndarray:
    """
    Asymmetric power law: a_i ∝ (i+½)^{-p_left}·(n‑½‑i)^{-p_right}
    (with an optional epsilon added to the product).
    """
    i = np.arange(n, dtype=float)
    left = (i + 0.5 + eps) ** (-p_left)
    right = (n - 0.5 - i + eps) ** (-p_right)
    vals = left * right
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at `peak`."""
    left = np.arange(peak + 1)
    right = np.arange(n - peak - 1)[::-1]
    seq = np.concatenate([left, right]).astype(float)
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# -----------------------------------------------------------------------------
# Optimisers (Adam on the soft‑max surrogate, exact sub‑gradient)
# -----------------------------------------------------------------------------

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) before the simplex projection.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length 2·n‑1
    max_c = np.max(conv)

    # numerically stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)            # weights, sum to 1

    # weighted sub‑gradient:   2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length n + (2n‑1) - 1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Exact sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t).  Projection onto simplex keeps Σ=1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a

# Frank–Wolfe style linear‑program refinement (optional, uses SciPy if present)
# -----------------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    import scipy.sparse as sp
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False


def lp_refine_step(a: np.ndarray,
                   deadline: float,
                   t_mix: float = 0.15,
                   tol_frac: float = 0.01) -> np.ndarray:
    """
    One Frank–Wolfe LP step.
    Maximise total mass of a vector `g` under the current convolution bound,
    then mix (1‑t)·a + t·g and project back onto the simplex.
    """
    if not _SCIPY_AVAILABLE:
        return a

    n = len(a)
    conv = convolution_fft(a, a)
    max_val = conv.max()

    # constraints that are “almost” tight
    active = np.where(conv >= max_val - tol_frac * max_val)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=int)

    # Build a sparse matrix A (|active| × n) where row k contains the
    # coefficients of the convolution at lag `active[k]`.
    rows, cols, data = [], [], []
    for row_idx, k in enumerate(active):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(a[i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))

    c = -np.ones(n)                                 # maximise sum(g)  → minimise -sum
    b_ub = np.full(len(active), max_val)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return a

    g = res.x
    if g.sum() == 0.0:
        return a
    g = g / g.sum()
    new_a = (1.0 - t_mix) * a + t_mix * g
    return project_to_simplex(new_a)

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative coefficient sequence with a low
    `evaluate_sequence` value.  The routine consists of

    1.  Generating a broad pool of candidates for several lengths.
    2.  Keeping the best few seeds and polishing each with a long
        Adam optimisation (soft‑max surrogate) followed by exact
        sub‑gradient descent.
    3.  Evaluating the official score and retaining the global best.
    4.  An optional Frank–Wolfe LP refinement + a final global
        polishing phase (sub‑gradient + Adam).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ----------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # 1.  Exploration over several lengths
    # ----------------------------------------------------------------------
    lengths = [2048, 4096]                 # larger lengths are cheap because we reuse the same pattern
    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        pool: List[Tuple[float, np.ndarray]] = []

        # deterministic starters
        pool.append((compute_score(np.full(n, 1.0 / n)), np.full(n, 1.0 / n)))
        pool.append((compute_score(arcsine_initial(n)), arcsine_initial(n)))

        # 1a. random beta family
        for _ in range(300):
            alpha = rng.uniform(0.05, 3.0)
            beta = rng.uniform(0.05, 3.0)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_beta_shape(n, alpha, beta, eps)
            pool.append((compute_score(cand), cand))

        # 1b. random power‑law family
        for _ in range(300):
            p = rng.uniform(0.1, 1.0)
            eps = rng.uniform(0.0, 1.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # 1c. asymmetric power‑law
        for _ in range(150):
            p_l = rng.uniform(0.05, 1.0)
            p_r = rng.uniform(0.05, 1.0)
            eps = rng.uniform(0.0, 1.0)
            cand = generate_asym_power_shape(n, p_l, p_r, eps)
            pool.append((compute_score(cand), cand))

        # 1d. mixture of arcsine & uniform
        for _ in range(100):
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # 1e. simple triangular bumps
        for _ in range(100):
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

        # keep the most promising seeds
        pool.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in pool[:7]]

        # ---------- 2. refinement of each seed ----------
        for seed_vec in top_seeds:
            if time.time() >= deadline - 5.0:
                break

            # Phase A – smooth Adam optimisation
            remaining = deadline - time.time()
            max_iter_adam = int(min(180_000, remaining * 35))
            refined, _ = adam_optimize_smooth(
                initial_a=seed_vec,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=2.0,
                lam_end=180.0,
                noise_every=3000,
                noise_scale=3e-5,
            )

            # Phase B – exact sub‑gradient descent
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_sub = int(min(100_000, remaining * 35))
                refined = subgradient_descent_exact(
                    refined,
                    deadline,
                    max_iter=max_iter_sub,
                    step0=0.04,
                )

            # Phase C – optional LP polishing (few cheap steps)
            if _SCIPY_AVAILABLE and time.time() < deadline - 5.0:
                lp_steps = 3
                cur = refined
                for i in range(lp_steps):
                    if time.time() >= deadline:
                        break
                    cur = lp_refine_step(cur, deadline, t_mix=0.12, tol_frac=0.015)
                refined = cur

            # Final official evaluation
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"  → refined official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, refined.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception:
                pass

    # ----------------------------------------------------------------------
    # 3. Global polishing of the overall best vector (optional up‑sampling)
    # ----------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        # Upsample to a larger length if we still have enough budget
        if len(a) < 8192 and (deadline - time.time()) > 30.0:
            new_n = min(8192, len(a) * 2)
            x_old = np.linspace(0, 1, len(a))
            x_new = np.linspace(0, 1, new_n)
            a = np.interp(x_new, x_old, a)
            a = a / a.sum()
            print(f"\nUpsampled to n = {new_n} for final polishing")

        remaining = deadline - time.time()
        total_iters = int(min(500_000, remaining * 40))

        # split the remaining budget between sub‑gradient and Adam
        it_sub = total_iters // 2
        it_adam = total_iters - it_sub

        print("\nFinal global polishing (sub‑gradient + Adam) …")
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.02)

        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=it_adam,
            lr0=0.08,
            lam_start=2.0,
            lam_end=200.0,
            noise_every=2000,
            noise_scale=2e-5,
        )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ----------------------------------------------------------------------
    # 4. Guard – guarantee a return value
    # ----------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_val:.6f}")

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
