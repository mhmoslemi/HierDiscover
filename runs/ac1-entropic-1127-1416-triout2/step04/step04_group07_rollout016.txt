"""
Minimal-step‐function optimizer

Strategy
--------
* Use many short Adam runs (sub‑gradient of the true max‑convolution) with
  different random starts.  Each run is given a small time slice (≈30 s
  or the remaining budget) and the best result is kept.
* After the restart phase a tiny “hill‑climbing” refinement moves a small
  amount of mass from a random index to another one and keeps the move
  only if the evaluation score improves.
* The algorithm works on the simplex (probability vectors) – scaling of
  the vector does not affect the objective, therefore we normalise to
  sum = 1 after every modification.
* No external files or network access are used – only NumPy, SciPy
  (for the FFT) and the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution of a, length 2·n‑1, using FFT length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Return (max_conv, sub‑gradient) for the aperiodic self‑convolution.
    The gradient is the average of the sub‑gradients for all active
    (max‑tight) positions.
    """
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Adam optimiser on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    restart_id: int,
    target: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex minimizing
        2·n·max(conv(a,a)) / (sum(a)**2)
    ``init_a`` must already be normalised to sum = 1.
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # logistic variables (log‑softmax) – tiny epsilon avoids log(0)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # evaluate the initial point
    a = np.exp(b - b.max())
    a /= a.sum()
    max_val, _ = _maxconv_and_subgrad(a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # probabilities from logits
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        # convolution & sub‑gradient of the max entry
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} iter {it:7d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if best_score < target:
            break

        # occasional random jitter to escape flat plateaus
        if stagn > 8000:
            b += 0.01 * np.random.randn(n)
            stagn = 0

        # Adam update (gradient of the objective 2·n·max(conv))
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)          # ⟨a,∇⟩
        grad_b = a * (grad_obj - avg)      # ∇_b of the log‑softmax parametrisation

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑mass‐transfer hill‑climbing (post‑processing)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Move a small amount of mass between two coordinates as long as the
    objective improves.  Operates directly on the ℓ∞‑objective."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    best_a = a.copy()
    # initial objective (sum is (very close to) 1)
    fft_a = np.fft.rfft(best_a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()

    rng = np.random.default_rng(987654)
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or best_a[i] <= 1e-15:
            continue
        # move up to 5 % of the mass at i
        delta = rng.random() * 0.05 * best_a[i]
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # compute new objective
        fft_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fft_c * fft_c, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_c.max()

        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best_a = cand
            # occasional progress report
            if int(time.time()) % 30 == 0:
                print(f"Hill‑climb: improved score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function sequence within the given time
    budget (seconds).  The routine mixes many short Adam runs with a
    final random‑mass‑transfer hill‑climbing phase and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(123456)          # deterministic RNG
    deadline = time.time() + budget_s - 0.5       # small safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse previously good construction if available
    # ------------------------------------------------------------------
    try:
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = previous.shape[0]
        # normalise previous vector to sum = 1
        prev = np.maximum(previous, 0.0)
        s = prev.sum()
        if s > 0:
            prev = prev / s
        else:
            prev = np.full(n, 1.0 / n)
    except Exception:
        prev = None
        n = 5146                                   # classic size from earlier runs

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_overall_a = None
    best_overall_score = float("inf")
    restart_id = 0
    strategies = ["prev", "uniform", "random_dirichlet", "geometric"]
    min_restart_slice = 12.0                        # seconds per restart (min)

    # ------------------------------------------------------------------
    # Multiple short Adam restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - min_restart_slice:
        kind = strategies[restart_id % len(strategies)]

        # ---- generate feasible start point ---------------------------------
        if kind == "prev" and prev is not None:
            a0 = prev.copy()
        elif kind == "uniform":
            a0 = np.full(n, 1.0 / n)
        elif kind == "random_dirichlet":
            # sparse Dirichlet (α=0.5)
            sample = rng.gamma(0.5, 1.0, size=n)
            a0 = sample / sample.sum()
        else:  # geometric shape
            r = 0.98 + 0.02 * rng.random()
            left = r ** np.arange(n)
            right = r ** np.arange(n - 1, -1, -1)
            a0 = left + right
            a0 = a0 / a0.sum()

        # ---- allocate a slice of the remaining budget for this restart -----
        remain = deadline - time.time()
        slice_deadline = time.time() + min(30.0, 0.45 * remain)

        # ---- run Adam optimiser (sub‑gradient) -----------------------------
        a_opt, score_opt = _adam_optimize(
            a0, slice_deadline, restart_id, target=1.503
        )

        # ---- update global best -------------------------------------------
        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()
            print(
                f"Global best after restart {restart_id:02d}: {best_overall_score:.6f}"
            )
            if best_overall_score < 1.503:
                print(
                    f"Target reached (score={best_overall_score:.6f}) after {restart_id+1} restarts."
                )
                break

        restart_id += 1

    # ------------------------------------------------------------------
    # Final random‑mass‑transfer hill climbing (makes tiny tweaks)
    # ------------------------------------------------------------------
    if best_overall_a is None:
        best_overall_a = np.full(n, 1.0 / n)

    # ensure a valid probability vector before hill climbing
    best_overall_a = np.maximum(best_overall_a, 0.0)
    best_overall_a /= best_overall_a.sum() + 1e-30

    hill_deadline = deadline
    best_overall_a, best_overall_score = _hill_climb(best_overall_a, hill_deadline)

    # ------------------------------------------------------------------
    # Return a plain Python list (the evaluator expects this)
    # ------------------------------------------------------------------
    return [float(v) for v in best_overall_a.tolist()]
