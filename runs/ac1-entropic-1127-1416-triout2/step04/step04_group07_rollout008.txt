```
def adam_run(init_a, deadline):
    n = init_a.shape[0]
    L = 1 << ((2*n - 2).bit_length())
    # logits
    eps = 1e-12
    logits = np.log(init_a + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    it = 0
    best_local_a = None
    best_local_score = float('inf')
    while time.time() < deadline:
        # compute probabilities from logits
        max_log = np.max(logits)
        a = np.exp(logits - max_log)
        a_sum = np.sum(a)
        a = a / a_sum
        
        # compute max conv and gradient
        max_val, grad_a = _maxconv_and_subgrad(a, L)
        cur_score = 2.0 * n * max_val
        
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_a = a.copy()
            # optional print
            if restart == 0:
                print(f"Adam iter {it} best = {cur_score:.6f}")
        
        # If we've reached target skip further?
        # Compute gradient wrt logits:
        avg = np.dot(a, 2.0 * n * grad_a)  #   grad of objective w.r.t a
        grad_logits = a * (2.0 * n * grad_a - avg)  # chain rule: d a_i / d log b_i = a_i - a_i a_j sum?
        # Actually we need variable mapping: b=logits; a_i = exp(b_i)/ sum_j exp(b_j); derivative of a w.r.t b: ∂a_i/∂b_j = a_i (δ_{ij} - a_j). So gradient b = a ⊙ (grad_a - avg) where avg = Σ a_j grad_a_j.
        # So compute as above.
        # But we need grad_a of objective w.r.t a: that's 2 n * grad_a (where grad_a is partial derivative of max conv w.r.t a). Indeed subgrad function returns gradient of max conv (∂max/∂a_i). Then objective = 2 n * max conv, gradient = 2 n * grad_a.
        # So grad_obj = 2 n * grad_a.
        # Then set grad_b = a * (grad_obj - avg).
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_logits = a * (grad_obj - avg)
        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = 0.5 / np.sqrt(it)  # maybe 0.5 initial
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    return best_local_a, best_local_score
```

```
def simulated_annealing(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    # compute initial
    max_val = np.fft.irfft(np.fft.rfft(a, n=L) ** 2, n=L)[:2*n-1].max()
    cur_score = 2.0 * n * max_val
    best_a = a.copy()
    best_score = cur_score
    temp = 0.01 * cur_score  # initial temperature
    # schedule: exponential decay
    decay = 0.9995
    steps = 0
    while time.time() < deadline:
        # propose move: pick i where a_i > 0, j arbitrary
        i = rng.integers(n)
        # ensure a_i > some epsilon
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        # propose transfer epsilon = delta * a[i] where delta random in [0, 0.1]
        delta = 0.1 * rng.random()
        eps = delta * a[i]
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # compute new max conv
        maxv_new = np.fft.irfft(np.fft.rfft(a_new, n=L) ** 2, n=L)[:2*n-1].max()
        new_score = 2.0 * n * maxv_new
        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / temp):
            # accept
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # temperature decay
        temp *= decay
        steps += 1
        # maybe occasional reheat?
        if steps % 20000 == 0:
            # Print progress
            print(f"SA step {steps}: best = {best_score:.6f}")
    return best_a, best_score
```

```
        # get probabilities
        bmax = np.max(logits)

        a_sum = a.sum()

                print(f"Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:

            logits += 0.01 * rng.standard_normal(n)
```

"""
Hybrid random-restart Adam + Simulated Annealing for minimizing
2·n·max(conv(a,a))/(sum a)^2.  Each restart starts from a diverse random
distribution, runs Adam on the exact max‑conv sub‑gradient (log‑softmax
parameterisation to stay on the simplex) and keeps the best solution.
When the time budget is low a short simulated‑annealing refinement is
performed.
"""

import time
import numpy as np

# global rng
_rng = np.random.default_rng(12345)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max_conv, sub‑gradient) of aperiodic self‑convolution."""
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad

def _adam_optimize(init_a: np.ndarray, stop_time: float, restart_id: int,
                   target: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser on the exact max‑convolution sub‑gradient."""
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    eps = 1e-12
    logits = np.log(init_a + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    it = 0
    best_a = None
    best_score = float('inf')
    stagn = 0

    while time.time() < stop_time:
        it += 1

        # probabilities from logits (soft‑max)
        bmax = np.max(logits)
        a = np.exp(logits - bmax)
        a = a / a.sum()

        # evaluate max‑conv and its sub‑gradient
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            if it % 5000 == 0:
                print(f"Adam {restart_id:02d} iter {it:7d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if best_score < target:
            # early exit
            break

        if stagn > 8000:
            # random jitter in log‑space to escape plateaus
            logits += 0.02 * _rng.standard_normal(n)
            stagn = 0

        # gradient of the original objective w.r.t. a
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_logits = a * (grad_obj - avg)   # chain rule through soft‑max

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.5 / np.sqrt(it)   # learning‑rate schedule
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

    if best_a is None:
        # fall back to the current a
        best_a = a
        best_score = cur_score
    return best_a, best_score

def _simulated_annealing(start_a: np.ndarray, stop_time: float):
    """Very cheap SA that moves a small amount of probability mass each step."""
    a = start_a.copy()
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    # initial score
    max_val = np.fft.irfft(np.fft.rfft(a, n=fft_len) ** 2, n=fft_len)[:2 * n - 1].max()
    cur_score = 2.0 * n * max_val
    best_a = a.copy()
    best_score = cur_score

    # temperature starts as a small fraction of the current score
    temp = cur_score * 0.05
    decay = 0.9999
    step = 0

    while time.time() < stop_time:
        i = _rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = _rng.integers(n)
        if i == j:
            continue
        # move a fraction of the mass from i to j
        frac = 0.03 * _rng.random()   # up to 3 % of a[i]
        eps = frac * a[i]
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # evaluate
        max_val_new = np.fft.irfft(np.fft.rfft(a_new, n=fft_len) ** 2,
                                   n=fft_len)[:2 * n - 1].max()
        new_score = 2.0 * n * max_val_new
        dE = new_score - cur_score

        if dE < 0 or _rng.random() < np.exp(-dE / temp):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        temp *= decay
        step += 1
        if step % 5000 == 0:
            print(f"SA step {step:6d}: best = {best_score:.6f}")

    return best_a, best_score

def _random_seed(n: int, kind: int) -> np.ndarray:
    """Generate a diverse random starting distribution."""
    if kind == 0:
        # Sparse Dirichlet (α in [0.2, 0.8])
        alpha = 0.2 + 0.6 * _rng.random()
        samples = _rng.gamma(alpha, 1.0, size=n)
        a = samples / samples.sum()
    elif kind == 1:
        # Two‑sided geometric decay
        r = 0.85 + 0.3 * _rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        a = left + right
        a = a / a.sum()
    elif kind == 2:
        # Random bumps: few positions receive most of the mass
        a = np.full(n, 0.0)
        num = max(2, int(0.01 * n * (0.5 + _rng.random())))  # 1–2 % of entries
        idx = _rng.choice(n, size=num, replace=False)
        weights = _rng.random(num) + 0.1
        a[idx] = weights
        a += 1e-6                      # avoid exact zeros
        a = a / a.sum()
    else:
        # Absolute normal (α≈1)
        a = np.abs(_rng.standard_normal(n))
        a = a / a.sum()
    return a

def propose_candidate(budget_s: int = 1000) -> list:

    The method mixes many short Adam runs from diverse random seeds and,

    """
    deadline = time.time() + budget_s - 0.5   # small safety margin
    # ------------------------------------------------------------------
    # Problem size – reuse the previous best if available
    # ------------------------------------------------------------------
    previous = globals().get('height_sequence_1')
    if previous is not None:
        try:

            a_prev = np.maximum(a_prev, 0.0)
            if a_prev.sum() < 1e-12:

            else:
                a_prev = a_prev / a_prev.sum()
        except Exception:
            a_prev = None
            n = 5146
    else:
        a_prev = None
        n = 5146                       # default length

    # ------------------------------------------------------------------
    # Baseline – evaluate the known construction (if any)
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    if a_prev is not None:
        # compute exact score using our own routine (scale‑invariant)
        fft_len = 1 << ((2 * n - 2).bit_length())
        max_val = np.fft.irfft(np.fft.rfft(a_prev, n=fft_len) ** 2,
                               n=fft_len)[:2 * n - 1].max()
        best_score = 2.0 * n * max_val
        best_a = a_prev.copy()
        print(f"Baseline (previous) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – many short Adam restarts
    # ------------------------------------------------------------------
    restart_id = 0
    seed_kinds = [0, 1, 2, 3]   # four diverse families
    while time.time() < deadline - 8.0:

        init_a = _random_seed(n, kind)

            print(f"Target reached after {restart_id+1} restarts: {best_score:.6f}")

        best_a, best_score = _simulated_annealing(best_a, deadline)

    # ------------------------------------------------------------------
    # Safety fallback – return a uniform distribution if nothing was found
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # Return a plain Python list as required by the evaluator
    return [float(v) for v in best_a.tolist()]

    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max(conv), sub‑gradient) of the aperiodic self‑convolution."""
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


# ----------------------------------------------------------------------
# Adam optimiser on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray, stop_time: float, restart_id: int,
                   target: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on the exact max‑convolution sub‑gradient."""
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    eps = 1e-12
    logits = np.log(init_a + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    it = 0
    best_a = None
    best_score = float("inf")
    stagn = 0

    while time.time() < stop_time:
        it += 1

        # probabilities from logits (soft‑max)
        bmax = np.max(logits)
        a = np.exp(logits - bmax)
        a = a / a.sum()

        # evaluate max‑conv and its sub‑gradient
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            if it % 5000 == 0:
                print(f"Adam {restart_id:02d} iter {it:7d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # early exit if we already beat the target
        if best_score < target:
            break

        if stagn > 8000:
            # random jitter in log‑space to escape plateaus
            logits += 0.02 * _rng.standard_normal(n)
            stagn = 0

        # gradient of the original objective w.r.t. a
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        # chain rule through the soft‑max
        grad_logits = a * (grad_obj - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.5 / np.sqrt(it)             # simple decreasing schedule
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

    if best_a is None:                      # fallback – should never happen
        best_a = a
        best_score = cur_score
    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap Simulated‑Annealing polishing
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray, stop_time: float):
    """Move a small amount of probability mass each step, accepting uphill moves with a temperature."""
    a = start_a.copy()
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # initial score
    max_val = np.fft.irfft(np.fft.rfft(a, n=fft_len) ** 2,
                           n=fft_len)[:2 * n - 1].max()
    cur_score = 2.0 * n * max_val
    best_a = a.copy()
    best_score = cur_score

    temp = cur_score * 0.05               # start temperature
    decay = 0.9999
    step = 0

    while time.time() < stop_time:
        i = _rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = _rng.integers(n)
        if i == j:
            continue
        # move up to a few percent of a[i] to a[j]
        frac = 0.03 * _rng.random()
        eps = frac * a[i]
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        max_val_new = np.fft.irfft(np.fft.rfft(a_new, n=fft_len) ** 2,
                                   n=fft_len)[:2 * n - 1].max()
        new_score = 2.0 * n * max_val_new
        dE = new_score - cur_score

        if dE < 0 or _rng.random() < np.exp(-dE / temp):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        temp *= decay
        step += 1
        if step % 5000 == 0:
            print(f"SA step {step:6d}: best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Random seed generation – several diverse families
# ----------------------------------------------------------------------
def _random_seed(n: int, kind: int) -> np.ndarray:
    """Create a random starting distribution on the simplex."""
    if kind == 0:
        # Sparse Dirichlet (α in [0.2, 0.8])
        alpha = 0.2 + 0.6 * _rng.random()
        samples = _rng.gamma(alpha, 1.0, size=n)
        a = samples / samples.sum()
    elif kind == 1:
        # Two‑sided geometric decay
        r = 0.85 + 0.3 * _rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        a = left + right
        a = a / a.sum()
    elif kind == 2:
        # Random bumps – a few entries hold most of the mass
        a = np.zeros(n)
        num = max(2, int(0.01 * n * (0.5 + _rng.random())))
        idx = _rng.choice(n, size=num, replace=False)
        weights = _rng.random(num) + 0.1
        a[idx] = weights
        a += 1e-6               # avoid exact zeros
        a = a / a.sum()
    else:
        # Absolute normal (≈ Dirichlet α=1)
        a = np.abs(_rng.standard_normal(n))
        a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function within the time budget.
    The routine mixes many short Adam runs from diverse random seeds and
    a final simulated‑annealing polishing phase.
    """
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – reuse the previous best if it exists
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            a_prev = np.asarray(previous, dtype=float)
            n = a_prev.shape[0]
            a_prev = np.maximum(a_prev, 0.0)
            if a_prev.sum() < 1e-12:
                a_prev = np.full(n, 1.0 / n)
            else:
                a_prev = a_prev / a_prev.sum()
        except Exception:
            a_prev = None
            n = 5146
    else:
        a_prev = None
        n = 5146                     # default length used in earlier runs

    # --------------------------------------------------------------
    # Baseline – evaluate the known construction (if any)
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    if a_prev is not None:
        fft_len = 1 << ((2 * n - 2).bit_length())
        max_val = np.fft.irfft(np.fft.rfft(a_prev, n=fft_len) ** 2,
                               n=fft_len)[:2 * n - 1].max()
        best_score = 2.0 * n * max_val
        best_a = a_prev.copy()
        print(f"Baseline (previous) score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop – many short Adam restarts
    # --------------------------------------------------------------
    restart_id = 0
    seed_kinds = [0, 1, 2, 3]            # four diverse families
    while time.time() < deadline - 8.0:  # keep a few seconds for polishing
        kind = seed_kinds[restart_id % len(seed_kinds)]

        init_a = _random_seed(n, kind)
        init_a = _project_simplex(init_a)   # just in case of tiny numerical drift

        # allocate a small time slice for this restart
        remaining = deadline - time.time()
        slice_time = time.time() + min(6.0, 0.25 * remaining)

        a_local, score_local = _adam_optimize(init_a, slice_time, restart_id)

        if score_local < best_score - 1e-12:
            best_score = score_local
            best_a = a_local.copy()
            print(f"Global best after restart {restart_id:02d}: {best_score:.6f}")

        if best_score < 1.503:
            print(f"Target reached after {restart_id+1} restarts: {best_score:.6f}")
            break

        restart_id += 1

    # --------------------------------------------------------------
    # Final polishing – simulated annealing (if time permits)
    # --------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 2.0:
        print("Starting final simulated‑annealing polishing.")
        best_a, best_score = _simulated_annealing(best_a, deadline)

    # --------------------------------------------------------------
    # Safety fallback
    # --------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # Return a plain Python list for the evaluator
    return [float(v) for v in best_a.tolist()]
