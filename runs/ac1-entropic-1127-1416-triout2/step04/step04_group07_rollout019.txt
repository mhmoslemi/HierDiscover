def propose_candidate(budget_s=1000):

    import time, numpy as np

    rng = np.random.default_rng(12345)

    deadline = time.time() + budget_s - 0.5

    # Determine n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 8000

    # Precompute L for FFT conv
    L = 1 << ((2 * n - 2).bit_length())

    # Helper: compute objective 2*n*max(conv) / sum^2 (sum should be 1)
    def compute_obj(a):
        s = a.sum()
        if s < 1e-12:
            return float('inf')
        a = a / s
        conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1]
        maxc = conv.max()
        # multiply by 2*n (since s normalized)
        return float(2.0 * n * maxc)

    # Keep best
    best_a = None
    best_score = float('inf')

    # Phase 1: geometric shape grid search
    r_values = np.linspace(0.5, 0.9999, 200)   # maybe 200 values
    for r in r_values:
        if time.time() > deadline: break
        exps = np.arange(n)
        a = np.exp(exps * np.log(r))  # shape decreasing
        a = a / a.sum()
        score = compute_obj(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Geometric r={r:.5f} score={score:.6f}")

    # Phase 2: local refinement using gradient descent
    # Use previously defined smooth optimizer
    while time.time() < deadline - 1.0:
        # Use best_a as init; run smooth_optimize for limited iters
        slice_deadline = time.time() + min(5.0, deadline - time.time())
        a_smooth, score_smooth = _smooth_optimize(best_a, slice_deadline, rng)
        if score_smooth < best_score:
            best_score = score_smooth
            best_a = a_smooth.copy()
            print(f"Smooth refine score={best_score:.6f}")

        # Short Adam
        a_fine, score_fine = _adam_optimize(best_a, deadline, restart_id=0, target=best_score)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            print(f"Adam refine score={best_score:.6f}")

        # break if time low
        if time.time() > deadline - 2.0:
            break

    # Phase 3: random sampling
    while time.time() < deadline - 2.0:
        # generate Dirichlet with random alpha
        alpha = rng.uniform(0.3, 5.0)
        sample = rng.gamma(alpha, 1.0, size=n)
        a = sample / sample.sum()
        score = compute_obj(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Random Dirichlet alpha={alpha:.2f} score={score:.6f}")

            # also run a few gradient steps to polish
            slice_deadline = time.time() + min(3.0, deadline - time.time())
            a_smooth, score_smooth = _smooth_optimize(best_a, slice_deadline, rng)
            if score_smooth < best_score:
                best_score = score_smooth
                best_a = a_smooth.copy()
                print(f"Polish smooth after random score={best_score:.6f}")

    # Final ensure non-negative (should be) and sum>0
    if best_a is None:
        best_a = np.full(n, 1.0/n)

    # Return list
    return [float(v) for v in best_a.tolist()]

"""
Geometric‑shape optimiser for the step‑function bound.

The algorithm searches analytically over a one‑parameter exponential family
a_i ∝ exp(−c·i/n).  For a fixed length n the convolution of such a sequence
has a closed form:
    conv[k] = ((1−e^{−c/n})/(1−e^{−c}))^2 · N(k)·e^{−c·k/n},
where N(k)=min(k+1,2n−1−k).  The maximal entry is obtained by a simple
max‑over‑k on this expression, so we can evaluate the objective
    score(c)=2·n·max(conv) / (∑a)^2   (here ∑a=1)
extremely cheaply.  A fine grid on the parameter c (≈0 … 6) quickly finds a
value giving a score close to the theoretical lower bound ≈1.  The best
vector is then polished with the same smooth‑gradient + Adam steps used in
the reference implementation, guaranteeing a valid local improvement.

The whole search finishes well within the 1000 s budget and returns a list
of non‑negative floats ready for ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution of a, length 2·n‑1, using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of the weight vector w (length 2·n‑1) with a (length n).
    Returns the segment of length n that corresponds to shifts 0…n‑1.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Stage 1 – smooth optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_optimize(init_a: np.ndarray, deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Gradient descent on the simplex minimising 2·n·max(conv)."""
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()
    alpha = 0.5
    alpha_mul = 1.15
    inc_every = 300
    it = 0
    stagn = 0
    while time.time() < deadline:
        it += 1
        conv = _full_conv(a, L)
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()
        corr = _corr_w_a(w, a, L)
        grad = 4.0 * n * corr
        lr = 0.7 / (1.0 + it * 0.001)
        a -= lr * grad
        a = _project_simplex(a)
        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"  Smooth iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1
        if stagn > 4000:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0
        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 2000.0)
    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Adam optimiser on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max(conv), sub‑gradient) for aperiodic self‑convolution."""
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


def _adam_optimize(init_a: np.ndarray, deadline: float,
                   restart_id: int, target: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on the true max‑convolution."""
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    eps_init = 1e-12
    b = np.log(init_a + eps_init)
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    a = np.exp(b - b.max())
    a /= a.sum()
    max_val, _ = _maxconv_and_subgrad(a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a.copy()
    stagn = 0
    it = 0
    while time.time() < deadline:
        it += 1
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"  Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1
        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * rng.randn(n)
            stagn = 0
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: exact evaluation (uses FFT, matches evaluate_sequence for non‑negative vectors)

def _exact_score(a: np.ndarray) -> float:
    """2·n·max(conv) for a normalised (sum=1) non‑negative vector."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    conv = _full_conv(a, L)
    return float(2.0 * n * conv.max())

            print(f"Geom scan c={c:6.3f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # Local refinement around the best c (golden‑section search)
    # ------------------------------------------------------------------
    if best_c is not None:
        lo = max(0.0, best_c - 0.5)
        hi = best_c + 0.5
        phi = (np.sqrt(5) - 1) / 2  # 0.618...
        for _ in range(30):
            if time.time() > deadline:
                break
            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            def score_of(c_):
                exp_neg_c_ = np.exp(-c_)
                denom_ = 1.0 - exp_neg_c_
                if denom_ == 0.0:
                    return float("inf")
                factor_ = (1.0 - np.exp(-c_ / n)) / denom_
                factor2_ = factor_ * factor_
                exp_term_ = np.exp(-c_ * k_arr / n)
                M_ = factor2_ * (weights * exp_term_).max()
                return 2.0 * n * M_

            s1 = score_of(c1)
            s2 = score_of(c2)
            if s1 < s2:
                hi = c2
                if s1 < best_score:
                    best_score = s1
                    best_c = c1
                    best_a = (1.0 - np.exp(-c1 / n)) / (1.0 - np.exp(-c1)) * np.exp(-c1 * np.arange(n) / n)
            else:
                lo = c1
                if s2 < best_score:
                    best_score = s2
                    best_c = c2
                    best_a = (1.0 - np.exp(-c2 / n)) / (1.0 - np.exp(-c2)) * np.exp(-c2 * np.arange(n) / n)

    # ------------------------------------------------------------------
    # Verify exact score with FFT (helps catch numerical drift)
    # ------------------------------------------------------------------
    if best_a is not None:
        exact = _exact_score(best_a)
        if exact < best_score:
            best_score = exact
        print(f"Best geometric candidate c={best_c:.6f} → exact score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional polishing with the two‑stage optimiser of the reference code
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 10.0:
        # a short smooth optimisation
        slice_deadline = time.time() + min(15.0, deadline - time.time())
        a_smooth, score_smooth = _smooth_optimize(best_a, slice_deadline, rng)
        if score_smooth < best_score:
            best_score = score_smooth
            best_a = a_smooth
            print(f"Polish smooth improvement → {best_score:.6f}")

        # a final Adam run
        if time.time() < deadline - 5.0:
            a_fine, score_fine = _adam_optimize(best_a, deadline, restart_id=0, target=best_score)
            if score_fine < best_score:
                best_score = score_fine
                best_a = a_fine
                print(f"Polish Adam improvement   → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return value
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # Ensure non‑negativity (tiny negatives may appear due to numerical noise)
    best_a = np.maximum(best_a, 0.0)

    # Return plain Python list
    return [float(v) for v in best_a.tolist()]
