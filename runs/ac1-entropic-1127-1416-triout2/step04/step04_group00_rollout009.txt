"""
Propose a step‑function whose evaluation score is as low as possible.

Algorithm
---------

1. **Random dense‑subset search**  
   Very large vectors (up to 32768 entries) are generated.  For each
   length *n* a density *p*∈[0.6,0.97] is chosen, a random set of
   ⌈p·n⌉ indices is selected and equal mass 1/k is placed on those
   positions (k = number of chosen indices).  The exact objective
   `2·n·max(conv(a,a))` (which coincides with `evaluate_sequence`
   after normalising the sum to 1) is evaluated via an FFT‑based
   convolution.  The best sequence found during this fast stochastic
   phase is kept.

2. **Smooth‑max gradient descent**  
   Starting from the best random candidate we minimise a smooth
   approximation of the max‑convolution:
        Φ(a)=2·n·log‑sum‑exp(α·(a∗a))/α .
   For a fixed soft‑max temperature α the gradient can be written
   analytically:
        ∇Φ = (2·n/α)·2·conv(p_rev , a) ,
   where `p` is the soft‑max over the convolution values and `p_rev`
   is `p` reversed.  The update is a projected (onto the probability
   simplex) stochastic gradient step with a diminishing step size.
   The temperature α is increased in several stages
   (5 → 20 → 80 → 300 → 1200) so that the optimisation first smoothes
   the objective and then concentrates on the true maximum.

3. **Fine‑grained hill‑climb**  
   A short stochastic hill‑climber (swap or mass‑transfer moves) is
   applied to polish the final vector.

All work respects the 1000 s budget; progress and intermediate scores
are printed.  The returned object is a plain Python list of non‑negative
floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using a real FFT (fast).
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for a vector whose entries sum to 1.
    Returns 2·n·max(a∗a) (exactly the value returned by `evaluate_sequence`
    when the input is normalised).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Very small stochastic hill‑climber that swaps indices or moves a fraction
    of the mass from one position to another.  The exact objective
    `compute_score` is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # typical fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)      # success → increase step
        else:
            step_scale *= 0.97                         # failure → decrease step

        max_steps -= 1

    return best_a.tolist(), best_score

# Smooth‑max gradient descent (stage 2)
# ----------------------------------------------------------------------


def smooth_gradient_descent(a_init: np.ndarray,
                           time_deadline: float,
                           alphas=(5, 20, 80, 300, 1200),
                           iters_per_alpha=5000,
                           step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed objective
        Φ(a)=2·n·log‑sum‑exp(α·(a∗a))/α .
    `alphas` is an increasing schedule; for each α a limited number of
    iterations is performed.  The step size decays as 1/√t.
    Returns the best vector found and its exact (non‑smoothed) score.
    """
    a = a_init.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for alpha in alphas:
        # protect against exceeding the global deadline
        for t in range(1, iters_per_alpha + 1):
            if time.time() >= time_deadline - 5.0:
                return best_a, best_score

            conv = convolution_fft(a, a)                      # full conv, length 2n‑1
            # soft‑max over the convolution values (numerically stable)
            weighted = alpha * conv
            max_w = np.max(weighted)
            exp_w = np.exp(weighted - max_w)
            p = exp_w / np.sum(exp_w)                         # shape (2n‑1,)

            # gradient of the smoothed objective (up to a constant factor)
            p_rev = p[::-1]
            grad_full = convolution_fft(p_rev, a)             # length 3n‑2
            grad = 2.0 * grad_full[n - 1:n - 1 + n]           # slice n elements

            # step size (diminishing)
            lr = step0 / np.sqrt(t)

            a = a - lr * grad
            a = project_to_simplex(a)

            # occasional exact evaluation to keep the best record
            if t % 100 == 0:
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"  ↪ α={alpha:4d}, iter={t:5d}, improved score={best_score:.6f}")

    return best_a, best_score
