def lp_step(f, M, t):
    n = f.shape[0]
    rows = 2*n - 1
    # Build convolution matrix A
    A = sp.diags(f, offsets=np.arange(n), shape=(rows, n), format='csc')
    b_ub = np.full(rows, M)
    # maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return f
    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0:
        return f
    g = b_opt / sum_b
    # update convex combination
    f_new = (1 - t) * f + t * g
    # ensure simplex
    f_new = project_to_simplex(f_new)

"""
Propose a low‑score step‑function (non‑negative vector) for the
evaluation function *evaluate_sequence*.

The search uses several complementary ideas:

* 1️⃣  Randomised exploration of the symmetric β‑family (including the
      classic arcsine distribution) and a simple power‑law family.
* 2️⃣  Multi‑stage local optimisation:
      – Adam on a smooth log‑sum‑exp surrogate of the max‑convolution,
      – projected sub‑gradient descent on the true max,
      – stochastic local search (Gaussian perturbations).
* 3️⃣  A short linear‑programming polishing phase (the “LP‑step” from the
      paper) which maximises the ℓ₁‑mass that can be added without
      increasing the current convolution bound.
* 4️⃣  A final polishing stage that repeats the above refinements.
All iterates are projected onto the probability simplex
(Σa = 1, a ≥ 0) so the fast surrogate `2·n·max(conv)` equals the
official score returned by *evaluate_sequence*.

The routine respects the supplied time budget (default 1000 s) and
always returns a non‑empty list of non‑negative numbers.
"""
import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (maximum value, index) of the self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)) for a already normalised
    (Σa = 1).  This equals the official evaluation value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Result is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_powerlaw_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Monotone decreasing power‑law shape a_i ∝ (i+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    base = (i + eps) ** (-p)
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # sum to 1
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of max

    # weighted gradient of the quadratic form
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 2n‑1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Adam optimiser (smooth surrogate)
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 120.0,
    noise_every: int = 1200,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """Adam on the smooth surrogate, projected on the simplex."""
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 120_000,
) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        k = int(np.argmax(conv))

        # gradient of the active quadratic term (k‑th convolution entry)
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """Random walk on the simplex, keeping improvements."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept improvements, or occasional uphill moves for escape
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(f"    Random local iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming polishing step (the “LP‑step” from the paper)
# ----------------------------------------------------------------------


def lp_improve(f: np.ndarray, M: float, t: float = 0.25) -> np.ndarray:
    """
    Solve  max Σ b_j   s.t.   f*b ≤ M, b ≥ 0.
    The LP is solved with SciPy’s high‑performance interior‑point
    method.  The returned vector is a convex combination
        f_new = (1‑t)·f + t·(b / Σb)
    projected back onto the simplex.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Convolution matrix: (f*b)_k = Σ_i f_i b_{k‑i}
    A = sp.diags(f, offsets=np.arange(n), shape=(rows, n), format="csc")
    b_ub = np.full(rows, M)

    # minimise -Σb  (i.e. maximise Σb)
    c = -np.ones(n)

    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs")
    if not res.success:
        return f

    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0:
        return f
    g = b_opt / sum_b
    f_new = (1.0 - t) * f + t * g
    f_new = project_to_simplex(f_new)
    return f_new

                max_iter=max_iter_sub,
            )

        # Stage C – stochastic local search
        if time.time() < deadline - 3.0:
            max_iter_loc = int(min(250_000, (deadline - time.time()) * 40))
            print(f"  Random local polishing (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.015,
                max_iter=max_iter_loc,
            )

        # Stage D – a few LP‑based polishing steps (the true LP from the paper)
        for lp_step in range(3):
            if time.time() >= deadline - 2.0:
                break
            M, _ = max_convolution_and_index(a)
            print(f"  LP polishing step {lp_step + 1}, current max conv = {M:.6e}")
            a = lp_improve(a, M, t=0.30)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3️⃣  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # Very simple safe fallback (uniform 256‑vector)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
