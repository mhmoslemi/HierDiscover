    """

    # Helper to compute objective

    candidates = []

    # previous best if exists
    if prev_arr is not None:

        s = a.sum()
        if s == 0:

def _maxconv_and_subgrad(a, fft_len):
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad
```

def _adam_optimize(init_a, deadline, restart_id, target=1.503):
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    eps_init = 1e-12
    b = np.log(init_a + eps_init)
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # evaluate init
    a0 = np.exp(b - b.max())
    a0 = a0 / a0.sum()
    max_val, _ = _maxconv_and_subgrad(a0, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0
    while time.time() < deadline:
        it += 1
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1
        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * np.random.randn(n)
            stagn = 0
        # Adam update
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return best_a, best_score

"""
Search for a low‑score step‑function sequence.

The algorithm consists of three stages:

1. **Warm‑start** – a few simple constructions (previous best, uniform,
   Dirichlet samples, symmetric geometric profiles) are evaluated and the
   best one is kept.

2. **Active‑set linear‑programming phase** – the positions where the
   aperiodic self‑convolution attains its maximum are identified.
   Keeping these constraints active we solve a tiny LP that maximises
   the ℓ₁‑norm of a new candidate while guaranteeing that its convolution
   with the current iterate does not exceed the current peak value.
   The new candidate is mixed with the current iterate (step‑size decays
   like 1/√t).  This rapidly reduces the peak of the convolution.

3. **Fine‑grained polishing** – after the LP phase we run a projected
   sub‑gradient method with a back‑tracking line‑search, finishing with a
   short Adam optimiser (the same routine used in the previous
   implementation).  The whole search respects a 1000 s budget and always
   returns a valid non‑negative list of floats.

The implementation only uses ``numpy`` and, when available, ``scipy`` for
the linear programmes.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full aperiodic convolution
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution of ``a`` (length 2·n‑1) using FFT length ``L``."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active entries are averaged)
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Return (max_conv, sub‑gradient) for the aperiodic self‑convolution of ``a``.
    The sub‑gradient is the average of the gradients from all maximising
    shift‑positions (ties are handled by averaging).
    """
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Small Adam optimiser on the true max‑convolution (used for polishing)
# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray, deadline: float,
                   restart_id: int, target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the non‑smooth objective.
    Returns the best pair (vector, score) found before ``deadline``.
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # initial evaluation
    a0 = np.exp(b - b.max())
    a0 = a0 / a0.sum()
    max_val, _ = _maxconv_and_subgrad(a0, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * np.random.randn(n)
            stagn = 0

        # Adam update (gradient of the original objective)
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to build the (active‑set) LP constraint matrix
# ----------------------------------------------------------------------
def _build_constraint_matrix(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """Return a matrix A such that (A @ b)_i = (a * b)_{shifts[i]}."""
    n = a.shape[0]
    K = shifts.shape[0]
    A = np.zeros((K, n), dtype=np.float64)

    for r, s in enumerate(shifts):
        low = max(0, s - (n - 1))
        high = min(s, n - 1)
        j = np.arange(low, high + 1, dtype=int)         # indices of b
        i = s - j                                         # corresponding indices of a
        A[r, j] = a[i]

    return A

    best_score = float("inf")
    for cand in candidates:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"Initial best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – active‑set LP improvement (uses at most ~30 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.33 * budget_s
    it_lp = 0
    stagn_lp = 0

    while time.time() < phase1_deadline and SCIPY_AVAILABLE:
        it_lp += 1
        conv = _full_conv(best_a, L)
        M = conv.max()
        cur_score = 2 * n * M

        # update global best if we are lucky (unlikely here, but cheap)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = best_a.copy()
            print(f"LP‑phase iter {it_lp:5d}: new best {best_score:.6f}")
            stagn_lp = 0
        else:
            stagn_lp += 1

        # choose the most violating shifts (top‑k of the convolution)
        top_k = min(30, max(5, int(0.005 * n)))      # a few dozen constraints
        idx = np.argpartition(conv, -top_k)[-top_k:]
        idx = np.sort(idx)                           # keep order stable

        # build the linear‑program matrix A·b ≤ M
        A = _build_constraint_matrix(best_a, idx)

        # linear programme: maximise   1ᵀb   subject to   A b ≤ M,  b ≥ 0
        c = -np.ones(n, dtype=np.float64)            # maximise sum(b)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=bounds, method='highs', options={'presolve': True})

        if res.success and res.x.sum() > 0:
            b_opt = res.x
            b_norm = b_opt / b_opt.sum()            # bring back onto simplex

            # a simple convex combination – step size decays like 1/√t
            t = 0.5 / math.sqrt(it_lp)
            new_a = (1.0 - t) * best_a + t * b_norm
            new_a /= new_a.sum()
            best_a = new_a
            stagn_lp = 0
        else:
            # fallback: add a tiny random jitter to escape a possible stall
            best_a += 0.001 * rng.standard_normal(n)
            best_a = _project_simplex(best_a)

        # occasional random restart if progress stalls for a while
        if stagn_lp > 2000:
            s = rng.gamma(0.5, 1.0, size=n)
            best_a = s / s.sum()
            stagn_lp = 0

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with back‑tracking line‑search
    # ------------------------------------------------------------------
    it_sg = 0
    stagn_sg = 0
    while time.time() < deadline - 5.0:
        it_sg += 1
        M, grad = _maxconv_and_subgrad(best_a, L)           # max‑conv and sub‑gradient
        cur_score = 2 * n * M

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = best_a.copy()
            stagn_sg = 0
            print(f"Sub‑grad iter {it_sg:5d}: new best {best_score:.6f}")
        else:
            stagn_sg += 1

        # back‑tracking line search
        step = 0.5 / math.sqrt(it_sg)          # initial guess
        while True:
            a_trial = best_a - step * grad
            a_trial = _project_simplex(a_trial)
            trial_score = _score(a_trial)
            if trial_score < cur_score:
                best_a = a_trial
                break
            step *= 0.5
            if step < 1e-12:                 # no progress possible
                break

        # occasional jitter when we stagnate
        if stagn_sg > 1500:
            best_a += 0.001 * rng.standard_normal(n)
            best_a = _project_simplex(best_a)
            stagn_sg = 0

        # random restart after a very long stall
        if stagn_sg > 5000:
            s = rng.gamma(0.5, 1.0, size=n)
            best_a = s / s.sum()
            stagn_sg = 0

    # ------------------------------------------------------------------
    # Phase 3 – short Adam polishing (if any time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_polished, score_polished = _adam_optimize(best_a, deadline,
                                                    restart_id=0, target=1.503)
        if score_polished < best_score:
            best_score = score_polished
            best_a = a_polished
            print(f"Adam polishing achieved new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & conversion to plain Python list
    # ------------------------------------------------------------------
    # Clip to the limits enforced by evaluate_sequence (0 ≤ x ≤ 1000)
    result = [float(min(1000.0, max(0.0, v))) for v in best_a.tolist()]

    # Show the value as seen by the official evaluator (helps debugging)
    try:                                   # pragma: no cover
        val = evaluate_sequence(result)
        print(f"Final evaluated score (via evaluate_sequence): {val:.6f}")
    except Exception:
        pass

    return result
