```
def fast_score(a):
    n = len(a)
    L = 1 << ((2*n -2).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a*fft_a, n=L)[:2*n-1]
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01: return np.inf
    return 2*n*max_conv/(s**2)
```

```
def anneal(a, deadline, rng):
    n = len(a)
    best_a = a.copy()
    best_score = fast_score(a)
    cur_a = a.copy()
    cur_score = best_score
    T0 = 0.1
    T_min = 1e-6
    # schedule based on remaining time: log-linear.
    while time.time() < deadline:
        # temperature schedule: linear down
        t = (deadline - time.time()) / (deadline - start_time)
        T = max(T_min, T0 * t)  # decreasing
        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # amount to transfer: delta = step * cur_a[i] maybe
        step = 0.05
        delta = step * cur_a[i]  # at most 5% of value
        if delta <= 0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # ensure non-negative
        # compute score
        score_new = fast_score(new_a)
        # accept
        if score_new < cur_score or rng.random() < np.exp(-(score_new - cur_score)/T):
            cur_a = new_a
            cur_score = score_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"Anneal iter {iters}: best = {best_score:.6f}")
        iters += 1
```

```
def _anneal_optimize(init_a: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = len(init_a)
    L = 1 << ((2*n - 2).bit_length())
    # Precompute function to compute score
    def compute_score(arr):
        fft_arr = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fft_arr * fft_arr, n=L)[:2*n-1]
        max_conv = conv.max()
        s = arr.sum()
        return 2 * n * max_conv / (s * s)

    cur_a = init_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    # Temperature schedule: start high maybe 0.1, gradually reduce
    start_T = 0.1
    end_T = 1e-6
    start_time = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # exponential cooling based on proportion of time left
        t_frac = (deadline - time.time()) / (deadline - start_time)
        T = end_T + (start_T - end_T) * t_frac

        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Transfer fraction of mass from i to j
        # If cur_a[i] small then move small amount
        max_transfer = cur_a[i] * 0.5  # can transfer up to 50% of mass at i
        if max_transfer <= 0:
            continue
        # Sample delta uniformly from [0, max_transfer]
        delta = rng.random() * max_transfer
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # Ensure sum unchanged, nonnegative.
        # Compute score
        new_score = compute_score(new_a)

        # Accept if improvement or by probability
        if new_score < cur_score - 1e-12 or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"Anneal iter {it}: best = {best_score:.6f}")

        # occasional jitter to avoid stagnation
        if it % 20000 == 0:
            # random perturbation small
            cur_a += 0.001 * rng.standard_normal(n)
            cur_a = np.maximum(cur_a, 0)
            s = cur_a.sum()
            if s > 0:
                cur_a /= s
            cur_score = compute_score(cur_a)
```

```
"""
Hybrid optimizer for the autocorrelation max‑norm problem.

1. Generates several feasible starting vectors (uniform, Dirichlet,
   geometric, previous best).
2. Refines each with a smooth log‑sum‑exp surrogate gradient descent
   projected onto the simplex.
3. Polishes the result with Adam on the true max‑convolution objective.
4. Final stage: guided simulated‑annealing / mass‑transfer moves that
   use the sub‑gradient of the max‑convolution to push mass from
   “high‑impact’’ coordinates to “low‑impact’’ ones.
5. Returns the best sequence found within the allotted time budget.
"""

import time
import math
import numpy as np

# Helper: projection onto simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# Full aperiodic self‑convolution via FFT
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


# Correlation of weight vector w (length 2n‑1) with a (length n)
def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


def _smooth_optimize(init_a: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Gradient descent on the simplex using a log‑sum‑exp surrogate."""
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())  # next power of two
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()

    alpha = 0.5
    alpha_mul = 1.15
    inc_every = 300
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1
        conv = _full_conv(a, L)
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()

        corr = _corr_w_a(w, a, L)
        grad = 4.0 * n * corr

        lr = 0.7 / (1.0 + it * 0.001)
        a -= lr * grad
        a = _project_simplex(a)

        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"Smooth iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if stagn > 4000:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 2000.0)

    return best_a, best_score


def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max_conv, sub‑gradient) for the aperiodic self‑convolution."""
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


def _adam_optimize(init_a: np.ndarray, deadline: float, restart_id: int, target: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on the true max‑convolution objective."""
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    a0 = np.exp(b - b.max())
    a0 = a0 / a0.sum()
    max_val, _ = _maxconv_and_subgrad(a0, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * np.random.randn(n)
            stagn = 0

        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


def _fast_score(a: np.ndarray, L: int) -> float:
    """Compute the exact evaluation score without the safety wrappers."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


def _anneal_optimize(init_a: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Guided simulated‑annealing that repeatedly transfers a small amount
    of mass from a high‑impact coordinate (large sub‑gradient) to a
    low‑impact one.  The temperature decays linearly with the remaining
    time.
    """
    n = init_a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    cur_a = init_a.copy()
    cur_score = _fast_score(cur_a, L)
    best_a = cur_a.copy()
    best_score = cur_score

    start_time = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # exponential cooling: temperature proportional to remaining fraction
        frac = (deadline - time.time()) / (deadline - start_time)
        T = max(1e-8, frac)  # simple linear schedule

        # with 30 % probability do a gradient‑guided transfer,
        # otherwise pick two random indices.
        if rng.random() < 0.3:
            # compute sub‑gradient of the current max
            max_val, grad = _maxconv_and_subgrad(cur_a, L)
            # indices with largest / smallest gradient
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                continue
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue

        # amount to move: a fraction of the source mass; shrink with temperature
        max_move = cur_a[i] * 0.5
        if max_move <= 0:
            continue
        # move size decreases as temperature drops
        move_frac = 0.05 * T
        delta = move_frac * cur_a[i]
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # normalise to sum‑1 (numerical drift)
        s = new_a.sum()
        if s <= 0:
            continue
        new_a /= s

        new_score = _fast_score(new_a, L)

        # Metropolis acceptance
        if new_score < cur_score - 1e-12 or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"Anneal iter {it:7d}: best = {best_score:.6f}")

        # occasional global jitter to avoid dead‑ends
        if it % 20000 == 0:
            cur_a += 0.001 * rng.standard_normal(n)
            cur_a = np.maximum(cur_a, 0)
            cur_a /= cur_a.sum()
            cur_score = _fast_score(cur_a, L)
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"Anneal jitter {it}: best = {best_score:.6f}")

    return best_a, best_score

"""
Hybrid optimiser for the autocorrelation max‑norm problem.

The algorithm proceeds in three stages:

1. **Smooth surrogate optimisation** – gradient descent on the
   log‑sum‑exp (soft‑max) surrogate of the convolution maximum,
   projected onto the simplex.

2. **Adam polishing** – optimisation of the true non‑smooth objective
   using the sub‑gradient of the max‑convolution (the active set).

3. **Guided simulated‑annealing** – repeatedly moves a small amount of
   mass from a “high‑impact’’ coordinate (large sub‑gradient) to a
   “low‑impact’’ one.  The temperature decays with the remaining time
   and Metropolis acceptance is used to escape shallow local minima.

Multiple random restarts with diverse initialisations (previous best,
uniform, Dirichlet, geometric) are performed until the time budget
expires.  The best sequence found is returned as a plain ``list`` of
floats, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (Wang & Carreira‑Perpiñán)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based)
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Aperiodic self‑convolution of a (length 2·n‑1) using FFT length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of weight vector w (length 2·n‑1) with a (length n).
    Returns the segment of length n corresponding to shifts 0…n‑1.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Stage 1 – smooth optimisation with a log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _smooth_optimize(init_a: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the simplex minimising
        2·n·M_α(a)   with   M_α = (1/α)·log Σ exp(α·conv)
    α is annealed during the run.  Returns the best vector found and its
    *exact* (non‑smooth) score.
    """
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())   # next power of two
    best_a = a.copy()
    best_score = 2.0 * n * _full_conv(a, L).max()

    alpha = 0.5
    alpha_mul = 1.15
    inc_every = 300
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1

        conv = _full_conv(a, L)
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)           # safe soft‑max
        w = exp_scaled / exp_scaled.sum()                  # Σ w = 1

        corr = _corr_w_a(w, a, L)
        grad = 4.0 * n * corr

        lr = 0.7 / (1.0 + it * 0.001)
        a -= lr * grad
        a = _project_simplex(a)

        score = 2.0 * n * _full_conv(a, L).max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"Smooth iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if stagn > 4000:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 2000.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (used by Adam & annealing)
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Return (max_conv, sub‑gradient) for the aperiodic self‑convolution.
    The sub‑gradient is the average of the gradients contributed by all
    active (max‑tight) convolution entries.
    """
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Stage 2 – Adam optimiser on the true max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   deadline: float,
                   restart_id: int,
                   target: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on the logits of a soft‑max representation."""
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    eps_init = 1e-12
    b = np.log(init_a + eps_init)               # logits

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    a0 = np.exp(b - b.max())
    a0 = a0 / a0.sum()
    max_val, _ = _maxconv_and_subgrad(a0, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * np.random.randn(n)
            stagn = 0

        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast exact evaluation (same formula as evaluate_sequence, without safety)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Stage 3 – guided simulated annealing
# ----------------------------------------------------------------------
def _anneal_optimize(init_a: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small amount of mass from a high‑impact index
    (large sub‑gradient) to a low‑impact one.  Acceptance follows a
    Metropolis rule with a temperature that decays linearly with the
    remaining time.
    """
    n = init_a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    cur_a = init_a.copy()
    cur_score = _fast_score(cur_a, L)
    best_a = cur_a.copy()
    best_score = cur_score

    start_time = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # Linear temperature schedule (remaining fraction of time)
        frac = (deadline - time.time()) / (deadline - start_time)
        T = max(1e-8, frac)

        # 30 % of the proposals are gradient‑guided
        if rng.random() < 0.3:
            _, grad = _maxconv_and_subgrad(cur_a, L)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                continue
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue

        # Move a fraction of the source mass; the fraction shrinks with T
        max_move = cur_a[i] * 0.5
        if max_move <= 0:
            continue
        move_frac = 0.05 * T
        delta = move_frac * cur_a[i]
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a /= new_a.sum()          # renormalise (scale‑invariant)

        new_score = _fast_score(new_a, L)

        # Metropolis acceptance
        if new_score < cur_score - 1e-12 or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"Anneal iter {it:7d}: best = {best_score:.6f}")

        # Global jitter every few thousand steps to avoid dead‑ends
        if it % 20000 == 0:
            cur_a += 0.001 * rng.standard_normal(n)
            cur_a = np.maximum(cur_a, 0)
            cur_a /= cur_a.sum()
            cur_score = _fast_score(cur_a, L)
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"Anneal jitter {it}: best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function sequence within the supplied
    time budget (seconds).  Returns the best sequence as a plain list
    of floats.
    """
    rng = np.random.default_rng(123456)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – use the previously found construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = prev.shape[0]
        a0 = np.maximum(prev, 0.0)
        tot = a0.sum()
        a0 = a0 / tot if tot > 0 else np.full(n, 1.0 / n)
    except Exception:
        # Fallback length – large enough to allow a good bound but still cheap
        n = 5000
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    L_global = 1 << ((2 * n - 2).bit_length())
    best_a = a0.copy()
    best_score = _fast_score(best_a, L_global)
    print(f"Initial score: {best_score:.6f}")

    restart = 0
    strategies = ["prev", "uniform", "dirichlet", "geometric"]

    while time.time() < deadline - 5.0:
        kind = strategies[restart % len(strategies)]

        # ------------------------------------------------------------------
        # Generate a feasible start point
        # ------------------------------------------------------------------
        if kind == "prev":
            a_start = best_a.copy()
            a_start += 0.001 * rng.standard_normal(n)
            a_start = np.maximum(a_start, 0)
            a_start /= a_start.sum()
        elif kind == "uniform":
            a_start = np.full(n, 1.0 / n)
        elif kind == "dirichlet":
            sample = rng.gamma(0.3, 1.0, size=n)
            a_start = sample / sample.sum()
        elif kind == "geometric":
            r = 0.95 + 0.05 * rng.random()
            left = r ** np.arange(n)
            right = r ** np.arange(n - 1, -1, -1)
            a_start = left + right
            a_start /= a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)

        # ------------------------------------------------------------------
        # Split the remaining budget for the three stages
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        slice1 = time.time() + min(25.0, 0.20 * remaining)   # smooth
        slice2 = time.time() + min(35.0, 0.30 * remaining)   # Adam

        # ------------------------------------------------------------------
        # Stage 1 – smooth optimisation
        # ------------------------------------------------------------------
        a_smooth, _ = _smooth_optimize(a_start, slice1, rng)

        # ------------------------------------------------------------------
        # Stage 2 – Adam polishing
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            a_adam, score_adam = _adam_optimize(a_smooth, slice2, restart, target=1.503)
        else:
            a_adam = a_smooth
            score_adam = _fast_score(a_adam, L_global)

        # ------------------------------------------------------------------
        # Stage 3 – guided annealing (uses the remaining time)
        # ------------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_final, score_final = _anneal_optimize(a_adam, deadline, rng)
        else:
            a_final = a_adam
            score_final = score_adam

        # ------------------------------------------------------------------
        # Update global best
        # ------------------------------------------------------------------
        for cand, sc in [(a_smooth, _fast_score(a_smooth, L_global)),
                         (a_adam, score_adam),
                         (a_final, score_final)]:
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"Global best after restart {restart:02d}: {best_score:.6f}")
                if best_score < 1.503:
                    print(f"Target reached (score={best_score:.6f})")
                    deadline = time.time()   # force early exit
                    break

        restart += 1

    # ------------------------------------------------------------------
    # Final sanity checks (non‑negative, reasonable sum)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total < 0.01:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a = best_a / total   # normalise – the score is scale‑invariant

    # Return a plain Python list as required by the evaluator
    return [float(v) for v in best_a.tolist()]
```
