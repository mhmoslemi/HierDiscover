"""
Simple but aggressive search for a low‑max‑autoconvolution step function.

The algorithm:

1.  Choose a set of lengths (we mainly use n=2048).  
2.  Generate a large pool of candidate shapes:
    * deterministic arcsine, uniform, triangular bumps,
    * systematic beta‑shapes (a = b) and power‑shapes
      covering a fine grid of the shape‑parameter,
    * random Dirichlet draws.
    All candidates are projected onto the probability simplex
    (Σa = 1) so that the official objective reduces to
    2·n·max(conv(a,a)).
3.  Keep the few best candidates (according to the fast FFT‑based
    surrogate) and refine each of them by three stages:
    * sub‑gradient descent on the exact max‑convolution,
    * Adam optimisation of a log‑sum‑exp smooth approximation,
    * a short simulated‑annealing polishing phase.
4.  Evaluate the refined vectors with the official ``evaluate_sequence``
    function and keep the best one.
5.  If time permits, a final polishing (sub‑gradient + Adam) is
    performed on the overall best vector.

All heavy work is guarded by the time‑budget (default 1000 s); the
routine always returns a non‑empty list, so the harness never receives
``inf``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for the official score when Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Duchi‑et‑al. (2008).
    """
    n = v.size
    if np.all(v <= 0):
        return np.full(n, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine shape a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, w_arc: float) -> np.ndarray:
    """Mixture (w)·arcsine + (1‑w)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Optimisers (all work on vectors that already satisfy Σ=1)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max); the sum‑constraint is
    enforced later by a projection onto the simplex.
    """
    n = a.size
    conv = convolution_fft(a, a)         # length 2n‑1
    max_c = conv.max()

    # stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / exp_term.sum()        # sum=1

    # weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)          # length n + (2n‑1) - 1 = 3n‑2
    L = 2 * n - 1
    weighted = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return 2.0 * n * weighted


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth (log‑sum‑exp) surrogate.
    Returns the best vector seen (projected onto the simplex) together
    with its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule for λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)
        cur_sc = compute_score(a_new)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Exact sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t); projection onto the simplex keeps Σ=1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()

        # tolerance to capture all indices attaining the max
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Very light Metropolis‑style local search.
    Small random transfers of mass preserve the sum and non‑negativity.
    The fast surrogate (2·n·max) is used for the Metropolis test.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # transfer up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        # Metropolis acceptance
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine respects the time budget, explores several families
    of shapes, refines the best ones with sub‑gradient, Adam and simulated
    annealing, and finally returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0        # spare second for cleanup

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback if a previous run stored a sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose lengths (larger lengths give a finer “continuum”
    #     approximation but are a bit slower).  We start with 2048.
    # ------------------------------------------------------------------
    lengths = [2048, 1024, 4096]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break

        print(f"\n=== Length n = {n} ===")

        # ----------------- 1a. Generate a rich pool of candidates -------------
        pool: List[Tuple[float, np.ndarray]] = []

        # deterministic starters
        pool.append((compute_score(arcsine_initial(n)), arcsine_initial(n)))
        pool.append((compute_score(np.full(n, 1.0 / n)), np.full(n, 1.0 / n)))

        # systematic beta / power grid
        for p in np.arange(0.35, 1.55, 0.05):          # p = exponent in power‑shape
            for eps in (0.0, 1e-3, 0.1, 1.0):
                cand = generate_power_shape(n, p, eps)
                pool.append((compute_score(cand), cand))

        for a in np.arange(0.2, 2.21, 0.2):           # symmetric beta parameters
            for eps in (0.0, 0.5, 2.0):
                cand = generate_beta_shape(n, a, a, eps)
                pool.append((compute_score(cand), cand))

        # random Dirichlet draws
        for _ in range(200):
            dir_vec = rng.random(n)
            dir_vec /= dir_vec.sum()
            pool.append((compute_score(dir_vec), dir_vec))

        # triangular bumps
        for _ in range(100):
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

        # mixture of arcsine & uniform
        for _ in range(100):
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # keep the most promising seeds
        pool.sort(key=lambda x: x[0])
        top_vectors = [vec for _, vec in pool[:6]]
        print(f"  Best raw surrogate after sampling: {pool[0][0]:.6f}")

        # ----------------- 2.  Refine each seed -----------------
        for idx, seed_vec in enumerate(top_vectors):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Refine seed #{idx+1}/{len(top_vectors)} (n={n})")

            # Sub‑gradient descent (exact max)
            remaining = deadline - time.time()
            max_iter_sub = int(min(120_000, remaining * 30))
            refined = subgradient_descent_exact(seed_vec, deadline,
                                                max_iter=max_iter_sub,
                                                step0=0.04)

            # Adam on smooth surrogate
            if time.time() < deadline - 3.0:
                remaining = deadline - time.time()
                max_iter_adam = int(min(100_000, remaining * 30))
                refined, _ = adam_optimize_smooth(refined,
                                                  deadline,
                                                  max_iter=max_iter_adam,
                                                  lr0=0.12,
                                                  lam_start=2.0,
                                                  lam_end=120.0,
                                                  noise_every=3000,
                                                  noise_scale=2e-5)

            # Simulated‑annealing polishing
            if time.time() < deadline - 2.0:
                remaining = deadline - time.time()
                max_iter_sa = int(min(120_000, remaining * 30))
                refined = simulated_annealing(refined,
                                              rng,
                                              deadline,
                                              max_iters=max_iter_sa,
                                              init_temp=0.025,
                                              decay=0.99995)

            # Official evaluation
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"  → official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        it_sub = int(min(250_000, remaining * 30 // 2))
        it_adam = int(min(200_000, remaining * 30 // 2))

        print("\nFinal global polishing (sub‑gradient + Adam)…")
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.02)
        a, _ = adam_optimize_smooth(a,
                                    deadline,
                                    max_iter=it_adam,
                                    lr0=0.09,
                                    lam_start=2.0,
                                    lam_end=150.0,
                                    noise_every=2000,
                                    noise_scale=1e-5)

        try:
            val = evaluate_sequence(a.tolist())
            print(f"Final polished official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print("*** Updated global best after final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_val:.6f}")

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
