"""
Search for a non‑negative step function that minimizes

        2·n·max(conv(a,a))   (a is normalised to Σa = 1)

The search proceeds in four phases:

1. **Exploratory sampling** – for each admissible length we evaluate
   a rich family of symmetric beta‑type shapes (including uniform,
   arcsine, quadratic, sinusoidal) on a fine grid of the shape
   exponent.  The best raw candidate for that length is kept.

2. **Adam optimisation** – the selected candidate is refined with
   Adam on the smooth log‑sum‑exp surrogate of the max‑convolution.
   The temperature λ is annealed from 1 to a large value (≈120) so
   that the surrogate converges to the true max.

3. **L2 flattening** – a second optimisation stage targets a
   *uniform* self‑convolution.  The loss is
       ‖a*a – 1/(2n‑1)·1‖₂²,
   whose gradient can be written as a simple convolution.
   Projected gradient descent with a diminishing stepsize reduces
   the peak of the convolution further.

4. **Final polishing** – the globally best vector found so far is
   refined again with Adam (long run) + L2 flattening.  The routine
   respects the overall time budget (default 1000 s) and guarantees
   a non‑empty return value.

All helper functions are top‑level (no closures) and the code uses
only NumPy and the already‑imported ``evaluate_sequence`` function.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, arg‑max index) of the self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate (exact score for a normalised vector):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Families of symmetric shapes
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    """
    Symmetric beta‑family shape:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to sum 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def shape_power_mirror(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """
    a_i ∝ ((i+½)*(n‑½‑i)+eps)^{p}.
    For p>0 the vector vanishes at the ends,
    for p<0 it has singularities (arcsine‑like).
    """
    i = np.arange(n, dtype=float)
    base = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def shape_sin(n: int) -> np.ndarray:
    """Symmetric sine shape, zero at the two boundaries."""
    i = np.arange(1, n + 1, dtype=float)
    vals = np.sin(np.pi * i / (n + 1))
    total = vals.sum()
    return vals / total


def shape_quadratic_zero(n: int) -> np.ndarray:
    """a_i ∝ i·(n‑1‑i) (exactly zero at the ends)."""
    i = np.arange(n, dtype=float)
    vals = i * (n - 1 - i)
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total

# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # softmax weights, sum to 1
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of max

    # weighted sum of sub‑gradients:
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    # derivative of the whole objective 2·n·B / s²
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 120.0,
    noise_every: int = 1200,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic schedule for λ (fast at the start, strong near the end)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        # occasional tiny stochastic push to escape plateaus
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# L2 flattening of the convolution (aim for a constant)
# ----------------------------------------------------------------------


def l2_convolution_gradient(a: np.ndarray) -> np.ndarray:
    """
    Gradient of ½·‖a*a – C·1‖₂² where C = 1/(2n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    C = 1.0 / (2 * n - 1)
    residual = conv - C
    rev = residual[::-1]
    conv_rev = convolution_fft(a, rev)
    L = 2 * n - 1
    grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return grad


def l2_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.4,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected gradient descent on the L2 flattening objective.
    The step size decays as step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad = l2_convolution_gradient(a)

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    L2‑grad iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exploration for a single length
# ----------------------------------------------------------------------


def explore_length(
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> Tuple[np.ndarray, float]:
    """
    Return the best vector found for a given length n before the
    supplied deadline.  The routine mixes deterministic starters,
    a fine grid over several shape families and a short random
    beta‑family sampling.  Only the fast ``compute_score`` is
    used during exploration.
    """
    best_a = None
    best_sc = float("inf")

    # 1. Deterministic starters
    starters = [
        np.full(n, 1.0 / n),                     # uniform
        generate_beta_shape(n, 0.5, 0.5, 1e-9),   # standard arcsine
        shape_sin(n),                            # sine
        shape_quadratic_zero(n),                 # quadratic vanishing at ends
    ]
    for st in starters:
        sc = compute_score(st)
        if sc < best_sc:
            best_sc = sc
            best_a = st

    # 2. Systematic grid over symmetric beta (α = β)
    alphas = np.linspace(0.2, 2.4, 23)   # ~0.1 step
    for alpha in alphas:
        cand = generate_beta_shape(n, alpha, alpha, eps=1e-9)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand

    # 3. Grid over power‑mirror exponent p
    ps = np.linspace(-0.5, 1.5, 21)     # –0.5 (arcsine) … 1.5 (strong interior)
    for p in ps:
        cand = shape_power_mirror(n, p, eps=1e-9)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand

    # 4. Random beta‑family samples (quick)
    max_samples = 300
    for _ in range(max_samples):
        if time.time() >= deadline - 8.0:
            break
        alpha = rng.uniform(0.05, 2.5)
        beta = rng.uniform(0.05, 2.5)
        eps = rng.uniform(1e-9, max(0.01, 0.02 * n))
        cand = generate_beta_shape(n, alpha, beta, eps)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand

    # Defensive fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_sc = compute_score(best_a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine combines an extensive grid search over beta‑family
    shapes with a two‑stage optimisation (Adam on a smooth max‑surrogate
    followed by an L2‑flattening of the convolution).  It tries several
    vector lengths up to 8192 and returns the best vector found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds spare for the final return
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Easy fallback (if the notebook already provided one)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Loop over candidate lengths
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --- exploration phase -------------------------------------------------
        local_a, local_sc = explore_length(n, deadline, rng)
        print(f"  Best raw score after exploration = {local_sc:.6f}")

        # --- refinement phase --------------------------------------------------
        # allocate a reasonable chunk of the remaining budget to this length
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # 1) Adam on smooth max
        max_iter_adam = int(min(120_000, remaining * 30))
        print(f"  Adam refinement (max_iter={max_iter_adam}) …")
        refined_a, _ = adam_optimize(
            initial_a=local_a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.18,
            lam_start=1.0,
            lam_end=120.0,
            noise_every=1000,
            noise_scale=5e-5,
        )

        # 2) L2‑flattening stage
        remaining = deadline - time.time()
        if remaining > 3.0:
            max_iter_l2 = int(min(100_000, remaining * 30))
            print(f"  L2‑flattening (max_iter={max_iter_l2}) …")
            refined_a, _ = l2_optimize(
                refined_a,
                time_deadline=deadline,
                step0=0.35,
                max_iter=max_iter_l2,
            )

        # Official (slow) evaluation – one per length
        val = evaluate_sequence(refined_a.tolist())
        print(f"  → official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined_a.tolist()
            print(f"  *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Final polishing of the global best ===")
        a = np.array(best_seq, dtype=float)

        # 2.A Adam long run
        remaining = deadline - time.time()
        max_iter_adam = int(min(350_000, remaining * 40))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.15,
            lam_start=1.0,
            lam_end=150.0,
            noise_every=1500,
            noise_scale=4e-5,
        )

        # 2.B L2‑flattening polishing
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iter_l2 = int(min(150_000, remaining * 35))
            print(f"  L2 polishing (max_iter={max_iter_l2}) …")
            a, _ = l2_optimize(
                a,
                time_deadline=deadline,
                step0=0.3,
                max_iter=max_iter_l2,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
