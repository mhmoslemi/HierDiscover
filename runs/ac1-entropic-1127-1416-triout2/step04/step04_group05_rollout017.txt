def smoothed_grad(a, conv, beta):
    # conv = a*a
    # compute w = softmax(beta * conv)
    maxc = conv.max()
    # for numeric stability: exponentiate (conv - maxc)
    exp_vals = np.exp(beta * (conv - maxc))
    w = exp_vals / exp_vals.sum()
    # compute gradient via FFT
    a_rev = a[::-1]
    size = len(conv)  # 2n-1
    fft_w = np.fft.rfft(w, n=size)
    fft_rev = np.fft.rfft(a_rev, n=size)
    corr = np.fft.irfft(fft_w * fft_rev, n=size)
    # gradient_i = 2 * corr[i + n - 1] (since a_rev length n gave conv length 2n-1)
    grad = 2.0 * corr[(n-1):(n-1 + n)]  # slice length n
    return grad

def project_simplex(v):
    # if sum <= 0? but we ensure nonnegative after subtract.
    # Implementation: compute sorted v in descending order, find rho, etc.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    # find rho = last true index
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑height sequence with a small value of ``evaluate_sequence``.

Algorithm
---------
1.  Determine the problem size n.  If a previous solution
    ``height_sequence_1`` exists we keep its length, otherwise we use
    n = 5146 (the length used in the reference run).

2.  Initialise a pool of candidate vectors:
    * uniform vector,
    * a few Beta‑distributed vectors  β(α,α) with 0.51 ≤ α ≤ 2.0,
    * a handful of random Dirichlet draws.
    The best of these candidates (in terms of the exact objective
    ``2·n·max(conv)``) becomes the starting point.

3.  Smoothed‑gradient descent.
    We minimise a smooth approximation of the max‑convolution:
        Fβ(a) = (1/β)·log Σk exp(β·c_k) ,   c = a * a .
    Its gradient is
        ∇Fβ(a)_i = 2· Σk w_k·a_{k‑i},
    where w_k = softmax(β·c_k) .
    The gradient is computed with a single FFT‑based convolution.
    After each step we project back onto the probability simplex
    (non‑negative entries, sum = 1).

    The smoothing parameter β is increased geometrically (β←β·1.05)
    from 0.5 up to ≈20, so the optimisation first sees a very smooth
    landscape and gradually focuses on the true max.

4.  The best vector observed during the whole run is returned.
    The vector is already normalised; the evaluation routine will
    therefore compute exactly the same score as our internal
    ``compute_score`` function.

The whole routine respects the 1000 s time budget and never produces
negative or NaN entries, guaranteeing that ``evaluate_sequence`` will
accept the output.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a probability vector (sum == 1).
    Returns 2·n·max(conv) where conv = a * a (linear convolution).
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    """
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow for extreme α,β
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | sum(x)=1, x_i >= 0 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    # shift to guarantee positivity
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / float(rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def smoothed_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max: F_beta(a) = (1/beta)·log Σ exp(beta·conv).
    Returns ∇F_beta(a) evaluated at the given probability vector a.
    """
    n = a.shape[0]
    # soft‑max weights (stable)
    maxc = conv.max()
    tmp = beta * (conv - maxc)               # shift to avoid overflow
    tmp = np.clip(tmp, -700, 700)            # safety for exp
    w = np.exp(tmp)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a_rev * w  (see analysis for indexing)
    a_rev = a[::-1]
    size = conv.size                          # 2*n‑1
    fft_w = np.fft.rfft(w, n=size)
    fft_rev = np.fft.rfft(a_rev, n=size)
    corr = np.fft.irfft(fft_w * fft_rev, n=size)

    # gradient_i = 2 * corr[i + n - 1]
    grad = 2.0 * corr[(n - 1):(n - 1 + n)]
    return grad

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for returning the result
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1 – problem size & initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        n = 5146
        init = None
        print(f"No existing sequence – starting from length {n}.")

    # --------------------------------------------------------------
    # 2 – build a small pool of simple initial vectors
    # --------------------------------------------------------------
    candidates = []

    # uniform start (always feasible)
    candidates.append(np.ones(n, dtype=float) / n)

    # Beta(α,α) family – sweep α in a region that favours edge mass
    alphas = np.concatenate((
        np.linspace(0.51, 0.90, 20),
        np.linspace(0.95, 2.0, 15)
    ))
    for a in alphas:
        cand = beta_sequence(a, a, n)
        candidates.append(cand)

    # a few random Dirichlet draws for diversity
    for _ in range(5):
        cand = np.random.dirichlet(np.ones(n))
        candidates.append(cand)

    # optionally include the previous solution (if any)
    if init is not None:
        candidates.append(init / init.sum())

    best_a = None
    best_score = float("inf")
    for v in candidates:
        sc = compute_score(v)
        if sc < best_score:
            best_score = sc
            best_a = v.copy()
    print(f"Best seed candidate score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3 – Smoothed gradient‑descent phase
    # --------------------------------------------------------------
    a = best_a.copy()
    beta = 0.5                     # start with a very smooth surrogate
    beta_max = 20.0                # final smoothing (close to true max)
    beta_mul = 1.05                # geometric increase per iteration
    step0 = 0.3                    # base learning rate
    it = 0
    no_improve = 0

    while time.time() < deadline - 5.0:          # keep a few seconds buffer
        # compute convolution (full length = 2*n‑1)
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)

        # gradient of the smoothed max
        grad = smoothed_gradient(a, conv, beta)

        # diminishing stepsize (square‑root schedule)
        step = step0 / math.sqrt(it + 1.0)

        # projected gradient step
        a_next = a - step * grad
        a_next = project_simplex(a_next)

        # evaluate the true (non‑smoothed) objective
        sc_next = compute_score(a_next)

        if sc_next < best_score - 1e-12:
            best_score = sc_next
            best_a = a_next.copy()
            no_improve = 0
            print(f"Iter {it:6d}: new best {best_score:.6f}")
        else:
            no_improve += 1

        a = a_next
        it += 1

        # increase smoothing parameter slowly
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # occasional random restart if we are stuck for a long time
        if no_improve >= 20000:
            a = np.random.dirichlet(np.ones(n))
            a = project_simplex(a)      # ensure exact sum‑1
            no_improve = 0
            print(f"Iter {it:6d}: random restart (no improvement)")

    print(f"Gradient descent finished after {it} iterations; best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4 – return best sequence (already normalised)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
