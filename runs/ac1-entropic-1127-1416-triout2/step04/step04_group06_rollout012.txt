"""
Propose a non‑negative step‑function (probability vector) that
approximately minimises
    2·n·max(conv(a,a))
by using a soft‑max smoothed max, Adam optimisation on the log‑space
variables, optional symmetry projection and a short simulated‑annealing
refinement.  Multiple random restarts are performed within the time
budget.  The best vector found is returned.
"""

import time
import math
import numpy as np

def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to sum = 1."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s

def compute_score(a: np.ndarray) -> float:
    """Exact evaluate_sequence value for a vector that already sums to 1."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()

def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    # Use log‑space to avoid under/overflow.
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()

def init_dirichlet_random(n: int) -> np.ndarray:
    """Sample a Dirichlet(1,…,1) vector."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)

def init_edge_concentrated_random(n: int) -> np.ndarray:
    """Linear “U‑shape”: larger values near the two ends."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx).astype(float)
    a = (n - dist) + 1.0      # edges receive larger weight
    return simplex_normalise(a)

def init_sparse_random(n: int, k: int = 12) -> np.ndarray:
    """Place mass on k random positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)

def init_exponential_decay_random(n: int) -> np.ndarray:
    """Exponential decay from both ends towards the centre."""
    lam = np.random.uniform(0.001, 0.01)   # decay rate
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    dist = np.abs(idx - centre)
    w = np.exp(-lam * dist)
    return simplex_normalise(w)

def init_double_spike_random(n: int) -> np.ndarray:
    """Two narrow spikes at the two ends."""
    a = np.zeros(n, dtype=float)
    # allocate a random split of the total mass
    split = np.random.rand()
    a[0] = split
    a[-1] = 1.0 - split
    return simplex_normalise(a)

def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over convolution values with smoothing parameter α."""
    maxc = conv.max()
    # shift for numerical stability
    shifted = alpha * (conv - maxc)
    # clip to avoid overflow
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s == 0 or not np.isfinite(s):
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s

def cross_correlation(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k p_k·a_{k‑i} for i=0,…,n‑1.
    Implements non‑circular correlation via FFT.
    """
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2
    fft_a = np.fft.rfft(a, n=L)
    fft_p = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_p), n=L)
    offset = len(p) - 1
    return corr[offset:offset + n]

def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force mirror symmetry a_i = a_{n‑1‑i} and renormalise."""
    a_sym = (a + a[::-1]) * 0.5
    return simplex_normalise(a_sym)

def adam_step(z: np.ndarray, m: np.ndarray, v: np.ndarray,
              grad: np.ndarray, t: int,
              lr: float = 0.02,
              beta1: float = 0.9, beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Perform one Adam update on logits z."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** (t + 1))
    v_hat = v / (1.0 - beta2 ** (t + 1))
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v

            # ----- 7) Adam update ----------------------------------------------------
            z, m, v = adam_step(z, m, v, grad_z, it, lr=lr,
                                beta1=beta1, beta2=beta2, eps=eps)

            # ----- 8) occasional exact evaluation ------------------------------------
            if (it & 31) == 0:
                score = compute_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    print(f"[Restart {restart}] it {it:5d} → new best {best_score:.6f}")

        cur_score = best_score
        # initialise a temperature proportional to the current objective

        T_end = cur_score * 1e-5
        sa_iter = 0

            # pick a distinct target uniformly

            cand[tgt] += delta

"""
Search for a non‑negative step‑function (probability vector) that
minimises the evaluation function

    evaluate_sequence(a) = 2·n·max(conv(a,a))   (sum(a)=1)

The algorithm uses:

1. Multiple random restarts with diverse simple initialisers.
2. Adam optimisation on the log‑space (softmax) parameters of the vector.
   The objective is the smooth approximation
       f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
   where a*a denotes linear self‑convolution.
   The gradient of f_α w.r.t. a is the cross‑correlation of a with the
   soft‑max weights over the convolution entries.
3. A hard projection onto the mirror‑symmetric subspace (a_i = a_{n‑1‑i})
   after every few iterations.
4. A short simulated‑annealing (mass‑transfer) refinement.
5. All candidates are kept on the simplex by the soft‑max parametrisation;
   positivity is therefore guaranteed.

The routine runs until the supplied time budget expires (≈1000 s) and
returns the best vector found (as a list of floats, sum≈1).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a vector that already sums to 1."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # protect against overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_dirichlet_random(n: int) -> np.ndarray:
    """Sample a Dirichlet(1,…,1) vector."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated_random(n: int) -> np.ndarray:
    """Linear “U‑shape”: larger values near the two ends."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx).astype(float)
    a = (n - dist) + 1.0          # edges receive larger weight
    return simplex_normalise(a)


def init_sparse_random(n: int, k: int = 12) -> np.ndarray:
    """Place mass on k random positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


def init_exponential_decay_random(n: int) -> np.ndarray:
    """Exponential decay from both ends towards the centre."""
    lam = np.random.uniform(0.001, 0.01)
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    dist = np.abs(idx - centre)
    w = np.exp(-lam * dist)
    return simplex_normalise(w)


def init_double_spike_random(n: int) -> np.ndarray:
    """Two narrow spikes at the two ends."""
    a = np.zeros(n, dtype=float)
    split = np.random.rand()
    a[0] = split
    a[-1] = 1.0 - split
    return simplex_normalise(a)


def random_initial_vector(n: int) -> np.ndarray:
    """Pick one of the simple families at random and generate a start vector."""
    choice = np.random.randint(6)
    if choice == 0:
        return beta_sequence(np.random.uniform(0.2, 5.0),
                             np.random.uniform(0.2, 5.0), n)
    if choice == 1:
        return init_dirichlet_random(n)
    if choice == 2:
        return init_edge_concentrated_random(n)
    if choice == 3:
        return init_sparse_random(n, k=np.random.randint(5, 30))
    if choice == 4:
        return init_exponential_decay_random(n)
    return init_double_spike_random(n)


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over convolution entries with smoothing parameter α."""
    maxc = conv.max()
    shifted = alpha * (conv - maxc)
    shifted = np.clip(shifted, -700.0, 700.0)   # safety for exp
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s == 0 or not np.isfinite(s):
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def cross_correlation(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k p_k·a_{k‑i}  for i = 0,…,n‑1 (non‑circular correlation).
    Implemented with FFT for O(n log n) cost.
    """
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2
    fft_a = np.fft.rfft(a, n=L)
    fft_p = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_p), n=L)
    offset = len(p) - 1
    return corr[offset:offset + n]


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry a_i = a_{n‑1‑i} and renormalise."""
    a_sym = (a + a[::-1]) * 0.5
    return simplex_normalise(a_sym)


def adam_step(z: np.ndarray, m: np.ndarray, v: np.ndarray,
              grad: np.ndarray, t: int,
              lr: float = 0.02,
              beta1: float = 0.9, beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam update on the log‑space variables z.
    Returns the updated (z, m, v).
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** (t + 1))
    v_hat = v / (1.0 - beta2 ** (t + 1))
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v
