def refine_exponential(a, deadline):
    a = normalize(a)
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    it_total = 0
    # set schedule of tau series
    tau_vals = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for tau in tau_vals:
        # adapt iteration count based on remaining time
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        max_iter = int(min(50000, max(1000, remaining * 12)))  # approximated speed
        for it in range(1, max_iter+1):
            if time.time() >= deadline:
                break
            conv = convolution_fft(a, a)
            max_val = np.max(conv)
            # compute softmax weights
            w_raw = np.exp(conv / tau)
            w = w_raw / np.sum(w_raw)
            grad = gradient_from_weights(a, w)  # proper shape
            # step size schedule decreasing
            eta = 0.5 / np.sqrt(it_total + it)  # constant schedule
            a = a * np.exp(-eta * grad)    # exponentiated
            a = normalize(a)
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # after each tau stage maybe also do a scarcity: if best not changed, continue
    return best_a, best_score

"""
propose_candidate.py

Search for a non‑negative step‑function (vector of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The code follows a multi‑stage strategy:

1. **Parametric seeding** – generate a diverse set of candidate shapes
   (arcsine, power‑law, symmetric beta, mixtures, Dirichlet draws,
   and up‑sampled best solutions from previous lengths).

2. **Soft‑max mirror descent** – for each seed run a few “smooth”
   optimisation phases where the non‑smooth max is replaced by a
   temperature‑controlled soft‑max.  The gradient of the surrogate is
   obtained analytically via a single FFT; updates are performed with an
   exponentiated‑gradient (multiplicative) step and a projection onto the
   probability simplex.

3. **Exact sub‑gradient polishing** – after the smooth phases the
   true max‑objective is tackled with a conventional sub‑gradient method
   (the same as in the reference implementation) to obtain the final
   answer.

The routine respects a user‑supplied time budget (default 1000 s).  It
always returns a feasible list of non‑negative floats; the official
`evaluate_sequence` function is called only for the final verification
of each promising candidate (reducing overhead).

The algorithm improves on the baseline by:
    * richer initial families (including sparse Dirichlet draws);
    * a temperature schedule that gradually sharpens the objective;
    * adaptive step‑size (1/√t) and exponentiated updates,
      keeping iterates on the simplex without explicit projections;
    * a final exact sub‑gradient polishing stage.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Zero‑pad to the next power of two for speed.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the scale‑invariant objective.
    Assumes a is non‑negative (it will be normalised to sum=1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # a is already on the probability simplex, sum == 1
    return 2.0 * n * max_val


def normalize(v: np.ndarray) -> np.ndarray:
    """
    Project a non‑negative vector onto the probability simplex
    (non‑negative, sum == 1).  If the total mass is zero, fall back
    to uniform.
    """
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k  w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    w : np.ndarray, shape (2n‑1,)
        Weight vector for each entry of the self‑convolution.
    Returns
    -------
    grads : np.ndarray, shape (n,)
        Gradient of the weighted sum of convolution entries.
    """
    # Reverse w because convolution implements (a * rev(w))[i] = Σ_j a_j w_{i-j}
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)                 # length = n + (2n-1) - 1 = 3n-2
    idx = 2 * len(a) - 2                             # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution ≈ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric beta‑like shape with equal parameters a."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet_alpha(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Sample from a Dirichlet distribution with concentration `alpha`."""
    samp = rng.gamma(shape=alpha, scale=1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Helper for cross‑length seeding
# ----------------------------------------------------------------------


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    interpolated = np.interp(x_new, x_old, seq)
    interpolated = np.maximum(interpolated, 0.0)
    return normalize(interpolated)


# ----------------------------------------------------------------------
# Mirror‑descent with a soft‑max surrogate
# ----------------------------------------------------------------------


def refine_softmax(
    a: np.ndarray,
    deadline: float,
    *,
    tau_schedule: Tuple[float, ...] = (0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005),
    base_eta: float = 0.7,
    max_iter_per_stage: int = 8_000,
) -> Tuple[np.ndarray, float]:
    """
    Smooth optimisation using a temperature‑controlled soft‑max
    approximation of the max‑function.
    Returns the best vector encountered and its exact (non‑smooth) score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    total_iters = 0
    n = len(a)

    for tau in tau_schedule:
        if time.time() >= deadline:
            break

        # Determine a safe iteration budget for this temperature.
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        it_budget = int(min(max_iter_per_stage, max(1_000, remaining * 12.0)))

        for it in range(1, it_budget + 1):
            if time.time() >= deadline:
                break

            conv = convolution_fft(a, a)                       # length = 2n‑1
            max_val = np.max(conv)

            # Stable soft‑max weights
            w_raw = np.exp((conv - max_val) / tau)
            w = w_raw / np.sum(w_raw)

            grad = gradient_from_weights(a, w)

            total_iters += 1
            eta = base_eta / np.sqrt(total_iters)

            # Exponentiated‑gradient update (multiplicative)
            a = a * np.exp(-eta * grad)
            a = normalize(a)

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

        # optional early break if improvement stalls – omitted for simplicity

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (as in the reference implementation)
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑objective.
    Multiplicative (exponentiated‑gradient) updates with projection onto
    the simplex.  Returns the best vector seen and its exact score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # Uniform weight on all maximising indices (ties are handled)
        w = np.zeros(conv_len, dtype=float)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------


def generate_candidates(n: int, rng: np.random.Generator, best_prev: np.ndarray = None) -> List[np.ndarray]:
    """
    Produce a diverse library of initial vectors of length n.
    The list includes analytically motivated shapes, random Dirichlet
    draws with several concentration parameters, and mixtures.
    If `best_prev` is supplied, it is up‑sampled to length n and added.
    """
    candidates: List[np.ndarray] = []

    # Classic families
    candidates.append(uniform_shape(n))
    candidates.append(arcsine_initial(n))

    # Power‑law family (randomised)
    for _ in range(8):
        p = rng.uniform(0.30, 0.70)          # exponent near 0.5 (arcsine)
        eps = rng.uniform(0.0, max(5.0, 0.015 * n))
        candidates.append(powerlaw_shape(n, p, eps))

    # Symmetric Beta family
    for _ in range(6):
        a = rng.uniform(0.2, 5.0)
        candidates.append(beta_shape(n, a))

    # Dirichlet draws with varying concentration (sparse ↔ flat)
    for alpha in (0.1, 0.3, 0.7, 2.0, 5.0):
        for _ in range(2):
            candidates.append(random_dirichlet_alpha(n, rng, alpha))

    # Mixtures of arcsine and uniform (smoothes the singularities)
    for _ in range(5):
        w = rng.uniform(0.0, 0.5)
        mix = (1.0 - w) * arcsine_initial(n) + w * uniform_shape(n)
        candidates.append(normalize(mix))

    # Mixtures of arcsine and a random Dirichlet (adds asymmetry)
    for _ in range(5):
        w = rng.uniform(0.0, 0.5)
        dirich = random_dirichlet_alpha(n, rng, rng.uniform(0.2, 2.0))
        mix = (1.0 - w) * arcsine_initial(n) + w * dirich
        candidates.append(normalize(mix))

    # Optional up‑sampled best from a previous length
    if best_prev is not None:
        candidates.append(upsample(best_prev, n))

    return candidates

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step function with a low ``evaluate_sequence``
    score.  The routine respects the time budget, explores several
    parametric families, refines them with a smooth soft‑max mirror‑descent
    and a final exact sub‑gradient polishing, and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Lengths to explore – we start with modest sizes and, if time
    #    permits, try a larger one.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    # Keep the best vector (as a NumPy array) for possible up‑sampling.
    best_global_vec: np.ndarray = None

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n>>> Exploring length n = {n}")

        # ----------------------------------------------------------------
        # 1a. Generate a pool of initial candidates.
        # ----------------------------------------------------------------
        candidates = generate_candidates(n, rng, best_prev=best_global_vec)

        # ----------------------------------------------------------------
        # 1b. Fast ranking (internal score) and keep the most promising.
        # ----------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 6
        top_candidates = [c for _, c in scored[:top_k]]
        print("  top initial scores:",
              " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]))

        # ----------------------------------------------------------------
        # 2. Refine each retained candidate with the smooth soft‑max
        #    mirror‑descent, then evaluate with the official function.
        # ----------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            cur_sc = compute_score(cand)
            print(f"  candidate {idx + 1}/{len(top_candidates)} start_score = {cur_sc:.6f}")

            # Allocate a slice of the remaining time for this candidate.
            remaining = max(0.0, deadline - time.time())
            # Empirically ~15 iterations per second for n≈2000.
            max_iters = int(min(30_000, max(2_000, remaining * 15)))

            refined, _ = refine_softmax(
                a=cand,
                deadline=deadline,
                max_iter_per_stage=max_iters,
            )

            # Official evaluation of the refined vector.
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_global_vec = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 3. Final polishing of the overall best vector (exact sub‑gradient).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best (exact sub‑gradient)")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(100_000, max(5_000, remaining * 15)))

        polished, _ = refine_subgradient(
            a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # Fallback to a safe uniform vector of modest size.
        fallback_vec = uniform_shape(256)
        best_seq = fallback_vec.tolist()
        best_val = evaluate_sequence(best_seq)
        print("[fallback] using uniform vector, score =", best_val)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
