Algorithm

   on the simplex.  The sub‑gradient of the maximal convolution entry
   is computed exactly; the step size decays as 1/√t.  The run is

   perform a large number of stochastic moves that transfer a random
   fraction of mass from a high‑gradient index to a low‑gradient one.
   A geometric temperature schedule (starting relatively hot) and a
   Metropolis acceptance rule allow escaping shallow local minima.
   Every 10 k iterations a tiny smoothing step (3‑point moving average)
   is applied; occasional random Dirichlet restarts improve exploration.

5. **Return** – The best vector seen is normalised to sum = 1 and
   returned as a plain list of floats.

def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector whose sum is 1.
    Returns 2·n·max(conv) where conv = a * a (linear convolution).
    """
    n = a.shape[0]
    # convolution via FFT (real‐to‐real)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    # work in log‑space to avoid overflow for extreme α,β
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)            # safe for exp
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    w /= w.sum()
    return w


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of ``max(conv)`` at the point ``a``.
    The gradient is the average of the exact gradients of all active
    convolution entries (those attaining the current maximum up to a tiny
    tolerance).
    """
    n = a.shape[0]
    max_b = conv.max()
    eps = max_b * 1e-6                     # tolerance for active set
    active = np.where(conv >= max_b - eps)[0]

    if active.size == 0:                   # fallback – not expected
        return 2.0 * a

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        # derivative ∂conv_k/∂a_i = 2·a_{k‑i}
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) update on the probability simplex.
    """
    factor = np.exp(-step * grad)          # multiplicative factor
    a_new = a * factor
    # avoid under‑flow / exact zeros that could stall the algorithm
    a_new = np.clip(a_new, 1e-16, None)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The algorithm runs for at most ``budget_s``
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1 – problem size & initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        n = 4096
        print(f"No existing sequence – using default length {n}.")
        init = np.ones(n, dtype=float) / n

    # Normalise to a probability vector (sum == 1)
    a = init / init.sum()

    # --------------------------------------------------------------
    # 2 – Global shape search (Beta family)
    # --------------------------------------------------------------
    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # Parameter grid (log‑spaced α, symmetric β; also a few asymmetric variants)
    alphas = np.concatenate((
        np.logspace(-2, -0.5, 12, base=10.0),
        np.logspace(-0.5, 0.0, 12, base=10.0),
        np.logspace(0.0, 1.0, 12, base=10.0)
    ))
    # Asymmetric β values to broaden search
    extra_betas = [0.5, 1.0, 2.0]

    for alpha in alphas:
        # symmetric case
        seq = beta_sequence(alpha, alpha, n)
        sc = compute_score(seq)
        if sc < best_score:
            best_score = sc
            best_a = seq
            print(f"Beta({alpha:.3g},{alpha:.3g}) new best {best_score:.6f}")

        # a few asymmetric pairs
        for beta in extra_betas:
            seq = beta_sequence(alpha, beta, n)
            sc = compute_score(seq)
            if sc < best_score:
                best_score = sc
                best_a = seq
                print(f"Beta({alpha:.3g},{beta:.3g}) new best {best_score:.6f}")

        # Early abort if time is nearly exhausted
        if time.time() > deadline - 200:
            break

    # --------------------------------------------------------------
    # 3 – Exponential‑gradient descent phase
    # --------------------------------------------------------------
    a = best_a.copy()               # start the descent from the best Beta vector
    it_exp = 0
    eta0 = 0.8                      # base step size (tuned)

    while time.time() < deadline - 150:
        # convolution & current score
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        cur_score = 2.0 * n * conv.max()

        # record improvement
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it_exp % 20000 == 0:
                print(f"Exp‑grad iter {it_exp:6d}: new best {best_score:.6f}")

        # sub‑gradient of the current maximal convolution entry
        grad = subgradient_of_max_conv(a, conv)

        # diminishing step size (1/√t schedule)
        step = eta0 / math.sqrt(it_exp + 1.0)

        # multiplicative update & renormalisation
        a = exponential_update(a, grad, step)

        it_exp += 1

    print(f"Exponentiated‑gradient finished after {it_exp} iters, best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4 – Guided simulated annealing phase
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    fft = np.fft.rfft(cur_a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    cur_score = 2.0 * n * conv.max()

    # temperature schedule – start relatively hot, end close to 0
    T_start = cur_score * 0.20          # 20 % of current score
    T_end = cur_score * 1e-5
    it_sa = 0

    while time.time() < deadline:
        # geometric cooling
        T = max(T_end, T_start * (0.9995 ** it_sa))

        # ---------------------------------------------------------
        # 4.1 – gradient‑guided proposal distribution
        # ---------------------------------------------------------
        grad = subgradient_of_max_conv(cur_a, conv)

        # source indices (high gradient -> likely to decrease max)
        src_probs = grad / grad.sum()
        src = np.random.choice(n, p=src_probs)

        # target indices (low gradient -> safe to increase)
        inv = 1.0 / (grad + 1e-12)
        tgt_probs = inv / inv.sum()
        tgt = np.random.choice(n, p=tgt_probs)

        if src == tgt or cur_a[src] <= 0.0:
            it_sa += 1
            continue

        # Transfer a random fraction of mass from src to tgt
        delta = cur_a[src] * np.random.rand() * 0.5   # at most 50 % of src mass
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # Every 10 k SA steps apply a tiny smoothing filter
        if it_sa % 10000 == 0 and it_sa > 0:
            kernel = np.array([0.25, 0.5, 0.25])
            cand = np.convolve(cand, kernel, mode='same')
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()                     # renormalise after smoothing

        # evaluate candidate (full convolution)
        fft_c = np.fft.rfft(cand, n=2 * n - 1)
        conv_c = np.fft.irfft(fft_c * fft_c, n=2 * n - 1)
        cand_score = 2.0 * n * conv_c.max()

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            cur_a = cand
            conv = conv_c
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:7d}: new best {best_score:.6f}")

        # occasional random Dirichlet restart to boost exploration
        if it_sa % 50000 == 0 and it_sa > 0:
            cur_a = np.random.dirichlet(np.ones(n))
            fft = np.fft.rfft(cur_a, n=2 * n - 1)
            conv = np.fft.irfft(fft * fft, n=2 * n - 1)
            cur_score = 2.0 * n * conv.max()
            print(f"SA iter {it_sa}: random restart performed")

        # periodic status output
        if it_sa % 20000 == 0 and it_sa > 0:
            print(f"SA iter {it_sa:7d}: current best {best_score:.6f}")

        it_sa += 1

    print(f"Simulated‑annealing finished after {it_sa} iterations; final best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5 – Return the best sequence (still normalised to sum == 1)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()       # safety normalisation
    return [float(v) for v in best_a.tolist()]
