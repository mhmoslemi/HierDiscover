def solve_lp(f, M, constraint_shifts):
    n=len(f)
    m=len(constraint_shifts)
    # A_ub shape (m,n)
    A_ub = np.zeros((m,n))
    for row_idx, k in enumerate(constraint_shifts):
        # For each variable j: consider i = k - j
        # if 0 <= i < n then coefficient = f[i]
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # For j in [i_start? Actually j = k - i]
        # Equivalent to j = k - i, as i loops over possible values.
        # As we fill A_ub[row_idx, j] = f[i] where i = k - j.
        # Let's loop j from max(0, k-(n-1)) to min(n-1, k):
        j_start = max(0, k-(n-1))
        j_end = min(n-1, k)
        # for j in this range:
        # i = k - j
        # coefficient = f[i]
        # fill.
        for j in range(j_start, j_end+1):
            i = k - j
            A_ub[row_idx, j] = f[i]
    b_ub = np.full(m, M)
    c = -np.ones(n) # maximize sum g => minimize -sum

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None
    return res.x

def solve_lp_linear(a, M, shifts):
    n = len(a)
    m = len(shifts)
    A = np.zeros((m, n))
    for idx, k in enumerate(shifts):
        # Compute valid j range 
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # Corresponding i indices
        i_vals = k - np.arange(j_start, j_end + 1)
        A[idx, j_start:j_end + 1] = a[i_vals]
    b = np.full(m, M)
    c = -np.ones(n)  # maximize sum (minimize negative)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Propose a step‑function (non‑negative vector) minimizing `evaluate_sequence`.

Algorithm
---------
1. Random initialisation  
   – generate many candidates (sparse uniform, sparse Dirichlet, dense Dirichlet)  
   – keep the best (fast FFT based score).

2. Frank‑Wolfe style refinement via a linear program  
   – given the current vector `a` (∑a = 1) compute M = max(a∗a).  
   – maximise Σ g_i under the *linear* constraints (a∗g)_k ≤ M for a
     carefully chosen set of shifts `k`.  This yields a vector `g` whose
     total mass S = Σ g_i is typically > 1.  
   – scale `ĝ = g / S` (now ∑ĝ = 1) and either replace `a` by `ĝ` or
     take a convex combination `a←(1−t)·a + t·ĝ` (t≈0.5, reduced until
     improvement).  The step is guaranteed to lower max‑convolution
     whenever possible.

3. Final polishing – a simple hill‑climber that swaps entries or moves a
   small fraction of mass between positions, always projecting back onto
   the probability simplex.

All work respects the 1000 s budget; the routine prints progress and
returns a plain Python list of floats ready for the harness.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a∗a)  (exactly the value returned by evaluate_sequence
    for a normalized vector).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# A tiny hill‑climber (swap or mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)  # success → increase step
        else:
            step_scale *= 0.97                     # failure → decrease step

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Linear‑program based refinement (Frank‑Wolfe style)
# ----------------------------------------------------------------------


def solve_lp_max_sum(a: np.ndarray, M: float, shifts: np.ndarray) -> np.ndarray | None:
    """
    Solve   max Σ g_i   s.t.   a * g ≤ M   at the given shifts.
    Returns the optimal g (non‑negative) or None if LP fails.
    """
    n = a.shape[0]
    m = shifts.shape[0]

    # Build the constraint matrix A_ub (shape m×n)
    A = np.zeros((m, n), dtype=float)
    for row, k in enumerate(shifts):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        j_range = np.arange(j_start, j_end + 1, dtype=int)
        i_vals = k - j_range
        A[row, j_start:j_end + 1] = a[i_vals]

    b = np.full(m, M, dtype=float)
    c = -np.ones(n, dtype=float)                # maximise Σ g_i  ⇔ minimise -Σ g_i
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def frank_wolfe_refine(a_init: np.ndarray,
                       time_deadline: float,
                       max_iter: int = 200) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve `a` by solving the linear programme described above.
    Stops when time runs out or no improvement is possible.
    Returns the improved vector and its score.
    """
    a = a_init.copy()
    n = len(a)
    best_score = compute_score(a)

    it = 0
    while time.time() < time_deadline and it < max_iter:
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # Choose a set of shifts that guarantees every variable appears.
        # Use all shifts r … r+n‑1 where r is the index of the largest entry.
        r = int(np.argmax(a))
        shifts = np.arange(r, r + n, dtype=int)

        g = solve_lp_max_sum(a, M, shifts)
        if g is None:
            break

        S = float(np.sum(g))
        if S <= 1.0 + 1e-12:
            # cannot enlarge the total mass → no further progress
            break

        # Scale g to the simplex (∑ĝ = 1)
        g_hat = g / S

        # Direct replacement test
        score_g = compute_score(g_hat)
        if score_g < best_score - 1e-12:
            a = g_hat
            best_score = score_g
            print(f"Iter {it}: direct LP‑vector improvement → {best_score:.6f}")
            it += 1
            continue

        # Otherwise try a convex combination a←(1−t)a + t·ĝ
        improved = False
        t = 0.5
        while t > 1e-4:
            a_candidate = (1.0 - t) * a + t * g_hat
            a_candidate = project_to_simplex(a_candidate)
            cand_score = compute_score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_score = cand_score
                improved = True
                print(f"Iter {it}: mixed (t={t:.4f}) improvement → {best_score:.6f}")
                break
            t *= 0.5

        if not improved:
            # No improvement with any t → stop.
            break

        it += 1

    return a, best_score


# ----------------------------------------------------------------------
# Random candidate generation
# ----------------------------------------------------------------------


def random_candidate(n: int) -> np.ndarray:
    """
    Produce a random vector with ∑=1.
    Two families are used:
      – sparse support with Dirichlet weights,
      – dense Dirichlet vector.
    """
    mode = np.random.choice(["sparse", "dense"], p=[0.7, 0.3])
    if mode == "sparse":
        # pick a density p∈[0.1,0.4] → k≈p·n
        p = np.random.uniform(0.10, 0.40)
        k = max(2, int(round(p * n)))
        idx = np.random.choice(n, size=k, replace=False)
        weights = np.random.dirichlet(np.ones(k))      # random positive weights
        a = np.zeros(n, dtype=float)
        a[idx] = weights
    else:
        # dense Dirichlet with moderate concentration
        alpha = np.random.uniform(0.05, 2.0)
        a = np.random.dirichlet(np.full(n, alpha))
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The search proceeds in three phases (random sampling, LP refinement,
    final hill‑climb) while respecting the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Optional fallback: use the globally supplied sequence if it exists
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = np.array(fallback, dtype=float)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – fast random search
    # --------------------------------------------------------------
    lengths = [1024, 2048, 4096]   # reasonable sizes (larger ⇒ slower)
    while time.time() < deadline - 200.0:   # leave ~200 s for refinements
        n = int(np.random.choice(lengths))
        a = random_candidate(n)
        cur_val = compute_score(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"Random best: score={best_val:.6f} (n={n})")

    # --------------------------------------------------------------
    # Phase 2 – LP (Frank‑Wolfe) refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 50.0:
        print("\nStarting LP‑based Frank‑Wolfe refinement")
        refined_a, refined_score = frank_wolfe_refine(
            best_seq,
            time_deadline=deadline - 30.0,
            max_iter=200,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.copy()
            print(f"LP refinement improved score to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final hill‑climb polish
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nFinal hill‑climb polishing")
        polished_seq, polished_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=deadline,
            max_steps=300_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = np.array(polished_seq, dtype=float)
            print(f"Hill‑climb improved score to {best_val:.6f}")

    # --------------------------------------------------------------
    # Safeguard: never return None
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = np.full(n, 1.0 / n, dtype=float)

    # Final official evaluation (uses the provided `evaluate_sequence`)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list.
    return best_seq.tolist()
