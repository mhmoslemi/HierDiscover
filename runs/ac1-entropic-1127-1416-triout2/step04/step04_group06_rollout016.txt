def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    # initialize best vector
    try:
        best = np.array(height_sequence_1, dtype=float)
        n = best.size
    except NameError:
        n = 5146
        best = np.full(n, 1.0 / n)

    best = simplex_normalize(best)
    best_score = compute_score(best)

    # create working vector a = best copy
    a = best.copy()
    # parameters
    alpha = 0.5
    alpha_max = 2000.0
    alpha_inc = 1.005  # per iteration multiplication

    eta0 = 0.5
    # Precompute L for correlation
    L_corr = 1 << ((3 * n - 2).bit_length())
    # keep track of iteration without improvement
    last_improve_iter = 0
    it = 0

    while time.time() < deadline - 5.0:
        # compute self-convolution
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
        maxc = conv.max()

        # determine plateau set
        eps = 0.001  # 0.1% tolerance
        plateau_mask = conv >= maxc * (1.0 - eps)
        num_plateau = plateau_mask.sum()
        if num_plateau <= 10:
            # softmax
            shifted = conv - maxc  # <=0
            exp_vals = np.exp(alpha * shifted)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                p = np.full_like(conv, 1.0 / conv.size)
            else:
                p = exp_vals / sum_exp
        else:
            # uniform over plateau
            p = np.zeros_like(conv)
            p[plateau_mask] = 1.0 / num_plateau

        # compute gradient via cross-correlation
        fft_a_full = np.fft.rfft(a, n=L_corr)
        fft_p_full = np.fft.rfft(p, n=L_corr)
        corr = np.fft.irfft(fft_a_full * np.conj(fft_p_full), n=L_corr)
        offset = convex_len = conv.size - 1  # 2n-2
        g = corr[offset:offset + n]

        # update
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_normalize(a)

        # update parameters
        if alpha < alpha_max:
            alpha = min(alpha * alpha_inc, alpha_max)

        # evaluate occasionally
        if (it & 15) == 0:  # every 16 iterations
            cur_score = compute_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve_iter = it
                print(f"[Iter {it}] new best {best_score:.12f}")

        # if no improvement for many iters, perturb
        if it - last_improve_iter > 2000:
            # add small random noise
            noise = np.random.rand(n) * 1e-4
            a = a + noise
            a = simplex_normalize(a)
            last_improve_iter = it
            print(f"[Iter {it}] perturbation applied")

        it += 1

    # After core loop, apply simulated annealing refinement
    cur_a = best.copy()
    cur_score = best_score
    T = cur_score * 0.02
    T_end = cur_score * 1e-6
    # estimate remaining time
    while time.time() < deadline - 0.5 and T > T_end:
        # choose two indices using probability proportional to a for source
        src = np.random.choice(n, p=cur_a)
        tgt = np.random.randint(n)
        if tgt == src:
            continue
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[SA] new best {best_score:.12f}")
        T *= 0.9995

    print(f"[Done] best_score {best_score:.12f}")
    return [float(v) for v in best.tolist()]

def gradient_update(a, alpha, eta, L_corr):
    # compute conv
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2*n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2*n - 1)
    maxc = conv.max()
    # compute p weights
    if np.random.rand() < 0.001: # maybe occasional random? Not needed.

    # plateau detection threshold
    eps = 0.001
    high_mask = conv >= maxc * (1.0 - eps)
    count_high = high_mask.sum()
    if count_high <= 15:
        # softmax
        shifted = conv - maxc
        exp_vals = np.exp(alpha * shifted)
        sum_exp = exp_vals.sum()
        if sum_exp == 0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp
    else:
        p = np.zeros_like(conv)
        p[high_mask] = 1.0 / count_high

    # gradient via correlation
    fft_a_full = np.fft.rfft(a, n=L_corr)
    fft_p_full = np.fft.rfft(p, n=L_corr)
    corr = np.fft.irfft(fft_a_full * np.conj(fft_p_full), n=L_corr)
    offset = conv.size - 1  # 2n-2
    g = corr[offset:offset + n]
    # update
    a_new = a * np.exp(-eta * g)
    a_new = simplex_normalise(a_new)
    return a_new, maxc

def simulated_annealing_refine(a_start, deadline):

    while time.time() < deadline - 0.1 and T > T_end:
        # pick source index according to a distribution
        src = np.random.choice(n, p=a)
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1
        if a[src] <= 0:
            continue
        delta = a[src] * np.random.rand()
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] new best {best_score:.12f}")
        T *= 0.9995
    return best_a, best_score

from scipy.optimize import linprog


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale ``v`` to have sum = 1 (preserves non‑negativity)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_self_conv(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2·n‑1."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector already normalised
    to sum = 1:  2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = compute_self_conv(a)
    return 2.0 * n * conv.max()


def gradient_update(a: np.ndarray, alpha: float, eta: float, L_corr: int):
    """
    One exponentiated‑gradient step on the smooth max of the
    self‑convolution.  Returns the updated vector and the current
    ``max(conv)``.
    """
    n = a.shape[0]

    # ----- 1️⃣  self‑convolution and max -----
    conv = compute_self_conv(a)
    maxc = conv.max()

    # ----- 2️⃣  weighting over the largest lags -----
    eps = 0.001                      # plateau tolerance
    high_mask = conv >= maxc * (1.0 - eps)
    if high_mask.sum() <= 15:        # use a soft‑max when the plateau is tiny
        shifted = conv - maxc        # ≤ 0, safe for exponentiation
        exp_vals = np.exp(alpha * shifted)
        sum_exp = exp_vals.sum()
        if sum_exp == 0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp
    else:                            # otherwise treat the plateau uniformly
        p = np.zeros_like(conv)
        p[high_mask] = 1.0 / high_mask.sum()

    # ----- 3️⃣  gradient = cross‑correlation (a ★ p) -----
    fft_a = np.fft.rfft(a, n=L_corr)
    fft_p = np.fft.rfft(p, n=L_corr)
    corr = np.fft.irfft(fft_a * np.conj(fft_p), n=L_corr)
    offset = conv.size - 1           # index where lag 0 appears
    g = corr[offset:offset + n]

    # ----- 4️⃣  exponentiated‑gradient update on the simplex -----
    a_new = a * np.exp(-eta * g)
    a_new = simplex_normalise(a_new)

    return a_new, maxc


def run_gradient_descent(a0: np.ndarray, deadline: float):
    """Phase 1 – long mirror‑descent run."""
    a = a0.copy()
    n = a.shape[0]
    L_corr = 1 << ((3 * n - 2).bit_length())   # next power‑of‑two ≥ 3n‑2

    best_a = a.copy()
    best_score = compute_score(best_a)

    # schedules
    alpha = 0.5
    alpha_max = 2000.0
    alpha_mul = 1.005
    eta0 = 0.8

    last_improve = 0
    it = 0

    while time.time() < deadline - 5.0:
        eta = eta0 / math.sqrt(it + 1.0)
        a, _ = gradient_update(a, alpha, eta, L_corr)

        # increase smoothing parameter
        if alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic evaluation
        if (it & 31) == 0:                     # every 32 iterations
            cur_score = compute_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = it
                print(f"[GD] iter {it:6d}  new best {best_score:.12f}")

        # stagnation → small random jitter
        if it - last_improve > 5000:
            noise = np.random.rand(n) * 1e-4
            a = simplex_normalise(a + noise)
            last_improve = it
            print(f"[GD] iter {it:6d}  perturbation applied")

        it += 1

    return best_a, best_score


def build_constraint_matrix(f: np.ndarray, lags: np.ndarray) -> np.ndarray:
    """
    Build the (|lags| × n) linear‑constraint matrix C where
    row k corresponds to  Σ_i f_i·g_{lag‑i} ≤ M.
    """
    n = f.shape[0]
    K = lags.size
    C = np.zeros((K, n), dtype=float)

    for row, k in enumerate(lags):
        start_j = max(0, k - (n - 1))
        end_j = min(k, n - 1)
        j = np.arange(start_j, end_j + 1, dtype=int)
        i = k - j
        C[row, j] = f[i]                     # coefficient f_{k‑j}
    return C


def lp_refine(a0: np.ndarray, deadline: float):
    """Phase 2 – LP‑based refinement using only the top convolution lags."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    while time.time() < deadline - 1.0:
        conv = compute_self_conv(a)
        M = conv.max()

        # pick the most critical lags (20‑30 of them)
        K = 20
        if K > conv.size:
            K = conv.size
        top_lags = np.argpartition(conv, -K)[-K:]

        C = build_constraint_matrix(a, top_lags)

        # linear program: maximise Σ g_j  ⇔ minimise –Σ g_j
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None)] * n
        res = linprog(c,
                      A_ub=C,
                      b_ub=np.full(C.shape[0], M, dtype=float),
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 10})

        if not res.success or res.x is None:
            print("[LP] LP failed – exiting refinement")
            break

        g = res.x
        gn = g.sum()
        if gn == 0.0:
            break
        g = g / gn                     # normalise to a probability vector

        # convex combination with the current vector
        t = 0.5
        a = (1.0 - t) * a + t * g
        a = simplex_normalise(a)

        cur_score = compute_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[LP] improvement to {best_score:.12f}")

    return best_a, best_score


def simulated_annealing_refine(a0: np.ndarray, deadline: float):
    """Phase 3 – short SA pass to escape tiny local minima."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)
    cur_score = best_score

    T = cur_score * 0.02
    T_end = cur_score * 1e-6

    while time.time() < deadline - 0.1 and T > T_end:
        src = np.random.choice(n, p=a)
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1
        if a[src] <= 0.0:
            continue
        delta = a[src] * np.random.rand()
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] new best {best_score:.12f}")

        T *= 0.9995

    return best_a, best_score

    """Search for a step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ---- initial vector (use the supplied one if available) ----
    try:
        init_vec = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_vec.size
    except NameError:

        init_vec = np.full(n, 1.0 / n, dtype=float)

    init_vec = simplex_normalise(init_vec)
    best_a = init_vec.copy()
    best_score = compute_score(best_a)
    print(f"[Init] n={n}, score={best_score:.12f}")

    # ------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent
    # ------------------------------------------------------------
    phase1_deadline = deadline - 30.0   # keep ~30 s for later phases
    if time.time() < phase1_deadline:
        best_a, best_score = run_gradient_descent(best_a, phase1_deadline)
        print(f"[Phase1] best score {best_score:.12f}")

    # ------------------------------------------------------------
    # Phase 2 – LP refinement on the worst lags
    # ------------------------------------------------------------
    phase2_deadline = deadline - 10.0   # keep a few seconds for SA
    if time.time() < phase2_deadline:
        best_a, best_score = lp_refine(best_a, phase2_deadline)
        print(f"[Phase2] best score {best_score:.12f}")

    # ------------------------------------------------------------
    # Phase 3 – final simulated‑annealing polishing
    # ------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_refine(best_a, deadline)
        print(f"[Phase3] final best {best_score:.12f}")

    # Return the best vector (already normalised to sum = 1)
    return [float(v) for v in best_a.tolist()]
