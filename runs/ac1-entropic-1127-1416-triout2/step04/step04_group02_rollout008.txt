```
def project_simplex(v):
    # using algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.full_like(v, 1.0/len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def optimise_expgd(initial_a, deadline, lr0=0.5, eps0=1.0, eps_min=0.05, eps_decay=0.9995, max_iter=200000, noise_every=0, noise_scale=0.0):
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # compute conv from a and a
        conv = convolution_fft(a, a)
        # compute softmax weights
        w = np.exp(eps0 * conv)
        w_sum = w.sum()
        w = w / w_sum
        # gradient
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[n-1:2*n-1][::-1]
        # step size schedule
        lr = lr0 / np.sqrt(it)
        # update
        a = a * np.exp(-lr * grad)
        # optional noise
        if noise_every and it % noise_every == 0:
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)
        # project onto simplex
        a = project_simplex(a)
        # evaluate
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # decay epsilon
        eps0 = max(eps0 * eps_decay, eps_min)
        # occasional prints
        if it % 2000 == 0:
            print(f"    expgd iter={it:,d} cur={cur_score:.6f} best={best_score:.6f} eps={eps0:.5f}")
```

The goal is to minimise  

\[
\Phi(a)=\frac{2\,n\;\max\bigl(a\ast a\bigr)}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the objective is homogeneous of degree 0 we may fix the
normalisation \(\sum_i a_i=1\).  
Let  

\[
c = a\ast a\in\mathbb R^{2n-1} .
\]

The non‑smooth term \(\max(c)\) is replaced by a smooth *soft‑max*  

\[
M_\varepsilon(a)=\frac{1}{\varepsilon}\log\!\Bigl(\sum_{k}e^{\varepsilon c_k}\Bigr)
\]

which converges to \(\max(c)\) as \(\varepsilon\downarrow 0\).  
For a fixed \(\varepsilon\) the gradient of \(M_\varepsilon\) can be written
very compactly:

```
w_k = softmax_ε(c)_k                # w_k ≥ 0 , Σ w_k = 1
∇_a M_ε = 2·(a * w_rev)[n‑1:2n‑1]  # ( * ) = linear convolution,
                                   # w_rev = w reversed,
                                   # slice is then reversed.
```

Thus a single iteration needs only two FFT‑based convolutions.
We perform **exponentiated‑gradient (mirror‑descent) updates**

```
a ← a ⊙ exp( -η_t ∇M_ε )        (⊙ = element‑wise product)
a ← projection of a onto the probability simplex
```

which guarantees non‑negativity.  
The step‑size follows \(\eta_t = \eta_0/\sqrt t\) and the temperature
\(\varepsilon\) is *annealed* from a large value (smooth landscape) down to a
small one (close to the true max).

The search proceeds in three phases

1. **exploration** – several lengths (1024 … 8192) are tried, a small library
   of diverse initial shapes is built (uniform, arcsine, beta‑like,
   power‑law, Dirichlet, and an up‑sampled best‑so‑far shape).
2. **refinement** – each promising start is optimised by the exponentiated‑gradient
   routine described above.  The best sequence seen is kept.
3. **final polishing** – the globally best vector is refined again with a
   smaller learning rate and a much lower temperature.

All time is watched with a hard deadline; the routine always returns a
feasible sequence before the budget expires.

```python
"""
Hybrid optimisation for the step‑function min‑max convolution problem.

The algorithm explores several lengths, builds a diverse set of
initial candidates, and refines each with an exponentiated‑gradient
method that minimises a smooth soft‑max approximation of the
max‑convolution.  The temperature ε is annealed, the step‑size follows
ηₜ = η₀/√t and after each update we project onto the probability simplex.
A final polishing step with a very small ε finishes the search.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant).
    a is assumed non‑negative and already normalised (∑a = 1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a small Gaussian perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    # re‑normalise onto the simplex
    up = up / up.sum()
    return up


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (soft‑max of the convolution)
# ----------------------------------------------------------------------


def optimise_expgd(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lr0: float = 0.6,
    eps0: float = 5.0,
    eps_min: float = 0.05,
    eps_decay: float = 0.9995,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed objective
    M_ε(a) = (1/ε)·log Σ_k exp(ε·(a∗a)_k).

    Returns the best vector seen and its *true* score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()                    # work on the probability simplex
    best_a = a.copy()
    # initial true score (cheaply obtained from the same convolution)
    conv = convolution_fft(a, a)
    best_sc = compute_score(a)

    n = len(a)
    eps = eps0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- 1. convolution a∗a  (used for soft‑max and for the true score) -----
        conv = convolution_fft(a, a)
        conv_max = conv.max()
        cur_sc = 2.0 * n * conv_max / (a.sum() ** 2)

        # ----- keep the best vector -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # ----- 2. soft‑max weights -----
        w = np.exp(eps * conv)               # shape (2n‑1)
        w_sum = w.sum()
        w = w / w_sum                         # now a probability distribution

        # ----- 3. gradient of M_ε -----
        w_rev = w[::-1]                       # reversed weights
        conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : 2 * n - 1][::-1]   # length n

        # ----- 4. step‑size & update (exponentiated‑gradient) -----
        eta = lr0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # optional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # ----- 5. projection onto the simplex -----
        a = project_simplex(a)

        # ----- 6. anneal epsilon -----
        eps = max(eps * eps_decay, eps_min)

        # occasional progress report (does not affect runtime critically)
        if it % 2000 == 0:
            print(
                f"    expgd iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} eps={eps:.5f}"
            )

    return best_a, best_sc

            candidates.append(generate_beta_like(n, alpha, beta))

        # random power‑law shapes
        for _ in range(3):
            p = rng.uniform(0.1, 1.0)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            candidates.append(generate_powerlaw(n, p, eps))

        # Dirichlet / exponential draws
        for conc in [0.5, 1.0, 2.0]:
            candidates.append(random_dirichlet(n, conc, rng))

        # up‑sample the best vector found so far (if any)
        if best_vec is not None:
            up = upsample(best_vec, n)
            candidates.append(up)

        # small perturbations of the arcsine shape (helps escape shallow minima)
        for _ in range(2):
            cand = perturb(arcsine_initial(n), rng, scale=0.02)
            candidates.append(cand)

        # --------------------------------------------------------------
        # 1b. quick cheap ranking, keep the most promising few
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:4]]   # keep four best

        # --------------------------------------------------------------
        # 1c. refine each retained candidate with the exponentiated‑gradient optimiser
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # remaining time → rough iteration budget
            remaining = max(0.0, deadline - time.time())
            # rough estimate: ~300 iterations per second for n≈4096,
            # ~120 for n≈8192 (convolutions dominate)
            it_per_sec = 300 if n <= 4096 else 120
            max_iter = int(min(200_000, max(2_000, remaining * it_per_sec)))

            refined, _ = optimise_expgd(
                initial_a=cand,
                deadline=deadline,
                lr0=0.6,
                eps0=5.0,
                eps_min=0.05,
                eps_decay=0.9995,
                max_iter=max_iter,
                noise_every=500,
                noise_scale=5e-6,
            )

            # final evaluation with the official evaluator
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_vec = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2. final polishing of the overall best sequence (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        it_per_sec = 300 if len(a) <= 4096 else 120
        max_iter = int(min(300_000, remaining * it_per_sec))

        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_expgd(
            initial_a=a,
            deadline=deadline,
            lr0=0.3,               # smaller steps for fine‑tuning
            eps0=1.0,
            eps_min=0.01,
            eps_decay=0.9999,
            max_iter=max_iter,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3. guard – always return a feasible sequence
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
```
