```
def proj_simplex(v):
    # expects numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1,n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Non‑negative step‑function optimisation.

We look for a vector a of length n (n is taken from the previously
known solution or defaults to 8000) with sum = 1 that minimises

        2·n·max(conv(a,a))

where conv is the linear (aperiodic) self‑convolution.
The problem is non‑convex; we use a multi‑restart stochastic
search:

1.  Several simple random initialisers (Dirichlet, beta‑density,
    edge‑heavy, sparse, Gaussian, uniform).
2.  On each restart we run *exponentiated‑gradient* mirror descent on
    the probability simplex.  The gradient is a **sub‑gradient of the
    ‖a∗a‖∞‑norm**, i.e. the average of the gradients of all convolution
    entries that are (almost) maximal.  The gradient is obtained by a
    single FFT‑based cross‑correlation (O(n log n)).
3.  After the descent a short greedy “mass‑transfer” local search
    (simulated‑annealing style) moves probability mass between
    coordinates.
4.  The best vector found during the whole budget is returned.

All helper functions are defined at top level; no closures or lambdas
are used.  The routine respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def simplex_project(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring / convolution helpers
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector already normalised
    to sum = 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_a = np.fft.rfft(a, n=conv_len)
    conv = np.fft.irfft(fft_a * fft_a, n=conv_len)
    return 2.0 * n * conv.max()


def convolution_full(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a∗a (full length 2n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_a = np.fft.rfft(a, n=conv_len)
    return np.fft.irfft(fft_a * fft_a, n=conv_len)


def subgrad_max(a: np.ndarray, conv: np.ndarray, conv_len: int,
                corr_len_pow2: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    The active set consists of all indices where conv is (almost) maximal.
    Returns a vector of length n (gradient of the objective 2·n·max(conv)).
    """
    n = a.shape[0]
    # Active mask (use a tiny relative tolerance)
    maxc = conv.max()
    eps = maxc * 1e-12
    active = conv >= maxc - eps
    # Normalised mask p (probability distribution over active entries)
    p = active.astype(float)
    if p.sum() == 0.0:
        p[:] = 1.0 / conv_len
    else:
        p /= p.sum()

    # Cross‑correlation a ⋆ p (full length n+conv_len‑1 = 3n‑2)
    fft_a = np.fft.rfft(a, n=corr_len_pow2)
    fft_p = np.fft.rfft(p, n=corr_len_pow2)
    corr = np.fft.irfft(fft_a * np.conj(fft_p), n=corr_len_pow2)

    offset = conv_len - 1  # position of lag 0 in the correlation array
    grad = 2.0 * corr[offset: offset + n]  # factor 2 from d(a_i a_j)/da
    return grad


# ----------------------------------------------------------------------
# Random initialisers (all return a simplex‑normalised vector)
# ----------------------------------------------------------------------


def init_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_project(a)


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)           # avoid overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return simplex_project(w)


def init_beta(n: int) -> np.ndarray:
    a = np.random.uniform(0.2, 5.0)
    b = np.random.uniform(0.2, 5.0)
    return beta_sequence(a, b, n)


def init_edge_heavy(n: int) -> np.ndarray:
    """Linear edge‑heavy shape (larger values near the two ends)."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)
    w = dist.astype(float) + 1.0
    return simplex_project(w)


def init_sparse(n: int) -> np.ndarray:
    """Place mass on a small random support."""
    k = np.random.randint(5, min(30, n) + 1)
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=k, replace=False)
    vals = np.random.rand(k) + 1e-6
    a[chosen] = vals
    return simplex_project(a)


def init_gaussian(n: int) -> np.ndarray:
    """Symmetric Gaussian‑shaped density."""
    sigma = np.random.uniform(0.05, 0.25) * n
    x = np.arange(n) - (n - 1) / 2.0
    w = np.exp(-0.5 * (x / sigma) ** 2)
    return simplex_project(w)


def init_uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=float)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence() within ``budget_s`` seconds.
    Returns the best vector found (as a list of floats, sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # ------------------------------------------------------------------
    # 0. Load a previously known solution if it exists
    # ------------------------------------------------------------------
    try:
        init_best = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_best.size
        best_a = simplex_project(init_best)
        best_score = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1 (n={n}), score={best_score:.6f}")
    except NameError:
        n = 8000
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_a)
        print(f"[Init] No previous vector – using uniform start, n={n}, score={best_score:.6f}")

    # Pre‑compute FFT sizes that will be reused
    conv_len = 2 * n - 1                     # length of a∗a
    corr_len = 3 * n - 2                     # length needed for a⋆p
    fft_corr_len = 1 << (corr_len.bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # 1. Collection of initialisers (functions take n → vector)
    # ------------------------------------------------------------------
    init_functions = [
        init_dirichlet,
        init_beta,
        init_edge_heavy,
        init_sparse,
        init_gaussian,
        init_uniform,
    ]

    # ------------------------------------------------------------------
    # 2. Main restart loop
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 8.0:       # keep a few seconds for final return
        # ----- pick a start vector -----
        init_fun = init_functions[np.random.randint(len(init_functions))]
        a = init_fun(n)

        # occasional blending with the current best (helps transfer good mass)
        if np.random.rand() < 0.2:
            a = 0.5 * a + 0.5 * best_a
            a = simplex_project(a)

        # ----- exponentiated‑gradient descent using a sub‑gradient of L∞ -----
        max_iters = 2500
        eta0 = 1.0                               # initial step‑size factor

        for it in range(max_iters):
            if time.time() > deadline - 4.0:
                break

            # 1) self‑convolution and maximal entry
            conv = convolution_full(a)           # length = conv_len
            maxc = conv.max()

            # 2) sub‑gradient of the max (average over all active indices)
            grad = subgrad_max(a, conv, conv_len, fft_corr_len)

            # 3) exponentiated‑gradient update (mirror descent on simplex)
            eta = eta0 / math.sqrt(it + 1.0)
            # Prevent overflow in the exponential
            exp_factor = np.exp(np.clip(-eta * grad, -700.0, 700.0))
            a = a * exp_factor
            a = simplex_project(a)

            # ----- record improvements regularly -----
            if (it & 31) == 0:           # every 32 iterations
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Restart {restart}] it {it:4d} → new best {best_score:.6f}")

        # ----- short greedy mass‑transfer refinement (simulated‑annealing style) -----
        cur_a = best_a.copy()
        cur_score = compute_score(cur_a)
        T = cur_score * 0.018                     # initial temperature
        T_min = cur_score * 1e-6
        while time.time() < deadline - 2.0 and T > T_min:
            # pick source proportional to its mass, target uniformly
            src = np.random.choice(n, p=cur_a)
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1
            if cur_a[src] <= 0.0:
                continue
            delta = cur_a[src] * np.random.rand() * 0.5
            cand = cur_a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand_score = compute_score(cand)

            if cand_score < cur_score:
                accept = True
            else:
                accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

            if accept:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[Refine R{restart}] iter → new best {best_score:.6f}")

            T *= 0.9995          # geometric cooling

        restart += 1

    # ------------------------------------------------------------------
    # 3. Return the best vector found (already on the simplex)
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {restart} restarts: {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
