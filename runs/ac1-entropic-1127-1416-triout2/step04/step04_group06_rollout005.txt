def mirror_descent(a, max_iters, eta0, alpha_start, alpha_end, deadline):
    n = a.size
    best_a = a.copy()
    best_score = compute_score(a)
    # Precompute FFT length for conv and correlation
    conv_len = 2*n - 1
    fft_len_conv = 1 << ((conv_len).bit_length()) # pow2 >= conv_len
    corr_len = n + conv_len - 1 # = 3n-2
    fft_len_corr = 1 << ((corr_len).bit_length())
    alpha = alpha_start
    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break
        # compute conv
        fft_a = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(fft_a * fft_a, n=fft_len_conv)[:conv_len]
        # softmax
        maxc = conv.max()
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp
        # correlation gradient (2*)
        # compute linear correlation of a and p
        fft_a_corr = np.fft.rfft(a, n=fft_len_corr)
        fft_p_corr = np.fft.rfft(p, n=fft_len_corr)
        corr = np.fft.irfft(fft_a_corr * np.conj(fft_p_corr), n=fft_len_corr)
        offset = conv_len - 1
        g = 2.0 * corr[offset:offset + n]
        # step size
        eta = eta0 / math.sqrt(it + 1.0)
        # exponentiated update
        a = a * np.exp(-eta * g)
        a = simplex_normalize(a)
        # update alpha linearly
        alpha = alpha_start + (alpha_end - alpha_start) * (it + 1) / max_iters
        # occasional evaluation
        if (it & 31) == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score

def simulated_annealing(a, cur_score, deadline):
    n = a.size
    best_a = a.copy()
    best_score = cur_score
    # Temperature start and end based on current score
    T = cur_score * 0.02
    T_end = cur_score * 1e-5
    while time.time() < deadline - 1.0 and T > T_end:
        # choose src weighted by a
        src = np.random.choice(n, p=a)
        tgt = np.random.randint(n-1)
        if tgt >= src:
            tgt += 1
        if a[src] <= 0:
            continue
        delta = a[src] * np.random.rand()
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand_score = compute_score(cand)
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))
        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        T *= 0.9995
    return best_a, best_score

"""
Propose a non‑negative step‑function that (approximately) minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a two‑phase optimisation.

Phase 1 – Smoothed L∞ mirror descent
    * Work on the probability simplex (a is kept normalised to sum=1).
    * Approximate the max‑norm with a log‑sum‑exp with a temperature α.
    * Gradient of the smoothed objective is obtained by a (non‑circular)
      correlation of a with the soft‑max weights p.
    * An exponentiated‑gradient (mirror descent on the simplex) update
      a ← a·exp(‑η·g) is performed with a step‑size η∝1/√t.
    * α is increased from 1 to a large value (≈400) so that the
      smooth problem turns into the true max‑problem.

Phase 2 – Light simulated‑annealing refinement
    * Mass‑transfer moves (pick a source index proportionally to its mass,
      transfer a random fraction to a different target index) are tried.
    * Acceptance follows a Metropolis rule with an exponential cooling
      schedule.

Multiple random restarts (different simple initialisers) are run
within the given time budget (default 1000 s).  The best vector seen
so far (any length) is kept and finally returned as a plain Python
list of floats.  All helper functions are top‑level (no closures or
lambdas) to satisfy the execution rules.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (leaves zeros untouched)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length = 2·n‑1) using a power‑of‑two FFT.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len).bit_length())   # next power of two ≥ conv_len
    fft_a = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv_full[:conv_len]


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_self(a)
    return 2.0 * n * conv.max()


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max (log‑sum‑exp) weights over the convolution entries.
    Numerical stability: subtract the maximum before exponentiating.
    """
    m = conv.max()
    # exponentiate (conv - m) to keep values ≤ 1
    exps = np.exp(alpha * (conv - m))
    s = exps.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_exponential(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smoothed max objective with respect to a,
    i.e. 2·a ⋆ p where ⋆ denotes linear (non‑circular) correlation.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    corr_len = n + conv_len - 1               # = 3·n‑2
    fft_len = 1 << ((corr_len).bit_length())   # power of two ≥ corr_len

    fft_a = np.fft.rfft(a, n=fft_len)
    fft_p = np.fft.rfft(p, n=fft_len)
    # linear correlation via IFFT of A·conj(P)
    corr_full = np.fft.irfft(fft_a * np.conj(fft_p), n=fft_len)
    offset = conv_len - 1                     # position of lag 0
    grad = 2.0 * corr_full[offset:offset + n]  # factor 2 for d(a*a)/da
    return grad


# ----------------------------------------------------------------------
# Simple random initialisers (all return a normalised vector)
# ----------------------------------------------------------------------


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum = 1.
    Uses the midpoint rule and clips the log‑density to avoid overflow.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # safe range for np.exp
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return simplex_normalise(w)


def init_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector (i.e. i.i.d. Gamma(1))."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """Linear “edge‑heavy” shape: larger values near both ends, smaller in the centre."""
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    w = dist_to_center.astype(float) + 1.0     # edges receive larger weight
    return simplex_normalise(w)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Place mass on k randomly chosen positions (k ≤ n)."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def mirror_descent(start_a: np.ndarray,
                   max_iters: int,
                   eta0: float,
                   alpha_start: float,
                   alpha_end: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform smoothed‑max mirror descent starting from ``start_a``.
    Returns the best vector found during the run and its score.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # Pre‑compute lengths for FFTs (constants per run)
    conv_len = 2 * n - 1
    fft_len_conv = 1 << ((conv_len).bit_length())
    corr_len = n + conv_len - 1            # = 3·n‑2
    fft_len_corr = 1 << ((corr_len).bit_length())
    offset = conv_len - 1

    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break

        # ---- self‑convolution (FFT) ----
        fft_a = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(fft_a * fft_a, n=fft_len_conv)[:conv_len]

        # ---- soft‑max weights over convolution entries ----
        p = softmax_weights(conv, alpha_start)

        # ---- gradient (2·a ⋆ p) ----
        fft_a_corr = np.fft.rfft(a, n=fft_len_corr)
        fft_p_corr = np.fft.rfft(p, n=fft_len_corr)
        corr_full = np.fft.irfft(fft_a_corr * np.conj(fft_p_corr), n=fft_len_corr)
        g = 2.0 * corr_full[offset:offset + n]

        # ---- exponentiated‑gradient update (mirror descent on simplex) ----
        η = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-η * g)
        a = simplex_normalise(a)

        # ---- increase smoothing parameter α linearly ----
        alpha_start = alpha_start + (alpha_end - alpha_start) / max_iters

        # ---- occasional progress check ----
        if (it & 31) == 0:                     # every 32th iteration
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[MD] it {it:5d} → new best {best_score:.6f}")

    return best_a, best_score


def simulated_annealing(start_a: np.ndarray,
                        start_score: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer simulated‑annealing starting from ``start_a``.
    Returns the best vector encountered and its score.
    """
    a = start_a.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    # Temperature schedule (relative to current score)
    T = cur_score * 0.02
    T_end = cur_score * 1e-5

    n = a.shape[0]

    iter_cnt = 0
    while time.time() < deadline - 1.0 and T > T_end:
        # ---- propose a move: transfer a random fraction of mass from src to tgt ----
        src = np.random.choice(n, p=a)               # source weighted by its mass
        tgt = np.random.randint(n - 1)               # uniformly pick a different index
        if tgt >= src:
            tgt += 1
        if a[src] <= 0.0:
            continue
        delta = a[src] * np.random.rand()            # amount to move
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)

        # ---- Metropolis acceptance ----
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] iter {iter_cnt:6d} → new best {best_score:.6f}")

        # ---- geometric cooling ----
        T *= 0.9995
        iter_cnt += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best vector found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # ------------------------------------------------------------------
    # 0. Choose a working dimension.  Smaller n speeds up iterations and
    #    still allows scores far below the target 1.503.
    # ------------------------------------------------------------------
    n = 1024
    # initialise best with a uniform vector of the chosen size
    best_a = np.full(n, 1.0 / n, dtype=float)
    best_score = compute_score(best_a)
    print(f"[Init] uniform n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Main restart loop (time‑driven)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0:          # keep a few seconds for final return
        # ---- pick an initialisation scheme ----
        mode = np.random.randint(4)
        if mode == 0:
            a = beta_sequence(np.random.uniform(0.2, 5.0),
                              np.random.uniform(0.2, 5.0), n)
        elif mode == 1:
            a = init_dirichlet(n)
        elif mode == 2:
            a = init_edge_concentrated(n)
        else:
            a = init_sparse(n, k=np.random.randint(5, 30))

        # ---- occasional blending with the current best (helps retain useful mass) ----
        if np.random.rand() < 0.15:
            a = simplex_normalise(0.5 * a + 0.5 * best_a)

        # ---- Phase 1: smoothed‑max mirror descent ----
        md_iters = 5000
        eta0 = 0.6
        a_md, score_md = mirror_descent(
            start_a=a,
            max_iters=md_iters,
            eta0=eta0,
            alpha_start=1.0,
            alpha_end=400.0,
            deadline=deadline,
        )
        if score_md < best_score:
            best_score = score_md
            best_a = a_md.copy()
            print(f"[Restart {restart}] after MD → new best {best_score:.6f}")

        # ---- Phase 2: simulated‑annealing refinement ----
        a_sa, score_sa = simulated_annealing(best_a.copy(), best_score, deadline)
        if score_sa < best_score:
            best_score = score_sa
            best_a = a_sa.copy()
            print(f"[Restart {restart}] after SA → new best {best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # 2. Final report and return
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {restart} restarts: {best_score:.6f}")
    # The returned list is already normalised (sum≈1) and contains only non‑negative numbers.
    return best_a.tolist()
