def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0
    best_seq = None
    best_score = float('inf')
    lengths = [1024, 2048, 4096, 8192]  # explore increasing sizes
    for n in lengths:
        if time.time() >= deadline - 5: break
        print(f"\n=== Length {n} ===")
        # generate candidates
        cand_pool = []
        # uniform
        uniform = np.full(n, 1.0/n)
        cand_pool.append((compute_score(uniform), uniform.copy()))
        # arcsine
        arcs = arcsine_initial(n)
        cand_pool.append((compute_score(arcs), arcs.copy()))
        # random Dirichlet
        for _ in range(100):
            alpha = rng.uniform(0.05, 5.0)  # small alpha yields sparsity
            vec = rng.dirichlet(np.full(n, alpha))
            cand_pool.append((compute_score(vec), vec))
        # sparse vectors
        for sparsity in [0.005, 0.01, 0.02, 0.05]:
            k = max(2, int(sparsity * n))
            for _ in range(30):
                idx = rng.choice(n, size=k, replace=False)
                vals = rng.random(k)
                vec = np.zeros(n)
                vec[idx] = vals
                vec /= vec.sum()
                cand_pool.append((compute_score(vec), vec))
        # mixture arcsine & uniform
        for _ in range(50):
            w = rng.random()
            vec = w*arcs + (1-w)*uniform
            vec /= vec.sum()
            cand_pool.append((compute_score(vec), vec))
        # keep best few seeds
        cand_pool.sort(key=lambda x: x[0])
        seeds = [vec for _, vec in cand_pool[:5]]   # top 5 seeds
        # refine each seed
        for seed_vec in seeds:
            if time.time() >= deadline - 3: break
            # Phase A: Adam smooth
            max_iter_adam = int(min(100_000, (deadline - time.time()) * 30))
            refined, _ = adam_optimize_smooth(seed_vec,
                                               deadline,
                                               max_iter=max_iter_adam,
                                               lr0=0.12,
                                               lam_start=2.0,
                                               lam_end=120.0,
                                               noise_every=3000,
                                               noise_scale=2e-5)
            # Phase B: Simulated annealing on true max
            if time.time() < deadline - 2:
                max_iter_sa = int(min(150_000, (deadline - time.time()) * 30))
                refined = simulated_annealing(refined, rng, deadline,
                                             max_iters=max_iter_sa,
                                             init_temp=0.03,
                                             decay=0.9995)
            # Phase C: Exact sub‑gradient descent
            if time.time() < deadline - 2:
                max_iter_sub = int(min(80_000, (deadline - time.time()) * 30))
                refined = subgradient_descent_exact(refined,
                                                    deadline,
                                                    max_iter=max_iter_sub,
                                                    step0=0.03)
            # Official evaluation
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"Length {n}, seed refined score={val:.6f}")
                if val < best_score:
                    best_score = val
                    best_seq = refined.tolist()
                    print(f"*** New best score {best_score:.6f} for n={n} ***")
            except Exception as e:
                print("evaluation error", e)
    # Global polishing on best
    if best_seq is not None and time.time() < deadline - 2:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        total_iters = int(min(400_000, remaining * 30))
        # use more aggressive subgradient then Adam
        a = subgradient_descent_exact(a, deadline, max_iter=total_iters//2, step0=0.02)
        a, _ = adam_optimize_smooth(a,
                                    deadline,
                                    max_iter=total_iters - total_iters//2,
                                    lr0=0.09,
                                    lam_start=5.0,
                                    lam_end=120.0,
                                    noise_every=2000,
                                    noise_scale=1e-5)
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_score:
            best_score = final_val
            best_seq = a.tolist()
            print("*** Updated best after final polishing ***")
    # Guard
    if best_seq is None:
        best_seq = [1.0] * 256
        best_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Best score {best_score:.6f}")
    return best_seq

def lp_improvement(f, max_iters, step_size, deadline):
    a = f.copy()
    n = len(a)
    m = 2 * n - 1
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        M = np.max(conv)
        # build A
        A = np.zeros((m, n), dtype=float)
        for j in range(n):
            A[j:j+n, j] = a
        # Solve LP
        c = -np.ones(n)
        b = np.full(m, M + 1e-14)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:

        g = g / sum_g  # scale to sum=1
        # mixing

"""
Search for a non‑negative step sequence that minimises the official
evaluate_sequence objective.

The algorithm mixes several complementary ideas:

1. **Initial pool** – uniform, arcsine, beta‑family, random Dirichlet
   and deliberately sparse vectors (few non‑zero entries).  All candidates
   are normalised to Σa = 1 (the official objective is scale‑invariant).

2. **LP‑improvement** – a single linear‑programming step (the one described
   in the paper) that, for a fixed sequence f, finds a partner g that
   maximises Σg under the constraint (f∗g)≤M where M = max(f∗f).  After a
   small convex combination f←(1‑t)f+tg the peak of the self‑convolution
   often drops dramatically.

3. **Smooth Adam optimisation** – a soft‑max (log‑sum‑exp) surrogate
   of max(f∗f) with analytically computed gradient via FFT, followed by
   Adam updates and projection onto the probability simplex.

4. **Exact sub‑gradient descent** – the true sub‑gradient of
   2·n·max(f∗f) is used with a diminishing step size and simplex projection.

5. **Simulated‑annealing polishing** – fast Metropolis moves that transfer
   a fraction of mass between coordinates, accepting uphill moves with a
   cooling schedule.

All phases check the global deadline (default 1000 s).  The best sequence
found is finally evaluated with the official ``evaluate_sequence`` and
returned.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score (sum=1 already).
    Returns 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """(i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1)
    right = np.arange(n - peak - 1)[::-1]
    seq = np.concatenate([left, right]).astype(float)
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_sparse_vector(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries at random positions (k≥1)."""
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vec = np.zeros(n, dtype=float)
    vec[idx] = vals
    if vec.sum() == 0:
        vec[idx[0]] = 1.0
    return vec / vec.sum()


def generate_dirichlet_vector(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration parameter ``conc`` (scalar)."""
    if conc <= 0:
        conc = 1e-3
    return rng.dirichlet(np.full(n, conc))


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max); the simplex constraint is handled
    later by a Euclidean projection.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # linear convolution, length 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)            # weights sum to 1

    # weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length n+(2n‑1)-1 = 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for temperature (lambda)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t); projection onto the simplex enforces Σ=1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # mask of indices attaining the maximum (within tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted gradient = 2·∑_{k∈K} shift(a,k)
        conv_rev = convolution_fft(a, mask[::-1])
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search using the true max‑convolution
    (fast ``compute_score``) as the energy.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = compute_score(cand)

        # Metropolis rule
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best


# ----------------------------------------------------------------------
# Linear‑programming improvement step (the “LP‑update” from the paper)
# ----------------------------------------------------------------------


def lp_improvement_step(f: np.ndarray,
                        M: float,
                        mix_weight: float,
                        deadline: float) -> np.ndarray:
    """
    One LP‑based improvement:
      maximise Σg  subject to  (f∗g)[k] ≤ M  (all k) , g ≥ 0 .
    The solution is scaled to Σg = 1 and mixed with the current f.
    """
    if not SCIPY_AVAILABLE:
        return f  # fall back silently if SciPy is missing

    n = len(f)
    m = 2 * n - 1

    # Build the Toeplitz‑type constraint matrix A with A_{k,j}=f_{k‑j}
    # (zero when the index is out of range).  Column j is f placed
    # starting at row j.
    A = np.zeros((m, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f

    # RHS – a tiny tolerance avoids infeasibility caused by rounding
    b = np.full(m, M + 1e-14)

    # maximise Σg  → minimise −Σg
    c = -np.ones(n)

    bounds = [(0.0, None)] * n

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})

    if not res.success:
        # LP failed – return the original vector
        return f

    g0 = res.x
    total = g0.sum()
    if total <= 0:
        return f

    g = g0 / total                     # scale to Σg = 1
    f_new = (1.0 - mix_weight) * f + mix_weight * g
    return project_to_simplex(f_new)


def lp_improvement(f: np.ndarray,
                   deadline: float,
                   mix_schedule: list[float]) -> np.ndarray:
    """
    Apply a short sequence of LP‑improvement steps with decreasing mixing
    weights (mix_schedule, e.g. [0.3, 0.2, 0.1]).
    """
    a = f.copy()
    for w in mix_schedule:
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        M = np.max(conv)
        a = lp_improvement_step(a, M, mix_weight=w, deadline=deadline)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback if a previous construction exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths (larger n gives more freedom)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]          # 8192 would be memory‑intensive

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        pool: list[tuple[float, np.ndarray]] = []

        # ----- Deterministic starters -----
        uniform = np.full(n, 1.0 / n)
        pool.append((compute_score(uniform), uniform.copy()))

        arc = arcsine_initial(n)
        pool.append((compute_score(arc), arc.copy()))

        # ----- Random Dirichlet (dense and sparse regimes) -----
        for _ in range(80):
            alpha = rng.uniform(0.05, 5.0)          # concentration
            vec = generate_dirichlet_vector(n, alpha, rng)
            pool.append((compute_score(vec), vec))

        # ----- Sparse random vectors (few non‑zeros) -----
        for sparsity in [0.005, 0.01, 0.02, 0.05]:
            k = max(2, int(sparsity * n))
            for _ in range(30):
                vec = generate_sparse_vector(n, k, rng)
                pool.append((compute_score(vec), vec))

        # ----- Beta‑family shapes -----
        for _ in range(80):
            a = rng.uniform(0.1, 2.5)
            b = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 1.0)
            vec = generate_beta_shape(n, a, b, eps)
            pool.append((compute_score(vec), vec))

        # ----- Generalised arcsine (power) -----
        for _ in range(80):
            p = rng.uniform(0.2, 2.0)
            eps = rng.uniform(0.0, 1.0)
            vec = generate_power_shape(n, p, eps)
            pool.append((compute_score(vec), vec))

        # ----- Mixture of arcsine & uniform -----
        for _ in range(40):
            w = rng.random()
            vec = generate_mixture_shape(n, w)
            pool.append((compute_score(vec), vec))

        # ----- Random rotations of arcsine / triangular shapes -----
        for _ in range(40):
            shift = rng.integers(0, n)
            vec = np.roll(arc, shift)
            pool.append((compute_score(vec), vec))

        # ----- Keep a handful of the best seeds -----
        pool.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in pool[:5]]

        # ------------------------------------------------------------------
        # 2.  Refine each promising seed
        # ------------------------------------------------------------------
        for seed_vec in top_seeds:
            if time.time() >= deadline - 5.0:
                break

            # ---- (a) LP‑based improvement (very cheap, often large gain) ----
            seed_vec = lp_improvement(seed_vec,
                                      deadline,
                                      mix_schedule=[0.3, 0.2, 0.1])

            # ---- (b) Adam on the smooth surrogate ----
            remaining = deadline - time.time()
            max_iter_adam = int(min(120_000, remaining * 30))
            print(f"  Adam smooth (max_iter={max_iter_adam}) …")
            refined, _ = adam_optimize_smooth(
                initial_a=seed_vec,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=2.0,
                lam_end=120.0,
                noise_every=3000,
                noise_scale=3e-5,
            )

            # ---- (c) Simulated annealing on the true max ----
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_sa = int(min(150_000, remaining * 30))
                print(f"  Simulated annealing (max_iter={max_iter_sa}) …")
                refined = simulated_annealing(
                    refined,
                    rng,
                    deadline,
                    max_iters=max_iter_sa,
                    init_temp=0.03,
                    decay=0.9995,
                )

            # ---- (d) Exact sub‑gradient descent ----
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_sub = int(min(80_000, remaining * 30))
                print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
                refined = subgradient_descent_exact(
                    refined,
                    deadline,
                    max_iter=max_iter_sub,
                    step0=0.03,
                )

            # ---- Official evaluation ----
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"  → official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception as exc:
                print("  Evaluation error:", exc)

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        total_iters = int(min(400_000, remaining * 30))

        # half sub‑gradient, half Adam
        it_sub = total_iters // 2
        it_adam = total_iters - it_sub

        print("\nFinal global polishing (sub‑gradient + Adam) …")
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.02)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=it_adam,
            lr0=0.09,
            lam_start=5.0,
            lam_end=120.0,
            noise_every=2000,
            noise_scale=1e-5,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
