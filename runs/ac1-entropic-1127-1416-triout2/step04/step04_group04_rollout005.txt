"""
search_step_function_v2.py

A more exhaustive heuristic search for a non‑negative step‑function that
minimises the official objective

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in four stages:

1. **Large parametric sweep** –  systematic grids over power‑law,
   symmetric beta, mixtures of the arcsine law with the uniform law,
   and many random Dirichlet draws.  Every candidate is normalised to
   sum = 1 and scored with a fast FFT‑based estimator.  The best few
   (default 15) are kept.

2. **Mirror‑descent (sub‑gradient) refinement** –  each retained
   candidate is improved by the same multiplicative‑gradient updates
   used in the previous baseline, but with a larger iteration budget
   that is dynamically adapted to the remaining time.

3. **Intensive polishing** – the globally best vector found so far is
   refined again with a longer run of the sub‑gradient method.

4. **Fallback** – if nothing is found (e.g. because of a timeout) a
   safe uniform vector is returned.

All operations are performed on the probability simplex (sum = 1),
so the value of the official objective is simply `2·n·max(conv)`.
The routine respects the supplied time budget (default 1000 s) and
always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the scale‑invariant objective.
    The vector is assumed non‑negative and (optionally) already normalised
    to sum = 1; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    # 2·n·max(conv) / (sum(a)²)   (exact official formula)
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    # reverse the weight vector to obtain the required convolution pattern
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = len(a)+len(w)-1 = 3·n‑2
    idx = 2 * len(a) - 2                            # index that aligns w_{m+j}
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex via exponential sampling."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Mirror‑descent (sub‑gradient) refinement
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true objective (max(conv)).
    Multiplicative updates (exponentiated gradient) keep the iterate
    non‑negative; after each step the vector is re‑projected onto the
    simplex.  The function returns the best vector seen and its fast
    (FFT‑based) score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # binary weight vector for the (sub‑)gradient of the max‑function
        w = np.zeros(conv_len, dtype=float)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)   # spread weight over all ties

        grad = gradient_from_weights(a, w)

        # classic 1/√t step size
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)      # multiplicative update
        a = normalize(a)                 # back to the simplex

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

        # fallback to a safe uniform vector (small length)
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)
        print("[fallback] using uniform 256‑length vector")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
