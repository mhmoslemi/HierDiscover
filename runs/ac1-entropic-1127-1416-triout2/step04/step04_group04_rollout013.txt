def refine_exponential(a0, deadline, max_iter=200000, eta0=0.5, beta0=1.0, beta_increment=0.001):
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    beta = beta0
    m = np.zeros_like(a)  # momentum term
    beta1 = 0.9  # momentum decay
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        # Update beta gradually
        beta = min(1000.0, beta + beta_increment)
        # softmax weights
        maxc = np.max(conv)
        w = np.exp(beta * (conv - maxc))
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.ones_like(conv) / conv.shape[0]
        else:
            w = w / w_sum
        grad = gradient_from_weights(a, w)
        # momentum
        m = beta1 * m + (1 - beta1) * grad
        # step size schedule
        eta = eta0 / (np.sqrt(t) * (np.max(np.abs(m)) + 1e-12))
        a = a * np.exp(- eta * m)
        a = normalize(a)
        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score = cur_sc
            best_a = a.copy()
        # optionally print intermediate info
    return best_a, best_score

   stabilises the updates.  The iterate is always projected onto the
   probability simplex.

3. **Final polish** – the current global best is refined once more with a
   longer run of the same optimiser.

All operations use FFT‑based convolutions for speed and the routine
observes the supplied time budget (default 1000 s).  The function
``propose_candidate`` returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative and is not re‑scaled here.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    tot = v.sum()
    if tot <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / tot


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_j = 2· Σ_m a_m·w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def exponential_decay_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ e^{‑λ·i}+e^{‑λ·(n‑1‑i)}."""
    i = np.arange(n, dtype=float)
    base = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return normalize(base)


def sine_shape(n: int, power: float) -> np.ndarray:
    """Shape a_i ∝ sin(π·(i+0.5)/n)^{power}."""
    i = np.arange(n, dtype=float)
    base = np.sin(np.pi * (i + 0.5) / n) ** power
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Smooth‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------


def refine_exponential(
    a0: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 1_000_000,
    eta0: float = 0.8,
    beta0: float = 1.0,
    beta_inc: float = 0.02,
) -> Tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent on the true objective.
    The non‑smooth max is approximated by a soft‑max with temperature β
    that is increased gradually.  A simple momentum term (β1≈0.9) is used.
    Returns the best vector seen and its fast‑score.
    """
    a = normalize(a0.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    # momentum accumulator (same shape as a)
    mom = np.zeros_like(a)
    beta1 = 0.9
    beta_cur = beta0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # 1) current self‑convolution
        conv = convolution_fft(a, a)

        # 2) update β (smoothness)
        beta_cur = min(5000.0, beta_cur + beta_inc)

        # 3) soft‑max weights for the (smoothed) max
        mx = np.max(conv)
        # shift by the max to avoid overflow
        w = np.exp(beta_cur * (conv - mx))
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # 4) gradient of the weighted sum (includes factor 2)
        grad = gradient_from_weights(a, w)

        # 5) momentum update
        mom = beta1 * mom + (1.0 - beta1) * grad

        # 6) step‑size schedule (classic 1/√t)
        eta = eta0 / np.sqrt(it)

        # 7) multiplicative (exponentiated) update and projection
        a = a * np.exp(-eta * mom)
        a = normalize(a)

        # 8) track best
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    score.  The routine respects the time budget and always returns a feasible
    list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin to avoid a hard timeout
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths we will explore.  4096 gives a finer discretisation, 2048 is
    # a cheaper backup, and 1024 ensures we still have something even if
    # time runs out early.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1. Build a modest library of initial candidates.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # deterministic families
        candidates.append(uniform_shape(n))
        candidates.append(arcsine_initial(n))
        candidates.append(beta_shape(n, 0.5))   # arcsine is beta(0.5,0.5)
        candidates.append(beta_shape(n, 0.3))
        candidates.append(beta_shape(n, 0.8))

        # power‑law variations
        for _ in range(6):
            p = rng.uniform(0.1, 0.9)
            eps = rng.uniform(0.0, 5.0)
            candidates.append(powerlaw_shape(n, p, eps))

        # exponential‑decay variations
        for _ in range(4):
            lam = rng.uniform(0.0005, 0.01) * n   # scale with length
            candidates.append(exponential_decay_shape(n, lam))

        # sine‑shape variations
        for _ in range(4):
            power = rng.uniform(1.0, 4.0)
            candidates.append(sine_shape(n, power))

        # mixtures of arcsine with uniform (helps smooth singularities)
        for _ in range(5):
            w = rng.uniform(0.0, 0.4)
            mix = (1.0 - w) * arcsine_initial(n) + w * uniform_shape(n)
            candidates.append(normalize(mix))

        # random Dirichlet draws
        for _ in range(8):
            candidates.append(random_dirichlet(n, rng))

        # --------------------------------------------------------------
        # 2. Quick ranking with the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored: List[Tuple[float, np.ndarray]] = [
            (compute_score(c), c) for c in candidates
        ]
        scored.sort(key=lambda x: x[0])
        top_k = 6
        top_candidates = [c for _, c in scored[:top_k]]
        print(
            "Top initial scores:",
            " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]),
        )

        # --------------------------------------------------------------
        # 3. Refine each retained candidate with smooth‑max exponential‑grad.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            start_sc = compute_score(cand)
            print(f"\nCandidate {idx + 1}/{len(top_candidates)} start_score = {start_sc:.6f}")

            # give each candidate a fair slice of the remaining budget
            remaining = max(0.0, deadline - time.time())
            # empirical budget: about 300 iterations per second for n≈4000
            max_iter = int(min(300_000, max(5_000, remaining * 300)))

            refined, _ = refine_exponential(
                a0=cand,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.8,
                beta0=0.5,
                beta_inc=0.03,
            )

            # final official evaluation
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 4. Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n===== Final polishing of the global best =====")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(500_000, max(20_000, remaining * 300)))

        polished, _ = refine_exponential(
            a0=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
            beta0=2.0,
            beta_inc=0.05,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector (short)
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
