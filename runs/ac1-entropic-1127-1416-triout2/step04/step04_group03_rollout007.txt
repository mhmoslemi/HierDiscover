"""
Step‑function minimiser (≈ 1000 s budget)

The optimiser works in three phases:

1. **Quick scouting** – test a handful of lengths (512 … 8192) with a
   few analytic shapes (uniform, arcsine, random symmetric beta) and keep
   the best normalised vector according to the fast surrogate
   `2·n·max(conv(a,a))`.

2. **Multi‑stage local optimisation** – starting from that vector we run
   *Adam* on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The gradient of the surrogate is now computed exactly (the previous
   version used a mismatched index).  After Adam we polish with a
   projected sub‑gradient descent on the true max and finish with a
   stochastic local walk on the simplex.  Each stage respects the global
   deadline.

3. **Final polishing** – the globally best candidate is refined once
   more with longer Adam / sub‑gradient / random‑walk budgets.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
so that the official `evaluate_sequence` never returns ``inf``.
The routine prints timing / improvement information and finally returns
the best sequence found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)) for a already normalised
    (Σa = 1).  This equals the official evaluation value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """Classic symmetric arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – corrected version
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient of the full objective, surrogate_value B).
    The surrogate B approximates max(conv) and the gradient uses the
    exact derivative of B (see derivation in the analysis).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = float(np.max(conv))

    # stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                           # soft‑max weights, sum to 1
    B = max_c + np.log(sum_exp) / lam                # smooth approximation of max

    # gradient of B w.r.t. a:
    #   ∂B/∂a = 2 * (w * a_rev)  (where * is linear convolution)
    a_rev = a[::-1]                                  # reversed a
    conv_w_rev = convolution_fft(w, a_rev)           # length 3n‑2
    grad_B = 2.0 * conv_w_rev[n - 1 : n - 1 + n]      # slice of size n

    s = np.sum(a)                                     # = 1 after projection
    term1 = (2.0 * n / (s ** 2)) * grad_B
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    noise_every: int = 1500,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature (quadratic schedule → smooth → sharper)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(*a.shape) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam it={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on   max(conv(a,a)).
    Step size decays like step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # active gradient: ∂ (a*a)_idx / ∂a_i = 2 * a_{idx-i} (if in range)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad it={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc

# Stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex: add Gaussian noise, project back,
    keep improvements (tiny chance of uphill moves for escape).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        # Accept if improvement or with a tiny escape probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk it={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc

    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Quick scouting to pick a promising length and shape
    # ------------------------------------------------------------------
    scout_lengths = [512, 1024, 2048, 4096, 8192]
    scout_best_a = None
    scout_best_score = float("inf")
    scout_best_n = None

    for n in scout_lengths:
        if time.time() >= deadline - 6.0:
            break
        print(f"\n--- Scouting length {n} ---")

        candidates = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),                     # uniform
        ]

        # a few random beta‑family members (symmetry enforced by α≈β)
        for _ in range(12):
            alpha = rng.uniform(0.05, 2.0)
            beta = rng.uniform(0.05, 2.0)
            eps = rng.uniform(1e-6, max(0.01, 0.02 * n))
            candidates.append(generate_beta_shape(n, alpha, beta, eps))

        for cand in candidates:
            sc = compute_score(cand)
            if sc < scout_best_score:
                scout_best_score = sc
                scout_best_a = cand.copy()
                scout_best_n = n

    if scout_best_a is None:
        # very unlikely, but keep a safe uniform fallback
        scout_best_n = 1024
        scout_best_a = np.full(scout_best_n, 1.0 / scout_best_n)

    print(f"\nChosen length for refinement: n = {scout_best_n}")
    print(f"Scouting score (fast surrogate) = {scout_best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Multi‑stage optimisation of the chosen candidate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate most of the remaining budget to Adam (the most powerful stage)
        max_iter_adam = int(min(600_000, remaining * 60))
        print(f"  Adam stage – max_iter = {max_iter_adam}")
        a_refined, _ = adam_optimize(
            initial_a=scout_best_a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.15,
            lam_start=0.5,
            lam_end=120.0,
            noise_every=800,
            noise_scale=5e-5,
        )

        # sub‑gradient polishing
        remaining = deadline - time.time()
        if remaining > 5.0:
            max_iter_sub = int(min(250_000, remaining * 35))
            print(f"  Sub‑gradient stage – max_iter = {max_iter_sub}")
            a_refined, _ = subgradient_optimize(
                a_refined,
                time_deadline=deadline,
                step0=0.25,
                max_iter=max_iter_sub,
            )

        # random‑local polishing
        remaining = deadline - time.time()
        if remaining > 4.0:
            max_iter_loc = int(min(300_000, remaining * 40))
            print(f"  Random‑walk stage – max_iter = {max_iter_loc}")
            a_refined, _ = random_local_search(
                a_refined,
                time_deadline=deadline,
                init_step=0.015,
                max_iter=max_iter_loc,
            )

        # Evaluate the candidate with the official (slow) function
        val = evaluate_sequence(a_refined.tolist())
        print(f"Refined candidate official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_refined.tolist()
            print(f"*** New global best after refinement: {best_val:.6f} ***")
    else:
        # Not enough time – just evaluate the scouted vector
        val = evaluate_sequence(scout_best_a.tolist())
        print(f"Scouted candidate official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = scout_best_a.tolist()

    # ------------------------------------------------------------------
    # 3.  Final polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Final polishing of the global best ===")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_adam = int(min(400_000, remaining * 50))
        print(f"  Final Adam – max_iter = {max_iter_adam}")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.12,
            lam_start=0.5,
            lam_end=150.0,
            noise_every=1000,
            noise_scale=4e-5,
        )

        remaining = deadline - time.time()
        if remaining > 3.0:
            max_iter_sub = int(min(150_000, remaining * 30))
            print(f"  Final sub‑gradient – max_iter = {max_iter_sub}")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.2,
                max_iter=max_iter_sub,
            )

        remaining = deadline - time.time()
        if remaining > 2.0:
            max_iter_loc = int(min(200_000, remaining * 35))
            print(f"  Final random walk – max_iter = {max_iter_loc}")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.012,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # Simple safe fallback (uniform 256‑vector)
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
```

This implementation fixes the gradient computation for the smooth log‑sum‑exp surrogate, expands the scouting phase to include a larger set of lengths, and allocates the remaining time to more aggressive Adam / sub‑gradient / random‑walk polishing. The corrected gradient typically leads to a steeper descent and, together with the broader search, should push the final score below the previous 1.5146 and hopefully reach the target ≈ 1.503.
