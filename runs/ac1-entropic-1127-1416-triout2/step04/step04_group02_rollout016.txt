"""
Hybrid smooth‑max gradient descent for the min‑max self‑convolution problem.

The algorithm works with a probability vector  a (a_i≥0, Σa_i=1)  so that the
objective reduces to  2·n·max_k (a∗a)_k .  We minimise the non‑smooth max via a
log‑sum‑exp “smooth‑max’’ with a temperature γ that is increased during the run.
The gradient of the smooth‑max can be written analytically and evaluated entirely
with FFTs, giving an O(N log N) update that works for vectors of several thousand
entries.  After a long annealing phase we finish with a short polishing run
using a smaller step‑size.  The best sequence found (scaled back to the original
range) is returned before the time budget expires.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = len(a) + len(b) - 1
    N = next_pow2(n)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:n]
    return conv


def valid_correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the “valid’’ (length‑n) correlation  g_j = Σ_i a_i·w_{i+j}
    using the FFT.  a has length n, w has length 2·n‑1.
    """
    n = len(a)
    m = len(w)                 # = 2·n‑1
    N = next_pow2(n + m - 1)   # ≥ 3·n‑2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr_full = np.fft.irfft(A * np.conj(W), N)[: n + m - 1]
    start = m - 1
    return corr_full[start: start + n]


def cheap_score(a: np.ndarray) -> float:
    """Fast proxy for the official score (assumes Σa = 1)."""
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Families for initialisation
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine law, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1}(n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑like vector (Γ‑distributed) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.9,
    gamma_start: float = 2.0,
    gamma_mul: float = 1.02,
    gamma_max: float = 200.0,
    eval_every: int = 2000,
) -> np.ndarray:
    """
    Exponentiated‑gradient minimisation of  max_k (a∗a)_k  via a smooth
    approximation (log‑sum‑exp).  The vector is kept on the simplex.
    """
    a = init_a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / len(a))
    a /= a.sum()                 # work on the probability simplex

    n = len(a)
    best_a = a.copy()
    best_sc = cheap_score(a)

    gamma = gamma_start
    it = 0

    while time.time() < deadline:
        it += 1
        # ----- convolution a∗a (full) -----
        conv = convolution_fft(a, a)        # length 2·n‑1

        # ----- smooth max: weights w_k ∝ exp(γ·conv_k) -----
        # subtraction of the max keeps the exponent well‑scaled
        max_gc = np.max(gamma * conv)
        w = np.exp(gamma * conv - max_gc)   # w ≥ 0, max w = 1
        Z = w.sum() + 1e-30

        # ----- gradient of smooth‑max (using correlation) -----
        grad = 2.0 * valid_correlation_fft(a, w) / Z   # length n

        # ----- exponentiated (mirror) update -----
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # ----- renormalise onto the simplex (avoid exact zeros) -----
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # ----- increase temperature slowly -----
        if gamma < gamma_max:
            gamma = min(gamma_max, gamma * gamma_mul)

        # ----- occasional exact evaluation -----
        if it % eval_every == 0:
            cur_sc = cheap_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
            # optional progress output
            print(f"    iter {it:,d}  γ={gamma:.1f}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a

            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
