"""
Propose a step‑function that (approximately) minimises
evaluate_sequence().

Algorithm
---------
* Load the previously found best vector (height_sequence_1) if it exists.
* Generate many random starting vectors from a variety of families
  (power‑law, exponential, Gaussian‑like, Dirichlet, edge‑heavy).  A
  random cyclic shift is applied to break symmetry.
* For each start run a long mirror‑descent on the smooth approximation
  f_α(a) = (1/α)·log Σ exp(α·(a∗a)_k).  The smoothing parameter α grows
  from a modest value up to several thousand, and the step size decays
  like 1/√t.  The update a←a·exp(−η·g) (exponentiated gradient) keeps
  a on the probability simplex (projected by the standard simplex‑
  projection routine).
* After the smoothed phase perform a few rounds of exact sub‑gradient
  descent on the true ℓ∞‑norm (using the max convolution entry only)
  with a projection onto the simplex.  This fine‑tunes the solution
  and often yields the last tiny improvements.
* Keep the best vector seen across all restarts and return it
  (already normalised to sum = 1, which is the natural scaling for
  evaluate_sequence).

The whole search respects the 1000 s budget; intermediate progress
(and the current best score) is printed for diagnostics.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v_i ≥ 0)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Return v scaled to have sum = 1 (preserves non‑negativity)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Convolution / gradient helpers (FFT based)
# ----------------------------------------------------------------------


def self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a (full length = 2·len(a)−1)."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return conv


def gradient_from_softmax(p: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Compute g_i = Σ_k p_k·a_{k‑i}  (cross‑correlation of a with p)
    using FFT of length L (≥ len(p)+len(a)−1).
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=L)
    fft_p = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_p), n=L)
    offset = p.shape[0] - 1
    return corr[offset:offset + n]


def subgradient_max_conv(a: np.ndarray, L: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    Takes the unique maximal index (if several, averages the contributions).
    """
    n = a.shape[0]
    conv = self_convolution(a)
    max_val = conv.max()
    # indices attaining the maximum (within a tiny tolerance)
    tol = 1e-12
    idxs = np.nonzero(conv >= max_val - tol)[0]
    # build indicator p for these indices (average if more than one)
    p = np.zeros_like(conv)
    p[idxs] = 1.0 / len(idxs)
    # cross‑correlation gives the sub‑gradient
    return gradient_from_softmax(p, a, L)


# ----------------------------------------------------------------------
# Families of initial vectors (all returned already normalised)
# ----------------------------------------------------------------------


def init_power(n: int) -> np.ndarray:
    """a_i ∝ (i+1)^γ with γ ∈ [−5,5]."""
    gamma = np.random.uniform(-5.0, 5.0)
    x = np.arange(1, n + 1, dtype=float)
    a = np.power(x, gamma)
    a = simplex_normalise(a)
    shift = np.random.randint(0, n)
    a = np.roll(a, shift)
    return a


def init_exponential(n: int) -> np.ndarray:
    """a_i ∝ exp(−λ·i) with λ ∈ [1e‑4, 2e‑2]."""
    lam = np.random.uniform(1e-4, 2e-2)
    x = np.arange(n, dtype=float)
    a = np.exp(-lam * x)
    a = simplex_normalise(a)
    shift = np.random.randint(0, n)
    a = np.roll(a, shift)
    return a


def init_gaussian(n: int) -> np.ndarray:
    """Discrete Gaussian with random centre and width."""
    mu = np.random.uniform(0, n - 1)
    sigma = np.random.uniform(0.1 * n, 0.4 * n)
    x = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = simplex_normalise(a)
    shift = np.random.randint(0, n)
    a = np.roll(a, shift)
    return a


def init_dirichlet(n: int) -> np.ndarray:
    """Dirichlet(1,…,1) (uniform‑Gamma)."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    a = simplex_normalise(a)
    shift = np.random.randint(0, n)
    a = np.roll(a, shift)
    return a


def init_edge_heavy(n: int) -> np.ndarray:
    """Linear weight larger near the two ends, smaller in the centre."""
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = dist_to_center.astype(float) + 1.0
    a = simplex_normalise(a)
    shift = np.random.randint(0, n)
    a = np.roll(a, shift)
    return a


def init_sparse(n: int) -> np.ndarray:
    """Place total mass on a small random support."""
    k = np.random.randint(5, 30)
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    a = simplex_normalise(a)
    shift = np.random.randint(0, n)
    a = np.roll(a, shift)
    return a

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimizes
    evaluate_sequence() within ``budget_s`` seconds.
    Returns the best sequence found (as a list of floats, sum == 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # Load previous solution if available; otherwise start from uniform.
    # ------------------------------------------------------------------
    try:
        init_best = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        init_best = simplex_normalise(init_best)
        n = init_best.size
        best_a = init_best.copy()
        best_score = 2.0 * n * self_convolution(best_a).max()
        print(f"[Init] Loaded height_sequence_1 (n={n}), score={best_score:.6f}")
    except NameError:
        n = 5146
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = 2.0 * n * self_convolution(best_a).max()
        print(f"[Init] No previous vector – using uniform start, n={n}, score={best_score:.6f}")

    # pre‑compute the FFT length needed for the cross‑correlation
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # Collection of initialisers
    # ------------------------------------------------------------------
    init_fns = [
        lambda: init_power(n),
        lambda: init_exponential(n),
        lambda: init_gaussian(n),
        lambda: init_dirichlet(n),
        lambda: init_edge_heavy(n),
        lambda: init_sparse(n),
    ]

    restart = 0
    # main restart loop – stop early if time runs out
    while time.time() < deadline - 6.0 and restart < 200:
        # ----------- pick a starting vector -----------------------------
        a = np.random.choice(init_fns)()
        # occasionally blend with current global best (helps keep good mass)
        if np.random.rand() < 0.15:
            a = 0.5 * a + 0.5 * best_a
            a = simplex_projection(a)

        # ----------- smoothed mirror descent ----------------------------
        max_iters = 6000
        eta0 = 1.0
        alpha = 5.0          # start smoothing
        alpha_target = 2000.0
        for it in range(max_iters):
            # time guard
            if time.time() > deadline - 4.0:
                break

            # 1) self‑convolution
            conv = self_convolution(a)
            maxc = conv.max()

            # 2) soft‑max (log‑sum‑exp) weights
            # shift by maxc for numerical stability
            exp_vals = np.exp(alpha * (conv - maxc))
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                p = np.full_like(conv, 1.0 / conv.size)
            else:
                p = exp_vals / sum_exp

            # 3) gradient of the smoothed objective
            g = gradient_from_softmax(p, a, L)

            # 4) exponentiated‑gradient update (mirror descent on simplex)
            eta = eta0 / math.sqrt(it + 1.0)
            a = a * np.exp(-eta * g)
            a = np.maximum(a, 1e-16)          # avoid exact zeros
            a = simplex_projection(a)

            # 5) increase smoothing parameter gradually
            alpha += (alpha_target - alpha) / max_iters

            # occasional reporting
            if (it & 31) == 0:
                cur_score = 2.0 * n * maxc
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Restart {restart}] it {it:4d} → new best {best_score:.6f}")

        # ----------- exact ℓ∞ sub‑gradient refinement -------------------
        # start from the current global best (may be already improved)
        cur_a = best_a.copy()
        cur_score = best_score
        # a simple geometric cooling for the step size
        step = cur_score * 0.02
        step_min = cur_score * 1e-6
        while time.time() < deadline - 2.0 and step > step_min:
            # sub‑gradient w.r.t. the true max
            g_sub = subgradient_max_conv(cur_a, L)

            # additive projected gradient step
            cand = cur_a - step * g_sub
            cand = simplex_projection(cand)

            # evaluate candidate
            cand_score = 2.0 * n * self_convolution(cand).max()
            if cand_score < cur_score:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[Refine R{restart}] step {step:.2e} → new best {best_score:.6f}")
                # keep the step size, maybe even increase a little
                step *= 0.98
            else:
                # failed – shrink the step
                step *= 0.95

        restart += 1

    print(f"[Done] Best score after {restart} restarts: {best_score:.6f}")
    # return as plain Python list (evaluate_sequence expects a list)
    return [float(v) for v in best_a.tolist()]
