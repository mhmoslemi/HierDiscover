def gradient_smoothmax(a: np.ndarray, beta: float) -> np.ndarray:
   conv = convolution_fft(a, a)
   # compute softmax weights
   # to avoid overflow, subtract max before exponentiate
   maxc = np.max(conv)
   expb = np.exp(beta * (conv - maxc))
   weights = expb / np.sum(expb)
   # compute weighted sum of gradients via cross-correlation
   # w = weights (len 2n-1)
   # compute conv between a and reversed w
   w_rev = weights[::-1]
   grad_conv = convolution_fft(a, w_rev)  # length = 3n-2
   n = len(a)
   grad = 2.0 * grad_conv[2*n - 2 - np.arange(n)]
   # now we have gradient of smoothmax w.r.t a.
   return grad

   # Use standard algorithm like sort.
   # If sum of max(v,0) is zero, return uniform.

   u = np.sort(v)[::-1]

   rho = ind[cond][-1]

def grad_smoothmax(a, beta):
    n = len(a)
    conv = convolution_fft(a, a)
    # Compute softmax weights
    maxc = conv.max()
    # subtract maxc for stability
    w = np.exp(beta * (conv - maxc))
    sumw = w.sum()
    if sumw == 0:
        w = np.full_like(w, 1.0 / len(w))
    else:
        w /= sumw
    # Compute cross-correlation gradient
    w_rev = w[::-1]
    # Convolution a and w_rev using FFT
    grad_conv = convolution_fft(a, w_rev)  # length n + (2n-1) -1 = 3n-2
    # indices 2n-2 .. n-1 inclusive for gradient
    idx_start = 2 * n - 2
    # grad = 2 * grad_conv[idx_start - np.arange(n)]
    grad = 2.0 * grad_conv[idx_start - np.arange(n)]
    return grad, conv.max()
```

def optimise_smooth(initial_a, deadline, max_iter=200_000,
                    beta_start=5.0, beta_end=200.0, eta0=0.5,
                    noise_every=0, noise_scale=0.0):
    a = np.maximum(initial_a, 1e-30)  # avoid zero
    a = a / a.sum()  # simplex
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    # Precompute ratio for linear schedule? Provide per iteration.
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break

        # schedule beta (linear)
        beta = beta_start + (beta_end - beta_start) * (it - 1) / max_iter
        # schedule stepsize
        eta = eta0 / np.sqrt(it)  # sqrt schedule

        grad, cur_max = grad_smoothmax(a, beta)

        # multiplicative update
        a = a * np.exp(-eta * grad)

        # optional noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # renorm
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # evaluate using compute_score (fast) for monitoring
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

"""
Optimisation of a non‑negative step function for the max‑convolution objective.

The algorithm works in three stages:

1. **Exploration** – try several lengths (powers of two) and a diverse
   collection of initial shapes (uniform, arcsine, power‑law, beta‑like,
   Dirichlet).  The cheap “compute_score’’ (the exact objective with the
   denominator normalised to 1) is used to keep the most promising candidates.

2. **Smooth‑max mirror descent** – each retained candidate is refined by a
   multiplicative (exponentiated‑gradient) update on the probability simplex.
   Instead of a hard max we minimise a *soft‑max* approximation
   `softmax_β(b) = log‑sum‑exp(β·b)/β`.  The gradient of this smooth objective
   can be written as a simple cross‑correlation, which we evaluate with an FFT
   based convolution (O(n log n)).  The temperature `β` is increased
   linearly from a small value (easy landscape) to a large one (close to the
   true max), while the step size follows a `1/√t` schedule.
   After each update the vector is re‑normalised and explicitly symmetrised
   (`a = (a + a[::-1]) / 2`) – symmetry is known to be optimal for this
   problem and reduces the search space.

3. **Final polishing** – the best sequence found is run again with a higher
   final temperature and a smaller initial step size to obtain a fine‑tuned
   solution.

All updates keep the entries non‑negative and the sum equal to one,
so the denominator of the original objective is constant and we only need
to drive the (smooth) maximum of the self‑convolution down.

The routine respects the supplied time budget (default 1000 s) and always
returns a feasible list of floats before the deadline.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution using a real FFT.  The result length is
    len(a) + len(b) – 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (the denominator is 1 because
    the vector is normalised to sum = 1).  Returns 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1}(n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (gamma draws) and normalise."""
    if concentration <= 0:
        # heavy‑tailed “sparse’’ draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smooth‑max gradient utilities
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the (β‑softmax) weights w_i = exp(β·c_i) / Σ_j exp(β·c_j)
    in a numerically stable way.
    """
    max_c = conv.max()
    # subtract max to avoid overflow; values are ≤ 0 after subtraction
    exp_term = np.exp(beta * (conv - max_c))
    sum_exp = exp_term.sum()
    if sum_exp == 0.0:
        # should never happen, but fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / sum_exp


def grad_softmax_max(a: np.ndarray, beta: float) -> Tuple[np.ndarray, float]:
    """
    Gradient (w.r.t. a) of the β‑soft‑max of the self‑convolution.
    Returns (gradient, current (hard) max of the convolution) for logging.
    """
    n = len(a)

    # Self‑convolution
    conv = convolution_fft(a, a)

    # Soft‑max weights
    w = softmax_weights(conv, beta)                # length = 2n‑1

    # Cross‑correlation a ⋆ w  (equivalently convolution with reversed w)
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)   # length = 3n‑2

    # Index of the element that aligns with offset p:
    idx_start = 2 * n - 2
    grad = 2.0 * corr[idx_start - np.arange(n)]

    # Return gradient and the *hard* maximum (useful for diagnostics)
    return grad, float(conv.max())


# ----------------------------------------------------------------------
# Optimiser based on exponentiated (mirror) gradient descent
# ----------------------------------------------------------------------


def optimise_smooth(
    initial_a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    eta0: float = 0.6,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the β‑soft‑max of the self‑convolution.
    The vector is kept on the probability simplex (non‑negative, sum = 1)
    and explicitly symmetrised after every update.
    Returns the best vector seen (as a NumPy array) and its fast score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()                     # initialise on the simplex

    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear annealing of the temperature β
        beta = beta_start + (beta_end - beta_start) * (it - 1) / max_iter

        # step size (standard 1/√t schedule)
        eta = eta0 / np.sqrt(it)

        grad, cur_max = grad_softmax_max(a, beta)

        # multiplicative (exponentiated) update
        a = a * np.exp(-eta * grad)

        # optional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # enforce symmetry (a_i = a_{n-1-i})
        a = 0.5 * (a + a[::-1])

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(
                f"    iter={it:,d}  β={beta:6.2f}  max_conv={cur_max:.6e}"
                f"  cur_score={cur_sc:.6f}  best_score={best_sc:.6f}"
            )

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine explores several lengths, tries a variety of
    initial shapes, refines them with the smooth‑max optimiser and finally
    performs a polishing run on the globally best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final official evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional fallback (if a previously discovered construction is
    #     available in the global namespace under the name
    #     ``height_sequence_1``).
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Lengths to explore – powers of two are cheap for FFTs.
    # --------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ----- 1a. Build a diverse pool of starting vectors -----
        candidates: List[np.ndarray] = []

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # arcsine (classic)
        candidates.append(arcsine_initial(n))

        # power‑law families
        for _ in range(4):
            p = rng.uniform(0.1, 1.2)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            candidates.append(generate_powerlaw(n, p, eps))

        # beta‑like families
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            candidates.append(generate_beta_like(n, alpha, beta))

        # Dirichlet draws (dense and sparse)
        for conc in [0.3, 0.7, 1.5, 2.0]:
            candidates.append(random_dirichlet(n, conc, rng))

        # perturbed arcsine – helps to escape a possible local basin
        for _ in range(2):
            candidates.append(perturb(arcsine_initial(n), rng, scale=0.015))

        # ----- 1b. Quick sweep with the cheap score, keep the top few -----
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:4]]   # keep the four most promising

        # ----- 1c. Refine each retained candidate with the smooth optimiser -----
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # allocate a slice of the remaining budget
            remaining = max(0.0, deadline - time.time())
            # Rough estimate: about 150 iterations per second for n≈8000,
            # 300 it/s for n≈2000.  Scale max_iter accordingly.
            it_per_sec = 300.0 if n <= 2048 else 150.0
            max_iter = int(min(150_000, max(4_000, remaining * it_per_sec)))

            refined, _ = optimise_smooth(
                initial_a=cand,
                deadline=deadline,
                max_iter=max_iter,
                beta_start=5.0,
                beta_end=200.0,
                eta0=0.6,
                noise_every=500,
                noise_scale=5e-6,
            )

            # official evaluation
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ----------------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence (if time permits).
    # ----------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        it_per_sec = 300.0 if len(a) <= 2048 else 150.0
        max_iter = int(min(250_000, remaining * it_per_sec))

        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_smooth(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter,
            beta_start=50.0,      # start already in a “warm’’ regime
            beta_end=600.0,       # push very close to the true max
            eta0=0.3,             # smaller steps for fine‑tuning
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ----------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ----------------------------------------------------------------------
    if not best_seq:
        # Very safe fallback: a modest uniform vector (length 256)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
