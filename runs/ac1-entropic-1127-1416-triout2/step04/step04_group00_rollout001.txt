"""
Propose a low‑score step function.

Algorithm
---------
1. **Broad random search** – generate many candidate vectors of various
   lengths (1024, 2048, 4096).  Candidates are created by several simple
   random models:
   * uniform weight on a random sparse subset,
   * Dirichlet‑distributed weights on a random subset,
   * one‑sided exponential decay,
   * symmetric exponential decay.
   The cheap fast‑FFT replica of `evaluate_sequence` (``compute_score``)
   is used for rapid screening; the best candidate is kept.

2. **Smooth p‑norm optimisation** – starting from the best candidate,
   minimise a smooth approximation of the true objective.
   For a probability vector `a` (∑a = 1) let
       c = a * a                (full convolution)
   and
       F_p(a) = ( Σ c_k^p )^{1/p} .
   As `p → ∞`, `F_p(a)` converges to `max_k c_k`.  The gradient of `F_p`
   can be written in closed form using a correlation of `a` with
   `c^{p‑1}`; both operations are performed with an FFT‑based convolution.
   Projected gradient descent (Euclidean simplex projection) with a
   decreasing step‑size and a continuation schedule for `p`
   (gradually from 4 to 32) is applied.  The best vector seen during the
   run is retained.

3. **Final hill‑climber** – a stochastic mass‑transfer / swap hill‑climber
   (the same routine used in the previous baseline) polish the result.

All work respects the 1000 s budget; progress and intermediate scores are
printed.  The function returns a plain Python list of non‑negative floats
ready for the official `evaluate_sequence` harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, projection, fast scoring)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution of real vectors ``a`` and ``b`` using a real FFT.
    Returns an array of length ``len(a)+len(b)-1``.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for a vector with sum 1.
    Returns 2·n·max(a*a) exactly as the official evaluator would.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Hill‑climber (swap / mass transfer) – same as in the baseline

def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    Uses Euclidean projection onto the simplex after each move.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # random swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
                    cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Smooth p‑norm optimisation (continuation from p=4 → p=32)
# ----------------------------------------------------------------------


def smooth_pnorm_optimize(init_a: np.ndarray,
                          time_deadline: float,
                          max_iter: int = 20_000,
                          p_start: float = 4.0,
                          p_end: float = 32.0,
                          step0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation F_p(a) of the max‑convolution.
    ``init_a`` must already sum to 1.
    Uses projected gradient descent with a diminishing step size and a
    linear continuation schedule for the norm exponent p.
    Returns the best vector (as a numpy array) and its corresponding
    (fast) score 2·n·maxconv.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    # Pre‑allocate buffer for reversed version – will be updated each iter
    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # continuation for p
        p = p_start + (p_end - p_start) * (t - 1) / (max_iter - 1)

        # convolution a*a
        conv = convolution_fft(a, a)                # length 2n‑1
        # p‑norm of the vector of convolution entries
        conv_pow = conv ** p
        Lp = np.sum(conv_pow) ** (1.0 / p)

        # weight w = conv^{p‑1} (used in the gradient)
        if p == 1.0:
            w = np.ones_like(conv)
        else:
            w = conv ** (p - 1.0)

        # gradient = 2·p / Lp^{p‑1} · corr(a, w)
        rev_a = a[::-1]
        corr_full = convolution_fft(rev_a, w)      # length 3n‑2
        grad = corr_full[n - 1:n - 1 + n]           # size n
        scale = 2.0 * p / (Lp ** (p - 1.0) + 1e-30)
        grad = scale * grad

        # step size – classic 1/√t schedule
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        # occasional evaluation (every 100 iterations)
        if t % 100 == 0:
            cur_score = compute_score(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()

        a = a_new

    # final check
    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = best_a.copy()

    return best_a, best_score

        cur_score4 = compute_score(a4)
        if cur_score4 < best_val:
            best_val = cur_score4
            best_seq = a4.tolist()
            print(f"[rand {iter_cnt:04d}] New best (exp‑sym) "
                  f"score={best_val:.6f} (n={n})")

        iter_cnt += 1

        # quick escape if we are very close to the deadline
        if time.time() > rand_search_end:
            break

    # ------------------------------------------------------------------
    # 2. Smooth p‑norm optimisation (continuation)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print("\n=== Smooth p‑norm optimisation phase ===")
        a_init = np.array(best_seq, dtype=float)
        # allocate most of the remaining budget to this phase
        phase_deadline = deadline - 150.0
        refined_a, refined_score = smooth_pnorm_optimize(
            a_init,
            time_deadline=phase_deadline,
            max_iter=30_000,
            p_start=4.0,
            p_end=32.0,
            step0=0.12,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Smooth optimisation improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n=== Final hill‑climb polishing ===")
        refined_seq, refined_val = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=400_000,
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Safeguard – fall back to a simple uniform vector if nothing found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (the harness expects this)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
