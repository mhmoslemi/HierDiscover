    print(f"Initial score = {best_score:.6f}")

    # logits initialization

    logits = np.log(a + 1e-12)

    v = np.zeros_like(logits)

    lam = 5.0
    lam_max = 2000.0
    lam_factor = 1.001

    it = 0
    while time.time() < deadline - 0.2:
        # compute probabilities
        max_logit = np.max(logits)
        exp_logits = np.exp(logits - max_logit)   # avoid overflow
        a = exp_logits / np.sum(exp_logits)
        # compute convolution via FFT (real, length 2n-1)
        L_conv = 2 * n - 1
        fft_a = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(fft_a * fft_a, n=L_conv)

        scaled = lam * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w_sum = np.sum(w)

"""
Smooth‑max Adam optimiser for the self‑convolution bound.

We view the problem as:  minimise   F(a)=max_k (a*a)[k]   under
a_i≥0, Σa_i=1.  The objective is scale‑invariant, so we keep a on the
probability simplex throughout.  Because the max‑operator is non‑smooth,
we replace it by the log‑sum‑exp (soft‑max) with a temperature λ which
is slowly increased – this yields a smooth approximation that gradually
focuses on the true maximal entry.

The variables are the logits z so that a_i = softmax(z)_i.
Gradient of the smoothed objective is obtained analytically:
    ∂F/∂a = 2·corr( w , a )       with   w_k ∝ exp( λ·conv_k )
where conv = a*a and corr(·,·) is the (valid) cross‑correlation.
The chain‑rule gives the gradient w.r.t. the logits,
which is fed to an Adam optimiser.
The best feasible vector (according to the true max) is recorded and
returned at the end of the allotted time.

The implementation runs in O(L log L) per iteration (L≈2ⁿ⁺¹ power of two)
using FFTs, so tens of thousands of updates fit easily into the 1000 s
budget.  It also works with the previously discovered
`height_sequence_1` if that variable exists.
"""

import time
import math
import numpy as np


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of ``evaluate_sequence`` for a normalised vector
    (Σa = 1).  Returns 2·n·max(conv) where conv = a * a (linear convolution).
    """
    n = a.shape[0]
    L = 2 * n - 1
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return 2.0 * n * np.max(conv)


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    """
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow for extreme α,β
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def softmax(logits: np.ndarray) -> np.ndarray:
    """
    Numerically stable soft‑max returning a probability vector.
    """
    m = np.max(logits)
    ex = np.exp(logits - m)
    return ex / np.sum(ex)


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence minimising
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence found (still normalised to
    sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # 1 – initialise a probability vector a
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.size
        a = np.maximum(0.0, init)
        s = a.sum()
        if s == 0.0:
            a = np.ones(n) / n
        else:
            a = a / s
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        # fall back to a U‑shaped Beta(0.5,0.5) start
        n = 5146                          # same size as the previous run
        a = beta_sequence(0.5, 0.5, n)
        print(f"No existing sequence – using Beta(0.5,0.5) start, length {n}.")

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – Adam optimisation on the logits (soft‑max parametrisation)
    # ------------------------------------------------------------------
    # logits such that a = softmax(logits)
    logits = np.log(a + 1e-16)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1                     # base stepsize for logits

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # schedule for the smooth‑max temperature λ
    lam = 5.0                    # start fairly smooth
    lam_max = 2000.0
    lam_factor = 1.001           # geometric increase each iteration

    # pre‑computed FFT length (next power of two >= 3·n)
    L_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    last_improve_it = 0

    while time.time() < deadline - 0.2:
        it += 1

        # ----- a = softmax(logits) -----
        a = softmax(logits)

        # ----- self‑convolution of a -----
        fft_a = np.fft.rfft(a, n=L_fft)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L_fft)
        conv = conv_full[: 2 * n - 1]          # the genuine linear convolution

        # ----- smooth max (log‑sum‑exp) -----
        scaled = lam * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)       # stable exponentiation
        w /= w.sum()                          # w_k = soft‑max weights, Σw = 1

        # ----- gradient w.r.t. a :   g = 2·corr(w, a)  (valid mode) -----
        # correlation via FFT: corr = np.correlate(w, a, mode='valid')
        fft_w = np.fft.rfft(w, n=L_fft)
        fft_rev_a = np.fft.rfft(a[::-1], n=L_fft)
        corr_full = np.fft.irfft(fft_w * fft_rev_a, n=L_fft)
        start = n - 1
        grad_a = 2.0 * corr_full[start : start + n]   # shape (n,)

        # ----- gradient w.r.t. logits (chain rule) -----
        # dF/dz_i = a_i * (grad_a_i - <grad_a, a>)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- record true (non‑smoothed) score -----
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_it = it
            print(f"Iter {it:7d}: new best {best_score:.6f}")

        # ----- increase smoothing temperature -----
        if lam < lam_max:
            lam = min(lam_max, lam * lam_factor)

        # ----- occasional random restart if stuck for long -----
        if it % 50000 == 0:
            if it - last_improve_it > 20000:
                # re‑initialise logits around the current best to inject diversity
                print(f"Iter {it}: restarting (no improvement for {it-last_improve_it} iters)")
                perturb = np.random.randn(n) * 0.1
                logits = np.log(best_a + 1e-16) + perturb
                m.fill(0.0)
                v.fill(0.0)
                last_improve_it = it

        # ----- status output -----
        if it % 10000 == 0:
            elapsed = time.time() - start_time
            print(f"Iter {it:7d} | score {cur_score:.6f} | λ={lam:.1f} | elapsed {elapsed:.1f}s")

    # ------------------------------------------------------------------
    # 3 – final output
    # ------------------------------------------------------------------
    best_a /= best_a.sum()          # final normalisation (should already be 1)
    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
