"""
Optimisation of the step‑height sequence for ``evaluate_sequence``.

Strategy
--------
1. **Initialisation** – use the previous global ``height_sequence_1`` (if
   available) normalised to a probability vector, otherwise start from a
   uniform distribution.

2. **Exploratory sampling** – generate a number of random Dirichlet draws
   and a dense grid of Beta‑distribution shapes.  Keep the best of those
   candidates (they are cheap to evaluate).

3. **Mirror‑descent (exponentiated‑gradient)** – starting from the best
   candidate, perform many KL‑mirror‑descent steps that minimise the
   ∞‑norm of the self‑convolution while staying on the simplex.

4. **Guided simulated annealing** – use the current best point as a seed
   for a simulated‑annealing phase.  Moves transfer a random amount of mass
   from an index that contributes strongly to the maximal convolution entry
   (large sub‑gradient) to an index that contributes weakly (small
   sub‑gradient).  The temperature decays geometrically; worsening moves
   are accepted with Metropolis probability.

5. **Return** – the best normalised sequence found during the whole run.
   The evaluation function is invariant to scaling, therefore the returned
   vector may be any non‑negative vector with sum ≈ 1.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines – all are top‑level as required
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector whose sum is 1.
    Returns ``2·n·max(conv)`` where ``conv = a * a`` (linear convolution).
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on ``n`` points, normalised to sum = 1.
    """
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow for extreme α,β
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    w /= w.sum()
    return w


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv)`` at ``a``.
    The gradient is the average of the exact gradients of all active
    convolution entries (those attaining the current maximum up to a tiny
    tolerance).
    """
    n = a.shape[0]
    max_b = conv.max()
    eps = max_b * 1e-9
    active = np.where(conv >= max_b - eps)[0]

    if active.size == 0:
        # safety fallback – gradient of the quadratic 2·a·aᵀ
        return 2.0 * a

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        # ∂conv_k/∂a_i = 2·a_{k‑i}
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """
    Mirror‑descent update on the probability simplex using the
    exponentiated‑gradient (multiplicative) rule.
    """
    factor = np.exp(-step * grad)
    a_new = a * factor
    a_new = np.clip(a_new, 1e-16, None)       # keep strictly positive
    a_new /= a_new.sum()
    return a_new

        # evaluate candidate
        fft_c = np.fft.rfft(cand, n=2 * n - 1)
        conv_c = np.fft.irfft(fft_c * fft_c, n=2 * n - 1)
        cand_score = 2.0 * n * conv_c.max()

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-12))

        if accept:
            cur_a, cur_score, conv_cur = cand, cand_score, conv_c
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
                print(f"SA iter {it_sa:07d}: new best {best_score:.6f}")

        # occasional full‑restart to inject diversity
        if it_sa % 60000 == 0 and it_sa > 0:
            cur_a = np.random.dirichlet(np.ones(n))
            fft_cur = np.fft.rfft(cur_a, n=2 * n - 1)
            conv_cur = np.fft.irfft(fft_cur * fft_cur, n=2 * n - 1)
            cur_score = 2.0 * n * conv_cur.max()
            print(f"SA iter {it_sa}: random restart performed")

        # periodic status report
        if it_sa % 20000 == 0 and it_sa > 0:
            print(f"SA iter {it_sa:07d}: best so far {best_score:.6f}")

        it_sa += 1

    print(f"Search finished – final best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence (still normalised)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
