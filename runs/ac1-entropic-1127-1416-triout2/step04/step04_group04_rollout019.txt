def refine_adam(a, deadline, max_iter, eta0=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    a = normalize(a.copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline: break
        conv = convolution_fft(a,a)
        max_val = conv.max()
        # binary indicator for max entries:
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w = np.zeros(len(conv))
        w[max_idx] = 1.0 / len(max_idx)
        grad = gradient_from_weights(a, w)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)  # multiplicative update
        a = normalize(a)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

"""
propose_candidate.py

A refined heuristic for the step‑function minimisation problem.
The algorithm builds many low‑dimensional families (arcsine, power‑law,
beta, exponential decay, random Dirichlet), selects the most promising
candidates and then improves them with a long exponentiated‑gradient
(Ada‑type) sub‑gradient optimisation on the true nonsmooth objective.
A final “polish’’ run on the global best further reduces the score.
All heavy work is done with FFT‑based convolutions and the routine
respect the supplied time budget (default 1000 s) and always return a
feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast (exact) evaluation of the official objective."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)                # length = 3·n‑2
    idx = 2 * len(a) - 2                             # position of the “center’’ term
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Families for the initial sweep
# ----------------------------------------------------------------------


def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def expo_decay_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-lam·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(i - centre))
    return normalize(vals)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (Adam‑style exponentiated gradient)
# ----------------------------------------------------------------------


def refine_adam(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 50_000,
    eta0: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, float]:
    """
    Projected multiplicative Adam on the true objective.
    Returns the best vector seen and its (exact) score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution and sub‑gradient weight vector
        conv = convolution_fft(a, a)
        max_val = conv.max()
        w = np.zeros(conv_len, dtype=float)
        # distribute weight equally over all maximising indices
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # exponentiated‑gradient step
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Candidate generation and selection
# ----------------------------------------------------------------------


def generate_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Return a modest library of initial vectors for length n."""
    cand: List[np.ndarray] = []

    cand.append(uniform_shape(n))
    cand.append(arcsine_initial(n))

    # power‑law family
    for _ in range(6):
        p = rng.uniform(0.2, 0.8)
        eps = rng.uniform(0.0, 5.0)
        cand.append(powerlaw_shape(n, p, eps))

    # symmetric beta‑like
    for _ in range(5):
        a = rng.uniform(0.2, 5.0)
        cand.append(beta_shape(n, a))

    # exponential decay shapes
    for _ in range(5):
        lam = rng.uniform(0.001, 0.02) * n   # scale with n
        cand.append(expo_decay_shape(n, lam))

    # mixtures of arcsine and uniform (helps smoothing)
    for _ in range(5):
        w = rng.uniform(0.0, 0.5)
        mix = (1.0 - w) * arcsine_initial(n) + w * uniform_shape(n)
        cand.append(normalize(mix))

    # random Dirichlet draws
    for _ in range(5):
        cand.append(random_dirichlet(n, rng))

    return cand


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    score.  The routine respects the time budget, explores several parametric
    families, refines promising candidates with a long Adam‑style
    sub‑gradient descent and finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin before the hard timeout
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – start small, then move to larger n if time permits.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n>>> Exploring length n = {n}")

        # --------------------------------------------------------------
        # 1.  Build a library of initial candidates.
        # --------------------------------------------------------------
        candidates = generate_candidates(n, rng)

        # --------------------------------------------------------------
        # 2.  Quick ranking – keep only the best few according to the fast
        #     (FFT‑based) internal score.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 8
        top_candidates = [c for _, c in scored[:top_k]]
        print(
            "  top initial scores:",
            " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]),
        )

        # --------------------------------------------------------------
        # 3.  Refine each retained candidate with the Adam‑style optimiser.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            init_sc = compute_score(cand)
            print(
                f"  candidate {idx + 1}/{len(top_candidates)} start_score = {init_sc:.6f}"
            )

            remaining = max(0.0, deadline - time.time())
            # roughly 20 iterations per second for n≈2000 (convolution dominated)
            max_iter = int(min(120_000, max(5_000, remaining * 20)))

            refined, _ = refine_adam(
                a=cand,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.03,  # a bit smaller than before – more stable
            )

            # official evaluation
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 4.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(300_000, max(20_000, remaining * 20)))

        polished, _ = refine_adam(
            a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.02,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
