    best_sc = compute_score(a)

    gamma = gamma_start

    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break

        # compute convolution a*a
        conv = convolution_fft(a, a)
        # compute softmax weights
        # Use stability: shift by max.
        max_c = conv.max()
        # compute exponent: (conv - max_c) * gamma
        exp_vals = np.exp(gamma * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # compute gradient
        grad = gradient_from_weights(a, w)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a + step
        # clip to non-negative
        a = np.maximum(a, 0.0)
        # renormalize to sum 1
        a = normalize(a)

        # evaluate score
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # increase gamma for sharper focus gradually
        gamma *= gamma_rate

        # optionally print

    return best_a, best_sc

"""
Algorithm
---------

We search for a non‑negative step‑function *a* (∑a = 1) that minimises

        2·n·max(conv(a,a)) / (∑a)²   =   2·n·max(conv(a,a))

because the ratio is scale‑invariant.  The quantity to be minimised is the
maximum entry of the linear convolution of *a* with its reversal.

The search proceeds in three phases:

1. **Parametric initialisation** – a handful of low‑dimensional families
   (arcsine, power‑law, symmetric Beta, uniform, random Dirichlet) supply
   promising starting points for a chosen length *n* (2048 is the main
   work‑horse).

2. **Adam + soft‑max sub‑gradient optimisation** –  the non‑smooth
   ``max`` is replaced by a smooth “soft‑max’’ with an increasing
   temperature.  The gradient of the soft‑max is obtained analytically via
   one extra FFT.  An Adam optimiser (multiplicative / momentum update)
   keeps the iterate on the probability simplex.

3. **Upscaling & polishing** – the best vector for *n = 2048* is duplicated
   (each entry split in two) to obtain a feasible vector for *n = 4096*.
   A short Adam run further reduces the objective.  The final candidate
   is returned.

The whole routine respects the supplied time budget (default 1000 s) and
always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast (scale‑invariant) evaluation of the official objective:
        2·n·max(conv(a,a)) / (∑a)² .
    The vector is assumed non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (∑ = 1)."""
    total = float(v.sum())
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k   w.r.t. a.
    Implements   g_j = 2· Σ_m a_m·w_{m+j} .
    Computed via a single FFT‑based convolution.
    """
    # w is a 1‑D array of length (2·len(a)‑1)
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)          # length = 3·n‑2
    idx = 2 * len(a) - 2                      # points to position 2·n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution   a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law   a_i ∝ ((i+ε)(n‑1‑i+ε))^{-p}."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape   a_i ∝ (i+½)^{a‑1}·(n‑1‑i+½)^{a‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex by exponential sampling."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Adam + soft‑max optimisation of the max‑convolution
# ----------------------------------------------------------------------


def refine_adam(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    lr: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    gamma_start: float = 400.0,
    gamma_rate: float = 1.015,
    gamma_max: float = 1e6,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of ``max(conv)``.
    The weight vector is a soft‑max of the convolution entries;
    its temperature (inverse gamma) is increased gradually, driving the
    objective towards the true max.

    Returns the best vector seen and its (exact) score computed by
    ``compute_score``.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    m = np.zeros_like(a)    # first moment
    v = np.zeros_like(a)    # second moment
    gamma = gamma_start

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # --- convolution of a with itself ---------------------------------
        conv = convolution_fft(a, a)               # length = 2·n‑1

        # --- soft‑max weights --------------------------------------------
        # stability: shift by the maximum
        max_c = conv.max()
        exp_vals = np.exp(gamma * (conv - max_c))
        w = exp_vals / exp_vals.sum()               # shape (2·n‑1,)

        # --- gradient ----------------------------------------------------
        grad = gradient_from_weights(a, w)

        # --- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a + step
        a = np.maximum(a, 0.0)          # enforce non‑negativity
        a = normalize(a)                # projection onto simplex

        # --- bookkeeping --------------------------------------------------
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    Adam iter={it:,d} score={cur_sc:.6f} best={best_sc:.6f}")

        # increase sharpness of the soft‑max
        gamma = min(gamma * gamma_rate, gamma_max)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper for up‑scaling a vector (duplicate each entry)
# ----------------------------------------------------------------------


def upscale_by_factor(seq: np.ndarray, factor: int = 2) -> np.ndarray:
    """
    Produce a new vector of length len(seq)·factor by repeating each entry
    ``factor`` times and scaling each copy by 1/factor, thereby preserving
    the total sum (=1).
    """
    if factor <= 0:
        raise ValueError("factor must be positive")
    repeated = np.repeat(seq, factor)
    return repeated / factor


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    evaluation function.  The routine respects the time budget and always
    returns a feasible list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin to avoid a hard timeout
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Primary optimisation on a medium size (n = 2048)
    # ------------------------------------------------------------------
    n_target = 2048
    if time.time() < deadline - 5:
        print(f"\n>>> Optimising length n = {n_target}")

        # ---- 1) build a modest library of initial candidates ------------
        candidates: List[np.ndarray] = []

        candidates.append(uniform_shape(n_target))
        candidates.append(arcsine_initial(n_target))

        for _ in range(6):
            p = rng.uniform(0.35, 0.70)           # around the arcsine exponent
            eps = rng.uniform(0.0, max(5.0, 0.015 * n_target))
            candidates.append(powerlaw_shape(n_target, p, eps))

        for _ in range(4):
            a = rng.uniform(0.1, 5.0)
            candidates.append(beta_shape(n_target, a))

        for _ in range(3):
            candidates.append(random_dirichlet(n_target, rng))

        # ---- 2) quick ranking (fast internal score) --------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 4
        top_candidates = [c for _, c in scored[:top_k]]
        print(
            "  top initial scores:",
            " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]),
        )

        # ---- 3) Adam refinement of the retained candidates -------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5:
                break
            start_sc = compute_score(cand)
            print(
                f"  candidate {idx + 1}/{len(top_candidates)} start_score = {start_sc:.6f}"
            )
            # allocate iterations based on remaining time
            remaining = max(0.0, deadline - time.time())
            # empirical iterations per second for n≈2048 is ~15
            it_per_sec = 15.0
            max_iter = int(min(30_000, max(2_000, remaining * it_per_sec)))
            refined, _ = refine_adam(
                a=cand,
                deadline=deadline,
                max_iter=max_iter,
                lr=0.07,
                gamma_start=400.0,
                gamma_rate=1.015,
                report_every=0,
            )
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n_target}) ***")

    # ------------------------------------------------------------------
    # Upscaling to a larger length (n = 4096) and final polishing
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5:
        a_best = np.array(best_seq, dtype=float)
        n_up = len(a_best) * 2
        print(f"\n>>> Upscaling to n = {n_up}")
        a_up = upscale_by_factor(a_best, factor=2)

        # a short Adam run on the up‑scaled vector
        remaining = max(0.0, deadline - time.time())
        it_per_sec = 8.0  # slower for the larger problem
        max_iter = int(min(20_000, max(5_000, remaining * it_per_sec)))
        refined_up, _ = refine_adam(
            a=a_up,
            deadline=deadline,
            max_iter=max_iter,
            lr=0.05,
            gamma_start=800.0,
            gamma_rate=1.02,
            report_every=0,
        )
        val_up = evaluate_sequence(refined_up.tolist())
        print(f"    up‑scaled evaluated score = {val_up:.6f}")
        if val_up < best_val:
            best_val = val_up
            best_seq = refined_up.tolist()
            print("*** Updated global best after up‑scaling ***")

    # ------------------------------------------------------------------
    # Optional final polishing (short Adam run on the current best)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        it_per_sec = 12.0
        max_iter = int(min(60_000, max(8_000, remaining * it_per_sec)))
        polished, _ = refine_adam(
            a=a,
            deadline=deadline,
            max_iter=max_iter,
            lr=0.04,
            gamma_start=600.0,
            gamma_rate=1.018,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
