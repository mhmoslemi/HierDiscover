"""
Hybrid parametric + smoothed‑max Adam descent for the step‑function problem.

The algorithm proceeds in three stages:

1.  **Broad initialization** – for each chosen length `n` we build a
    modest library of plausible shapes (arcsine, symmetric power‑law,
    Beta‑like, mixtures with the uniform distribution and random Dirichlet draws).
    The shapes are normalised to the probability simplex (sum = 1).

2.  **Refinement** – every top candidate is improved with a gradient‑based
    optimiser that works on the *smoothed* maximum of the self‑convolution.
    The smoothed max is realised by a soft‑max (log‑sum‑exp) with a temperature
    parameter `α`.  `α` is increased linearly from a small value (very smooth)
    to a large one (close to the true max).  The gradient of the smoothed
    objective is obtained analytically; we then apply an Adam‑style
    exponentiated‑gradient update (multiplicative update followed by a
    projection onto the simplex).  This gives a much better descent direction
    than the hard sub‑gradient used previously.

3.  **Final polish** – the globally best vector is refined again with a
    longer run of the same optimiser, using a larger `α_max` to converge
    to the true objective.

All operations are FFT‑based (linear convolution) and respect the
time‑budget (default 1000 s).  The routine always returns a feasible
non‑negative list of floats; if nothing is found the uniform vector is used.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT, zero‑padded to the next power of two.
    Returns the full convolution of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the scale‑invariant objective
        2·n·max(conv(a,a)) / (sum(a)²)
    a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j} .
    Computed via one FFT‑based convolution.
    """
    # reverse w for the convolution that yields the needed shift pattern
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)                # length = 3·n‑2
    idx = 2 * len(a) - 2                             # centre index (2n‑2)
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def smoothed_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max (log‑sum‑exp) weights for the smoothed max.
    Returns a probability vector `w` of the same length as `conv`.
    """
    # stability: subtract the maximum before exponentiation
    shift = np.max(conv)
    exp_term = np.exp(alpha * (conv - shift))
    total = np.sum(exp_term)
    if total == 0.0:
        # fallback – uniform weights (should never happen for finite alpha)
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / total


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution ≈ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with equal parameters `a`."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def edge_heavy_shape(n: int, eps: float = 0.0) -> np.ndarray:
    """
    Very heavy at both ends: w_i ∝ 1/(i+1) + 1/(n-i).
    Optional epsilon smooths the singularities.
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / (i + 1 + eps) + 1.0 / (n - i + eps)
    return normalize(vals)


def generate_initial_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """
    Build a modest library of candidate shapes for a given length `n`.
    """
    cand: List[np.ndarray] = []

    # deterministic families
    cand.append(uniform_shape(n))
    cand.append(arcsine_initial(n))
    cand.append(edge_heavy_shape(n, eps=0.0))

    # power‑law grid
    for p in np.linspace(0.30, 0.90, 7):
        for eps in [0.0, 0.001 * n, 0.01 * n]:
            cand.append(powerlaw_shape(n, p, eps))

    # symmetric beta grid
    for a in np.linspace(0.2, 5.0, 7):
        cand.append(beta_shape(n, a))

    # mixtures arcsine + uniform
    arcs = arcsine_initial(n)
    uniform = uniform_shape(n)
    for _ in range(5):
        w = rng.uniform(0.0, 0.5)          # weight of the uniform part
        mix = (1.0 - w) * arcs + w * uniform
        cand.append(normalize(mix))

    # random draws
    for _ in range(6):
        cand.append(random_dirichlet(n, rng))

    return cand


# ----------------------------------------------------------------------
# Smoothed‑max Adam (exponentiated‑gradient) refinement
# ----------------------------------------------------------------------


def refine_adam(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.5,
    alpha0: float = 0.5,
    alpha_max: float = 200.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the *smoothed* max
    of the self‑convolution.
    The smoothing parameter α is linearly increased from `alpha0` to
    `alpha_max` over the course of the run, providing a continuation scheme.
    Returns the best vector seen (as a NumPy array) and its fast‑score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current smoothing coefficient
        alpha = alpha0 + (alpha_max - alpha0) * it / max_iter

        # convolution a*a
        conv = convolution_fft(a, a)

        # soft‑max weights for the smoothed max
        w = smoothed_weights(conv, alpha)

        # gradient of the smoothed objective w.r.t. a
        grad = gradient_from_weights(a, w)

        # Adam moment updates (on the *gradient* before the exponentiated step)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # multiplicative (exponentiated) update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)     # element‑wise
        a = normalize(a)

        # fast internal score (no clipping, no INF checks)
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    Adam iter={it:,d} α={alpha:.2f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
