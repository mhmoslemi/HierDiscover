"""
Hybrid search for a low‑max‑autocorrelation step function.

The routine explores several sequence lengths and many families of
initial shapes (uniform, arcsine, beta‑like, power‑law, sinusoidal,
Gaussian‑like, Dirichlet draws).  For each promising candidate we first
run a *smooth* projected gradient descent on

    Fβ(a) = 2·n·β·log‑sum‑exp( (a∗a)/β ),

which is a differentiable approximation of the true objective
`2·n·max(a∗a)`.  The gradient can be computed in O(n log n) via FFT.
An adaptive step‑size (ηₜ = η₀ / √t) and an exponential decay of β
gradually sharpen the approximation.

After the smoothed phase we switch to a short *sub‑gradient* (exponentiated
mirror‑descent) optimisation that directly minimises the true max‑convolution.
The best result over all lengths is finally polished once more with the
sub‑gradient method.

All vectors are kept on the probability simplex (non‑negative, sum = 1);
the official evaluator is scale‑invariant, so we can return the simplex
vector directly.

The algorithm respects the supplied time budget and always returns a
feasible list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT.
    Returns a length‑len(a)+len(b)‑1 array.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the official evaluator (scale‑invariant).
    a is assumed non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, sum x = 1}.
    Implements the O(n log n) algorithm of [Duchi et al. 2008].
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (unnormalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def beta_like_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    alpha = rng.uniform(0.2, 5.0)
    beta = rng.uniform(0.2, 5.0)
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def powerlaw_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))^{‑p}."""
    p = rng.uniform(0.1, 1.0)
    eps = rng.uniform(0.001, max(1.0, 0.02 * n))
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def sinusoid_initial(n: int) -> np.ndarray:
    """Half‑period sine shape, positive on (0,π)."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    sigma = rng.uniform(0.08, 0.25) * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals += 1e-12
    return vals / vals.sum()


def dirichlet_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample from a Dirichlet‑like distribution."""
    conc = rng.choice([0.3, 0.5, 1.0, 2.0])
    if conc <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a small random perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0:
        return np.full_like(a, 1.0 / len(a))
    return b / s

# Smoothed projected gradient descent (β‑log‑sum‑exp)
# ----------------------------------------------------------------------


def optimise_smooth(
    initial_a: np.ndarray,
    deadline: float,
    *,
    beta_start: float = 0.05,
    beta_decay: float = 0.98,
    min_beta: float = 5e-4,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    line_search_steps: int = 6,
    verbose: bool = False,
) -> Tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the objective using projected gradient descent.
    Returns the best vector (on the simplex) seen and its exact (non‑smooth) score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    beta = beta_start

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # full convolution a∗a
        conv = convolution_fft(a, a)                     # length 2n‑1

        # soft‑max weights (stable)
        max_c = conv.max()
        exp_vals = np.exp((conv - max_c) / beta)
        w = exp_vals / exp_vals.sum()                    # length 2n‑1, sum = 1

        # gradient = 2 * (w_rev ∗ a) restricted to n entries
        grad_full = convolution_fft(w[::-1], a)          # length 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]          # size n

        # step‑size schedule
        eta = eta0 / np.sqrt(it)

        # simple backtracking line‑search
        step = eta
        improved = False
        for _ in range(line_search_steps):
            a_new = a - step * grad
            a_new = project_onto_simplex(a_new)
            cur_sc = compute_score(a_new)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            # accept the last trial to keep moving (may be neutral)
            a = a_new

        # gradual cooling of β
        if it % 2000 == 0:
            beta = max(beta * beta_decay, min_beta)

        if verbose and it % 5000 == 0:
            print(
                f"    smooth iter={it:,d} β={beta:.6f} cur_score={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (mirror‑descent)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    If several offsets attain the maximum, the gradients are averaged.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
    verbose: bool = False,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient / mirror descent on the exact max‑convolution.
    Works on the probability simplex.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a = project_onto_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if verbose and it % 2000 == 0:
            print(
                f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
