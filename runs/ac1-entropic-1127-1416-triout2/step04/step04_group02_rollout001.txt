    # returns array length len(w) + len(a) -1; we will slice first len(a) for results.
    L = len(w)

    W = np.fft.rfft(w, n=N)

    # Since for i up to n-1, i+j < L always holds because L=2n-1, and max i+j = (n-1)+(n-1)=2n-2 = L-1. Good.
    return corr[:n]

"""
SMOOTHED‑MAX GRADIENT DESCENT (Adam) FOR MINIMISING THE PEAK OF THE AUTO‑CONVOLUTION.

The original problem is scale invariant, so we fix  Σ a_i = 1  and minimise
    Φ(a) = 2·n·max_k (a∗a)_k .
Instead of the non‑smooth max we minimise a smooth approximation

    f_λ(a) = (1/λ)·log Σ_k exp( λ·(a∗a)_k )      (λ≫1 ⇒ f_λ≈max)

The gradient of f_λ is cheap to obtain:
    w_k = softmax_λ( (a∗a)_k )
    ∇_a f_λ = 2·corr(w, a)                     (linear cross‑correlation)

We represent a via a soft‑max of free variables u (a_i = e^{u_i}/∑e^{u}),
so the constraints a_i≥0 and Σa_i=1 are always satisfied.
Adam (β₁=0.9, β₂=0.999) updates u using the gradient above.
λ is raised gradually (5 → 20 → 50 → 100 → 200) to sharpen the objective.

The search loops over a few sequence lengths (2048, 4096, 8192), builds a
small library of diverse initial shapes (arcsine, beta‑U‑shapes,
Dirichlet draws) and runs the optimiser for each.  The best result
according to the official ``evaluate_sequence`` is kept and, if time
permits, polished with a longer run at the highest λ.

All operations are O(n log n) thanks to FFTs, and the routine respects
the 1000 s wall‑clock budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full) via real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_linear(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a) restricted to the first |a| entries:
        (w ⋆ a)_i = Σ_j w_{i+j}·a_j   for i = 0 … len(a)-1
    (|w| = 2·len(a)-1, so the needed indices are always valid.)
    """
    L = len(w)
    n = len(a)
    N = next_pow2(L + n - 1)
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a, n=N)
    C = np.fft.irfft(W * np.conj(A), n=N)[: L + n - 1]
    return C[:n]   # shape = (n,)


# ----------------------------------------------------------------------
# Fast (scale‑invariant) internal score – useful for intermediate ranking
# ----------------------------------------------------------------------


def internal_score(a: np.ndarray) -> float:
    """2·n·max(a∗a)/(sum a)² – the same quantity as ``evaluate_sequence`` but
    without the clipping and without the 0.01 guard."""
    n = len(a)
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * mx / (s * s)


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------


def arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(0.5,0.5)) – U‑shaped."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def dirichlet_shape(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Simple Dirichlet‑like draw (Gamma with shape=conc)."""
    if conc <= 0:
        # heavy‑tailed sparse draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()


# ----------------------------------------------------------------------
# Smoothed‑max optimiser (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def _lambda_schedule(it: int) -> float:
    """Piecewise linear schedule for the smoothing parameter λ."""
    if it < 20_000:
        return 5.0
    if it < 60_000:
        return 20.0
    if it < 120_000:
        return 50.0
    if it < 180_000:
        return 100.0
    return 200.0


def smoothed_max_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    lr: float = 0.05,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max (see module docstring).

    Parameters
    ----------
    init_a : np.ndarray
        Initial non‑negative vector (will be normalised to sum = 1).
    deadline : float
        Absolute time (time.time()) after which the function returns.
    max_iter : int, optional
        Upper bound on iterations.
    lr : float, optional
        Base learning rate for Adam.

    Returns
    -------
    best_a : np.ndarray
        The best vector (still normalised to sum = 1) seen during optimisation.
    best_score : float
        Internal score (2·n·max(conv)/sum²) of ``best_a``.
    """
    n = len(init_a)
    # initialise the logits u so that softmax(u) = init_a
    eps = 1e-30
    u = np.log(np.maximum(init_a, eps))
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_a = None
    best_sc = float("inf")

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # --- obtain a from logits (soft‑max) ---
        u_shift = u - np.max(u)          # for numerical stability
        exp_u = np.exp(u_shift)
        a = exp_u / exp_u.sum()          # guaranteed sum‑to‑one, a_i≥0

        # --- convolution of a with itself ---
        conv = convolution_fft(a, a)     # length 2n‑1

        # --- smooth max -------------------------------------------------
        lam = _lambda_schedule(it)
        scaled = lam * conv
        max_scaled = scaled.max()
        # stable soft‑max
        exp_scaled = np.exp(scaled - max_scaled)
        Z = exp_scaled.sum()
        w = exp_scaled / Z                # w_k = softmax_λ(conv_k)

        # smoothed objective (useful for diagnostics)
        smooth_obj = (max_scaled + np.log(Z)) / lam

        # --- gradient w.r.t a (∇_a f_λ) ----------------------------------
        grad_a = 2.0 * correlation_linear(w, a)   # shape (n,)

        # --- gradient w.r.t logits u (chain rule through soft‑max) ------
        dot = np.dot(a, grad_a)                    # scalar = a·∇_a
        grad_u = a * (grad_a - dot)                # shape (n,)

        # --- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- keep best seen by the true internal score ------------------
        cur_sc = internal_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report
        if it % 10_000 == 0:
            print(
                f"      iter={it:,d} λ={lam:.0f} smooth={smooth_obj:.6f} "
                f"score≈{cur_sc:.6f}"
            )

    # if the loop never updated best_a (unlikely) fall back to final a
    if best_a is None:
        best_a = a
        best_sc = internal_score(a)

    return best_a, best_sc

            print(f"    official score = {val:.6f}")
