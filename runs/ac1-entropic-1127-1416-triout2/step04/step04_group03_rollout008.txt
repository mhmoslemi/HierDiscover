    for it in range(max_iters):
        if time.time() >= time_deadline: break
        # compute M
        conv_self = convolution_fft(a, a)
        M = float(np.max(conv_self))
        # LP to find g0 maximizing sum(g) s.t. conv(a,g) <= M
        # Build A_ub
        # Build rows, cols, data as above
        rows = np.empty(n*n, dtype=np.int32)
        cols = np.empty(n*n, dtype=np.int32)
        data = np.empty(n*n, dtype=np.float64)
        pos = 0
        arange_n = np.arange(n, dtype=np.int32)
        for i in range(n):
            rows[pos:pos+n] = i + arange_n
            cols[pos:pos+n] = arange_n
            data[pos:pos+n] = a[i]
            pos += n
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M, dtype=float)
        # Solve LP
        res = scipy.optimize.linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                                   bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"LP failed at iter {it}: {res.message}")
            break
        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 == 0:
            # No improvement possible
            break
        # Scale g to sum=1
        g = g0 / sum_g0
        # Combine with a
        a_new = (1 - t) * a + t * g
        # Compute new M
        conv_self_new = convolution_fft(a_new, a_new)
        M_new = float(np.max(conv_self_new))
        # Accept if improved
        if M_new < M - 1e-12:  # improvement
            a = a_new
            # Possibly adapt t based on improvement magnitude
        else:
            # possibly reduce t and try again
            t = t * 0.5
            if t < 1e-3:
                break
            continue
    return a

def lp_refine(a, time_deadline, max_iters=30, t_init=0.5):
    n = len(a)
    a = a.copy()
    # ensure sum=1
    a = a / np.sum(a)
    t = t_init
    for it in range(max_iters):
        if time.time() >= time_deadline:
            break
        conv_self = convolution_fft(a, a)
        M = float(np.max(conv_self))
        # Build constraint matrix
        # using sparse
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)
        pos = 0
        arange_n = np.arange(n, dtype=np.int32)
        for i in range(n):
            rows[pos:pos+n] = i + arange_n
            cols[pos:pos+n] = arange_n
            data[pos:pos+n] = a[i]
            pos += n
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n - 1, M, dtype=float)
        # solve LP max sum(g)
        c = -np.ones(n)  # minimize -sum => maximize sum
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                    bounds=bounds, method='highs',
                                    options={'presolve': True})
        if not res.success:
            print(f"LP failed at it {it}: {res.message}")
            break
        g0 = res.x
        sum_g0 = np.sum(g0)
        # If sum_g0 is zero break
        if sum_g0 <= 0:
            break
        # Scale to sum=1
        g = g0 / sum_g0
        # Combine
        a_new = (1 - t) * a + t * g
        # Evaluate new max
        conv_new = convolution_fft(a_new, a_new)
        M_new = float(np.max(conv_new))
        # Accept if reduction, else reduce t
        if M_new < M - 1e-12:
            a = a_new
            # maybe increase t for next iteration?
            t = min(0.9, t * 1.1)
        else:
            t *= 0.5
            if t < 1e-4:
                break
    return a

"""
Improved search for a non‑negative step‑function that minimizes

    2·n·max(conv(a,a)) / ( Σ a )²

The algorithm mixes three ideas:

1. **Exploration** – a few analytic starters (uniform, arcsine) and a
   random Beta‑family sampling are used to get a decent initial shape.

2. **LP refinement** – following the idea from the paper, we repeatedly
   solve the linear programme

        max   Σ g_j
        s.t.  a * g ≤ M    (convolution of the current a with g)
               g ≥ 0

   where `M = max(conv(a,a))`.  The LP gives a direction that provably
   reduces the maximal convolution value; we take a convex combination
   `a ← (1‑t) a + t·g` and repeat.  The LP is built with a sparse Toeplitz
   matrix and solved by ``scipy.optimize.linprog`` (HiGHS).

3. **Local polishing** – after the LP phase we run the three‑stage
   optimiser (Adam on a smooth soft‑max surrogate, sub‑gradient descent
   on the true max, and a stochastic Gaussian walk) exactly as in the
   previous best solution.

The whole search respects the 1000 s budget, tries several vector
lengths (256 … 4096) and keeps the best result according to the official
``evaluate_sequence`` function.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT convolution, simplex projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (output length len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)) for a already normalised (Σa = 1).
    This equals the official evaluation value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic starters and random beta‑family generator
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic (symmetrised) arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric Beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # softmax weights, sum to 1
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of max

    # weighted sum of sub‑gradients:
    # ∂ (Σ_k w_k (a*a)_k) / ∂a = 2 * (a * w_rev) evaluated at appropriate indices
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    noise_every: int = 1500,
    noise_scale: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(len(a)) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on   max(conv(a,a)).
    The step size decays like step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        idx = int(np.argmax(conv))

        i = np.arange(len(a))
        j = idx - i
        mask = (j >= 0) & (j < len(a))
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc

def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex: add Gaussian noise, project back,
    keep improvements (tiny probability of uphill moves for escape).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=len(a))
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# LP refinement (core of the new algorithm)
# ----------------------------------------------------------------------


def lp_refine(
    a0: np.ndarray,
    time_deadline: float,
    max_iters: int = 15,
    t_init: float = 0.5,
) -> np.ndarray:
    """
    Repeatedly solve the linear programme

        max Σ g_j   s.t.   a * g ≤ M,   g ≥ 0

    where M = max(conv(a,a)).  The LP gives a direction that reduces the
    maximal self‑convolution.  The current vector is updated by a convex
    combination a ← (1‑t) a + t·g (with an adaptive step size t).
    The routine works with the probability simplex (Σ a = 1).
    """
    import scipy.sparse

    a = a0.copy()
    a = a / np.sum(a)          # guarantee Σa = 1
    t = t_init
    n = len(a)

    # Pre‑allocate index arrays once (they are reused with different data)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    arange_n = np.arange(n, dtype=np.int32)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = i + arange_n
        cols[pos:pos + n] = arange_n
        pos += n
    row_idx = rows
    col_idx = cols
    shape = (2 * n - 1, n)

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break

        # current maximal convolution value
        conv_self = convolution_fft(a, a)
        M = float(np.max(conv_self))

        # Build the sparse constraint matrix A_ub where
        #   (A_ub @ g)_k = Σ_i a_i g_{k-i}
        data = np.repeat(a, n)          # a_i repeated n times, order i‑major
        A_ub = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=shape)

        b_ub = np.full(2 * n - 1, M, dtype=float)

        # Linear programme: maximise Σ g_j   (i.e. minimise -Σ g_j)
        c = -np.ones(n, dtype=float)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # If the LP fails we abort the refinement for this length.
            print(f"    LP failed at iter {it}: {res.message}")
            break

        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 <= 0:
            break

        # Normalise g to the simplex
        g = g0 / sum_g0

        # Convex combination
        a_new = (1.0 - t) * a + t * g

        # Check improvement
        conv_new = convolution_fft(a_new, a_new)
        M_new = float(np.max(conv_new))

        if M_new < M - 1e-14:                # genuine decrease
            a = a_new
            # slowly increase step size to accelerate convergence
            t = min(0.9, t * 1.1)
        else:
            # back‑off
            t *= 0.5
            if t < 1e-4:
                break

    # final projection (numerical drift)
    return project_to_simplex(a)

# Main entry point required by the competition harness

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine explores several vector lengths, builds an initial
    shape (arcsine / random Beta), refines it by the LP scheme, then
    finishes with the three‑stage Adam → sub‑gradient → stochastic walk
    polishing.  The best vector found within the allotted time is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback (previously discovered good construction)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Working with length n = {n} ===")

        # ----- 1a. Deterministic starters -----
        candidates = [
            np.full(n, 1.0 / n),   # uniform
            arcsine_initial(n),    # arcsine
        ]

        # ----- 1b. Random Beta‑family sampling -----
        for _ in range(200):
            if time.time() >= deadline - 8.0:
                break
            alpha = rng.uniform(0.05, 2.0)
            beta = rng.uniform(0.05, 2.0)
            eps = rng.uniform(1e-6, max(0.01, 0.02 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)
            candidates.append(cand)

        # ----- 1c. Pick the best according to the fast surrogate -----
        local_best_a = None
        local_best_sc = float("inf")
        for cand in candidates:
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand.copy()

        # ----- 2. LP refinement (only for moderate sizes) -----
        if local_best_a is not None and n <= 2048 and time.time() < deadline - 5.0:
            max_lp_iters = 10 if n > 1024 else 15
            print("  LP refinement …")
            local_best_a = lp_refine(local_best_a, deadline,
                                     max_iters=max_lp_iters, t_init=0.5)
            local_best_sc = compute_score(local_best_a)

        # ----- 3. Adam optimisation (smooth surrogate) -----
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(120_000, remaining * 30))
            print(f"  Adam optimisation (max_iter={max_iter_adam}) …")
            local_best_a, _ = adam_optimize(
                initial_a=local_best_a,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.22,
                lam_start=1.0,
                lam_end=80.0,
                noise_every=1200,
                noise_scale=6e-5,
            )

        # ----- 4. Sub‑gradient polishing -----
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(80_000, remaining * 25))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            local_best_a, _ = subgradient_optimize(
                local_best_a,
                time_deadline=deadline,
                step0=0.25,
                max_iter=max_iter_sub,
            )

        # ----- 5. Stochastic local walk -----
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_loc = int(min(120_000, remaining * 30))
            print(f"  Random local walk (max_iter={max_iter_loc}) …")
            local_best_a, _ = random_local_search(
                local_best_a,
                time_deadline=deadline,
                init_step=0.02,
                max_iter=max_iter_loc,
            )

        # ----- 6. Official evaluation -----
        val = evaluate_sequence(local_best_a.tolist())
        print(f"  → official score for n={n} : {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = local_best_a.tolist()
            print(f"  *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 7. Final polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Final polishing of the global best ===")
        a = np.array(best_seq, dtype=float)

        # Stage A – long Adam run
        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 35))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.15,
            lam_start=1.0,
            lam_end=100.0,
            noise_every=1000,
            noise_scale=4e-5,
        )

        # Stage B – sub‑gradient polishing
        if time.time() < deadline - 3.0:
            max_iter_sub = int(min(120_000, (deadline - time.time()) * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.2,
                max_iter=max_iter_sub,
            )

        # Stage C – random local walk
        if time.time() < deadline - 2.0:
            max_iter_loc = int(min(200_000, (deadline - time.time()) * 40))
            print(f"  Random local polishing (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.015,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 8. Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
