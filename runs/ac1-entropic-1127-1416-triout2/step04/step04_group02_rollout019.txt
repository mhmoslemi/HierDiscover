import time, math, numpy as np
from typing import List, Tuple

def next_pow2(x):
    return 1 << (x-1).bit_length()

def convolution_fft(a,b):
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A*B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a):
    n = len(a)
    conv = convolution_fft(a,a)
    max_val = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_val / (s * s)

def _max_conv_gradient(a, conv):
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    if len(active) > 1:
        grad /= len(active)
    return grad

                  max_iter: int = 200_000,
                  eta0: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,

                  temp_decay: float = 0.9995) -> Tuple[np.ndarray, float]:
    """
    Adam on the log‑softmax parametrisation of the probability simplex.
    Uses a smoothed max‑convolution (log‑sum‑exp) with a decaying temperature.
    Returns the best vector encountered.
    """
    n = len(initial_a)
    # initialise log‑probabilities (avoid infinities)
    y = np.log(np.maximum(initial_a, 1e-30))
    m = np.zeros(n)
    v = np.zeros(n)

    best_a = None
    best_sc = float('inf')

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # softmax to obtain a on the simplex
        # improve numerical stability by subtracting max
        y_shift = y - np.max(y)
        a = np.exp(y_shift)
        a /= a.sum()

        # convolution
        conv = convolution_fft(a, a)

        # temperature for smoothing
        temp = max(temp0 * (temp_decay ** t), 1e-12)

        # compute softmax weights over convolution values
        # we work with conv - max(conv) to avoid overflow
        shifted = conv - conv.max()
        w = np.exp(shifted / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective w.r.t a:
        #   g = 2 * correlation(w, a)
        a_rev = a[::-1]
        corr_full = convolution_fft(w, a_rev)
        # slice the part that corresponds to indices 0..n-1
        g_a = 2.0 * corr_full[n - 1 : n - 1 + n]

        # convert gradient to the log‑parameter space (softmax Jacobian)
        dot = np.dot(a, g_a)
        g_y = a * (g_a - dot)

        # Adam update on y
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y -= eta0 * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0 or t == max_iter:
            sc = compute_score(a)   # same as official eval up to scale invariance
            if sc < best_sc:

        a = np.exp(y - np.max(y))

"""
Heuristic optimiser for the max‑convolution objective.

The algorithm builds a small library of diverse initial vectors
(uniform, arcsine, Dirichlet‑type, random power‑law), discards the
worst by a cheap scale‑invariant score, and then refines the survivors
with an Adam optimiser that works on the simplex via a soft‑max
parameterisation.  A temperature‑controlled log‑sum‑exp smoothes the
max‑convolution during the early phase, after which the temperature is
driven to zero so that the optimiser targets the true non‑smooth
objective.  A short post‑hoc mixing with the uniform distribution helps
to flatten residual peaks.  The routine respects the supplied time
budget (default 1000 s) and always returns a feasible list of non‑negative
floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full length, zero‑padded to a power of two).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    The scale‑invariant official score:
        2·n·max(a*a) / (sum(a)²)
    a must be non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (non‑smoothed) max‑convolution.
    For every active offset k we add 2·a_{k‑j} to the corresponding entries.
    If several offsets are active we average the gradients.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametric form)
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 200_000,
    eta0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp0: float = 1e-3,
    temp_decay: float = 0.9995,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the max‑convolution objective.

    Parameters
    ----------
    initial_a : np.ndarray
        Starting probability vector (non‑negative, sum≈1).
    deadline : float
        Unix time when the optimisation must stop.
    max_iter : int
        Upper bound on the number of iterations.
    eta0 : float
        Base learning‑rate for Adam.
    beta1, beta2, eps : float
        Standard Adam hyper‑parameters.
    temp0, temp_decay : float
        Starting temperature and geometric decay factor for the
        log‑sum‑exp smoothing of the max.

    Returns
    -------
    best_a : np.ndarray
        The best vector encountered (still on the simplex).
    best_sc : float
        Its `compute_score`.
    """
    n = len(initial_a)

    # initialise log‑probabilities safely
    y = np.log(np.maximum(initial_a, 1e-30))

    m = np.zeros(n)
    v = np.zeros(n)

    best_a = None
    best_sc = float("inf")

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- soft‑max to obtain a point on the simplex -----------------
        y_shift = y - np.max(y)                # stability
        a = np.exp(y_shift)
        a /= a.sum()

        # ----- convolution a * a -----------------------------------------
        conv = convolution_fft(a, a)          # length 2n‑1

        # ----- temperature for smoothing ---------------------------------
        temp = max(temp0 * (temp_decay ** t), 1e-12)

        # ----- smoothed weights over the convolution values -------------
        shifted = conv - conv.max()            # avoid overflow in exp
        w = np.exp(shifted / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective w.r.t. a ----------------
        # g_a = 2 * correlation(w, a)
        a_rev = a[::-1]
        corr_full = convolution_fft(w, a_rev)          # length 3n‑2
        g_a = 2.0 * corr_full[n - 1 : n - 1 + n]       # slice n entries

        # ----- convert to gradient w.r.t. the log‑parameters (soft‑max Jacobian)
        dot = np.dot(a, g_a)
        g_y = a * (g_a - dot)

        # ----- Adam update on the unconstrained variables y ---------------
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y -= eta0 * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional bookkeeping (cheap score based on the same invariant) -----
        if t % 5000 == 0 or t == max_iter:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"    Adam iter={t:,d} cur={sc:.6f} best={best_sc:.6f}")

    # fall back to the final iterate if nothing was stored
    if best_a is None:
        a = np.exp(y - np.max(y))
        a /= a.sum()
        best_a = a
        best_sc = compute_score(best_a)

    return best_a, best_sc

        # 1b. Quick cheap ranking – keep the most promising few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]   # retain three best

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate with Adam optimisation.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # allocate a slice of the remaining budget
            remaining = max(0.0, deadline - time.time())
            # rough estimate: ~260 it/s for n≈2000 (empirically observed)
            max_it = int(min(200_000, max(5_000, remaining * 260)))

            refined, _ = adam_optimize(
                initial_a=cand,
                deadline=deadline,
                max_iter=max_it,
                eta0=0.6,
                temp0=1e-3,
                temp_decay=0.9995,
            )

            # official evaluation with the supplied evaluator
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Cheap post‑hoc mixing with the uniform distribution.
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        uniform = np.full_like(np.array(best_seq), 1.0 / len(best_seq))
        best_arr = np.array(best_seq, dtype=float)

        for alpha in np.linspace(0.0, 0.10, 11):   # try 0 % … 10 % uniform
            mixed = (1.0 - alpha) * best_arr + alpha * uniform
            mixed /= mixed.sum()
            val = evaluate_sequence(mixed.tolist())
            if val < best_val:
                best_val = val
                best_seq = mixed.tolist()
                print(f"    Mixing alpha={alpha:.2f} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        n = 512
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
