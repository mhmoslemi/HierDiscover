"""
Propose a low‑score step function.

Algorithm outline
-----------------
1.  Randomly initialise a sparse, normalised vector `a` (∑a = 1).
2.  Repeatedly improve it by solving a linear programme:
        maximise   1·b
        subject to  a ∗ b  ≤  M   (M = max(a∗a))
                    b ≥ 0
    The LP gives a vector `b` that keeps the convolution bound `M`
    but has the largest possible ℓ₁‑norm.  After scaling `b` back to
    a probability vector we move a fraction `γ` towards it.
    This “mass‑inflation’’ step equalises the convolution values and
    drives the factor `c = 2·n·max(a∗a)` down.
3.  A short projected sub‑gradient descent (as in the previous
    submission) refines the result.
4.  Finally a stochastic hill‑climber (swap / mass‑transfer moves)
    cleans up any remaining spikes.
All work respects the 1000 s budget; the best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT‑based, very fast)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a∗a) which equals the official score
    because the denominator (∑a)² is 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Linear‑programme that, for a fixed f, finds the densest b with
# a∗b ≤ max(a∗a)
# ----------------------------------------------------------------------


def solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   1·b
        subject to  f ∗ b ≤ M          (element‑wise)
                    b ≥ 0
    The convolution constraints are linear in b, therefore we can build
    a (2·n‑1) × n matrix T where (T·b)[k] = (f∗b)[k].
    Returns the optimal b (not normalised) or None on failure.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz‑like matrix T efficiently.
    # T[k, j] = f[k‑j]  when 0 ≤ k‑j < n, otherwise 0.
    T = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        i_vals = k - np.arange(j_start, j_end + 1)
        T[k, j_start:j_end + 1] = f[i_vals]

    # Linear programme (HiGHS interior) – maximise ∑b  ⇔ minimise –∑b
    c = -np.ones(n)                     # objective
    bounds = [(0.0, None)] * n           # b_j ≥ 0
    b_ub = np.full(rows, M, dtype=float)

    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        return res.x
    # if HiGHS fails fall back to a simple “do‑nothing’’ vector
    return None


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (same as the previous solution)
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Simple projected sub‑gradient descent on the exact objective."""
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)

        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, deadline, max_steps=300_000):
    """Random walk that swaps entries or moves a fraction of mass."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
                    cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
