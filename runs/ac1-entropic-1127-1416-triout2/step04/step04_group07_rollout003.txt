def maxconv_and_subgrad(a, fft_len):
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft*fft, n=fft_len)[:2*a.shape[0]-1]
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi+1)
        grad[idx] += 2.0 * a[k-idx]
    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad

def adam_optimize(init_a, time_limit, rng):
    n = init_a.shape[0]
    fft_len = 1 << ((2*n -2).bit_length())
    b = np.log(init_a + 1e-12)
    m = np.zeros_like(b); v = np.zeros_like(b)
    beta1 = 0.9; beta2 = 0.999; eps=1e-8
    best_a = init_a.copy()
    best_score = score_of(init_a)
    it = 0
    deadline_local = time.time() + time_limit
    stagn=0
    while time.time() < deadline_local:
        it += 1
        # compute a
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a_sum = a.sum()
        a = a / a_sum

        max_val, grad_a = maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn=0
        else:
            stagn+=1

        # early break if stagn may be
        if stagn > 2000:
            # jitter log-probabilities
            b += 0.01 * rng.standard_normal(n)
            stagn=0

        # gradient for b
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

"""
propose_candidate
-----------------

Search for a low‑value step‑function for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Parametric sweep** – construct a handful of analytically‑motivated
   symmetric distributions (uniform, geometric, Beta, cosine window,
   semicircle, triangle, sine, arcsine).  The best among them gives a very
   good starting point (often already close to the target 1.503).

2. **Many short Adam runs** – repeatedly initialise a (log‑softmax)
   vector from the best sweep candidate, from random Dirichlet samples,
   or by jittering the current best.  For each start a short
   Adam‑optimisation on the *exact* ‑max‑convolution (sub‑gradient) is
   performed.  The runs are tiny (≈5 s each) so that many independent
   restarts can be executed within the total budget.

3. **Final polishing** – a longer Adam run starting from the overall best
   sequence discovered in step 2.

All operations stay on the probability simplex (the vector is always
normalised to sum = 1), therefore the scaling invariance of the objective
is respected.  Convolutions are computed with an FFT of length
`next_pow2(2·n‑1)` for speed.

The function returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT length, convolution, sub‑gradient)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Compute the maximal entry of the aperiodic self‑convolution of ``a``
    and a sub‑gradient of that maximal entry w.r.t. ``a``.
    Returns (max_value, grad) where ``grad`` has the same shape as ``a``.
    """
    # FFT based convolution, keep only the 2·n‑1 valid entries
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    # tolerance to collect all (almost) maximal entries
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # derivative of a_i a_{k-i} w.r.t. a_i is a_{k-i}
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Fast (exact) score – sum(a) is always 1 in our code
# ----------------------------------------------------------------------
def _score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed to be 1)."""
    max_val, _ = _conv_max_and_subgrad(a, fft_len)
    return 2.0 * a.shape[0] * max_val


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(
    init_a: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
    target: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑softmax (``b``) parameters.
    Returns the best (a, score) seen before *time_limit* expires.
    """
    n = init_a.shape[0]
    fft_len = _next_pow2(2 * n - 1)

    # Initialise logits (add a tiny epsilon to avoid log(0))
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # initial evaluation
    a = np.exp(b - b.max())
    a /= a.sum()
    best_score = _score(a, fft_len)
    best_a = a.copy()
    stagn = 0
    it = 0
    deadline = time.time() + time_limit

    while time.time() < deadline:
        it += 1

        # probabilities from logits
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        max_val, grad_a = _conv_max_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
        else:
            stagn += 1

        # early exit if we already beat the target
        if best_score < target:
            break

        # occasional jitter to escape plateaus
        if stagn > 8000:
            b += 0.01 * rng.standard_normal(n)
            stagn = 0

        # sub‑gradient of the objective (objective = 2·n·max_val)
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)          # ⟨a,∇f⟩
        grad_b = a * (grad_obj - avg)      # ∇f in the log‑softmax coordinates

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.5 / np.sqrt(it)            # a slightly larger initial learning rate
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Families of symmetric candidate shapes (all are normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _geometric(n: int, r: float) -> np.ndarray:
    """Symmetric geometric decay a_i ∝ r^{|i‑mid|}."""
    mid = (n - 1) / 2.0
    d = np.abs(np.arange(n) - mid)
    a = r ** d
    a /= a.sum()
    return a


def _beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    x = np.linspace(0.0, 1.0, n)
    # avoid exact 0/1 when α<1 (division by zero)
    a = np.where(
        (x == 0) | (x == 1),
        0.0,
        x ** (alpha - 1) * (1.0 - x) ** (alpha - 1),
    )
    a /= a.sum()
    return a


def _raised_cosine(n: int, power: int = 1) -> np.ndarray:
    """Hann‑type window raised to an integer power."""
    mid = (n - 1) / 2.0
    d = np.arange(n) - mid
    half = (n - 1) / 2.0
    t = d / half
    a = np.maximum(0.0, np.cos(np.pi * t / 2)) ** power
    a /= a.sum()
    return a


def _semicircle(n: int) -> np.ndarray:
    """Discrete semicircle ‑ a_i ∝ sqrt(1‑(2i/(n‑1)‑1)²)."""
    mid = (n - 1) / 2.0
    d = (np.arange(n) - mid) / mid
    a = np.sqrt(np.maximum(0.0, 1.0 - d ** 2))
    a /= a.sum()
    return a


def _triangle(n: int) -> np.ndarray:
    """Tent shape (linear rise then fall)."""
    a = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    a = a.astype(float)
    a /= a.sum()
    return a


def _sine(n: int) -> np.ndarray:
    """Half‑sine window."""
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a /= a.sum()
    return a


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1 / sqrt( (i+0.5)(n-0.5‑i) )."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
