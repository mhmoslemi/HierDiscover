def mirror_descent_phase(a, deadline, max_iters=3000, alpha_end=200.0, eta0=0.8):
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    alpha = 1.0
    alpha_inc = (alpha_end - alpha) / max_iters if max_iters>0 else 0

    for it in range(max_iters):
        if time.time() > deadline - 2:  # keep small margin
            break
        # conv of a with itself
        ft_a = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(ft_a * ft_a, n=2*n-1)
        maxc = conv.max()
        # compute softmax probabilities
        # shift for stability: exp(alpha*(conv - maxc))
        # note conv is np.ndarray length 2n-1
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp
        # correlation gradient
        # compute FFT of p of length L
        ft_p = np.fft.rfft(p, n=L)
        ft_a_full = np.fft.rfft(a, n=L)
        corr = np.fft.irfft(ft_a_full * np.conj(ft_p), n=L)
        offset = len(p) - 1
        g = corr[offset:offset+n]
        # adapt step size
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)
        # increase alpha
        alpha += alpha_inc
    return a

def max_gradient_phase(a, deadline, max_iters=4000, eta0=0.5):
    n = a.shape[0]
    L_conv = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        # compute convolution
        ft_a = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(ft_a * ft_a, n=L_conv)
        k0 = int(conv.argmax())
        # compute gradient g
        # Determine range of i where partner index exists
        start = max(0, k0 - (n - 1))
        end = min(n - 1, k0)
        g = np.zeros_like(a)
        # vectorized assignment:
        i_idx = np.arange(start, end+1)
        j_idx = k0 - i_idx
        g[i_idx] = 2.0 * a[j_idx]
        # step size schedule
        eta = eta0 / (it + 1.0)  # maybe decreasing
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)
    return a

def simulated_annealing_refine(a, cur_score, deadline):
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()
    cur_score = cur_score
    # Temperature schedule
    T = cur_score * 0.02
    T_end = cur_score * 1e-5
    iter_count = 0
    while time.time() < deadline - 0.5 and T > T_end:
        # choose src weighted by a to favor larger masses
        src = np.random.choice(n, p=cur_a)
        # choose different target uniformly
        tgt = np.random.randint(n-1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] <= 0:
            continue
        delta = cur_a[src] * np.random.rand() * 0.5   # limit to half mass
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand_score = compute_score(cand)
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))
        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        # geometric cooling
        T *= 0.9993  # maybe faster cooling
        iter_count += 1
    return best_a, best_score

def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (preserves non‑negativity)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector that already satisfies
    sum(a) == 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution length = 2*n‑1
    ft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(ft * ft, n=2 * n - 1)
    return 2.0 * n * conv.max()

def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)           # avoid overflow/underflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """Linear “edge‑heavy” shape – larger values near the two ends."""
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = dist_to_center.astype(float) + 1.0
    return simplex_normalise(a)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Place mass on k randomly chosen positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


# ----------------------------------------------------------------------
# Phase‑A : smoothed soft‑max mirror descent
# ----------------------------------------------------------------------


def smooth_mirror_descent(a: np.ndarray,
                          deadline: float,
                          max_iters: int = 3000,
                          alpha_end: float = 200.0,
                          eta0: float = 0.8) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp approximation of
    max(conv(a,a)).  The smoothing parameter α grows linearly from 1 to
    ``alpha_end``; the step size η decays as 1/√t.
    """
    n = a.shape[0]
    L_corr = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2
    alpha = 1.0
    delta_alpha = (alpha_end - alpha) / max_iters if max_iters > 0 else 0.0

    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break

        # self‑convolution (FFT length = 2n‑1)
        ft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(ft_a * ft_a, n=2 * n - 1)

        # soft‑max probabilities over convolution entries
        maxc = conv.max()
        # subtract maxc for numerical stability
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp

        # gradient = a *cross‑correlation* p  (non‑circular)
        ft_a_full = np.fft.rfft(a, n=L_corr)
        ft_p = np.fft.rfft(p, n=L_corr)
        corr = np.fft.irfft(ft_a_full * np.conj(ft_p), n=L_corr)
        offset = len(p) - 1
        g = corr[offset:offset + n]

        # exponentiated‑gradient update on the simplex
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)

        # increase smoothing
        alpha += delta_alpha

    return a


# ----------------------------------------------------------------------
# Phase‑B : exact max gradient descent (active‑set)
# ----------------------------------------------------------------------


def max_gradient_descent(a: np.ndarray,
                         deadline: float,
                         max_iters: int = 4000,
                         eta0: float = 0.5) -> np.ndarray:
    """
    Direct descent on the true max of the convolution.
    Uses the sub‑gradient gᵢ = 2·a_{k₀‑i} where k₀ is the current peak index.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1

    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break

        ft = np.fft.rfft(a, n=conv_len)
        conv = np.fft.irfft(ft * ft, n=conv_len)
        k0 = int(conv.argmax())

        # build gradient g
        start = max(0, k0 - (n - 1))
        end = min(n - 1, k0)
        g = np.zeros_like(a)
        if start <= end:
            i_idx = np.arange(start, end + 1)
            j_idx = k0 - i_idx
            # each pair contributes the same factor twice, hence *2
            g[i_idx] = 2.0 * a[j_idx]

        # step size schedule (simple 1/(t+1) decay)
        eta = eta0 / (it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_normalise(a)

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (local stochastic search)
# ----------------------------------------------------------------------


def simulated_annealing_refine(a: np.ndarray,
                               cur_score: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer SA that tries to improve the current vector.
    The temperature starts at a small fraction of the current score
    and cools geometrically.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()
    cur_score = cur_score

    T = cur_score * 0.02          # initial temperature
    T_end = cur_score * 1e-5
    # a slightly faster cooling to spend more iterations on fine tuning
    decay = 0.9995

    while time.time() < deadline - 0.5 and T > T_end:
        # select source index with probability proportional to its mass
        src = np.random.choice(n, p=cur_a)
        # pick a different target uniformly
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        if cur_a[src] <= 0.0:
            continue

        # move a random fraction (up to 50 %) of the source mass
        delta = cur_a[src] * np.random.rand() * 0.5
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        T *= decay

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence() within ``budget_s`` seconds.
    Returns the best sequence found (as a list of floats, sum == 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    # 0. Load previous solution or fall back to a uniform start
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        best_a = simplex_normalise(init_seq)
        n = best_a.size
        best_score = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1 (n={n}), score={best_score:.6f}")
    except NameError:
        # No previous sequence – use a uniform vector of moderate size
        n = 5146
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_a)
        print(f"[Init] No previous data – start uniform, n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Optional light random restarts to escape possible deep local minima
    # ------------------------------------------------------------------
    restart_budget = 5          # few cheap restarts – most time is spent in deep refinement
    for r in range(restart_budget):
        if time.time() > deadline - 4.0:
            break

        # start from current best with a tiny random perturbation
        noise = np.random.rand(n) * 1e-4
        a = simplex_normalise(best_a + noise)

        # -------- Phase A: soft‑max mirror descent ----------
        a = smooth_mirror_descent(a, deadline,
                                   max_iters=2500,
                                   alpha_end=200.0,
                                   eta0=0.9)

        # -------- Phase B: exact max descent -----------------
        a = max_gradient_descent(a, deadline,
                                 max_iters=3000,
                                 eta0=0.6)

        # evaluate current candidate
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Restart {r}] New best after deterministic phases: {best_score:.6f}")

        # -------- Local SA refinement ------------------------
        best_a, best_score = simulated_annealing_refine(best_a, best_score, deadline)
        print(f"[Restart {r}] After SA: best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Final long refinement on the best vector (uses the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        # Smooth phase (more iterations now that we have all remaining time)
        best_a = smooth_mirror_descent(best_a, deadline,
                                       max_iters=4000,
                                       alpha_end=300.0,
                                       eta0=0.8)

        # Exact max phase
        best_a = max_gradient_descent(best_a, deadline,
                                      max_iters=5000,
                                      eta0=0.5)

        # Final SA pass
        best_a, best_score = simulated_annealing_refine(best_a,
                                                        compute_score(best_a),
                                                        deadline)

    total_time = time.time() - start_time
    print(f"[Done] Best score = {best_score:.6f} after {total_time:.1f}s")
    return [float(v) for v in best_a.tolist()]
