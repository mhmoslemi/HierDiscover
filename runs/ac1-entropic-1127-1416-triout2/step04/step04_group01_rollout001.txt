def adam_optimize(initial_a, grad_func, deadline, max_iter, lr0=0.15, beta1=0.9, beta2=0.999, eps=1e-8, clip=None):

    a = project_to_simplex(initial_a.copy())

    best_a = a.copy(); best_score = compute_score(a)

    m = np.zeros_like(a); v = np.zeros_like(a)

    for it in range(1, max_iter+1):
        if time.time() >= deadline: break
        grad = grad_func(a)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1-beta1**it)
        v_hat = v / (1-beta2**it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_to_simplex(a_new)
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        # maybe occasional jitter skip
    return best_a, best_score

"""
A fast multi‑phase optimizer for the step‑function challenge.

Strategy
--------
* Try several lengths (up to 16384).  Larger n lets the optimal
  “arcsine‑type” distribution approach its asymptotic constant ≈1.5,
  so we devote most time to the largest feasible size.
* For each length generate a modest pool of parametric seeds
  (arcsine, uniform, Beta‑family, power‑law, mixtures, triangular bumps)
  and keep the few with the lowest *fast* surrogate score
  `2·n·max(conv(a,a))` (the true objective after normalising Σa=1).
* Refine each seed with three optimisation phases:
    – Temperature‑controlled Adam on a soft‑max surrogate of the max
      (lam ramps from 2 to 250).
    – Adam on the exact max (hard sub‑gradient with momentum).
    – Light simulated‑annealing polishing.
* Keep the best vector seen (according to the official
  `evaluate_sequence`).  After the outer loop a final long Adam+hard‑gradient
  polish is performed on the global best.
* All updates are projected onto the probability simplex to keep the
  sum fixed to 1 (scale‐invariant objective).  The code stays within the
  1000 s budget by allocating iterations from the remaining time.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate score for a vector already normalised to Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1} (Duchi et al., 2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family shape on {0,…,n‑1}: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max (log‑sum‑exp) surrogate of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max).  The sum constraint will be enforced
    by a subsequent projection onto the simplex.
    """
    n = a.size
    conv = convolution_fft(a, a)                # length 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # weighted sub‑gradient = 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length n + (2n‑1) - 1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def hard_gradient(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of 2·n·max(conv(a,a)).
    The sub‑gradient is 2·n·∑_{k∈K} shift(a,k) where K are the
    indices attaining the maximum of the convolution.
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_val = np.max(conv)
    # tolerance to collect all maximisers
    tol = max_val * 1e-12 + 1e-14
    mask = (np.abs(conv - max_val) <= tol).astype(float)   # length 2·n‑1
    mask_rev = mask[::-1]
    L = 2 * n - 1
    conv_rev = convolution_fft(a, mask_rev)
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return 2.0 * n * weighted_grad

# Optimisers
# ----------------------------------------------------------------------


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 250.0,
                         noise_every: int = 2500,
                         noise_scale: float = 2e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule for the soft‑max
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # occasional jitter to escape plateaus
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def adam_optimize_hard(initial_a: np.ndarray,
                       deadline: float,
                       max_iter: int = 100_000,
                       lr0: float = 0.12,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the exact objective (hard max).
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad = hard_gradient(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search that transfers a small mass
    between two randomly chosen coordinates.
    Uses the fast surrogate score for acceptance.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass at i towards j
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

    return best
