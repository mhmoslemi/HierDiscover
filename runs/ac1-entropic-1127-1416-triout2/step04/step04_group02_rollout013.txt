"""
Hybrid optimisation for the min‑max self‑convolution problem.

The search works in three stages:

1. **Exploration** – for a few lengths (powers of two) a diverse library of
   initial probability vectors is built (uniform, arcsine, power‑law
   families, Dirichlet draws and small perturbations).  The cheap scale‑invariant
   score 2·n·max(a∗a) is computed and we keep the most promising candidates.

2. **Smoothed exponential‑gradient descent** – starting from each retained
   candidate we minimise a smooth approximation of the true objective:
        Φ_α(a) = 2·n·softmax_α(a∗a)  ,
   where `softmax_α` is the log‑sum‑exp with temperature α.
   The gradient of Φ_α has a closed‑form convolution, allowing an O(n log n)
   update per iteration.  A continuation on α (0.5 → 2 → 5 → 15) helps to
   escape plateaus.  The update is multiplicative (mirrored‑descent) and the
   vector is re‑normalised onto the probability simplex after each step.

3. **Final polishing** – the globally best vector is refined with a classic
   sub‑gradient method that directly minimises the true max‑convolution.
   This removes the smoothing bias and yields the final candidate.

All work respects the supplied time budget (default 1000 s) and always returns
a feasible non‑negative list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear full convolution using a real FFT.
    The result length is len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine law (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # avoid singularities at the boundaries
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """
    Power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ,
    normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑like vector (Gamma / exponential) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny zero‑mean Gaussian noise and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the softmax weights w_k = exp(α·conv_k) / Σ_j exp(α·conv_j).
    The returned vector has the same length as ``conv`` and sums to 1.
    """
    z = alpha * conv
    m = np.max(z)
    exp_z = np.exp(z - m)
    w = exp_z / exp_z.sum()
    return w


def smooth_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a∗a)_k  with respect to a.
    The formula reduces to 2·( w ∗ a_rev )[n‑1 : n‑1+n] .
    """
    # a_rev = a reversed
    a_rev = a[::-1]
    conv = convolution_fft(w, a_rev)          # length = 3·n‑2
    n = len(a)
    grad = 2.0 * conv[n - 1 : n - 1 + n]      # slice of length n
    return grad


# ----------------------------------------------------------------------
# Exponential‑gradient descent on the smoothed loss
# ----------------------------------------------------------------------


def optimise_smooth(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.8,
    max_iter: int = 120_000,
    alphas: Tuple[float, ...] = (0.5, 2.0, 5.0, 15.0),
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Minimise Φ_α(a)=2·n·softmax_α(a∗a) on the probability simplex.
    Uses a multiplicative (exponentiated) update and a continuation on α.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()                     # work on the simplex
    n = len(a)

    best_a = a.copy()
    # initial true score from the hard max‑convolution
    conv = convolution_fft(a, a)
    best_score = 2.0 * n * conv.max()   # sum(a)=1 -> denominator =1

    it = 0
    # allocation of iterations per temperature
    iters_per_alpha = max_iter // len(alphas)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            # current convolution (full) and the smooth weights
            conv = convolution_fft(a, a)
            w = softmax_weights(conv, alpha)

            # gradient of the smooth objective
            grad = smooth_gradient(a, w)

            # step size schedule η_t = η0 / √t
            eta = eta0 / np.sqrt(it)

            # exponentiated (mirror) update
            a = a * np.exp(-eta * grad)

            # optional exploration noise
            if noise_every and (it % noise_every == 0):
                a += np.random.randn(n) * noise_scale
                a = np.maximum(a, 0.0)

            # project back onto the simplex
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            # every few iterations evaluate the true objective (max‑convolution)
            if it % 2000 == 0:
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                print(
                    f"    smooth it={it:,d} α={alpha:.2f} cur={cur_score:.6f} best={best_score:.6f}"
                )

    # final check at loop end
    conv = convolution_fft(a, a)
    final_score = 2.0 * n * conv.max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (hard objective)
# ----------------------------------------------------------------------


def max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k (where conv_k == max), adds the standard gradient
    g_j = 2·a_{k‑j} (ignoring out‑of‑range indices).  Averages if multiple
    active offsets.
    """
    n = len(a)
    max_val = conv.max()
    # Numerical tolerance – the values are well‑behaved, an absolute atol works.
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 120_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient descent on Φ(a)=max_k (a∗a)_k.
    Returns the best vector seen and its true objective value.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_score = 2.0 * n * conv.max()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 2000 == 0:
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            print(f"    subgrad it={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    # final evaluation
    conv = convolution_fft(a, a)
    final_score = 2.0 * n * conv.max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine respects the supplied time budget and always returns a feasible
    list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Exploration over a few lengths (powers of two are FFT‑friendly).
    # ------------------------------------------------------------------
    lengths = [1024, 2048]  # keep the search space modest to stay within budget

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a modest library of diverse start vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # classic arcsine
        candidates.append(arcsine_initial(n))

        # power‑law family with a few hand‑picked exponents
        for p in (0.35, 0.40, 0.45, 0.50, 0.55, 0.60):
            candidates.append(generate_powerlaw(n, p, eps=0.0))

        # Dirichlet / exponential draws (different concentration parameters)
        for conc in (0.3, 0.6, 1.0, 2.0):
            candidates.append(random_dirichlet(n, conc, rng))

        # small perturbations of the arcsine shape
        for _ in range(2):
            cand = perturb(arcsine_initial(n), rng, scale=0.02)
            candidates.append(cand)

        # --------------------------------------------------------------
        # 1b. Quick sweep with the cheap internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]   # retain three most promising

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # Roughly allocate iterations based on remaining time.
            remaining = max(0.0, deadline - time.time())
            # Empirically about 300 iterations per second for n≈2000.
            max_iter = int(min(120_000, max(5_000, remaining * 300)))

            # ---- Stage 2: smoothed exponential‑gradient descent ----
            refined, _ = optimise_smooth(
                initial_a=cand,
                deadline=deadline,
                eta0=0.8,
                max_iter=max_iter,
                alphas=(0.5, 2.0, 5.0, 15.0),
                noise_every=500,
                noise_scale=3e-6,
            )

            # ---- Stage 3: final hard‑max polishing (optional) ----
            # Give the best vector a short sub‑gradient polish if time permits.
            if time.time() < deadline - 3.0:
                refined, _ = optimise_subgradient(
                    initial_a=refined,
                    deadline=deadline,
                    eta0=0.5,
                    max_iter=int(min(40_000, (deadline - time.time()) * 300)),
                    noise_every=0,
                    noise_scale=0.0,
                )

            # ----- Official evaluation -----
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(200_000, remaining * 300))

        print("\nFinal polishing of the global best (smooth → hard)")
        a_refined, _ = optimise_smooth(
            initial_a=a,
            deadline=deadline,
            eta0=0.4,
            max_iter=max_iter,
            alphas=(0.5, 2.0, 5.0, 15.0, 30.0),
            noise_every=0,
            noise_scale=0.0,
        )
        a_refined, _ = optimise_subgradient(
            initial_a=a_refined,
            deadline=deadline,
            eta0=0.3,
            max_iter=int(min(80_000, (deadline - time.time()) * 300)),
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
