"""
Propose a step‑function (non‑negative vector) that minimizes

        2·n·max(conv(a,a)) / (Σ a)²

which, after projecting onto the probability simplex (Σ a = 1), is just
`2·n·max(conv(a,a))`.  The search consists of three stages:

1. **Initial shape** – start from the classic arcsine distribution (the
   empirically best known shape).

2. **Linear‑programming refinement** – given a current vector `a` with
   `M = max(conv(a,a))`, solve the linear program

        max   Σ b_i
        s.t.  (a * b)_k ≤ M   for all k      (convolution is linear in b)
              b_i ≥ 0

   The solution `b` has a larger ℓ₁‑norm than `a` while keeping the mixed
   convolution bounded by `M`.  After normalising `b` to sum 1 we move
   partially towards it (`a←(1‑t)a + t b`).  Repeating this step (a few
   times, with a decreasing step size `t`) strictly reduces the maximal
   self‑convolution.

3. **Fast local polishing** – a short projected sub‑gradient descent on
   the true `max(conv(a,a))` followed by a tiny stochastic walk on the
   simplex.  All expensive operations are guarded by a global time budget
   (`budget_s`, default 1000 s).

The whole routine respects the time limit, prints progress, and always
returns a feasible (non‑empty, non‑negative, normalised) list.
"""

import time
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, arg‑max index) of the self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)) for a already normalised (Σa = 1).
    This equals the official evaluation value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """Classic symmetric arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP‑step)
# ----------------------------------------------------------------------


def lp_refine_step(a: np.ndarray, t: float, M: float) -> Tuple[np.ndarray, bool]:
    """
    Perform one LP refinement step.
    Returns (new_a, success_flag).
    """
    n = a.shape[0]

    # Construct the (2n‑1)×n constraint matrix for (a * b) ≤ M.
    # Row k (0‑based) contains the coefficients a_{k‑j} for variable b_j.
    A = np.zeros((2 * n - 1, n), dtype=float)
    # The coefficient pattern is simple: each column j contains the whole
    # vector a shifted down by j rows.
    for j in range(n):
        A[j:j + n, j] = a

    # Linear program: maximize Σ b_i  <=>  minimize -Σ b_i
    c = -np.ones(n, dtype=float)
    ub = np.full(2 * n - 1, M, dtype=float)

    # Bounds b_i ≥ 0 (upper bound left unbounded)
    bounds = [(0.0, None)] * n

    # Solve with HiGHS (fast dense solver shipped with SciPy ≥1.6)
    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # Occasionally the LP can be numerically unstable; treat as failure.
        return a, False

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return a, False

    # Normalise to the simplex (Σ b = 1) before mixing.
    b_norm = b / sum_b

    # Mix towards the LP solution.
    a_new = (1.0 - t) * a + t * b_norm
    # Numerical safety: enforce non‑negativity and re‑project.
    a_new = np.maximum(a_new, 0.0)
    a_new = project_to_simplex(a_new)
    return a_new, True


def lp_refine(a0: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Repeated LP refinements with a decreasing step size.
    Returns the best vector discovered (and its fast score).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    while time.time() < deadline - 2.0:
        it += 1
        # Current maximal self‑convolution.
        max_conv, _ = max_convolution_and_index(a)

        # Step size schedule: start with 0.6 and shrink like 1/√it.
        t = 0.6 / np.sqrt(it)

        a_new, ok = lp_refine_step(a, t, max_conv)
        if not ok:
            break

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            print(
                f"    LP iter={it:,d}  t={t:.3f}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

        a = a_new

        # Stop after a modest number of LP steps (the LP is the most expensive part).
        if it >= 15:
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (light version)
# ----------------------------------------------------------------------


def subgradient_optimize_light(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 30_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max(conv(a,a)).
    Uses a decaying step size and stops early when out of time.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # Self‑convolution and index of the active quadratic term.
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # Gradient of the active term: (∂/∂a_i) Σ_j a_j a_{idx-j}
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            if it % 2000 == 0:
                print(
                    f"    Sub‑grad light iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
                )

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local search (very light)
# ----------------------------------------------------------------------


def random_local_search_light(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 120_000,
) -> Tuple[np.ndarray, float]:
    """
    Tiny Gaussian perturbation walk on the simplex.
    Accepts improvements; occasional uphill moves help escape.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = project_to_simplex(a + noise)

        cur_sc = compute_score(a_proposed)

        # Accept if better, otherwise very rarely accept.
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()
                if it % 20000 == 0:
                    print(
                        f"    Random walk light iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
                    )

    return best_a, best_sc
