"""
Improved search for the low‑max‑self‑convolution step function.

The algorithm proceeds in three phases:

1. **Exploratory sampling** – for several candidate lengths we draw many
   symmetric beta‑family densities.  The sampling region is enlarged to
   include extreme parameters (α,β as low as 1e‑5) and very small ε,
   which places a lot of mass near the two ends of the interval – a
   pattern known to flatten the self‑convolution.

2. **Multi‑stage refinement** – the best sampled shape for each length
   is refined by:
      • Adam optimisation of a log‑sum‑exp (soft‑max) surrogate of the
        max‑convolution.  The temperature λ is annealed from 1 to a large
        value (200‑300) so that the surrogate becomes tight.
      • Projected sub‑gradient descent on the true max‑convolution.
        The sub‑gradient is formed from *all* active indices (those whose
        value is within 1 e‑12 of the current maximum) rather than only
        a single index.
      • A stochastic Gaussian walk on the simplex to escape shallow local
        minima.

   All updates are projected onto the probability simplex
   (Σ a = 1, a ≥ 0).

3. **Global polishing** – the globally best vector found so far is
   polished with a longer Adam run (λ up to 300) followed by the same
   sub‑gradient and random‑walk phases.

The routine respects the 1000‑second budget, prints progress, and
guarantees a non‑empty return value.  It can use the optional global
fallback `height_sequence_1` if it exists.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (fast for the lengths we use).
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluation:
        2·n·max(conv(a, a))   (a already normalised to Σa = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """Symmetric arcsine density on {0,…,n‑1} (the classic starter)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returns a normalised vector (Σa = 1).
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                      # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights, sum to 1
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of max

    # weighted sum of sub‑gradients:
    # ∂ (Σ_k w_k (a*a)_k) / ∂a = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)          # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 200.0,
    noise_every: int = 1200,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # Quadratic temperature schedule
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.2,
    max_iter: int = 150_000,
    tol: float = 1e-12,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   max(conv(a,a)).
    Uses *all* indices whose value is within `tol` of the current maximum,
    averaged to form a smoother descent direction.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        active = np.where(conv >= max_val - tol)[0]

        # aggregate gradient from all active indices
        grad = np.zeros_like(a)
        for idx in active:
            i = np.arange(n)
            j = idx - i
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        grad /= max(1, len(active))

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

# Stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with occasional uphill acceptance.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        # accept improvement; occasionally accept small uphill moves
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

            a_opt, _ = adam_optimize(
                initial_a=local_best_a,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.2,
                lam_start=1.0,
                lam_end=250.0,
                noise_every=1000,
                noise_scale=4e-5,
            )

            # Sub‑gradient polishing
            if time.time() < deadline - 4.0:
                max_iter_sub = int(min(120_000, (deadline - time.time()) * 30))
                a_opt, _ = subgradient_optimize(
                    a_opt,
                    time_deadline=deadline,
                    step0=0.18,
                    max_iter=max_iter_sub,
                )

            # Random local walk
            if time.time() < deadline - 3.0:
                max_iter_loc = int(min(180_000, (deadline - time.time()) * 30))
                a_opt, _ = random_local_search(
                    a_opt,
                    time_deadline=deadline,
                    init_step=0.012,
                    max_iter=max_iter_loc,
                )

            # Official (slow) evaluation
            val = evaluate_sequence(a_opt.tolist())
            print(f"    Restart {r+1} official score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = a_opt.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

            # keep the best refined candidate for possible next restart
            if val < local_best_score:
                local_best_score = val
                local_best_a = a_opt.copy()

    # ------------------------------------------------------------------
    # 3. Final polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\n=== Final polishing of the global best ===")
        a = np.array(best_seq, dtype=float)

        # Long Adam run with very high λ
        remaining = deadline - time.time()
        max_iter_adam = int(min(400_000, remaining * 35))
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.15,
            lam_start=1.0,
            lam_end=300.0,
            noise_every=800,
            noise_scale=3e-5,
        )

        # Sub‑gradient polishing
        if time.time() < deadline - 2.0:
            max_iter_sub = int(min(150_000, (deadline - time.time()) * 30))
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.15,
                max_iter=max_iter_sub,
            )

        # Random local walk for final tweaks
        if time.time() < deadline - 1.0:
            max_iter_loc = int(min(250_000, (deadline - time.time()) * 40))
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.008,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4. Guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # Very simple safe fallback (uniform 256‑vector)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
