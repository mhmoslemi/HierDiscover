def refine_sequence_lp(f, time_deadline, max_iters=20, tol=1e-8):
    n = len(f)
    best_f = f.copy()
    best_M = max_convolution_and_index(f)[0]
    for it in range(max_iters):
        if time.time() >= time_deadline:
            break
        # solve LP
        g_opt, sumg = lp_optimize_max_sum(best_f)
        # scale g
        g = g_opt / sumg
        # compute M_new_f as max conv of best_f (already known)
        M_current = max_convolution_and_index(best_f)[0]
        # compute M_g = max conv(g,g)
        M_g = max_convolution_and_index(g)[0]
        # try direct replacement
        if M_g < M_current - tol:
            best_f = g
            best_M = M_g
            continue
        # else try blending
        # we try t=0.5 and maybe half steps
        improved = False
        for t in [0.5, 0.25, 0.75]:
            f_new = (1 - t) * best_f + t * g
            f_new = f_new / f_new.sum()
            M_new = max_convolution_and_index(f_new)[0]
            if M_new < M_current - tol:
                best_f = f_new
                best_M = M_new
                improved = True
                break
        if not improved:
            # no improvement; break
            break
    return best_f

"""
Optimiser for the “minimum max‑self‑convolution’’ problem.

The objective for a non‑negative vector a with Σa=1 is
    score(a) = 2·n·max(conv(a,a))
where n = len(a) and conv denotes the full linear convolution.
The evaluation routine used by the competition returns exactly this value.

The algorithm consists of three phases:

1. **Exploratory seeding** – a handful of analytic families (uniform,
   arcsine, symmetric beta) provide a few good starting vectors for each
   candidate length.

2. **LP‑based refinement** – given a current vector f we fix the
   bound M = max(conv(f,f)) and solve the linear program

        maximise   Σ g_i
        subject to  conv(f,g) ≤ M   (component‑wise)
                    g_i ≥ 0

   The LP has n variables (the entries of g) and 2·n‑1 linear
   constraints.  The optimal g typically has Σg≫1.  By scaling
   g←g/Σg we obtain a new probability vector whose self‑convolution
   bound is reduced to M/Σg.  If this scaled vector improves the score
   we accept it; otherwise we take a short convex combination of the
   old and new vectors.  Repeating the step strictly decreases the
   score, and a few dozen iterations are enough to reach a local
   optimum.

3. **Final polishing** – a lightweight stochastic local search
   (Gaussian perturbations with simplex projection) cleans up any
   remaining imperfections.

All heavy work respects the global time budget (default 1000 s).  The
function ``propose_candidate`` returns the best vector found.
"""

import time
import numpy as np
from typing import List
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)).  a is assumed to be normalised (Σa = 1)."""
    conv = convolution_fft(a, a)
    return float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Symmetric arcsine shape (the classic β(½,½) density)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Linear‑programming refinement (core of the method)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (size (2n‑1)×n) such that A @ g = conv(f,g).
    Each diagonal i (offset i) consists of the constant f[i] repeated n times.
    """
    n = f.shape[0]
    # Every diagonal i contains the constant f[i] on n rows.
    data = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def lp_max_sum_under_bound(f: np.ndarray) -> (np.ndarray, float):
    """
    Solve the LP:
        maximise Σ g_i
        subject to conv(f,g) ≤ M  (component‑wise), g ≥ 0,
    where M = max(conv(f,f)).
    Returns the optimal g and its sum Σg.
    """
    n = f.shape[0]
    M = max_convolution(f)

    # constraint matrix A_ub g ≤ b_ub   (b_ub = M·1)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)

    # maximise Σ g_i  ⇔ minimise –Σ g_i
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback to the current vector (no improvement)
        return f, f.sum()

    g_opt = np.maximum(res.x, 0.0)   # numerical safety
    return g_opt, float(g_opt.sum())


def lp_refine(f0: np.ndarray, deadline: float, max_iters: int = 30) -> np.ndarray:
    """
    Repeated LP refinement of a single vector.
    The process stops when no improvement is observed or the time budget expires.
    """
    f = f0.copy()
    n = len(f)
    best_M = max_convolution(f)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # 1) solve LP for the current bound
        g_raw, sumg = lp_max_sum_under_bound(f)

        # 2) scale to restore Σg = 1
        if sumg <= 0:
            break
        g = g_raw / sumg

        # 3) try the scaled vector directly
        M_g = max_convolution(g)

        if M_g < best_M - 1e-12:
            f = g
            best_M = M_g
            continue

        # 4) otherwise try a short convex combination
        improved = False
        for t in (0.5, 0.25, 0.75):
            f_candidate = (1.0 - t) * f + t * g
            f_candidate = project_to_simplex(f_candidate)
            M_cand = max_convolution(f_candidate)
            if M_cand < best_M - 1e-12:
                f = f_candidate
                best_M = M_cand
                improved = True
                break
        if not improved:
            # no improvement this round → stop
            break

    return f


# ----------------------------------------------------------------------
# Very cheap stochastic polishing (same routine as the previous baseline)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 200_000,
) -> np.ndarray:
    """
    Gaussian perturbations with projection back onto the simplex.
    Only improvements (or a tiny chance of uphill moves) are kept.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = 2 * len(a) * max_convolution(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        proposal = a + rng.normal(scale=step, size=n)
        proposal = project_to_simplex(proposal)

        cur_sc = 2 * n * max_convolution(proposal)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = proposal
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = proposal.copy()

    return best_a

def propose_candidate(

    The routine combines analytic seeding, a specialised LP refinement
    and a final stochastic polishing.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # a tiny safety margin

    # --------------------------------------------------------------

        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_score:
            best_score = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Search over several lengths
    # --------------------------------------------------------------
    lengths = [512, 1024, 2048]                     # larger lengths become costly
    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Length {n} ===")

        # ---- 1a. generate a few seed vectors
        seeds: List[np.ndarray] = [
            np.full(n, 1.0 / n),                     # uniform
            arcsine_initial(n),                      # classic arcsine
        ]

        # a short random Beta sweep to diversify the starting points
        for _ in range(12):
            alpha = rng.uniform(0.2, 1.8)
            beta = rng.uniform(0.2, 1.8)
            eps = rng.uniform(1e-8, min(0.02, 0.01 * n))
            seeds.append(generate_beta_shape(n, alpha, beta, eps))

        # ---- 1b. refine each seed via the LP routine
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline - 8.0:
                break

            seed_vec = project_to_simplex(seed_vec)   # guarantee Σ=1
            local_deadline = deadline - 2.0

            # a modest number of LP iterations; enough to reach a local optimum
            refined = lp_refine(seed_vec, local_deadline, max_iters=25)

            # a final cheap stochastic polish (uses remaining budget)
            if time.time() < deadline - 2.0:
                refined = random_local_search(
                    refined,
                    time_deadline=deadline,
                    init_step=0.015,
                    max_iter=80_000,
                )

            val = evaluate_sequence(refined.tolist())
            print(
                f"  seed #{idx+1:02d} → official score = {val:.6f}"
            )
            if val < best_score:
                best_score = val
                best_seq = refined.tolist()
                print(f"  *** New global best: {best_score:.6f} (n={n}) ***")

    # --------------------------------------------------------------
    # 2.  Global polishing of the overall best vector (if time permits)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing ===")
        a = np.array(best_seq, dtype=float)

        # a final short random walk
        a = random_local_search(
            a,
            time_deadline=deadline,
            init_step=0.012,
            max_iter=120_000,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final official score = {final_val:.6f}")
        if final_val < best_score:
            best_score = final_val
            best_seq = a.tolist()
            print("*** Updated global best after polishing ***")

    # --------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return value
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_score = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_score:.6f}")
    return best_seq
