def multiplicative_subgradient(a, deadline, step0=0.2):
    # a sum to 1
    cur = a.copy()
    best = cur.copy()
    best_sc = compute_score(cur)
    it = 0
    while time.time()<deadline:
        it += 1
        grad = compute_subgradient(cur) # using mask as subgradient (size n)
        # step size decays: eta = step0 / sqrt(it)
        eta = step0 / (np.sqrt(it))
        # update
        cur = cur * np.exp(-eta * grad)
        # renorm
        cur_sum = cur.sum()
        if cur_sum <= 0:
            cur = np.full_like(cur, 1.0 / cur.size)
        else:
            cur = cur / cur_sum
        # Evaluate
        cur_sc = compute_score(cur)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = cur.copy()
        # Optionally early stop if no improvement for many it.
    return best
```

"""
Improved search for a low‑ℓ∞ self‑convolution step function.

Main ideas
----------
*   Sample a diverse pool of seed vectors (uniform, arcsine, Beta‑family,
    power‑law, triangular bumps, and mixtures).
*   For each promising seed run three optimisation phases:
    1.  Adam on a temperature‑controlled soft‑max surrogate of max(conv).
    2.  Multiplicative (exponentiated) sub‑gradient descent on the exact
        objective  f(a)=2·n·max(conv(a,a)).  The update
        a←a·exp(−η·∇f) automatically respects non‑negativity and,
        after renormalisation, stays on the probability simplex.
    3.  A short simulated‑annealing polishing pass.
*   Keep the best official score (using ``evaluate_sequence``) seen across
    all lengths and seeds.
*   Finally perform a global polishing on the overall best vector
    (multiplicative sub‑gradient + Adam) before returning it.

All heavy work is time‑checked against the supplied budget (default
1000 s).  The implementation uses only NumPy, SciPy (optional) and the
standard library and is written without closures so it can be safely
re‑imported by the evaluation harness.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (length = len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed normalised (Σa = 1), therefore the denominator
    of evaluate_sequence disappears: score = 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family shape (i+eps)^{α‑1}(n‑1‑i+eps)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at index `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max).  The simplex constraint is handled by
    the subsequent projection onto the simplex.
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 150_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2500,
                         noise_scale: float = 3e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) and its surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)
        cur_sc = compute_score(a_new)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def compute_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of f(a)=2·n·max(conv(a,a)).
    Returns a vector of length len(a).
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_val = np.max(conv)
    tol = max_val * 1e-12 + 1e-14
    mask = (np.abs(conv - max_val) <= tol).astype(float)
    mask_rev = mask[::-1]
    L = 2 * n - 1
    grad = 2.0 * convolution_fft(a, mask_rev)[L - 1 - np.arange(n)]
    return grad


def multiplicative_subgradient(initial_a: np.ndarray,
                               deadline: float,
                               max_iter: int = 150_000,
                               step0: float = 0.6) -> np.ndarray:
    """
    Exponentiated (multiplicative) sub‑gradient descent on the exact objective.
    The update a←a·exp(-η·∇f) automatically preserves non‑negativity;
    after each step we renormalise to stay on the probability simplex.
    """
    cur = project_to_simplex(initial_a.copy())
    best = cur.copy()
    best_sc = compute_score(cur)
    n = cur.size

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad = compute_subgradient(cur)
        eta = step0 / np.sqrt(it)

        # Prevent overflow/underflow in the exponent.
        exp_factor = np.exp(-eta * grad)
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)

        cur = cur * exp_factor
        s = cur.sum()
        if s <= 0.0 or not np.isfinite(s):
            cur = np.full_like(cur, 1.0 / n)
        else:
            cur /= s

        sc = compute_score(cur)
        if sc < best_sc:
            best_sc = sc
            best = cur.copy()

        if it % 5000 == 0:
            print(f"    MultiSubgrad it={it:,d} cur={sc:.6f} best={best_sc:.6f}")

    return best


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style annealing.
    Randomly transfers a small fraction of mass from i to j.
    Acceptance is based on the fast surrogate (2·n·max(conv)).
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluate_sequence value.
    The routine explores several lengths, samples diverse seeds, refines each
    seed with Adam (smooth surrogate), multiplicative sub‑gradient descent,
    and a short simulated‑annealing polishing phase.
    The best official score seen is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Exploration over several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]   # longer lengths are much heavier
    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ------- 1a. Create a pool of diverse candidates -------------
        pool: List[Tuple[float, np.ndarray]] = []

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for st in starters:
            pool.append((compute_score(st), st.copy()))

        # random beta family
        for _ in range(180):
            alpha = rng.uniform(0.1, 2.5)
            beta = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)
            pool.append((compute_score(cand), cand))

        # inverse‑polynomial (generalised arcsine)
        for _ in range(180):
            p = rng.uniform(0.2, 2.0)
            eps = rng.uniform(0.0, 1.0)

        # mixtures of arcsine & uniform
        for _ in range(80):

        # simple triangular bumps
        for _ in range(80):

            pool.append((compute_score(cand), cand))

        # keep a few most promising seeds
        pool.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in pool[:6]]

        # ------- 2. Refinement of each seed -------------------------
        for seed_vec in top_seeds:
            if time.time() >= deadline - 5.0:
                break

            # Phase A – smooth Adam optimisation
            remaining = deadline - time.time()
            max_iter_adam = int(min(120_000, remaining * 25))
            print(f"  Adam smooth (max_iter={max_iter_adam}) …")
            refined, _ = adam_optimize_smooth(
                initial_a=seed_vec,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.15,
                lam_start=2.0,

                noise_every=2500,
                noise_scale=3e-5,
            )

            # Phase B – multiplicative sub‑gradient descent (exact)
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_mul = int(min(150_000, remaining * 25))
                print(f"  Multiplicative sub‑grad (max_iter={max_iter_mul}) …")
                refined = multiplicative_subgradient(
                    refined,
                    deadline,
                    max_iter=max_iter_mul,
                    step0=0.6,
                )

            # Phase C – simulated‑annealing polishing
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_sa = int(min(120_000, remaining * 25))

            # Final official evaluation

                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception:

    # ------------------------------------------------------------------
    # 3. Global polishing of the overall best vector (if time allows)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if not best_seq:

        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
