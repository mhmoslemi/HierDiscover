def improve_via_lp(a, max_iters, deadline):
    f = a.copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = np.max(convolution_fft(f, f))
        # Build constraints matrix
        n = len(f)
        rows = []
        cols = []
        data = []
        for i in range(n):
            coeff = f[i]
            for j in range(n):
                rows.append(i + j)
                cols.append(j)
                data.append(coeff)
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        b = np.full(2*n-1, M)
        res = scipy.optimize.linprog(-np.ones(n), A_ub=A, b_ub=b,
                                    bounds=[(0, None)]*n, method='highs',
                                    options={'presolve': True})
        if not res.success:
            print("LP failed", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        # Combine

def exponentiated_gradient_refine(a, deadline, max_iter=100_000, eta0=0.3):
    a = a.copy()
    n = len(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # compute subgradient of max conv
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)
        mask_rev = mask[::-1]
        L = 2*n - 1
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # step size schedule decreasing
        eta = eta0 / (np.sqrt(it) + 1)
        a = a * np.exp(-eta * subgrad)
        # renormalize to simplex
        a = a / np.sum(a)
        # project to avoid numeric errors
        a = np.maximum(a, 0)
        # stop check maybe
    return a

"""
Propose a low‑discrepancy step function.

The search is divided into three independent stages that together
exploit complementary ideas:

1️⃣  **Parameterised families** –  Randomly sample arcsine‑like,
    beta‑family and simple mixtures.  For each length we also run a
    cheap hill‑climbing optimisation on the two main families
    (power‑shape and beta‑shape) directly on the fast surrogate
    `2·n·max(conv(a,a))`.

2️⃣  **Projected sub‑gradient descent** –  Starting from the few most
    promising seeds we perform a long run of exact sub‑gradient
    descent (the true objective is `2·n·max(conv)`, the denominator
    is fixed by normalising the vector to the probability simplex).
    The step‑size is `η_t = step0/√t` with a relatively large
    `step0` to escape shallow minima.

3️⃣  **Exponentiated‑gradient polishing** –  A Mirror‑Descent style
    update `a_i ← a_i·exp(‑η·subgrad_i)` keeps the iterate on the
    simplex automatically and often reduces the bound a bit further.

All phases respect a global time budget (default 1000 s).  The best
vector found is finally evaluated with the official ``evaluate_sequence``
and returned.  The implementation uses only NumPy and the standard
library and stays well below the time limit.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # uniform fallback – should never happen in practice
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    The step size is η_t = step0 / √t.  Projection onto the simplex
    enforces Σa = 1.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # identify all indices attaining the maximum (tolerance for FP errors)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    SubGrad it={it:,d} score={cur_sc:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """
    Mirror‑descent style polishing using an exponentiated‑gradient update.
    Keeps the iterate on the probability simplex automatically.
    """
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        # normalise back onto the simplex (numerical safety)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    EG it={it:,d} score={cur_sc:.6f}")

    return a


def hillclimb_power_shape(n: int,
                          init_p: float,
                          init_eps: float,
                          deadline: float,
                          max_iters: int = 300) -> Tuple[np.ndarray, float]:
    """
    Simple random‑walk hill‑climbing on the (p,eps) parameters of the
    generalized arcsine family.  Uses the fast surrogate.
    """
    p, eps = init_p, init_eps
    best_seq = generate_power_shape(n, p, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        # multiplicative perturbation for p (keeps it positive)
        p_new = max(0.01, p * (1.0 + np.random.uniform(-0.15, 0.15)))
        eps_new = max(0.0, eps + np.random.uniform(-0.1, 0.1))
        cand = generate_power_shape(n, p_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            p, eps, best_seq, best_sc = p_new, eps_new, cand, sc
    return best_seq, best_sc


def hillclimb_beta_shape(n: int,
                         init_alpha: float,
                         init_beta: float,
                         init_eps: float,
                         deadline: float,
                         max_iters: int = 300) -> Tuple[np.ndarray, float]:
    """
    Random‑walk hill‑climbing on the three parameters of the beta family.
    """
    alpha, beta, eps = init_alpha, init_beta, init_eps
    best_seq = generate_beta_shape(n, alpha, beta, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        alpha_new = max(0.01, alpha * (1.0 + np.random.uniform(-0.15, 0.15)))
        beta_new  = max(0.01, beta  * (1.0 + np.random.uniform(-0.15, 0.15)))
        eps_new   = max(0.0, eps + np.random.uniform(-0.1, 0.1))
        cand = generate_beta_shape(n, alpha_new, beta_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            alpha, beta, eps, best_seq, best_sc = alpha_new, beta_new, eps_new, cand, sc
    return best_seq, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.  The routine combines cheap random
    sampling, hill‑climbing on parametric families, a long sub‑gradient
    descent, and a final exponentiated‑gradient polishing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Generate a diverse pool of candidates (different lengths)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]   # larger sizes are slower but may improve bound
    global_candidates: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Length n = {n} ===")

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for st in starters:
            sc = compute_score(st)
            global_candidates.append((sc, st.copy()))

        # random beta‑family
        for _ in range(150):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # random power‑shape (generalised arcsine)
        for _ in range(150):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # mixtures of arcsine & uniform
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # simple triangular bumps
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # ------------------------------------------------------------------
        # 1b.  Small local parameter optimisation on the two main families
        # ------------------------------------------------------------------
        # power shape hill‑climb (start from arcsine‑like parameters)
        p0, eps0 = 0.5, 0.0
        cand, sc = hillclimb_power_shape(n, p0, eps0, deadline, max_iters=250)
        global_candidates.append((sc, cand))

        # beta shape hill‑climb (start from α=β=0.5)
        cand, sc = hillclimb_beta_shape(n, 0.5, 0.5, 0.0, deadline, max_iters=250)
        global_candidates.append((sc, cand))

    # Keep the very best few seeds for intensive refinement
    global_candidates.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in global_candidates[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds for refinement (fast score range "
          f"{global_candidates[0][0]:.6f} – {global_candidates[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  Intensive refinement of each seed (sub‑gradient + EG polish)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n={n}) …")

        # Sub‑gradient descent: allocate the bulk of the remaining time
        remaining = deadline - time.time()
        # roughly 150 000 iterations per seed if time permits
        max_iter_sub = int(min(150_000, remaining * 30))
        refined = subgradient_descent_exact(seed_vec, deadline,
                                            max_iter=max_iter_sub,
                                            step0=0.18)  # a relatively large initial step

        # EG polishing (lighter, but often nudges down the bound)
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(80_000, remaining * 30))
            refined = eg_refine(refined, deadline,
                                max_iter=max_iter_eg,
                                eta0=0.25)

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3.  Global polish on the current best (if we have any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(200_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline, max_iter=max_iter_sub, step0=0.12)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(120_000, remaining * 30))
            a = eg_refine(a, deadline, max_iter=max_iter_eg, eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a simple uniform vector (guaranteed to be valid)
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
