import time
import numpy as np

def next_pow2(x): ...
def convolution_fft(...): ...

def project_to_simplex(v): ...

def compute_score(a):
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def softmax_weights(conv, tau):
    # compute numerically stable softmax
    max_c = np.max(conv)
    exps = np.exp((conv - max_c) / tau)
    return exps / np.sum(exps)

def gradient_from_weights(a, w):
    # w is vector length 2n-1 corresponding to weights for each conv index.
    a_rev = a[::-1]
    conv2 = convolution_fft(w, a_rev)
    grad = 2.0 * conv2[:len(a)][::-1]
    return grad

def adam_optimize(a_init, time_deadline, max_iters=200000,
                  tau_start=0.5, tau_end=5e-4, step0=0.5):
    a = a_init.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_sc = compute_score(a)
    # compute schedule for tau linear in log scale
    total_time = time_deadline - time.time()
    while time.time() < time_deadline:
        t += 1
        # compute current tau based on elapsed fraction
        elapsed = time.time() - (time_deadline - total_time)
        frac = elapsed / total_time  # 0..1
        tau = tau_start * (tau_end / tau_start) ** frac

        conv = convolution_fft(a, a)
        w = softmax_weights(conv, tau)  # shape (2n-1)

        grad = gradient_from_weights(a, w)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a = a - step0 * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        if t >= max_iters:
            break
    return best_a, best_sc

"""
Propose a low‑score step function.

The algorithm consists of three stages:

1. **Diverse random seeding** – generate many random vectors on the simplex,
   including uniform weight on a random sparse index set and fully dense
   Dirichlet‑type vectors. Keep the best few candidates.

2. **Smoothed ‑max optimisation** – for each kept candidate run an
   Adam‑style projected gradient descent on a smooth surrogate of the
   max‑convolution (soft‑max with a temperature that is exponentially
   annealed). The gradient is obtained via FFT‑based convolution, and the
   iterates are projected onto the probability simplex to respect the
   non‑negativity and sum‑to‑one constraints.

3. **Final hill‑climb** – a cheap stochastic hill‑climber (swaps and mass
   transfers) polishes the best vector found in the previous stage.

All work respects the 1000 s budget.  The returned list can be fed directly
into `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Scoring (fast replica of evaluate_sequence when sum = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a vector a that already sums to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Random vector generators
# ----------------------------------------------------------------------


def random_sparse_vector(n: int, density: float) -> np.ndarray:
    """Uniform weight on a random subset of size ≈ density·n."""
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_simplex_vector(n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponential (Dirichlet) sampling."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Gradient of the smoothed max (soft‑max) objective
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Stable soft‑max of a convolution vector with temperature tau."""
    # conv is non‑negative; shift by its max for stability
    shift = np.max(conv)
    exps = np.exp((conv - shift) / tau)
    return exps / np.sum(exps)


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a vector a (sum = 1) and a weight vector w of length 2·n‑1,
    return the gradient of  Σ w_k·(a*a)_k with respect to a,
    i.e. 2·(w ⋆ a) where ⋆ is the appropriate correlation.
    """
    n = a.shape[0]
    # correlation can be expressed as a convolution with a reversed w
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)          # length = n + (2n‑1) – 1 = 3n‑2
    # slice the part that corresponds to indices 0 … n‑1 (see derivation)
    slice_section = conv[n - 1 : 2 * n - 1]    # length n
    grad = 2.0 * slice_section[::-1]          # reverse to obtain proper order
    return grad


# ----------------------------------------------------------------------
# Smoothed max optimisation with Adam
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    tau_start: float = 0.5,
    tau_end: float = 5e-4,
    step0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max surrogate of max(a*a).
    The temperature τ decays exponentially from tau_start to tau_end.
    Returns the best vector seen and its (exact) score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_sc = compute_score(a)

    t0 = time.time()
    total_budget = time_deadline - t0
    it = 0

    while time.time() < time_deadline and it < max_iters:
        it += 1
        # --- temperature schedule (exponential decay) ---
        elapsed = time.time() - t0
        frac = elapsed / total_budget
        tau = tau_start * (tau_end / tau_start) ** frac
        if tau < tau_end:
            tau = tau_end

        # --- compute convolution and soft‑max weights ---
        conv = convolution_fft(a, a)        # length 2n‑1
        w = softmax_weights(conv, tau)      # also length 2n‑1, sums to 1

        # --- gradient ---
        grad = gradient_from_weights(a, w)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - step0 * m_hat / (np.sqrt(v_hat) + eps)

        # --- project back onto simplex ---
        a = project_to_simplex(a)

        # --- evaluate exact objective (the true score) ---
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """Stochastic hill‑climber that swaps positions or moves a small fraction of mass."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.1      # fraction of mass moved during a transfer

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 1e-12:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget and returns a plain list of
    floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (previous best known)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1: random exploration on several lengths
    # ------------------------------------------------------------------
    cand_pool = []                     # list of (score, vector) tuples
    pool_size = 5                      # keep the best few candidates
    lengths = [4096, 8192, 16384]      # larger lengths give a better theoretical bound

    print("Phase 1 – random exploration")
    while time.time() < deadline - 30.0:    # reserve ≈30 s for refinements
        n = int(np.random.choice(lengths))
        if np.random.rand() < 0.7:
            # uniform weight on a random sparse subset
            density = np.random.uniform(0.2, 0.9)
            a = random_sparse_vector(n, density)
        else:
            # completely dense random point on the simplex
            a = random_simplex_vector(n)

        sc = compute_score(a)
        # keep the pool sorted (smallest first)
        if len(cand_pool) < pool_size:
            cand_pool.append((sc, a))
            cand_pool.sort(key=lambda x: x[0])
        elif sc < cand_pool[-1][0]:
            cand_pool[-1] = (sc, a)
            cand_pool.sort(key=lambda x: x[0])

        if sc < best_val:
            best_val = sc
            best_seq = a.tolist()
            print(f"New random best: n={n}, score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2: smoothed‑max refinement (Adam) on the best few candidates
    # ------------------------------------------------------------------
    if cand_pool and time.time() < deadline - 10.0:
        remaining = deadline - time.time() - 5.0   # keep ~5 s for final hill‑climb
        per_candidate = max(10.0, remaining / len(cand_pool))  # at least a few seconds each
        print(f"Phase 2 – Adam refinement on {len(cand_pool)} candidates")
        for sc, a in cand_pool:
            sub_deadline = time.time() + per_candidate
            refined_a, refined_sc = adam_optimize(
                a,
                time_deadline=sub_deadline,
                max_iters=200_000,
                tau_start=0.5,
                tau_end=1e-4,
                step0=0.5,
            )
            if refined_sc < best_val:
                best_val = refined_sc
                best_seq = refined_a.tolist()
                print(f"Adam improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3: final hill‑climber polish
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("Phase 3 – final hill‑climb")
        final_seq, final_sc = hill_climb_permutation(best_seq, deadline, max_steps=300_000)
        if final_sc < best_val:
            best_val = final_sc
            best_seq = final_seq
            print(f"Hill‑climb final best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: fall back to a small uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # final official evaluation
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
