def refine_with_adam(a_start, n, deadline):
    # convert to log-space
    eps = 1e-12
    z = np.log(np.maximum(a_start, eps))
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    # hyperparams
    lr = 1.0
    alpha_start = 0.5
    alpha_end = 400.0
    max_iter = 8000
    best_a = a_start.copy()
    best_score = compute_score(best_a)
    for t in range(max_iter):
        if time.time() > deadline - 2.0:
            break
        # compute a from z
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)  # avoid overflow
        a = exp_z / exp_z.sum()
        # compute conv via FFT
        fft_a = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft_a * fft_a, n=2*n-1)
        # softmax weights
        maxc = conv.max()
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp if sum_exp > 0 else np.full_like(conv, 1.0/conv.size)

        # gradient w.r.t a (approx)
        L = 1 << ((3*n-2).bit_length())
        fft_a_full = np.fft.rfft(a, n=L)
        fft_p = np.fft.rfft(p, n=L)
        corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L)
        offset = conv.size - 1
        g = corr[offset:offset+n]  # g_i = Σ_k p_k a_{k-i}
        # approximate full gradient by 2*g (since missing symmetric term)
        grad_a = 2.0 * g

        # gradient w.r.t z = a * (grad_a - dot(a, grad_a))
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1-beta1) * grad_z
        v = beta2 * v + (1-beta2) * (grad_z**2)
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        d = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - d

        # optional: re-normalize to avoid drift (softmax handles)
        if (t & 63) == 0:
            cur_a = np.exp(z - np.max(z))
            cur_a = cur_a / cur_a.sum()
            cur_score = compute_score(cur_a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # print some log
                print(f"[ADAM] iter {t} new best {best_score:.6f}")
    final_a = best_a
    final_score = best_score
    return final_a, final_score

                best_score = cand_score
                best_a = cand.copy()
                print(f"[SAfinal] new best {best_score:.6f}")
        T *= 0.9995

"""Improved search for a step‑function with minimal evaluate_sequence() score.

The algorithm combines a diverse set of deterministic initialisers (uniform,
edge‑heavy, monotone power‑law, exponential decay and Beta densities) and
evaluates them quickly with an FFT‑based exact score function.  The best few
candidates are refined with a long Adam‑style exponentiated‑gradient descent
on the smooth log‑sum‑exp approximation of the ‖f∗f‖∞ norm.  A short
simulated‑annealing phase finally escapes remaining local minima.  All work
fits comfortably inside the 1000 s budget.
"""

import time
import math
import numpy as np

# Basic linear‑algebra helpers (top level – no closures)
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (preserves non‑negativity)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector that already satisfies
    sum(a) == 1.  Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution length = 2·n‑1
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Various initialisers (all return a probability vector of length n)
# ----------------------------------------------------------------------


def init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=float)


def init_edge_concentrated(n: int) -> np.ndarray:
    """
    Linear “edge‑heavy’’ shape: larger values near the two ends,
    smaller in the middle.
    """
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = dist_to_center.astype(float) + 1.0
    return simplex_normalise(a)


def init_linear_decreasing(n: int) -> np.ndarray:
    """
    Strictly decreasing linear shape: a_i ∝ (n‑i).
    """
    a = (np.arange(n, 0, -1).astype(float))
    return simplex_normalise(a)


def init_power_decreasing(n: int, power: float) -> np.ndarray:
    """
    Decreasing power‑law: a_i ∝ (n‑i)^power .
    power ≥ 0.  For power=0 the distribution is uniform.
    """
    a = (np.arange(n, 0, -1).astype(float)) ** power
    return simplex_normalise(a)


def init_exponential_decreasing(n: int, lam: float) -> np.ndarray:
    """
    Exponential decay: a_i ∝ exp(-lam·i) .
    """
    a = np.exp(-lam * np.arange(n, dtype=float))
    return simplex_normalise(a)


def init_beta(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum = 1.
    Mid‑point rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # protect against overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return simplex_normalise(w)


def init_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector (i.e. i.i.d. Gamma(1))."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def adam_refine(a0: np.ndarray,
                deadline: float,
                max_iters: int = 8000,
                lr: float = 1.0,
                alpha_start: float = 0.5,
                alpha_end: float = 400.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the log‑sum‑exp smoothed
    max‑convolution objective.  The iterate is kept on the probability simplex
    by optimising in the log‑space (soft‑max) of the components.
    Returns the best vector seen and its exact score.
    """
    n = a0.shape[0]
    # log‑space variables (soft‑max parametrisation)
    eps = 1e-14
    z = np.log(np.maximum(a0, eps))

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # length for correlation (next power of two ≥ 3·n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    best_a = a0.copy()
    best_score = compute_score(best_a)

    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break

        # ---- 1) obtain a from the soft‑max of z -----------------------
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # ---- 2) self‑convolution ------------------------------------
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

        # ---- 3) soft‑max (log‑sum‑exp) weights -----------------------
        maxc = conv.max()
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp

        # ---- 4) gradient w.r.t. a (using one side of the symmetry) ---
        fft_a_full = np.fft.rfft(a, n=L)
        fft_p = np.fft.rfft(p, n=L)
        corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L)
        offset = conv.size - 1                     # index where lag 0 appears
        g = corr[offset:offset + n]               # g_i = Σ_k p_k·a_{k‑i}

        # The true gradient is 2·g minus a tiny diagonal correction;
        # using 2·g works fine for optimisation.
        grad_a = 2.0 * g

        # ---- 5) chain rule to obtain gradient w.r.t. z ---------------
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)               # ∂f/∂z_i = a_i(∂f/∂a_i - ⟨a,∂f/∂a⟩)

        # ---- 6) Adam update in log‑space -----------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z -= step

        # ---- 7) occasional exact evaluation ---------------------------
        if (it & 31) == 0:          # every 32 iterations
            cur_a = np.exp(z - np.max(z))
            cur_a /= cur_a.sum()
            cur_score = compute_score(cur_a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Adam] iter {it:5d} → new best {best_score:.6f}")

    return best_a, best_score


def simulated_annealing_refine(a: np.ndarray,
                               cur_score: float,
                               deadline: float,
                               start_T_factor: float = 0.02,
                               end_T_factor: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Very light mass‑transfer annealing: pick a source proportional to the mass,
    move a random fraction of it to a random different target.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = cur_score

    T = cur_score * start_T_factor
    T_end = cur_score * end_T_factor

    it = 0
    while time.time() < deadline - 0.5 and T > T_end:
        src = np.random.choice(n, p=cur_a)            # source that actually has mass
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1            # guarantee src != tgt

        if cur_a[src] <= 0.0:
            continue

        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] iter {it:6d} → new best {best_score:.6f}")

        T *= 0.9995
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point – called by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative height sequence (list of floats) that minimises
    evaluate_sequence() under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # tiny safety margin

    # ------------------------------------------------------------------
    # Determine problem size (use the length of the provided solution if any)
    # ------------------------------------------------------------------
    try:
        init_best = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_best.size
        print(f"[Info] Loaded existing height_sequence_1, n={n}")
    except NameError:
        n = 5146
        print(f"[Info] No external vector – using default n={n}")

    # ------------------------------------------------------------------
    # 1. Build an initial pool of diverse candidates
    # ------------------------------------------------------------------
    pool = []

    # deterministic families
    pool.append(init_uniform(n))
    pool.append(init_edge_concentrated(n))
    pool.append(init_linear_decreasing(n))

    # power‑law decreasing with several exponents
    for p in [0.5, 1.0, 2.0, 4.0, 8.0]:
        pool.append(init_power_decreasing(n, p))

    # exponential decay with several rates
    for lam in [0.001, 0.003, 0.006, 0.012]:
        pool.append(init_exponential_decreasing(n, lam))

    # Beta densities that are heavily front‑loaded (α<1, β>1) and back‑loaded
    for a, b in [(0.4, 5.0), (0.6, 3.0), (1.2, 1.2), (2.0, 0.5)]:
        pool.append(init_beta(a, b, n))

    # a handful of random Dirichlet draws for extra diversity
    for _ in range(5):
        pool.append(init_dirichlet(n))

    # ------------------------------------------------------------------
    # 2. Evaluate the pool and keep the best few entries
    # ------------------------------------------------------------------
    scores = [compute_score(v) for v in pool]
    best_idx = int(np.argmin(scores))
    best_a = pool[best_idx].copy()
    best_score = scores[best_idx]
    print(f"[Init] Best pool entry {best_idx} → score {best_score:.6f}")

    # Keep the top 5 distinct candidates for intensive refinement
    top_k = 5
    top_indices = np.argsort(scores)[:top_k]
    top_candidates = [pool[i] for i in top_indices]

    # ------------------------------------------------------------------
    # 3. Refine each top candidate with a long Adam optimisation
    # ------------------------------------------------------------------
    for idx, cand in enumerate(top_candidates, 1):
        if time.time() > deadline - 5.0:
            break
        print(f"[Refine] Starting Adam on candidate {idx}/{len(top_candidates)}")
        refined_a, refined_score = adam_refine(cand, deadline,
                                               max_iters=8000,
                                               lr=1.0,
                                               alpha_start=0.5,
                                               alpha_end=400.0)
        if refined_score < best_score:
            best_score = refined_score
            best_a = refined_a.copy()
            print(f"[Refine] New global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final short simulated‑annealing pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[SA] Starting final annealing")
        best_a, best_score = simulated_annealing_refine(best_a,
                                                        best_score,
                                                        deadline,
                                                        start_T_factor=0.02,
                                                        end_T_factor=1e-5)

    # ------------------------------------------------------------------
    # 5. Return the best vector (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[Done] Best score after search: {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
