def convolution_matrix(f):
    n = len(f)
    L = 2*n - 1
    rows = []
    cols = []
    data = []
    for k in range(L):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # indices range.
        # For each j in j_start..j_end:
        # coefficient = f[k - j]
        # We can vectorize by using np.arange and f values.
        js = np.arange(j_start, j_end+1)
        vals = f[k - js]
        rows.extend([k]*len(js))
        cols.extend(js.tolist())
        data.extend(vals.tolist())
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(L, n))
```

"""
Propose a low‑discrepancy step function.

The algorithm combines two complementary ideas:

1. **LP‑based polishing** –  Starting from a simplex‑projected
   candidate (arcsine or uniform) it repeatedly solves the linear
   program  

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M      for all k,
                    g_i ≥ 0,

   where * denotes the linear convolution, M = max(f*f) and the
   convolution matrix is built once per sequence length.
   The optimal g typically has a larger total mass than the current
   f while respecting the same autocorrelation bound.  Mixing f and
   g (with a fixed mixing factor t) and projecting back onto the
   simplex yields a monotone decrease of the true objective
   `2·n·max(conv)`.  Only a few LP iterations (≈10) are needed.

2. **Smooth Adam + exact sub‑gradient finish** –  After the LP phase
   we run a short Adam optimisation on a soft‑max surrogate of the
   max‑convolution and a few steps of exact sub‑gradient descent.
   The final candidate is evaluated with the official `evaluate_sequence`.

All work is performed under a hard time budget (default 1000 s) and
the best sequence found (as a plain list of non‑negative floats) is
returned.
"""

import time
from typing import List, Tuple

import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The sequence is assumed to be on the probability simplex
    (Σ a = 1), thus the denominator of the official formula disappears.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # uniform fallback – never happens in practice
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def build_convolution_structure(n: int) -> Tuple[np.ndarray, np.ndarray, np.ndarray, Tuple[int, int]]:
    """
    Pre‑compute the sparsity pattern of the convolution matrix
    (f * g)_k = Σ_i f_i * g_{k-i}.  For a fixed length n only the
    values of f change, not the pattern.
    Returns:
        i_arr – index of the coefficient from f for each non‑zero entry,
        rows  – row index (k) for each entry,
        cols  – column index (j) for each entry,
        shape – shape of the matrix (2*n-1, n).
    """
    i_arr = np.repeat(np.arange(n), n)                 # length n²
    j_arr = np.tile(np.arange(n), n)                   # length n²
    rows = i_arr + j_arr                               # k = i + j
    cols = j_arr                                        # column = j
    shape = (2 * n - 1, n)
    return i_arr, rows, cols, shape


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) where the sum constraint is handled
    by a subsequent projection onto the simplex.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length L = 2·n‑1
    max_c = np.max(conv)

    # numerically stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)            # weights, sum to 1

    # weighted sub‑gradient:   2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length n + L - 1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # objective factor 2·n
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 60_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 3e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # occasional jitter to avoid stagnation
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 10_000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 50_000,
                              step0: float = 0.04) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective  f(a)=2·n·max(conv(a,a)).
    The step size is η_t = step0 / sqrt(t).  Projection onto the simplex
    keeps the sum fixed to 1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # indices attaining the maximum (tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 10_000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def lp_refine(f: np.ndarray,
              t: float,
              max_iter: int,
              deadline: float,
              i_arr: np.ndarray,
              rows: np.ndarray,
              cols: np.ndarray,
              shape: Tuple[int, int]) -> Tuple[np.ndarray, float]:
    """
    LP‑based polishing.
    Repeatedly solves
        maximise Σ g_i
        s.t.    conv(f,g) ≤ M   (component‑wise)
                g ≥ 0
    mixes the solution back with the current f and projects onto the
    simplex.  Returns the best vector seen and its fast score.
    """
    n = len(f)
    best_f = f.copy()
    best_sc = compute_score(f)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = np.convolve(f, f)
        M = np.max(conv)

        # build the convolution matrix for the current f
        data = f[i_arr]               # shape (n*n,)
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=shape)

        # linear programme: maximise Σ g_i  ⇔ minimise -Σ g_i
        c = -np.ones(n, dtype=float)
        b_ub = np.full(shape[0], M, dtype=float)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=(0, None), method='highs')
        if not res.success:
            # LP failed – abort polishing for this seed
            break

        g = np.maximum(res.x, 0.0)
        sum_g = np.sum(g)
        if sum_g <= 0 or not np.isfinite(sum_g):
            break

        # normalise g to keep the total mass = 1
        g_norm = g / sum_g

        # convex combination and projection
        f = (1.0 - t) * f + t * g_norm
        f = project_to_simplex(f)

        cur_sc = compute_score(f)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_f = f.copy()
            print(f"    LP iter {it}/{max_iter} – new best score {cur_sc:.6f}")

    return best_f, best_sc

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The search proceeds as follows:

    * Try a few sequence lengths.
    * For each length start from two simple seeds (arcsine, uniform)
      and run a short LP‑based polishing phase.
    * Refine the result with a few hundred ADAM steps on a soft‑max
      surrogate and a brief exact sub‑gradient descent.
    * Keep the globally best sequence and finally perform a longer
      polishing pass.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Fallback: use a previously discovered construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]      # larger lengths become expensive

    for n in lengths:
        if time.time() >= deadline - 20.0:
            break
        print(f"\n=== Starting search for length n = {n} ===")

        # Pre‑compute the convolution matrix sparsity pattern once per n
        i_arr, rows, cols, shape = build_convolution_structure(n)

        # Two deterministic starters for each length
        starters = [
            arcsine_initial(n),
            np.full(n, 1.0 / n)               # uniform
        ]

        for start_idx, start_vec in enumerate(starters):
            if time.time() >= deadline - 10.0:
                break
            print(f"  Starter {start_idx + 1} (initial score {compute_score(start_vec):.6f})")

            f = project_to_simplex(start_vec.astype(float))

            # ------------------------------------------------------------------
            # LP‑based polishing (core of the new approach)
            # ------------------------------------------------------------------
            t_mix = 0.5                     # mixing factor
            max_lp_iters = 12
            f_lp, sc_lp = lp_refine(
                f,
                t=t_mix,
                max_iter=max_lp_iters,
                deadline=deadline,
                i_arr=i_arr,
                rows=rows,
                cols=cols,
                shape=shape,
            )
            # ------------------------------------------------------------------
            # Smooth Adam optimisation (fine‑grained improvement)
            # ------------------------------------------------------------------
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_adam_iters = int(min(60_000, remaining * 30))
                f_adam, _ = adam_optimize_smooth(
                    initial_a=f_lp,
                    deadline=deadline,
                    max_iter=max_adam_iters,
                    lr0=0.12,
                    lam_start=2.0,
                    lam_end=80.0,
                    noise_every=2500,
                    noise_scale=3e-5,
                )
                f_cur = f_adam
            else:
                f_cur = f_lp

            # ------------------------------------------------------------------
            # Exact sub‑gradient descent (polish down to the true objective)
            # ------------------------------------------------------------------
            if time.time() < deadline - 2.0:
                remaining = deadline - time.time()
                max_sub_iters = int(min(50_000, remaining * 30))
                f_cur = subgradient_descent_exact(
                    f_cur,
                    deadline,
                    max_iter=max_sub_iters,
                    step0=0.035,
                )

            # ------------------------------------------------------------------
            # Official evaluation
            # ------------------------------------------------------------------
            try:
                val = evaluate_sequence(f_cur.tolist())
                print(f"  → official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = f_cur.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception as e:
                print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 2.  Global polishing of the overall best vector (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)

        # a short sub‑gradient phase
        remaining = deadline - time.time()
        it_sub = int(min(80_000, remaining * 20))
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.02)

        # a longer Adam phase
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            it_adam = int(min(120_000, remaining * 20))
            a, _ = adam_optimize_smooth(
                a,
                deadline,
                max_iter=it_adam,
                lr0=0.1,
                lam_start=2.0,
                lam_end=80.0,
                noise_every=2000,
                noise_scale=2e-5,
            )

        # final official check
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
