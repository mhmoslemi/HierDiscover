def annealing_optimize(a0, time_deadline, init_temp=0.01, final_temp=1e-6, step_factor=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    n = a.shape[0]
    rng = np.random.default_rng()
    # schedule: T = init_temp * (final_temp/init_temp)^(t/Tmax)
    max_iter = 0
    while time.time() < time_deadline:
        # choose move size delta
        # sample two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # sample delta as fraction of a_i (or set absolute)
        max_delta = a[i]
        if max_delta <= 0:
            continue
        delta = rng.uniform(0, max_delta)
        # propose new vector
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # no need to project; sum unchanged, nonneg satisfied
        # compute score
        # we could compute explicit conv via FFT
        new_score = compute_score(a_new)
        # schedule temperature based on iteration count or progress
        # compute iteration count
        max_iter += 1
        # temperature schedule: geometric decay
        # T = init_temp * (final_temp / init_temp) ** (max_iter / total_est_iter)
        # total_est_iter = maybe 1e6 or derived from time left
        # We'll compute approximate fraction based on elapsed time fraction.
        elapsed = time.time() - start_time
        total_budget = time_deadline - start_time
        # fraction = elapsed / total_budget
        # T = init_temp * (final_temp/init_temp)**fraction
        fraction = elapsed / (time_deadline - start_time)
        # avoid zero division
        T = init_temp * (final_temp / init_temp) ** fraction
        # Accept if new_score < cur_score or random accept
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        # optionally update temperature adaptively
    return best_a, best_score
```

def simulated_annealing(a0, time_deadline, init_temp=0.05, final_temp=1e-6):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    # Determine total allowed time
    total = time_deadline - start
    while time.time() < time_deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = init_temp * (final_temp / init_temp) ** frac
        # delta step size also decays: step scale = initial_step * (1-frac)
        step_scale = max(1e-6, (0.1 * (1-frac)))  # adjust
        # pick i,j
        i, j = rng.choice(n, size=2, replace=False)
        # delta as random fraction of a[i] or step_scale
        if a[i] <= 0:
            continue
        delta = rng.uniform(0, min(a[i], step_scale))
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = compute_score(a_new)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            accept_prob = np.exp(-(new_score - cur_score) / (T + 1e-12))
            if rng.random() < accept_prob:
                accept = True
        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
    return best_a, best_score

"""
Propose a non‑negative step sequence that minimizes the official
evaluate_sequence() value.

Algorithm (≈ 1000 s total):
  1.  Exploration – For several lengths (512, 1024, 2048, 4096)
      generate a collection of candidate shapes:
          * uniform, classic arcsine, sine‑based, and many random
            symmetric beta‑family members.
      Keep the best of each length according to the cheap surrogate
      S(a)=2·n·max(conv(a,a)) (identical to the official score when
      Σa=1).

  2.  Refinement – The best candidate of a length is polished by three
      stages that respect the remaining time budget:
          • Adam optimisation on a smooth log‑sum‑exp approximation of
            the max‑convolution.
          • Projected sub‑gradient descent on the true max‑convolution.
          • Simulated‑annealing on the simplex (mass‑transfer moves that
            occasionally accept uphill steps to escape local minima).

  3.  Global polishing – After the length loop the best vector found
      overall is refined again (long Adam → sub‑gradient → annealing).

All vectors are kept on the probability simplex (Σa=1, a≥0) so that
evaluate_sequence will never return ``inf``.  The routine respects the
budget by checking the deadline before each heavy loop and reduces the
iteration counts when time is scarce.

The implementation uses only NumPy and the provided evaluate_sequence().
"""

import time
import numpy as np
from typing import List, Tuple

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate 2·n·max(conv(a,a)) for a vector already normalised
    (Σa = 1).  This equals the official evaluation value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# --------------------------------------------------------------
# Shape factories
# --------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def sine_initial(n: int) -> np.ndarray:
    """Simple half‑sine shape, normalised."""
    i = np.arange(n, dtype=float) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# --------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# --------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                      # softmax weights

    B = max_c + np.log(sum_exp) / lam           # smooth approximation of max

    # weighted sum of sub‑gradients:
    # ∂ (Σ_k w_k (a*a)_k) / ∂a = 2 * (a * w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    noise_every: int = 1500,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# --------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# --------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on max(conv(a,a)).
    Step size decays like step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# --------------------------------------------------------------
# Simulated annealing on the simplex (mass‑transfer moves)
# --------------------------------------------------------------


def simulated_annealing_optimize(
    a0: np.ndarray,
    time_deadline: float,
    init_temp: float = 0.01,
    final_temp: float = 1e-6,
    min_step_factor: float = 0.02,
) -> Tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing.
    Repeatedly picks two indices (i → j) and moves a random fraction of a[i]
    to a[j].  Acceptance follows the Metropolis rule with a temperature that
    decays geometrically to ``final_temp``.
    Returns the best vector seen.
    """
    a = a0.copy()
    best_a = a.copy()
    cur_score = compute_score(a)
    best_score = cur_score

    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    total_budget = max(time_deadline - start, 1e-6)

    while time.time() < time_deadline:
        # Temperature schedule (geometric decay)
        elapsed = time.time() - start
        frac = elapsed / total_budget
        temp = init_temp * (final_temp / init_temp) ** frac

        # Choose donor and receiver
        i, j = rng.choice(n, size=2, replace=False)

        if a[i] <= 0.0:
            continue

        # Step size shrinks with time; we move at most a fraction of a[i]
        step_factor = max(min_step_factor, (0.5) * (1.0 - frac))
        delta = rng.uniform(0.0, a[i] * step_factor)

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # Evaluate
        new_score = compute_score(a_new)

        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            accept = rng.random() < np.exp(-(new_score - cur_score) / (temp + 1e-12))

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.  The routine combines random beta‑family
    sampling, multi‑stage optimisation (Adam → sub‑gradient → simulated
    annealing) and a final polishing phase.
    """
    rng = np.random.default_rng(seed)

    # 0.  Fallback to a previously found construction if available

    except Exception:

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 8.0:

        print(f"\n=== Exploring length n = {n} ===")

        # ----- deterministic starters -----
        local_best_a = None
        local_best_score = float("inf")
        starters = [
            np.full(n, 1.0 / n),   # uniform
            arcsine_initial(n),
            sine_initial(n),
        ]
        for st in starters:
            sc = compute_score(st)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = st.copy()

        # ----- random beta‑family sampling -----
        max_samples = 800
        for _ in range(max_samples):
            if time.time() >= deadline - 8.0:
                break
            alpha = rng.uniform(0.05, 1.8)

            eps = rng.uniform(1e-6, max(0.01, 0.02 * n))

                )

            # Official evaluation (slow) – one per length
            val = evaluate_sequence(refined_a.tolist())
            print(f"  → refined official score = {val:.6f}")
            if val < best_val:

        else:
            # If refinement was skipped, just test the raw best candidate

    # ------------------------------------------------------------------

    if best_seq and time.time() < deadline - 5.0:

            noise_scale=4e-5,
        )

        # Stage B – sub‑gradient polish
        if time.time() < deadline - 3.0:

        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
