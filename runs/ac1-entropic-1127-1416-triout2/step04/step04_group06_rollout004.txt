"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().

Algorithm outline
-----------------
* Load an existing best sequence (if any) and keep it as the global
  incumbent.
* Perform many independent restarts.  Each restart begins from a
  randomly generated distribution (Beta, Dirichlet, edge‑heavy or
  sparse) optionally blended with the incumbent.
* On each restart run three optimisation phases:
    1. **Entropy‑smoothed mirror descent** (exponentiated gradient) with
       a log‑sum‑exp approximation of the L∞‑norm.  The smoothing
       parameter α is increased from 1 to 300 while an Adam‑style
       adaptive step‑size is used.
    2. **True‑max sub‑gradient descent** on the exact max‑convolution
       value (projected onto the probability simplex).
    3. **Short simulated‑annealing (SA)** consisting of mass‑transfer
       moves to escape possible local minima.
* Throughout the run the best vector (lowest score) is recorded and
  returned at the end of the allotted time budget.
"""

import time
import math
import numpy as np

def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have unit ℓ₁‑norm."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / v.size)
    return v / s


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a (possibly negative) vector onto the
    probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_conv(a: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) convolution a * a using FFT.
    Length returned = 2·len(a)‑1.
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector that already satisfies
    sum(a) == 1.  Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = compute_conv(a)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)            # guard against overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_dirichlet(n: int) -> np.ndarray:
    """Sample a Dirichlet(1,…,1) vector (i.e. i.i.d. Gamma(1))."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """Linear “edge‑heavy” shape: larger values near the two ends."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = dist.astype(float) + 1.0                 # edges get larger weight
    return simplex_normalise(a)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Place mass on *k* random positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)

def mirror_descent_phase(a: np.ndarray,
                         max_iters: int,
                         alpha_end: float,
                         eta0: float,
                         L_corr: int,
                         deadline: float,
                         best_score: float,
                         best_a: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Entropy‑smoothed mirror descent (exponentiated gradient) with an
    Adam‑style adaptive step size.  The smoothing parameter α is linearly
    increased from 1 to *alpha_end*.
    """
    n = a.shape[0]
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    alpha = 1.0
    for it in range(max_iters):
        # time guard
        if time.time() > deadline - 2.0:
            break

        # 1) self‑convolution
        conv = compute_conv(a)

        # 2) soft‑max weights (log‑sum‑exp smoothing)
        maxc = conv.max()
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp

        # 3) gradient of the smoothed max (cross‑correlation)
        fft_a = np.fft.rfft(a, n=L_corr)
        fft_p = np.fft.rfft(p, n=L_corr)
        corr = np.fft.irfft(fft_a * np.conj(fft_p), n=L_corr)
        offset = conv.size - 1   # lag 0 of correlation appears here
        grad = 2.0 * corr[offset:offset + n]   # factor 2 from derivative of a*a

        # 4) Adam‑style moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # 5) exponentiated‑gradient update on the simplex
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        a = simplex_normalise(a)

        # 6) increase smoothing parameter
        alpha += (alpha_end - 1.0) / max_iters

        # occasional score check
        if (it & 31) == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Mirror] iter {it:4d} → new best {best_score:.6f}")

    return a, best_score, best_a


def subgradient_max_phase(a: np.ndarray,
                          max_iters: int,
                          deadline: float,
                          best_score: float,
                          best_a: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution value.
    """
    n = a.shape[0]
    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break

        conv = compute_conv(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Subgrad] iter {it:5d} → new best {best_score:.6f}")

        # sub‑gradient of the max‑entry (pick the first maximal index)
        idx = int(conv.argmax())
        # build the gradient vector: g_j = 2 * a_{idx‑j} if the index exists
        grad = np.zeros_like(a)
        start = max(0, idx - (n - 1))
        end = min(idx, n - 1)
        js = np.arange(start, end + 1)
        grad[js] = 2.0 * a[idx - js]

        # step size (simple diminishing schedule)
        eta = 0.5 / math.sqrt(it + 1.0)

        # Euclidean projected gradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a = project_to_simplex(a_new)

    return a, best_score, best_a


def simulated_annealing_phase(a: np.ndarray,
                              deadline: float,
                              best_score: float,
                              best_a: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Very short SA phase consisting of mass‑transfer moves.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = compute_score(cur_a)

    # start temperature as a small fraction of current score
    T = cur_score * 0.02
    T_end = cur_score * 1e-5
    it = 0
    while time.time() < deadline - 0.2 and T > T_end:
        # pick a source index (probability proportional to its mass)
        src = np.random.choice(n, p=cur_a)
        # pick a distinct target index uniformly
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        if cur_a[src] <= 0.0:
            continue

        delta = cur_a[src] * np.random.rand()   # amount to move
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] iter {it:6d} → new best {best_score:.6f}")

        T *= 0.9995   # geometric cooling
        it += 1

    return cur_a, best_score, best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence() within `budget_s` seconds.
    Returns the best sequence found (as a list of floats, sum ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # 0. Load a previously known solution (if available)
    # ------------------------------------------------------------------
    try:
        # The surrounding environment may provide `height_sequence_1`.
        init_best = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        init_best = simplex_normalise(init_best)
        n = init_best.size
        best_a = init_best.copy()
        best_score = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1 (n={n}), score={best_score:.6f}")
    except NameError:
        # Fallback: uniform vector of modest size (still large enough)
        n = 4096
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_a)
        print(f"[Init] No previous sequence – using uniform start, n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Pre‑compute helper values
    # ------------------------------------------------------------------
    # Length needed for the linear correlation (a with p) -> at most n + (2n‑1) - 1 = 3n‑2
    L_corr = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # 2. Collection of simple initialisers (all produce sum = 1)
    # ------------------------------------------------------------------
    init_functions = [
        lambda: beta_sequence(np.random.uniform(0.2, 5.0),
                              np.random.uniform(0.2, 5.0), n),
        lambda: init_dirichlet(n),
        lambda: init_edge_concentrated(n),
        lambda: init_sparse(n, k=np.random.randint(5, 30)),
    ]

    restart_counter = 0

    # ------------------------------------------------------------------
    # 3. Main restart loop – each restart runs the three phases
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:     # leave a few seconds for final return
        # Choose a random starter
        a = np.random.choice(init_functions)()
        # Occasionally blend with the incumbent to keep good mass distribution
        if np.random.rand() < 0.2:
            a = 0.5 * a + 0.5 * best_a
            a = simplex_normalise(a)

        # --------------------------------------------------------------
        # Phase 1 – entropy‑smoothed mirror descent
        # --------------------------------------------------------------
        max_iters_md = 2500
        eta0 = 0.5          # base learning rate
        alpha_end = 300.0   # more aggressive smoothing than the baseline
        a, best_score, best_a = mirror_descent_phase(
            a,
            max_iters_md,
            alpha_end,
            eta0,
            L_corr,
            deadline,
            best_score,
            best_a,
        )

        # --------------------------------------------------------------
        # Phase 2 – true‑max sub‑gradient descent (projected)
        # --------------------------------------------------------------
        max_iters_sg = 800
        a, best_score, best_a = subgradient_max_phase(
            a,
            max_iters_sg,
            deadline,
            best_score,
            best_a,
        )

        # --------------------------------------------------------------
        # Phase 3 – brief simulated annealing refinement
        # --------------------------------------------------------------
        a, best_score, best_a = simulated_annealing_phase(
            a,
            deadline,
            best_score,
            best_a,
        )

        restart_counter += 1
        print(f"[Restart {restart_counter}] best so far {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best vector (already normalised to sum ≈ 1)
    # ------------------------------------------------------------------
    print(f"[Done] Completed {restart_counter} restarts, final best {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
