def proj_simplex(v):
    # ensure sum = 1 and v >= 0
    # from https: efficient projection O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def optimise_adam(initial_a, deadline, max_iter, epsilon_start=5.0, epsilon_end=200.0, alpha=0.05, beta1=0.9, beta2=0.999, eps=1e-8)

    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # epsilon schedule linear
        eps_cur = epsilon_start + (epsilon_end - epsilon_start) * min(1.0, t / max_iter)
        conv = convolution_fft(a, a)
        max_conv = np.max(conv)
        shifted = conv - max_conv
        exp_vals = np.exp(eps_cur * shifted)  # max exponent zero
        w = exp_vals / np.sum(exp_vals)
        grad = 2 * convolution_fft(w, a[::-1])[n-1:2*n-1]  # length n
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

"""
Improved search for a step‑function with a low max‑convolution value.

The algorithm consists of three parts:
1.  Build a single long candidate (arcsine‑type density) on a fairly large
    length (4096).  This shape is known to be a good starting point.
2.  Optimise the candidate with an Adam‑style projected gradient descent
    on a *smooth* approximation of the max‑convolution:
        φε(a)= (1/ε)·log Σ exp(ε·(a∗a)_k) .
    The gradient of φε is computed analytically via a second convolution,
    which is evaluated with a real‑FFT (O(n log n)).  The iterates are kept
    on the probability simplex (non‑negative, sum = 1) by a fast projection.
    The smoothing parameter ε is increased from a small value to a large
    one, so the optimisation first sees a smooth landscape and then
    gradually focuses on the true non‑smooth objective.
3.  If enough time remains a second short run is started from a different
    random Dirichlet draw, and the best sequence seen overall is returned.
The whole routine respects the supplied time budget (default 1000 s) and
always produces a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    a is assumed non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def proj_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (gamma draws, then normalise)."""
    if concentration <= 0:
        # heavy‑tailed “sparse” draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution objective
# ----------------------------------------------------------------------


def optimise_adam(
    init_a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    *,
    eps_start: float = 0.5,
    eps_end: float = 200.0,
    max_iters: int = 200_000,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    report_every: int = 5000,
) -> np.ndarray:
    """
    Projected Adam optimisation of the smooth max‑convolution objective.
    Returns the best vector encountered (in terms of the true objective).
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()                     # work on the probability simplex
    n = len(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- smoothing parameter (geometric increase) ----
        ratio = min(t, max_iters) / max_iters
        eps_cur = eps_start * (eps_end / eps_start) ** ratio

        # ---- convolution a∗a ----
        conv_aa = convolution_fft(a, a)                # length 2n‑1

        # ---- softmax weights for the smooth max ----
        max_conv = np.max(conv_aa)
        shifted = conv_aa - max_conv                    # max becomes 0
        w_raw = np.exp(eps_cur * shifted)               # under‑flow is fine
        w = w_raw / np.sum(w_raw)                       # sum‑to‑1

        # ---- gradient of φε ----
        # grad = 2 * (w ∗ a_rev) restricted to indices n‑1 … 2n‑2
        grad_full = convolution_fft(w, a[::-1])        # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : 2 * n - 1]       # slice of length n

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---- optional tiny noise to escape plateaus ----
        if t % (report_every * 2) == 0:
            a += rng.normal(0.0, 1e-6, size=n)

        # ---- projection onto the simplex ----
        a = np.maximum(a, 0.0)
        a = proj_onto_simplex(a)

        # ---- keep the best according to the true (non‑smooth) score ----
        if t % report_every == 0:
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
            print(f"    Adam iter={t:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine spends the bulk of the time refining an arcsine‑shaped

    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

            best_seq = fallback

        pass

    # ------------------------------------------------------------------

    if not best_seq:

        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
