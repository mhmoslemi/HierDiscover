def compute_softmax_objective_and_gradient(a, beta):
    # compute conv
    n = len(a)
    L = pow2next
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[:2*n-1]  # linear conv
    max_conv = conv.max()
    # compute softmax weights
    # w_k = exp(beta*(conv_k - max_conv)) / sum_exp
    # For numerical stability, subtract max before exponent.
    exp_term = np.exp(beta * (conv - max_conv))
    w = exp_term / np.sum(exp_term)
    # compute gradient w.r.t a: grad_a = 2 * (conv of w_rev with a) slice
    w_rev = w[::-1]
    fft_w_rev = np.fft.rfft(w_rev, n=L)
    tmp = np.fft.irfft(fft_a * fft_w_rev, n=L)  # length L
    # extract slice: grad_vec = 2 * tmp[2*n-2 : n-2 : -1]
    grad_full = tmp
    grad_a = 2.0 * grad_full[2*n-2 : n-2 : -1]
    return max_conv, grad_a, conv

"""
Propose a step‑height sequence that minimises ``evaluate_sequence``.

The algorithm consists of two stages:

1. **Deterministic Adam optimisation** –   
   The objective ``max(conv)`` (where ``conv = a*a``) is smoothed by a
   log‑sum‑exp (softmax) with a gradually increasing temperature β.
   Using the soft‐max we obtain an analytical gradient that can be
   evaluated with two FFTs per iteration.  The optimisation variables are
   the logits of the probability vector (a = softmax(y)), which guarantees
   non‑negativity and the sum‑to‑one constraint.  Adam adapts the step
   size per coordinate and a simple learning‑rate decay is used.  Several
   diverse initialisations (previous best, uniform, random Dirichlet,
   Beta‑distributions, two‑level patterns) are tried; the best point found
   across all runs is kept.

2. **Guided simulated annealing** –   
   Starting from the best point of stage 1 we perform a Monte‑Carlo
   annealing that moves a random amount of mass from an index with a
   large sub‑gradient (i.e. contributes much to the current maximal
   convolution entry) to an index with a small sub‑gradient.  The temperature
   decays geometrically; worsening moves are accepted with the Metropolis
   probability.  Occasional random restarts keep the search from getting
   stuck.

Both stages stay on the simplex, therefore the returned sequence always
satisfies the validity checks of ``evaluate_sequence``.
"""

import time

def compute_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a vector whose sum is 1.
    Returns 2·n·max(conv) where conv = a * a (linear convolution)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())           # power‑of‑two ≥ 2·n‑1
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum‑to‑1."""
    x = (np.arange(n) + 0.5) / n
    # log‑space to avoid overflow for extreme α,β
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def two_level_pattern(n: int, high_frac: float, ratio: float) -> np.ndarray:
    """
    Build a vector that contains two values.
    ``high_frac`` is the proportion of entries that receive the larger value.
    ``ratio`` = high / low  (>1).  The vector is normalised to sum‑to‑1.
    """
    low = 1.0 / (high_frac * ratio + (1.0 - high_frac))
    high = ratio * low
    a = np.full(n, low, dtype=float)
    high_cnt = max(1, int(round(high_frac * n)))
    idx = np.random.choice(n, size=high_cnt, replace=False)
    a[idx] = high
    a /= a.sum()
    return a


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return the soft‑max weights w_k = exp(β·(conv_k‑max))/∑exp(...)."""
    max_c = conv.max()
    diff = conv - max_c                      # ≤ 0
    # clip to avoid overflow/underflow
    exp_term = np.exp(np.clip(beta * diff, -700.0, 0.0))
    s = exp_term.sum()
    if s == 0.0:
        # all entries under‑flowed → fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / s


def gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k·conv_k   where conv = a*a.
    This equals 2·(a ★ w_rev) restricted to the first n entries,
    where ★ denotes linear convolution and w_rev is w reversed.
    """
    n = a.shape[0]
    # length needed for linear convolution a (n) * w_rev (2n‑1) = 3n‑2
    L = 1 << ((3 * n - 2).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    w_rev = w[::-1]
    fft_w = np.fft.rfft(w_rev, n=L)
    tmp = np.fft.irfft(fft_a * fft_w, n=L)
    # extract grad_i = 2·tmp[2n‑2‑i]   (i = 0..n‑1)
    grad = 2.0 * tmp[2 * n - 2 : n - 2 : -1]
    return grad


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv)`` at ``a`` – average of the exact gradients
    of all active convolution entries (those attaining the current maximum
    up to a tiny tolerance).
    """
    n = a.shape[0]
    max_b = conv.max()
    eps = max_b * 1e-9
    active = np.where(conv >= max_b - eps)[0]

    if active.size == 0:          # safety net
        return 2.0 * a

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


def adam_optimize(init_a: np.ndarray, deadline: float,
                  max_iter: int = 20000,
                  beta_start: float = 5.0,
                  beta_factor: float = 1.005,
                  beta_max: float = 500.0,
                  lr: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth objective,
    starting from ``init_a`` (already normalised to sum = 1).
    Returns the best vector seen and its exact score.
    """
    n = init_a.shape[0]
    # logits y such that a = softmax(y)
    y = np.log(init_a + 1e-16)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta = beta_start
    best_a = init_a.copy()
    best_score = compute_score(best_a)

    t = 0
    while time.time() < deadline - 0.5 and t < max_iter:
        # current probability vector
        a = np.exp(y - np.max(y))
        a /= a.sum()

        # ----- compute convolution and smooth gradient -----
        L = 1 << ((3 * n - 2).bit_length())
        fft_a = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
        # exact max (for true score)
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Adam iter {t:6d}: new best score {best_score:.6f}")

        # soft‑max weights for the smoothed objective
        w = softmax_weights(conv_full, beta)

        # gradient wrt a
        grad_a = gradient_wrt_a(a, w)

        # chain‑rule to logits
        avg = np.dot(a, grad_a)
        grad_y = a * (grad_a - avg)

        # Adam update
        t += 1
        beta = min(beta_max, beta * beta_factor)
        m = 0.9 * m + 0.1 * grad_y
        v = 0.999 * v + 0.001 * (grad_y ** 2)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

    return best_a, best_score


def simulated_annealing(start_a: np.ndarray,
                        deadline: float,
                        init_temp_factor: float = 0.12,
                        final_temp_factor: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Guided simulated annealing starting from ``start_a``.
    Returns the best vector encountered and its exact score.
    """
    n = start_a.shape[0]
    cur_a = start_a.copy()
    _, cur_score = None, compute_score(cur_a)  # compute once
    best_a = cur_a.copy()
    best_score = cur_score

    # initial and final temperatures expressed as a fraction of the current score
    T_start = cur_score * init_temp_factor
    T_end = cur_score * final_temp_factor
    it = 0

    while time.time() < deadline - 0.2:
        # geometric cooling
        T = max(T_end, T_start * (0.9995 ** it))

        # compute a sub‑gradient for source/target biasing
        L = 1 << ((3 * n - 2).bit_length())
        fft_a = np.fft.rfft(cur_a, n=L)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
        grad = subgradient_of_max_conv(cur_a, conv_full)

        # source – high gradient, target – low gradient
        src_probs = grad / grad.sum()
        src = np.random.choice(n, p=src_probs)

        inv = 1.0 / (grad + 1e-12)
        tgt_probs = inv / inv.sum()
        tgt = np.random.choice(n, p=tgt_probs)

        if src == tgt or cur_a[src] <= 0.0:
            it += 1
            continue

        # move a random fraction of mass
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # evaluate candidate
        fft_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fft_c * fft_c, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_c.max()

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it:7d}: new best {best_score:.6f}")

        # opportunistic restart
        if it % 70000 == 0 and it > 0:
            cur_a = np.random.dirichlet(np.ones(n))
            cur_score = compute_score(cur_a)

        if it % 25000 == 0 and it > 0:
            print(f"SA iter {it:7d}: current best {best_score:.6f}")

        it += 1

    return best_a, best_score

    for idx, cand in enumerate(candidates):
        # give each start a slice of the remaining budget (but never exceed)
        remaining = deadline - time.time()
        if remaining < 20.0:       # not enough time for another run
            break
        # allow up to 15 % of the total budget per start (capped by remaining)
        max_iter_time = min(remaining * 0.15, 30.0)   # 30 s cap per run
        # we translate a time budget into an iteration limit (roughly 200 iters/s)
        max_iter = int(max_iter_time * 200)

        print(f"\n=== Adam optimisation run {idx} (max_iter={max_iter}) ===")
        a_opt, score_opt = adam_optimize(cand, deadline, max_iter=max_iter)

        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()
            print(f"*** New overall best after run {idx}: {best_overall_score:.6f}")

    if best_overall_a is None:
        # fallback – should never happen
        best_overall_a = np.ones(n, dtype=float) / n
        best_overall_score = compute_score(best_overall_a)

    # --------------------------------------------------------------
    # Stage 2 – guided simulated annealing starting from the best point
    # --------------------------------------------------------------
    print("\n=== Starting simulated annealing ===")
    best_a, best_score = simulated_annealing(best_overall_a, deadline)

    print(f"\nFinished optimisation. Final best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence (still on the simplex)
    # --------------------------------------------------------------
    best_a /= best_a.sum()
    return [float(v) for v in best_a.tolist()]
