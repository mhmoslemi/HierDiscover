# ----------------------------------------------------------------------

def _full_conv(a: np.ndarray) -> np.ndarray:
    """Aperiodic self‑convolution using FFT (full length 2·n‑1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)

    """

    Sub‑gradient is the average of the gradients of the active entries.
    """
    n = a.shape[0]

    Search for a low‑score step‑function sequence within the allotted time.
    The method combines a short Adam warm‑up, a long simulated‑annealing
    walk, and a final Adam polishing phase.  The best vector found is
    returned as a plain ``list`` of floats.
    """
    rng = np.random.default_rng(123456789)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse a previous construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = prev.shape[0]
    except Exception:
        n = 5146

    # ------------------------------------------------------------------
    # Simple geometric seed (high mass at the two ends, low in the centre)
    # ------------------------------------------------------------------
    def _geometric_initial():
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n, dtype=np.float64)
        right = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
        a0 = left + right
        return a0 / a0.sum()

    if 'height_sequence_1' in globals():
        a0 = np.maximum(prev, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = _geometric_initial()
    else:
        a0 = _geometric_initial()

    # ------------------------------------------------------------------
    # Stage 1 – quick Adam warm‑up (≈ 20 s)
    # ------------------------------------------------------------------
    now = time.time()
    stage1_deadline = min(deadline - 30.0, now + 20.0)
    if now < stage1_deadline:
        a_adam, _ = _adam_optimize(a0, stage1_deadline, restart_id=0, target=1.503)
    else:
        a_adam = a0.copy()

    # ------------------------------------------------------------------
    # Initialise SA state
    # ------------------------------------------------------------------
    cur_a = a_adam.copy()
    cur_conv = _full_conv(cur_a)
    cur_score = 2.0 * n * cur_conv.max()
    best_a = cur_a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Stage 2 – long simulated‑annealing walk
    # ------------------------------------------------------------------
    start_temp = 0.05 * best_score           # modest initial temperature
    cooling = 0.9999                         # exponential cooling factor

    moves = 0
    while time.time() < deadline - 10.0:     # reserve ~10 s for final polishing
        T = start_temp * (cooling ** moves)

        # pick a source index that actually carries mass
        i = rng.integers(n)
        if cur_a[i] == 0.0:
            continue
        # pick a different destination index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount to move – up to half of the source mass
        eps = rng.random() * cur_a[i] * 0.5
        if eps <= 0.0:
            continue

        # perform the move
        _move_mass(cur_a, cur_conv, i, j, eps)
        new_score = 2.0 * n * cur_conv.max()
        delta = new_score - cur_score

        accept = False
        if delta < 0.0:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"SA iter {moves:9d}: new best {best_score:.6f}")
        else:
            # revert the change
            _revert_move(cur_a, cur_conv, i, j, eps)

        moves += 1

    # ------------------------------------------------------------------
    # Stage 3 – final short Adam polishing (≈ 10 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, final_score = _adam_optimize(best_a, deadline,
                                              restart_id=1, target=best_score)
        if final_score < best_score:
            best_score = final_score
            best_a = a_final.copy()
            print(f"Final Adam polishing improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector found (plain list of floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
