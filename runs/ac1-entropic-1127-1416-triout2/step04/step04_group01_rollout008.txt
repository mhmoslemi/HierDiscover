def lp_improve(f, M, deadline):
    n = len(f)
    L = 2*n - 1
    C_mat = build_conv_matrix(f) # dense

    # Create c = -np.ones(n) for maximize sum(g)
    c = -np.ones(n)

    # Build constraints A_ub and b_ub
    A_ub = C_mat
    b_ub = np.full(L, M)

    # bounds for each var: (0,None)
    bounds = [(0, None)] * n

    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
    # Output: success?
    if not res.success:
        return None, None
    g = res.x
    sum_g = g.sum()
    return g, sum_g

def lp_improve(a, M, deadline):
    import numpy as np
    from scipy.optimize import linprog

    n = a.shape[0]
    L = 2*n - 1
    # Build matrix
    C = np.empty((L, n), dtype=np.float64)
    # Zero-initialized
    C[:] = 0.0
    for i in range(n):
        C[i:i+n, i] = a[i]
    # Solve LP
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None)] * n
    # Upper bounds vector
    b_ub = np.full(L, M, dtype=np.float64)

    # Set time limit option if available
    options = {'presolve': True}
    # possibly set 'max_iter'?
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs', options=options)

    if not res.success:
        return None, None

    g = res.x
    sum_g = g.sum()
    return g, sum_g

"""
Propose a low‑discrepancy step sequence.

Approach
--------
1.  Choose a modest length (≤ 2048) and initialise with a classic
    arcsine‑Law (the discrete Beta(½,½) distribution).  A few random
    Beta‑shapes are also generated and the best few are kept.

2.  **Linear‑programming refinement** –  For a current vector `a`
    (∑a = 1) compute `M = max(conv(a,a))`.  Solve the LP
        maximise   ∑ g_i
        subject to  conv(a,g) ≤ M ,   g_i ≥ 0 .
    Because `conv(a,g)` is linear in `g` the constraints are a simple
    dense matrix `C @ g ≤ M`.  If the optimal sum `∑g` exceeds 1 we can
    move a fraction `t` towards the normalised solution `g/∑g`; the
    new vector has a strictly smaller `max(conv)`.

    The LP is solved with SciPy's HiGHS interior‑point method; a few
    (≤ 8) alternations are sufficient to reach a stationary point.

3.  **Exact sub‑gradient descent** –  After the LP stage a few
    thousand steps of sub‑gradient descent on the true objective
    `f(a)=2·n·max(conv(a,a))` (with the simplex projection) polish the
    vector.

4.  **Local simulated‑annealing** –  A cheap Metropolis search on the
    fast surrogate `2·n·max(conv)` finishes the refinement.

5.  The best vector seen for any length is finally evaluated with the
    official ``evaluate_sequence`` and returned.  All steps respect the
    1000 s time budget.

The whole routine is self‑contained, uses only NumPy/SciPy and prints
progress information.
"""

import time
import numpy as np
from scipy.optimize import linprog
from typing import List

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator disappears and the score is simply
    2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of candidate shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine shape:
    a_i ∝ 1/√((i+½)*(n‑½‑i)).
    Normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Linear‑programming based improvement
# ----------------------------------------------------------------------


def build_conv_matrix_left(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix C (size (2n‑1) × n) such that
    conv(a, g) = C @ g   for any g.
    """
    n = a.size
    L = 2 * n - 1
    C = np.empty((L, n), dtype=np.float64)
    C[:] = 0.0
    # each column i contributes a[i] to rows i … i+n‑1
    for i in range(n):
        C[i:i + n, i] = a[i]
    return C


def lp_improve(a: np.ndarray, M: float, deadline: float):
    """
    Solve the LP:
        maximise   Σ g_i
        subject to   C @ g ≤ M      (C built from a)
                     g ≥ 0
    Returns the optimal g and its sum, or (None, None) on failure.
    """
    n = a.size
    C = build_conv_matrix_left(a)          # shape (2n‑1, n)
    L = C.shape[0]

    # Objective – maximise Σg ⇒ minimise –Σg
    c = -np.ones(n, dtype=np.float64)

    # Upper‑bound constraints
    b_ub = np.full(L, M, dtype=np.float64)

    # Variable bounds (non‑negative)
    bounds = [(0.0, None)] * n

    # Respect the remaining time (HiGHS accepts a time_limit option)
    remaining = max(0.0, deadline - time.time())
    options = {"presolve": True}
    if remaining > 0:
        options["time_limit"] = remaining

    res = linprog(
        c,
        A_ub=C,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options=options,
    )
    if not res.success:
        return None, None
    g = res.x
    return g, float(g.sum())


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 50_000,
                              step0: float = 0.04) -> np.ndarray:
    """Standard diminishing‑step sub‑gradient method with simplex projection."""
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # active set (indices attaining the maximum, tolerance for float noise)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient: 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        grad_full = convolution_fft(a, mask_rev)            # length n+L-1
        weighted_grad = 2.0 * grad_full[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing (uses the fast surrogate)
# ----------------------------------------------------------------------


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 60_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 20 % of the mass at i to j
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = compute_score(cand)

        # Metropolis acceptance
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc
        temp = max(temp * decay, 1e-8)

    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The algorithm combines:
        * diverse beta‑family initialisations,
        * a linear‑programming refinement that strictly reduces the
          supremum of the self‑convolution,
        * exact sub‑gradient descent, and
        * a light simulated‑annealing polishing step.
    All work is bounded by the provided time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths (LP works comfortably up to 2048)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ----- generate a small pool of seeds -------------------------
        seed_vectors = []

        # deterministic sensible starters
        seed_vectors.append(arcsine_initial(n))
        seed_vectors.append(np.full(n, 1.0 / n))                # uniform

        # a few random Beta‑family members
        for _ in range(3):
            alpha = rng.uniform(0.15, 2.5)
            beta = rng.uniform(0.15, 2.5)
            eps = rng.uniform(0.001, max(0.5, 0.01 * n))
            seed_vectors.append(generate_beta_shape(n, alpha, beta, eps))

        # keep the most promising few (fast surrogate score)
        scored = [(compute_score(v), v) for v in seed_vectors]
        scored.sort(key=lambda x: x[0])
        top_vectors = [v for _, v in scored[:2]]

        # ----- refine each top seed ----------------------------------
        for idx, init_vec in enumerate(top_vectors, 1):
            if time.time() >= deadline - 5.0:
                break
            print(f"\n  Seed {idx} – initial fast score = {compute_score(init_vec):.6f}")

            a = project_to_simplex(init_vec.copy())   # enforce Σ=1

            # ---------- LP refinement loop ----------
            for lp_it in range(8):
                if time.time() >= deadline - 5.0:
                    break
                M = float(np.max(convolution_fft(a, a)))   # current max(conv)

                g, sum_g = lp_improve(a, M, deadline)
                if g is None or sum_g is None:
                    print("    LP failed – stop refinement")
                    break
                if sum_g <= 1.0 + 1e-8:
                    # no slack left – optimum for this M reached
                    print("    LP found no slack (sum_g≈1) – stop")
                    break

                # mixture update
                t = 0.5
                g_norm = g / sum_g
                a = (1.0 - t) * a + t * g_norm
                a = project_to_simplex(a)

                cur_sc = compute_score(a)
                print(f"    LP {lp_it + 1:2d}: sum_g={sum_g:.3f}, fast score={cur_sc:.6f}")

            # ---------- Sub‑gradient descent ----------
            if time.time() < deadline - 5.0:
                a = subgradient_descent_exact(
                    a,
                    deadline,
                    max_iter=20000,
                    step0=0.04,
                )
                print(f"    after sub‑grad fast score = {compute_score(a):.6f}")

            # ---------- Simulated annealing ----------
            if time.time() < deadline - 5.0:
                a = simulated_annealing(
                    a,
                    rng,
                    deadline,
                    max_iters=50000,
                    init_temp=0.02,
                    decay=0.9995,
                )
                print(f"    after SA fast score = {compute_score(a):.6f}")

            # ----- final official evaluation -----
            try:
                val = evaluate_sequence(a.tolist())
                print(f"  → official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = a.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception as exc:
                print("  Official evaluation failed:", exc)

    # ------------------------------------------------------------------
    # 2.  Global polishing of the overall best vector (if time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        it_sub = int(min(40000, remaining * 30))
        print("\nFinal global polishing – sub‑gradient")
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.02)

        # a final short simulated‑annealing pass
        if time.time() < deadline - 3.0:
            print("Final global polishing – simulated annealing")
            a = simulated_annealing(a, rng, deadline, max_iters=30000,
                                    init_temp=0.015, decay=0.9997)

        try:
            final_val = evaluate_sequence(a.tolist())
            print(f"Final polished official score = {final_val:.6f}")
            if final_val < best_val:
                best_val = final_val
                best_seq = a.tolist()
                print("*** Updated global best after final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform vector – score =", best_val)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
