"""
Improved search for a step function that minimizes
evaluate_sequence(...).

The algorithm combines three ideas:

1. **Random Dirichlet start** – choose a random set of indices and
   assign positive Dirichlet‑distributed masses.  The vector is
   normalised so that  ∑a = √(2 n);  with that normalisation the
   evaluation score equals the maximal entry of the self‑convolution
   (max a∗a).

2. **Linear‑programming exchange step** – given the current vector f
   and its maximal autocorrelation M = max(f∗f), solve the LP

        maximise   1ᵀb
        subject to  f∗b ≤ M ,   b ≥ 0

   (the convolution f∗b is linear in b).  The optimal b has the
   largest possible total mass while still respecting the current bound
   M.  After scaling b to the required normalisation we mix it with f:
   f ← (1‑t) f + t g (t≈0.5).  The mix never increases the bound and
   typically lowers it.  Repeating this step drives the maximal
   autocorrelation down.

3. **Local stochastic hill‑climber** – a cheap projection‑gradient
   search that swaps entries or transfers a fraction of mass between
   coordinates while keeping the vector on the simplex
   (∑a = √(2 n), a ≥ 0).  It polishes the result found by the LP
   exchange.

All work respects the 1000 s budget.  The final list of floats is
returned directly for the evaluation harness.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Fast convolution utilities
# --------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full discrete convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_self_convolution(a: np.ndarray) -> float:
    """Return max_i (a ∗ a)_i."""
    conv = convolution_fft(a, a)
    return float(np.max(conv))


# --------------------------------------------------------------
# Simple simplex projection (∑x = 1, x ≥ 0)
# --------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1)."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def project_to_target_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the simplex and then scale to the desired total mass."""
    w = _project_to_simplex(v)
    return w * target


# --------------------------------------------------------------
# Linear‑programming exchange step (requires SciPy)
# --------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _has_scipy = True
except Exception:                     # pragma: no cover
    _has_scipy = False


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  maximise   1ᵀb   s.t.   f∗b ≤ M , b ≥ 0 .
    Returns the optimal b (or None if the LP fails).
    """
    if not _has_scipy:                # pragma: no cover
        return None

    n = len(f)
    rows = 2 * n - 1
    # Build the convolution matrix A (rows = 2n-1, cols = n)
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f
    # Linear programme (SciPy minimises, so maximise sum(b) → minimise -sum(b))
    c = -np.ones(n, dtype=float)
    b_ub = np.full(rows, M, dtype=float)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options={"presolve": True})
    if res.success:
        return np.array(res.x, dtype=float)
    return None


# --------------------------------------------------------------
# Very lightweight stochastic hill‑climber
# --------------------------------------------------------------
def hill_climb(seq: list[float],
               deadline: float,
               max_steps: int = 200_000) -> tuple[list[float], float]:
    """
    Simple projected hill‑climber.
    The objective is the maximal entry of a∗a (the score when the sum
    equals √(2 n)).  The algorithm swaps entries or transfers a fraction
    of mass between two coordinates while staying on the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # normalise to the target sum (just in case)
    a = project_to_target_sum(a, target_sum)

    best = a.copy()
    best_val = max_self_convolution(best)

    step_scale = 0.1
    while max_steps > 0 and time.time() < deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # project back onto the simplex with the required total mass
        cand = project_to_target_sum(cand, target_sum)

        cur_val = max_self_convolution(cand)
        if cur_val < best_val:
            best_val = cur_val
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.95

        max_steps -= 1

    return best.tolist(), best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low score.
    The routine follows the three‑phase strategy described above.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously discovered candidate
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]           # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_score:
            best_score = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameter ranges
    # ------------------------------------------------------------------
    lengths = [128, 256, 512, 1024]          # small enough for fast LP solves
    restarts_per_len = 6
    lp_iters = 12                            # LP‑mixing iterations per restart

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        for r in range(restarts_per_len):
            if time.time() >= deadline:
                break

            # ---- random Dirichlet start --------------------------------
            density = np.random.uniform(0.25, 0.65)                 # p = k/n
            k = max(2, int(round(density * n)))
            idx = np.random.choice(n, size=k, replace=False)

            # Dirichlet gives a variety of weight shapes (α < 1 -> spikier)
            alpha = np.full(k, 0.6)
            weights = np.random.dirichlet(alpha)                    # sum to 1
            weights = weights * target_sum
            f = np.zeros(n, dtype=float)
            f[idx] = weights

            cur_obj = max_self_convolution(f)                       # equals eval‑score
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = f.tolist()
                print(f"Init  n={n:4d}  restart={r}  obj={cur_obj:.6f}")

            # ---- LP‑exchange phase ------------------------------------
            t_mix = 0.5
            for it in range(lp_iters):
                if time.time() >= deadline:
                    break
                M = max_self_convolution(f)                         # current bound
                b_opt = solve_lp_max_sum(f, M)
                if b_opt is None:
                    break

                # rescale to the required total mass
                scale = target_sum / b_opt.sum()
                g = b_opt * scale

                # convex combination
                f_new = (1.0 - t_mix) * f + t_mix * g
                new_obj = max_self_convolution(f_new)

                if new_obj < cur_obj - 1e-10:                       # improvement
                    f = f_new
                    cur_obj = new_obj
                    t_mix = min(t_mix * 1.08, 0.9)
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = f.tolist()
                        print(f"LP‑mix n={n:4d} restart={r} it={it} obj={cur_obj:.6f}")
                else:
                    t_mix *= 0.5
                    if t_mix < 1e-3:
                        break

            # ---- final stochastic refinement ---------------------------
            if time.time() < deadline - 5.0:
                refined_seq, refined_val = hill_climb(f.tolist(),
                                                      deadline,
                                                      max_steps=150_000)
                if refined_val < best_score:
                    best_score = refined_val
                    best_seq = refined_seq
                    print(f"Hill‑climb n={n:4d} restart={r} improved to {refined_val:.6f}")

    # ------------------------------------------------------------------
    # Safety net – fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        target_sum = np.sqrt(2.0 * n)
        best_seq = [target_sum / n] * n
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (the harness will also call it)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
