def optimise_smooth(initial_a, deadline, max_iter=50000, eta0=0.5, beta0=5.0, beta_inc=1.001, beta_max=5000.0):
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    beta = beta0
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        maxc = float(conv.max())
        # compute softmax weights
        # subtract maxc to prevent overflow
        shifted = beta * (conv - maxc)
        # clamp large negative to avoid underflow? exp(-large) ~ 0 okay.
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
        # compute gradient: g = 2*cross_corr(a, w)
        # Use FFT cross-correlation
        L = next_pow2(len(a) + len(w) - 1)
        A = np.fft.rfft(a, n=L)
        W = np.fft.rfft(w, n=L)
        # cross-correlation: ifft(A * conj(W))
        corr = np.fft.irfft(A * np.conj(W), n=L)
        # gradient components are first n entries (shift 0..n-1)
        grad = 2.0 * corr[:n]
        # step size
        eta = eta0 / np.sqrt(it)
        # update a multiplicatively
        a = a * np.exp(-eta * grad)
        # reproject onto simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # optional evaluate score
        if it % 1000 == 0:
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                #print progress
                print(f"    iter={it} cur={cur_sc:.6f} best={best_sc:.6f}")
        # update beta
        beta = min(beta_max, beta * beta_inc)

    # final best check
    final_sc = compute_score(best_a)
    return best_a, final_sc

"""
Propose a low‑max‑convolution step function.

The search combines:
  * a diverse library of initial shapes (uniform, arcsine, power‑law,
    beta‑like, Dirichlet samples and small perturbations);
  * a *continuation* optimisation that gradually sharpens a smooth
    approximation of the max‑convolution using an exponentiated‑gradient
    (mirror‑descent) update on the probability simplex;
  * a final polishing pass with higher “temperature” (large β) to
    approach the true nonsmooth objective.

All computation respects the wall‑clock budget (default 1000 s) and
always returns a feasible non‑negative sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for evaluate_sequence.
    Returns 2·n·max(a*a) / (sum(a)^2); a may be any non‑negative vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like draw (Gamma or heavy‑tailed exponential)."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny zero‑mean Gaussian perturbation and re‑normalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smoothed max‑convolution optimiser (continuation method)
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the β‑smoothed objective
        f_β(a) = logsumexp(β·(a*a))   (up to a constant factor).

    The gradient is 2 * cross‑correlation of a with the softmax weights
    w_k ∝ exp(β·(conv_k - max(conv))).
    """
    n = len(a)
    conv = convolution_fft(a, a)            # length = 2n‑1
    maxc = np.max(conv)

    # softmax weights (numerically stable)
    shifted = beta * (conv - maxc)          # ≤ 0
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w  (g_j = Σ_i a_i w_{i+j})
    L = next_pow2(len(a) + len(w) - 1)
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(A * np.conj(W), n=L)   # real‑valued
    grad = 2.0 * corr[:n]
    return grad


def optimise_continuation(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    beta0: float = 5.0,
    beta_max: float = 2000.0,
    max_iter: int = 200_000,
    verbose: bool = False,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max‑convolution.
    Works on the probability simplex (sum = 1).
    Returns the best vector seen and its official score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()                     # simplex
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for β (continuation)
        beta = beta0 + (beta_max - beta0) * (it / max_iter)

        # gradient of the smooth surrogate
        grad = _smooth_gradient(a, beta)

        # step‑size schedule η_t = η0 / √t
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad)

        # projection onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional cheap score check
        if it % 2000 == 0 or it == max_iter:
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                if verbose:
                    print(f"    iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
