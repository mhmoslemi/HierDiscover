def compute_score(a):
    n = len(a)
    L = 1 << ((2*n-2).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[:2*n-1]
    max_val = conv.max()
    return 2.0 * n * max_val

def _simulated_annealing(a_start, deadline):
    a = a_start.copy()
    best = a.copy()
    best_score = compute_score(best)
    cur_score = best_score
    n = len(a)
    # temperature schedule parameters
    T0 = best_score * 0.1  # initial temperature maybe 10% of score
    it = 0
    while time.time() < deadline:
        it += 1
        # choose random move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move
        delta_max = a[i]
        if delta_max <= 0:
            continue
        # propose delta as random fraction up to some proportion
        frac = rng.random() * 0.1  # up to 10% of a[i]
        delta = delta_max * frac
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # maintain non-negative; ok.

        # compute score
        cand_score = compute_score(a_candidate)
        # accept if better or via Metropolis
        if cand_score < cur_score:
            accept = True
        else:
            T = T0 * (0.99 ** (it // 1000))  # schedule decay
            if T < 1e-12:
                accept = False
            else:
                prob = np.exp(-(cand_score - cur_score) / T)
                accept = rng.random() < prob

        if accept:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = a_candidate.copy()
                print(f"SA iter {it:,}: new best = {best_score:.6f}")

        # occasional large jitter to escape local minima
        if it % 5000 == 0:
            # random perturbation of a while preserving sum
            perturb = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + perturb, 0)
            total = a.sum()
            if total == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= total

    return best, best_score

assistantfinal**Solution Overview**

The evaluator computes  

\[
\mathrm{score}(a)=\frac{2\,n\,\max(a*a)}{(\sum a)^2},
\]

where `a*a` is the *aperiodic* self‑convolution of the non‑negative vector `a`.  
The score is invariant under scaling, therefore the search can be performed on the simplex  
\(\sum a_i = 1,\; a_i\ge 0\).

The program mixes three complementary optimisation ideas

1. **Smooth surrogate optimisation** – minimise a temperature‑controlled log‑sum‑exp of the
   convolution (`_smooth_optimize`).  The surrogate is differentiable,
   so we can use projected gradient descent on the simplex.
2. **Exact sub‑gradient polishing** – Adam on the true non‑smooth objective
   (`_adam_optimize`).  The gradient of the max‑convolution is obtained analytically.
3. **Simulated‑annealing refinement** – a cheap coordinate‑wise random walk
   (`_simulated_annealing`) that can escape the neighbourhood of a local optimum
   found by the gradient steps.

All three stages run inside the allotted time budget, each stage receiving a
time‑slice that shrinks as the overall deadline approaches.  
The best vector seen at any moment is kept as the global answer.

The code is completely self‑contained, uses only `numpy` and the supplied
`evaluate_sequence`, and respects the required API (`propose_candidate`).  

```python
"""
search for a low‑score step‑function sequence.

The optimiser combines three phases:
1. smooth log‑sum‑exp surrogate minimisation on the simplex,
2. Adam with the exact sub‑gradient of the max‑convolution,
3. a simulated‑annealing walk that perturbs probability mass between
   entries and accepts moves with a temperature schedule.

All phases share the same FFT‑based convolution utilities.
The routine runs until the time budget expires and returns the best
vector (as a list of floats) found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Full aperiodic self‑convolution of a (length 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of weight vector w (length 2·n‑1) with a (length n).
    Returns the segment of length n that corresponds to shifts 0…n‑1.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Exact score (no clipping – the evaluator does the same)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """2·n·max(a*a) for a probability vector (sum(a)=1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Stage 1 – smooth optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_optimize(init_a: np.ndarray, deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    best_a = a.copy()
    best_score = _exact_score(a)

    alpha = 0.5
    alpha_mul = 1.15
    inc_every = 300
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1

        conv = _full_conv(a, L)
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()

        corr = _corr_w_a(w, a, L)
        grad = 4.0 * n * corr

        lr = 0.7 / (1.0 + it * 0.001)
        a -= lr * grad
        a = _project_simplex(a)

        score = _exact_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"Smooth  iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if stagn > 4000:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 2000.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max_conv, sub‑gradient) for a*a."""
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * a.shape[0] - 1]
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Stage 2 – Adam optimisation on the true max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray, deadline: float,
                   restart_id: int, target: float = 1e20) -> tuple[np.ndarray, float]:
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # start from a probability vector (logits)
    eps = 1e-12
    b = np.log(init_a + eps)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    a_cur = np.exp(b - b.max())
    a_cur /= a_cur.sum()
    max_val, _ = _maxconv_and_subgrad(a_cur, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a_cur.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} iter {it:7d}: best = {best_score:.6f}")
        else:
            stagn += 1

        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * np.random.randn(n)
            stagn = 0

        # sub‑gradient of the objective
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‐transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray, deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    best = a.copy()
    best_score = _exact_score(best)
    cur_score = best_score
    n = a.shape[0]
    it = 0
    T0 = best_score * 0.05  # initial temperature

    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue

        # move a random fraction of a[i] to a[j]
        frac = rng.random() * 0.2  # up to 20 % of the source mass
        delta = a[i] * frac

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        cand_score = _exact_score(a_candidate)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            # linear temperature decay
            elapsed = it
            T = max(1e-12, T0 * (1.0 - elapsed / (deadline - time.time() + 1e-12)))
            prob = np.exp(-(cand_score - cur_score) / T)
            accept = rng.random() < prob

        if accept:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = a_candidate.copy()
                print(f"SA iter {it:,}: new best = {best_score:.6f}")

        # occasional global jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            if s > 0:
                a /= s

    return best, best_score

# ----------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_overall_a is None:
        best_overall_a = np.full(n, 1.0 / n)

    # Return a plain Python list (the evaluator expects this)
    return [float(v) for v in best_overall_a.tolist()]
```
