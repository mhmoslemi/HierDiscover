"""
Propose a low‑discrepancy step function.

The search builds on the previous “Adam → sub‑gradient → annealing” pipeline
but adds several new ideas:

*   A wider set of lengths (up to 16384) – the constant factor in
    2·n·max(conv) improves with finer discretisation.
*   Richer random families (beta‑shapes, power‑shapes, mixtures of several
    base shapes) and a cheap grid‑search over the beta parameters.
*   A stronger soft‑max surrogate (λ up to 200) and finer Adam learning‑rate
    schedules.
*   Longer sub‑gradient runs with a smaller initial step.
*   A final “polishing’’ stage that repeats Adam with a very low learning
    rate after the annealing phase.
*   Automatic time‑budget handling – the algorithm constantly checks the
    remaining wall‑clock time and scales the number of iterations accordingly.
*   If a previous solution is available in the global variable
    ``height_sequence_1`` it is used as an extra seed.

The objective is invariant under scaling, therefore all intermediate vectors
are kept on the probability simplex (Σa = 1).  The official
``evaluate_sequence`` function is called only for promising candidates,
so the whole search comfortably fits into the 1000 s limit.
"""

import time
import numpy as np
from typing import List, Tuple

# Low‑level utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = len(a) + len(b) - 1
    n = next_pow2(m)
    A = np.fft.rfft(a, n=n)
    B = np.fft.rfft(b, n=n)
    C = A * B
    conv = np.fft.irfft(C, n=n)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be on the simplex, therefore the denominator
    disappears and score = 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # avoid division by zero – uniform vector is a safe fallback
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump, apex at `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture: weight_arc·arcsine + (1-weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def random_combination(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a random convex combination of 2‑3 basic shapes.
    This enlarges the search space without increasing dimensionality.
    """
    shapes = []
    weights = []

    # always include a base arcsine component (helps with low max)
    shapes.append(arcsine_initial(n))
    weights.append(rng.random())

    # randomly decide how many extra components (1 or 2)
    extra = rng.integers(1, 3)
    for _ in range(extra):
        choice = rng.integers(0, 4)
        if choice == 0:   # beta shape
            a = rng.uniform(0.1, 2.5)
            b = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            shapes.append(generate_beta_shape(n, a, b, eps))
        elif choice == 1:  # power shape
            p = rng.uniform(0.2, 2.0)
            eps = rng.uniform(0.0, 1.0)
            shapes.append(generate_power_shape(n, p, eps))
        elif choice == 2:  # triangular bump
            peak = rng.integers(0, n)
            shapes.append(generate_triangular_shape(n, peak))
        else:               # uniform component
            shapes.append(np.full(n, 1.0 / n))

        weights.append(rng.random())

    # convex combination
    w = np.array(weights)
    w /= w.sum()
    combo = sum(w_i * s for w_i, s in zip(w, shapes))
    return combo / combo.sum()

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max).  The sum constraint is enforced later
    by projection onto the simplex.
    """
    n = a.size
    conv = convolution_fft(a, a)               # length 2*n-1
    max_c = conv.max()
    # softmax weights (stable)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / exp_term.sum()              # sum to 1

    # weighted sum of shifted copies:
    # grad_i = 2 * sum_{k} w_k * a_{k-i}  (with zero‑padding)
    # Using convolution: (a * rev(w))_{n-1+i}
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length n + (2n-1) -1 = 3n-2
    L = 2 * n - 1
    weighted = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return 2.0 * n * weighted


def adam_optimize_smooth(
    init: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.12,
    lam_start: float = 2.0,
    lam_end: float = 200.0,
    noise_every: int = 3000,
    noise_scale: float = 2e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate (soft‑max of the convolution).
    Returns the best vector seen (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(init.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule for the soft‑max temperature λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += rng.normal(scale=noise_scale, size=a_new.shape)

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent(
    init: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step0: float = 0.02,
) -> np.ndarray:
    """
    Exact sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t).
    """
    a = project_to_simplex(init.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()
        # active set (indices attaining the maximum, with tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2 * Σ_{k∈K} shift(a, k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(
    init: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 120_000,
    init_temp: float = 0.02,
    decay: float = 0.9999,
) -> np.ndarray:
    """
    Simple Metropolis‑style mass‑transfer local search.
    Uses the fast surrogate (2·n·max) for acceptance.
    """
    cur = init.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # transfer up to 20 % of mass from i to j
        delta = cur[i] * 0.2 * rng.random()
        if delta < 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.

    The routine explores several lengths, generates a diverse pool of
    parametrised candidates, refines the top few with a multi‑phase optimiser
    and keeps the best vector found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback from a previous run (if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192, 16384]   # larger n gives better asymptotics

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a pool of candidate vectors
        # --------------------------------------------------------------
        pool: List[Tuple[float, np.ndarray]] = []

        # deterministic starters
        pool.append((compute_score(np.full(n, 1.0 / n)), np.full(n, 1.0 / n)))
        pool.append((compute_score(arcsine_initial(n)), arcsine_initial(n)))

        # random beta shapes
        for _ in range(300):
            a = rng.uniform(0.05, 2.5)
            b = rng.uniform(0.05, 2.5)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            pool.append((compute_score(cand), cand))

        # random power shapes (generalised arcsine)
        for _ in range(300):
            p = rng.uniform(0.2, 2.0)
            eps = rng.uniform(0.0, 1.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # mixtures of arcsine & uniform
        for _ in range(150):
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # random convex combinations of several basic shapes
        for _ in range(150):
            cand = random_combination(n, rng)
            pool.append((compute_score(cand), cand))

        # simple triangular bumps
        for _ in range(100):
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

        # --------------------------------------------------------------
        # 1b.  Keep the most promising seeds
        # --------------------------------------------------------------
        pool.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in pool[:6]]   # six seeds per length

        # --------------------------------------------------------------
        # 1c.  Refine each seed
        # --------------------------------------------------------------
        for seed_vec in top_seeds:
            if time.time() >= deadline - 5.0:
                break

            print("\n  Refining a seed ...")
            refined = refine_candidate(seed_vec, rng, deadline)

            # Official evaluation
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"  → official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 2.  Global polishing of the overall best vector (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best vector ===")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        # Sub‑gradient phase
        max_iter_sub = int(min(200_000, remaining * 30))
        print(f"  Global sub‑gradient (max_iter={max_iter_sub})")
        a = subgradient_descent(a, deadline, max_iter=max_iter_sub, step0=0.015)

        remaining = deadline - time.time()
        if remaining > 0:
            max_iter_adam = int(min(150_000, remaining * 30))
            print(f"  Global Adam polish (max_iter={max_iter_adam})")
            a, _ = adam_optimize_smooth(
                init=a,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.07,
                lam_start=20.0,
                lam_end=200.0,
                noise_every=2000,
                noise_scale=1e-5,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a simple arcsine shape of moderate size
        best_seq = arcsine_initial(2048).tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback to arcsine: official score = {best_val:.6f}")

    print(f"\nFinished search. Best reported official score = {best_val:.6f}")
    return best_seq
