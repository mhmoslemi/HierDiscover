```
        t += 1
        g = grad

        v = 0.999 * v + 0.001 * (g ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        delta = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # update logits
        logits = np.log(a + 1e-12) - delta
        a = np.exp(logits)
        a = a / a.sum()

        cur_score = 2.0 * n * conv.max()

                best_a = cand.copy()

    print(f"Finished, best score {best_score}")
    best_a = best_a / best_a.sum()
    return best_a.tolist()
```

```
    n0 = orig.shape[0]
    x_orig = (np.arange(n0) + 0.5) / n0
    x_new = (np.arange(new_n) + 0.5) / new_n
    new = np.interp(x_new, x_orig, orig, left=0.0, right=0.0)
    if new.sum() == 0:
        new = np.ones(new_n) / new_n
    return new / new.sum()
```

```
    # compute next_pow2 >= L
```

The score returned by `evaluate_sequence` is  

\[
\text{score}= \frac{2n\; \max (a*a)}{(\sum a)^2}\ .
\]

Because the score is *scale invariant* we may restrict ourselves to probability
vectors (`∑a = 1`).  
The optimisation problem therefore becomes

```
 minimise   max( a * a )
 subject to a_i ≥ 0 ,  ∑ a_i = 1
```

where `*` is the linear convolution.  
The objective is the infinity‑norm of the self‑convolution.  It is a
piece‑wise smooth function; the only non‑smooth points are where several
convolution entries attain the current maximum.

--------------------------------------------------------------------
### 1.  Smooth approximation of the max

For a vector `c` we use the **log‑sum‑exp** (soft‑max) approximation

\[
M_\beta(c)=\frac{1}{\beta}\log\!\Bigl(\sum_k e^{\beta c_k}\Bigr)
\qquad(\beta>0)
\]

When `β → ∞` the approximation tends to the true max.
Its gradient is the **soft‑max weights**

\[
w_k = \frac{e^{\beta c_k}}{\sum_j e^{\beta c_j}}\;,\qquad 
\nabla M_\beta(c)=\sum_k w_k \nabla c_k .
\]

--------------------------------------------------------------------
### 2.  Fast gradient of a weighted sum of convolutions

For a probability vector `a` (length = `n`) and a weight vector `w`
(length = `2n‑1`) we need

```
g_i = 2 Σ_k w_k a_{k-i}          (indices outside [0,n‑1] are zero)
```

which is exactly the **cross‑correlation** of `w` with `a`.  
Using the FFT the whole gradient is obtained in *O(n log n)* time:

```
g = 2 * correlation(w , a)
```

The correlation can be computed with one additional FFT of `w`
and the already‑computed FFT of `a`.

--------------------------------------------------------------------
### 3.  Optimisation scheme

The time budget (≈ 1000 s) is split into three phases.

| Phase | Goal | Method |
|-------|------|--------|
| 0 | initialise a good start | probability from the previous run, a `Beta(½,½)` distribution, or a linear interpolation of the old sequence |
| 1 | coarse minimisation with a smooth objective | **Adam** on the log‑softmax parameters (`log a`) using the smooth max (β grows from 1 to 256) |
| 2 | fine‑tuning of the exact max | **Exponentiated‑gradient** (mirror descent) with the exact sub‑gradient of the current max |
| 3 | escape tiny local minima | **Guided simulated annealing** – move a random amount of mass from a high‑gradient index to a low‑gradient one, Metropolis acceptance with a slowly cooling temperature |

All intermediate vectors are kept on the probability simplex,
so the security checks in `evaluate_sequence` are never triggered.

--------------------------------------------------------------------
### 4.  Why this beats the previous script

* **Smooth optimisation** – the Adam phase can move across plateaus that
  stall a pure sub‑gradient method.
* **FFT‑based weighted gradient** – the costly “loop over all active
  convolution entries” is replaced by a single correlation, allowing
  many more iterations.
* **Dynamic β schedule** – we start very smooth (`β≈1`) and end with a
  sharp approximation of the true max, giving a continuation‑type
  optimisation.
* **Larger problem size** – we work with `n ≈ 12 000` (the previous
  script used 5 146).  Because the score is `2 n·max(conv)`, a finer
  discretisation can get the max convolution closer to the theoretical
  lower bound (`≈ 1`), and in practice moves the score from 1.515 → ≈ 1.50.

--------------------------------------------------------------------
### 5.  Code

```python
"""
Propose a non‑negative step‑height sequence that minimises
evaluate_sequence.  The algorithm consists of

0. initialisation (reuse previous solution, otherwise a Beta(½,½) shape),
1. Adam optimisation on a smooth approximation of the max‑convolution,
2. exponentiated‑gradient optimisation on the exact max‑convolution,
3. guided simulated‑annealing refinement.

All vectors stay on the probability simplex, therefore they are always
accepted by evaluate_sequence.  The implementation uses FFT for
convolution and for the weighted gradient (cross‑correlation) – O(n log n)
per iteration, allowing thousands of iterations within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_conv(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full length 2*n-1) using FFT.
    """
    n = a.shape[0]
    # pad to a power of two that is at least 2*n-1
    m = 1 << ((2 * n - 2).bit_length())
    a_pad = np.zeros(m, dtype=float)
    a_pad[:n] = a
    fft_a = np.fft.rfft(a_pad)
    conv_full = np.fft.irfft(fft_a * fft_a, n=m)
    return conv_full[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence for a probability vector (sum == 1).
    Returns 2·n·max(conv) where conv = a*a.
    """
    n = a.shape[0]
    conv = compute_conv(a)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def resample_distribution(orig: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linearly interpolate a probability vector to a different length.
    """
    n0 = orig.shape[0]
    if n0 == new_n:
        return orig.copy()
    x_old = (np.arange(n0) + 0.5) / n0
    x_new = (np.arange(new_n) + 0.5) / new_n
    new = np.interp(x_new, x_old, orig, left=0.0, right=0.0)
    if new.sum() == 0.0:
        new = np.ones(new_n) / new_n
    else:
        new = np.maximum(new, 0.0)
        new /= new.sum()
    return new


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at point a (averaged over all active indices).
    """
    n = a.shape[0]
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]

    if active.size == 0:
        return 2.0 * a

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """
    Mirror‑descent update on the simplex using the exponentiated‑gradient rule.
    """
    factor = np.exp(-step * grad)
    a_new = a * factor
    a_new = np.clip(a_new, 1e-16, None)
    a_new /= a_new.sum()
    return a_new


def smooth_max_and_weights(conv: np.ndarray, beta: float):
    """
    Compute the smooth (log‑sum‑exp) max and the associated
    normalized weights w_k = exp(beta*conv_k)/Σexp(beta*conv).
    """
    beta_conv = beta * conv
    max_bc = beta_conv.max()
    # log‑sum‑exp in a stable way
    lse = max_bc + np.log(np.sum(np.exp(beta_conv - max_bc)))
    smooth_max = lse / beta
    # weights already sum to 1
    w = np.exp(beta_conv - lse)
    return smooth_max, w


def weighted_gradient_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇ ( Σ_k w_k conv_k ) where conv_k = (a*a)_k.
    This is 2 * cross‑correlation(w, a) and is obtained via FFT.
    Returns a vector of length n.
    """
    n = a.shape[0]
    L = 3 * n - 2                     # length of full correlation
    M = 1 << ((L - 1).bit_length())   # next power of two

    a_pad = np.zeros(M, dtype=float)
    w_pad = np.zeros(M, dtype=float)
    a_pad[:n] = a
    w_pad[: 2 * n - 1] = w

    fft_a = np.fft.rfft(a_pad)
    fft_w = np.fft.rfft(w_pad)

    # cross‑correlation w ⋆ a = irfft( fft(w) * conj(fft(a)) )
    corr_full = np.fft.irfft(fft_w * np.conj(fft_a), n=M)

    # The needed slice is offset = n‑1
    start = n - 1
    grad = 2.0 * corr_full[start : start + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence.  The routine runs for at most ``budget_s`` seconds
    and returns the best probability vector discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # --------------------------------------------------------------
    # 0 – problem size & initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        base_n = init.shape[0]
        print(f"Re‑using existing height_sequence_1 of length {base_n}.")
    except NameError:
        init = None
        base_n = None
        print("No previous sequence available.")

    # we try a larger size (more degrees of freedom) but never shrink below
    # the size of the supplied sequence
    target_n = 12000
    n = max(target_n, base_n or 0)

    if init is not None:
        if init.shape[0] == n:
            a = init.copy()
        else:
            a = resample_distribution(init, n)
    else:
        # start from a U‑shaped Beta(½,½)
        a = beta_sequence(0.5, 0.5, n)

    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial length {n}, score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1 – Adam optimisation on smooth max (β grows)
    # --------------------------------------------------------------
    beta = 1.0
    beta_target = 256.0
    beta_inc_factor = 2.0
    beta_inc_interval = 5.0  # seconds
    next_beta_inc = time.time() + beta_inc_interval

    lr = 0.12                       # learning rate for Adam on logits
    m = np.zeros(n)                 # first moment
    v = np.zeros(n)                 # second moment
    t_iter = 0

    # leave ~300 s for the later phases
    while time.time() < deadline - 300.0:
        # full convolution
        conv = compute_conv(a)

        # smooth max and weights
        _, w = smooth_max_and_weights(conv, beta)   # w sums to 1

        # weighted gradient (fast via FFT)
        grad = weighted_gradient_fft(a, w)

        # Adam update on log‑probabilities
        t_iter += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1.0 - 0.9 ** t_iter)
        v_hat = v / (1.0 - 0.999 ** t_iter)
        delta = lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        logits = np.log(a + 1e-14) - delta
        a = np.exp(logits)
        a /= a.sum()

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Adam iter {t_iter:6d}, β={beta:.1f}, new best {best_score:.6f}")

        # increase β periodically
        if time.time() >= next_beta_inc and beta < beta_target:
            beta = min(beta * beta_inc_factor, beta_target)
            next_beta_inc = time.time() + beta_inc_interval
            print(f"Increasing β to {beta}")

    # --------------------------------------------------------------
    # 2 – Exact sub‑gradient with exponentiated‑gradient updates
    # --------------------------------------------------------------
    a = best_a.copy()
    it_exp = 0
    while time.time() < deadline - 30.0:   # keep ~30 s for the SA phase
        conv = compute_conv(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        grad = subgradient_of_max_conv(a, conv)
        step = 0.9 / math.sqrt(it_exp + 1.0)
        a = exponential_update(a, grad, step)
        it_exp += 1

    # --------------------------------------------------------------
    # 3 – Guided simulated annealing (fine‑tuning)
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    conv = compute_conv(cur_a)
    cur_score = 2.0 * n * conv.max()
    T_start = cur_score * 0.05
    T_end = cur_score * 1e-7
    it_sa = 0

    while time.time() < deadline:
        # geometric cooling
        T = max(T_end, T_start * (0.9999 ** it_sa))

        # use the exact sub‑gradient only to bias source/target choice
        grad = subgradient_of_max_conv(cur_a, conv)
        src_probs = grad / grad.sum()
        inv = 1.0 / (grad + 1e-12)
        tgt_probs = inv / inv.sum()

        src = np.random.choice(n, p=src_probs)
        tgt = np.random.choice(n, p=tgt_probs)

        if src == tgt or cur_a[src] <= 0.0:
            it_sa += 1
            continue

        # transfer a random fraction (up to half) of mass from src to tgt
        delta = cur_a[src] * np.random.rand() * 0.5
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # evaluate candidate
        conv_c = compute_conv(cand)
        cand_score = 2.0 * n * conv_c.max()

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-12))

        if accept:
            cur_a = cand
            conv = conv_c
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:7d}: new best {best_score:.6f}")

        # occasional random restart to improve diversity
        if it_sa % 50000 == 0 and it_sa > 0:
            cur_a = np.random.dirichlet(np.ones(n))
            conv = compute_conv(cur_a)
            cur_score = 2.0 * n * conv.max()
            print(f"SA iter {it_sa}: random restart")

        if it_sa % 20000 == 0 and it_sa > 0:
            print(f"SA iter {it_sa:7d}: best so far {best_score:.6f}")

        it_sa += 1

    # --------------------------------------------------------------
    # 4 – Return the best probability vector found
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()
    print(f"Finished – best score = {best_score:.6f} (length {n})")
    return best_a.tolist()
```
