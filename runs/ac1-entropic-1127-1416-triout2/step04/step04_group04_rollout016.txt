def project_simplex(v):
    # Euclidean projection onto simplex sum=1, v >= 0
    # Implementation from Duchi et al. (2008)
    # returns projected vector.
    # v shape (n,)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    """Linear convolution of a and b using a real FFT (zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

    """
    Fast evaluation of the objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    a is assumed non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))

    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n>>> Exploring length n = {n}")

        # --------------------------------------------------------------
        # 1. Build a modest library of initial candidates.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # deterministic families
        candidates.append(uniform_shape(n))
        candidates.append(arcsine_initial(n))

        # grid of power‑law shapes
        for p in np.linspace(0.35, 0.65, 7):
            eps = rng.uniform(0.0, max(5.0, 0.015 * n))
            candidates.append(powerlaw_shape(n, p, eps))

        # grid of beta‑like shapes
        for a in np.linspace(0.3, 3.0, 6):
            candidates.append(beta_shape(n, a))

        # mixtures of arcsine with a uniform component
        for _ in range(5):
            w_mix = rng.uniform(0.0, 0.5)
            mix = (1.0 - w_mix) * arcsine_initial(n) + w_mix * uniform_shape(n)
            candidates.append(normalize(mix))

        # a handful of random Dirichlet draws
        for _ in range(5):
            candidates.append(random_dirichlet(n, rng))

        # --------------------------------------------------------------
        # 2. Quick ranking with the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = min(6, len(scored))
        top_candidates = [c for _, c in scored[:top_k]]
        print(
            "  top initial scores:",
            " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]),
        )

        # --------------------------------------------------------------
        # 3. Refine each retained candidate with smoothed mirror‑descent.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            start_sc = compute_score(cand)
            print(
                f"  candidate {idx + 1}/{len(top_candidates)} start_score = {start_sc:.6f}"
            )

            remaining = max(0.0, deadline - time.time())
            # empirical rule: ≈15 iterations per second for n≈4000
            max_iter = int(min(120_000, max(5_000, remaining * 15)))

            refined, _ = refine_candidate(
                a=cand,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.8,
                gamma0=0.5,
                gamma_max=8.0,
                report_every=0,
            )

            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 4. Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(250_000, max(20_000, remaining * 15)))
        polished, _ = refine_candidate(
            a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.6,
            gamma0=0.5,
            gamma_max=10.0,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
