"""
Propose a step‑height sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation**
   * If a global ``height_sequence_1`` exists it is used, otherwise a
     uniform vector of length 5000 is taken.
   * The vector is normalised to a probability vector (sum == 1).

2. **Smooth‑max projected gradient descent**
   * We minimise a smooth approximation of the quantity
        max_i (a * a)_i                (self‑convolution)
     using the log‑sum‑exp (soft‑max) with a temperature μ.
     For a fixed μ the objective is  

        F_μ(a) = (1/μ)·log Σ_i exp( μ·conv_i ),

     whose gradient can be written in closed form and is evaluated
     efficiently with two FFT‑based convolutions.
   * After each gradient step the iterate is projected onto the
     probability simplex (non‑negative, sum = 1).  The projection is
     performed with the O(n log n) algorithm of Duchi et al.
   * μ is increased linearly from 10 to 200 during the run, turning the
     smooth objective into the true max as the optimisation proceeds.
   * A simple learning‑rate schedule  η_t = η₀ / sqrt(t) together with
     momentum (β=0.9) is used.

3. **Best‑ever tracking**
   * The exact score 2·n·max(conv) (which equals the value returned by
     ``evaluate_sequence`` when the vector sums to one) is computed at
     every iteration.  The best vector seen so far is stored.

4. **Return**
   * The best vector (still normalised to sum = 1) is returned as a list
     of floats.  All coefficients are non‑negative, therefore the
     security checks in ``evaluate_sequence`` never reject the answer.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all components are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_exact_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum(a)=1):
       2·n·max(a*a) where * denotes linear convolution."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def softmax_weights(conv: np.ndarray, mu: float) -> np.ndarray:
    """Return soft‑max weights w_i = exp(μ·conv_i)/Σ_j exp(μ·conv_j)."""
    # stabilise the exponential
    scaled = mu * conv
    max_scaled = scaled.max()
    exp_shifted = np.exp(scaled - max_scaled)
    return exp_shifted / exp_shifted.sum()


def gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max objective w.r.t. a.
    Implements g = 2·(a_rev * w)[n‑1 : n‑1+n] using FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]                           # reversed copy
    L = n + (2 * n - 1) - 1                    # 3·n‑2, length of full convolution
    # FFT of the two signals (real → complex)
    fft_ar = np.fft.rfft(a_rev, n=L)
    fft_w = np.fft.rfft(w, n=L)
    # convolution in time domain
    conv_full = np.fft.irfft(fft_ar * fft_w, n=L)
    # slice the needed segment (indices n‑1 … n‑1+n‑1)
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]
    return grad


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    A fallback sub‑gradient of the non‑smooth max‑convolution.
    Used only in the very last few seconds if time permits.
    """
    n = a.shape[0]
    max_val = conv.max()
    eps = max_val * 1e-8
    active = np.where(conv >= max_val - eps)[0]

    if active.size == 0:
        return 2.0 * a

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad
