"""
Propose a low‑discrepancy step sequence.

The search builds on the previous “Adam + sub‑gradient + SA” pipeline,
but adds a **linear‑programming polishing** step (the method described in
the original paper).  For a fixed vector `f` we solve

    maximize   Σ_j g_j
    subject to  g ≥ 0,
                (f * g)[k] ≤ M   ∀k,

where `M = max(f*f)` and `*` is the linear convolution.
The LP pushes mass onto coordinates that do **not** increase the
convolution bound, thereby lowering the maximal self‑overlap.
After a few LP updates we finish with a short Adam refinement.

The algorithm explores several lengths (up to 4096), generates a rich
pool of shapes (uniform, arcsine, beta‑family, power‑law, mixtures,
random Dirichlet), refines each promising seed with Adam, exact
sub‑gradient descent and simulated‑annealing, then applies the LP
polish.  The best candidate found (according to the fast surrogate
`2·n·max(conv)`) is finally evaluated with the official
`evaluate_sequence` and returned.

All steps respect the global 1000 s budget and fall back to a uniform
vector if nothing else succeeds.
"""

import time
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The sequence is assumed to be on the simplex Σa = 1,
    therefore the denominator of the official objective disappears
    and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for the initial pool)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (U‑shaped)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1)[::-1].astype(float)
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, w: float) -> np.ndarray:
    """Mixture w·arcsine + (1‑w)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = w * arc + (1.0 - w) * uni
    mix /= mix.sum()
    return mix


def random_dirichlet_shape(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet distribution with concentration `conc`."""
    x = rng.gamma(conc, 1.0, size=n)
    x_sum = x.sum()
    if x_sum == 0:
        return np.full(n, 1.0 / n)
    return x / x_sum

# Optimisers (smooth Adam, exact sub‑gradient, simulated annealing)

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max).  The simplex constraint is handled
    later by a projection step.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                      # length 2·n‑1
    max_c = conv.max()
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / exp_term.sum()                     # soft‑max weights

    # weighted sub‑gradient = 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)              # length n + (2n‑1) - 1 = 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 120_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate (soft‑max of the convolution).
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Exact sub‑gradient descent for f(a)=2·n·max(conv(a,a)).
    Step size ηₜ = step0 / sqrt(t).  Projection onto the simplex keeps Σa=1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()

        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search on the fast surrogate.
    Randomly moves a small fraction of mass between two coordinates.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def solve_lp_max_sum(f: np.ndarray, M: float) -> Optional[np.ndarray]:
    """
    Solve   maximize Σ g_j   s.t.   g ≥ 0,   (f * g)[k] ≤ M  ∀k .
    Returns the optimal g (not normalised) or None on failure.
    """
    if not _LP_AVAILABLE:
        return None

    n = f.shape[0]
    L = 2 * n - 1

    # Build sparse constraint matrix A (L × n)
    rows = []
    cols = []
    data = []

    # For each column i the coefficient f[i] appears in rows k = i … i+n‑1
    for i in range(n):
        fi = f[i]
        if fi == 0.0:
            continue
        ks = np.arange(i, i + n, dtype=np.int64)  # L‑range for this i
        rows.extend(ks.tolist())
        cols.extend([i] * n)
        data.extend([fi] * n)

    A = sp.csr_matrix((data, (rows, cols)), shape=(L, n))
    b = np.full(L, M, dtype=float)
    c = -np.ones(n, dtype=float)          # maximise sum(g) ↔ minimise –sum(g)

    bounds = [(0.0, None)] * n

    res = scipy.optimize.linprog(c,
                                 A_ub=A,
                                 b_ub=b,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    g[g < 0] = 0.0
    if g.sum() == 0:
        return None
    return g


def lp_polish(f: np.ndarray,
              deadline: float,
              max_iters: int = 4,
              t: float = 0.3) -> np.ndarray:
    """
    Apply a few LP‑updates to `f`.  Each iteration solves the LP described
    above, rescales the solution to Σg = 1 and mixes it with the current
    vector:  f ← (1‑t)·f + t·g .
    """
    if not _LP_AVAILABLE:
        return f

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv_f = convolution_fft(f, f)
        M = conv_f.max()

        g_raw = solve_lp_max_sum(f, M)
        if g_raw is None:
            break

        # Normalise g to the simplex
        g = g_raw / g_raw.sum()

        # Linear blend
        f = (1.0 - t) * f + t * g
        f = project_to_simplex(f)

        cur_sc = compute_score(f)
        print(f"    LP polish iter={it+1}/{max_iters} score={cur_sc:.6f}")

    return f

            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

        # random Dirichlet (various concentrations)
        for _ in range(100):
            conc = rng.uniform(0.05, 2.0)
            cand = random_dirichlet_shape(n, conc, rng)
            pool.append((compute_score(cand), cand))

        # keep the most promising seeds (few of them)
        pool.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in pool[:6]]

        # ------- 1b. Refine each seed -------
        for seed_vec in top_seeds:
            if time.time() >= deadline - 5.0:
                break

            # Phase A – smooth Adam optimisation
            remaining = deadline - time.time()
            max_iter_adam = int(min(150_000, remaining * 25))
            print(f"  Adam smooth (max_iter={max_iter_adam}) …")
            vec, _ = adam_optimize_smooth(
                initial_a=seed_vec,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=2.0,
                lam_end=80.0,
                noise_every=2500,
                noise_scale=3e-5,
            )

            # Phase B – exact sub‑gradient descent
            if time.time() < deadline - 3.0:
                remaining = deadline - time.time()
                max_iter_sub = int(min(100_000, remaining * 25))
                print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
                vec = subgradient_descent_exact(
                    vec,
                    deadline,
                    max_iter=max_iter_sub,
                    step0=0.04,
                )

            # Phase C – simulated‑annealing
            if time.time() < deadline - 3.0:
                remaining = deadline - time.time()
                max_iter_sa = int(min(120_000, remaining * 25))
                print(f"  Simulated annealing (max_iter={max_iter_sa}) …")
                vec = simulated_annealing(
                    vec,
                    rng,
                    deadline,
                    max_iters=max_iter_sa,
                    init_temp=0.02,
                    decay=0.9999,
                )

            # Phase D – LP polishing (only for moderate lengths)
            if n <= 2048 and time.time() < deadline - 2.0:
                print("  LP polishing …")
                vec = lp_polish(vec, deadline, max_iters=3, t=0.35)

            # Final official evaluation
            try:
                val = evaluate_sequence(vec.tolist())
                print(f"  → official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = vec.tolist()
                    print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 2.  Global polishing of the overall best vector (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        # allocate remaining time roughly half/half between sub‑gradient and Adam
        it_sub = int(min(150_000, remaining * 20))
        it_adam = int(min(150_000, remaining * 20))

        print("\nFinal global polishing (sub‑gradient → Adam → LP) …")
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.02)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=it_adam,
            lr0=0.1,
            lam_start=2.0,
            lam_end=80.0,
            noise_every=2000,
            noise_scale=2e-5,
        )
        if a.shape[0] <= 2048:
            a = lp_polish(a, deadline, max_iters=4, t=0.3)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty answer
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
