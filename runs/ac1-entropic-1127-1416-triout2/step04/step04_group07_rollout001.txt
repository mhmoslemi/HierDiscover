def local_search(a_init, deadline):
    a = a_init.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # Starting temperature T related to improvements (maybe 1e-4)
    T = 0.1  # depends
    it = 0
    while time.time() < deadline:
        it += 1
        # step size: decreasing
        step_factor = max(0.001, 0.5 * (deadline - time.time()) / (deadline - start))
        # choose i weighted by a (larger likely to reduce)
        i = np.random.choice(len(a), p=a)
        # choose j (different)
        j = np.random.randint(len(a)-1)
        if j >= i:
            j += 1
        # delta up to step_factor * a[i]
        max_delta = min(a[i], step_factor)
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non-negative (some rounding)
        # evaluate
        score_new = evaluate_sequence(a_new.tolist())
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"Local best iter {it}: {best_score:.6f}")
        # acceptance criterion
        cur_score = evaluate_sequence(a.tolist())
        if score_new < cur_score or np.random.rand() < np.exp(-(score_new - cur_score) / max(T, 1e-12)):
            a = a_new
        # anneal temperature gradually
        if it % 1000 == 0:
            T *= 0.99
    return best_a, best_score

"""
Propose a low‐score step function for the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm combines three phases:

1️⃣  A few stochastic restarts of a smooth log‑sum‑exp surrogate
    optimisation on the simplex (projected gradient descent).

2️⃣  A short Adam run that uses the exact sub‑gradient of the true
    max‑convolution (still on the simplex).

3️⃣  A pure‑Python simulated‑annealing / coordinate‑descent phase that
    updates the convolution incrementally (O(n) vectorised updates) and
    refines the best vector found in the first two phases.

All phases respect the non‑negative simplex constraint,
so the returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Full aperiodic convolution a*a (length 2·n‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of the weight vector w (length 2·n‑1) with a.
    Returns the n‑long segment corresponding to shifts 0…n‑1.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    n = a.shape[0]
    return corr[n - 1: n - 1 + n]


# ----------------------------------------------------------------------
# Phase‑1 – smooth surrogate optimisation
# ----------------------------------------------------------------------
def _smooth_optimize(init_a: np.ndarray, deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    a = init_a.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())  # next power of two
    best_a = a.copy()
    conv = _full_conv(a, L)
    best_score = 2.0 * n * conv.max()          # exact score (sum(a)=1)

    # annealing schedule
    alpha = 0.5
    alpha_mul = 1.15
    inc_every = 300
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1
        conv = _full_conv(a, L)                   # full convolution
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)   # safe soft‑max
        w = exp_scaled / exp_scaled.sum()          # Σ w = 1

        corr = _corr_w_a(w, a, L)                 # length n
        grad = 4.0 * n * corr                     # gradient of surrogate

        # simple diminishing step size
        lr = 0.7 / (1.0 + it * 0.001)
        a -= lr * grad
        a = _project_simplex(a)

        # exact objective (no smoothing)
        conv = _full_conv(a, L)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"Smooth  iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # occasional jitter when stuck
        if stagn > 4000:
            a += 1e-3 * rng.standard_normal(n)
            a = _project_simplex(a)
            stagn = 0

        if it % inc_every == 0:
            alpha = min(alpha * alpha_mul, 2000.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 – Adam on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max(conv), sub‑gradient) for a·a."""
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * a.shape[0] - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


def _adam_optimize(init_a: np.ndarray, deadline: float,
                   restart_id: int, target: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on the exact (non‑smoothed) objective."""
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # logits (log‑softmax) – tiny epsilon avoids log(0)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # initial evaluation
    a0 = np.exp(b - b.max())
    a0 /= a0.sum()
    max_val, _ = _maxconv_and_subgrad(a0, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a0.copy()
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1
        # probabilities from logits
        a = np.exp(b - b.max())
        a /= a.sum()

        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"Adam {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagn += 1

        # early stop if target reached
        if best_score < target:
            break
        if stagn > 8000:
            b += 0.01 * np.random.randn(n)   # jitter in log‑space
            stagn = 0

        # Adam update (gradient of the true objective)
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – fast local refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def _local_refine(a_start: np.ndarray, conv_start: np.ndarray,
                  deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Incremental coordinate updates that preserve the simplex.
    Uses the exact convolution (updated in O(n) via vectorised slices)
    and a simple annealing schedule.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = conv_start.copy()            # length 2·n‑1
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    cur_score = best_score

    # temperature schedule (starts a little aggressive)
    T = 0.02
    it = 0
    start = time.time()

    while time.time() < deadline:
        it += 1
        # pick a source index proportionally to its current weight
        if a.sum() == 0:
            break
        i = rng.choice(n, p=a)
        if a[i] == 0.0:
            continue
        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # step size slowly shrinks as we approach the deadline
        time_frac = (deadline - time.time()) / (deadline - start)
        step_scale = max(0.001, 0.05 * time_frac)

        max_delta = a[i]
        delta = rng.random() * max_delta * step_scale
        if delta <= 0.0:
            continue

        # Incremental convolution update (vectorised)
        conv_new = conv.copy()
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a
        ii = 2 * i
        if ii < conv_new.size:
            conv_new[ii] += delta * delta
        jj = 2 * j
        if jj < conv_new.size:
            conv_new[jj] += delta * delta
        ij = i + j
        conv_new[ij] -= 2.0 * delta * delta

        # new probabilities (simple transfer)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = 2.0 * n * conv_new.max()

        # Acceptance test
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"Local  iter {it:6d}: best = {best_score:.6f}")

        # Anneal temperature every few hundred iterations
        if it % 2000 == 0:
            T *= 0.95

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function within ``budget_s`` seconds.
    The routine runs several gradient‑descent restarts, a short Adam
    polishing phase and finishes with a fast incremental local search.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(123456)               # deterministic RNG
    deadline = time.time() + budget_s - 0.5            # tiny safety margin

    # ------------------------------------------------------------------
    # Determine problem size (use previous solution if available)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = prev.shape[0]
        prev = np.maximum(prev, 0.0)
        if prev.sum() == 0:
            prev = np.full(n, 1.0 / n)
        else:
            prev = prev / prev.sum()
    except Exception:
        # fallback size – a few thousand entries give a good trade‑off
        n = 5146
        prev = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_overall_a = prev.copy()
    best_overall_score = 2.0 * n * _full_conv(prev, 1 << ((2 * n - 2).bit_length())).max()
    restart_id = 0
    strategies = ["prev", "uniform", "random_dirichlet", "geometric"]

    # ------------------------------------------------------------------
    # Stochastic restarts of the two‑stage optimiser
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:   # leave ~30 s for the final refinement
        kind = strategies[restart_id % len(strategies)]

        # ---- generate a feasible start point ---------------------------------
        if kind == "prev":
            a0 = prev.copy()
        elif kind == "uniform":
            a0 = np.full(n, 1.0 / n)
        elif kind == "random_dirichlet":
            # α = 0.5 gives a fairly sparse start
            sample = rng.gamma(0.5, 1.0, size=n)
            a0 = sample / sample.sum()
        elif kind == "geometric":
            r = 0.98 + 0.02 * rng.random()
            left = r ** np.arange(n)
            right = r ** np.arange(n - 1, -1, -1)
            a0 = left + right
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)

        # ---- allocate a slice of the remaining budget for this restart -----
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(40.0, 0.45 * remaining)

        # ---- Phase 1 – smooth surrogate optimisation ------------------------
        a_smooth, score_smooth = _smooth_optimize(a0, slice_deadline, rng)

        # ---- Phase 2 – Adam polishing ---------------------------------------
        if time.time() < deadline - 5.0:
            a_fine, score_fine = _adam_optimize(
                a_smooth, deadline, restart_id, target=1.503
            )
        else:
            a_fine, score_fine = a_smooth, score_smooth

        # ---- Global best update ---------------------------------------------
        if score_fine < best_overall_score:
            best_overall_score = score_fine
            best_overall_a = a_fine.copy()
            print(f"Global best after restart {restart_id:02d}: {best_overall_score:.6f}")

        if best_overall_score < 1.503:
            print(f"Target reached (score={best_overall_score:.6f}) after {restart_id+1} restarts.")
            break

        restart_id += 1

    # ------------------------------------------------------------------
    # Phase 3 – fast local refinement (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        # start with the best vector so far
        conv_best = _full_conv(best_overall_a, 1 << ((2 * n - 2).bit_length()))
        ref_deadline = deadline
        refined_a, refined_score = _local_refine(best_overall_a, conv_best,
                                                ref_deadline, rng)
        if refined_score < best_overall_score:
            best_overall_score = refined_score
            best_overall_a = refined_a.copy()
            print(f"After local refinement: score = {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best_overall_a.tolist()]
