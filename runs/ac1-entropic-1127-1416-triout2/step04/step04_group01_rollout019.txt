"""
LP‑plus‑gradient search for a low‑max‑convolution step sequence.

1.  Initialise a modest‑size distribution (n≈1 k) with the classic
    arcsine shape – already known to give a good bound.
2.  Repeatedly solve a *linear* programme:
        maximise   Σ g_i
        subject to  g ≥ 0  and  f∗g ≤ M,
    where M = max(f∗f) for the current f.
    Since the convolution is linear in g the constraints are a set of
    ordinary linear inequalities.  The optimum g pushes mass into
    regions that do not increase the current worst convolution value.
3.  Mix the new g into the current f (convex combination) and
    renormalise – the theory guarantees a non‑increasing sequence of
    M‑values.
4.  Upsample the best f to a larger length (≈4 k) and finish with the
    smooth‑Adam, exact sub‑gradient and simulated‑annealing refinements
    used in the reference code.
5.  The best vector found is finally evaluated with the official
    `evaluate_sequence` and returned.

The whole routine respects the 1000 s wall‑clock budget and falls back
to a uniform vector if nothing else succeeds.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(n log n))."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    After projection onto the simplex Σa = 1 the denominator disappears,
    therefore the score is simply 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of the soft‑max surrogate of max(conv(a,a))."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_c = np.max(conv)

    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the smooth surrogate."""
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """Exact sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """Simple Metropolis annealing on the fast surrogate."""
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = compute_score(cand)
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc
        temp = max(temp * decay, 1e-8)
    return best

def build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A where (A @ g)[k] = Σ_i f_i g_{k-i}
    for the linear constraints f*g ≤ M.
    The matrix has shape (2·n‑1, n).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(f: np.ndarray,
              deadline: float,
              mix: float = 0.2,
              max_iters: int = 12) -> Tuple[np.ndarray, List[float]]:
    """
    Perform a few LP‑updates on ``f``.
    Returns the refined vector and a list of fast scores observed.
    """
    scores = []
    try:
        import cvxpy as cp
    except Exception as e:
        print("cvxpy not available, skipping LP phase:", e)
        return f, scores

    n = f.size
    for it in range(max_iters):
        if time.time() >= deadline - 2.0:
            break
        M = float(convolution_fft(f, f).max())
        A = build_constraint_matrix(f)               # (2n‑1) × n
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2 * n - 1)]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception as ee:
            print(f"LP solve failed at iter {it+1}:", ee)
            break
        if g.value is None:
            print("LP did not return a solution")
            break
        g0 = np.array(g.value).ravel()
        s = g0.sum()
        if s <= 0:
            print("LP produced zero‑mass solution")
            break
        g_norm = g0 / s                     # renormalise to Σ=1
        # convex combination with the old iterate
        f = (1.0 - mix) * f + mix * g_norm
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        sc = compute_score(f)
        scores.append(sc)
        print(f"LP iteration {it+1:2d}: fast score = {sc:.6f}")
    return f, scores
