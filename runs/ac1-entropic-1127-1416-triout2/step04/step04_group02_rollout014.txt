def solve_lp_direction(a: np.ndarray, M: float, solver: str = "ECOS") -> np.ndarray:
    n = len(a)
    a_param = cp.Parameter(n, nonneg=True)
    a_param.value = a
    b = cp.Variable(n, nonneg=True)
    conv = cp.conv(b, a_param)  # length 2n-1
    constraints = [conv <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=solver, warm_start=True, max_iters=5000)
    if b.value is None:
        return a.copy()
    return np.maximum(0.0, np.array(b.value).astype(float))

def line_search_mix(a: np.ndarray, g: np.ndarray, steps: List[float]) -> Tuple[np.ndarray, float]:

"""
Hybrid LP‑Frank‑Wolfe + exponentiated‑gradient search for the
min‑max self‑convolution problem.

The algorithm works in three stages:

1. Build a diverse library of initial shapes (uniform, arcsine,
   symmetric Beta, power‑law, Dirichlet draws, small perturbations).

2. For each promising start run a few Frank‑Wolfe iterations.
   In every iteration we solve the linear program

        max   Σ b_j
        s.t.  (a * b)_k ≤ M   for all k,
              b ≥ 0,

   where ``M = max_k (a*a)_k``.
   The LP is solved with CVXPY (``cp.conv`` provides the convolution
   expression).  The optimal vector `b` is renormalised to a probability
   distribution `g = b / Σb` and we perform a short line‑search over a
   handful of mixing coefficients `t` and keep the best mixture
   ``a←(1−t)·a + t·g``.  This step is cheap (a few hundred ms) and
   quickly flattens the convolution.

3. The best sequence found so far is refined with a long
   exponentiated‑gradient (mirror‑descent) sub‑gradient method on the
   true non‑smooth objective
        Φ(a) = 2·n·max_k (a*a)_k / (Σa)² .
   The update
        a ← a · exp(−η_t·∇Φ)
   guarantees non‑negativity; after each step we project onto the
   simplex.  A 1/√t step‑size schedule, occasional tiny Gaussian noise
   and a final polishing run provide the final answer.

The routine respects a wall‑clock budget (default 1000 s) and always
returns a feasible list of non‑negative floats.
"""

import time
import math
import numpy as np

# Try to import cvxpy – the LP part is optional.
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official (scale‑invariant) objective:
        2·n·max(a*a) / (sum(a)²)
    a is assumed non‑negative; any shape works.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (symmetrised, normalised to sum=1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """U‑shaped power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like draw (Gamma / exponential, then normalise)."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.shape[0])
    return b / s


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exponentiated gradient on the true max‑conv)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For every active offset k (where the max is attained) we add
        g_j = 2·a_{k‑j}   (if the index is valid)
    and then average over all active offsets.
    """
    n = len(a)
    max_val = conv.max()
    # mask of offsets achieving the maximal value
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the exact objective.
    Returns the best vector seen and its score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()               # stay on the simplex
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / math.sqrt(it)          # 1/√t schedule
        a = a * np.exp(-eta * grad)         # multiplicative update

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # project back onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming sub‑problem (Frank‑Wolfe direction)
# ----------------------------------------------------------------------


def solve_lp_direction(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ b_j   s.t.   (a * b)_k ≤ M   ∀k,   b ≥ 0.
    Returns the optimal b (as a NumPy array).  If the LP cannot be solved,
    the current vector ``a`` is returned unchanged.
    """
    if not _CVXPY_AVAILABLE:
        # Fall back to a no‑op.
        return a.copy()

    n = len(a)
    # Parameter for the fixed vector a (constant during the solve)
    a_param = cp.Parameter(n, nonneg=True)
    a_param.value = a

    b = cp.Variable(n, nonneg=True)

    # Linear convolution a*b (length 2n‑1).  cvxpy's ``conv`` is linear in b.
    conv_expr = cp.conv(b, a_param)            # shape (2n‑1,)

    constraints = [conv_expr <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:  # pragma: no cover
        return a.copy()

    if b.value is None:
        return a.copy()
    # Numerical safety – clip tiny negatives.
    b_opt = np.maximum(0.0, np.asarray(b.value, dtype=float))
    return b_opt


def line_search_mix(a: np.ndarray, g: np.ndarray, steps: list[float]) -> tuple[np.ndarray, float]:
    """
    Perform a tiny line‑search among a list of mixing coefficients.
    Returns the best mixture and its score.
    """
    best_a = a
    best_sc = compute_score(a)
    for t in steps:
        a_new = (1.0 - t) * a + t * g
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new
    return best_a, best_sc


def frank_wolfe_optimize(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 15,
    step_candidates: list[float] | None = None,
) -> np.ndarray:
    """
    Frank‑Wolfe / LP based improvement.
    Starting from ``initial_a`` we repeatedly:
        * compute M = max (a*a)
        * solve the LP that maximises Σb under a*b ≤ M
        * normalise b → g, mix a and g (tiny line‑search)
    The process stops after ``max_iter`` iterations or when no
    improvement is observed.
    """
    if step_candidates is None:
        step_candidates = [0.05, 0.1, 0.2, 0.3, 0.5]

    a = np.maximum(initial_a, 0.0)
    a = a / a.sum()                     # work on the simplex
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iter):
        if time.time() >= deadline - 2.0:
            break

        conv = convolution_fft(a, a)
        M = float(conv.max())

        b = solve_lp_direction(a, M)
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # No direction that can increase the total mass → stop.
            break

        g = b / sum_b                     # a feasible direction with Σg = 1
        a_new, sc_new = line_search_mix(a, g, step_candidates)

        if sc_new + 1e-12 < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # No improvement – terminate early.
            break

    return best_a
