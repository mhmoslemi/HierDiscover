def arcsine_bin(n: int) -> np.ndarray:
    # Compute probabilities for bins of length 1/n of arcsine density.
    i = np.arange(n)
    left = i / n
    right = (i + 1) / n
    vals = (2.0 / np.pi) * (np.arcsin(np.sqrt(right)) - np.arcsin(np.sqrt(left)))
    # Numerical issues near 0 or 1 produce zeros; fine.
    return vals
```

"""
Propose a step‑function (a non‑negative vector) that minimises

    2·n·max(a*a) / (Σa)²

where ``a*a`` denotes the linear (non‑circular) convolution.
The known optimal continuous solution is the arcsine distribution,
because the sum of two independent arcsine variables is exactly uniform.
We discretise this distribution *by integrating the density on each bin*,
which yields a far more accurate approximation than using the
mid‑point density (the previous code’s “arcsine_initial”).  For moderate
sizes (n≈2 k … 16 k) the resulting max‑convolution is already close to the
theoretical lower bound 1 (the value 2·n·avg(conv)=1).  The algorithm:

1.  Scan a handful of lengths and build the exact “bin‑integrated” arcsine
    vector.
2.  Keep the best vector according to a fast FFT‑based surrogate
    ``2·n·max(conv)``.
3.  If time permits, polish the best candidate with three stages:
        – Adam on a smooth log‑sum‑exp surrogate of the max,
        – simple projected sub‑gradient descent on the true max,
        – stochastic Gaussian local search.
    All iterates are projected onto the probability simplex
    (Σa = 1, a ≥ 0).

The whole search respects the 1000 s budget and always returns a valid
non‑empty sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    # trivial case: all non‑positive → uniform distribution
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)) for a already normalised
    (Σa = 1).  This equals the official evaluation value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Exact arcsine‑distribution discretisation (bin integration)
# ----------------------------------------------------------------------


def arcsine_bin(n: int) -> np.ndarray:
    """
    Return the probability vector obtained by integrating the arcsine
    density (Beta(0.5, 0.5)) over the n equal sub‑intervals of [0, 1].
    The result is already normalised, but we project onto the simplex
    to guard against round‑off.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    # clamp arguments of sqrt for safety
    left = np.clip(left, 0.0, 1.0)
    right = np.clip(right, 0.0, 1.0)
    vals = (2.0 / np.pi) * (np.arcsin(np.sqrt(right)) - np.arcsin(np.sqrt(left)))
    # numerical noise may produce tiny negatives; clip and re‑project
    vals = np.maximum(vals, 0.0)
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights, sum to 1
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of max(conv)

    # weighted sum of sub‑gradients:
    # ∂ Σ_k w_k (a*a)_k / ∂a = 2 * (a * w_rev) evaluated at appropriate indices
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    noise_every: int = 1500,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature (square‑root schedule)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        # occasional tiny isotropic noise to escape plateaus
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   max(conv(a,a)).
    The step size decays like step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex: add Gaussian noise, project back,
    keep improvements (tiny probability of uphill moves for escape).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # decaying step size
        step = init_step / np.sqrt(it)

        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept if improvement, otherwise with a tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"    Random local iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine builds the exact discrete arcsine distribution for
    several lengths, keeps the best surrogate, and then refines it with
    Adam → sub‑gradient → random local search.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Scan a few lengths using the exact arcsine‑bin discretisation
    # ------------------------------------------------------------------
    # Choose lengths that still allow a final official ``np.convolve`` call.
    # Up to ~12000 is safe on typical hardware.
    lengths = [512, 1024, 2048, 4096, 8192, 12288]

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Testing length n = {n} ===")
        a = arcsine_bin(n)                     # already normalised
        sc = compute_score(a)
        print(f"  Fast surrogate score = {sc:.6f}")

        # keep the best according to the fast surrogate
        if sc < best_val:
            best_val = sc
            best_seq = a.tolist()
            print(f"  *** New best (surrogate) with n={n}: {sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  If we have time, polish the best candidate with a multi‑stage
    #     optimiser.
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Polishing the global best ===")
        a = np.array(best_seq, dtype=float)

        # ---- Stage A – Adam on smooth surrogate ----
        remaining = deadline - time.time()
        max_iter_adam = int(min(600_000, remaining * 30))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.15,
            lam_start=1.0,
            lam_end=120.0,
            noise_every=1000,
            noise_scale=4e-5,
        )

        # ---- Stage B – projected sub‑gradient descent ----
        if time.time() < deadline - 3.0:
            max_iter_sub = int(min(150_000, (deadline - time.time()) * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.2,
                max_iter=max_iter_sub,
            )

        # ---- Stage C – stochastic local walk ----
        if time.time() < deadline - 2.0:
            max_iter_loc = int(min(250_000, (deadline - time.time()) * 40))
            print(f"  Random local polishing (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.015,
                max_iter=max_iter_loc,
            )

        # final official evaluation
        final_val = evaluate_sequence(a.tolist())
        print(f"Final official evaluation = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a uniform vector of moderate size
        best_seq = [1.0] * 256
        best_seq = project_to_simplex(np.array(best_seq)).tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence official score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
