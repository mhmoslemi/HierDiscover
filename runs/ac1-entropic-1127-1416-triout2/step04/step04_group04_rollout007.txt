  for it in range(1, max_iter+1):
      if time.time()>= deadline:
         break
      # linear interpolation of gamma
      if gamma_step is None:
           # gradually increase as iteration fraction
           gamma = gamma_start + (gamma_end - gamma_start) * (it-1)/(max_iter-1)
      else:
           gamma = gamma_start + (gamma_end - gamma_start) * (it-1)/(max_iter-1)
      a = np.exp(z)
      a = a / np.sum(a) # ensures sum=1
      conv = convolution_fft(a, a)
      # compute smoothed weights
      w = np.exp(gamma * (conv - np.max(conv))) # subtract max for stability
      w_sum = np.sum(w)
      if w_sum == 0:
          w = np.full_like(conv, 1.0/len(conv))
      else:
          w = w / w_sum
      # compute gradient wrt a of smoothed objective: grad_a = 2 * (conv of a with w_rev)
      grad_a = gradient_from_weights(a, w)  # returns shape n
      # compute grad_z using chain rule: grad_z = (grad_a - np.dot(a, grad_a)) * a

      # Adam update
      m = beta1*m + (1-beta1)*grad_z
      v = beta2*v + (1-beta2)*(grad_z**2)
      m_hat = m / (1 - beta1**it)
      v_hat = v / (1 - beta2**it)
      # step size adaptation: choose base learning rate; maybe 0.1?

      if it % 200 == 0 or it == max_iter:

          # track best

              best_score = val; best_a = a.copy()
              # we might keep best_z as well? not needed.

        lr = base_lr / np.sqrt(t)
        a = np.exp(z)

        # compute conv and gradient

        # compute smoothing param gamma

        else:

        # gradient wrt a
        grad_a = gradient_from_weights(a, w)   # returns shape n
        # chain rule to get grad wrt z
        dot = np.dot(a, grad_a)
        grad_z = (grad_a - dot) * a
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # update z
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasionally evaluate exact
        if t % 200 == 0 or t == max_iter:
            a_eval = np.exp(z)
            a_eval = a_eval / np.sum(a_eval)
            val = evaluate_sequence(a_eval.tolist())
            if val < best_score:
                best_score = val
                best_a = a_eval.copy()
    return best_a, best_score

"""
Search for a non‑negative step‑function that minimizes the official
objective 2·n·max(conv(a,a)) / (sum(a)^2).

The strategy
-------------
1.  Build a modest library of parametric families (arcsine, power‑law,
    symmetric beta, uniform, random Dirichlet and mixtures).  Each
    candidate is normalised to sum‑to‑one (the objective is
    scale‑invariant).

2.  Keep the few best candidates (according to a fast FFT based score).

3.  Refine each candidate with a *smooth* version of the objective:
    fγ(a) = (1/γ)·log Σ_i exp(γ·(a∗a)_i).  When γ is large the
    function approaches the true max, but for small γ it is smooth
    and has a well‑behaved gradient.  The gradient w.r.t. a is
        ∇_a fγ = 2·conv(a , w) ,
    where w is the soft‑max of the convolution values.

    Optimisation is performed on the *log‑probability* variables
    z_i = log a_i (so that a = softmax(z) automatically satisfies a_i≥0
    and Σa_i = 1).  An Adam optimiser is used on z with a linear
    annealing schedule for γ (from 1 up to 200).  This gives fast,
    stable progress and avoids the plateaus typical for sub‑gradient
    methods.

4.  The best refined shape is polished once more with a longer Adam run
    (higher final γ).

All work is timed to stay inside the supplied budget (default 1000 s) and
the routine always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.  The result has length
    len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    This uses the FFT convolution; it is accurate enough for ranking.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k w.r.t. the vector a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via a single FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)            # length = n + len(w) - 1 = 3·n‑2
    idx = 2 * len(a) - 2                         # centre of the convolution result
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine shape ≈ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Soft‑max helpers (smooth approximation of the max)
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Compute a soft‑max (probability) vector over the convolution entries.
    The max is subtracted for numerical stability.
    """
    if gamma <= 0:
        # fallback to a uniform weight vector
        return np.full_like(conv, 1.0 / conv.size)
    max_c = np.max(conv)
    scaled = gamma * (conv - max_c)
    # clipping prevents overflow for large negative values
    scaled = np.clip(scaled, -700, 700)          # np.exp under/overflow guard
    w = np.exp(scaled)
    s = np.sum(w)
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w / s


def a_from_z(z: np.ndarray) -> np.ndarray:
    """Convert log‑probabilities to a probability vector (softmax)."""
    shift = z - np.max(z)        # numerical stability
    exp_z = np.exp(shift)
    return exp_z / np.sum(exp_z)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int,
    gamma_start: float = 1.0,
    gamma_end: float = 200.0,
    base_lr: float = 0.05,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑probability parameters.
    The objective is the smooth max defined by a soft‑max with a
    linearly increasing temperature parameter γ.
    Returns the best vector (as a probability distribution) and its
    *exact* evaluation using ``evaluate_sequence``.
    """
    n = len(a0)
    # initialise log‑probabilities safely (avoid log(0))
    eps = 1e-12
    z = np.log(np.maximum(a0, eps))

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # keep track of the best exact score we have seen
    best_a = a_from_z(z)
    try:
        best_score = evaluate_sequence(best_a.tolist())
    except Exception:                     # safety – should never happen
        best_score = float("inf")

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # learning‑rate schedule (simple 1/√t decay)
        lr = base_lr / np.sqrt(t)

        # current probability vector
        a = a_from_z(z)

        # convolution a∗a
        conv = convolution_fft(a, a)

        # γ increases linearly from start to end
        if max_iter > 1:
            gamma = gamma_start + (gamma_end - gamma_start) * (t - 1) / (max_iter - 1)
        else:
            gamma = gamma_end

        # soft‑max weights for the smoothed max
        w = softmax_weights(conv, gamma)

        # gradient w.r.t. a of the smooth objective
        grad_a = gradient_from_weights(a, w)          # shape (n,)

        # chain rule to obtain gradient w.r.t. z (log‑probs)
        dot = np.dot(a, grad_a)
        grad_z = (grad_a - dot) * a

        # Adam update on z
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation (every 200 steps or at the end)
        if t % 200 == 0 or t == max_iter:
            a_eval = a_from_z(z)
            val = evaluate_sequence(a_eval.tolist())
            if val < best_score:
                best_score = val
                best_a = a_eval.copy()
    return best_a, best_score

    except NameError:

    except Exception:

    scored.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [c for _, c in scored[:top_k]]
    print("  top initial fast scores:",
          " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]))

    # --------------------------------------------------------------
    # 3.  Refine each retained candidate with Adam‑based optimisation.
    # --------------------------------------------------------------
    for idx, cand in enumerate(top_candidates):
        if time.time() >= deadline - 5.0:
            break
        init_sc = compute_score(cand)
        print(f"\n  refining candidate {idx + 1}/{len(top_candidates)} (init score = {init_sc:.6f})")
        remaining = max(0.0, deadline - time.time())
        # empirical budget: roughly 30–40 iterations per second for n≈2000
        max_iter = int(min(80_000, max(5_000, remaining * 35)))
        refined, _ = adam_optimize(
            a0=cand,
            deadline=deadline,
            max_iter=max_iter,
            gamma_start=1.0,
            gamma_end=200.0,
            base_lr=0.06,
        )
        # official evaluation
        val = evaluate_sequence(refined.tolist())
        print(f"    exact score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print(f"    *** New global best: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 4.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(120_000, max(8_000, remaining * 35)))
        polished, _ = adam_optimize(
            a0=a,
            deadline=deadline,
            max_iter=max_iter,
            gamma_start=5.0,
            gamma_end=500.0,
            base_lr=0.04,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished final score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector (should never happen)
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f} (length = {len(best_seq)})")
    return best_seq
