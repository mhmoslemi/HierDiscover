"""
search_step_function_v2.py

Hybrid search for a non‑negative step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases:

1. **Broad parametric sweep** –  For a few increasing lengths
   (512, 1024, 2048, 4096) a collection of simple parametric families
   (uniform, arcsine, triangular, Gaussian, symmetric beta, power‑law,
   mixtures and random Dirichlet draws) is generated.  The cheap FFT‑based
   score `compute_score` is used to keep the few best candidates.

2. **Heavy mirror‑descent refinement** –  The current global best candidate
   is refined with a long run of the exponentiated‑gradient (mirror‑descent)
   sub‑gradient method (`refine_subgradient`).  The whole remaining budget
   (≈ 60 % of the time) is spent on this step.

3. **Smooth‑max polishing** –  Finally the solution is polished with a
   smooth approximation of the “max’’ (soft‑max with a temperature that is
   annealed) while still staying on the probability simplex
   (`refine_smooth`).  The last ≈ 40 % of the time budget is used here.

All helper functions are defined at the top level – no closures or lambdas.
The routine respects the supplied time budget (default 1000 s) and always
returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official (scale‑invariant) objective:
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed to be non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a (the convolution entry weights w sum to 1).
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads

# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/sqrt(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def triangle_shape(n: int) -> np.ndarray:
    """Triangular (tent) distribution – linear rise then fall."""
    i = np.arange(n, dtype=float)
    vals = np.minimum(i + 1, n - i)
    return normalize(vals)


def gaussian_shape(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete centred Gaussian with σ = sigma_factor·n."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return normalize(vals)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return normalize(new_seq)


# ----------------------------------------------------------------------
# Mirror‑descent (hard‑max) refinement
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 500_000,
    eta0: float = 0.8,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the true objective
    ``max(conv(a,a))``.  The iterate stays on the simplex.
    Returns the best vector seen and the corresponding fast score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # binary sub‑gradient: give equal weight to all maximising indices
        w = np.zeros(conv_len, dtype=float)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)          # classic 1/√t schedule
        a = a * np.exp(-eta * grad)       # multiplicative update
        a = normalize(a)                  # project onto simplex

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) refinement – final polishing
# ----------------------------------------------------------------------


def refine_smooth(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 300_000,
    tau_start: float = 0.5,
    tau_end: float = 12.0,
    eta0: float = 0.6,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent using a soft‑max approximation of the max‑function.
    The temperature ``tau`` is annealed from ``tau_start`` to ``tau_end``.
    Returns the best vector seen and its fast score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for the temperature
        tau = tau_start + (tau_end - tau_start) * (it - 1) / (max_iter - 1)

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # soft‑max weights (stable implementation)
        exp_term = np.exp(tau * (conv - max_val))   # all ≤ 1
        w = exp_term / np.sum(exp_term)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    score.  The routine respects the time budget, explores several parametric
    families, refines the best candidate with a long mirror‑descent run and
    finally polishes it with a smooth‑max stage.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Broad parametric sweep over several lengths.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]

    # Collect the best few candidates across all lengths
    global_best_a = None
    global_best_score = float("inf")

    for n in lengths:
        # stop early if we are about to run out of time
        if time.time() >= deadline - 5.0:
            break

        print(f"\n>>> Exploring length n = {n}")

        candidates = []

        # deterministic families
        candidates.append(uniform_shape(n))
        candidates.append(arcsine_initial(n))
        candidates.append(triangle_shape(n))
        candidates.append(gaussian_shape(n, sigma_factor=0.25))

        # beta shapes with a variety of exponents
        for _ in range(4):
            a = rng.uniform(0.15, 1.2)
            candidates.append(beta_shape(n, a))

        # power‑law shapes (randomised)
        for _ in range(4):
            p = rng.uniform(0.30, 0.70)
            eps = rng.uniform(0.0, max(2.0, 0.01 * n))
            candidates.append(powerlaw_shape(n, p, eps))

        # mixtures of arcsine and uniform
        for _ in range(4):
            w = rng.uniform(0.0, 0.4)
            mix = (1.0 - w) * arcsine_initial(n) + w * uniform_shape(n)
            candidates.append(normalize(mix))

        # random Dirichlet draws
        for _ in range(4):
            candidates.append(random_dirichlet(n, rng))

        # up‑sample the current global best (if any) to the new length
        if global_best_a is not None:
            candidates.append(upscale(global_best_a, n))

        # ------------------------------------------------------------------
        # 1b. Fast ranking (FFT‑based score) – keep the top few.
        # ------------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 5
        for sc, _ in scored[:top_k]:
            print(f"   init score = {sc:.6f}")

        # keep the absolute best across all lengths for the heavy refinement
        if scored[0][0] < global_best_score:
            global_best_score = scored[0][0]
            global_best_a = scored[0][1].copy()

    if global_best_a is None:
        # as a final safety net return a uniform vector of modest size
        best_seq = uniform_shape(256).tolist()
        return best_seq

    # ------------------------------------------------------------------
    # 2.  Heavy mirror‑descent refinement on the *single* best candidate.
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    # allocate roughly 60 % of the remaining budget to the hard‑max stage
    sub_deadline = time.time() + 0.60 * remaining

    print("\n--- Heavy mirror‑descent refinement (hard max) ---")
    refined_hard, _ = refine_subgradient(
        a=global_best_a,
        deadline=sub_deadline,
        max_iter=500_000,
        eta0=0.8,
        report_every=0,
    )

    # ------------------------------------------------------------------
    # 3.  Smooth‑max polishing (the last ≈ 40 % of the time budget).
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n--- Smooth‑max polishing ---")
        polished, _ = refine_smooth(
            a=refined_hard,
            deadline=deadline,
            max_iter=300_000,
            tau_start=0.5,
            tau_end=12.0,
            eta0=0.6,
        )
        final_candidate = polished
    else:
        final_candidate = refined_hard

    # ------------------------------------------------------------------
    # 4.  Official evaluation and return.
    # ------------------------------------------------------------------
    final_val = evaluate_sequence(final_candidate.tolist())
    print(f"\nFinal evaluated score = {final_val:.6f}")

    if final_val < best_val:
        best_val = final_val
        best_seq = final_candidate.tolist()

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
