def adam_optimize(a_init, time_deadline, max_iter=200000):
    a = a_init.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5   # may adjust.
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        # schedule beta
        beta = 2.0 + (48.0 * t / max_iter)  # from 2 to 50
        conv = convolution_fft(a, a)
        maxconv = np.max(conv)
        # compute softmax probabilities:
        maxc = maxconv  # same
        exp_beta = np.exp(beta * (conv - maxc))
        probs = exp_beta / np.sum(exp_beta)
        # compute gradient
        a_rev = a[::-1]
        grad_full = convolution_fft(probs, a_rev)
        grad = 4.0 * n * grad_full[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex to maintain sum=1 and nonnegativity
        a = project_to_simplex(a)

        score = 2.0 * n * maxconv  # compute_score from conv
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # maybe reduce lr if we want
    return best_a, best_score
```

"""
Propose a low‑score step function.

Algorithm
---------

1. **Random sparse seeding** – generate many short vectors where a random
   subset of the entries carries equal mass (weights = 1/k).  This yields a
   decent starting point for the later optimisation and provides an initial
   bound.

2. **Smooth‑max Adam optimisation** – minimise the true objective
   `2·n·max(a∗a)` under the simplex constraint `a≥0, Σa = 1`.  The
   nondifferentiable maximum is replaced by a smooth approximation

        Sβ(a) = (1/β)·log Σ exp(β·(a∗a))

   whose gradient can be written analytically (using one extra convolution).
   An Adam optimiser with an annealing schedule for `β` (starting at 2,
   ending at ≈50) drives the vector toward a *uniform* auto‑convolution,
   i.e. a distribution where every shift receives roughly the same amount
   of mass.  After each update the vector is projected onto the probability
   simplex to keep the constraints satisfied.

3. **Final hill‑climb polish** – a cheap stochastic hill‑climber that swaps
   entries or moves a small fraction of mass between positions is run for a
   short while to escape tiny local plateaus.

All work respects the 1000 s budget; the best sequence found is returned as
a plain list of floats which can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolutions, simplex projection, scoring)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact version of `evaluate_sequence` for vectors with Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed maximum of the auto‑convolution
# ----------------------------------------------------------------------


def adam_optimize(
    a_init: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max Adam optimisation.

    Parameters
    ----------
    a_init : np.ndarray
        Starting vector (already projected to the simplex).
    time_deadline : float
        Unix time at which the optimiser must stop.
    max_iter : int
        Hard iteration cap (may stop earlier because of the time limit).
    lr : float
        Base learning rate for Adam.
    Returns
    -------
    best_a : np.ndarray
        Best vector found (projected onto the simplex).
    best_score : float
        Corresponding value of 2·n·max(a∗a).
    """
    a = a_init.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # ---- smooth max -------------------------------------------------
        # β grows from 2 up to about 50 (linear schedule)
        beta = 2.0 + 48.0 * (t / max_iter)

        conv = convolution_fft(a, a)            # a∗a, length 2n‑1
        max_c = np.max(conv)

        # softmax over the convolution entries (numerically stable)
        shifted = conv - max_c
        exp_beta = np.exp(beta * shifted)
        p = exp_beta / np.sum(exp_beta)         # probabilities, sum = 1

        # ---- gradient of the smooth max (4·n·conv(p, a_rev)) ----------
        a_rev = a[::-1]
        grad_full = convolution_fft(p, a_rev)   # length 3n‑2
        grad = grad_full[n - 1 : n - 1 + n]      # central n entries
        grad *= 4.0 * n

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # keep the simplex constraints
        a = project_to_simplex(a)

        # evaluate true (non‑smoothed) score
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny jitter to avoid getting stuck on exact plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (lr * 0.01 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber used for a final polish
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    Works directly with the exact objective.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of the source mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a small fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)  # increase step on success
        else:
            step_scale *= 0.97                     # decrease on failure

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine combines a fast random‑subset pre‑search, a smooth‑max
    Adam optimiser, and a light hill‑climber polish.  It respects the
    1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # a small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback (if a previous good construction is present)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = np.array(fallback)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Random sparse‑subset pre‑search
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    # we will keep generating candidates until we have a few seconds left
    while time.time() < deadline - 30.0:
        n = int(rng.choice(lengths))
        # pick density between 30 % and 80 %
        p = rng.uniform(0.30, 0.80)
        k = max(2, int(round(p * n)))
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k                     # uniform weight on the chosen positions
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"New random‑subset best: n={n}, k={k}, score={cur_val:.6f}")

    if best_seq is None:                     # safety net – never happen, but guards against bugs
        n = 256
        best_seq = np.full(n, 1.0 / n)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 2. Smooth‑max Adam optimisation (multiple restarts if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 15.0:
        # start from the current best, add a little noise to escape possible flat regions
        a0 = best_seq.copy()
        noise = rng.normal(scale=0.02, size=a0.shape)
        a0 += noise
        a0 = np.maximum(a0, 0.0)
        a0 = project_to_simplex(a0)

        # estimate how many iterations we can afford (very rough)
        remaining = deadline - time.time()
        # empirical cost: ≈6 FFTs per iteration → ~0.0007 s for n≈4096.
        # we conservatively allocate an upper bound.
        max_iter = int(min(150_000, remaining / 0.0008))
        if max_iter < 5_000:
            break

        refined_a, refined_score = adam_optimize(
            a0,
            time_deadline=deadline - 5.0,
            max_iter=max_iter,
            lr=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.copy()
            print(f"Adam improvement: score={refined_score:.6f} (iters={max_iter})")

        # if we are already very close to the theoretical lower bound, stop early
        if best_val < 1.10:
            break

    # ------------------------------------------------------------------
    # 3. Final hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\nRunning final hill‑climb polish")
        polished_seq, polished_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=deadline,
            max_steps=250_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = np.array(polished_seq)
            print(f"Hill‑climb polish improved best to {polished_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain list of floats
    # ------------------------------------------------------------------
    final_seq = [float(x) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"\nFinished search – final reported score = {final_score:.6f}")
    return final_seq
