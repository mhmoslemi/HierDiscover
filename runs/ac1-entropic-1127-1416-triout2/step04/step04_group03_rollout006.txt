The problem is to make the self‑convolution as flat as possible – i.e. to

The new search routine consists of three independent ideas that together

   * Keep the best candidate for each length according to the fast surrogate
     ``score = 2·n·max(conv(a,a))`` (the official evaluation after normalising).

   * Starting from the best grid point run a few hundred random perturbations
     of (α,β,ε) (Gaussian steps in log‑space for ε, additive steps for α,β).
   * Accept a perturbation only if it improves the fast surrogate.
   * The best parameter triple is turned into a full vector.

3. **Vector polishing** (three stages, all respect the remaining time)
   * Adam optimisation on a smooth log‑sum‑exp approximation of the max‑convolution.
   * Projected sub‑gradient descent directly on the true max‑convolution.
   * Stochastic local search with Gaussian noises on the simplex.

If the budget runs out a simple arcsine vector is returned as a safe
fallback.  The routine prints progress and the best official score found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)) for a already normalised (∑a = 1).
    This coincides with the official evaluation value.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric family – symmetric / asymmetric Beta shapes
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (α=β=0.5, small ε)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, smooth_max) where smooth_max approximates max(conv).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights, Σw = 1
    smooth_max = max_c + np.log(sum_exp) / lam

    # weighted sum of quadratic sub‑gradients:
    # ∂ (a*a)_k / ∂a = 2·a shifted by k
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * smooth_max / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, smooth_max


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    noise_every: int = 1500,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast surrogate value.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on   max(conv(a,a)).
    The step size decays like step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term at the maximal index
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local search on the simplex
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    time_deadline: float,
    init_step: float = 0.02,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex: add Gaussian noise, project back,
    keep improvements (with a tiny probability of uphill moves for escape).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)

        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept improvement, otherwise occasional uphill move
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"    Random local iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Parameter‑space hill climbing (α,β,ε)
# ----------------------------------------------------------------------


def hill_climb_parameters(
    n: int,
    start_alpha: float,
    start_beta: float,
    start_eps: float,
    time_deadline: float,
    max_steps: int = 400,
) -> Tuple[float, float, float]:
    """
    Small stochastic local search on the three Beta parameters.
    Uses the fast surrogate ``compute_score``.
    """
    best_alpha, best_beta, best_eps = start_alpha, start_beta, start_eps
    best_vec = generate_beta_shape(n, best_alpha, best_beta, best_eps)
    best_sc = compute_score(best_vec)

    rng = np.random.default_rng()
    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        # Gaussian proposals (log‑space for eps)
        cand_alpha = max(0.01, best_alpha + rng.normal(scale=0.05))
        cand_beta = max(0.01, best_beta + rng.normal(scale=0.05))
        cand_eps = abs(best_eps * np.exp(rng.normal(scale=0.1)))  # log‑normal

        # keep eps in a safe interval
        cand_eps = np.clip(cand_eps, 1e-6, 0.01)

        cand_vec = generate_beta_shape(n, cand_alpha, cand_beta, cand_eps)
        cand_sc = compute_score(cand_vec)

        if cand_sc < best_sc:
            best_alpha, best_beta, best_eps = cand_alpha, cand_beta, cand_eps
            best_sc = cand_sc
            best_vec = cand_vec
    return best_alpha, best_beta, best_eps

                    time_deadline=loc_dead,
                    init_step=0.018,
                    max_iter=max_iter_loc,
                )

            # official evaluation (one call per length)
            val = evaluate_sequence(refined_vec.tolist())
            print(f"  → refined official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_vec.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:
            # If refinement never happened, fall back to raw best candidate
            if local_best_vec is not None:
                val = evaluate_sequence(local_best_vec.tolist())
                print(f"  Raw candidate official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = local_best_vec.tolist()
                    print("*** Updated global best (raw) ***")

    # ------------------------------------------------------------------
    # 3. Final polishing of the overall best vector (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\n=== Final polishing of the global best ===")
        a = np.array(best_seq, dtype=float)

        # Stage A – extended Adam run
        remain = deadline - time.time()
        adam_dead = time.time() + remain * 0.45
        max_iter_adam = int(min(300_000, (adam_dead - time.time()) * 35))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=adam_dead,
            max_iter=max_iter_adam,
            lr0=0.15,
            lam_start=1.0,
            lam_end=160.0,
            noise_every=1000,
            noise_scale=3e-5,
        )

        # Stage B – sub‑gradient polishing
        if time.time() < deadline - 2.0:
            sub_dead = time.time() + (deadline - time.time()) * 0.30
            max_iter_sub = int(min(180_000, (sub_dead - time.time()) * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=sub_dead,
                step0=0.22,
                max_iter=max_iter_sub,
            )

        # Stage C – stochastic local walk
        if time.time() < deadline - 2.0:
            loc_dead = time.time() + (deadline - time.time()) * 0.25
            max_iter_loc = int(min(250_000, (loc_dead - time.time()) * 40))
            print(f"  Random local polishing (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                time_deadline=loc_dead,
                init_step=0.014,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4. Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # safe fallback – classic arcsine with moderate size
        best_seq = arcsine_initial(2048).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
