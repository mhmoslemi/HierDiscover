The routine builds many inexpensive random candidates (random‑phase,
Dirichlet and Beta‑shaped vectors), keeps the best one and finally
polishes it with an exponentiated‑gradient (mirror‑descent) run on the

    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft = np.fft.rfft(a, n=conv_len)
    conv = np.fft.irfft(fft * fft, n=conv_len)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    """
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow for extreme α,β
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of ``max(conv)`` at the point ``a``.
    The gradient is the average of the exact gradients of all active
    convolution entries (those attaining the current maximum up to a tiny
    tolerance).
    """
    n = a.shape[0]
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]

    if active.size == 0:                     # safety net
        grad = 2.0 * a
        return grad

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        # ∂conv_k/∂a_i = 2·a_{k‑i}
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """
    Mirror‑descent update on the probability simplex using the
    exponentiated‑gradient (multiplicative) rule.
    """
    factor = np.exp(-step * grad)
    a_new = a * factor
    a_new = np.clip(a_new, 1e-16, None)      # keep strictly positive
    a_new /= a_new.sum()
    return a_new

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The algorithm runs for at most ``budget_s``
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # --------------------------------------------------------------
    # 1 – problem size & initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        # fallback – a plain uniform start (any reasonable start works)
        n = 5146
        init = np.ones(n, dtype=float) / n
        print(f"No existing sequence – using uniform start, length {n}.")

    conv_len = 2 * n - 1

    def evaluate_and_normalise(vec: np.ndarray):
        """Clamp to ≥0, renormalise to sum = 1 and compute the score."""
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return float("inf"), None
        vec = vec / s
        sc = compute_score(vec)
        return sc, vec

    # keep the best vector seen so far
    best_vec = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 2 – evaluate a few deterministic starts
    # --------------------------------------------------------------
    for cand in (init, np.ones(n) / n):
        sc, vec = evaluate_and_normalise(cand)
        if sc < best_score:
            best_score = sc
            best_vec = vec
            print(f"Initial deterministic candidate score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3 – Random‑phase sampling (high‑throughput exploration)
    # --------------------------------------------------------------
    phase_end = deadline - 30.0           # reserve ~30 s for refinement
    samples = 0
    while time.time() < phase_end:
        # random unit‑magnitude FFT, zero‑phase on DC
        phases = np.random.rand(n // 2 + 1) * 2.0 * np.pi
        phases[0] = 0.0
        spectrum = np.exp(1j * phases)
        a_complex = np.fft.irfft(spectrum, n=n)
        a = np.abs(a_complex)               # force non‑negativity
        # optional symmetry enforcement (cheap)
        a = (a + a[::-1]) * 0.5
        sc, vec = evaluate_and_normalise(a)
        samples += 1
        if sc < best_score:
            best_score = sc
            best_vec = vec
            print(f"[Random‑phase] sample {samples:,d} → new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 4 – Dirichlet draws (different concentration levels)
    # --------------------------------------------------------------
    dirichlet_end = deadline - 15.0        # keep a bit of slack
    conc_vals = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws = 0
    while time.time() < dirichlet_end:
        c = np.random.choice(conc_vals)
        a = np.random.dirichlet(np.full(n, c))
        # symmetrise – symmetry is natural for the self‑convolution problem
        a = (a + a[::-1]) * 0.5
        sc, vec = evaluate_and_normalise(a)
        draws += 1
        if sc < best_score:
            best_score = sc
            best_vec = vec
            print(f"[Dirichlet] draw {draws:,d} (c={c}) → new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 5 – Discretised Beta family sampling
    # --------------------------------------------------------------
    beta_end = deadline - 8.0
    alphas = np.linspace(0.2, 5.0, 12)
    betas = np.linspace(0.2, 5.0, 12)
    beta_trials = 0
    while time.time() < beta_end:
        alpha = float(np.random.choice(alphas))
        beta = float(np.random.choice(betas))
        a = beta_sequence(alpha, beta, n)
        # enforce symmetry (Beta is already symmetric when α=β;
        # we symmetrise anyway to keep the search space consistent)
        a = (a + a[::-1]) * 0.5
        sc, vec = evaluate_and_normalise(a)
        beta_trials += 1
        if sc < best_score:
            best_score = sc
            best_vec = vec
            print(f"[Beta] trial {beta_trials:,d} (α={alpha:.2f},β={beta:.2f}) → new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 6 – Local refinement (exponentiated‑gradient / mirror descent)
    # --------------------------------------------------------------
    if best_vec is None:
        # extremely unlikely, but fall back to uniform
        best_vec = np.ones(n) / n
        best_score = compute_score(best_vec)

    a = best_vec.copy()
    refine_end = deadline - 0.5
    it = 0
    eta0 = 1.0

    while time.time() < refine_end:
        # compute convolution and current score
        fft = np.fft.rfft(a, n=conv_len)
        conv = np.fft.irfft(fft * fft, n=conv_len)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[Refine] it {it:,d} → new best {best_score:.6f}")

        # sub‑gradient of the maximal convolution entry
        grad = subgradient_of_max_conv(a, conv)

        # diminishing step size
        step = eta0 / math.sqrt(it + 1.0)

        # mirror‑descent update on the simplex
        a = exponential_update(a, grad, step)

        # enforce symmetry (helps keep the peak balanced)
        a = (a + a[::-1]) * 0.5

        it += 1

    # --------------------------------------------------------------
    # 7 – Final normalisation and return
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    best_vec /= best_vec.sum()
    print(f"Finished – best score = {best_score:.6f} after {it:,d} refinement steps")
    return [float(v) for v in best_vec.tolist()]
