"""
Algorithm:  Broad parametric sweep → smoothed exponentiated‑gradient (mirror)
            descent on the simplex with enforced symmetry → final long polishing.
            The method searches several sequence lengths, generates many low‑dimensional
            families (arcsine, power‑law, beta, Gaussian, comb‑spike, random Dirichlet
            and mixtures), keeps the best few by a fast FFT based score and improves
            them with a sub‑gradient method that uses a temperature‑controlled
            smooth approximation of the max‑convolution.  Symmetry (a_i = a_{n‑1‑i})
            is enforced after each update, which reduces the feasible space and
            accelerates convergence.  The routine respects the given time budget
            (default 1000 s) and always returns a feasible non‑negative list.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for the official objective.
    Input a is assumed non‑negative; the function works for any vector.
    Returns 2·n·max(conv(a,a)) / (sum(a)²).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)  # length = len(a) + len(w_rev) - 1 = 3·n‑2
    idx = 2 * len(a) - 2
    # pick the slice that corresponds to indices (idx - i) for i=0..n‑1
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution ≈ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters = a."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return normalize(vals)


def comb_shape(n: int, k: int) -> np.ndarray:
    """
    Put equal mass on `k` equally spaced spikes.
    If k>n, returns uniform.
    """
    if k <= 1:
        vals = np.zeros(n, dtype=float)
        vals[0] = 1.0
        return normalize(vals)
    vals = np.zeros(n, dtype=float)
    step = max(1, (n - 1) // (k - 1))
    indices = np.arange(0, n, step)[:k]
    vals[indices] = 1.0 / len(indices)
    return normalize(vals)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def refine_exp_grad(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    tau0: float = 0.5,
    tau_max: float = 20.0,
    enforce_sym: bool = True,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true objective using a smooth approximation
    of the max‑convolution.  The temperature `tau` is annealed from
    `tau0` to `tau_max`.  Updates are multiplicative (exponentiated‑gradient)
    and the iterate is projected onto the simplex after each step.
    Symmetry a_i = a_{n-1-i} is enforced if `enforce_sym` is True.
    Returns the best vector seen and its (exact) score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution and its maximum
        conv = convolution_fft(a, a)
        max_c = np.max(conv)

        # temperature schedule (linear annealing)
        tau = tau0 + (tau_max - tau0) * (it / max_iter)

        # smooth weights that approximate the sub‑gradient of max
        # subtract max_c for numerical stability (values ≤ 0)
        exp_vals = np.exp(tau * (conv - max_c))
        w = exp_vals / np.sum(exp_vals)  # sum to 1

        # gradient of the weighted sum
        grad = gradient_from_weights(a, w)

        # centre the gradient (adds only a global factor after the exp)
        grad_centered = grad - grad.mean()

        # step‑size schedule (classical 1/√t)
        eta = eta0 / np.sqrt(it)

        # multiplicative update, then enforce symmetry and simplex projection
        a = a * np.exp(-eta * grad_centered)
        if enforce_sym:
            a = 0.5 * (a + a[::-1])
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 8  # keep a reasonable number for deeper refinement
        top_candidates = [c for _, c in scored[:top_k]]
        print("  top initial scores:",
              " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]))

        # --------------------------------------------------------------
        # 3.  Refine each retained candidate with smoothed sub‑gradient descent.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            start_sc = compute_score(cand)
            print(f"  candidate {idx + 1}/{len(top_candidates)} start_score = {start_sc:.6f}")

            # allocate iteration budget depending on remaining time
            remaining = max(0.0, deadline - time.time())
            # roughly 8–12 iterations per second for n≈2000; scale with n
            it_per_sec = max(4.0, 12.0 * 2048 / n)
            max_iter = int(min(40_000, max(4_000, remaining * it_per_sec)))

            refined, _ = refine_exp_grad(
                a=cand,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.7,
                tau0=0.5,
                tau_max=15.0,
                enforce_sym=True,
                report_every=0,
            )

            # official evaluation
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")
        # end of candidates loop
    # end of lengths loop

    # ------------------------------------------------------------------
    # 4.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        # give the polishing a generous number of iterations
        it_per_sec = max(4.0, 12.0 * 2048 / len(a))
        max_iter = int(min(120_000, max(20_000, remaining * it_per_sec)))

        polished, _ = refine_exp_grad(
            a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
            tau0=0.3,
            tau_max=20.0,
            enforce_sym=True,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
