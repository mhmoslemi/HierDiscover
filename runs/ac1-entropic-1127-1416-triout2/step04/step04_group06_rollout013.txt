def gradient_descent_adam(a, deadline, max_iters=5000,
                         alpha_start=1.0, alpha_end=400.0,
                         eta=0.2, beta1=0.9, beta2=0.999,
                         eps=1e-12):

        if time.time() > deadline - 0.5:

        # compute self-convolution
        conv = np.fft.irfft(np.fft.rfft(a, n=2*n-1) * np.fft.rfft(a, n=2*n-1), n=2*n-1)
        # softmax weights

        p_fft = np.fft.rfft(p, n=L_corr)

        offset = len(p) - 1

        # Adam update of gradient before exponentiated step
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**it)

def simulated_annealing(a, best_score, deadline):
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = best_score
    T = cur_score * 0.03
    T_end = cur_score * 1e-6
    while time.time() < deadline - 0.2:
        # choose src weighted by cur_a
        src = np.random.choice(n, p=cur_a)
        tgt = np.random.randint(n)
        if tgt == src:
            continue
        # random transfer delta
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand_score = compute_score(cand)
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T+1e-12))
        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] new best {best_score:.6f}")
        # cooling
        T *= 0.9996
    return best_a, best_score

"""
Propose a non‑negative step‑function (a simplex vector) that minimizes
``evaluate_sequence``.  

The search consists of three stages:

1. **Diverse random initialisers** – beta‑densities, Dirichlet draws,
   edge‑heavy shapes, sparse spikes and an up‑sampled previous best
   (if available).

2. **Adam‑style exponentiated‑gradient mirror descent** on a smooth
   approximation of the L∞‑norm of the self‑convolution.
   The smoothing parameter α is increased exponentially from 1 to a large
   value (default 400) so the iterates approach the true maximum.
   The update is multiplicative (mirror descent on the simplex) and the
   adaptive Adam moments help to navigate the high‑dimensional space
   efficiently.

3. **Simulated‑annealing refinement** – small mass‑transfer moves are
   accepted with a Metropolis rule.  This cheap local search escapes
   remaining shallow minima.

All computation respects a user‑provided time budget (default 1000 s);
the best vector found is returned as a plain list of floats (sum ≈ 1,
non‑negative).  No closures or lambda expressions are used – every helper
is defined at top level.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities – all are top‑level, no closures
# ------------------------------------------------------------

def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (preserving non‑negativity)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector whose sum is (approximately) 1.
    Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution via FFT, length 2*n‑1
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum = 1.
    Mid‑point rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow/underflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)   # safe range for exp
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet(1,…,1) (i.e. i.i.d. Gamma(1)) and normalise."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """Linear ‘edge‑heavy’ shape – larger values near the two ends."""
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = (dist_to_center.astype(float) + 1.0)
    return simplex_normalise(a)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Place mass on k random positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)


def upsample_previous(prev: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation of a previous vector to a new length.
    The result is re‑normalised to sum = 1 and clipped to non‑negative.
    """
    if prev.size == target_n:
        return simplex_normalise(prev)
    x_old = np.linspace(0.0, 1.0, prev.size)
    x_new = np.linspace(0.0, 1.0, target_n)
    new = np.interp(x_new, x_old, prev)
    new = np.clip(new, 0.0, None)
    return simplex_normalise(new)


def next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ------------------------------------------------------------
# Adam‑style exponentiated‑gradient mirror descent
# ------------------------------------------------------------

def gradient_descent_adam(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 4000,
                         alpha_start: float = 1.0,
                         alpha_end: float = 400.0,
                         eta: float = 0.15,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution using
    exponentiated‑gradient with Adam moments.  Returns the best vector
    encountered and its score.
    """
    a = start_a.copy()
    n = a.shape[0]

    # FFT size for the correlation (full length = n + (2n‑1) - 1 = 3n‑2)
    L_corr = next_pow_two(3 * n - 2)

    # Adam moment vectors
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        # time‑check – keep a small safety margin
        if time.time() > deadline - 0.5:
            break

        # ----------------------------------------------------
        # 1) self‑convolution (FFT)
        # ----------------------------------------------------
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

        # ----------------------------------------------------
        # 2) soft‑max weights (smooth max)
        # ----------------------------------------------------
        maxc = conv.max()
        # exponential schedule for the smoothing parameter α
        alpha = alpha_start * ((alpha_end / alpha_start) ** (it / max_iters))
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp

        # ----------------------------------------------------
        # 3) cross‑correlation = gradient of the smooth max
        # ----------------------------------------------------
        fft_a_full = np.fft.rfft(a, n=L_corr)
        fft_p_full = np.fft.rfft(p, n=L_corr)
        corr = np.fft.irfft(fft_a_full * np.conj(fft_p_full), n=L_corr)
        offset = len(p) - 1               # lag 0 appears at this index
        grad = corr[offset:offset + n]

        # ----------------------------------------------------
        # 4) Adam moments
        # ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----------------------------------------------------
        # 5) exponentiated‑gradient update (mirror descent on simplex)
        # ----------------------------------------------------
        factor = np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        # avoid numerical blow‑up / under‑flow
        factor = np.clip(factor, 1e-12, 1e12)
        a = a * factor
        a = simplex_normalise(a)

        # ----------------------------------------------------
        # 6) keep track of the best vector seen so far
        # ----------------------------------------------------
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[GD] iter {it:4d} → new best {best_score:.6f}")

    return best_a, best_score

def simulated_annealing(start_a: np.ndarray,
                        start_score: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer SA that keeps the sum exactly 1.
    Returns the best vector found and its score.
    """
    n = start_a.shape[0]
    cur_a = start_a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    # start temperature proportional to the current score
    T = cur_score * 0.02
    T_end = cur_score * 1e-5
    iteration = 0

    while time.time() < deadline - 0.2 and T > T_end:
        # choose a source index with probability proportional to its mass
        src = np.random.choice(n, p=cur_a)
        # pick a distinct target uniformly
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        if cur_a[src] <= 0.0:
            continue
        # transfer a random fraction of the source mass
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[SA] iter {iteration:5d} → new best {best_score:.6f}")

        # geometric cooling
        T *= 0.9995
        iteration += 1

    return best_a, best_score
