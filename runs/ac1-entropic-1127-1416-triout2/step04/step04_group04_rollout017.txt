    sum_exp = np.sum(exp_vals)
    softmax = exp_vals / sum_exp

    return smooth_max_val, softmax

def refine_smooth(a_initial, deadline, max_iter=None, extra_iters=False):
    a = normalize(a_initial.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    conv_len = 2*n-1

    # hyperparameters
    alpha = 1.0   # start low
    alpha_max = 200.0
    alpha_inc = (alpha_max - alpha) / (max_iter/2 if max_iter else 1000)
    # learn rate schedule
    eta0 = 0.5

    it = 0
    if max_iter is None:
        max_iter = 20000 if not extra_iters else 50000

    while it < max_iter and time.time() < deadline:
        it += 1
        # compute conv and smoothed weights
        conv = convolution_fft(a, a)
        # compute softmax weights
        # use incremental alpha schedule
        cur_alpha = min(alpha_max, alpha + (it/ max_iter) * (alpha_max - alpha))
        max_val = conv.max()
        # shift for numerical stability
        exp_vals = np.exp(cur_alpha * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # gradient
        grad = gradient_from_weights(a, w)
        # step size schedule
        eta = eta0 / np.sqrt(it)
        # multiplicative update
        a = a * np.exp(-eta * grad)
        a = normalize(a)
        # evaluate
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # optionally reduce step size if not improving?
    return best_a, best_sc

def refine_smooth(a_init, deadline, max_iter=None):
    a = normalize(a_init.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2*n - 1

    # schedule parameters
    eta0 = 0.5
    p_start = 2.0   # start of power
    p_end = 64.0
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        # compute weight using power law
        p = p_start + (p_end - p_start) * (it / max_iter)
        # use exponential smoothing approximating L^p norm: w ∝ conv^{p-1}
        # But conv may have zeros -> add small epsilon
        eps = 1e-12
        w = np.power(conv + eps, p-1)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full(conv_len, 1.0/conv_len)
        else:
            w = w / w_sum
        grad = gradient_from_weights(a, w)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

"""
propose_candidate
=================

A new search strategy for the non‑negative step‑function minimising

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm differs from the previous approach in several ways:

* **Smoothed max objective** –  instead of a hard sub‑gradient on the
  max‑entry we minimise a smooth approximation
  f(a)= (1/α)·log Σ_k exp(α·conv_k).  For α→∞ this converges to the true
  max, but for moderate α the gradient is a weighted sum of all
  convolution entries, giving a more informative direction.

* **Exponentiated‑gradient on the simplex** –  the update
  a ← a·exp(−η·∇f) keeps the variables non‑negative and a single
  normalisation step projects back onto the probability simplex
  (∑a_i=1).  This is the natural analogue of mirror descent for the
  simplex.

* **Richer initial families** –  besides the arcsine/power‑law families
  used before we also generate Gaussian, exponential, linear‑ramp and
  random Dirichlet shapes, as well as asymmetric permutations of a good
  seed.  The initial candidates are ranked with a fast FFT‑based score
  and only the best few are refined.

* **Annealing of the smoothness parameter** –  the exponent α is
  increased from a small value to a large one during the optimisation,
  gradually turning the smoothed objective into the true maximum.

* **Time‑aware budgeting** –  the routine constantly checks the time
  budget (default 1000 s) and adapts the number of optimisation
  iterations accordingly, guaranteeing termination with a feasible
  sequence.

The final result is the best sequence found, evaluated with the official
`evaluate_sequence` routine.  The code respects all competition rules:
no file‑IO, no closures, only top‑level functions, and it prints progress
information for debugging.

"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    # Numerical noise may produce tiny negative values – clip them.
    conv = np.maximum(conv, 0.0)
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the scale‑invariant objective:
        2·n·max(conv(a,a)) / (sum(a)²)
    No clipping or security checks – purely for internal ranking.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                           # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads

# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution ≈ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian (truncated to the domain)."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return normalize(vals)


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam * i)
    return normalize(vals)


def linear_ramp_shape(n: int) -> np.ndarray:
    """Linear ramp from 0 to 1."""
    i = np.arange(n, dtype=float)
    vals = i
    return normalize(vals)


def interior_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet on the interior (first/last entry forced to zero)."""
    if n <= 2:
        return uniform_shape(n)
    interior = rng.exponential(1.0, size=n - 2)
    interior = normalize(interior)
    vals = np.concatenate(([0.0], interior, [0.0]))
    return normalize(vals)


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) gradient descent on the simplex
# ----------------------------------------------------------------------


def refine_smooth(
    a_init: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.6,
    alpha_start: float = 2.0,
    alpha_end: float = 120.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient minimisation of a smoothed version of
    max(conv(a,a)).  The smoothing parameter α grows from ``alpha_start``
    to ``alpha_end`` during the run (log‑sum‑exp annealing).
    Returns the best vector seen and its fast (FFT‑based) score.
    """
    a = normalize(a_init.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)                # length 2n‑1, non‑negative
        # anneal the smoothing exponent
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)

        # log‑sum‑exp weighting (numeric stability by shifting by max)
        max_c = conv.max()
        exp_weights = np.exp(alpha * (conv - max_c))
        w = exp_weights / exp_weights.sum()         # Σ w = 1, all ≥0

        grad = gradient_from_weights(a, w)

        # classic 1/√t step‑size schedule
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)                 # multiplicative update
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` score.  The routine respects the time budget,
    builds a diverse set of initial candidates, refines promising ones
    with the smoothed‑max gradient descent, and finally returns the best
    sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin to avoid a hard timeout
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Length(s) to explore – 2048 is a good trade‑off between quality
    # and speed.  We also try 1024 as a cheap backup.
    # ------------------------------------------------------------------
    lengths = [2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n>>> Exploring length n = {n}")

        # --------------------------------------------------------------
        # 1. Build a diverse library of initial candidates.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        candidates.append(uniform_shape(n))
        candidates.append(arcsine_initial(n))

        # Gaussian mixtures with several bandwidths
        for sigma in [n / 8.0, n / 6.0, n / 4.0]:
            candidates.append(gaussian_shape(n, sigma))

        # Exponential decays
        for lam in [0.001, 0.002, 0.005, 0.010]:
            candidates.append(exponential_shape(n, lam))

        # Linear ramp (asymmetric)
        candidates.append(linear_ramp_shape(n))

        # Random Dirichlet draws
        for _ in range(8):
            candidates.append(random_dirichlet(n, rng))

        # Interior Dirichlet (zeros at both ends)
        for _ in range(5):
            candidates.append(interior_dirichlet(n, rng))

        # Permuted arcsine – breaks symmetry
        base_arcs = arcsine_initial(n)
        for _ in range(5):
            candidates.append(rng.permutation(base_arcs))

        # --------------------------------------------------------------
        # 2. Quick ranking with the cheap FFT‑based score; keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 5
        top_candidates = [c for _, c in scored[:top_k]]
        print(
            "  top initial scores:",
            " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]),
        )

        # --------------------------------------------------------------
        # 3. Refine each retained candidate with smoothed‑max descent.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            start_sc = compute_score(cand)
            print(f"  candidate {idx + 1}/{len(top_candidates)} start_score = {start_sc:.6f}")

            # Allocate iterations based on remaining time.
            remaining = max(0.0, deadline - time.time())
            # Empirical throughput: ~15‑20 iterations per second for n≈2000.
            it_per_sec = 17
            max_iter = int(min(50_000, max(3_000, remaining * it_per_sec)))

            refined, _ = refine_smooth(
                a_init=cand,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.6,
                alpha_start=2.0,
                alpha_end=120.0,
            )

            # official evaluation (includes all safety checks)
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 4. Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        it_per_sec = 18
        max_iter = int(min(80_000, max(6_000, remaining * it_per_sec)))

        polished, _ = refine_smooth(
            a_init=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
            alpha_start=5.0,
            alpha_end=200.0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
