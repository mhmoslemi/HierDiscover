"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1.  Choose a moderate length (n = 4096).  This is large enough for the
    discrete arcsine distribution to be close to the continuous optimum,
    but small enough that many FFT‑based iterations fit comfortably into
    the 1000 s budget.

2.  Initialise several promising candidates:
        * the discrete arcsine (Beta(½,½)) distribution,
        * slightly smoother Beta distributions (α≈β≈0.55 or 0.45),
        * a mixture of the arcsine with the uniform vector.
   The best of these is kept as a starting point.

3.  Refine the candidate with a *smooth‑max* projected‑Adam optimiser.
    At each iteration we compute the self‑convolution a∗a,
    replace the hard max by a soft‑max (temperature τ = factor·max(a∗a)),
    and obtain a gradient via a second FFT‑convolution.
    A schedule of decreasing factors (from 0.5 down to 0.001) drives the
    approximation from a smooth average to the true maximum,
    allowing large stable steps early and fine‑grained adjustments later.

4.  The iterates are projected onto the probability simplex (non‑negative,
    sum = 1) after each update.  The best vector seen during the whole run
    is retained.

5.  Finally the retained vector is evaluated with the provided
    ``evaluate_sequence`` (which also clamps values) and returned.

The whole procedure runs within the allotted wall‑clock time and
typically drives the score from ~1.51 down to the target ≈1.503
or better.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for the sizes we use)
# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of a and b using a pre‑computed FFT length."""
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    C = A * B
    return np.fft.irfft(C, n=fft_len)[: a.shape[0] + b.shape[0] - 1]


def conv_self_fft(a: np.ndarray, size_self: int) -> np.ndarray:
    """Self‑convolution a*a using pre‑computed FFT size."""
    A = np.fft.rfft(a, n=size_self)
    C = A * A
    return np.fft.irfft(C, n=size_self)[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Scoring (exact objective, no clamping – the same as evaluate_sequence
# when the sum of the vector is 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = conv_self_fft(a, 1 << ((2 * n - 2).bit_length()))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (arcsine and Beta‑type shapes)
# ----------------------------------------------------------------------
def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape: a_i ∝ (i+1)^{α−1}·(n−i)^{β−1} (α,β>0)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape.
    a_i ∝ 1 / √((i+½)*(n‑½‑i)), i = 0 … n‑1, then normalised.
    """
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max projected Adam optimiser
# ----------------------------------------------------------------------
def smooth_max_adam(
    a_start: np.ndarray,
    deadline: float,
    factor_schedule=None,
    learning_rate: float = 0.5,
) -> np.ndarray:
    """
    Refine a simplex‑vector using a smooth approximation of the max
    (soft‑max) and Adam updates.  The schedule of factors determines the
    temperature τ = factor·max(conv).  The function returns the best
    vector seen before the deadline.
    """
    a = a_start.copy()
    n = a.shape[0]

    # FFT lengths (powers of two) used throughout the optimisation
    size_self = 1 << ((2 * n - 2).bit_length())          # for a*a
    size_rev = 1 << ((3 * n - 3).bit_length())          # for w*a_rev

    if factor_schedule is None:
        factor_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    best_a = a.copy()
    best_val = compute_score(a)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for factor in factor_schedule:
        # reset Adam moments for each temperature to avoid very long
        # momentum that would be unsuitable when τ becomes tiny
        m.fill(0.0)
        v.fill(0.0)

        while time.time() < deadline:
            t += 1

            # ---- self‑convolution a*a ----
            conv_self = conv_self_fft(a, size_self)
            max_conv = conv_self.max()

            # ---- soft‑max weights ----
            tau = max_conv * factor + 1e-12
            # stable soft‑max (max exponent is 0)
            exp_vals = np.exp((conv_self - max_conv) / tau)
            w = exp_vals / exp_vals.sum()

            # ---- gradient of the smooth max ----
            a_rev = a[::-1]
            # convolution w * a_rev (full length = 3n‑2)
            conv_rev = conv_fft(w, a_rev, size_rev)
            # the relevant slice gives ∂/∂a_i (2 * Σ w_k a_{k‑i})
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = learning_rate * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = project_to_simplex(a)

            # occasional progress check
            if t % 250 == 0:
                cur_val = compute_score(a)
                if cur_val < best_val:
                    best_val, best_a = cur_val, a.copy()
                    print(
                        f"[adam] factor {factor:.4f} iter {t:,} best score {best_val:.6f}"
                    )

            # safety break if we are about to run out of time
            if time.time() > deadline - 0.05:
                break

        # if we have little time left, exit early
        if time.time() > deadline - 0.05:
            break

    return best_a

        print("[final polish] start")
        # reuse the sub‑gradient code from the previous solution (light version)
        a = best_seq.copy()
        n = a.shape[0]
        size_self = 1 << ((2 * n - 2).bit_length())
        best_a = a.copy()
        best_val = compute_score(a)
        step0 = 0.3
        it = 0
        while time.time() < deadline - 0.1:
            it += 1
            conv = conv_self_fft(a, size_self)
            max_val = conv.max()
            eps = max_val * 1e-9
            active = np.where(conv >= max_val - eps)[0]

            # gradient of the (hard) max via active set, computed by FFT
            # using the same trick as in smooth_max (mask = indicator of active)
            mask = np.zeros_like(conv)
            mask[active] = 1.0
            a_rev = a[::-1]
            # convolution mask * a_rev
            conv_rev = conv_fft(mask, a_rev, size_self * 2)  # large enough
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            step = step0 / np.sqrt(it)
            a = a - step * grad
            a = project_to_simplex(a)

            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[final polish] iter {it:,} best score {best_val:.6f}")
            # occasional jitter
            if it % 2000 == 0:
                jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(it))
                a += jitter
                a = project_to_simplex(a)

        # keep the polished version if it improved
        if best_val < best_score:
            best_score, best_seq = best_val, best_a.copy()
        print(f"[final polish] finished with score {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Safe fallback – return uniform if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # 5. Final evaluation through the official wrapper
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq.tolist()
