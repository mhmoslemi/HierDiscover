    resphi = 2 - phi

            d = b - resphi*(b-a)

def golden_section(f, lo, hi, tol, max_iter, deadline):
    phi = (1 + math.sqrt(5))/2
    resphi = 2 - phi
    x1 = lo + resphi * (hi - lo)
    x2 = hi - resphi * (hi - lo)
    f1 = f(x1)
    f2 = f(x2)
    # evaluate and update best globally inside loop (if f1 < best)
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + resphi*(hi - lo)
            f1 = f(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = hi - resphi*(hi - lo)
            f2 = f(x2)
        if abs(hi - lo) < tol:
            break
    # final best among x1,x2 and boundaries
    if f1 < f2:
        return x1, f1
    else:
        return x2, f2
```

def upscale(a_small, n_large):
    n_small = a_small.shape[0]
    # compute positions of small shape as centers of intervals
    xs_small = np.linspace(0, 1, n_small, endpoint=False) + 0.5/n_small
    xs_large = np.linspace(0, 1, n_large, endpoint=False) + 0.5/n_large
    a_large = np.interp(xs_large, xs_small, a_small, left=a_small[0], right=a_small[-1])
    a_large = a_large / a_large.sum()
    return a_large
```

"""
Propose a step‑function sequence with a low value of
evaluate_sequence().  The algorithm proceeds in three phases:

1. Fast parametric sweep on a modest size (n≈2048) over several
   low‑dimensional families (reverse‑power‑law, Gaussian, sine,
   raised‑cosine, geometric, etc.).  Fine grids (step 1e‑3 or 5e‑4)
   are used to pinpoint the best parameters.

2. The best shape is interpolated to a larger resolution (n≈4096 if
   time permits) and then polished by projected sub‑gradient descent
   on the exact max‑convolution objective.  An adaptive tolerance
   enlarges the active set, and gentle jitter avoids stagnation.

3. A stochastic hill‑climb with adaptive step size and occasional
   restarts finishes the optimisation.

All vectors are kept on the probability simplex (Σa=1); the
evaluation function is scale‑invariant, so the scale does not matter.
The final sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}, decreasing front‑loaded power‑law (p≥0)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p} (decreasing tail‑heavy power‑law)."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}, 0<q<1."""
    if not (0.0 < q < 1.0):
        return None
    a = q ** np.arange(n, dtype=np.float64)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x^{α}(1‑x)^{β}, x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle.  sigma is a fraction of n."""
    if sigma <= 0.0:
        return None
    i = np.arange(n, dtype=np.float64) + 0.5
    centre = n / 2.0
    x = (i - centre) / (sigma * n)
    a = np.exp(-0.5 * x * x)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sine_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^{p}, p≥0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = np.sin(np.pi * i / n)
    a = np.power(x, p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def raised_cosine_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1‑cos(2π(i+0.5)/n))^{p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = (1.0 - np.cos(2.0 * np.pi * i / n)) / 2.0
    a = np.power(x, p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    convex combination  mix·reverse_power_law(p) + (1‑mix)·uniform,
    0 ≤ mix ≤ 1.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

    """Linear interpolation of a probability vector to a larger size."""
    n_small = a_small.shape[0]

    a_large = np.interp(xs_large, xs_small, a_small, left=a_small[0], right=a_small[-1])
    a_large /= a_large.sum()
    return a_large

# Sub‑gradient polishing (more aggressive tolerance & jitter)

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """
    Projected sub‑gradient descent on  max(conv(a,a)).
    Uses a relative tolerance for the active set and occasional jitter.
    """
    a = a0.copy()
    best_a = a.copy()
    n = a.shape[0]
    best_M = full_convolution(a).max()
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # active indices: within a small relative band of the maximum
        tol = max(1e-12, 1e-8 * M)
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient for the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to disturb stagnation
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

    it = 0

            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            # restart from current best with fresh jitter
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0            # safety margin

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # 0. Optional fallback (if a previous construction is present)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Fast parametric sweep on a modest size (≈200 s)
    # ------------------------------------------------------------------
    n_small = 2048
    remaining = deadline - time.time()
    if remaining > 250:
        sweep_deadline = time.time() + min(200.0, remaining - 50.0)
        print(f"[phase 1] parametric sweep on n={n_small}, deadline={sweep_deadline:.2f}")
        a_small = parametric_search(n_small, sweep_deadline)
        if a_small is not None:
            val = evaluate_sequence(a_small.tolist())
            print(f"[phase 1] best raw score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_small.tolist()
    else:
        print("[phase 1] Skipped parametric sweep due to time constraints")

    # ------------------------------------------------------------------
    # 2. Upscale to a larger resolution if time permits
    # ------------------------------------------------------------------
    cur_time = time.time()
    remaining = deadline - cur_time
    n_large = 4096 if remaining > 200 else n_small

    if best_seq is not None and n_large != n_small:
        a_small = np.array(best_seq, dtype=float)
        a_large = upscale(a_small, n_large)
        val = evaluate_sequence(a_large.tolist())
        print(f"[upscale] n={n_large} score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_large.tolist()
    elif best_seq is None:
        # never got a candidate -> start from a generic shape
        a_large = mixed_reverse_uniform(n_large, mix=0.8, p=1.8)
        best_seq = a_large.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] using mixed shape, score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient polishing (bulk of remaining time)
    # ------------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.15)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 4.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # 5. Safety fallback (should never be hit)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_vec = mixed_reverse_uniform(n_large, mix=0.85, p=2.0)
        best_seq = best_vec.tolist()
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
