def lp_step(a, M, t):
    n = a.shape[0]
    # Build constraint matrix
    rows = []
    cols = []
    data = []
    for i in range(n):
        start = i
        end = i + n
        # rows indices from start to end-1
        rows.extend(range(start, end))
        cols.extend([i] * n)
        data.extend([a[i]] * n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # define variable b
    b = cvxpy.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    # maximize sum b
    objective = cvxpy.Maximize(cvxpy.sum(b))
    prob = cvxpy.Problem(objective, constraints)
    prob.solve(solver='ECOS', warm_start=True, verbose=False, max_iters=5000) # adjust
    # After solving, get b value
    b_val = b.value
    if b_val is None:
        return a  # fallback
    # Normalize b to sum=1
    b_norm = b_val / b_val.sum()
    # convex combination
    a_new = (1 - t) * a + t * b_norm
    return a_new

"""
Non‑negative step‑function optimiser – improved version.

The optimiser works in several stages:

1.   Initialise a population of random Dirichlet vectors and keep the best.
2.   Run a moderate‑length Adam optimiser on the smooth‑max surrogate
     τ·log Σ exp(conv/τ).  This quickly lowers the peak of the convolution.
3.   Refine the solution using a linear‑programming step described in the
     problem statement: maximise Σ b_j under the linear constraints
     (a * b)_k ≤ M, where M is the current maximum convolution value.
     The returned b is mixed with the current a.
4.   Perform a deterministic projected‑gradient descent that uses the
     gradient of the active (maximum) convolution entry.
5.   Finish with a small random mass‑transfer hill‑climb for polishing.

All vectors are kept on the probability simplex (∑a=1, a_i≥0); the objective
2·n·max(conv) is invariant under scaling, so the normalisation does not
affect the final score returned by ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (fast for many calls)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 2.0,
    tau_end: float = 1e-6,
    tau_decay: float = 0.995,
    lr_start: float = 0.05,
    lr_decay: float = 0.999,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate
        f_τ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    Returns the best vector observed before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # temperature schedule
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # full convolution
        conv = full_convolution(a)               # length m
        shift = conv.max()
        # stable soft‑max
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # shape (m,)

        # gradient of smooth‑max: 2·(a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter={t}, τ={tau:.1e}, lr={lr:.3e}, score={cur_val:.6f}")

        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-6)

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “g‑step” from the statement)
# ----------------------------------------------------------------------


def lp_refine(a0: np.ndarray, deadline: float, current_best: float):
    """
    Perform a sequence of LP‑based refinements.
    Each iteration solves:
        max Σ b_j
        s.t. (a * b)_k ≤ M   for all k,
             b_j ≥ 0,
    where M is the current maximum of a*a.
    The solution b is mixed with a using a shrinking step size.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    # static index structure for the constraint matrix (size (2n‑1)×n)
    row_idx = (np.arange(n)[:, None] + np.arange(n)).ravel().astype(np.int32)
    col_idx = np.repeat(np.arange(n), n).astype(np.int32)

    best_a = a.copy()
    best_val = current_best
    step = 0.30  # initial mixing coefficient

    while time.time() < deadline:
        conv = full_convolution(a)
        M = conv.max()
        # Data for A_ub = a[col_idx] placed at (row_idx, col_idx)
        data = a[col_idx]
        A_ub = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        c = -np.ones(n)                     # maximise Σ b_j  → minimise –Σ b_j
        b_ub = M * np.ones(2 * n - 1)

        # solve LP with HiGHS (fast, supports sparse A)
        res = scipy.optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 5.0},
        )

        if not res.success or res.x is None:
            break

        b = res.x
        if b.sum() == 0.0:
            break

        b_norm = b / b.sum()
        a = (1.0 - step) * a + step * b_norm
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[lp] step={step:.3f}, score={cur_val:.6f}")

        # shrink the mixing factor slowly
        step = max(step * 0.95, 0.02)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic projected gradient descent using the active convolution entry
# ----------------------------------------------------------------------


def gradient_refine(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the exact max‑convolution objective.
    Uses the gradient of the active (maximal) convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]

    eta = 0.02               # initial step size
    it = 0

    while time.time() < deadline:
        conv = full_convolution(a)
        kmax = int(np.argmax(conv))

        # gradient of conv_{kmax} w.r.t. a: 2 * a shifted by kmax
        partner = kmax - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[partner[mask]]

        # projected gradient step
        a_new = a - eta * grad
        a_new = project_to_simplex(a_new)

        cur_val = compute_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            if it % 5000 == 0:
                print(f"[grad] iter={it}, score={cur_val:.6f}")

        a = a_new
        it += 1

        # decay step size
        if it % 2000 == 0:
            eta = max(eta * 0.7, 1e-4)

    return best_a


# ----------------------------------------------------------------------
# Simple hill‑climbs on the exact objective (final polish)
# ----------------------------------------------------------------------


def mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Move a small fraction of mass from one index to another."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        frac = np.random.rand() * 0.30
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 10000 == 0:
                print(f"[mass] iter={it}, score={val:.6f}")
        it += 1
    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5.0   # small safety margin

    # --------------------------------------------------------------
    # 0. Load a known construction (fallback) if available.
    # --------------------------------------------------------------
    fallback = None
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
    except Exception:
        pass

    if fallback is not None:
        a0 = np.array(fallback, dtype=float)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        n = a0.shape[0]
        print(f"[init] using fallback of length {n}")
    else:
        n = 2048
        a0 = np.full(n, 1.0 / n)
        print("[init] no fallback – start with uniform vector")

    best_a = a0.copy()
    best_val = compute_score(best_a)
    print(f"[init] initial score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Quick Dirichlet seeding.
    # --------------------------------------------------------------
    for _ in range(12):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[dirichlet] improved seed score = {val:.6f}")

    # --------------------------------------------------------------
    # 2. Adam‑based smooth‑max optimisation (moderate time budget).
    # --------------------------------------------------------------
    adam_deadline = min(deadline, time.time() + 250.0)
    print("[adam] starting smooth‑max optimisation")
    a_adam = smooth_max_adam(best_a, adam_deadline, lr_start=0.03)
    val = compute_score(a_adam)
    if val < best_val:
        best_val = val
        best_a = a_adam.copy()
        print(f"[adam] improved to {val:.6f}")

    # --------------------------------------------------------------
    # 3. Linear‑programming refinement.
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        lp_deadline = min(deadline, time.time() + 250.0)
        print("[lp] starting LP refinement")
        a_lp, val_lp = lp_refine(best_a, lp_deadline, best_val)
        if val_lp < best_val:
            best_val = val_lp
            best_a = a_lp.copy()
            print(f"[lp] improved to {val_lp:.6f}")

    # --------------------------------------------------------------
    # 4. Deterministic projected‑gradient descent.
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        grad_deadline = min(deadline, time.time() + 250.0)
        print("[grad] starting gradient refinement")
        a_grad = gradient_refine(best_a, grad_deadline)
        val = compute_score(a_grad)
        if val < best_val:
            best_val = val
            best_a = a_grad.copy()
            print(f"[grad] improved to {val:.6f}")

    # --------------------------------------------------------------
    # 5. Final small‑scale hill‑climb for polishing.
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mass_deadline = deadline - 2.0
        print("[mass] starting final mass‑transfer hill‑climb")
        a_final = mass_transfer_hill_climb(best_a, mass_deadline)
        val = compute_score(a_final)
        if val < best_val:
            best_val = val
            best_a = a_final.copy()
            print(f"[final mass] improved to {val:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best sequence found (normalised to sum = 1).
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq
