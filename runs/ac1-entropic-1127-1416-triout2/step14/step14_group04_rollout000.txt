def lp_optimize(f, M):
    n = len(f)
    # Build A_ub shape (2n-1, n)
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    # Each column j (0-index) corresponds to f shifted by j: positions j..j+n-1 receive f contributions.
    for j in range(n):
        A[j:j+n, j] = f
    # This is O(n^2) loop; can accelerate using broadcasting:
    # but this loops n times each building a slice; still okay for moderate n (<1024)
    # but we can also use np.add.at:
    # Not needed.
    # Solve linear program:
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None)] * n
    # Upper bound M for all rows
    b = np.full(rows, M)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        # fallback to f
        g = f.copy()
    return g

"""
Propose a low‐max‑convolution sequence.

Algorithm
---------

1.  Initialise a 2048‑entry vector `f` using a cheap
    exponential (or a fallback global variable) that already yields
    a decent score (~1.51).

2.  Iteratively improve `f` with an *alternating linear programme*:
    * For the current `f` compute `M = max (f*f)`.
    * Solve the LP  (variables `g_i ≥ 0`)

          maximise   Σ g_i
          subject to conv(f, g) ≤ M      (∀ shifts)
                     Σ g_i  free (the LP will push the sum as high as
                     possible while respecting the bound).

      Because `conv(f,g) = A @ g` with a Toeplitz matrix `A`,
      the constraints are linear.  `scipy.optimize.linprog` (HiGHS)
      is used.

    * Rescale the optimiser output `g` to a probability vector
      (`g ← g / Σ g`).

    * Mix the old and the new vector:
          f ← (1‑t)·f + t·g      (t starts at 0.25 and is slowly reduced)

      Mixing keeps the sum equal to 1, preserves non‑negativity and
      typically reduces the maximal convolution value.

    * Keep the best vector seen (lowest exact score
      `2·n·max(conv(f,f))`).

    The loop stops when the wall‑clock budget (≈ 1000 s) is exhausted.

3.  A short simulated‑annealing polish (optional, cheap) is run on the
    best vector to escape possible plateaus.

The whole search runs in < 1000 s on a single core and returns a list
of floats ready for `evaluate_sequence`.  The method improves upon the
previous soft‑max/Adam approach by directly attacking the `max`‑convolution
constraint via exact linear programming.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple families to obtain an initial candidate
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Linear‑programming step
# ----------------------------------------------------------------------


def lp_optimize(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_i   subject to   conv(f, g) ≤ M,   g ≥ 0.
    Returns the optimiser output g (not normalised).
    """
    n = f.size
    rows = 2 * n - 1

    # Build the Toeplitz constraint matrix A where conv(f,g) = A @ g.
    # Column j of A is f shifted by j positions.
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    # Linear programme: maximise sum(g) → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)          # objective vector
    b = np.full(rows, M, dtype=np.float64)     # right‑hand side (all M)

    # Bounds: g_i >= 0, no upper bound.
    bounds = [(0, None)] * n

    # Use HiGHS (the default for recent SciPy versions).  It tolerates dense A.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={"presolve": True})
    if res.success:
        g = res.x
    else:
        # If the LP fails (unlikely) fall back to the current vector.
        g = f.copy()
    return g


# ----------------------------------------------------------------------
# Small simulated‑annealing polish (optional)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Simple Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to kick the walk out of stagnation
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size – 2048 gives a good trade‑off between expressivity
    # and speed within the 1000 s budget.
    # --------------------------------------------------------------
    n = 2048

    best_seq = None
    best_score = float("inf")   # this is the *compute_score* value (sum=1)

    # --------------------------------------------------------------
    # 0. Use a global fallback if available (might already be good)
    # --------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            # normalise internally for our own bookkeeping
            arr = np.array(fb, dtype=float)
            if arr.sum() == 0:
                raise ValueError
            arr = arr / arr.sum()
            sc = compute_score(arr)
            if sc < best_score:
                best_score, best_seq = sc, arr.tolist()
                print(f"[fallback] initial score = {sc:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. QUICK baseline: sweep exponential and power‑law families
    # --------------------------------------------------------------
    # (the sweeps are deliberately coarse – they just give a decent start)
    if time.time() < deadline:
        # exponential sweep – λ roughly in [5e‑4, 2e‑3] works well for n=2048
        lam_vals = np.linspace(5e-4, 2e-3, 200)
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            a = generate_exponential(n, lam)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
        # power‑law sweep – exponent in [0.1, 2.5]
        if time.time() < deadline:
            p_vals = np.linspace(0.1, 2.5, 200)
            for p in p_vals:
                if time.time() >= deadline:
                    break
                a = generate_powerlaw(n, p)
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.tolist()
        print(f"[baseline] best after param sweeps = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. MAIN loop – alternating LP + mixing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        f = np.array(best_seq, dtype=float)
        it = 0
        # mixing factor schedule – start aggressive, then taper
        while time.time() < deadline:
            it += 1
            # compute current maximal convolution
            conv_f = full_convolution(f)
            M = conv_f.max()

            # solve LP to find a partner g
            g = lp_optimize(f, M)

            # normalise g to a probability vector
            sum_g = g.sum()
            if sum_g <= 0:
                break
            g = g / sum_g

            # mixing weight t – larger early, shrinking later
            if it < 6:
                t = 0.25
            elif it < 20:
                t = 0.12
            else:
                t = 0.05

            # convex combination, keep sum=1 automatically
            f = (1.0 - t) * f + t * g
            # guard against numeric drift
            f = np.maximum(f, 0.0)
            f_sum = f.sum()
            if f_sum == 0:
                break
            f /= f_sum

            # evaluate
            sc = compute_score(f)
            if sc < best_score:
                best_score, best_seq = sc, f.tolist()
                print(f"[LP‑iter {it:3d}] new best score = {sc:.6f}")

            # terminate the loop if improvements have stalled for many iterations
            if it > 30 and best_score - sc < 1e-7:
                # small improvement threshold – likely converged
                break

    # --------------------------------------------------------------
    # 3. OPTIONAL simulated‑annealing polish (cheap, improves few 1e‑4)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, deadline)
        sc = compute_score(final)
        print(f"[anneal] polished score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, final.tolist()

    # --------------------------------------------------------------
    # 4. Fallback – uniform distribution (never reached in practice)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq))

    # --------------------------------------------------------------
    # 5. Official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    official_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {official_score:.6f}")
    return best_seq
