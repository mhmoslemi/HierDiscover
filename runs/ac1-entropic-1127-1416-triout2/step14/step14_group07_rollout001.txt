def fft_convolve(x, y):
    n = len(x) + len(y) - 1
    N = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:n]
    return conv

    # Phase 1: smooth descent
    smooth_deadline = deadline - 300.0  # allocate 300 seconds for smoothing
    if time.time() < smooth_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_smooth = smooth_descent(a0, smooth_deadline)
        val = evaluate_sequence(a_smooth.tolist())
        print(f"[smooth] final score = {val:.6f}")

        a_polished = subgradient_polish(a0, polish_deadline, step0=0.12)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # Phase 3: hill climb
    hill_deadline = deadline - 5.0

        a_start = np.array(best_seq, dtype=float)

"""
search_step_function_v4.py

A multi‑phase optimiser for ``evaluate_sequence`` that builds on the
previous implementation but adds a *smooth‑max* gradient‑descent phase.
The idea is to replace the non‑differentiable max‑operator by a
softmax (log‑sum‑exp) approximation, compute its gradient efficiently
via FFT‑based correlation, and perform projected gradient descent on
the probability simplex.  The smoothing parameter α is increased
gradually, so the method behaves like a sub‑gradient descent on the
original objective but follows a smoother path and can escape the
shallow local minima that limited earlier hill‑climbs.

Phases (time‑scheduled):
  0. Use the known global variable ``height_sequence_1`` as a fallback.
  1. Smooth‑max gradient descent (large budget, main improvement).
  2. Projected sub‑gradient polishing (fine‑tuning, short budget).
  3. Stochastic hill‑climb (final polishing, very short budget).

All vectors are kept normalised (∑a = 1); the objective is
scale‑invariant, so this does not affect the score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        # If everything is non‑positive, return the uniform distribution.
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two arbitrary real vectors using an FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    N = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:n]
    return conv


# ----------------------------------------------------------------------
# Simple analytic families (only reverse power‑law is needed for the fallback)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p   (p ≥ 0).  Result is normalised to sum‑1."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------
def smooth_descent(a_start: np.ndarray, deadline: float,
                   alpha0: float = 0.5,
                   alpha_factor: float = 1.2,
                   alpha_growth_iter: int = 2000,
                   step0: float = 0.4,
                   eval_every: int = 500,
                   jitter_every: int = 15000) -> np.ndarray:
    """
    Projected gradient descent on the smooth approximation
        f_α(a) = max(conv(a,a)) + (1/α)·log Σ exp(α·(conv - max))
    The gradient is 2·correlate(p, a) where p is the soft‑max over the
    convolution entries.  α is increased periodically to tighten the
    approximation to the true max.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = float('inf')
    last_improve = 0
    t = 0
    alpha = alpha0

    while time.time() < deadline:
        t += 1

        # 1) current convolution
        conv = full_convolution(a)
        maxc = conv.max()

        # 2) soft‑max weights (stable computation)
        exps = np.exp(alpha * (conv - maxc))
        p = exps / exps.sum()                         # shape (2n‑1,)

        # 3) gradient = 2·correlate(p, a)   (valid mode → length n)
        grad = 2.0 * np.correlate(p, a, mode='valid')

        # 4) step size (standard 1/√t schedule)
        step = step0 / math.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)

        # 5) increase smoothing parameter periodically
        if t % alpha_growth_iter == 0:
            alpha = min(alpha * alpha_factor, 200.0)

        # 6) occasional jitter to leave plateaus
        if jitter_every and t % jitter_every == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        # 7) sporadic official evaluation
        if t % eval_every == 0:
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val, best_a = val, a.copy()
                last_improve = t
                print(f"[smooth] iter {t:>6d}, new best {val:.6f}")

        # 8) restart if stuck for a long period
        if t - last_improve > 200_000:
            # small random perturbation around the best seen so far
            a = best_a + np.random.randn(n) * 0.0005
            a = project_to_simplex(a)
            last_improve = t
            print(f"[smooth] restart at iter {t}")

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (same as in version 3)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on the true max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (identical to version 3)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """Greedy hill‑climb moving a fraction of mass between two random indices."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

    np.random.seed(seed)
    start_time = time.time()

    n = 2048                                          # fixed length used throughout

    # ------------------------------------------------------------------
    # (0) Fallback – use the known global variable if it exists
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore

    except Exception:

    # ------------------------------------------------------------------
    # (0b) If no fallback, start from a simple analytic shape
    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = a0.tolist()
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    smooth_deadline = deadline - 300.0
    if time.time() < smooth_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_smooth = smooth_descent(a_start, smooth_deadline)
        val = evaluate_sequence(a_smooth.tolist())
        print(f"[smooth] final official score = {val:.6f}")
        if val < best_val:

    # ------------------------------------------------------------------
    # (2) Sub‑gradient polishing – fine‑tuning
    # ------------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.12)
        val = evaluate_sequence(a_polished.tolist())

    # ------------------------------------------------------------------
    # (3) Final stochastic hill‑climb – last few seconds
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")
    return best_seq
