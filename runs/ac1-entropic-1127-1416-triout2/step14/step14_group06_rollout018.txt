"""
Propose candidate step sequence using extensive parametric families,
random Dirichlet sampling, subgradient polishing, and multi‑restart
hill‑climbing. Designed to improve on previous best (~1.51) within a
1000 s budget.

import time
import math
import numpy as np

# Helper: projection onto simplex (sum=1, nonnegative)
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto probability simplex."""
    if np.all(v <= 0):
        # fallback to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a * a using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(a,a)) for a normalized to sum=1.
    Lower is better.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Families of parametric shapes
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, 0, -1, dtype=np.float64)
    a = idx ** p
    a /= a.sum()
    return a

def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a

def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a

def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    if not 0.0 <= mix <= 1.0:
        return None
    rp = reverse_power_law(n, p)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    return project_to_simplex(a)

def double_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a /= a.sum()
    return a

def two_sided_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p + (n-i)^p (p ≥ 0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a /= a.sum()
    return a

def logarithmic_shape(n: int) -> np.ndarray:
    """a_i ∝ -log((i+0.5)/n) (shifted to be non‑negative)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = -np.log(x)
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Sub‑gradient polishing (projected)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # gradient: for each active index k, add 2*a shifted by k
        g = np.zeros_like(a)
        for k in active:
            # compute contribution of a shifted by k (with zero padding)
            # we can do via slice
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        if t % 2000 == 0:
            # random jitter to escape shallow plateaus
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    print(f"[sub‑grad] finished, best max conv = {best_M:.8f}")
    return best_a

# ----------------------------------------------------------------------
# Stochastic hill‑climb with restarts
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 150_000) -> np.ndarray:
    """Greedy hill‑climb that periodically restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            # restart around the best found so far
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    print(f"[hill‑climb] finished after {it} iterations, best max conv = {best_M:.8f}")
    return best_a

# ----------------------------------------------------------------------
# Multi‑restart hill‑climb wrapper
# ----------------------------------------------------------------------
def multi_restart_hill_climb(best_vec: np.ndarray, deadline: float) -> np.ndarray:
    """Run several short hill‑climbs from diverse starts."""
    n = best_vec.shape[0]
    global_best = best_vec.copy()
    global_best_val = compute_score(global_best)

    while time.time() < deadline - 2.0:
        # pick a random start: either best with jitter or a fresh random Dirichlet
        if np.random.rand() < 0.6:
            start = global_best + np.random.randn(n) * 0.001
            start = project_to_simplex(start)
        else:
            # sparse random start
            rnd = np.random.gamma(0.3, 1.0, size=n)
            start = rnd / rnd.sum()
        # allocate a modest slice of time for this climb
        sub_deadline = min(deadline, time.time() + 8.0)  # ~8 s per climb
        cand = hill_climb(start, sub_deadline)
        cand_val = compute_score(cand)
        if cand_val < global_best_val:
            global_best_val, global_best = cand_val, cand
            print(f"[multi‑restart] new global best = {global_best_val:.6f}")

    return global_best

    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0  # safety margin

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Try fallback if present
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose length
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 250 else 2048
    print(f"[setup] using vector length n = {n}")

    # -----------------------------------------------------------------
    # 2. Broad parametric sweep
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val_raw = compute_score(a_initial)
            val_real = evaluate_sequence(a_initial.tolist())
            print(f"[parametric sweep] raw score = {val_raw:.6f}, official = {val_real:.6f}")
            if val_real < best_val:
                best_val, best_seq = val_real, a_initial.tolist()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (use whichever best we have)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.15)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Multi‑restart hill‑climb (aggressive exploration)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 4.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = multi_restart_hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[final hill‑climb] final official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best = mixed_reverse_uniform(n, mix=0.7, p=1.2)
        best_seq = best.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback generation] score = {best_val:.6f}")

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")
    return best_seq

Propose a step‑function sequence that minimizes ``evaluate_sequence``.

The optimiser combines
* an extensive parametric sweep over many analytic families,
* random Dirichlet sampling,
* a projected sub‑gradient polish,
* a multi‑restart stochastic hill‑climb.

The search respects the 1000 s budget, uses a single CPU and never
produces negative or NaN entries.

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(conv(a,a))
    Lower is better.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}   (front‑loaded decreasing power‑law), p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)
    a = idx ** p
    a /= a.sum()
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """convex combination: mix·reverse_power_law + (1‑mix)·uniform."""
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    return project_to_simplex(a)


def double_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends: e^{-lam·i}+e^{-lam·(n‑1‑i)}."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a /= a.sum()
    return a


def two_sided_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{p} + (n‑i)^{p} (p ≥ 0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a /= a.sum()
    return a


def logarithmic_shape(n: int) -> np.ndarray:
    """a_i ∝ –log((i+0.5)/n).  Shifted to be non‑negative."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = -np.log(x)
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a

# Projected sub‑gradient polishing

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # gradient contribution of every active index
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        # occasional jitter to escape shallow plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    print(f"[sub‑grad] best max conv = {best_M:.8f}")
    return best_a

# Stochastic hill‑climb with restarts
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 150_000) -> np.ndarray:
    """Greedy hill‑climb that restarts after prolonged stagnation."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    print(f"[hill‑climb] finished after {it} iters, best max conv = {best_M:.8f}")
    return best_a

def multi_restart_hill_climb(best_vec: np.ndarray, deadline: float) -> np.ndarray:
    """Run a series of brief hill‑climbs from diverse starts."""
    n = best_vec.shape[0]
    global_best = best_vec.copy()
    global_best_val = compute_score(global_best)

    while time.time() < deadline - 2.0:
        if np.random.rand() < 0.6:
            start = global_best + np.random.randn(n) * 0.001
            start = project_to_simplex(start)
        else:
            # a fresh, possibly sparse Dirichlet start

            start = rnd / rnd.sum()

        if cand_val < global_best_val:
            global_best_val, global_best = cand_val, cand
            print(f"[multi‑restart] new global best = {global_best_val:.6f}")

    return global_best
