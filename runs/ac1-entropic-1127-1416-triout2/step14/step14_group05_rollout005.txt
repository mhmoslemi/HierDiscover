def simulated_annealing(a0, deadline, init_temp=0.01, final_temp=1e-6, decay=0.995):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    cur_val = best_val
    t = 0
    temp = init_temp
    n = len(a)
    while time.time() < deadline:
        # propose move
        i = np.random.randint(n)
        if a[i] < 1e-15:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # choose delta fraction of a[i], maybe uniform random up to 0.5?
        delta = a[i] * np.random.rand() * 0.5  # up to half of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_score(cand)
        # acceptance
        if val < cur_val or np.random.rand() < np.exp(-(val - cur_val) / max(temp, 1e-12)):
            a = cand
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = cand.copy()
        # update temp
        if t % 1000 == 0:
            temp = max(final_temp, temp * decay)
        t += 1
    return best_a
```

"""
Non‑negative step‑function optimiser.

The optimiser works in three stages:

1. Initialise a good start.
   * If a known construction ``height_sequence_1`` is present we use it.
   * Otherwise a uniform vector is taken and a short burst of random
     Dirichlet seeds is used to find a better seed.

2. Simulated‑annealing (SA) on the exact objective
   `2·n·max(conv(a))`.  Each SA step proposes one of three moves:
      – targeted mass transfer (move mass away from entries that
        contribute most to the current peak of the convolution);
      – random mass transfer;
      – a simple swap of two entries.
   The move is accepted if it improves the score or with a probability
   depending on a temperature that is slowly cooled.  The SA loop runs
   until the time budget is exhausted, keeping the best vector seen.

3. The final vector is returned as a plain Python list.  The official
   ``evaluate_sequence`` function is called once at the end for a
   sanity check and to report the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Guarantees v_i >= 0 and sum(v) == 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (real FFT implementation).
    Length of result is 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score_and_conv(a: np.ndarray):
    """
    Exact objective for a normalised vector (∑a = 1).
    Returns (score, convolution).
    """
    conv = full_convolution(a)
    n = a.shape[0]
    score = float(2.0 * n * conv.max())
    return score, conv


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser
# ----------------------------------------------------------------------


def simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Direct SA on the exact objective.
    Moves:
      1. targeted mass transfer (reduces the contribution to the current
         maximum of the convolution);
      2. random mass transfer;
      3. swap of two entries.
    """
    a = a0.copy()
    cur_score, cur_conv = compute_score_and_conv(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]

    # temperature schedule
    T = 0.5               # initial temperature
    T_final = 1e-7
    decay = 0.99995       # per‑iteration multiplicative decay
    steps = 0

    while time.time() < deadline:
        r = np.random.rand()

        # ---------- move generation ----------
        if r < 0.45:        # targeted mass transfer
            k = int(cur_conv.argmax())
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_vals = np.arange(i_min, i_max + 1)
            j_vals = k - i_vals
            # product a_i * a_j for the current peak
            prod = a[i_vals] * a[j_vals]
            src = i_vals[np.argmax(prod)]
            # choose a random destination ≠ src
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1
            delta = a[src] * np.random.rand() * 0.30   # up to 30 % of src
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta

        elif r < 0.90:      # random mass transfer
            src = np.random.randint(n)
            if a[src] <= 1e-14:
                continue
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1
            delta = a[src] * np.random.rand() * 0.30
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta

        else:               # swap two entries
            i, j = np.random.choice(n, size=2, replace=False)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # ---------- evaluation ----------
        cand_score, cand_conv = compute_score_and_conv(cand)

        # ---------- Metropolis acceptance ----------
        if cand_score < cur_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = cand
            cur_score = cand_score
            cur_conv = cand_conv
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                # occasional log
                if steps % 5000 == 0:
                    print(f"[SA] step={steps}, T={T:.2e}, new best={best_score:.6f}")

        # temperature decay
        steps += 1
        T = max(T_final, T * decay)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    t0 = time.time()
    # small safety margin so that the final evaluation can be performed
    deadline = t0 + budget_s - 5.0

    # --------------------------------------------------------------
    # 0. Load an existing construction if available.
    # --------------------------------------------------------------
    fallback = None
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    if fallback is not None:
        a0 = np.array(fallback, dtype=float)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        n = a0.shape[0]
        print(f"[init] using fallback of length {n}")
    else:
        # default length – 2048 was used in earlier experiments
        n = 2048
        a0 = np.full(n, 1.0 / n)
        print(f"[init] no fallback – start with uniform vector of length {n}")

    best_a = a0.copy()
    best_score, _ = compute_score_and_conv(best_a)
    print(f"[init] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Quick Dirichlet restarts to obtain a decent seed.
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:   # keep ~30 s for the main SA
        # draw a random concentration parameter (sparsity control)
        conc = 10 ** np.random.uniform(-1.5, 0.7)   # roughly 0.03 … 5.0
        a = np.random.dirichlet(np.full(n, conc))
        score, _ = compute_score_and_conv(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[seed] improved seed score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Simulated‑annealing refinement.
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[SA] starting simulated annealing")
        best_a = simulated_annealing(best_a, deadline)
        best_score, _ = compute_score_and_conv(best_a)
        print(f"[SA] finished – best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Final report using the official evaluation function.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq
