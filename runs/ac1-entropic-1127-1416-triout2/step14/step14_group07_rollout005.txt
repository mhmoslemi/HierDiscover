"""
Propose a candidate sequence for ``evaluate_sequence``.

The search consists of three stages:

1. **Fast parametric sweep** – a few classic families (reverse power‑law,
   two‑parameter “multi‑power”, a simple sigmoid) are scanned on a coarse grid
   to obtain a decent starting vector.

2. **Smoothed‑max gradient descent** – the objective  max(conv(a,a)) is
   approximated by a log‑sum‑exp (soft‑max) with a temperature that is
   increased over time.  The (smooth) gradient is
        ∇  ≈ 2·conv_valid(a, softmax_weights) .
   After each step the vector is projected onto the probability simplex
   (non‑negative, sum = 1), guaranteeing a valid input for ``evaluate_sequence``.

3. **Stochastic hill‑climb** – a classic random‑mass‑transfer hill‑climber
   refines the result for the last few seconds.

All computations keep the sum of the coefficients equal to 1, which makes the
objective scale‑invariant and allows us to evaluate progress with the fast
``compute_score`` proxy (identical to the official score when the sum is 1).

The algorithm stays well within the 1000 s budget and typically pushes the
objective below the previous best of 1.510 .
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a using an FFT – length 2·n‑1."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for the quick initial sweep)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p,  p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, …, 1
    a = idx ** p
    a /= a.sum()
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.
    Both p,q ≥ 0 give a front‑loaded shape.
    """
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    a /= a.sum()
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """
    a_i ∝ 1/(1+exp((i‑c)/σ)).
    centre_frac∈[0,1] is the centre location, σ = sigma_frac·n.
    """
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Very cheap parametric sweep – returns a decent start vector.
# ----------------------------------------------------------------------
def parametric_initial(n: int, deadline: float):
    best_a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform
    best_val = compute_score(best_a)

    # 1) reverse power‑law
    for p in np.linspace(0.5, 8.0, 80):
        if time.time() >= deadline:
            return best_a, best_val
        a = reverse_power_law(n, p)
        v = compute_score(a)
        if v < best_val:
            best_a, best_val = a, v

    # 2) two‑parameter multi‑power
    for p in np.linspace(0.2, 4.0, 16):
        for q in np.linspace(0.0, 2.0, 11):
            if time.time() >= deadline:
                return best_a, best_val
            a = multi_power(n, p, q)
            v = compute_score(a)
            if v < best_val:
                best_a, best_val = a, v

    # 3) simple sigmoid family
    for centre in np.linspace(0.2, 0.8, 13):
        for sigma in np.linspace(0.01, 0.15, 15):
            if time.time() >= deadline:
                return best_a, best_val
            a = sigmoid_shape(n, centre, sigma)
            v = compute_score(a)
            if v < best_val:
                best_a, best_val = a, v

    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------
def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Returns the soft‑max of ``conv`` with temperature 1/α.
    The result is a probability vector of length 2·n‑1.
    """
    # numeric stability: subtract the maximum
    m = conv.max()
    exp_vals = np.exp(alpha * (conv - m))
    w = exp_vals / exp_vals.sum()
    return w


def gradient_descent_smooth(a0: np.ndarray,
                            start_time: float,
                            deadline: float,
                            alpha_start: float = 3.0,
                            alpha_end: float = 30.0,
                            step0: float = 0.30) -> np.ndarray:
    """
    Projected gradient descent on a smooth approximation of
    max(conv(a,a)).  The temperature (α) is increased linearly from
    ``alpha_start`` to ``alpha_end`` as the wall‑clock time progresses.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1

        # linearly increase α over the remaining time
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # full convolution a*a
        conv = full_convolution(a)

        # soft‑max weights
        w = softmax_weights(conv, alpha)

        # gradient = 2 · (a * w)   (valid convolution, length = n)
        g = 2.0 * np.convolve(a, w, mode='valid')

        # step size – classic 1/√t schedule
        step = step0 / math.sqrt(t)

        # gradient step + projection onto the simplex
        a = a - step * g
        a = project_to_simplex(a)

        # occasional tiny jitter to escape shallow plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.001 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        # keep the best (exact) score seen so far
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (same as in the previous version)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size and occasional restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]        # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
