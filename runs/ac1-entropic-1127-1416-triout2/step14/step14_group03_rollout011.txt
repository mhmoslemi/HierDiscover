"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.

Main ideas
----------
1. **Coarse global baseline** – a very fast sweep of a handful of simple
   parametric families (uniform, power‑law, exponential, Gaussian) and a
   few Dirichlet draws.  The best vector (normalised to sum = 1) is kept
   as a starting point.

2. **Linear‑programming polishing** – the convex‐combination iteration
   described in the literature:
      * fix the current vector f,
      * solve a linear program   maximize  Σ gⱼ   s.t.   f * g ≤ M,
        where M = max(f * f) (the current objective value).
      * rescale g to have the same ℓ¹‑norm as f and replace
        f ← (1‑t)·f + t·g with a step size *t* that is adapted
        automatically.
      This step strictly decreases M while keeping the sum of the
      coefficients equal to 1.  For n≈2048 the LP is only a few MB and
      solves in a few seconds with HiGHS.

3. **Upscaling** – the improved 2048‑vector is linearly interpolated to
   length 4096 and projected back onto the probability simplex.

4. **Smooth Adam optimisation** – minimise a soft‑max surrogate of
   max(f * f) using a log‑softmax parametrisation (the “soft‑max” trick
   from the original code).  The optimiser runs for the remaining time
   with a slowly decaying temperature.

5. **Light simulated‑annealing polish** – a final Metropolis walk to
   escape tiny plateaux.

All stages respect the 1000 s wall‑clock budget and print progress.
The returned list is a plain Python list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear convolution via FFT (real)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b (FFT based)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a) = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(a*a)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for a quick baseline)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linearly interpolate ``vec`` (length n) to length ``new_n``.
    The result is projected back onto the simplex.
    """
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


# ----------------------------------------------------------------------
# Linear‑programming polishing (the “LP‑step”)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1)×n) such that
    ``A @ g`` equals the full convolution ``f * g``.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_improve(f: np.ndarray, max_iter: int, deadline: float) -> np.ndarray:
    """
    Repeatedly apply the LP‑step to ``f`` while time permits.
    Returns the best vector observed (still normalised to sum = 1).
    """
    n = f.shape[0]
    best = f.copy()
    best_M = full_convolution(best).max()
    t = 0.4                     # initial mixing factor

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        M = full_convolution(best).max()
        # ----- build LP matrix -----
        A = build_conv_matrix(best)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g
        bounds = [(0, None)] * n

        # ----- solve LP -----
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as exc:  # pragma: no cover
            print("[LP] linprog exception:", exc)
            break

        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-12:
            # no improvement possible – the current f is already “LP‑optimal”
            break

        # ----- rescale g to unit ℓ¹‑norm -----
        g = g0 / sum_g0

        # ----- convex combination -----
        cand = (1.0 - t) * best + t * g
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()          # re‑normalise to avoid drift

        M_cand = full_convolution(cand).max()

        if M_cand < best_M - 1e-12:
            print(f"[LP] iter {it+1:3d}  M {best_M:.9f} → {M_cand:.9f}  t={t:.3f}")
            best = cand
            best_M = M_cand
            t = min(0.95, t * 1.08)       # accelerate when we make progress
        else:
            # back‑off
            t = max(0.05, t * 0.55)

        # stop early if the mixing factor becomes too small
        if t < 0.04:
            break

    print(f"[LP] final M = {best_M:.9f}")
    return best


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 10.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9995) -> np.ndarray:

    Minimise max_k (a*a)_k using a smooth soft‑max approximation.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)  # log‑softmax parameters

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

        it += 1

        # ----- soft‑max mapping -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = full_convolution(a)
        max_c = conv.max()

        # ----- soft‑max weights -----
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient w.r.t. a -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the soft‑max (x‑space) -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature -----
        tau = max(min_tau, tau * tau_decay)

        # ----- periodic exact evaluation -----
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    best_seq = None
    best_val = float("inf")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Very fast baseline at moderate size (n_small = 2048)
    # -----------------------------------------------------------------
    n_small = 2048
    candidates = []

    # uniform
    candidates.append(np.full(n_small, 1.0 / n_small))

    # a few power‑law exponents
    for p in [0.5, 1.0, 2.0, 3.0, 4.0, 6.0]:
        candidates.append(generate_powerlaw(n_small, p))

    # a few exponentials
    for lam in [0.02, 0.05, 0.10, 0.15, 0.20]:
        candidates.append(generate_exponential(n_small, lam))

    # a few Gaussians
    for sigma in [n_small / 8.0, n_small / 5.0, n_small / 3.0]:
        candidates.append(generate_gaussian(n_small, sigma))

    # some Dirichlet draws (moderate concentration)
    for _ in range(200):
        a = np.random.dirichlet(np.full(n_small, 0.5))
        candidates.append(a)

    # keep the best
    best_f = None
    best_M = float("inf")
    for vec in candidates:
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total == 0:
            continue
        vec = vec / total          # project onto the simplex
        M = full_convolution(vec).max()
        if M < best_M:
            best_M, best_f = M, vec.copy()
    if best_f is None:  # extremely unlikely
        best_f = np.full(n_small, 1.0 / n_small)

    print(f"[baseline] best max‑conv = {best_M:.9f}")

    # -----------------------------------------------------------------
    # 2. Linear‑programming polishing (≈ 200 s)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 500.0           # reserve most time for later stages
    if time.time() < lp_deadline:
        best_f = lp_improve(best_f, max_iter=30, deadline=lp_deadline)
        best_M = full_convolution(best_f).max()
        print(f"[post‑LP] max‑conv = {best_M:.9f}")

    # -----------------------------------------------------------------
    # 3. Upsample to the larger dimension (4096) for fine‑grained polishing
    # -----------------------------------------------------------------
    n_big = 4096
    a_big = upsample(best_f, n_big)
    a_big = np.maximum(a_big, 0.0)
    a_big = a_big / a_big.sum()   # ensure sum = 1

    # -----------------------------------------------------------------
    # 4. Smoothed Adam optimisation on the large problem (≈ 650 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 120.0
    if time.time() < smooth_deadline:
        refined = smooth_optimize(a_big, smooth_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()
    else:
        refined = a_big

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if time.time() < anneal_deadline:
        final = simulated_annealing(np.array(best_seq or refined), anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
