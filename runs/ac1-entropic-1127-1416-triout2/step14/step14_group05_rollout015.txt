"""
Non‑negative step‑function optimiser.

The optimiser proceeds through the following phases (time‑aware):
  1. Initialise a pool of promising vectors:
       – the supplied fallback (if any)
       – uniform probability vector
       – several Dirichlet samples
       – exponential‑decay families (a_i ∝ r^i and mirrored versions)
  2. Run a few Adam‑optimisations on the smooth‑max surrogate
     τ·log∑exp(conv/τ).  The optimiser works on the probability simplex.
  3. Greedy “peak‑mass‑transfer’’ hill‑climb.  At each step the
     current maximal convolution entry is identified; mass is moved
     from the index that contributes most to that peak towards a random
     other index.  Moves are accepted only if they improve the exact
     objective.
  4. A final random mass‑transfer polish (the routine from the reference
     implementation).

All phases stop before the global deadline and always keep the best
solution seen so far.  The final sequence is returned as a plain list of
floats (the evaluation function rescales it internally, so the sum is
kept equal to 1 throughout the search).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

    lr_decay: float = 0.995,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Adam optimiser (projected onto the simplex) for the surrogate

        f_τ(a) = τ·log Σ_k exp(conv_k(a)/τ),

    where conv = a*a.  The function returns the best vector observed
    before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # ----- temperature schedule -----
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # ----- compute smoothed max -----
        conv = full_convolution(a)               # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # (m,)

        # ----- gradient of the surrogate (2·a⋆w) -----
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # ----- Adam update -----
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # ----- bookkeeping -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter={t:6d} τ={tau:.1e} lr={lr:.1e} score={cur_val:.6f}")

        # ----- learning‑rate decay -----
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-6)

        # ----- occasional jitter to avoid stagnation -----
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Greedy peak‑mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def peak_mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Directed hill‑climb: repeatedly locate the index k where the
    convolution attains its maximum, move a fraction of the mass from
    the index that contributes most to that peak towards a random
    other index, accept the move only if the exact objective improves.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        conv = full_convolution(a)
        k = int(np.argmax(conv))                     # position of current peak

        # donor = index i that maximises a[i] * a[k-i] (valid i only)
        if k < n:
            i_candidates = np.arange(k + 1)
        else:
            i_candidates = np.arange(k - (n - 1), n)

        partner = k - i_candidates
        contrib = a[i_candidates] * a[partner]
        donor = i_candidates[np.argmax(contrib)]

        # pick a random receiver different from donor
        receiver = np.random.randint(0, n)
        if receiver == donor:
            continue

        # move a small fraction of donor's mass
        if a[donor] <= 1e-15:
            continue
        frac = np.random.rand() * 0.20                # up to 20 % of donor mass
        delta = a[donor] * frac

        a[donor] -= delta
        a[receiver] += delta

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[peak] iter={it:6d} score={cur_val:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Small random mass‑transfer hill‑climb (as in the reference implementation)
# ----------------------------------------------------------------------


def mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Randomly move a fraction of mass from one index to another."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        frac = np.random.rand() * 0.30
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:
                print(f"[mass] iter={it:6d} score={val:.6f}")
        it += 1
    return best_a


# ----------------------------------------------------------------------
# Helper to generate exponential‑decay vectors
# ----------------------------------------------------------------------


def exponential_vector(n: int, epsilon: float, reverse: bool = False) -> np.ndarray:
    """
    Return a normalised vector a_i ∝ (1‑ε)^{i} (or reversed).
    epsilon = 1‑r,  r close to 1.  The vector is projected onto the simplex
    so the sum equals one.
    """
    r = 1.0 - epsilon
    if not reverse:
        idx = np.arange(n, dtype=float)
    else:
        idx = np.arange(n - 1, -1, -1, dtype=float)
    a = (1.0 - r) * (r ** idx)
    a = a / a.sum()
    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5.0               # small safety margin

    # --------------------------------------------------------------
    # 0. Load the known construction (if any) and initialise a start.
    # --------------------------------------------------------------
    fallback = None
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
    except Exception:
        pass

    n = 2048                                      # default length
    if fallback is not None:
        a0 = np.array(fallback, dtype=float)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        print(f"[init] using fallback of length {a0.shape[0]}")
        n = a0.shape[0]
    else:
        a0 = np.full(n, 1.0 / n)                  # uniform start
        print("[init] no fallback – start with uniform vector")

    best_a = a0.copy()
    best_val = compute_score(best_a)
    print(f"[init] initial score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Build a pool of promising seeds (uniform, Dirichlet, exponentials).
    # --------------------------------------------------------------
    seeds = [best_a]
    seeds.append(np.full(n, 1.0 / n))                # uniform again
    for _ in range(6):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # exponential families – epsilon≈c/n with c∈{0.8,1.2,1.6,2.0}
    for c in (0.8, 1.2, 1.6, 2.0):
        eps = c / n
        seeds.append(exponential_vector(n, eps, reverse=False))
        seeds.append(exponential_vector(n, eps, reverse=True))

    # --------------------------------------------------------------
    # 2. Quick evaluation of all seeds, keep the best.
    # --------------------------------------------------------------
    for a in seeds:
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[seed] improved seed score = {val:.6f}")

    # --------------------------------------------------------------
    # 3. Adam‑based smooth‑max optimisation on a few selected starts.
    # --------------------------------------------------------------
    # we keep a short pool (best so far + a few random Dirichlet)
    adam_starts = [best_a]
    for _ in range(3):
        adam_starts.append(np.random.dirichlet(np.ones(n)))

    remaining = deadline - time.time()
    per_start = remaining / max(len(adam_starts), 1)

    for idx, start_vec in enumerate(adam_starts):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_start - 2.0   # safety margin
        if local_deadline <= time.time():
            continue
        print(f"[adam] start {idx + 1}/{len(adam_starts)} – budget ≈ {per_start:.1f}s")
        a_ref = smooth_max_adam(start_vec, local_deadline,
                                tau_start=1.0, tau_end=1e-8,
                                tau_decay=0.99,
                                lr_start=0.05, lr_decay=0.995)
        val = compute_score(a_ref)
        print(f"[adam] finished start {idx + 1}, score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 4. Directed peak‑mass‑transfer hill‑climb (the main polishing step).
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print("[peak] starting directed peak‑mass‑transfer hill‑climb")
        a_ref = peak_mass_transfer_hill_climb(best_a, deadline - 5.0)
        val = compute_score(a_ref)
        print(f"[peak] finished, score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 5. Final random mass‑transfer polish (as in the reference code).
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[mass] starting final random mass‑transfer polish")
        a_ref = mass_transfer_hill_climb(best_a, deadline - 2.0)
        val = compute_score(a_ref)
        print(f"[mass] finished, score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 6. Report using the official evaluation function.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq
