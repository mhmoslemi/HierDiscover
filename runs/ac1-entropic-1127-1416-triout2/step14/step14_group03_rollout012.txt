def direct_max_optimize(a0, deadline, init_lr=0.1, decay=0.9995):
    a = a0.copy()
    n = len(a)
    # log-softmax param x as before
    eps = 1e-30
    x = np.log(a + eps)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    best_val = compute_score(np.exp(x)/np.exp(x).sum())
    best_a = np.exp(x)/np.exp(x).sum()
    it = 0
    while time.time() < deadline:
        it += 1
        # softmax for a
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        conv = full_convolution(a)
        max_c = conv.max()
        # compute mask of active indices (within tolerance)
        # tolerance relative: maybe 1e-12 absolute
        mask = (conv >= max_c - 1e-15)
        w = mask.astype(float)
        # if no active? then continue (should be at least one)
        # gradient calculation:
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n] / max(1, w.sum())  # average across active.
        # chain rule to x-space
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a
        # Adam update
        m = beta1*m + (1-beta1)*grad_x
        v = beta2*v + (1-beta2)*(grad_x**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat)+eps_adam)
        # evaluate periodically
        if it % 200 == 0:
            val = compute_score(np.exp(x - np.max(x))/np.exp(x - np.max(x)).sum())
            if val < best_val:
                best_val = val
                best_a = np.exp(x - np.max(x))/np.exp(x - np.max(x)).sum()
                print(f"[hard_max_opt] iter {it} val {val:.6f}")
    return best_a

"""
Optimisation of a non‑negative step function for the objective

    2·n·max(conv(a,a))   (with sum(a)=1)

The algorithm consists of three stages:

1. **Broad baseline sweep** – parametric families (power‑law,
   exponential, beta, Gaussian, sine‑power, cosine‑window) and many
   Dirichlet draws (including monotone versions).  The best few
   candidates are retained.

2. **Upscaling & smooth optimisation** – the current best candidate
   is linearly interpolated to a larger dimension (here 8192) and
   refined with Adam on a soft‑max smoothed approximation of the
   ``max`` of the convolution.  Several random restarts are performed
   while time permits.

3. **Hard‑max sub‑gradient refinement** – after the smoothing phase
   the temperature is set to zero and a genuine sub‑gradient of the
   ``max`` is used (average over all active shifts).  Adam on the
   log‑softmax parametrisation drives the solution further down.

A short simulated‑annealing polish is applied at the end and the
final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b (FFT based)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a)=1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(a*a)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·i/(n‑1))**p  (zero at the ends)."""
    i = np.arange(n, dtype=float)
    arr = np.sin(np.pi * i / (n - 1)) ** p
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_cosine_window(n: int) -> np.ndarray:
    """Raised‑cosine window a_i ∝ (1‑cos(2π·i/(n‑1)))/2."""
    i = np.arange(n, dtype=float)
    arr = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) * 0.5
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Helpers for the baseline stage
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise vec, evaluate it and possibly store it among the best."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return
    vec = vec / total  # simplex projection
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Sweeps a rich set of families and Dirichlet draws.
    Returns up‑to ``max_keep`` best vectors (as ndarrays)."""
    best = []   # list of (score, vector)

    # ---- 1) power‑law -------------------------------------------------
    for p in np.linspace(0.05, 8.0, 350):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # ---- 2) exponential ------------------------------------------------
    for lam in np.linspace(0.00005, 0.4, 350):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # ---- 3) beta -------------------------------------------------------
    al_vals = np.linspace(0.2, 20.0, 30)
    be_vals = np.linspace(0.2, 20.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # ---- 4) Gaussian ---------------------------------------------------
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # ---- 5) sine‑power -------------------------------------------------
    for p in np.linspace(0.5, 8.0, 100):
        if time.time() >= deadline:
            break
        _try_candidate(generate_sin_power(n, p), best, max_keep)

    # ---- 6) cosine window -----------------------------------------------
    _try_candidate(generate_cosine_window(n), best, max_keep)

    # ---- 7) Dirichlet draws (both random and monotone) -----------------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 600
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            # descending (monotone) version
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolates ``vec`` to length ``new_n`` and projects to simplex."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray,
                    deadline: float,
                    init_lr: float = 0.09,
                    init_tau_factor: float = 8.0,
                    min_tau: float = 1e-12,
                    tau_decay: float = 0.9999) -> np.ndarray:
    """
    Minimises a smooth approximation of max(conv) using Adam on the
    log‑softmax (x) parametrisation.  Returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise temperature from current max
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- soft‑max mapping ----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---- convolution ----
        conv = full_convolution(a)
        max_c = conv.max()

        # ---- soft‑max weights for the max ----
        if tau < 1e-15:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ---- gradient w.r.t a (g_i = 2·∑_k w_k·a_{k-i}) ----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ---- chain rule to x‑space ----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- anneal temperature ----
        tau = max(min_tau, tau * tau_decay)

        # ---- occasional exact evaluation ----
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient optimisation (τ → 0)
# ----------------------------------------------------------------------


def hard_max_optimize(a0: np.ndarray,
                      deadline: float,
                      init_lr: float = 0.10) -> np.ndarray:
    """
    Adam optimiser on the true max‑convolution (sub‑gradient obtained
    from all active shifts).  Returns the best vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- soft‑max (log‑softmax) ----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---- convolution ----
        conv = full_convolution(a)
        max_c = conv.max()

        # ---- active set (indices attaining the maximum up to tolerance) ----
        tol = max_c * 1e-12 + 1e-15
        mask = (conv >= max_c - tol).astype(float)   # shape 2n‑1
        act = mask.sum()
        if act == 0.0:
            act = 1.0
            mask[np.argmax(conv)] = 1.0

        # ---- sub‑gradient of the max (averaged over the active set) ----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, mask)
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n] / act

        # ---- chain rule to x ----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- periodic exact evaluation ----
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[hard_max] iter {it:6d}  score {val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Very light Metropolis hill‑climb that accepts worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem sizes – coarse search on 2048, refinement on 8192
    # -----------------------------------------------------------------
    n_small = 2048
    n_big = 8192

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 720.0  # reserve most time for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n_small, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Upsample the current best to the larger dimension
    # -----------------------------------------------------------------
    if best_seq is None:
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)

    a0_big = upsample(a0_small, n_big)

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation on the large problem (≈ 650 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if time.time() < polish_deadline:
        # primary run from the up‑sampled best
        refined = smooth_optimize(a0_big, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # further random restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)

            # random Dirichlet start – try both random and monotone shapes
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n_big, conc))
            if np.random.rand() < 0.45:
                a_start = np.sort(a_start)[::-1]  # monotone decreasing

            cand = smooth_optimize(a_start, sub_deadline)
            val = compute_score(cand)
            if val < best_val:
                best_val, best_seq = val, cand.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Hard‑max sub‑gradient refinement (≈ 80 s)
    # -----------------------------------------------------------------
    hard_deadline = deadline - 60.0
    if best_seq is not None and time.time() < hard_deadline:
        a_start = np.array(best_seq, dtype=float)
        refined = hard_max_optimize(a_start, hard_deadline)
        val = compute_score(refined)
        print(f"[hard‑max‑refine] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported evaluation = {final_score:.6f}")
    return best_seq
