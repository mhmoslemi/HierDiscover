"""
Improved optimiser for the “max‑convolution” problem.

Key ideas
---------
* Use a richer baseline (more Dirichlet draws and the whole β‑family,
  especially edge‑heavy shapes).
* Run a *scheduled* Adam optimiser:
      – start with a strong flattening term (large λ) and a large
        smooth‑max temperature τ,
      – gradually decay both λ and τ, ending with a pure max‑objective
        minimisation.
* After Adam, run a Nesterov‑accelerated projected sub‑gradient
  descent on the exact max‑objective for a long polishing phase.
* Finish with a fast simulated‑annealing hill‑climb.
* All steps respect the 1000 s wall‑clock limit and constantly keep the
  best vector seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of ½·∥a*a - vec∥².
    Returns 2·(a ⋆ vec) where ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assumes Σa = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families – used for the cheap global baseline
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_inverse_sqrt(n: int) -> np.ndarray:
    """U‑shaped a_i ∝ 1/√(i+1) + 1/√(n-i)."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global baseline exploration (cheap, many random draws)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 20,
    random_draws: int = 30000,
) -> list[np.ndarray]:
    """Generate a small pool of promising vectors."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta grid
    al_vals = np.linspace(0.02, 2.0, 31)
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # random beta draws
    for _ in range(300):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    # Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # sine‑power sweep
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # edge‑U shape
    try_candidate(generate_inverse_sqrt(n))

    # mixtures of families
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam_mix = np.random.rand()
        try_candidate(mixture(a, b, lam_mix))

    # Dirichlet draws with many concentrations
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Scheduled Adam optimiser (smooth‑max + flattening, λ and τ decay)
# ----------------------------------------------------------------------


def scheduled_adam(
    a0: np.ndarray,
    deadline: float,
    lam_start: float = 0.8,
    lam_end: float = 0.02,
    lam_decay: float = 0.95,
    lam_interval: int = 400,
    tau_start: float = 1.0,
    tau_end: float = 1e-6,
    tau_decay: float = 0.94,
    tau_interval: int = 250,
    lr_start: float = 0.30,
    lr_decay: float = 0.97,
    lr_interval: int = 600,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on the blended surrogate

        L(a) = (1‑λ)·τ·log‑sum‑exp(conv/τ) + λ·½·‖conv – 1/(2n‑1)‖² .

    λ (flatten‑weight) and τ (smooth‑max temperature) decay
    exponentially during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m

    size_fft = 1 << ((3 * n - 3).bit_length())

    # Adam moment estimates
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    lam = lam_start
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # ----- schedules -----
        if t % lam_interval == 0 and lam > lam_end:
            lam = max(lam_end, lam * lam_decay)

        if t % tau_interval == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        if t % lr_interval == 0:
            lr = max(lr * lr_decay, 1e-5)

        # ----- core quantities -----
        conv = full_convolution(a)                     # length m
        shift = conv.max()
        # stable log‑sum‑exp
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()                 # soft‑max weights

        err = conv - target

        # ----- gradients -----
        grad_max = correlation_grad(a, w, size_fft)
        grad_err = correlation_grad(a, err, size_fft)
        grad = (1.0 - lam) * grad_max + lam * grad_err

        # ----- Adam update -----
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # ----- bookkeeping -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Accelerated projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def accelerated_subgradient(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.40,
) -> np.ndarray:
    """Nesterov‑accelerated sub‑gradient descent on max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    a_prev = a.copy()
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        # Nesterov momentum
        momentum = (t - 1) / (t + 2)
        a_mom = a_new + momentum * (a_new - a_prev)
        a_mom = project_to_simplex(a_mom)

        a_prev, a = a, a_mom

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-7,
    cooling_factor: float = 0.9995,
    step_scale_init: float = 0.06,
    step_scale_min: float = 5e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a = a + jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    # problem size – larger n gives a lower theoretical bound and more freedom
    n = 8192

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global cheap baseline (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 150.0)
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=20, random_draws=30000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Multi‑start scheduled Adam optimisation (≈ 550 s)
    # -----------------------------------------------------------------
    # reserve about 200 s for later polishing/annealing
    remaining = deadline - time.time() - 200.0
    if remaining > 40.0 and best_seq is not None:
        starts = [
            np.array(best_seq, dtype=float),
            np.full(n, 1.0 / n),
        ]
        # add a few random Dirichlet starts
        for _ in range(4):
            starts.append(project_to_simplex(np.random.rand(n)))

        for idx, start_vec in enumerate(starts):
            if time.time() >= deadline - 200.0:
                break
            local_deadline = time.time() + min(55.0, remaining / len(starts))
            refined = scheduled_adam(
                start_vec,
                local_deadline,
                lam_start=0.8,
                lam_end=0.02,
                lam_decay=0.955,
                lam_interval=400,
                tau_start=1.0,
                tau_end=5e-7,
                tau_decay=0.94,
                tau_interval=250,
                lr_start=0.30,
                lr_decay=0.97,
                lr_interval=600,
            )
            val = compute_score(refined)
            print(f"[scheduled‑adam {idx}] refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
        print(f"[scheduled‑adam] best after this phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Accelerated sub‑gradient polishing (≈ 200 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = accelerated_subgradient(a0, polish_deadline, step0=0.38)
        val = compute_score(a_polished)
        print(f"[accelerated‑polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (final ≈ 80 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(
            a_start,
            anneal_deadline,
            init_temp=2e-3,
            final_temp=1e-7,
            cooling_factor=0.9995,
            step_scale_init=0.05,
        )
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
