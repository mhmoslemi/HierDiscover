def lp_refine(a, max_iters=10, t=0.5, time_deadline=some):
    a = a.copy()
    sum_a = a.sum()
    for it in range(max_iters):
        if time.time() >= time_deadline:
            break
        n = len(a)
        M = np.max(full_convolution(a))
        # Build A matrix as described (conv matrix with f)
        # Use sparse
        A = build_conv_matrix(a)    # shape (2n-1, n)
        # Solve LP: maximize sum(g) subject to A @ g <= M, g>=0
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0,None)]*n, method='highs')
        if not res.success:
            break
        g = res.x
        # optionally scale g to same sum
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g * (sum_a / sum_g)
        # mix
        a_new = (1 - t) * a + t * g
        # project to nonneg and renorm to sum_a (just to be safe)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum() * sum_a
        # compute new M
        M_new = np.max(full_convolution(a_new))
        # If decrease > tolerance, accept update and continue; else reduce t or break.
        if M_new < M - 1e-9:
            a = a_new
            print(f"[lp_iter] iter {it} M improved {M} -> {M_new}")
        else:
            # maybe reduce t
            t = t * 0.5
            if t < 1e-4:
                break
    return a

    n = len(a)

"""
Improved search: start from a good baseline (previous best or simple
parametric families).  Then apply a convex-programming refinement based on
the linear‑program described in the problem statement (maximising the
L1‑norm of a vector g under the constraint (f*g)_k ≤ M).  The LP is solved
with SciPy’s high‑performance interior‑point method.  After a few LP
iterations we up‑sample the result and perform a short Adam‑type
optimisation to polish the solution.
"""

import time, math, numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# convolution via FFT
def _linear_convolution(a, b):
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A * B
    return np.fft.irfft(C, n_fft)[:m]

def full_convolution(a):
    return _linear_convolution(a, a)

def compute_score(a):
    """Exact objective used by evaluate_sequence (ignores clipping)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

# ========================= Parametric families =========================
def generate_powerlaw(n, p):
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return arr / s

def generate_exponential(n, lam):
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return arr / s

def generate_beta(n, alpha, beta):
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 1.0 / n)
    s = arr.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return arr / s

# ========================= Helpers =========================
def upsample(vec, new_n):
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0:
        return np.full(new_n, 1.0 / new_n)
    return up / total

def _try_candidate(vec, best, max_keep):
    """Normalise, evaluate, keep if good."""
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0:
        return
    vec = vec / total
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])

def baseline_search(n, deadline, max_keep=8):
    best = []
    # power law sweep 0.1..6
    for p in np.linspace(0.1, 6.0, 250):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)
    # exponential sweep
    for lam in np.linspace(0.0001, 0.2, 250):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)
    # beta grid
    al_vals = np.linspace(0.2, 8.0, 20)
    be_vals = np.linspace(0.2, 8.0, 20)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)
    # Dirichlet draws
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws = 400
    for conc in concs:
        for _ in range(draws):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(np.full(n, conc))
            _try_candidate(a, best, max_keep)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)
    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# ========================= LP refinement =========================
def build_conv_matrix(a):
    """Return sparse matrix A with A_{k,i}=a_{k-i} (zero outside). Shape (2n-1,n)."""
    n = a.shape[0]
    # each diagonal d (0..n-1) filled with a[d]
    data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A

def lp_refine(a0, deadline, max_outer=6, inner_t=0.15):
    """
    Perform a few iterations of the convex‑programming improvement.
    Returns a refined vector (still normalised to sum=1).
    """
    a = a0.copy()
    sum_a = a.sum()
    it_out = 0
    while time.time() < deadline and it_out < max_outer:
        it_out += 1
        M = np.max(full_convolution(a))
        # Build matrix
        A = build_conv_matrix(a)
        # Solve LP: max sum(g)  <=> min -1·g
        c = -np.ones_like(a)  # minimize -sum(g)
        b = np.full(A.shape[0], M)
        bounds = [(0, None)] * len(a)
        # SciPy's hi‑gs method can accept sparse A_ub
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print("[lp] LP failed:", res.message)
            break
        g = res.x
        sum_g = g.sum()
        # If LP could not increase L1‑norm, stop
        if sum_g <= sum_a + 1e-12:
            print("[lp] No L1 improvement, sum_g=", sum_g)
            break
        # Scale g back to original sum
        g = g * (sum_a / sum_g)
        # Mix f and g with a modest step size
        t = inner_t
        while t > 1e-4:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum() * sum_a
            M_new = np.max(full_convolution(a_new))
            if M_new < M - 1e-12:
                print(f"[lp] iter {it_out}, step {t:.4f}, M {M:.6e} -> {M_new:.6e}")
                a = a_new
                break
            t *= 0.5
        else:
            # no improvement found
            print("[lp] step size exhausted without M reduction")
            break
    return a

# ========================= Smoothed Adam (reuse) =========================
def smooth_optimize(a0, deadline,
                    init_lr=0.12, init_tau_factor=10.0,
                    min_tau=1e-9, tau_decay=0.9995):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-30
    x = np.log(a + eps)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        conv = full_convolution(a)
        max_c = conv.max()
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum
        # gradient of smooth max
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # len = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        tau = max(min_tau, tau * tau_decay)
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")
    return best_a

# ========================= Simulated annealing =========================
def simulated_annealing(a_start, deadline,
                        init_temp=1e-3, final_temp=1e-6,
                        cooling_factor=0.9995,
                        step_scale_init=0.03, step_scale_min=1e-6):
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val
    T = init_temp
    step_scale = step_scale_init
    it = 0
    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        val = compute_score(cand)
        dval = val - cur_val
        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)
        T = max(final_temp, T * cooling_factor)
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)
    return best_a

# ========================= Main entry point =========================
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function that minimises ``evaluate_sequence``.
    The algorithm combines a quick parametric baseline, a linear‑programming
    refinement (the “LP‑step” described in the problem statement) and a
    short Adam‑based smooth optimisation, finishing with a tiny simulated‑
    annealing polish.
    """
    import numpy as np  # ensure local reference
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 0. Load a fallback solution if present (often the previous best)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Baseline exploration (small size for speed)
    # --------------------------------------------------------------
    n_small = 2048
    baseline_deadline = deadline - 400.0   # leave time for LP + polish
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n_small, baseline_deadline, max_keep=6)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # --------------------------------------------------------------
    # 2. If we still have no candidate, start from uniform distribution
    # --------------------------------------------------------------
    if best_seq is None:
        a_small = np.full(n_small, 1.0 / n_small)
    else:
        a_small = np.array(best_seq, dtype=float)
        # ensure correct length
        if a_small.size != n_small:
            a_small = upsample(a_small, n_small)

    # --------------------------------------------------------------
    # 3. LP refinement (core of our new contribution)
    # --------------------------------------------------------------
    lp_deadline = deadline - 200.0
    if time.time() < lp_deadline:
        refined = lp_refine(a_small, lp_deadline, max_outer=8, inner_t=0.2)
        val = compute_score(refined)
        print(f"[lp] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # --------------------------------------------------------------
    # 4. Upsample to a larger dimension (4096) for a final polish
    # --------------------------------------------------------------
    n_big = 4096
    a_big = upsample(np.array(best_seq, dtype=float), n_big)

    # --------------------------------------------------------------
    # 5. Short smooth optimisation on the large problem
    # --------------------------------------------------------------
    opt_deadline = deadline - 30.0
    if time.time() < opt_deadline:
        refined_big = smooth_optimize(a_big, opt_deadline)
        val = compute_score(refined_big)
        print(f"[smooth‑opt‑final] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined_big.tolist()

    # --------------------------------------------------------------
    # 6. Very brief simulated‑annealing polish
    # --------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        final = simulated_annealing(np.array(best_seq, dtype=float), anneal_deadline)
        val = compute_score(final)
        print(f"[anneal‑final] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # --------------------------------------------------------------
    # 7. Fallback uniform if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 8. Report official evaluation
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq

"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.

The algorithm proceeds in four phases:

1. **Fast baseline** – a short sweep of simple parametric families
   (power‑law, exponential, beta) and a few Dirichlet draws.  The
   best few candidates are kept.

2. **Linear‑programming refinement** – implements the LP described in the
   problem statement: for a current vector `f` (normalised to sum = 1) we
   maximise `∑ g_i` subject to the linear constraints
   `(f * g)_k ≤ M` where `M = max(f * f)`.  The solution `g` is scaled
   back to the same ℓ₁‑norm and mixed with `f`.  Repeating this step a
   small number of times flattens the convolution peak and typically
   pushes the objective below the previous best (≈ 1.510).

3. **Upscaling & smooth Adam polish** – the refined vector is linearly
   interpolated to length 4096 and a short Adam‑style optimisation on a
   soft‑max approximation of the max‑convolution is performed.

4. **Final simulated‑annealing polish** – a lightweight Metropolis walk
   finishes the search.

All stages respect the 1000 s wall‑clock budget and the function returns
a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A * B
    return np.fft.irfft(C, n_fft)[:m]


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)^2) (same as ``evaluate_sequence``)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Simple parametric families (used in the baseline stage)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 1.0 / n)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``vec`` to length ``new_n`` and project onto simplex."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


def _try_candidate(vec: np.ndarray, best, max_keep: int):
    """Normalise ``vec`` onto the simplex, evaluate and keep if it improves."""
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0:
        return
    vec = vec / total
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 6):
    """
    Very light sweep of a few families and Dirichlet draws.
    Returns up‑to ``max_keep`` best vectors.
    """
    best = []
    # 1) power‑law
    for p in np.linspace(0.1, 6.0, 200):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # 2) exponential
    for lam in np.linspace(0.0001, 0.2, 200):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # 3) beta
    al_vals = np.linspace(0.2, 8.0, 15)
    be_vals = np.linspace(0.2, 8.0, 15)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # 4) Dirichlet draws (including sorted version)
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws = 300
    for conc in concs:
        for _ in range(draws):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(np.full(n, conc))
            _try_candidate(a, best, max_keep)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP step” from the statement)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Construct the sparse matrix A with A_{k,i}=a_{k-i} (zero outside
    the valid index range).  Shape = (2·n‑1, n).
    """
    n = a.shape[0]
    # each diagonal d (0…n‑1) is filled with the constant a[d]
    data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)               # 0,1,…,n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def lp_refine(a0: np.ndarray, deadline: float,
              max_outer: int = 8, init_step: float = 0.2) -> np.ndarray:
    """
    Repeatedly solve the LP:
        maximise   ∑ g_i
        subject to  (a * g)_k ≤ M   (for all k)
                    g_i ≥ 0
    where M = max(a * a).  The solution is mixed with the current vector.
    Returns a refined vector (still normalised to sum = 1).
    """
    a = a0.copy()
    total = a.sum()
    outer = 0
    while time.time() < deadline and outer < max_outer:
        outer += 1
        M = np.max(full_convolution(a))
        A = build_conv_matrix(a)

        # LP: maximise sum(g) → minimise -∑g
        c = -np.ones_like(a)
        b = np.full(A.shape[0], M)
        bounds = [(0, None)] * len(a)

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print("[lp] LP failed:", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= total + 1e-12:
            print("[lp] No ℓ₁‑norm improvement (sum_g = {:.6f})".format(sum_g))
            break

        # Scale g back to the original ℓ₁‑norm
        g = g * (total / sum_g)

        # Mix with a modest step size, shrinking if necessary
        t = init_step
        while t > 1e-4:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = a_candidate / a_candidate.sum() * total
            M_new = np.max(full_convolution(a_candidate))
            if M_new < M - 1e-12:
                print("[lp] iter {}, step {:.4f}, M {:.6e} → {:.6e}".format(
                    outer, t, M, M_new))
                a = a_candidate
                break
            t *= 0.5
        else:
            # step size exhausted without improvement → stop
            print("[lp] No improvement found for this outer iteration")
            break
    return a


# ----------------------------------------------------------------------
# Smooth Adam optimisation (soft‑max approximation of max)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 10.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9995) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-30
    x = np.log(a + eps)                     # log‑softmax parametrisation
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0

    while time.time() < deadline:
        it += 1
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = full_convolution(a)
        max_c = conv.max()
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient of the smooth max (see derivation in analysis)
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        tau = max(min_tau, tau * tau_decay)

        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# Light simulated‑annealing polish
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Quick baseline exploration (≈ 200 s)
    # -----------------------------------------------------------------
    n_small = 2048
    baseline_deadline = deadline - 400.0   # reserve time for later stages
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n_small, baseline_deadline, max_keep=6)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Initialise a vector of length n_small
    # -----------------------------------------------------------------
    if best_seq is None:
        a_small = np.full(n_small, 1.0 / n_small)
    else:
        a_small = np.array(best_seq, dtype=float)
        if a_small.size != n_small:
            a_small = upsample(a_small, n_small)

    # -----------------------------------------------------------------
    # 3. Linear‑programming refinement (core improvement)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 200.0
    if time.time() < lp_deadline:
        refined = lp_refine(a_small, lp_deadline, max_outer=8, init_step=0.2)
