def local_hill_climb(a, deadline):
    n = len(a)
    cur = a.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    step = 0.01
    while time.time() < deadline:
        i = np.random.randint(n)
        if cur[i] <= 0: continue
        j = np.random.randint(n)
        if i==j: continue
        delta = step * cur[i]
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # enforce nonnegative and renorm
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s == 0: continue
        cand = cand / s
        sc = compute_score(cand)
        if sc < best_score:

            cur = cand
            cur_score = sc
        else:
            # accept maybe with probability decreasing
            if np.random.rand() < 0.01:
                cur = cand
                cur_score = sc
        # adapt step size
        step = min(step*1.01, 0.5)
    return best

"""
Search for a non‑negative coefficient sequence that minimises
``evaluate_sequence`` (the maximal autocorrelation bound).

Algorithm (≈ 1000 s total):
  0. Load fallback solution if available.
  1. Broad random exploration (Dirichlet, monotone, mixed‑exponential).
  2. Parameter sweeps for exponential, power‑law and Beta families
     (including extreme β‑parameters < 0.2).
  3. Keep the 12 best candidates (score, vector) found so far.
  4. For each retained candidate run a short Adam‑based optimisation on the
     smooth‑max of the convolution (soft‑max temperature annealed to the
     exact max).  The variant uses a smaller learning‑rate and longer
     temperature decay than the reference implementation.
  5. Finish with a tiny simulated‑annealing polishing step.
The best vector found is returned as a plain ``list`` of floats.
"""

import time
import numpy as np
import operator

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    if s == 0.0:
        return np.zeros(n, dtype=float)
    return a / s


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    if s == 0.0:
        return np.zeros(n, dtype=float)
    return a / s


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.zeros(n, dtype=float)
    s = a.sum()
    if s == 0.0:
        return np.zeros(n, dtype=float)
    return a / s


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        return np.zeros(n, dtype=float)
    return a / s


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random vector, sort it decreasingly, and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]               # decreasing
    s = a.sum()
    if s == 0.0:
        return np.zeros(n, dtype=float)
    return a / s


def generate_mixed_exponential(n: int, lam1: float, lam2: float, w: float) -> np.ndarray:
    """Mixture w·exp(-lam1·i)+(1‑w)·exp(-lam2·i)."""
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    s = a.sum()
    if s == 0.0:
        return np.zeros(n, dtype=float)
    return a / s


def generate_mixed_powerlaw(n: int, p1: float, p2: float, w: float) -> np.ndarray:
    """Mixture w·(i+1)^{-p1}+(1‑w)·(i+1)^{-p2}."""
    i = np.arange(1, n + 1, dtype=float)
    a = w * i ** (-p1) + (1.0 - w) * i ** (-p2)
    s = a.sum()
    if s == 0.0:
        return np.zeros(n, dtype=float)
    return a / s

def _try_candidate(vec: np.ndarray, best):
    """Insert ``vec`` into ``best`` if it improves the recorded score.
    ``best`` is a list of (score, vector) sorted by score (smallest first)."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return
    vec = vec / total
    sc = compute_score(vec)

    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort()
    else:
        if sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort()

def baseline_search(n: int, deadline: float):
    """Collect a set of diverse promising candidates."""
    best = []   # list of (score, vector) – sorted by score (smallest first)

    # 1) Dirichlet draws with a range of concentrations
    alphas = [0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 4.0, 8.0]
    draws_per_alpha = 1500
    for conc in alphas:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    # 2) Random monotone vectors
    for _ in range(4000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # 3) Mixed‑exponential families (lightweight random sampling)
    for _ in range(3000):
        if time.time() >= deadline:
            break
        lam1 = np.random.uniform(0.0001, 0.0030)
        lam2 = np.random.uniform(0.0001, 0.0030)
        w = np.random.rand()
        a = generate_mixed_exponential(n, lam1, lam2, w)
        _try_candidate(a, best)

    # 4) Mixed‑power‑law families
    for _ in range(2000):
        if time.time() >= deadline:
            break
        p1 = np.random.uniform(0.05, 4.0)
        p2 = np.random.uniform(0.05, 4.0)
        w = np.random.rand()
        a = generate_mixed_powerlaw(n, p1, p2, w)
        _try_candidate(a, best)

    # 5) Coarse beta sweep (including very small parameters)
    al_vals = np.linspace(0.05, 6.0, 12)
    be_vals = np.linspace(0.05, 6.0, 12)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best)

    # 6) Gaussian sweep (different widths)
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 20)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        a = generate_gaussian(n, sigma)
        _try_candidate(a, best)

    return [vec for sc, vec in best]

def fine_exponential_search(n: int, deadline: float):
    """Dense scan of the exponential family around the empirically good region."""
    best_sc = float("inf")
    best_vec = None
    lam_vals = np.linspace(0.0003, 0.0030, 3500)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        a = generate_exponential(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec

    best_vec = None
    p_vals = np.linspace(0.05, 5.0, 3000)

        if time.time() >= deadline:

        a = generate_powerlaw(n, p)
        sc = compute_score(a)
        if sc < best_sc:

def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.04,
                    init_tau_factor: float = 8.0,
                    min_tau: float = 1e-10,
                    tau_decay: float = 0.9998) -> np.ndarray:
    """
    Adam on the soft‑max variables ``x`` (so that a = softmax(x)).
    The smooth max uses a temperature ``tau`` that is gradually annealed.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution
        conv = full_convolution(a)
        max_c = conv.max()

        # soft‑max weighting of the max
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = w / w_sum

        # gradient wrt a (2·∑_k w_k·a_{k−i})
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)    # length = 3·n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for the logits
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # occasional exact check
        if (it % 250) == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                # tiny progress report
                print("[smooth] iter", it, "score", val)

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,

                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9996,
                        step_scale_init: float = 0.025,

    """Metropolis‑style hill‑climb that can accept worsening moves."""

    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.zeros(n, dtype=float)
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random vector, sort it decreasingly, and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]               # decreasing
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_mixed_exponential(n: int, lam1: float, lam2: float, w: float) -> np.ndarray:
    """Mixture w·exp(-lam1·i)+(1‑w)·exp(-lam2·i)."""
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_mixed_powerlaw(n: int, p1: float, p2: float, w: float) -> np.ndarray:
    """Mixture w·(i+1)^{-p1}+(1‑w)·(i+1)^{-p2}."""
    i = np.arange(1, n + 1, dtype=float)
    a = w * i ** (-p1) + (1.0 - w) * i ** (-p2)
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)

def _try_candidate(vec: np.ndarray, best):
    """Insert ``vec`` into ``best`` if it improves the recorded score.
    ``best`` is a list of (score, vector) sorted by score (smallest first)."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return
    vec = vec / total
    sc = compute_score(vec)

    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort()
    else:
        if sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort()

def baseline_search(n: int, deadline: float):
    """Collect a set of diverse promising candidates."""
    best = []   # list of (score, vector) – sorted by score (smallest first)

    # 1) Dirichlet draws with many concentration values
    alphas = [0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 4.0, 8.0]
    draws_per_alpha = 1500
    for conc in alphas:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    # 2) Random monotone vectors
    for _ in range(4000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # 3) Mixed‑exponential families (light random sampling)
    for _ in range(3000):
        if time.time() >= deadline:
            break
        lam1 = np.random.uniform(0.0001, 0.0030)
        lam2 = np.random.uniform(0.0001, 0.0030)
        w = np.random.rand()
        a = generate_mixed_exponential(n, lam1, lam2, w)
        _try_candidate(a, best)

    # 4) Mixed‑power‑law families
    for _ in range(2000):
        if time.time() >= deadline:
            break
        p1 = np.random.uniform(0.05, 5.0)
        p2 = np.random.uniform(0.05, 5.0)
        w = np.random.rand()
        a = generate_mixed_powerlaw(n, p1, p2, w)
        _try_candidate(a, best)

    # 5) Coarse beta sweep (including very small shape parameters)
    al_vals = np.linspace(0.05, 6.0, 12)
    be_vals = np.linspace(0.05, 6.0, 12)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best)

    # 6) Gaussian sweep (different widths)
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 20)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        a = generate_gaussian(n, sigma)
        _try_candidate(a, best)

    return [vec for sc, vec in best]

        sc, vec = fine_exponential_search(n, sweep_deadline)
        if sc is not None and sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print("[exp‑sweep] new best =", sc)

        if time.time() < sweep_deadline:
            sc, vec = fine_powerlaw_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print("[power‑sweep] new best =", sc)

    # -----------------------------------------------------------------
    # 3. Adam‑based smoothing optimisation (≈ 300 s)
    # -----------------------------------------------------------------
    opt_deadline = deadline - 150.0
    if best_seq is not None and time.time() < opt_deadline:
        start_vec = np.array(best_seq, dtype=float)
        refined = smooth_optimize(start_vec, opt_deadline)
        sc = compute_score(refined)
        print("[smooth‑opt] refined score =", sc)
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

        # extra random restarts while time permits
        while time.time() < opt_deadline:
            remaining = opt_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            conc = np.random.choice([0.2, 0.6, 1.2, 2.5])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print("[smooth‑restart] new best =", sc)

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 80 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(final)
        print("[anneal] final polished score =", sc)
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 6. Official evaluation (the harness will call ``evaluate_sequence``)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print("\nSearch finished. Official reported score =", final_score)
    return best_seq
