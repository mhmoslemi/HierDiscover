"""
step‑function optimiser – targeted max‑conv descent.

The search consists of three stages:

1. **Seeding** – generate several random Dirichlet vectors (moderate
   concentration) together with any supplied fallback.  Keep the best
   one (according to the exact objective).

2. **Targeted sub‑gradient descent** – repeatedly locate the index
   `k` of the maximal convolution entry and move a small amount of
   probability mass from the positions that contribute most to that
   entry to the positions that contribute least.
   The gradient of `max_k (a*a)[k]` w.r.t. `a` is simply
   `g_i = 2·a_{k‑i}` (if the index is valid, otherwise `0`).  By a
   simple transfer of probability mass guided by `g` we obtain a
   monotone decrease of the objective.  The step size is adapted
   automatically.

3. **Polishing** – a short permutation‑swap hill‑climb and a tiny
   random‑mass‑transfer hill‑climb finish the optimisation.

All vectors are kept on the probability simplex (∑a=1, a_i≥0) using a
Euclidean projection.  The whole search respects the 1000 s budget
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (fast for large vectors)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Targeted sub‑gradient descent on the L∞‑norm of the convolution
# ----------------------------------------------------------------------


def max_convolution_info(a: np.ndarray):
    """Return (max_value, argmax_index, full_convolution)."""
    conv = full_convolution(a)
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Reduce `max_k (a*a)[k]` by repeatedly moving probability mass
    from positions that contribute most to the current maximum to
    those that contribute least.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    n = a.shape[0]
    # initial transfer fraction (how much of the source mass we move)
    frac = 0.5

    it = 0
    while time.time() < deadline:
        max_val, k, _ = max_convolution_info(a)

        # gradient of the active convolution term:
        #   g_i = 2 * a_{k - i}   (only when the index is inside the vector)
        g = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        j_range = k - i_range
        g[i_range] = 2.0 * a[j_range]

        # pick the most and least influential coordinates
        i_max = int(g.argmax())
        i_min = int(g.argmin())

        if i_max == i_min:
            # nothing to move – break out
            break

        # amount of probability to move (bounded to stay non‑negative)
        delta = a[i_max] * frac
        if delta < 1e-12:
            # source already empty – shrink step size
            frac *= 0.5
            continue

        # perform the transfer
        a[i_max] -= delta
        a[i_min] += delta

        # evaluate
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # if we improve we can be a bit more aggressive
            frac = min(frac * 1.02, 0.9)
        else:
            # revert – the move was not beneficial
            a[i_max] += delta
            a[i_min] -= delta
            # be more conservative
            frac *= 0.7

        it += 1
        # occasional safety re‑normalisation (numerical drift)
        if it % 1024 == 0:
            a = a / a.sum()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbs (permutation & tiny mass‑transfer)
# ----------------------------------------------------------------------


def permute_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Swap two entries at a time; keep the best permutation."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # perform swap
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        else:
            a[i], a[j] = a[j], a[i]      # revert
        it += 1
        if it % 50000 == 0:
            print(f"[permute] iter={it}, best={best_val:.6f}")
    return best_a


def tiny_mass_transfer(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very small random mass transfers – final polishing."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # transfer at most 5 % of the source mass
        delta = a[i] * (np.random.rand() * 0.05)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
        it += 1
        if it % 20000 == 0:
            print(f"[tiny] iter={it}, best={best_val:.6f}")
    return best_a

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 5.0

    fallback = None

        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        n = a0.shape[0]
        print(f"[init] using fallback of length {n}")
    else:

        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
