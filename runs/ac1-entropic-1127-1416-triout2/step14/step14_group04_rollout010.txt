"""
Propose a coefficient sequence that minimises the max‑convolution bound.

Algorithm (≈ 1000 s total)
-------------------------------------------------------------
1. **Very wide random sampling** – Dirichlet draws with many
   concentration values, plus sorted‑exponential vectors.
2. **Dense parametric sweeps** – exponential, power‑law,
   Gaussian and a very fine beta (α,β) sweep around the
   uniform case (α≈β≈1).  All candidates are evaluated with the
   exact objective 2·n·max(conv).
3. **Adam optimisation on a smooth max** – starts from the best
   candidate of step 2, uses a soft‑max approximation of the
   max‑convolution and anneals the temperature to zero.
4. **Deterministic hill‑climbing** – repeatedly moves a tiny
   amount of mass from the entry that contributes most to the
   current maximal convolution entry to the entry with the
   smallest mass.  Each improvement is kept; step size is
   adaptively increased/decreased.
5. Return the best sequence found (as a plain ``list[float]``).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = rfft(a, n_fft)
    B = rfft(b, n_fft)
    C = A * B
    conv = irfft(C, n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors (all normalised to sum = 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    """Sorted exponential draws (decreasing) and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]               # decreasing
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Adam optimiser on a soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.08,
                    init_tau_factor: float = 5.0,
                    min_tau: float = 1e-12,
                    tau_decay: float = 0.9997) -> np.ndarray:
    """
    Adam optimisation on the *soft* maximum of the convolution.
    Uses the parametrisation a = softmax(x).  Returns the best
    vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)                     # logits

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise temperature from the current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- soft‑max weighting (smooth max) ----------
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient wrt a (2·∑_k w_k·a_{k−i}) ----------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)    # length = 3 n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the logits ----------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature ----------
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact check ----------
        if it % 500 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Deterministic hill‑climbing around the current optimum
# ----------------------------------------------------------------------


def local_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy local search: repeatedly move a tiny amount of probability
    mass from the entry that contributes most to the current maximal
    convolution entry to the entry with the smallest mass.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    # adaptive step size (fraction of the donor mass)
    step_scale = 0.05
    it = 0

    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        max_c = conv.max()
        k = conv.argmax()                     # shift where the maximum occurs

        # indices that actually contribute to conv[k]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i

        # contributions a_i * a_j
        contrib = a[idx_i] * a[idx_j]
        max_pos = np.argmax(contrib)
        i_high = idx_i[max_pos]
        j_high = idx_j[max_pos]

        # pick a low‑mass index (avoid the two high ones)
        low_candidates = np.where(a > 0)[0]
        mask = (low_candidates != i_high) & (low_candidates != j_high)
        if not np.any(mask):
            break
        low_idx = low_candidates[mask][np.argmin(a[mask])]

        # amount to transfer
        eps = step_scale * a[i_high]
        if eps <= 0:
            break

        cand = a.copy()
        cand[i_high] -= eps
        cand[low_idx] += eps
        # safety: clip negatives, renormalise
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            break
        cand /= s

        sc = compute_score(cand)
        if sc < best_sc - 1e-12:          # improvement
            best_sc, best_a = sc, cand.copy()
            a = cand.copy()
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale = max(step_scale * 0.97, 1e-6)

        if it % 2000 == 0:
            print(f"[hill] it {it:7d}  best {best_sc:.6f}  step {step_scale:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # problem size – 2048 gives a good trade‑off between expressivity
    # and speed for the 1000 s budget.
    n = 2048

    best_seq = None
    best_sc = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            sc = evaluate_sequence(fb)               # official evaluation
            # internal normalised score for comparison
            a = np.array(fb, dtype=float)
            a = np.maximum(a, 0.0)
            if a.sum() == 0:
                raise ValueError
            a /= a.sum()
            internal_sc = compute_score(a)
            if internal_sc < best_sc:
                best_sc, best_seq = internal_sc, fb
                print(f"[fallback] official score = {sc:.6f}")
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Very wide random baseline (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0          # leave 10 min for later stages
    if time.time() < baseline_deadline:
        alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
        draws_per_alpha = 4000
        for conc in alphas:
            alpha_vec = np.full(n, conc)
            for _ in range(draws_per_alpha):
                if time.time() >= baseline_deadline:
                    break
                a = np.random.dirichlet(alpha_vec)
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc, best_seq = sc, a.tolist()
                    print(f"[random] new best = {sc:.6f}")

        # sorted exponential vectors
        for _ in range(8000):
            if time.time() >= baseline_deadline:
                break
            a = generate_monotone_random(n)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_seq = sc, a.tolist()
                print(f"[monotone] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Fine parametric sweeps (≈ 200 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    if time.time() < sweep_deadline:
        # exponential
        lam_vals = np.linspace(0.0005, 0.0030, 2500)
        for lam in lam_vals:
            if time.time() >= sweep_deadline:
                break
            a = generate_exponential(n, lam)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_seq = sc, a.tolist()
                print(f"[exp] new best = {sc:.6f}")

        # power‑law
        p_vals = np.linspace(0.05, 5.0, 2500)
        for p in p_vals:
            if time.time() >= sweep_deadline:
                break
            a = generate_powerlaw(n, p)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_seq = sc, a.tolist()
                print(f"[pow] new best = {sc:.6f}")

        # Gaussian
        sigma_vals = np.linspace(n / 64.0, n / 2.0, 40)
        for sigma in sigma_vals:
            if time.time() >= sweep_deadline:
                break
            a = generate_gaussian(n, sigma)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_seq = sc, a.tolist()
                print(f"[gauss] new best = {sc:.6f}")

        # fine beta sweep (α,β around 1 – the most promising region)
        al_vals = np.linspace(0.5, 2.0, 80)
        be_vals = np.linspace(0.5, 2.0, 80)
        for alpha in al_vals:
            for beta in be_vals:
                if time.time() >= sweep_deadline:
                    break
                a = generate_beta(n, alpha, beta)
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc, best_seq = sc, a.tolist()
                    print(f"[beta] α={alpha:.3f} β={beta:.3f} → {sc:.6f}")

    # -----------------------------------------------------------------
    # 3. Adam on the smooth max (≈ 200 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 150.0
    if best_seq is not None and time.time() < smooth_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, smooth_deadline)
        sc = compute_score(refined)
        print(f"[smooth‑opt] refined score = {sc:.6f}")
        if sc < best_sc:
            best_sc, best_seq = sc, refined.tolist()

        # few extra random restarts while time permits
        while time.time() < smooth_deadline:
            remaining = smooth_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            conc = np.random.choice([0.3, 0.7, 1.5, 3.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline)
            sc = compute_score(refined)
            if sc < best_sc:
                best_sc, best_seq = sc, refined.tolist()
                print(f"[smooth‑restart] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Deterministic hill‑climbing polish (≈ 150 s)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        refined = local_hill_climb(a_start, hill_deadline)
        sc = compute_score(refined)
        print(f"[hill‑climb] final polished score = {sc:.6f}")
        if sc < best_sc:
            best_sc, best_seq = sc, refined.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_sc = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 6. Official evaluation (the harness will call ``evaluate_sequence``)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Internal best score = {best_sc:.6f}")
    print(f"Official reported score = {final_score:.6f}")
    return best_seq
