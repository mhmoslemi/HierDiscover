    step = init_step

        k_max = np.argmax(conv)
        max_val = conv[k_max]

Now implement simplex projection:

Algorithm: Given vector v, sort descending. Compute cumulative sum, find rho = max {j | v_j - (cumsum_j - 1) / (j+1) > 0}. Then theta = (cumsum_rho -1)/ (rho+1). Return w = max(v - theta, 0).

Implementation using numpy.

Let's write:

```
def project_simplex(v):
    """Project onto probability simplex (non-negative, sum=1)."""
    v = np.asarray(v, dtype=float)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0 => return uniform vector
        return np.full_like(v, 1.0/len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimise a non‑negative step‑function (probability vector) to minimise
``evaluate_sequence``.  The objective for a vector a with sum = 1 is

    score = 2·n·max(a * a)                (a*a = full linear convolution)

We combine a broad baseline sweep of parametric families (including a new
cosine‑power family) with a cheap up‑sampling, then apply a dedicated
projected sub‑gradient descent that directly reduces the maximal
convolution entry, and finish with a short targeted simulated‑annealing
polish.  The whole pipeline respects the 1000 s wall‑clock budget and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real vectors)
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective for a vector with sum == 1
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """2·n·max(a*a) – the quantity we minimise (n = len(a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used after every update
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=float)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (already used in the previous code)
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_cosine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ |cos(π·i/(n‑1))|^p – high at the edges, low in the centre."""
    i = np.arange(n, dtype=float)
    arr = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_quadratic(n: int, exp: float) -> np.ndarray:
    """a_i ∝ (i·(n‑i+1))^exp  – symmetric bump, tunable flatness."""
    i = np.arange(1, n + 1, dtype=float)
    arr = (i * (n - i + 1)) ** exp
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Helper to evaluate a candidate and keep the best few
# ----------------------------------------------------------------------
def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate and keep it if good."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    tot = vec.sum()
    if tot == 0.0:
        return
    vec = vec / tot
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])

    # ---- 2) exponential ------------------------------------------------
    for lam in np.linspace(0.0001, 0.2, 300):

# ----------------------------------------------------------------------
# Upsampling (linear interpolation + simplex projection)
# ----------------------------------------------------------------------
def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``vec`` to length ``new_n`` and renormalise."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def subgradient_refine(a0: np.ndarray, deadline: float,
                       init_step: float = 0.2,
                       step_decay: float = 0.9995,
                       min_step: float = 1e-8) -> np.ndarray:
    """
    Run a simple projected sub‑gradient descent that directly targets the
    index where the convolution reaches its maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    step = init_step
    it = 0
    while time.time() < deadline:
        it += 1

        # convolution and location of the peak
        conv = full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the peak entry (2·a_{k_max‑i})
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[partner[mask]]

        # candidate move and cheap projection (clip + renorm)
        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            cand = np.full_like(a, 1.0 / n)
        else:
            cand /= s

        cand_score = compute_score(cand)
        cur_score = compute_score(a)

        # accept if improvement, otherwise occasional uphill move
        if cand_score < cur_score:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
            step = min(step * 1.01, 1.0)   # slightly increase step on success
        else:
            # Metropolis‑like acceptance to escape plateaux
            prob = np.exp(-(cand_score - cur_score) / max(step, 1e-12))
            if np.random.rand() < prob:
                a = cand
            step = max(step * step_decay, min_step)

        # occasional random jitter to avoid stagnation
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step * 0.05)
            a = np.maximum(a * np.exp(jitter), 0.0)
            a /= a.sum()

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (targeted moves)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick a source index proportionally to its contribution to the current peak
        conv = full_convolution(a)
        k_max = int(np.argmax(conv))
        contrib = np.zeros_like(a)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib[mask] = a[partner[mask]]
        if contrib.sum() == 0:
            i = np.random.randint(n)
        else:
            prob = contrib / contrib.sum()
            i = np.random.choice(n, p=prob)

        # choose a destination index where a is small
        j = np.random.choice(n, p=(1.0 - a) / (n - a.sum()))

        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            # small random perturbation to shake the system
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a

        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
