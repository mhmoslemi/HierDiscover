def improve_by_lp(a, remaining_time, max_iter=30):
    best = a.copy()
    best_val = compute_score(a)
    t = 0.2 # mixing proportion initial

    while time.time() < remaining_time and max_iter>0:
        M = full_convolution(a).max()
        # Build matrix
        n = len(a)
        rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
        cols = np.repeat(np.arange(n), n)
        data = np.repeat(a, n)
        A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        c = -np.ones(n)
        ub = np.full(2*n-1, M)
        # solve LP
        res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("LP failed", res.message)
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        g = b / sum_b
        # mixing
        a_new = (1 - t) * a + t * g
        # project onto simplex (ensuring sum 1)
        a_new = project_to_simplex(a_new)
        cur_val = compute_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best = a_new.copy()
        # adapt t: if improvement, keep same; else reduce
        if cur_val < best_val:
            # maybe increase t?
            t = min(t * 1.1, 0.5)
        else:
            t = max(t * 0.9, 0.05)
        a = a_new
        max_iter -= 1

    return best

"""
Hybrid search for a step‑function with a low max‑self‑convolution.
================================================================

1. **Baseline** – cheap random / analytic families, keep the best vector.
2. **LP‑mixing** – repeatedly solve a linear programme
       max ∑b_i  s.t.   A(a)·b ≤ M , b≥0
   where M = max(conv(a,a)) and A(a) is the convolution matrix
   (A_{k,i}=a_i for k∈[i,i+n‑1]).  
   The optimiser uses the LP solution *g* (renormalised) and mixes it
   with the current a: a←(1‑τ)a+τg, τ slowly adapted.  This step
   pushes weight to positions where a_i is small and flattens the
   convolution, lowering the objective.
3. **Sub‑gradient polish** – projected sub‑gradient descent on the true
   max‑convolution objective.
4. **Simulated‑annealing fine tune** – Metropolis hill‑climb for the last
   few seconds.

All steps respect a 1000 s wall‑clock budget, repeatedly keep the best
sequence and finally return it as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, zero‑padding)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using the real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact score (objective with sum(a)=1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(conv(a,a))  (vector assumed to sum to 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for the cheap baseline)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Baseline exploration (fast, cheap)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float) -> np.ndarray:
    """Return a good starting vector (sum = 1)."""
    best_a = np.full(n, 1.0 / n)
    best_score = compute_score(best_a)

    def consider(vec):
        nonlocal best_a, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_a = sc, vec.copy()

    # power‑law sweep
    for p in np.linspace(0.1, 5.0, 40):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.5, 30):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # beta sweep (grid)
    al_vals = np.linspace(0.1, 3.0, 15)
    be_vals = np.linspace(0.1, 3.0, 15)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # random Dirichlet draws
    for _ in range(800):
        if time.time() >= deadline:
            break
        vec = np.random.dirichlet(np.ones(n))
        consider(vec)

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming mixing step
# ----------------------------------------------------------------------


def lp_mix_step(a: np.ndarray, M: float):
    """
    Solve   max ∑b_i   s.t.   A(a)·b ≤ M , b≥0.
    Return the normalised solution g (sum(g)=1) or None on failure.
    """
    n = a.shape[0]
    # build the convolution matrix A where A_{k,i}=a_i for k∈[i,i+n‑1]
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(a, n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                # maximise sum b_i  → minimise –sum b_i
    ub = np.full(2 * n - 1, M)     # upper bound for each convolution entry
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    b = np.maximum(res.x, 0.0)
    s = b.sum()
    if s < 1e-12:
        return None
    g = b / s                      # renormalise to sum = 1
    return g


# ----------------------------------------------------------------------
# Sub‑gradient polish (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.33) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
        # occasional jitter to leave plateaus
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing final hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3, final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a
