def max_adam_refine(a0, deadline, lr_start=0.15, lr_decay=0.97, beta1=0.9, beta2=0.999, eps=1e-8, eps_margin=1e-12):
    a = a0.copy()
    n = a.shape[0]
    m = 2*n-1
    size_fft = 1 << ((3*n-3).bit_length()) # as before
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    best_val = compute_score(a)
    best_a = a.copy()
    t=0
    while time.time() < deadline:
        t+=1
        conv = full_convolution(a)
        max_val = conv.max()
        # mask active values with tolerance
        mask = (conv >= max_val - eps_margin).astype(float)
        # compute gradient of max: sum of active gradients = correlation_grad
        grad = correlation_grad(a, mask, size_fft)
        # Adam
        m_t = beta1*m_t + (1-beta1)*grad
        v_t = beta2*v_t + (1-beta2)*(grad**2)
        m_hat = m_t/(1-beta1**t)
        v_hat = v_t/(1-beta2**t)
        step = lr_start * m_hat/(np.sqrt(v_hat)+eps)
        a = a - step
        # project to simplex
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # optionally reduce lr.
        if t % 500 == 0:
            lr_start = max(lr_start*lr_decay, 1e-5)
    return best_a

"""
Propose a low‑scoring step‑function for the max‑convolution task.

The algorithm proceeds in four stages:
1. **Baseline exploration** – cheap analytic families and many
   Dirichlet draws are projected onto the simplex and the best few
   candidates are kept (≈150 s).
2. **Smooth‑max Adam optimisation** – a blended surrogate of the
   max‑convolution (log‑sum‑exp) plus a flattening term is minimised.
   Several starts are refined (≈200 s total).
3. **Exact‑max Adam refinement** – the true L∞ objective is tackled
   using Adam with a sub‑gradient computed via FFT‑based correlation.
   This gives a fine‑grained improvement (≈300 s).
4. **Final simulated‑annealing polish** – a Metropolis hill‑climb
   with slow cooling extracts the last few bits of improvement.

All vectors are kept on the probability simplex, so the denominator
`sum(a)^2` in `evaluate_sequence` is constant (=1).  The whole
procedure respects the 1000 s wall‑clock budget and prints progress
information.  The returned list is a plain Python list of non‑negative
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (size≈2·len(a))."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of ½·∑(a⋆a·vec)  i.e. 2·(a *vec_rev)  where * is linear convolution.
    `vec` must have length 2·len(a)‑1.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Exact score (objective for a simplex‑normalised vector)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a))  (a assumed to sum to 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used by baseline search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global baseline exploration (many cheap candidates)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 15000,
) -> list[np.ndarray]:
    """Explore analytic families and Dirichlet draws, keep the best `max_keep`."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta grid
    al_vals = np.linspace(0.02, 2.0, 31)
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # random beta draws
    for _ in range(300):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    # Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # sine‑power sweep
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # mixtures of families
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # Dirichlet draws (wide concentration range)
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Combined Adam optimiser (smooth max + flattening)
# ----------------------------------------------------------------------


def combined_adam(
    a0: np.ndarray,
    deadline: float,
    lam: float = 0.15,
    tau_start: float = 0.30,
    tau_end: float = 1e-7,
    tau_decay: float = 0.94,
    lr_start: float = 0.15,
    lr_decay: float = 0.97,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam on a surrogate L = (1‑λ)·τ·log‑sum‑exp(conv/τ) + λ·½·‖conv‑target‖².
    `lam` controls the flattening weight; τ anneals to a tiny value,
    turning the first term into the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m

    size_fft = 1 << ((3 * n - 3).bit_length())
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # schedules
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # convolution & smooth‑max weights
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)      # stable log‑sum‑exp
        w = exp_vals / exp_vals.sum()

        # flattening error term
        err = conv - target

        # gradients
        grad_max = correlation_grad(a, w, size_fft)
        grad_err = correlation_grad(a, err, size_fft)
        grad = (1.0 - lam) * grad_max + lam * grad_err

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑objective (sub‑gradient via FFT)
# ----------------------------------------------------------------------


def max_adam_refine(
    a0: np.ndarray,
    deadline: float,
    lr_start: float = 0.20,
    lr_decay: float = 0.90,
    beta1: float = 0.90,
    beta2: float = 0.990,
    eps: float = 1e-8,
    active_eps: float = 1e-12,
) -> np.ndarray:
    """
    Adam on the true max‑convolution objective.
    The gradient is the correlation of a with the indicator of active
    (max‑attaining) shifts.
    """
    a = a0.copy()
    n = a.shape[0]
    size_fft = 1 << ((3 * n - 3).bit_length())

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    lr = lr_start

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # mask of active constraints (within a tiny tolerance)
        mask = (conv >= max_val - active_eps).astype(float)

        # gradient of the max (sum of sub‑gradients for all active shifts)
        grad = correlation_grad(a, mask, size_fft)

        # Adam update
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # simple learning‑rate schedule
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # occasional jitter
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.0008 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-8,
    cooling_factor: float = 0.9996,
    step_scale_init: float = 0.04,
    step_scale_min: float = 5e-6,
) -> np.ndarray:
    """Metropolis search that can accept non‑improving moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # periodic jitter to keep diversity
        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0       # safety margin

    # Problem size – 2048 gives a good trade‑off between quality & speed
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈150 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 150.0)
    if time.time() < baseline_deadline:
        baseline_candidates = baseline_search(
            n, baseline_deadline, max_keep=20, random_draws=20000
        )
        for vec in baseline_candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Multi‑start combined Adam optimisation (smooth‑max)
    # -----------------------------------------------------------------
    # Reserve ~250 s for the exact‑max refinement + annealing
    time_for_adam = max(0.0, deadline - time.time() - 250.0)
    if time_for_adam >= 30.0 and best_seq is not None:
        # Build a modest pool of start vectors
        starts = [
            np.array(best_seq, dtype=float),
            np.full(n, 1.0 / n),                 # uniform
        ]
        # a few random simplexes
        for _ in range(4):
            starts.append(project_to_simplex(np.random.rand(n)))
        # add the top baseline candidates (if any)
        for vec in baseline_candidates[:5]:
            starts.append(vec)

        per_start = max(1.0, time_for_adam / len(starts))
        for idx, start_vec in enumerate(starts):
            if time.time() > deadline - 250.0:
                break
            local_deadline = time.time() + min(45.0, per_start)
            refined = combined_adam(
                start_vec,
                local_deadline,
                lam=0.12,
                tau_start=0.25,
                tau_decay=0.93,
                lr_start=0.14,
                lr_decay=0.96,
            )
            val = compute_score(refined)
            print(f"[combined‑adam {idx}] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 3. Exact‑max Adam refinement (sub‑gradient via FFT)
    # -----------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        # Leave ~120 s for the final annealing step
        refine_deadline = time.time() + min(remaining - 120.0, 240.0)
        if refine_deadline > time.time():
            a0 = np.array(best_seq, dtype=float)
            refined = max_adam_refine(a0, refine_deadline, lr_start=0.22)
            val = compute_score(refined)
            print(f"[max‑adam] refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing final polish (≈ remaining time)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        anneal_deadline = deadline - 5.0
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(
            a_start,
            anneal_deadline,
            init_temp=1e-3,
            final_temp=1e-8,
            cooling_factor=0.9996,
            step_scale_init=0.04,
        )
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
