def es_local_search(a0, deadline):
    a = a0.copy()
    best_a = a0.copy()
    best_score = compute_score(a) # returns 2*n*max_b

    sigma = 0.05  # initial perturb sigma relative scale
    it = 0
    while time.time()<deadline:
        noise = np.random.randn(len(a))*sigma
        cand = a + noise
        cand = np.clip(cand,0, None)
        cand = cand / cand.sum()
        val = compute_score(cand)
        if val < best_score:
            best_score = val
            best_a = cand.copy()
            a = cand
            sigma *= 1.02
        else:
            sigma *= 0.999
        it+=1
        if it % 5000 == 0:
            # occasional jitter
            jam = np.random.randn(len(a))*0.001
            a = best_a + jam
            a = np.clip(a, 0, None); a=a/a.sum()
    return best_a

    if r < 1e-12:  # r ~ zero => uniform tail
        total = L + tail_len
        A = 1.0 / total
        tail = np.full(tail_len, A)
    else:
        exp_term = np.exp(-r)
        geom_sum = (1 - np.exp(-r * tail_len)) / (1 - exp_term)
        total = L + geom_sum
        A = 1.0 / total
        tail = A * np.exp(-r * np.arange(tail_len))
    head = np.full(L, A)
    a = np.concatenate([head, tail])
    a /= a.sum()  # but already normalized? Actually total was sum before division; but if we divide again safe.
    return a
```

"""
optim_step_function.py

Search for a sequence of non‑negative heights that minimises
``evaluate_sequence``.  The algorithm works in four phases:

1️⃣  A broad but cheap parametric sweep over several low‑dimensional
    families (reverse power, multi‑power, sigmoid, plateau+exp and beta).
    Each family is explored on a coarse grid and then locally refined
    around its current best parameters.

2️⃣  Randomised Dirichlet samples and random convex mixtures of a few strong
    bases to jump out of the parametric basin.

3️⃣  Projected sub‑gradient descent on the exact objective
    (max of the auto‑convolution).  The step‑size decays as 1/√t and we
    occasionally add jitter to avoid stagnation.

4️⃣  A simple (1+1)‑ES style stochastic local search that perturbs the
    current best vector with Gaussian noise, accepts improvements and
    adapts the mutation strength.

All phases respect a global deadline (≈budget_s – 5 s).  The best sequence
found is returned as a plain Python list of floats.
"""

import time

import numpy as np

# ----------------------------------------------------------------------
# Linear algebra helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Proxy objective for a normalised vector (∑a = 1):
        2·n·max(conv(a,a))
    The official evaluate_sequence is scale‑invariant, therefore we keep the
    vector on the simplex throughout the optimisation.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Parametric families

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n−i)^p,   p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.
    Both p,q ≥ 0.
    """
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """
    a_i ∝ 1/(1 + exp((i−c)/σ)).
    centre_frac ∈ (0,1) is the midpoint location,
    sigma_frac >0 controls the steepness (σ = sigma_frac·n).
    """
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^α (1‑x_i)^β,   x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def plateau_exp(n: int, L_frac: float, r: float) -> np.ndarray:
    """
    Front plateau of relative length L_frac followed by a geometric decay
    with rate r (per step).  r≈0 reproduces a pure plateau.
    """
    L = int(max(1, min(L_frac * n, n - 1)))
    tail_len = n - L
    if r < 1e-12:
        # pure plateau → uniform distribution over the whole segment
        total = n
        a = np.full(n, 1.0 / total, dtype=np.float64)
        return a
    # The plateau height A is chosen such that the whole vector sums to 1.
    exp_term = np.exp(-r)
    geom_sum = (1.0 - np.exp(-r * tail_len)) / (1.0 - exp_term)
    total = L + geom_sum
    A = 1.0 / total
    head = np.full(L, A, dtype=np.float64)
    tail = A * np.exp(-r * np.arange(tail_len, dtype=np.float64))
    a = np.concatenate([head, tail])
    # Numerical safety
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Projected sub‑gradient polishing
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.14) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simple (1+1)‑ES style stochastic local search
# ----------------------------------------------------------------------
def es_local_search(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """Gaussian perturbation + elitist selection on the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]

    sigma = 0.03  # initial mutation strength
    it = 0

    while time.time() < deadline:
        noise = np.random.randn(n) * sigma
        cand = a + noise
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            sigma = min(sigma * 1.05, 0.2)
        else:
            sigma = max(sigma * 0.994, 0.001)

        it += 1
        if it % 5000 == 0:
            # occasional restart jitter
            a = best_a + np.random.randn(n) * (sigma * 0.2)
            a = project_to_simplex(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative integer sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    best_seq = None
    best_val = float('inf')
    n = 2048

    # -------------------------------------------------------------

    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore

        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:

    # -------------------------------------------------------------
    # (1)  Parametric sweep – bulk of the time (~250 s)

    # -------------------------------------------------------------
    # (2)  Sub‑gradient polishing – till ~150 s remain
    # -------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.14)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # (3)  Final stochastic ES‑style local search
    # -------------------------------------------------------------
    final_deadline = deadline - 5.0
    if best_seq is not None and time.time() < final_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = es_local_search(a_start, final_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[es‑search] final official score = {val:.6f}")
        if val < best_val:

    # (4)  Safeguard – deterministic analytic fallback

    if best_seq is None:

    # Final report
    # -------------------------------------------------------------
