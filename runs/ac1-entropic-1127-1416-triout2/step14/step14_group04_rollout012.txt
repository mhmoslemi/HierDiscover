def permutation_annealing(a_start, deadline, init_temp=0.01, final_temp=1e-6, cooling=0.9995):
    a = a_start.copy()
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val
    T = init_temp
    while time.time() < deadline:
        # randomly choose two indices i, j
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        # swap
        a[i], a[j] = a[j], a[i]
        val = compute_score(a)
        delta = val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]
        T = max(final_temp, T * cooling)
    return best_a
```

"""
Propose a coefficient sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv is a linear convolution.  
The search consists of five stages

1. **Broad random sampling** – Dirichlet draws with many concentrations,
   random monotone vectors and a coarse sweep of the beta family.
2. **Targeted 1‑D sweeps** – dense scans of the exponential and power‑law
   families (the shape that gave the previous best result).
3. **Smoothed Adam optimisation** – optimises the soft‑max of the
   convolution (a smooth approximation of the max).  The temperature
   schedule is much slower than in the reference implementation, which
   helps to escape shallow local minima.
4. **Permutation annealing** – the multiset of values is kept fixed,
   but their positions are repeatedly swapped (Metropolis rule).  This
   reduces the peak of the convolution by decorrelating large entries.
5. **Final hill‑climb** – a classic simulated‑annealing step that moves
   small amounts of mass between indices.

All stages respect a global wall‑clock budget (≈ 1000 s).  The best
sequence found is returned as a plain list of floats; the harness will
call ``evaluate_sequence`` on it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (the sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    if lam <= 0:
        return np.full(n, 1.0 / n)
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    if p <= 0:
        return np.full(n, 1.0 / n)
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 1.0 / n)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random vector, sort it decreasingly, and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]               # decreasing
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Helper that updates the global best list
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best):
    """Update the (score, vector) list ``best`` if ``vec`` improves it."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        return
    vec = vec / s
    sc = compute_score(vec)
    # keep at most 12 best (small enough for later polishing)
    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Stage 1 – wide random sampling
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float):
    """Collect a diverse set of promising candidates."""
    best = []   # (score, vector)
    # Dirichlet draws with several concentration parameters
    alphas = [0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 4.0, 8.0]
    for conc in alphas:
        for _ in range(1500):
            if time.time() >= deadline:
                return [vec for _, vec in best]
            a = np.random.dirichlet(np.full(n, conc))
            _try_candidate(a, best)

    # Random monotone vectors
    for _ in range(4000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # Coarse beta sweep
    al_vals = np.linspace(0.2, 8.0, 12)
    be_vals = np.linspace(0.2, 8.0, 12)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best)

    # Gaussian sweep
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 25)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        a = generate_gaussian(n, sigma)
        _try_candidate(a, best)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Stage 2 – fine one‑dimensional sweeps
# ----------------------------------------------------------------------


def fine_exponential_search(n: int, deadline: float):
    """Dense scan of the exponential family."""
    best_sc = float("inf")
    best_vec = None
    lam_vals = np.linspace(0.0004, 0.0030, 4000)   # a bit wider than before
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        a = generate_exponential(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec


def fine_powerlaw_search(n: int, deadline: float):
    """Dense scan of the power‑law family."""
    best_sc = float("inf")
    best_vec = None
    p_vals = np.linspace(0.03, 6.0, 3500)
    for p in p_vals:
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec


# ----------------------------------------------------------------------
# Stage 3 – smoothed Adam optimisation (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.08,
                    init_tau_factor: float = 20.0,
                    min_tau: float = 1e-12,
                    tau_decay: float = 0.99996) -> np.ndarray:
    """
    Adam optimiser on the soft‑max of the convolution.
    Uses a slower temperature decay (larger tau) than the reference.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits safely (avoid log(0))
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise temperature from the current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- soft‑max weighting of the max ----------
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient wrt a (2·∑_k w_k·a_{k−i}) ----------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)    # length = 3 n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the logits ----------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature ----------
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact check ----------
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Stage 4 – permutation annealing (re‑ordering the multiset of values)
# ----------------------------------------------------------------------


def permutation_annealing(a_start: np.ndarray, deadline: float,
                          init_temp: float = 0.01,
                          final_temp: float = 1e-6,
                          decay: float = 0.99995) -> np.ndarray:
    """
    Metropolis‑style annealing that swaps two entries of the vector.
    The multiset of values is preserved – only the ordering changes.
    """
    a = a_start.copy()
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    n = a.shape[0]

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # try swapping
        a[i], a[j] = a[j], a[i]
        val = compute_score(a)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            cur_val = val
            if val < best_val:
                best_val, best_a = val, a.copy()
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]

        T = max(final_temp, T * decay)

    return best_a


# ----------------------------------------------------------------------
# Stage 5 – final hill‑climb (small mass transfers)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9997,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 5e-7) -> np.ndarray:
    """Very light SA that moves a tiny amount of mass between two slots."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

    return best_a


# ----------------------------------------------------------------------
# Main entry point – orchestrates the whole pipeline
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time (≈ 1000 s).
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0     # small safety margin

    n = 2048

    best_seq = None
    best_val = float("inf")               # scale‑invariant score (2·n·max(conv))

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] official score = {val:.6f}")
            sc = compute_score(np.array(fb) / np.sum(fb))
            if sc < best_val:
                best_val, best_seq = sc, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Broad random baseline (≈ 180 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0          # leave ~10 min for later stages
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Focused 1‑D sweeps (≈ 150 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    if time.time() < sweep_deadline:
        sc, vec = fine_exponential_search(n, sweep_deadline)
        if sc is not None and sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[exp‑sweep] new best = {sc:.6f}")

        if time.time() < sweep_deadline:
            sc, vec = fine_powerlaw_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[power‑sweep] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation (≈ 200 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if best_seq is not None and time.time() < smooth_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, smooth_deadline)
        sc = compute_score(refined)
        print(f"[smooth‑opt] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

        # extra random restarts while time permits
        while time.time() < smooth_deadline:
            sub_dead = time.time() + 30.0
            conc = np.random.choice([0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_dead)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[smooth‑restart] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Permutation annealing (≈ 250 s)
    # -----------------------------------------------------------------
    perm_deadline = deadline - 150.0
    if best_seq is not None and time.time() < perm_deadline:
        a0 = np.array(best_seq, dtype=float)
        permuted = permutation_annealing(a0, perm_deadline)
        sc = compute_score(permuted)
        print(f"[perm‑anneal] score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, permuted.tolist()

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (≈ 100 s)
    # -----------------------------------------------------------------
    final_deadline = deadline - 5.0
    if best_seq is not None and time.time() < final_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, final_deadline)
        sc = compute_score(final)
        print(f"[final‑anneal] polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 7. Official evaluation (the harness will call ``evaluate_sequence``)
    # -----------------------------------------------------------------
    official = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {official:.6f}")
    return best_seq
