"""
A multi‑stage search for a step‑function with a minimal max‑autocorrelation.
The algorithm explores several problem sizes, builds a cheap pool of
candidate shapes from analytic families and Dirichlet draws, refines the
most promising ones with a smooth‑max Adam optimizer, then C‑gradient
(polished) sub‑gradient descent and a final simulated‑annealing climb.
All steps respect the 1000 s wall‑clock limit and keep the best vector
found so far (projected onto the probability simplex, i.e. sum = 1).

The objective of ``evaluate_sequence`` for a normalized vector a is
    2·n·max(conv(a,a))
so we optimise this exact quantity (``compute_score``) and only call
``evaluate_sequence`` once at the very end.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility – Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    if np.all(v <= 0):
        # degenerate case – return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full linear convolution using real FFT (a*a)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using an FFT of appropriate size."""
    n = a.shape[0]
    m = 2 * n - 1
    # size must be power‑of‑2 for speed
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Correlation‑gradient used by the smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Compute 2·(a ⋆ vec) where ⋆ denotes cross‑correlation.
    ``vec`` length must be 2·len(a)-1 (e.g. a convolution or a soft‑max weight).
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Exact objective – 2·n·max(convolution) for a vector with sum = 1
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact value returned by ``evaluate_sequence`` for a simplex‑normalised a."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used by the cheap baseline generator)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Cheap baseline exploration – many analytic families + Dirichlet draws
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 8,
    random_draws: int = 6000,
) -> list[np.ndarray]:
    """
    Generate a small pool of promising vectors for a given length `n`.
    The pool size is bounded by `max_keep`.  Candidates are projected onto
    the simplex before they are scored.
    """
    best = []                     # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- families ----------------------------------------------------
    # power‑law
    for p in np.linspace(0.1, 5.0, 51):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential
    for lam in np.linspace(0.0001, 0.5, 60):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta (grid)
    al_vals = np.linspace(0.05, 2.0, 25)
    be_vals = np.linspace(0.05, 2.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # beta (random)
    for _ in range(80):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    # Gaussian family
    for sigma in np.linspace(0.1 * n, 0.45 * n, 20):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # sine‑power
    for p in np.linspace(0.1, 8.0, 25):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # mixtures of two families
    for _ in range(120):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # Dirichlet draws (wide concentration range)
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (blended max‑objective / flattening)
# ----------------------------------------------------------------------


def combined_adam(
    a0: np.ndarray,
    deadline: float,
    lam: float = 0.12,
    tau_start: float = 0.4,
    tau_end: float = 1e-7,
    tau_decay: float = 0.93,
    lr_start: float = 0.18,
    lr_decay: float = 0.96,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on a surrogate:
        L(a) = (1‑λ)·τ·log∑exp(conv/τ) + λ·½·‖conv – 1/(2n‑1)‖² .
    The first term drives down the max, the second flattens the whole
    convolution towards the uniform distribution.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m

    size_fft = 1 << ((3 * n - 3).bit_length())

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # schedule updates
        if t % 300 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 600 == 0:
            lr = max(lr * lr_decay, 5e-5)

        # convolution and smooth‑max weighting
        conv = full_convolution(a)
        shift = conv.max()                      # for numerical stability
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()           # soft‑max distribution

        # flattening error term
        err = conv - target

        # gradients (correlation of a with the two weight vectors)
        grad_max = correlation_grad(a, w, size_fft)
        grad_err = correlation_grad(a, err, size_fft)

        grad = (1.0 - lam) * grad_max + lam * grad_err

        # Adam step
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional tiny jitter to escape plateaus
        if t % 2600 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / math.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polish (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.33,
) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient is the sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if t % 3200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill climb (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-8,
    cooling_factor: float = 0.9996,
    step_scale_init: float = 0.05,
    step_scale_min: float = 4e-6,
) -> np.ndarray:
    """
    Metropolis‐type hill‑climb that can accept worsening moves.
    Each move transfers a small fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < math.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # periodic global perturbation
        if it % 9000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

            a_fin = simulated_annealing(a_start, anneal_deadline)
            val = compute_score(a_fin)
            print(f"[anneal {n}] score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_fin.tolist()

        print(f"[overall best after n={n}] score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Global final annealing (if time permits) – works on the best so far
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        final_deadline = deadline - 2.0
        a_final = simulated_annealing(a_start, final_deadline)
        val = compute_score(a_final)
        print(f"[final anneal] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()

    # -----------------------------------------------------------------
    # 3. Safeguard – if everything failed, fall back to uniform distribution
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 4. Final report using the official evaluation routine
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    print(f"Length of the returned sequence: {len(best_seq)}")
    return best_seq
