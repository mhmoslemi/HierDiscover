"""
propose_candidate – Linear‑programming driven search for a step‑function
with a low value of ``evaluate_sequence``.

Algorithm
---------

1.  Start from a known good sequence (the global ``height_sequence_1``)
    or, if unavailable, from a reverse power‑law shape.
2.  Repeatedly solve a linear program:
        maximise   Σ g_i
        subject to  g_i ≥ 0
                    (f * g)_k ≤ M   for all k
    where ``f`` is the current sequence,
          ``M = max(f * f)`` and ``*`` denotes the linear (full) convolution.
    The LP is feasible (e.g. ``g = f``) and usually yields a
    ``g`` with a larger total mass than ``f``.
3.  Mix the current sequence with the LP solution:
        f ← (1‑t)·f + t·g   (projected onto the probability simplex)
    with a geometrically decreasing step‑size ``t``.
    Because the total mass of the mixture is rescaled to 1,
    the bound ``max(f*f)`` typically decreases.
4.  After a few LP iterations (or when the time budget is low) run a
    stochastic hill‑climb to polish the result.
5.  Return the best sequence seen (as a list of floats).

The LP uses ``scipy.optimize.linprog`` (HiGHS solver) and a dense
convolution matrix built from the current ``f``.  All heavy numeric work
is done with NumPy/FFT, which is fast enough for the 2048‑dimensional
vectors we use.

The code respects the 1000‑second budget and prints progress
information for debugging.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        # return the centre of the simplex
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using FFT (fast, O(n log n))."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    This equals the value returned by ``evaluate_sequence`` for the same
    vector because the function is scale‑invariant.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (used only for a quick start)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p, p ≥ 0, normalised to sum 1."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


# ----------------------------------------------------------------------
# Build the convolution matrix needed for the LP:
#   (f * g)_k = Σ_j f_j * g_{k-j}
# For fixed f this is linear in g, hence A @ g = f * g.
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Returns a dense matrix A of shape (2*n-1, n) such that
    A @ g == np.convolve(f, g, mode='full')
    (both arguments are length n).
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j, val in enumerate(f):
        A[j:j + n, j] = val
    return A


# ----------------------------------------------------------------------
# Linear‑programming step: maximise Σg subject to f*g ≤ M
# ----------------------------------------------------------------------
def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve the LP described in the docstring; returns the optimal g or None."""
    n = f.shape[0]
    A = build_conv_matrix(f)                 # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σg ⇒ minimise −Σg
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:          # linprog may raise if the problem is pathological
        return None

    if res.success:
        return np.maximum(res.x, 0.0)   # safety: clip tiny negatives
    return None


# ----------------------------------------------------------------------
# Stochastic hill‑climb (same idea as the previous version but a bit
# tuned – used only as a polishing phase after the LP iterations)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size; stays on the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]      # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.04, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            # a small random restart to escape plateaus
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 8.0   # a few seconds safety margin

    n = 2048                     # moderate dimension – good trade‑off
    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # (0) Try the global fallback if it exists (usually already good)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # (1) If we have no good start, initialise with a reverse power law
    # ------------------------------------------------------------------
    if best_seq is None:
        a0 = reverse_power_law(n, 2.0)
        if a0 is None:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = a0.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[init] reverse_power p=2 score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # (2) LP‑driven improvement loop
    # ------------------------------------------------------------------
    # Convert best_seq back to a NumPy array (sum = 1)
    f = np.array(best_seq, dtype=np.float64)
    f = project_to_simplex(f)

    lp_iter = 0
    t = 0.5                     # initial mixing weight
    while time.time() < deadline - 12.0 and lp_iter < 20:
        M = full_convolution(f).max()
        # Solve LP for a better direction g
        g_raw = solve_lp_for_g(f, M)
        if g_raw is None:
            print("[LP] solver failed – exiting LP loop")
            break

        # Normalise g to the simplex (scaling down preserves feasibility)
        if g_raw.sum() == 0:
            print("[LP] got zero vector – aborting")
            break
        g = g_raw / g_raw.sum()
        g = project_to_simplex(g)

        # Convex combination and projection back onto the simplex
        f_new = (1.0 - t) * f + t * g
        f_new = project_to_simplex(f_new)

        # Evaluate the new candidate (fast proxy)
        proxy_val = compute_score(f_new)          # = 2·n·max conv
        val_new = evaluate_sequence(f_new.tolist())

        print(f"[LP iter {lp_iter:02d}] t={t:.3f} proxy={proxy_val:.6f} official={val_new:.6f}")

        if val_new < best_val:
            best_val, best_seq = val_new, f_new.tolist()
            print(f"   ** new best (LP) = {best_val:.6f}")

        # Accept the move only if it improves the true objective;
        # otherwise keep the old f but still shrink t.
        cur_val = evaluate_sequence(f.tolist())
        if val_new < cur_val:
            f = f_new
        else:
            # still move a little to avoid being stuck
            f = f_new

        # Decrease the mixing weight geometrically
        t = max(0.02, t * 0.85)
        lp_iter += 1

    # ------------------------------------------------------------------
    # (3) Final stochastic hill‑climb polishing
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=np.float64)
        a_final = hill_climb(a_start, hill_deadline, max_iter=1_200_000)
        val = evaluate_sequence(a_final.tolist())
        print(f"[final hill] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # (4) Safeguard – if nothing succeeded, fall back to a uniform shape
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64).tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] uniform shape, score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
