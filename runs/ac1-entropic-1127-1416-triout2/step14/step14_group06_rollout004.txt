assistantfinal**Solution Explanation**

The goal is to minimise  

\[
E(a)=\frac{2n\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the full discrete convolution of the non‑negative vector `a`.
The expression is scale‑invariant, therefore we may (and will) keep the
vector normalised to \(\sum_i a_i=1\); the objective then reduces to  

\[
E(a)=2n\;\max (a*a).

\]

The search consists of three stages

1. **Broad parametric sweep** – families that are known to give good
   shapes (reverse power law, geometric, exponential, Beta‑distribution,
   mixtures with a uniform component).  The sweep is limited only by a
   time‑deadline; every candidate is evaluated with a fast FFT based
   convolution (`compute_score`).

2. **Random mixture search** – a cheap way of exploring a high‑dimensional
   design space.  A candidate is built as a convex combination of a few
   elementary shapes (reverse‑power‑law, geometric, exponential,
   uniform).  Weight vectors are drawn from a Dirichlet distribution.
   The best mixture is kept.

3. **Local refinement** – two complementary polishing phases
   * **Projected sub‑gradient descent** (the same sub‑gradient that
     appears in the reference algorithm, but with a slightly larger
     initial step and occasional jitter).
   * **Stochastic hill‑climb** – moves mass from one index to another,
     always keeping the vector on the probability simplex.  The step‑size
     adapts automatically and a restart with a small random perturbation
     is performed when progress stalls.

All operations are performed on a vector of length `n`.  
If the time budget is large we use `n = 4096`, otherwise `n = 2048`.  
The helper `project_to_simplex` guarantees a feasible vector after every
update.

The final vector is returned as a plain Python list; the official
`evaluate_sequence` is called only for the final reporting (so the whole
search runs on the fast `compute_score`).

"""
search_step_function_v4.py

A diversified optimiser for `evaluate_sequence`.
The algorithm consists of:
  1) an extended parametric sweep over several families of shapes,
  2) a large random‑mixture search,
  3) projected sub‑gradient polishing,
  4) a stochastic hill‑climb with adaptive step‑size and restarts.
All work is done on the probability simplex (∑a_i = 1), therefore the
objective simplifies to 2·n·max(conv(a,a)).
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# 0. Helper utilities
# --------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (O(n log n))."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective when the vector is normalised ( Σa = 1 ) :

        score = 2·n·max(conv(a,a))

    This equals the value returned by `evaluate_sequence` for the same
    normalised vector.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# 1. Families of candidate vectors (all are normalised to sum=1)
# --------------------------------------------------------------

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i) ** p   (front‑loaded decreasing power law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q**i   (0 < q < 1)."""
    if not (0.0 < q < 1.0):
        return None
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i/(n‑1)).  lam > 0."""
    if lam <= 0:
        return None
    t = np.arange(n, dtype=np.float64) / max(1.0, n - 1)
    a = np.exp(-lam * t)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i**α (1‑x_i)**β ,   x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """mix * reverse_power_law(p) + (1‑mix) * uniform,   0 ≤ mix ≤ 1."""
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

def random_mixture_search(n: int, deadline: float,
                          current_best_a: np.ndarray,
                          current_best_val: float) -> (np.ndarray, float):
    """Generate many random convex combinations of simple shapes."""
    # primitive shapes that are cheap to compute
    primitives = [
        ('uniform', lambda: np.full(n, 1.0 / n)),
        ('rp1', lambda: reverse_power_law(n, 1.0)),
        ('rp2', lambda: reverse_power_law(n, 2.0)),
        ('geo09', lambda: geometric(n, 0.9)),
        ('exp1', lambda: exponential_decay(n, 1.0)),
    ]
    # pre‑compute them once
    basic = [func() for _, func in primitives]

    best_a = current_best_a
    best_val = current_best_val

    while time.time() < deadline:
        k = np.random.randint(2, 5)               # 2‑4 components
        idxs = np.random.choice(len(basic), size=k, replace=False)
        weights = np.random.dirichlet(np.ones(k))
        a = np.zeros(n, dtype=np.float64)
        for w, i in zip(weights, idxs):
            a += w * basic[i]
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a
    return best_a, best_val

# --------------------------------------------------------------

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.18) -> np.ndarray:
    """
    Projected sub‑gradient descent on  max(conv(a,a)).
    A slowly decaying step size and occasional jitter help escaping
    shallow local minima.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: for each active shift k add the shifted copy of a
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        # occasional jitter to escape plateaus
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# --------------------------------------------------------------
# 5. Stochastic hill‑climb with restarts
# --------------------------------------------------------------

def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 250_000) -> np.ndarray:
    """Greedy pair‑wise mass transfer with adaptive step size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a fraction of the source mass towards the destination
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] += delta
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            jitter = np.random.randn(n) * 0.0009
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# 6. Main entry point required by the evaluation harness
# --------------------------------------------------------------

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of a few seconds – the harness kills the process
    deadline = start_time + budget_s - 4.0

    best_seq = None
    best_val = float('inf')

    # ----------------------------------------------------------
    # 0. Optional fallback (if a previous construction is present)
    # ----------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ----------------------------------------------------------
    # 1. Choose vector length (larger → finer discretisation)
    # ----------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 200 else 2048
    print(f"[info] using n = {n}")

    # ----------------------------------------------------------
    # 2. Broad parametric sweep
    # ----------------------------------------------------------
    sweep_deadline = deadline - 300.0
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep_extended(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] raw score = {val:.6f}")
            official = evaluate_sequence(a_initial.tolist())
            print(f"[parametric sweep] official score = {official:.6f}")
            if official < best_val:
                best_val, best_seq = official, a_initial.tolist()

    # ----------------------------------------------------------
    # 3. Random mixture search (if we still have time)
    # ----------------------------------------------------------
    mixture_deadline = deadline - 120.0
    if best_seq is not None and time.time() < mixture_deadline:
        a_cur = np.array(best_seq, dtype=float)
        cur_val = compute_score(a_cur)
        a_mix, mix_val = random_mixture_search(n, mixture_deadline, a_cur, cur_val)
        official = evaluate_sequence(a_mix.tolist())
        print(f"[random mixture] official score = {official:.6f}")
        if official < best_val:
            best_val, best_seq = official, a_mix.tolist()

    # ----------------------------------------------------------
    # 4. Sub‑gradient polishing (bulk of remaining time)
    # ----------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.18)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # ----------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # ----------------------------------------------------------
    hill_deadline = deadline - 4.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ----------------------------------------------------------
    # 6. Safeguard – ensure we always return something
    # ----------------------------------------------------------
    if best_seq is None:
        # simple mixed shape as a last resort
        fallback = mixed_reverse_uniform(n, mix=0.85, p=2.0)
        best_seq = fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # ----------------------------------------------------------
    # Final report
    # ----------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
