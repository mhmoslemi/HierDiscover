    best_f = f.copy()

    while time.time() < deadline:
        conv_f = full_convolution(f)
        M = conv_f.max()

        A = build_conv_matrix(f)  # returns csr
        # Right-hand side vector
        b_ub = np.full(2*n - 1, M)
        c = -np.ones(n)  # minimize -sum = maximize sum
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print("[LP] failed", res.message)
            break
        b_vec = res.x
        sum_b = b_vec.sum()
        if sum_b <= 1e-12:
            # no progress
            break
        g_raw = np.sqrt(2 * n) / sum_b * b_vec
        # Choose t maybe adaptive: try a few values
        best_local_score = None
        best_local_f = None
        best_local_t = None
        for t in [0.05, 0.1, 0.2, 0.3]:
            f_candidate = (1 - t) * f + t * g_raw
            f_candidate = f_candidate / f_candidate.sum()
            sc = compute_score(f_candidate)
            if best_local_score is None or sc < best_local_score:
                best_local_score = sc
                best_local_f = f_candidate
                best_local_t = t

        best_f = f.copy()
        iter_count += 1
        print(f"[LP] iter {iter_count} score {best_score:.6f}")
        # stop if improvement less than tolerance?
        if time.time() >= deadline - 1:
            break

"""
search_step_function.py

A multi‑stage optimiser for the max‑convolution bound used in
``evaluate_sequence``.  The pipeline is:

0.  (Optional) use a previously discovered sequence as a warm start.
1.  Very fast random Dirichlet sampling to obtain a decent starting vector.
2.  Iterative linear‑programming refinement:
    – for the current f compute M = max(f*f).
    – solve   maximise   Σ b_j   s.t.   b ≥ 0  and  f * b ≤ M   (linear constraints).
    – scale the solution to the “canonical” L1‑norm √(2 n) and blend a small
      step towards it, re‑normalising to unit sum.
    – repeat while time permits, always keeping the best vector seen.
3.  A short Adam‑style smoothing optimisation on the soft‑max of the
    convolution (the same idea as the original code, but now applied to the
    LP‑refined vector).
4.  A light simulated‑annealing polish.

All stages respect the global 1000 s wall‑clock budget.
The function ``propose_candidate`` returns a list of non‑negative floats
ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using a real FFT.  Both a and b are
    one‑dimensional real arrays.
    """
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (the quantity returned by evaluate_sequence when the
# vector is normalised to sum = 1).
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a)) assuming ``a`` sums to 1.
    The smaller the value, the better.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Helper: construct the convolution matrix for a fixed left vector.
#    (f * b)_i = Σ_j f_j · b_{i‑j}  ↔  A_f b = conv(f,b)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that A @ b == full_convolution(f, b).
    """
    n = f.size
    # row indices: d + j   (d = index in f, j = index in b)
    rows = np.add.outer(np.arange(n, dtype=np.int32),
                        np.arange(n, dtype=np.int32)).ravel()
    # column indices are just the index j repeated for each d
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    # data = f[d] repeated n times
    data = np.repeat(f, n)
    return sparse.csr_matrix((data, (rows, cols)),
                             shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Very fast random baseline (Dirichlet draws)
# ----------------------------------------------------------------------


def baseline_random(n: int, deadline: float) -> np.ndarray:
    """
    Return a vector of length n with sum = 1 that is the best among a
    modest number of Dirichlet draws.  The routine stops early if time runs
    out.
    """
    best_vec = None
    best_sc = math.inf
    # a few concentration parameters give a wide variety of shapes
    alphas = [0.1, 0.5, 1.0, 2.0, 5.0]
    draws_per_alpha = 3000
    for a0 in alphas:
        alpha = np.full(n, a0)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                return best_vec if best_vec is not None else np.full(n, 1.0 / n)
            cand = np.random.dirichlet(alpha)
            sc = compute_score(cand)
            if sc < best_sc:
                best_sc, best_vec = sc, cand.copy()
    # guarantee a return
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)
    return best_vec


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def lp_refine(f0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Starting from the normalised vector ``f0`` (sum = 1), repeatedly
    solve the LP

        maximise   Σ b_j
        subject to  b ≥ 0
                    f * b ≤ M   (M = max(f*f))

    The solution is scaled to the canonical L1‑norm √(2·n) and a short
    convex combination with the current point is taken.  The best vector
    seen before the deadline is returned.
    """
    n = f0.size
    f = f0.copy()
    best_f = f.copy()
    best_sc = compute_score(f)

    # pre‑compute the row/column index pattern that does not depend on f
    rows = np.add.outer(np.arange(n, dtype=np.int32),
                        np.arange(n, dtype=np.int32)).ravel()
    cols = np.tile(np.arange(n, dtype=np.int32), n)

    # main loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current bound
        conv_f = full_convolution(f)
        M = conv_f.max()

        # build the constraint matrix A (data depends on the current f)
        data = np.repeat(f, n)
        A = sparse.csr_matrix((data, (rows, cols)),
                              shape=(2 * n - 1, n))

        # linear programme: maximise sum(b)  <=> minimise -1·b
        c = -np.ones(n, dtype=float)
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # solve
        lp_res = linprog(c, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not lp_res.success:
            # If linprog fails we abort the refinement phase.
            print(f"[LP] iteration {iteration}: failure – {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # scale to the canonical L1‑norm √(2·n)
        scale = math.sqrt(2.0 * n) / sum_b
        g = scale * b

        # try a few mixing coefficients and keep the best improvement
        best_local_sc = math.inf
        best_local_f = None
        for t in (0.05, 0.10, 0.15, 0.20):
            cand = (1.0 - t) * f + t * g
            cand /= cand.sum()
            sc = compute_score(cand)
            if sc < best_local_sc:
                best_local_sc, best_local_f = sc, cand.copy()

        # if none of the candidates improve the global best, we stop early
        if best_local_sc >= best_sc - 1e-12:
            # no improvement – the LP cannot push the bound further
            break

        # accept the best local move
        f = best_local_f
        best_sc, best_f = best_local_sc, f.copy()
        print(f"[LP] iter {iteration:3d} – score {best_sc:.6f}")

        # occasional safety break if improvement stalls
        if iteration >= 12:
            # after a dozen successful steps we limit the total number;
            # the heuristic prevents spending all time in LP mode.
            if time.time() > deadline - 30.0:
                break

    return best_f


# ----------------------------------------------------------------------
# Soft‑max Adam optimiser (identical in spirit to the original code)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.07,
                    init_tau_factor: float = 4.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9995) -> np.ndarray:
    """
    Adam optimisation on a soft‑max (log‑sum‑exp) surrogate for the max
    convolution entry.  The vector is always kept on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)                # logits

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution and soft‑max weighting of the max
        conv = full_convolution(a)
        s = conv.max()
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - s) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient wrt a (2·∑_k w_k·a_{k-i})
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain‑rule for logits
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        tau = max(min_tau, tau * tau_decay)

        # occasional exact check
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:6d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional final touch)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """A very light Metropolis‑style hill‑climb able to escape tiny plateaus."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < math.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter to keep the walk mobile
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    """

    start_time = time.time()
    # safety margin for the harness (5 s)

    # ------------------------------------------------------------------

                best_sc = compute_score(best_vec)
                print(f"[fallback] initial score = {best_sc:.6f}")
            else:
                best_vec = None
                best_sc = math.inf
        except Exception:
            best_vec = None
            best_sc = math.inf
    else:
        best_vec = None
        best_sc = math.inf

    # ------------------------------------------------------------------

        sc = compute_score(cand)
        if sc < best_sc:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        # a few quick random restarts while time permits
        while time.time() < smooth_deadline:

            sub_deadline = time.time() + min(25.0, remaining)
            a0 = np.random.dirichlet(np.full(n, 0.7))

            sc = compute_score(refined)
            if sc < best_sc:
                best_sc, best_vec = sc, refined.copy()
                print(f"[smooth‑restart] new best = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 30 s)
    # ------------------------------------------------------------------
    anneal_deadline = deadline
    if best_vec is not None and time.time() < anneal_deadline:
        final = simulated_annealing(best_vec, anneal_deadline)

        if sc < best_sc:

            print(f"[anneal] final best = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 6. Official evaluation (the harness will call ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return best_vec.tolist()
