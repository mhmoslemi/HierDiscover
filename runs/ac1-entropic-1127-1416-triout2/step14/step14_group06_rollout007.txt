"""
search_step_function_v4.py

A new optimiser for ``evaluate_sequence``.

Key ideas
---------
*   More diverse parametric families (truncated reverse‐power‑law,
    finer geometric sweep, Dirichlet front‑loads) to obtain a strong
    starting point.
*   A *smooth* gradient descent on a soft‑max approximation of the
    max‑convolution.  The objective
        f(a) = (1/γ)·log Σ_k exp(γ·(a*a)[k])
    is a differentiable surrogate of the original max.  Its gradient can
    be computed cheaply via two FFT‑based convolutions.  An Adam‑style
    update together with projection onto the probability simplex drives the
    search efficiently.
*   The smooth phase is followed by a short stochastic hill‑climb that
    moves small mass packets between coordinates – this removes the last
    tiny bumps that the smooth optimiser can miss.
*   Time is divided dynamically: a quick parametric sweep (≈15 % of the
    budget) finds a good seed, the smooth Adam optimiser consumes the
    bulk of the time, and the final hill‑climb cleans up the result.

The routine ``propose_candidate`` returns a plain Python list of floats
that respects all safety checks of ``evaluate_sequence``.
"""

import time

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    The returned vector is non‑negative and sums to 1.
    """
    if np.all(v <= 0):
        # completely non‑positive – fall back to the uniform simplex point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of two real vectors using an FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:n]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Fast linear (full) convolution a * a via FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law), p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def truncated_reverse_power_law(n: int, p: float, frac: float) -> np.ndarray:
    """
    Front‑loaded power‑law truncated after a fraction `frac` of the vector.
    Only the first m = floor(frac·n) entries are non‑zero.
    """
    m = max(1, int(frac * n))
    idx = np.arange(m, 0, -1, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = idx ** p
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    convex combination   mix * reverse_power_law(p) + (1‑mix) * uniform,
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Parametric sweep – cheap but broad
# ----------------------------------------------------------------------
def parametric_sweep_extended(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    A quick sweep over several families.
    Returns (best_vector, its compute_score) or (None, inf) on timeout.
    """
    best_vec = None
    best_val = float('inf')

    # ---- reverse power‑law (coarse) ----
    for p in np.arange(0.1, 12.0 + 1e-12, 0.2):
        if time.time() >= deadline:
            return best_vec, best_val
        a = reverse_power_law(n, float(p))
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- truncated reverse power‑law ----
    for frac in np.linspace(0.05, 0.5, 9):
        for p in np.arange(0.1, 12.0, 0.5):
            if time.time() >= deadline:
                return best_vec, best_val
            a = truncated_reverse_power_law(n, p, frac)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- geometric sweep ----
    for q in np.linspace(0.5, 0.999, 30):
        if time.time() >= deadline:
            return best_vec, best_val
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- mixed reverse‑uniform ----
    for mix in np.linspace(0.1, 0.9, 9):
        for p in np.linspace(0.5, 5.0, 22):
            if time.time() >= deadline:
                return best_vec, best_val
            a = mixed_reverse_uniform(n, mix, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- random Dirichlet (front‑loaded) ----
    # small concentration gives sparse, front‑loaded vectors.
    for conc in (0.05, 0.1, 0.2, 0.5):
        for _ in range(150):
            if time.time() >= deadline:
                return best_vec, best_val
            a = np.random.dirichlet(np.full(n, conc))
            # force a monotone decreasing shape (helps the structure)
            a = np.sort(a)[::-1]
            a = project_to_simplex(a)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- random convex combos of a few basic shapes ----
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rp2 = reverse_power_law(n, 2.0)
    geom = geometric(n, 0.97)
    basics = [uniform, rp2, geom]

    for _ in range(1000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec, best_val


# ----------------------------------------------------------------------
# Smooth Adam‑style descent on a soft‑max surrogate of max(conv)
# ----------------------------------------------------------------------
def smooth_adam_descent(a0: np.ndarray, deadline: float,
                        gamma_start: float = 5.0,
                        gamma_end: float = 80.0,
                        lr0: float = 0.5) -> np.ndarray:
    """
    Adam optimiser on the smooth surrogate
        fγ(a) = (1/γ)·log Σ_k exp(γ·(a*a)[k])
    The algorithm projects onto the probability simplex after each step.
    """
    a = project_to_simplex(a0.copy())
    best_a = a.copy()
    best_M = full_convolution(a).max()

    n = a.shape[0]
    m1 = np.zeros_like(a)  # first moment
    m2 = np.zeros_like(a)  # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    iteration = 0

    start = time.time()
    total_allowed = deadline - start

    while time.time() < deadline:
        iteration += 1
        now = time.time()
        elapsed = now - start
        frac = elapsed / total_allowed
        # schedule for γ and learning‑rate
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        lr = lr0 * (1.0 - frac) + 0.001   # never drop to zero

        # ----- soft‑max weights -----
        conv = full_convolution(a)                 # length 2n‑1
        c_max = conv.max()
        w = np.exp(gamma * (conv - c_max))         # stable exponentiation
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (2·a_rev * w) -----
        a_rev = a[::-1]
        grad_full = fft_convolve(a_rev, w)         # length 3n‑2
        # slice the middle n entries (indices n‑1 … 2n‑2)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # ----- Adam update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - beta1 ** iteration)
        m2_hat = m2 / (1.0 - beta2 ** iteration)

        step = lr * m1_hat / (np.sqrt(m2_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # ----- keep best ----- 
        M = full_convolution(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

        # ----- occasional jitter to escape plateaus -----
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * (0.0004 * (1.0 - frac))
            a = best_a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (local refinement)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 150_000) -> np.ndarray:
    """Greedy stochastic mass‑transfer moves, with occasional restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            jitter = np.random.randn(n) * 0.0006
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid overshooting the external timeout
    deadline = start_time + budget_s - 4.0

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0. Optional fallback (if a previous construction is present)
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Choose vector length – longer vectors give finer granularity
    # -------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 500:
        n = 8192
    elif remaining > 250:
        n = 4096
    else:
        n = 2048
    print(f"[info] vector length selected = {n}")

    # -------------------------------------------------------------
    # 2. Quick parametric sweep to obtain a decent seed
    # -------------------------------------------------------------
    sweep_deadline = time.time() + min(120.0, (deadline - time.time()) * 0.25)
    a_seed, seed_score = parametric_sweep_extended(n, sweep_deadline)
    if a_seed is not None:
        official_score = evaluate_sequence(a_seed.tolist())
        print(f"[parametric sweep] raw score = {seed_score:.6f}")
        print(f"[parametric sweep] official score = {official_score:.6f}")
        if official_score < best_val:
            best_val, best_seq = official_score, a_seed.tolist()

    # -------------------------------------------------------------
    # 3. Smooth Adam descent (bulk of remaining time)
    # -------------------------------------------------------------
    if best_seq is not None:
        a0 = np.array(best_seq, dtype=float)
    else:
        # fallback to a simple mixed shape
        a0 = mixed_reverse_uniform(n, mix=0.85, p=1.8)

    smooth_deadline = deadline - 80.0          # keep ~1 min for final hill‑climb
    if time.time() < smooth_deadline:
        a_smooth = smooth_adam_descent(a0, smooth_deadline)
        val = evaluate_sequence(a_smooth.tolist())
        print(f"[smooth Adam] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_smooth.tolist()

    # -------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # -------------------------------------------------------------
    hill_deadline = deadline - 4.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # 5. Ultimate fallback (should never be hit)
    # -------------------------------------------------------------
    if best_seq is None:
        best_seq = mixed_reverse_uniform(n, mix=0.8, p=1.7).tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final report
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
