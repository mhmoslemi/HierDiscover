Pseudo:

```
def project_simplex(v):
    # Project onto simplex {x>=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Search for a non‑negative step‑function (vector) that minimises
``evaluate_sequence``.  The programme consists of

0.  Helper utilities (FFT convolution, simplex projection).
1.  A broad “baseline” sweep over many parametric families and many
    Dirichlet draws – the K best candidates are kept.
2.  Upsampling of the current best to a larger dimension.
3.  Two complementary refinements:
        • Smoothed Adam optimisation on the soft‑max (log‑sum‑exp)
          approximation of the max‑convolution.
        • Projected sub‑gradient descent on the exact max‑convolution.
4.  A final long simulated‑annealing polish.
All stages respect a global wall‑clock deadline (≈1000 s) and always
keep the best vector seen so far.  The returned list of floats is a
plain Python list suitable for the external ``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 0.  Low‑level helpers
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT (real‑to‑real)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used many times)."""
    return _linear_convolution(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective assumed sum(a) = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implementation follows the classic O(n log n) algorithm.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform
        return np.full(n, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    if w.sum() == 0:
        return np.full(n, 1.0 / n)
    return w / w.sum()

def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_raised_cosine(n: int, width_frac: float) -> np.ndarray:
    """Half‑cosine bump of fractional width (smooth, compact support)."""
    width = max(2, int(width_frac * n))
    # cosine raised from 0 to pi
    i = np.arange(width, dtype=float)
    bump = (1 - np.cos(np.pi * i / (width - 1))) / 2.0
    full = np.zeros(n, dtype=float)
    start = (n - width) // 2
    full[start:start + width] = bump
    s = full.sum()
    return full / s if s > 0 else np.full(n, 0.0)


def generate_generalized_gaussian(n: int, sigma: float, p: float) -> np.ndarray:
    """a_i ∝ exp(-(|i-µ|/σ)^p)  (p>0)."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-np.abs(i - mu) ** p / (sigma ** p))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)

def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise *vec* onto the simplex, evaluate and keep if it improves."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    tot = vec.sum()
    if tot == 0:
        return
    vec = vec / tot
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """
    Very wide sweep over many families and Dirichlet draws.
    Returns the up‑to *max_keep* best vectors (as a list of ndarrays).
    """
    best = []   # list of (score, vector)

    # ----- power‑law -------------------------------------------------
    for p in np.linspace(0.1, 8.0, 400):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # ----- exponential -----------------------------------------------
    for lam in np.linspace(1e-4, 0.3, 400):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # ----- beta -------------------------------------------------------
    al_vals = np.linspace(0.2, 10.0, 30)
    be_vals = np.linspace(0.2, 10.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # ----- Gaussian ----------------------------------------------------
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # ----- Raised cosine -----------------------------------------------
    width_vals = np.linspace(0.05, 0.5, 25)   # fraction of total length
    for w in width_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_raised_cosine(n, w), best, max_keep)

    # ----- Generalised Gaussian ----------------------------------------
    for p in [1.5, 2.0, 3.0, 4.0]:
        sigma_vals = np.linspace(n / 80.0, n / 3.0, 20)
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_generalized_gaussian(n, sigma, p), best, max_keep)

    # ----- Dirichlet draws (random & sorted) --------------------------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 600
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            # also a monotone version (sorted descending)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(vec) to new_n and re‑project onto simplex."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, vec)
    return project_simplex(up)

# 4.  Smoothed Adam optimisation (soft‑max approximation of max)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 12.0,
                    min_tau: float = 1e-10,
                    tau_decay: float = 0.9996,
                    report_every: int = 500) -> np.ndarray:
    """
    Adam optimisation on the smooth approximation
        fτ(a) = τ·log Σ_k exp(conv_k/τ)
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # log‑softmax parameters (guarantee simplex)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initialise temperature from current convolution
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution -----
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- smooth max weights -----
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient w.r.t. a (g_i = 2·∑_k w_k·a_{k-i}) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule (soft‑max) -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature -----
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact evaluation -----
        if it % report_every == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth_opt] iter {it:7d}  best {val:.6f}  τ {tau:.2e}")

    return best_a

def subgradient_descent(a0: np.ndarray, deadline: float,
                        init_step: float = 0.5,
                        step_decay: float = 0.99995,
                        min_step: float = 1e-8,
                        report_every: int = 1000) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective  max(conv),
    with projection onto the simplex after each step.
    Returns the best vector found before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    step = init_step
    it = 0
    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        m_idx = int(conv.argmax())

        # gradient of conv[m_idx] w.r.t. a:
        grad = np.zeros_like(a)
        lo = max(0, m_idx - (n - 1))
        hi = min(n - 1, m_idx)
        # For i in [lo, hi] we have j = m_idx - i in range
        # contribution = a[j] (derivative of a_i * a_j)
        # Double counting handled by the factor 2 below.
        for i in range(lo, hi + 1):
            j = m_idx - i
            grad[i] = a[j]
        grad = 2.0 * grad

        # take a step opposite to the sub‑gradient and re‑project
        a = project_simplex(a - step * grad)

        # evaluate occasionally
        if it % report_every == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[subgrad] iter {it:7d}  best {val:.6f}  step {step:.3e}")

        # decay step size, keep it above min_step
        step = max(min_step, step * step_decay)

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """
    Light Metropolis‑style hill‑climb that may accept worsening moves.
    Adjusts two components of the vector at a time.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter to shake the system
            jitter = np.random.randn(n) * (step_scale * 0.02)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

    # -----------------------------------------------------------------

    return best_seq
