"""
Exponential‑subgradient optimizer for the self‑convolution L∞ problem.

Algorithm
---------
1. **Rich baseline exploration** – generate a diverse pool of
   candidates (power‑law, exponentials, Beta, cosine‑U, Gaussian,
   random Dirichlet draws, and random mixtures of Beta components).
   Keep the top few vectors.

2. **Upscaling** – linearly interpolate the current best vector to a
   larger dimension (4096) and re‑normalise onto the probability simplex.

3. **Exponential‑subgradient descent** – minimise the exact objective
   `max(a*a)` by repeatedly
   * taking the set of convolution entries attaining the current maximum,
   * forming a uniform weight on that set,
   * computing the sub‑gradient `g = 2·reverse(a) * w`,
   * updating multiplicatively `a ← a·exp(-η·g)` and re‑normalising.
   The learning rate η decays as 1/√t.  Many short restarts from random
   Dirichlet starts are performed while time permits.

4. **Final simulated‑annealing polish** – a lightweight Metropolis
   hill‑climber that moves a random ε‑fraction of mass between two
   coordinates, accepting worsening moves with a slowly cooling temperature.

The whole pipeline respects the 1000 s wall‑clock budget and returns the
best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a) == 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a).  Sum of a is assumed to be 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_cosine_power(n: int, p: float) -> np.ndarray:
    """U‑shaped: a_i ∝ |cos(π·i/(n‑1))|^{p}."""
    i = np.arange(n, dtype=float)
    arr = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_mixture_beta(n: int, comps: int = 3) -> np.ndarray:
    """Random mixture of `comps` Beta components."""
    weights = np.random.dirichlet(np.ones(comps))
    mixture = np.zeros(n, dtype=float)
    for w in weights:
        a = np.random.uniform(0.05, 5.0)
        b = np.random.uniform(0.05, 5.0)
        mixture += w * generate_beta(n, a, b)
    s = mixture.sum()
    return mixture / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Baseline exploration – keep the `max_keep` best vectors
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise, evaluate and keep the candidate if it improves the pool."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return
    vec = vec / total
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(
    n: int, deadline: float, max_keep: int = 12
) -> list[np.ndarray]:
    """Sweep a wide range of parametric families and random draws."""
    best = []   # (score, vector)

    # 1) power‑law (exponent up to 12)
    for p in np.linspace(0.05, 12.0, 300):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # 2) exponential (rates up to 1.0)
    for lam in np.linspace(0.0001, 1.0, 200):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # 3) beta (including very U‑shaped exponents)
    al_vals = np.concatenate(
        (np.linspace(0.05, 0.5, 30), np.linspace(0.6, 5.0, 30))
    )
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # 4) cosine‑power (U‑shaped)
    for p in np.linspace(0.5, 8.0, 120):
        if time.time() >= deadline:
            break
        _try_candidate(generate_cosine_power(n, p), best, max_keep)

    # 5) Gaussian (varied width)
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # 6) Dirichlet draws (various concentrations) + sorted versions
    concs = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 300
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            # monotone version – sorted descending
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    # 7) Random mixtures of Beta components
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = generate_mixture_beta(n, comps=np.random.choice([2, 3, 4]))
        _try_candidate(a, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves ordering roughly)
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `vec` to length `new_n` and re‑normalise."""
    n = vec.size
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


# ----------------------------------------------------------------------
# Exponential‑subgradient optimiser (direct subgradient of max)
# ----------------------------------------------------------------------


def exp_subgradient_opt(
    a0: np.ndarray,
    deadline: float,
    lr0: float = 0.8,
    grad_clip: float = 50.0,
    abs_tol: float = 1e-12,
) -> np.ndarray:
    """
    Minimise `max(a*a)` with a multiplicative (exponentiated) sub‑gradient.
    Returns the best vector encountered before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0

    while time.time() < deadline:
        # ----- convolution and weight on maximal entries -----
        conv = full_convolution(a)
        max_c = conv.max()
        mask = conv >= (max_c - abs_tol)
        if not np.any(mask):
            mask = np.zeros_like(conv, dtype=bool)
            mask[np.argmax(conv)] = True
        w = np.zeros_like(conv)
        w[mask] = 1.0 / np.count_nonzero(mask)

        # ----- sub‑gradient (same formula as smooth version) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)            # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]            # size n

        # ----- learning‑rate schedule -----
        lr = lr0 / np.sqrt(it + 1.0)

        # ----- multiplicative update (exponentiated gradient) -----
        g_clipped = np.clip(grad, -grad_clip, grad_clip)
        factor = np.exp(-lr * g_clipped)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            # fallback to previous best if we lost all mass
            a = best_a.copy()
            s = a.sum()
        a = a / s

        # ----- bookkeeping -----
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[exp_subgrad] iter {it:7d}  score {val:.6f}  η {lr:.2e}")

        # occasional jitter to escape very flat plateaux
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * (lr * 0.05)
            a = a * np.exp(jitter)
            a = a / a.sum()

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same as the previous implementation)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 1e-3,
    final_temp: float = 1e-6,
    cooling_factor: float = 0.9995,
    step_scale_init: float = 0.03,
    step_scale_min: float = 1e-6,
) -> np.ndarray:
    """Metropolis hill‑climb with decreasing temperature."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point demanded by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises `evaluate_sequence`.  Returns the best sequence found as a
    plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the surrounding harness
    deadline = start_time + budget_s - 5.0

    # problem sizes
    n_small = 2048
    n_big = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. optional fallback (e.g. from a previous run)
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline sweep (≈ 120 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 730.0   # reserve the bulk of the time for refinement
    if time.time() < baseline_deadline:
        candidates = baseline_search(n_small, baseline_deadline, max_keep=12)
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Upscale the current best to the larger dimension
    # -----------------------------------------------------------------
    if best_seq is None:
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)

    a0_big = upsample(a0_small, n_big)

    # -----------------------------------------------------------------
    # 3. Exponential‑subgradient optimisation (≈ 720 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 130.0
    if time.time() < polish_deadline:
        # primary run from the up‑sampled best
        refined = exp_subgradient_opt(a0_big, polish_deadline, lr0=0.8)
        val = compute_score(refined)
        print(f"[exp‑subgrad‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional short restarts while we still have time
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)

            # pick a start point: occasionally reuse the current best,
            # otherwise random Dirichlet (different concentrations)
            if np.random.rand() < 0.35 and best_seq is not None:
                a_start = np.array(best_seq, dtype=float)
            else:
                conc = np.random.choice([0.02, 0.05, 0.1, 0.2, 0.5, 1.0])
                a_start = np.random.dirichlet(np.full(n_big, conc))
                if np.random.rand() < 0.4:
                    a_start = np.sort(a_start)[::-1]   # monotone variant

            refined = exp_subgradient_opt(a_start, sub_deadline, lr0=0.9)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[exp‑subgrad‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 100 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be required)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
