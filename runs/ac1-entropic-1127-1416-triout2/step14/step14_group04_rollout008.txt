"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence`` (i.e. 2·n·max(convolution)/(sum²)).
The algorithm works in three stages:

1. **Initialisation** – use the previously found best sequence
   (global ``height_sequence_1``) or fall back to a uniform vector.
   A very short random‑perturbation phase tries to improve it cheaply.

2. **Linear‑program improvement** – repeatedly solve the LP
      maximise   ∑ b_j
      subject to   (f * b)_k ≤ M   for all k,
                  b_j ≥ 0,
   where ``f`` is the current vector and ``M = max(f*f)``.
   The solution ``b`` is re‑normalised to sum = 1 and mixed with ``f``
   (convex combination).  By a small step size the bound ``M`` is
   provably non‑increasing, and in practice the objective drops
   substantially after a few dozen LP calls.

3. **Fine polishing** – a short Adam optimisation on the smooth‑max
   approximation of the max‑convolution (identical to the previous
   implementation) cleans up remaining irregularities.

All work is performed inside the 1000 s budget; the best vector found
is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A * B
    conv = np.fft.irfft(C, n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Return the exact value 2·n·max(conv(a,a)).
    The vector ``a`` is assumed to have L1‑norm = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Soft‑max optimiser (identical to the previous implementation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.08,
                    init_tau_factor: float = 5.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9997) -> np.ndarray:
    """
    Adam on the soft‑max variables ``x`` (so that a = softmax(x)).
    The smooth max uses a temperature ``tau`` that is gradually annealed.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise temperature from the current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution
        conv = full_convolution(a)
        max_c = conv.max()

        # smooth‑max weighting
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a  (2·∑_k w_k·a_{k−i})
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)      # length = 3 n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for logits
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # occasional exact check
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Linear‑program step (core of the new search method)
# ----------------------------------------------------------------------


def lp_step(f: np.ndarray, M: float) -> tuple[np.ndarray, float] | tuple[None, float]:
    """
    Solve the LP:
        maximise   ∑ b_j
        subject to (f * b)_k ≤ M   for all k,
                  b_j ≥ 0.
    Returns the normalised solution ``b`` (sum = 1) and the bound ``M``.
    If the LP fails, ``None`` is returned for the vector.
    """
    n = f.size
    m = 2 * n - 1

    # Build the dense convolution matrix A where A[k, j] = f[k-j] (zero where out of range)
    # A can be filled efficiently column‑wise.
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f   # contributes f[0..n-1] to rows j..j+n-1

    b_ub = np.full(m, M, dtype=np.float64)          # upper bound for each convolution entry
    c = -np.ones(n, dtype=np.float64)               # maximise sum(b)  →  minimise -sum(b)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, M

    b = res.x
    s = b.sum()
    if s <= 0.0:
        return None, M

    # normalise to L1‑norm = 1 (the optimisation works with any positive scale)
    b_norm = b / s
    return b_norm, M


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 gives a good trade‑off between expressivity
    # and speed for the 1000 s budget.
    # -----------------------------------------------------------------
    n = 2048

    # -----------------------------------------------------------------
    # 0. Initialise from the previously known best construction if possible
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            f0 = np.array([float(x) for x in fallback], dtype=float)
            f0 = np.maximum(f0, 0.0)
            if f0.sum() == 0:
                raise ValueError
            f0 = f0 / f0.sum()
        except Exception:
            f0 = None
    else:
        f0 = None

    if f0 is None:
        # uniform start – trivial but safe
        f0 = np.full(n, 1.0 / n, dtype=float)

    best_f = f0.copy()
    best_score = compute_score(best_f)
    print(f"[init]   score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 1. Very quick random perturbation phase (≈ 20 s) – may give a tiny jump
    # -----------------------------------------------------------------
    quick_deadline = start_time + min(30.0, (deadline - start_time) * 0.03)
    while time.time() < quick_deadline:
        i, j = np.random.choice(n, size=2, replace=False)
        delta = np.random.rand() * best_f[i] * 0.05
        cand = best_f.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_f = sc, cand
            print(f"[perturb] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Linear‑program driven improvement (core of the new method)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 250.0                     # reserve time for polishing
    it = 0
    while time.time() < lp_deadline:
        # current bound M = max(f * f)
        conv_f = full_convolution(best_f)
        M = conv_f.max()

        b_norm, _ = lp_step(best_f, M)
        if b_norm is None:
            print("[LP]   LP failed – exiting LP loop")
            break

        # try a few mixing coefficients, stop at first improvement
        improved = False
        for t in (0.4, 0.2, 0.1, 0.05, 0.02):
            cand = (1.0 - t) * best_f + t * b_norm
            cand /= cand.sum()
            sc = compute_score(cand)
            if sc < best_score - 1e-8:
                best_score, best_f = sc, cand
                improved = True
                print(f"[LP]   it {it:4d}  t {t:.3f}  score {sc:.6f}")
                break

        if not improved:
            # Tiny step as a last resort
            t = 1e-3
            cand = (1.0 - t) * best_f + t * b_norm
            cand /= cand.sum()
            sc = compute_score(cand)
            if sc < best_score - 1e-8:
                best_score, best_f = sc, cand
                print(f"[LP]   it {it:4d}  tiny t {t:.1e}  score {sc:.6f}")
                improved = True

        if not improved:
            # No progress – exit the LP phase
            print("[LP]   No further improvement, terminating LP loop")
            break

        it += 1

    # -----------------------------------------------------------------
    # 3. Final fine‑grained Adam polishing (≈ 150 s)
    # -----------------------------------------------------------------
    if time.time() < deadline - 5.0:
        polish_deadline = deadline
        refined = smooth_optimize(best_f, polish_deadline)
        sc = compute_score(refined)
        print(f"[smooth] refined score = {sc:.6f}")
        if sc < best_score:
            best_score, best_f = sc, refined

    # -----------------------------------------------------------------
    # 4. Return the best candidate found
    # -----------------------------------------------------------------
    final_seq = best_f.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return final_seq
