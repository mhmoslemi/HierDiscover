"""
Simple Simulated Annealing optimizer for the step‑function attack.

The algorithm works on a probability vector `a` (non‑negative, sum = 1) of
length `n`.  The objective that we minimise is exactly the value returned
by ``evaluate_sequence`` (up to the harmless scaling that the evaluator
applies).  We use a simulated‑annealing loop that repeatedly proposes

* a targeted transfer of a small amount of mass from a coordinate that
  contributes heavily to the current maximal convolution entry,
* a random swap of two coordinates,
* a small Gaussian perturbation followed by a projection onto the simplex.

Accepted moves reduce the current objective or are accepted with a
Metropolis probability that depends on a temperature that decays
exponentially.  The best solution seen so far is kept and a final
mass‑transfer hill‑climb (many thousands of cheap moves) is performed to
polish the result.

The whole search stops a few seconds before the given time budget so
that the final sequence can be returned safely.
"""

import time
import numpy as np

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT (zero‑padded)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            a = cand
            if it % 20000 == 0:
                print(f"[mass‑climb] iter={it}, best={best_val:.6f}")
        it += 1
    return best_a

            # targeted mass transfer (uses the current maximal convolution entry)
            conv = full_convolution(cur_a)

            idx = np.arange(n)

            contrib = np.zeros(n)

            total_contrib = contrib.sum()

                src = np.random.choice(n, p=contrib / total_contrib)

    # 4. Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    final_seq = best_global_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq

Simulated‑annealing optimiser for the step‑function objective.

The optimiser works on a probability vector a (non‑negative, sum = 1) of
length n.  The value returned by ``evaluate_sequence`` (after the
internal sanitisation) equals

    2·n·max_k (a * a)[k] .

We therefore minimise the exact quantity ``2*n*max(conv)``.  The search
procedure is:

1. Initialise a pool of random Dirichlet vectors (including the uniform
   vector and, if available, the fallback construction) and keep the
   best seed.

2. Run a simulated‑annealing loop:
   * with probability 0.55 perform a *targeted* mass transfer – the
     source coordinate is chosen proportionally to its contribution to
     the current maximal convolution entry;
   * with probability 0.25 swap two coordinates;
   * otherwise apply a small Gaussian jitter followed by a Euclidean
     projection onto the probability simplex.
   Moves are accepted if they improve the score or with a Metropolis
   probability that depends on an exponentially decaying temperature.

3. After the annealing phase a long pure mass‑transfer hill‑climb
   (thousands of cheap moves) is performed to polish the best solution.

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (zero‑padded)."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for the FFT – fast and exact
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:

    Exact objective for a normalised vector (∑a = 1).

    Returns 2·n·max(conv) where conv = a*a (linear convolution).
    """
    n = a.shape[0]
