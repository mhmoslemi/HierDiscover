"""
Propose a step‑function (non‑negative vector on the simplex) that minimises
``evaluate_sequence``.

Strategy
--------
1. Choose a relatively large dimension (a power of two).  The theoretical
   lower bound 1 is approached as the discretisation gets finer, therefore
   a larger ``n`` is beneficial and still cheap thanks to FFT‑based convolutions.

2. Initialise the vector with the discrete arcsine law (Beta(½,½)).  Its
   self‑convolution is (asymptotically) uniform, giving a good starting
   point.

3. Optionally improve the start by **grid searching** a few Beta‑shapes
   (α,β ∈[0.3,0.8]) and by mixing the arcsine with the uniform vector.

4. **Exponentiated‑gradient polishing**:
   – minimise the true objective 2·n·max(conv(a,a)) while keeping
     a on the probability simplex;
   – the sub‑gradient of the max‑convolution is computed from the
     currently active (max‑attaining) indices;
   – an exponentiated‑gradient (mirror descent with KL‑divergence) update
     keeps the iterate strictly positive and only requires a cheap
     renormalisation.
   – step size decreases like 1/√t, occasional jitter helps escape plateaus.

5. Return the best vector found (as a plain list of ``float``).  The final
   official score is printed using ``evaluate_sequence`` for reference.

The whole search respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    # length of FFT: next power of two for speed
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape.
    a_i ∝ 1 / √((i+½)*(n‑½‑i)),   i = 0 … n‑1.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=float) + 0.5                # bin centre
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑type shape:
        a_i ∝ (i+1)^{α−1}·(n−i)^{β−1}   (α,β>0)
    Normalised to sum 1.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (mirror descent on KL‑divergence)
# ----------------------------------------------------------------------


def eg_polish(a0: np.ndarray, deadline: float,
              step0: float = 0.45,
              jitter_interval: int = 5000) -> np.ndarray:
    """
    Mirror‑descent / exponentiated‑gradient minimisation of
    2·n·max(conv(a,a)) under the simplex constraint.

    Parameters
    ----------
    a0 : ndarray
        Starting probability vector (sum == 1, non‑negative).
    deadline : float
        Unix time at which the routine must stop.
    step0 : float
        Initial step size (scaled by 1/√t).
    jitter_interval : int
        How often to add a tiny random perturbation (helps escape plateaus).

    Returns
    -------
    best_a : ndarray
        The best vector encountered (still on the simplex).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        # 1) current convolution and active set
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-12                # very tight tolerance
        active = np.where(conv >= max_val - eps)[0]

        # 2) sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # 3) exponentiated‑gradient update
        step = step0 / np.sqrt(t)
        a *= np.exp(-step * g)               # always stays positive
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)          # safety fallback
        else:
            a /= s

        # 4) occasional jitter
        if t % jitter_interval == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # 5) bookkeeping
        if t % 2000 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
    # final check after loop
    cur_val = compute_score(a)
    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()
    return best_a
