def lp_refine_step(a, t, deadline):
    n = a.shape[0]
    # compute M
    conv = full_convolution(a)
    M = conv.max()
    # build constraint matrix A of shape (2*n-1, n)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    # solve LP
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    # plus upper bound constraints
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback to current a
        return a
    b = res.x
    sum_b = b.sum()
    if sum_b == 0:
        return a
    g = b / sum_b
    a_new = (1 - t) * a + t * g
    # re-normalize to sum=1 to avoid any floating drift
    a_new = a_new / a_new.sum()
    return a_new
```

def permutation_anneal(a_start, deadline, init_temp=1e-3, final_temp=1e-6, cooling=0.9995):
    a = a_start.copy()
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val
    T = init_temp
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(n, size=2)
        if i==j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]
        val = compute_score(a)
        d = val - cur_val
        if d < 0 or np.random.rand() < np.exp(-d / max(T, 1e-12)):
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

            # random permutation of small fraction?
            idx = np.random.choice(n, size=int(0.01*n), replace=False)
            np.random.shuffle(a[idx])
    return best_a

"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.

Main ideas
----------
1. **Global baseline** – cheap sweep of parametric families and Dirichlet
   draws (as in the reference code) to obtain a decent starter.

2. **Linear‑programming refinement** – fix the current shape `f`,
   solve a LP that maximises the total L1‑mass of a new vector `g`
   while keeping every entry of the convolution `f*g` ≤ the current
   peak `M = max(f*f)`.  Mixing `f` and the normalised `g` strictly
   reduces the peak.  The LP is tiny (≈2 n constraints, n variables)
   and can be solved repeatedly for a moderate size (n≈1024).

3. **Upscaling** – the refined vector is linearly interpolated to a
   larger dimension (4096) to give the optimiser more freedom.

4. **Smooth Adam optimisation** – minimise a soft‑max surrogate of the
   max‑convolution using the log‑softmax parametrisation (the same
   “smooth_optimize’’ routine from the reference).

5. **Permutation annealing** – finally perform a Metropolis‑style
   hill‑climb that only swaps two entries at a time.  Swaps change the
   sum‑set of indices and therefore the distribution of convolution
   values; many swaps quickly lower the peak.

The whole pipeline respects the 1000 s wall‑clock budget and returns the
best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, real valued)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b (FFT based)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())       # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(a*a)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (used for the cheap global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    return arr / arr.sum()


def generate_exponential(n: int, lam: float) -> np.ndarray:
    arr = np.exp(-lam * np.arange(n, dtype=float))
    return arr / arr.sum()


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    return arr / arr.sum()


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return arr / arr.sum()


# ----------------------------------------------------------------------
# Baseline search (identical to the reference implementation, but
# trimmed for speed – we keep only the strongest candidates)
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        return
    vec = vec / s                         # simplex projection
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 6):
    best = []   # list of (score, vector)

    # Power‑law
    for p in np.linspace(0.1, 6.0, 120):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # Exponential
    for lam in np.linspace(0.0001, 0.2, 120):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # Beta
    al_vals = np.linspace(0.2, 8.0, 12)
    be_vals = np.linspace(0.2, 8.0, 12)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # Gaussian
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 12)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # Dirichlet draws (incl. sorted desc.)
    concs = [0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 400
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Upsampling (linear interpolation + simplex projection)
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0:
        return np.full(new_n, 1.0 / new_n)
    return up / s


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 10.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9995) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)               # log‑softmax parameters

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max -> probability vector
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution of the current a
        conv = full_convolution(a)
        max_c = conv.max()

        # smooth max weights
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            ws = w.sum()
            if ws == 0 or not np.isfinite(ws):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= ws

        # gradient wrt a  (g_i = 2·∑_k w_k·a_{k‑i})
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for the soft‑max (x‑space)
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # periodic exact evaluation
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing that swaps two entries (permutation optimiser)
# ----------------------------------------------------------------------


def permutation_anneal(a_start: np.ndarray, deadline: float,
                       init_temp: float = 1e-3,
                       final_temp: float = 1e-6,
                       cooling_factor: float = 0.9995) -> np.ndarray:
    """Swap‑based simulated annealing that changes the ordering of a."""
    a = a_start.copy()
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val
    T = init_temp
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # perform swap
        a[i], a[j] = a[j], a[i]
        val = compute_score(a)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T = max(final_temp, T * cooling_factor)

        # occasional randomised shake‑up of a tiny fraction
        if it % 3000 == 0:
            idx = np.random.choice(n, size=max(2, n // 100), replace=False)
            np.random.shuffle(a[idx])

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming refinement (single step)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False
    print("[warning] scipy.optimize.linprog not available – LP refinement will be skipped.")


def lp_refine_step(f: np.ndarray, mix: float = 0.2) -> np.ndarray:
    """
    One LP refinement step:
        maximise Σ b_j   subject to   b_j ≥ 0  and  (f * b)_k ≤ M  ∀k
    where M = max(f*f).  The returned vector is the convex mixture
    (1‑mix)·f + mix·(b / Σb) and is re‑normalised to sum = 1.
    """
    if not SCIPY_AVAILABLE:
        return f

    n = f.shape[0]
    conv = full_convolution(f)
    M = conv.max()

    # Build the (2n‑1) × n convolution matrix A where A[k, j] = f_{k‑j}
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f

    # Linear programme: minimise -Σ b_j  (i.e. maximise Σ b_j)
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n
    b_ub = np.full(rows, M + 1e-12, dtype=float)   # tiny slack for numerics

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # fallback: keep the current vector
        return f

    b = res.x
    sb = b.sum()
    if sb == 0:
        return f

    g = b / sb                                # normalise to Σg = 1
    f_new = (1.0 - mix) * f + mix * g
    f_new = f_new / f_new.sum()               # re‑project onto simplex
    return f_new


def lp_refine(f: np.ndarray, deadline: float,
              start_mix: float = 0.3, min_mix: float = 0.05,
              max_iters: int = 30) -> np.ndarray:
    """Repeated LP refinement until time runs out (or max_iters reached)."""
    mix = start_mix
    it = 0
    best = f.copy()
    best_score = compute_score(best)

    while time.time() < deadline and it < max_iters:
        f = lp_refine_step(f, mix=mix)
        sc = compute_score(f)
        if sc < best_score:
            best_score = sc
            best = f.copy()
            print(f"[lp_refine] iter {it:2d}  score {sc:.6f}  mix {mix:.3f}")
        # slowly decay the mixing factor to allow finer moves later
        mix = max(min_mix, mix * 0.93)
        it += 1
    return best


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin

    # -----------------------------------------------------------------
    # Problem sizes
    # -----------------------------------------------------------------
    n_small = 1024          # used for quick LP refinement
    n_big = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional reusable solution from a previous run
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Very cheap baseline sweep (≈ 80 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 750.0
    if time.time() < baseline_deadline:
        cand_vecs = baseline_search(n_small, baseline_deadline, max_keep=4)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Initialise the working vector (fallback to uniform)
    # -----------------------------------------------------------------
    if best_seq is None:
        a_small = np.full(n_small, 1.0 / n_small)
    else:
        a_small = np.array(best_seq, dtype=float)

    # -----------------------------------------------------------------
    # 3. Linear‑programming refinement (≈ 150 s)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 550.0
    if SCIPY_AVAILABLE and time.time() < lp_deadline:
        a_small = lp_refine(a_small, lp_deadline)
        val = compute_score(a_small)
        print(f"[lp] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_small.tolist()

    # -----------------------------------------------------------------
    # 4. Upsample to the larger dimension
    # -----------------------------------------------------------------
    a_big = upsample(a_small, n_big)

    # -----------------------------------------------------------------
    # 5. Smooth Adam optimisation (≈ 300 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if time.time() < smooth_deadline:
        a_big = smooth_optimize(a_big, smooth_deadline)
        val = compute_score(a_big)
        print(f"[smooth‑opt] post‑smooth score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_big.tolist()

    # -----------------------------------------------------------------
    # 6. Permutation‑based simulated annealing (≈ 180 s)
    # -----------------------------------------------------------------
    perm_deadline = deadline - 5.0
    if time.time() < perm_deadline:
        a_big = permutation_anneal(a_big, perm_deadline)
        val = compute_score(a_big)
        print(f"[perm‑anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_big.tolist()

    # -----------------------------------------------------------------
    # 7. Fallback – uniform (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 8. Official evaluation report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
