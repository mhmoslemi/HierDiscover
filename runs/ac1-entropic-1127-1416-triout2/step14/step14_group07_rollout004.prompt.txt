def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:

def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.

    exponent on the left‑hand side (p) and a gentle lift on the tail

    right = np.arange(1, n + 1, dtype=np.float64) ** q    # i+1
    a = left * right

def subgradient_polish(a0: np.ndarray, deadline: float,

    n = a.shape[0]

        t += 1

        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:

            j = k - idx

            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size and restart on stagnation."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]        # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            # jump start from best with a tiny random perturbation
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
