def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    n = 2048
    uniform = np.full(n, 1.0/n)
    best_seq = None
    best_score = float('inf')

    # fallback
    try:
        fallback = [float(x) for x in height_sequence_1]  # may exist
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_seq, best_score = fallback, val
    except Exception:
        pass

    # baseline arcsine
    arcsine = generate_arcsine(n)
    sc = compute_score(arcsine)
    print(f"[arcsine] score = {sc:.6f}")
    if sc < best_score:
        best_score, best_seq = sc, arcsine.copy()

    # baseline uniform
    sc_u = compute_score(uniform)
    if sc_u < best_score:
        best_score, best_seq = sc_u, uniform.copy()

    # random Beta and mixes
    trials = 0
    max_trials = 5000
    while time.time() < deadline - 15.0 and trials < max_trials:
        trials += 1
        alpha = np.random.uniform(0.2, 0.8)
        beta = np.random.uniform(0.2, 0.8)
        a = generate_beta(n, alpha, beta)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[beta] trial {trials} α={alpha:.3f} β={beta:.3f} score={sc:.6f}")

        # mixing with uniform
        lam = np.random.rand()
        mixed = lam * uniform + (1.0 - lam) * a
        mixed /= mixed.sum()
        sc = compute_score(mixed)
        if sc < best_score:
            best_score, best_seq = sc, mixed.copy()
            print(f"[mix] trial {trials} λ={lam:.3f} score={sc:.6f}")

        # also try mixing arcsine with a
        lam2 = np.random.rand()
        mixed2 = lam2 * arcsine + (1 - lam2) * a
        mixed2 /= mixed2.sum()
        sc = compute_score(mixed2)
        if sc < best_score:
            best_score, best_seq = sc, mixed2.copy()
            print(f"[mix2] trial {trials} λ={lam2:.3f} score={sc:.6f}")

    # fine grid search around best alpha,beta if we have them? Not tracking.

    # optional local hill climb on alpha,beta if we stored best params
    # but perhaps omit.

    # Subgradient polishing
    if best_seq is not None and time.time() < deadline - 5.0:
        polish_deadline = deadline - 5.0
        print(f"[polish] start with best_score={best_score:.6f}")
        a_polished = subgradient_polish(np.array(best_seq), polish_deadline)
        sc = compute_score(a_polished)
        print(f"[polish] final score={sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_polished.copy()

    # Simulated annealing final refinement
    if best_seq is not None and time.time() < deadline - 1.0:
        print("[anneal] final")
        a_final = simulated_annealing(np.array(best_seq), deadline)
        sc = compute_score(a_final)
        print(f"[anneal] final score={sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_final.copy()

    # Guarantee something
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = compute_score(best_seq)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished. final reported score = {final_score:.6f}")

"""
Search for a non‑negative step‑function that minimizes
``evaluate_sequence``.

Algorithm
---------
1. Build a few analytic families (uniform, arcsine, Beta(α,β)).
2. Randomly sample many Beta‑shapes (α,β ∈ (0.2,0.8)) and
   form convex mixtures with the uniform and arcsine vectors.
   Keep the best candidate seen so far (all vectors are
   normalised to sum‑1, i.e. lie on the probability simplex).
3. Refine the best candidate with a long projected sub‑gradient
   descent on the true objective 2·n·max(conv(a,a)).
4. Finish with a short simulated‑annealing hill‑climb to escape
   possible shallow minima.
5. Return the best sequence (plain list of floats).  The whole
   search respects the provided wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families of candidates
# ----------------------------------------------------------------------


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta–type shape:
       a_i ∝ (i+1)^{α−1}·(n−i)^{β−1},   α,β>0."""
    i = np.arange(1, n + 1, dtype=float)   # 1 … n
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) shape."""
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (uses the true objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active set (indices where conv is within machine epsilon of the max)
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of max(conv) = sum of gradients of active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape flat plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (keeps an upper bound)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises

    """

    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # -----------------------------------------------------------------
    # Configuration
    # -----------------------------------------------------------------
    n = 2048                                   # moderate size – fast FFTs
    uniform = np.full(n, 1.0 / n)

    best_seq = None
    best_score = float("inf")

    # 0. Optional fallback (if ``height_sequence_1`` exists)
    # -----------------------------------------------------------------

    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Deterministic strong starts: arcsine and uniform
    # -----------------------------------------------------------------
    arcsine = generate_arcsine(n)
    sc = compute_score(arcsine)
    print(f"[arcsine] score = {sc:.6f}")
    if sc < best_score:
        best_score, best_seq = sc, arcsine.copy()

    sc_u = compute_score(uniform)
    print(f"[uniform] score = {sc_u:.6f}")
    if sc_u < best_score:
        best_score, best_seq = sc_u, uniform.copy()

    # -----------------------------------------------------------------
    # 2. Randomised exploration of Beta families and mixtures
    # -----------------------------------------------------------------
    max_trials = 4000
    trial = 0
    # keep a few of the best Alpha/Beta pair for fine‑tuning later
    best_alpha = best_beta = None

    while time.time() < deadline - 30.0 and trial < max_trials:
        trial += 1
        # sample parameters in a range that produces non‑trivial shapes
        alpha = np.random.uniform(0.20, 0.80)
        beta = np.random.uniform(0.20, 0.80)

        a_beta = generate_beta(n, alpha, beta)
        sc = compute_score(a_beta)
        if sc < best_score:
            best_score, best_seq = sc, a_beta.copy()
            best_alpha, best_beta = alpha, beta
            print(f"[beta] trial {trial:4d} α={alpha:.3f} β={beta:.3f} score={sc:.6f}")

        # mix the beta shape with uniform
        lam = np.random.rand()
        mixed = lam * uniform + (1.0 - lam) * a_beta
        mixed /= mixed.sum()
        sc = compute_score(mixed)
        if sc < best_score:
            best_score, best_seq = sc, mixed.copy()
            print(f"[mixU] trial {trial:4d} λ={lam:.3f} score={sc:.6f}")

        # mix the beta shape with arcsine (helps cure discretisation artefacts)
        lam2 = np.random.rand()
        mixed2 = lam2 * arcsine + (1.0 - lam2) * a_beta
        mixed2 /= mixed2.sum()
        sc = compute_score(mixed2)
        if sc < best_score:
            best_score, best_seq = sc, mixed2.copy()
            print(f"[mixA] trial {trial:4d} λ={lam2:.3f} score={sc:.6f}")

    # -----------------------------------------------------------------
    # 3. Fine‑grid refinement around the best (α,β) if found
    # -----------------------------------------------------------------
    if best_alpha is not None and best_beta is not None and time.time() < deadline - 20.0:
        print("[fine grid] around best (α,β)")
        for da in np.linspace(-0.05, 0.05, 11):
            for db in np.linspace(-0.05, 0.05, 11):
                a_try = generate_beta(n, best_alpha + da, best_beta + db)
                sc = compute_score(a_try)
                if sc < best_score:
                    best_score, best_seq = sc, a_try.copy()
                    print(f"[grid] α={best_alpha+da:.3f} β={best_beta+db:.3f} score={sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing (takes the bulk of the remaining budget)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        polish_deadline = deadline - 2.0   # small reserve for the final eval
        print(f"[polish] start (best score {best_score:.6f})")
        a_polished = subgradient_polish(np.array(best_seq, dtype=float),
                                        polish_deadline)
        sc = compute_score(a_polished)
        print(f"[polish] refined score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_polished.copy()

    # -----------------------------------------------------------------
    # 5. Simulated annealing (final fine‑tuning)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[anneal] start")
        a_final = simulated_annealing(np.array(best_seq, dtype=float),
                                      deadline)
        sc = compute_score(a_final)
        print(f"[anneal] final score = {sc:.6f}")

            best_score, best_seq = sc, a_final.copy()

    # 6. Safeguard – return uniform if everything failed
    # -----------------------------------------------------------------

        best_seq = uniform.copy()
        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # 7. Official evaluation (clamping etc. done inside evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print("\nSearch finished. Reported score = {:.6f}".format(final_score))
    return best_seq.tolist()
