def l_bfgs_optimize(a_start, tau_factor, max_iter_per_stage, deadline):
    # compute initial conv
    a = a_start.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        conv = full_convolution(a)
        max_c = conv.max()
        if max_c == 0:
            break
        tau = max(tau_factor * max_c, 1e-12)   # tau scaling
        # define objective
        def fun_and_grad(x):
            obj, grad = smooth_obj_and_grad(x, tau)
            return obj, grad
        x0 = np.log(a + 1e-30)
        # call optimizer with limited maxiter
        try:
            res = opt.minimize(fun_and_grad, x0, method='L-BFGS-B', jac=True,
                               options={'maxiter': max_iter_per_stage, 'disp': False},
                               callback=lambda x: time.time() >= deadline)
        except Exception as e:
            break
        a = np.exp(res.x - np.max(res.x))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[l-bfgs] new best {val:.6f} tau {tau:.3e}")
        # reduce tau for next stage
        tau_factor *= 0.5
        # exit if tau too small or not improving
        if tau < 1e-12:
            break
    return best_a

"""
Search for a non‑negative step‑function minimizing evaluate_sequence.

Algorithm
---------
0. **Safety‑first** – try to reuse an existing global solution if present.
1. **Broad baseline** – generate many candidate vectors from simple
   parametric families (power‑law, exponential, beta, Gaussian,
   linear‑decreasing, quadratic‑decreasing) and random Dirichlet draws
   (including sorted descending versions).  Keep the best few.
2. **Multi‑scale up‑sampling** – take the best candidate and linearly
   interpolate it to a larger dimension (4096).  This gives the optimiser
   more degrees of freedom while preserving the overall shape.
3. **Smooth L‑BFGS optimisation** – minimise a smooth approximation of
   max(conv) on the simplex by optimising the log‑softmax parameters
   with L‑BFGS.  The temperature τ is decreased in stages, turning the
   smooth objective into the true max.  Several restarts from random
   Dirichlet seeds are performed while time permits.
4. **Long simulated‑annealing polish** – a Metropolis hill‑climb with
   adaptive step‑size explores the neighbourhood of the best vector.
5. **Final reporting** – evaluate the best sequence with the official
   evaluate_sequence() function and return it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution (FFT based, real)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b (FFT based)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a) = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(a*a)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return probability vectors)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_linear_decreasing(n: int) -> np.ndarray:
    """a_i decreasing linearly from large to zero (sums to 1)."""
    arr = np.arange(n, 0, -1, dtype=float)      # n, n-1, ..., 1
    s = arr.sum()
    return arr / s


def generate_quadratic_decreasing(n: int, p: float = 2.0) -> np.ndarray:
    """a_i ∝ (n-i)^{p} (smoothly decreasing, sums to 1)."""
    arr = np.arange(n, 0, -1, dtype=float) ** p
    s = arr.sum()
    return arr / s


# ----------------------------------------------------------------------
# Helpers for baseline stage
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate and keep if good."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return
    vec = vec / total
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """Sweep many families and Dirichlet draws, keep the best ``max_keep``."""
    best = []   # list of (score, vector)

    # ----- Power‑law -------------------------------------------------
    for p in np.linspace(0.05, 12.0, 500):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # ----- Exponential -----------------------------------------------
    for lam in np.linspace(0.0005, 1.0, 500):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # ----- Beta -------------------------------------------------------
    al_vals = np.linspace(0.2, 30.0, 40)
    be_vals = np.linspace(0.2, 30.0, 40)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # ----- Gaussian ---------------------------------------------------
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # ----- Linear / quadratic decreasing -------------------------------
    _try_candidate(generate_linear_decreasing(n), best, max_keep)
    for p in [1.5, 2.0, 3.0, 4.0]:
        _try_candidate(generate_quadratic_decreasing(n, p), best, max_keep)

    # ----- Dirichlet draws --------------------------------------------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    draws_per_conc = 500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            # sorted descending version (strongly monotone)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation + simplex projection)
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``vec`` to length ``new_n`` and re‑normalise."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


# ----------------------------------------------------------------------
# Smoothed max objective + exact gradient (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(x: np.ndarray, tau: float):
    """
    Soft‑max approximation of max(conv) together with its gradient
    w.r.t. the log‑softmax parameters ``x``.
    Returns (obj, grad_x).
    """
    # ----- soft‑max mapping -----
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    a = exp_x / exp_x.sum()                     # a lies on the simplex

    # ----- convolution -----
    conv = full_convolution(a)
    max_c = conv.max()

    # ----- weights w (smooth max) -----
    if tau < 1e-12:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        # stable soft‑max on conv
        w = np.exp((conv - max_c) / tau)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

    # ----- objective (smooth max) -----
    obj = np.dot(w, conv)                       # expectation under w

    # ----- gradient w.r.t. a -----
    a_rev = a[::-1]
    grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
    start = a.shape[0] - 1
    grad_a = 2.0 * grad_full[start:start + a.shape[0]]

    # ----- chain rule for soft‑max (x‑space) -----
    dot = np.dot(grad_a, a)
    grad_x = (grad_a - dot) * a                 # same as (diag(a) - a aᵀ)·grad_a

    return obj, grad_x


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smoothed problem
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray, deadline: float,
                    initial_tau_factor: float = 10.0,
                    max_iter_stage: int = 150) -> np.ndarray:
    """
    Run a L‑BFGS optimisation on the smooth max objective.
    The temperature τ is halved after each stage.
    Returns the best vector found before *deadline*.
    """
    import scipy.optimize as opt

    # initialise
    a = a_start.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    # initial temperature based on the current convolution peak
    conv0 = full_convolution(a)
    tau_factor = initial_tau_factor

    while time.time() < deadline:
        max_c = conv0.max()
        tau = max(tau_factor * max_c, 1e-12)

        # ----------- L‑BFGS on the current τ -----------------------
        x0 = np.log(a + 1e-30)

        def fun_and_grad(x):
            # returns (obj, grad) – required form when jac=True
            obj, grad = smooth_obj_and_grad(x, tau)
            return obj, grad

        # callback aborts if we run out of time
        callback = lambda x: time.time() >= deadline

        try:
            res = opt.minimize(fun_and_grad, x0,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': max_iter_stage,
                                        'disp': False},
                               callback=callback)
        except Exception:
            break

        # recover a from the solution
        a = np.exp(res.x - np.max(res.x))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[l‑bfgs] new best {val:.6f}  τ={tau:.2e}")

        # prepare for next stage (reduce τ)
        tau_factor *= 0.5
        if tau_factor * max_c < 1e-12:
            break
        # recompute convolution for the next stage
        conv0 = full_convolution(a)

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9992,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-14)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to free tiny plateaux
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem sizes – start cheap, finish on a richer grid.
    # -----------------------------------------------------------------
    n_small = 2048
    n_big = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 120 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 700.0   # reserve most time for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n_small, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Upsample the current best to the larger dimension
    # -----------------------------------------------------------------
    if best_seq is None:
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)

    a0_big = upsample(a0_small, n_big)

    # -----------------------------------------------------------------
    # 3. L‑BFGS optimisation on the big problem (≈ 650 s)
    # -----------------------------------------------------------------
    lbfgs_deadline = deadline - 200.0

            if val < best_val:
