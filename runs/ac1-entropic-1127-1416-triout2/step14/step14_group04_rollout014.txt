"""
Propose a non‑negative coefficient sequence that minimises the
``evaluate_sequence`` bound.

The search combines several ingredients that improve on the previous
implementation:

* **More expressive families** – symmetric exponential, U‑shaped
  (Beta with α=β<1) and explicit two‑block “end‑mass’’ shapes, all of
  which concentrate mass towards the ends of the interval while keeping
  each entry O(1/n).  These families were not explored before and can
  lower the peak of the self‑convolution.

* **Finer parametric sweeps** – dense grids for the new families
  (λ for the symmetric exponential, α for the U‑Beta, block‑width for
  the two‑block shape) together with the old exponential/power‑law
  sweeps.

* **Multiple Adam refinements** – each promising candidate is polished
  by Adam on a smooth‑max surrogate of the convolution, with a slower
  temperature decay than before.

* **Final simulated‑annealing polish** – a lightweight Metropolis walk
  to escape possible plateaus.

All stages respect the 1000 s wall‑clock budget; the code prints a few
status messages and finally returns the best list of floats it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sym_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) + exp(-lam·(n‑1‑i)) (symmetric, end‑heavy)."""
    left = np.exp(-lam * np.arange(n, dtype=float))
    right = left[::-1]
    a = left + right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_u_beta(n: int, alpha: float) -> np.ndarray:
    """U‑shaped Beta distribution: α=β<1 gives peaks at both ends."""
    return generate_beta(n, alpha, alpha)


def generate_two_block(n: int, frac: float) -> np.ndarray:
    """Two equal blocks of width frac·n at the ends, each carrying ½ mass."""
    w = max(1, int(frac * n / 2.0))
    a = np.zeros(n, dtype=float)
    val = 0.5 / w
    a[:w] = val
    a[-w:] = val
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]               # decreasing
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sym_monotone_random(n: int) -> np.ndarray:
    """Generate a decreasing half‑vector and mirror it to enforce symmetry."""
    half = n // 2
    v = np.random.exponential(scale=1.0, size=half)
    left = np.sort(v)[::-1]
    if n % 2 == 0:
        a = np.concatenate([left, left[::-1]])
    else:
        mid = np.array([np.random.exponential(scale=1.0)])
        a = np.concatenate([left, mid, left[::-1]])
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Simple bookkeeping utilities
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best):
    """Update the (score, vector) list ``best`` if ``vec`` improves it."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        return
    vec = vec / s
    sc = compute_score(vec)
    # keep at most 12 best candidates (small enough for later polishing)
    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Baseline exploration (wide random and parametric sampling)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float):
    best = []   # (score, vector) – smallest scores first

    # 1) Dirichlet draws with a range of concentration parameters
    alphas = [0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 4.0]
    draws_per_alpha = 1500
    for conc in alphas:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    # 2) Random monotone vectors (decreasing)
    for _ in range(3000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # 3) Random symmetric monotone vectors
    for _ in range(2000):
        if time.time() >= deadline:
            break
        a = generate_sym_monotone_random(n)
        _try_candidate(a, best)

    # 4) Coarse beta sweep (including very U‑shaped cases)
    al_vals = np.concatenate([np.linspace(0.05, 0.9, 9),
                              np.linspace(1.0, 5.0, 9)])
    be_vals = np.concatenate([np.linspace(0.05, 0.9, 9),
                              np.linspace(1.0, 5.0, 9)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best)

    # 5) U‑shaped beta (α=β) fine sweep (focus on the low‑α region)
    for alpha in np.linspace(0.05, 0.9, 18):
        if time.time() >= deadline:
            break
        a = generate_u_beta(n, alpha)
        _try_candidate(a, best)

    # 6) Symmetric exponential family sweep
    for lam in np.linspace(0.0003, 0.0030, 1500):
        if time.time() >= deadline:
            break
        a = generate_sym_exponential(n, lam)
        _try_candidate(a, best)

    # 7) Two‑block end‑mass family sweep
    for frac in np.linspace(0.01, 0.45, 45):   # block half‑width fraction
        if time.time() >= deadline:
            break
        a = generate_two_block(n, frac)
        _try_candidate(a, best)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]

# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on a soft‑max surrogate)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.07,
                    tau_decay: float = 0.9995,
                    min_tau: float = 1e-9) -> np.ndarray:
    """
    Adam optimisation on a soft‑max approximation of max(conv(a,a)).
    The soft‑max temperature τ is annealed slowly (τ_decay) so that the
    landscape becomes increasingly sharp.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-30
    x = np.log(a + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise τ from the current max‑convolution (scale‑free)
    conv = full_convolution(a)
    tau = max(min_tau, 5.0 * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping a = softmax(x)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution and soft‑max weighting
        conv = full_convolution(a)
        max_c = conv.max()
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient wrt a (2·∑_k w_k·a_{k−i})
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)    # length = 3·n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for logits
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # intermittent exact check
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 5e-7,
                        cooling_factor: float = 0.9994,
                        step_scale_init: float = 0.025,
                        step_scale_min: float = 5e-7) -> np.ndarray:
    """Metropolis hill‑climb with an annealing schedule."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasionally jitter the whole vector
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – a little larger than before (3072) gives finer
    # resolution without exploding the O(N log N) cost.
    # -----------------------------------------------------------------
    n = 3072

    best_seq = None
    best_val = float("inf")               # based on compute_score (sum=1)

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            sc = compute_score(np.array(fb) / np.sum(fb))
            if sc < best_val:
                best_val, best_seq = sc, fb
                print(f"[fallback] score {sc:.6f}")
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Broad random baseline (≈ 250 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 550.0          # reserve time for refinements
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Focused 1‑D sweeps (≈ 180 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 350.0
    if time.time() < sweep_deadline:
        # exponential (non‑symmetric)
        sc, vec = fine_exponential_search(n, sweep_deadline)
        if sc is not None and sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[exp‑sweep] new best = {sc:.6f}")

        # power‑law
        if time.time() < sweep_deadline:
            sc, vec = fine_powerlaw_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[pow‑sweep] new best = {sc:.6f}")

        # symmetric exponential
        if time.time() < sweep_deadline:
            sc, vec = fine_sym_exponential_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[sym‑exp‑sweep] new best = {sc:.6f}")

        # U‑shaped beta
        if time.time() < sweep_deadline:
            sc, vec = fine_u_beta_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[u‑beta‑sweep] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation (≈ 250 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, polish_deadline)
        sc = compute_score(refined)
        print(f"[smooth‑opt] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

        # extra random restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(35.0, remaining)

            # start from a fresh symmetric random draw (often a good basin)
            a_start = generate_sym_monotone_random(n)

            refined = smooth_optimize(a_start, sub_deadline)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[smooth‑restart] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 120 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(final)
        print(f"[anneal] final polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 6. Official evaluation (the harness will call evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return best_seq
