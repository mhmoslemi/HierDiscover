"""
Improved search for a low‑max‑auto‑convolution step function.

The code proceeds in three phases:

1.  Quick parametric sweep over a few simple families (reverse‑power‑law,
    a convex mixture with the uniform distribution) to obtain a decent
    starting vector.

2.  Smooth annealed gradient descent on a differentiable surrogate of the
    max‑convolution (soft‑max with an increasing temperature).  The
    gradient is obtained analytically via a second FFT‑based convolution
    and the iterate is projected onto the probability simplex after each
    step (mirrors the sub‑gradient method of the reference but with a
    smoother objective and Adam‑style step‑size adaptation).

3.  A final stochastic hill‑climb (same spirit as the reference) that
    makes tiny mass transfers between coordinates and occasionally restarts
    with jitter.  This polishes the result and typically pushes the
    reported value a little further down.

All heavy work uses FFT‑based full convolutions (O(n log n)) and the
implementation respects the 1000‑second budget by checking the deadline
frequently.
"""

# ----------------------------------------------------------------------
# Imports and tiny utilities
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum x = 1}."""
    if np.all(v <= 0):
        # avoid division by zero – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (real inputs)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """
    Compute the full linear convolution a * b using a real FFT.
    If b is None the function returns a * a.
    """
    if b is None:
        b = a
    n = a.shape[0]
    m = b.shape[0]
    size = n + m - 1
    n_fft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective for a normalized vector (Σa = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)          # a * a
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """Vector proportional to (n‑i)^{p}, i = 0..n‑1, normalized to sum 1."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n ... 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """mix·reverse_power_law(p) + (1‑mix)·uniform, projected onto simplex."""
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    return project_to_simplex(a)

def quick_seed(n: int, deadline: float) -> np.ndarray:

# ----------------------------------------------------------------------
# Phase 2 – annealed smooth gradient descent (soft‑max surrogate)
# ----------------------------------------------------------------------
def smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    beta_start: float = 2.0,
    beta_end: float = 150.0,
    lr_start: float = 0.25,
    lr_end: float = 0.02,
) -> np.ndarray:
    """
    Perform projected Adam‑style gradient descent on the soft‑max
    approximation of max(conv(a,a)).  The temperature parameter β is
    increased from ``beta_start`` to ``beta_end`` (annealing).  The
    learning rate decays from ``lr_start`` to ``lr_end``.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    best_a = a.copy()
    best_M = full_convolution(a).max()
    last_improve = time.time()

    while time.time() < deadline:
        t += 1
        # ----- current convolution and temperature -----
        conv = full_convolution(a)               # length 2n‑1
        M = conv.max()
        # track best (exact objective, not the surrogate)
        if M < best_M:
            best_M = M
            best_a = a.copy()
            last_improve = time.time()

        # ----- schedule -----
        now = time.time()
        tot = deadline - (deadline - (deadline - now))   # just for clarity
        prog = (now - (deadline - (deadline - now))) / (deadline - now + 1e-12)  # dummy, will recompute correctly below
        # Simpler: use elapsed fraction relative to sweep window
        elapsed = now - (deadline - (deadline - now))  # zero – we recompute proper fraction later

        # Actually compute fraction based on total allowed time for this phase:
        # We store the start time outside; easier to pass start_time.
        # To avoid extra arguments we compute fraction linearly using t.
        # We'll use a simple linear decay based on iteration count.
        # For robustness we keep beta and lr bounded.
        # (A more precise schedule would require the start time; this simple
        #  version works well in practice.)

        # ----- temperature and learning rate (linear schedule) -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 20000.0)
        lr = lr_start + (lr_end - lr_start) * min(1.0, t / 20000.0)

        # ----- soft‑max weights -----
        # subtract the max for numerical stability
        exp_weights = np.exp(beta * (conv - M))
        w = exp_weights / exp_weights.sum()          # shape (2n‑1,)

        # ----- gradient of the smooth objective -----
        # g_i = 2 * sum_t w_{i+t} * a_t   (t = 0..n‑1, i+t within 0..2n‑2)
        rev_a = a[::-1]
        conv_w = full_convolution(w, rev_a)          # length 3n‑2
        g = 2.0 * conv_w[n - 1 : n - 1 + n]          # slice length n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # ----- occasional jitter if stuck for a while -----
        if time.time() - last_improve > 30.0:
            # re‑inject a tiny random perturbation around the best so far
            a = best_a + np.random.randn(n) * 0.0005
            a = project_to_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = time.time()

    return best_a


# ----------------------------------------------------------------------
# Phase 3 – classic stochastic hill‑climb (fine polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 150_000) -> np.ndarray:
    """Greedy mass‑transfer hill‑climb with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            # restart from the current best with a gentle jitter
            jitter = np.random.randn(n) * 0.0006
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The method consists of a quick parametric sweep,
    an annealed smooth gradient descent, and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid being killed exactly at the deadline
    deadline = start_time + budget_s - 3.0

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0.  Optional fallback from a previous construction
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Choose problem dimension (larger n → finer discretisation)
    # -------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n = 8192
    elif remaining > 300:
        n = 4096
    else:
        n = 2048
    print(f"[setup] using n = {n}")

    # -------------------------------------------------------------
    # 2. Quick parametric sweep to obtain a solid seed
    # -------------------------------------------------------------
    seed_deadline = deadline - 250.0
    if time.time() < seed_deadline:
        a0 = quick_seed(n, seed_deadline)
        if a0 is not None:
            val_raw = compute_score(a0)
            val_off = evaluate_sequence(a0.tolist())
            print(f"[seed] raw score = {val_raw:.6f}, official = {val_off:.6f}")
            if val_off < best_val:
                best_val, best_seq = val_off, a0.tolist()
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # -------------------------------------------------------------
    # 3. Smooth annealed gradient descent on the surrogate objective
    # -------------------------------------------------------------
    gd_deadline = deadline - 80.0
    if time.time() < gd_deadline:
        a_start = np.array(best_seq, dtype=float) if best_seq is not None else a0
        a_polished = smooth_gradient_descent(a_start, gd_deadline)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[grad‑descent] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # 4. Final stochastic hill‑climb (fine polishing)
    # -------------------------------------------------------------
    hc_deadline = deadline - 2.0
    if time.time() < hc_deadline and best_seq is not None:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hc_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # 5. Fallback in the extremely unlikely case that nothing was found
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = mixed_reverse_uniform(n, mix=0.8, p=1.7)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final report
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
