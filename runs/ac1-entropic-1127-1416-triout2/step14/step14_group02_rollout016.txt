def solve_lp(f, M):
    n = len(f)
    m = 2*n - 1
    # Build A (m, n) matrix.
    # Using toeplitz: first column = np.concatenate([f, np.zeros(n-1)]) length m.
    # first row = np.concatenate([f[0], np.zeros(n-1)]) length n.
    # Actually to compute convolution, we need f placed as "kernel" sliding over b. The Toeplitz matrix built from f as column, zero padded.
    from scipy.linalg import toeplitz
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([[f[0]], np.zeros(n - 1)])
    A = toeplitz(col, row)  # shape (m, n)
    # Set up LP
    c = -np.ones(n)  # minimize -sum b
    bounds = [(0, None)] * n
    # Constraint A @ b <= M
    # Usually we need A_ub = A, b_ub = M * np.ones(m)
    b_ub = np.full(m, M)
    # Use linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = np.maximum(res.x, 0.0)  # ensure nonnegative due to tolerance.
    return b

"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of four stages:

1. **Global baseline** – cheap random draws from several analytic families
   (power‑law, exponential, beta, Gaussian, sine‑power) together with a large
   pool of Dirichlet samples.  The best few candidates are kept.

2. **Linear‑programming refinement** – for a current vector `f` we compute
   `M = max(conv(f,f))` and then solve the LP

        maximise   Σ b_j
        subject to ∀k : (f * b)[k] ≤ M ,   b ≥ 0

   The optimal `b` is normalised to a probability vector `g`.  A convex
   combination `f←(1‑t)f + t·g` (with a line‑search on `t`) never raises the
   maximal autocorrelation, and the process is repeated while time permits.

3. **Smooth‑max Adam optimisation** – a blended surrogate consisting of a
   smooth‑max of the convolution and a flattening term is minimised with a
   customised Adam optimiser.  The surrogate encourages a flat convolution
   (the theoretical lower bound) while still reducing the peak.

4. **Projected sub‑gradient polishing** – a final exact sub‑gradient descent
   on the true max‑objective to polish the result.

All stages respect the global time budget (≈ 1000 s).  The best vector found
is returned as a plain Python list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for many calls)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a)=1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return the exact evaluation score for a (sum(a) must be 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families for baseline search
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Baseline exploration (cheap, many random draws)
# ----------------------------------------------------------------------
def baseline_search(n: int, deadline: float, max_keep: int = 12,
                    random_draws: int = 12000) -> list[np.ndarray]:
    """Return a list of up to `max_keep` promising vectors for length `n`."""
    best = []                         # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- analytic families -----
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    al_vals = np.linspace(0.02, 2.0, 31)
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    for _ in range(300):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # ----- mixture of families -----
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # ----- Dirichlet draws -----
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    # ----- random permutations of the current best -----
    if best:
        top_vec = best[0][1]
        for _ in range(150):
            if time.time() >= deadline:
                break
            perm = np.random.permutation(top_vec)
            try_candidate(perm)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Linear‑programming improvement (see description above)
# ----------------------------------------------------------------------
def solve_lp_for_b(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max  Σ b_j  s.t.  (f * b)[k] ≤ M  for all k,  b ≥ 0.
    Returns the optimal b (may have arbitrary sum) or None on failure.
    """
    n = f.shape[0]
    m = 2 * n - 1
    # Build the Toeplitz matrix that performs linear convolution with f.
    # First column = f padded with n-1 zeros, first row = [f[0], 0, …, 0].
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([[f[0]], np.zeros(n - 1)])
    A = toeplitz(col, row)               # shape (m, n)

    c = -np.ones(n)                       # maximise Σb -> minimise -Σb
    bounds = [(0, None)] * n
    b_ub = np.full(m, M)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    b = np.maximum(res.x, 0.0)
    return b


def lp_refine(f_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly solve the LP, mix the result with the current vector and
    keep the improvement while time permits.
    """
    f = f_start.copy()
    n = f.shape[0]
    best_score = compute_score(f)

    while time.time() < deadline:
        conv_ff = full_convolution(f)
        M = conv_ff.max()

        b = solve_lp_for_b(f, M)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 0:
            break

        g = b / sum_b                     # turn b into a probability vector

        # line‑search on mixing factor t (start relatively aggressive)
        t = 0.5
        improved = False
        while t > 1e-5 and time.time() < deadline:
            f_candidate = (1.0 - t) * f + t * g
            f_candidate = project_to_simplex(f_candidate)
            new_max = full_convolution(f_candidate).max()
            # we only accept moves that do not raise the peak
            if new_max <= M * (1.0 + 1e-12):
                f = f_candidate
                improved = True
                break
            t *= 0.5

        if not improved:
            # no monotone improvement possible – stop the LP phase
            break

        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            # keep the best vector seen so far (might be a previous iterate)
    return f

# Combined Adam optimiser (smooth‑max + flattening term)

def combined_adam(a0: np.ndarray, deadline: float,
                  lam: float = 0.35,

                  tau_end: float = 1e-7,
                  tau_decay: float = 0.95,
                  lr_start: float = 0.12,
                  lr_decay: float = 0.97,
                  beta1: float = 0.90,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Adam optimiser of a blended smooth‑max / flattening surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m

    size_fft = 1 << ((3 * n - 3).bit_length())

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        conv = full_convolution(a)                     # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)       # stable log‑sum‑exp
        w = exp_vals / exp_vals.sum()                 # soft‑max weights

        err = conv - target

        # correlation gradients (cross‑correlation)
        def corr_grad(vec):
            rev = vec[::-1]
            A = np.fft.rfft(a, n=size_fft)
            B = np.fft.rfft(rev, n=size_fft)
            full = np.fft.irfft(A * B, n=size_fft)[: n + vec.shape[0] - 1]
            idx = (vec.shape[0] - 1) - np.arange(n)
            return 2.0 * full[idx]

        grad_max = corr_grad(w)
        grad_err = corr_grad(err)
        grad = (1.0 - lam) * grad_max + lam * grad_err

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (exact max objective)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.33) -> np.ndarray:
    """Exact sub‑gradient descent on the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    """Metropolis hill‑climb with temperature decay."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a simple list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # -----------------------------------------------------------------
    # 1. Problem size – a modest length gives a good speed/quality trade‑off
    # -----------------------------------------------------------------
    n = 1024

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previously known good construction)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 2. Global baseline exploration (≈ 120 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 120.0)
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=12, random_draws=20000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Linear‑programming refinement (major chunk of remaining time)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 320.0                     # keep time for final polish
    if best_seq is not None and time.time() < lp_deadline:
        f_current = np.array(best_seq, dtype=float)
        f_refined = lp_refine(f_current, lp_deadline)
        val = compute_score(f_refined)
        print(f"[LP‑refine] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, f_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Combined Adam optimisation (≈ 180 s)
    # -----------------------------------------------------------------
    adam_deadline = deadline - 130.0
    if best_seq is not None and time.time() < adam_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_adam = combined_adam(a0, adam_deadline, lam=0.35)
        val = compute_score(a_adam)
        print(f"[Adam] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_adam.tolist()

    # -----------------------------------------------------------------
    # 5. Sub‑gradient polish (≈ 100 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 30.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.33)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 6. Simulated annealing (final ~30 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline,
                                      init_temp=2e-3, final_temp=1e-7,
                                      cooling_factor=0.9995,
                                      step_scale_init=0.05)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – return a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
