def adam_polish(a0, deadline, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time()<deadline:
        t+=1
        conv = full_convolution(a)
        max_val = conv.max()
        eps_tol = max_val * 1e-8
        active = np.where(conv >= max_val - eps_tol)[0]
        if active.size ==0:
            # No active? could break
            continue
        g = np.zeros_like(a)
        for k in active:
            # add 2*a shifted by k-i for each i
            # Equivalent: for i from max(0, k-(n-1)) to min(n-1, k):
            i_min = max(0, k-(n-1))
            i_max = min(n-1, k)
            # vectorized: indices i and j = k-i
            i = np.arange(i_min, i_max+1)
            j = k - i
            g[i] += 2 * a[j]
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex to ensure sum=1 and nonneg
        a = project_to_simplex(a)
        # possibly early exit if no improvement
    return a
```

"""
Improved alternating‑LP + Adam polishing for the step‑height problem.

The goal is to minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant.  We therefore keep the vector normalised to a
probability distribution (sum = 1).  The algorithm consists of two phases:

1.  Alternating linear programmes (LP) – implemented with SciPy's
    ``linprog`` (so we avoid the cvxpy KKT errors seen before).  In each
    iteration we solve

        maximise   Σ b_i
        subject to   b_i ≥ 0,
                     conv(f, b) ≤ M   (full linear convolution)

    where ``f`` is the current distribution and ``M`` is the current peak
    of ``conv(f,f)``.  The optimal ``b`` is renormalised to the simplex and
    we move a convex combination step towards it.  This strictly reduces
    the peak of the autocorrelation.

2.  Adam‑style projected gradient descent.  The (non‑smooth) objective
    ``max(conv)`` is minimised by repeatedly moving opposite the
    sub‑gradient obtained from the active offsets (those attaining the
    current maximum).  An Adam update gives adaptive step sizes and fast
    convergence.  After each update we project back onto the probability
    simplex.

Multiple random starts (Dirichlet draws) are tried until the time budget
expires; the best sequence found (according to the provided
``evaluate_sequence``) is returned.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for the LP solver
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_LP_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_LP_AVAILABLE = False

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that A @ b = conv(f, b) (full convolution).

    The matrix has shape (2n‑1, n) and entry A[k, i] = f[k‑i] if the index
    is inside the support of ``f``; otherwise 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f  # place the whole filter shifted by i rows
    return A


# ----------------------------------------------------------------------
# Alternating LP refinement (SciPy implementation)
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP

        maximise   Σ b_i
        subject to   b ≥ 0,
                     conv(f, b) ≤ M

    Returns the optimal vector ``b`` (may be any total mass) or ``None`` on
    failure.
    """
    if not _SCIPY_LP_AVAILABLE:
        return None

    n = f.shape[0]
    A = build_conv_matrix(f)          # shape (2n‑1, n)
    c = -np.ones(n, dtype=np.float64)  # minimise -Σb  == maximise Σb
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS is the default for modern SciPy and handles the size well.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return None
    return np.maximum(res.x, 0.0)


def lp_refine(start_f: np.ndarray,
              deadline: float,
              t_step: float = 0.20,
              max_iter: int = 30) -> np.ndarray:
    """
    Perform a handful of alternating LP iterations.
    Returns the best vector encountered (still normalised to sum = 1).
    """
    f = start_f.copy()
    best_f = f.copy()
    best_val = 2.0 * len(f) * full_convolution(f).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = full_convolution(f)
        M = conv.max()

        b = _lp_step(f, M)
        if b is None:
            print("[LP] LP failed – stopping refinement")
            break

        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Normalise b to the simplex (sum = 1)
        b_norm = b / sum_b

        # Convex combination
        f_new = (1.0 - t_step) * f + t_step * b_norm

        cur_val = 2.0 * len(f_new) * full_convolution(f_new).max()
        if cur_val < best_val:
            best_val = cur_val
            best_f = f_new.copy()
            print(f"[LP] iter {it:3d} – new best val {best_val:.6f}")

        f = f_new

        # Reduce step size if progress stalls
        if it % 10 == 0 and cur_val >= best_val * 0.9999:
            t_step = max(t_step * 0.5, 0.02)

    return best_f


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the exact objective
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray,
                deadline: float,
                lr: float = 0.12,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                jitter_interval: int = 1500) -> np.ndarray:
    """
    Adam optimisation of  max(conv(a,a))  with projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0

    best_a = a.copy()
    best_val = 2.0 * n * full_convolution(a).max()

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active offsets (those attaining the maximum up to tiny tolerance)
        eps_tol = max_val * 1e-9
        active = np.where(conv >= max_val - eps_tol)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # i runs i_min … i_max, j = k - i
            i = np.arange(i_min, i_max + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = project_to_simplex(a)

        # occasional jitter to escape shallow minima
        if jitter_interval and t % jitter_interval == 0:
            jitter = np.random.randn(n) * (lr * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        cur_val = 2.0 * n * full_convolution(a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # optional progress print
            if t % 200 == 0:
                print(f"[Adam] iter {t:5d} – improved val {best_val:.6f}")

    return best_a
