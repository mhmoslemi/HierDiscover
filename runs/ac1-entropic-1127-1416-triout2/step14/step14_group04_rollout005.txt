"""
Implementation of a multistage stochastic optimisation to minimise ``evaluate_sequence``.
We employ:
  - a broad baseline sampling with Dirichlet draws and simple parametric families,
  - a multiscale refinement: start with moderate length (~1024), then upsample to
    a larger length (4096) and repeat the local Adam optimisation,
  - a final simulated‑annealing polish.

The optimizer works on the log‑softmax parametrisation of the vector, uses a smooth
max (log‑sum‑exp) with an annealing temperature and Adam updates.
"""

import time
import numpy as np

# define convolution utilities
def _linear_convolution(a,b):
    m = a.size + b.size - 1
    n_fft = 1 << ((m-1).bit_length())
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A*B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def full_convolution(a):
    return _linear_convolution(a, a)

def compute_score(a):
    """Score = 2*n*max(conv), assuming sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())
```

def generate_exponential(n, lam):
    a = np.exp(-lam*np.arange(n))
    s = a.sum()
    return a/s if s>0 else np.full(n,0.0)

def generate_powerlaw(n,p):
    a = (np.arange(1,n+1,dtype=float))**(-p)
    s=a.sum()
    return a/s if s>0 else np.full(n,0.0)

def generate_beta(n, alpha,beta):
    i = np.arange(1,n+1,dtype=float)
    left = np.power(i, alpha-1)
    right = np.power(n-i+1, beta-1)
    a = left*right
    if not np.all(np.isfinite(a)):
        return np.full(n,0.0)
    s=a.sum()
    return a/s if s>0 else np.full(n,0.0)

def generate_gaussian(n,sigma):
    i=np.arange(n,float)
    mu=(n-1)/2.0
    a=np.exp(-0.5*((i-mu)/sigma)**2)
    s=a.sum()
    return a/s if s>0 else np.full(n,0.0)

def generate_monotone_random(n):
    v = np.random.exponential(scale=1.0,size=n)
    a = np.sort(v)[::-1]
    s=a.sum()
    return a/s if s>0 else np.full(n,0.0)

def generate_piecewise_linear(n, peak_pos, left_slope, right_slope):
    # linear up to peak, then linear down
    x = np.arange(n,float)
    a = np.where(x <= peak_pos,
                 1 + left_slope * (peak_pos - x),   # decreasing to peak? adjust
                 1 + right_slope * (x - peak_pos))
    a = np.maximum(a,0.0)
    s=a.sum()
    return a/s if s>0 else np.full(n,0.0)

def upscale_vector(vec, new_n):
    # linear interpolation from old points to new_n points
    old_n = len(vec)
    old_x = np.linspace(0,1,old_n)
    new_x = np.linspace(0,1,new_n)
    new_vec = np.interp(new_x, old_x, vec)
    # ensure non-negativity
    new_vec = np.maximum(new_vec,0.0)
    new_vec /= new_vec.sum()
    return new_vec

def smooth_optimize(a0, deadline, init_lr=0.08, init_tau_factor=1.0, min_tau=1e-9,
                    tau_decay=0.9995):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-30
    x = np.log(a + eps)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    # initial temperature based on conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # softmax
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        # conv
        conv = full_convolution(a)
        max_c = conv.max()
        # smooth max weighting
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            sumw = w.sum()
            if not np.isfinite(sumw) or sumw == 0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= sumw
        # gradient wrt a: 2 * conv_rev(a, w)
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)
        start = n-1
        grad_a = 2.0 * grad_full[start:start+n]
        # chain rule
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # temperature decay
        tau = max(min_tau, tau * tau_decay)
        # occasional exact evaluation
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth] it {it:7d} score {val:.6f} τ {tau:.2e}")
    return best_a

def simulated_annealing(a_start, deadline,
                        init_temp=1e-3, final_temp=1e-7,
                        cooling_factor=0.9995,
                        step_scale_init=0.03, step_scale_min=1e-7):
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val
    T = init_temp
    step_scale = step_scale_init
    it=0
    while time.time() < deadline:
        it+=1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j+=1
        # propose shift
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand,0.0)
        cand /= cand.sum()
        val = compute_score(cand)
        dval = val - cur_val
        if dval < 0 or np.random.rand() < np.exp(-dval / max(T,1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)
        T = max(final_temp, T * cooling_factor)
        if it % 5000 == 0:
            # jitter
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)
    return best_a

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    # choose problem size based on available time
    # if we have plenty of time, use n=4096 else fallback to n=2048
    # We'll try n=4096, but if not enough time after baseline will early exit.
    n = 4096
    # best placeholder
    best_seq=None
    best_val=float('inf')
    # fallback if global variable height_sequence_1 exists
    fallback = globals().get('height_sequence_1')
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            sc = compute_score(np.array(fb)/np.sum(fb))
            if sc < best_val:
                best_val, best_seq = sc, fb
                print(f"[fallback] official score = {val:.6f}")
        except Exception:
            pass
    # Stage 0: baseline search - allocate ~250s
    baseline_deadline = deadline - 500.0  # leave ~500 sec for refinement
    if time.time() < baseline_deadline:
        cand_vecs = baseline_search(n, baseline_deadline)
        for vec in cand_vecs:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best score {sc:.6f}")
    # Stage 1: fine 1-D sweeps - allocate ~150s
    sweep_deadline = deadline - 350.0
    if time.time() < sweep_deadline:
        sc, vec = fine_exponential_search(n, sweep_deadline)
        if sc is not None and sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[exp sweep] new best {sc:.6f}")
        if time.time() < sweep_deadline:
            sc, vec = fine_powerlaw_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[power sweep] new best {sc:.6f}")
    # Stage 2: local optimization via smooth Adam - allocate ~250s
    opt_deadline = deadline - 80.0
    if best_seq is not None and time.time() < opt_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, opt_deadline, init_lr=0.1, init_tau_factor=1.0,
                                  min_tau=1e-10, tau_decay=0.9995)
        sc = compute_score(refined)
        print(f"[smooth] refined score {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()
        # Extra restarts while time permits
        while time.time() < opt_deadline:
            remaining = opt_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            # random start
            conc = np.random.choice([0.1,0.3,0.7,1.5,3.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline, init_lr=0.1,
                                      init_tau_factor=1.0, min_tau=1e-10, tau_decay=0.9995)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[smooth restart] new best {sc:.6f}")
    # Stage 3: simulated annealing polish - allocate rest time
    anneal_deadline = deadline
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline,
                                    init_temp=1e-3, final_temp=1e-7,
                                    cooling_factor=0.9995,
                                    step_scale_init=0.02,
                                    step_scale_min=1e-7)
        sc = compute_score(final)
        print(f"[anneal] final score {sc:.6f}")
        if sc < best_val:

    # If no best found, fallback to uniform

        best_val = compute_score(np.array(best_seq))
        print("[fallback] using uniform")

"""
Multistage stochastic optimisation for ``evaluate_sequence``.

The optimiser works with a log‑soft‑max parametrisation of the coefficient
vector and minimises the smooth approximation of the max‑convolution
(soft‑max with an annealing temperature).  Stages:

1.  Broad random sampling (Dirichlet draws, monotone vectors,
    beta‑distribution, Gaussian, mixed exponentials).
2.  Dense 1‑D sweeps of exponential / power‑law families.
3.  Local Adam optimisation on the smooth objective, started from the
    best candidates and from several random restarts.
4.  A final simulated‑annealing polish.
5.  Simple linear blends of the best vector with a few‑other shapes
    (uniform, reversed, exponential) to escape possible plateaus.

The code keeps tight track of the wall‑clock (≈ 1000 s budget) and
returns the best sequence it has found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A * B
    conv = np.fft.irfft(C, n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector that already sums to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors that sum to 1)
# ----------------------------------------------------------------------
def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]               # decreasing
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_mixed_exponential(n: int, lam1: float, lam2: float, w: float) -> np.ndarray:
    """Convex mixture of a left‑hand and a right‑hand exponential."""
    left = np.exp(-lam1 * np.arange(n, dtype=float))
    right = np.exp(-lam2 * (np.arange(n, dtype=float)[::-1]))
    a = w * left + (1.0 - w) * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Book‑keeping utilities
# ----------------------------------------------------------------------
def _try_candidate(vec: np.ndarray, best, keep=20):
    """Update the list ``best`` with ``vec`` if it improves the score."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return
    vec = vec / s
    sc = compute_score(vec)
    if len(best) < keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Global baseline exploration (wide random search)
# ----------------------------------------------------------------------
def baseline_search(n: int, deadline: float):
    """Collect a set of promising candidates (up to ``keep`` best)."""
    keep = 20
    best = []   # list of (score, vector)

    # 1) Dirichlet draws with a range of concentrations
    alphas = [0.05, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_alpha = 1200
    for conc in alphas:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, keep)

    # 2) Random monotone vectors (sorted exponentials)
    for _ in range(2000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best, keep)

    # 3) Coarse Beta family sweep (covers U‑shapes and central peaks)
    al_vals = np.linspace(0.1, 8.0, 10)
    be_vals = np.linspace(0.1, 8.0, 10)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best, keep)

    # 4) Gaussian family sweep (different widths)
    sigma_vals = np.linspace(n / 80.0, n / 2.0, 14)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        a = generate_gaussian(n, sigma)
        _try_candidate(a, best, keep)

    # 5) Mixed‑exponential family (left/right asymmetry)
    for lam1 in [0.0003, 0.0008, 0.0015]:
        for lam2 in [0.0003, 0.0008, 0.0015]:
            for w in [0.2, 0.5, 0.8]:
                if time.time() >= deadline:
                    break
                a = generate_mixed_exponential(n, lam1, lam2, w)
                _try_candidate(a, best, keep)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]   # sorted from best to worst

def fine_exponential_search(n: int, deadline: float):
    best_sc = float("inf")
    best_vec = None
    # Slightly enlarged range – a logarithmic grid gives better coverage
    lam_vals = np.logspace(np.log10(2e-4), np.log10(5e-3), 5000)
    for lam in lam_vals:
        if time.time() >= deadline:

        a = generate_exponential(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec


def fine_powerlaw_search(n: int, deadline: float):
    best_sc = float("inf")
    best_vec = None
    p_vals = np.linspace(0.03, 8.0, 5000)   # extend a bit beyond the previous range
    for p in p_vals:
        if time.time() >= deadline:

        a = generate_powerlaw(n, p)
        sc = compute_score(a)
        if sc < best_sc:

    return best_sc, best_vec


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max of the convolution)
# ----------------------------------------------------------------------
def smooth_optimize(a0: np.ndarray,
                    deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 30.0,
                    min_tau: float = 1e-10,
                    tau_decay: float = 0.9996) -> np.ndarray:
    """
    Adam optimiser on the soft‑max approximation of ``max(conv(a,a))``.
    ``a0`` is assumed to sum to 1 (any constant factor is irrelevant).
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits safely
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise temperature from the current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---------- soft‑max ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---------- convolution ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # ---------- smooth max weights ----------
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if not np.isfinite(w_sum) or w_sum == 0.0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ---------- gradient wrt a ----------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3·n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ---------- back‑prop through the soft‑max ----------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- temperature annealing ----------
        tau = max(min_tau, tau * tau_decay)

        # occasional exact check
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# Light simulated annealing (final polish)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

# Helper for cheap linear blends of a few simple shapes

def linear_blends(base: np.ndarray, deadline: float):
    """Try a few convex combinations of ``base`` with simple reference shapes."""
    n = len(base)
    candidates = []

    # uniform distribution
    uniform = np.full(n, 1.0 / n)

    # reversed version (useful if the optimum is asymmetric)
    rev = base[::-1]

    # exponential with a moderate decay (hand‑crafted)
    exp_mid = generate_exponential(n, lam=0.001)

    # list of secondary vectors
    others = [uniform, rev, exp_mid]

    for other in others:
        for alpha in np.linspace(0.0, 1.0, 21):
            if time.time() >= deadline:
                break
            mixed = (1.0 - alpha) * base + alpha * other
            mixed = np.maximum(mixed, 0.0)
            mixed /= mixed.sum()
            candidates.append(mixed)

    # keep only the 10 best blends
    scored = [(compute_score(v), v) for v in candidates]
    scored.sort(key=lambda x: x[0])
    return [v for _, v in scored[:10]]

    blend_deadline = deadline - 80.0
    if best_seq is not None and time.time() < blend_deadline:
        base = np.array(best_seq, dtype=float)
        blend_candidates = linear_blends(base, blend_deadline)
        for vec in blend_candidates:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[blend] improved score = {sc:.6f}")

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (remaining time)
    # -----------------------------------------------------------------
    anneal_deadline = deadline
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline,
                                    init_temp=1e-3,
                                    final_temp=1e-7,
                                    cooling_factor=0.9995,
                                    step_scale_init=0.02,
                                    step_scale_min=1e-7)
        sc = compute_score(final)
        print(f"[anneal] final polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))
        print("[fallback] using uniform distribution")

    # -----------------------------------------------------------------
    # 7. Official evaluation (the harness will call ``evaluate_sequence``)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return best_seq
