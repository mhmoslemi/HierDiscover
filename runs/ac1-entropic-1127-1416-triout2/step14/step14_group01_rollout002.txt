def full_convolve(u, v):
    m = len(u) + len(v) - 1
    n_fft = 1 << ((m - 1).bit_length())
    U = np.fft.rfft(u, n=n_fft)
    V = np.fft.rfft(v, n=n_fft)
    return np.fft.irfft(U * V, n=n_fft)[:m]

"""
Gradient‑based smoothing of the peak autocorrelation.

The objective of ``evaluate_sequence`` (ignoring clipping) is
    2·n·max(conv(a,a)) / (sum(a)**2) .
Because the problem is scale‑invariant we restrict the search to the
probability simplex (sum(a)=1).  For a fixed β>0 the smooth function

    Lβ(a) = (1/β)·log Σ exp(β·conv(a,a))

approximate the max‑operator, and ∇Lβ can be written in closed form.
Using this we run a projected (simplex) gradient descent with a schedule
of increasing β, starting from several random/simplex points (including the
known fallback).  The best vector found is returned as a plain Python
list of floats.

The code respects the 1000 s budget (a 5 s safety margin) and never
uses closures or lambda expressions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    condition = u - cssv / ind > 0
    if not np.any(condition):
        return np.full_like(v, 1.0 / n)
    rho = np.where(condition)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _fft_len(m: int) -> int:
    """Smallest power of two ≥ m, for zero‑padded FFT."""
    return 1 << ((m - 1).bit_length())

def conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = _fft_len(m)
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    return conv

def full_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors (real FFT)."""
    m = u.shape[0] + v.shape[0] - 1
    n_fft = _fft_len(m)
    U = np.fft.rfft(u, n=n_fft)
    V = np.fft.rfft(v, n=n_fft)
    conv = np.fft.irfft(U * V, n=n_fft)[:m]
    return conv

# ----------------------------------------------------------------------
# Core optimisation: smooth approximation of max‑convolution
# ----------------------------------------------------------------------
def smooth_max_optimize(init_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max‑convolution.
    Returns the best vector found (still normalised to sum=1) and its
    objective value 2·n·max(conv).
    """
    a = np.array(init_a, dtype=float)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a /= a.sum()                     # keep on the simplex

    n = a.shape[0]
    best_a = a.copy()
    conv = conv_self(a)
    best_val = 2.0 * n * conv.max()
    it = 0

    # geometric schedule for β – grows → sharper approximation of max
    betas = np.geomspace(1.0, 2000.0, num=30)

    for beta in betas:
        # step size roughly independent of β (β appears in the gradient)
        step0 = 0.5 / beta
        while time.time() < deadline:
            it += 1

            # -------- current objective ----------
            conv = conv_self(a)
            maxc = conv.max()
            cur_val = 2.0 * n * maxc
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[opt] β={beta:.2f} it={it:06d}  best={best_val:.6f}")

            # -------- smooth weight (soft‑max) ----------
            shift = maxc                     # prevents overflow
            exp_vals = np.exp(beta * (conv - shift))
            w = exp_vals / exp_vals.sum()   # weight vector – length 2n‑1

            # -------- gradient of (1/β)·log‑sum‑exp ----------
            a_rev = a[::-1]                  # reversed a
            conv_w = full_convolve(w, a_rev)  # length 3n‑2
            # slice the part that corresponds to ∂/∂a_j
            grad = 2.0 * beta * conv_w[n - 1:n - 1 + n]

            # -------- simple step and projection ----------
            step = step0 / np.sqrt(it)
            a = a - step * grad
            a = project_to_simplex(a)

            # occasional tiny jitter to escape shallow traps
            if it % 2000 == 0:
                jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(it))
                a = project_to_simplex(a + jitter)

            # stop inner loop if the update becomes negligible
            if np.linalg.norm(step * grad) < 1e-13:
                break

        if time.time() >= deadline:
            break

    return best_a, best_val

# ----------------------------------------------------------------------
# Main entry point required by the sandbox harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑height sequence that minimises
    ``evaluate_sequence``.  The method mixes a known construction,
    several random simplex starts and a β‑schedule gradient descent
    on a smoothed max‑convolution objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    best_seq: list[float] | None = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Try the provided fallback construction (if it exists)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore[name-defined]
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Decide on a problem size (use the length of the fallback if any)
    # --------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
    else:
        n = 2048                       # default size that gave good results before

    # --------------------------------------------------------------
    # 2.  Build a list of starting vectors (simplex, uniform, random)
    # --------------------------------------------------------------
    starts: list[np.ndarray] = []

    # a)  the fallback (already normalised in the evaluate call)
    if best_seq is not None:
        vec = np.array(best_seq, dtype=float)
        if vec.sum() == 0:
            vec = np.full_like(vec, 1.0 / vec.shape[0])
        else:
            vec = np.maximum(vec, 0.0)
            vec /= vec.sum()
        starts.append(vec)

    # b)  uniform distribution
    starts.append(np.full(n, 1.0 / n, dtype=float))

    # c)  a few random Dirichlet draws
    for _ in range(3):
        rnd = np.random.rand(n)
        rnd = np.maximum(rnd, 0.0)
        rnd /= rnd.sum()
        starts.append(rnd)

    # --------------------------------------------------------------
    # 3.  Optimise each start within its own time slice
    # --------------------------------------------------------------
    slice_budget = (deadline - time.time()) / len(starts)
    for idx, start_vec in enumerate(starts):
        now = time.time()
        if now >= deadline - 5.0:
            break
        slice_deadline = now + min(slice_budget, deadline - now - 2.0)
        print(f"[optim] start {idx:02d} – slice ≈ {slice_deadline - now:.1f}s")
        opt_vec, _ = smooth_max_optimize(start_vec, slice_deadline)

        # evaluate with the official scorer
        seq = opt_vec.tolist()
        try:
            val = evaluate_sequence(seq)
            print(f"[optim] start {idx:02d} evaluated score = {val:.6f}")
            if val < best_score:
                best_score, best_seq = val, seq
        except Exception as exc:
            print(f"[optim] start {idx:02d} evaluation failed: {exc}")

    # --------------------------------------------------------------
    # 4.  Safeguard – if nothing succeeded, fall back to uniform
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 5.  Final report
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
