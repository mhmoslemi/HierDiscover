"""
Sidon‑set based optimiser.

The idea: place the mass on indices that form a *B₂* (Sidon) set, i.e.
all unordered sums i+j are distinct.  With such a set the linear
convolution a∗a has at most two equal contributions for any index,
so the maximal value of the convolution is 2·(max a)².  If we use k
non‑zero entries with equal weight 1/k, the score becomes

    score = 2·n·max(a∗a) = 4·n / k² .

The theoretical upper bound for k in a vector of length n is
k(k+1)/2 ≤ 2·n−1, i.e. k ≤ ⌊(√(8·n−7)−1)/2⌋ ≈ 2·√n.
For n=2048 the bound is k≤90, giving a best possible score ≈1.0.
The optimiser repeatedly builds random greedy Sidon sets, keeps the
largest one found before the deadline and returns the corresponding
uniform weight vector.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper for greedy Sidon set construction
# ----------------------------------------------------------------------


def greedy_sidon_set(n: int, time_deadline: float, max_attempts: int = 5000):
    """
    Search for a large additive Sidon set (all unordered sums distinct)
    inside {0,…,n‑1}.  Returns the best set (list of indices) found
    before the deadline.
    """
    # Theoretical maximum k from k(k+1)/2 ≤ 2·n−1.
    k_max = int(((8 * n - 7) ** 0.5 - 1) // 2)

    best_set = []
    best_len = 0
    attempts = 0

    while time.time() < time_deadline and attempts < max_attempts:
        attempts += 1
        # Random order – different orders give different maximal sets.
        order = np.random.permutation(n)
        used = np.zeros(2 * n, dtype=bool)   # used[sum] = True if already taken
        cur = []

        for x in order:
            # self‑sum must be free
            s2 = x + x
            if used[s2]:
                continue

            # check all sums with existing elements
            conflict = False
            for y in cur:
                if used[x + y]:
                    conflict = True
                    break
            if conflict:
                continue

            # accept x
            cur.append(int(x))
            used[s2] = True
            for y in cur[:-1]:          # only with previous elements
                used[x + y] = True

        cur_len = len(cur)
        if cur_len > best_len:
            best_len = cur_len
            best_set = cur
            print(f"[sidon] new best |S| = {best_len} (attempt {attempts})")
            if best_len >= k_max:       # cannot be improved theoretically
                break

    return best_set
