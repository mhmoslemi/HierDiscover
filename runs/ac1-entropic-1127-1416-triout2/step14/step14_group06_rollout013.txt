    best_a = a.copy()

    v = np.zeros_like(a) # momentum
    beta = 0.9
    t=0
    lambda_ = lambda_start
    while time.time() < deadline:
        t+=1
        conv = full_convolution(a)
        conv_max = conv.max()
        # compute softmax weights

            w = w / w_sum

        # gradient
        grad = 2.0 * np.correlate(w, a, mode='valid')

        step = step0 / math.sqrt(t)

                best_a = a.copy()

        if inc_every and t % inc_every == 0 and lambda_ < lambda_max:
            lambda_ = min(lambda_ + lambda_inc, lambda_max)
        # restart jitter if stagnation: if no improvement after some steps, jitter
        # we can keep a counter.
        # Not implement for now.

    return best_a

"""
search_step_function_v4.py

A new optimiser for the ``evaluate_sequence`` objective.

Key ideas
---------
*   Initialise the search with a small parametric sweep (reverse power‑law,
    geometric and a mixed reverse‑uniform shape).  The sweep is cheap
    because it works on a single vector of length *n*.
*   Refine the candidate with a **smooth‑max** gradient descent.
    The non‑smooth objective  max(conv) is replaced by a
    soft‑max (log‑sum‑exp) that is differentiable.  The gradient is a
    simple correlation that can be evaluated by FFT in O(n log n).
    A schedule on the soft‑max temperature (λ) progressively sharpens
    the surrogate, driving the solution towards the true maximum.
*   A momentum term and periodic jitter avoid shallow local minima.
*   Finally a stochastic hill‑climb (pairwise mass transfers) cleans
    up any remaining tiny overshoots.

The whole routine respects the 1000 s budget, falls back to a simple
mixed shape if nothing else succeeds, and always returns a list of
non‑negative numbers (the “heights” of the step function).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors (all are normalised to sum = 1)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p} (front‑loaded decreasing power‑law), p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    Convex combination   mix·reverse_power_law(p) + (1‑mix)·uniform,
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

# Very fast parametric sweep (produces a decent starting point)
# ----------------------------------------------------------------------
def parametric_sweep_basic(n: int, deadline: float) -> np.ndarray:
    """Coarse sweep over a few useful families – returns the best vector."""
    best_vec = None
    best_val = float('inf')

        a = reverse_power_law(n, float(p))

# Smooth‑max (log‑sum‑exp) polishing
# ----------------------------------------------------------------------
def smooth_max_polish(a0: np.ndarray, deadline: float,
                      step0: float = 0.4,
                      lambda_start: float = 5.0,
                      lambda_max: float = 3000.0,
                      lambda_inc: float = 250.0,
                      inc_every: int = 1200) -> np.ndarray:
    """
    Gradient descent on a smooth approximation of max(conv).
    The surrogate is   L(a)= (1/λ)·log Σ exp(λ·conv_k) .
    λ is increased progressively (annealing) to focus on the true max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    v = np.zeros_like(a)          # momentum term
    beta = 0.9                    # momentum coefficient
    lam = lambda_start
    last_improve = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        conv_max = conv.max()

        # soft‑max weights (numerically stable)
        shifted = conv - conv_max            # ≤ 0
        w = np.exp(lam * shifted)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the surrogate: 2·corr(w, a)
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # step size with classic 1/√t decay
        step = step0 / math.sqrt(t)

        # momentum update
        v = beta * v - step * grad
        a = a + v
        a = project_to_simplex(a)          # keep feasibility

        # periodic evaluation of the true objective
        if t % 200 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                last_improve = t

        # anneal λ (sharpen the surrogate)
        if inc_every and t % inc_every == 0 and lam < lambda_max:
            lam = min(lam + lambda_inc, lambda_max)

        # if we have been idle for a while, add a small jitter
        if t - last_improve > 3000:
            jitter = np.random.randn(a.shape[0]) * 0.0003
            a = a + jitter
            a = project_to_simplex(a)
            last_improve = t

    return best_a

def hill_climb(a_start: np.ndarray, deadline: float,
               step_scale: float = 0.02,
               max_no_improve: int = 200_000) -> np.ndarray:
    """Greedy pairwise mass transfers with occasional restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    it = 0
    no_improve = 0

        cand = a.copy()

        cand[j] += delta
        cand = project_to_simplex(cand)

            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

        val = evaluate_sequence(a_polished.tolist())
        print(f"[smooth max polish] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # -------------------------------------------------------------
    hill_deadline = deadline - 4.0
    if time.time() < hill_deadline:
        a_start = np.array(best_seq if best_seq is not None else a_initial,
                           dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # 5. Safeguard – if nothing succeeded, return a simple shape
    # -------------------------------------------------------------
    if best_seq is None:
        fallback_vec = mixed_reverse_uniform(n, mix=0.8, p=1.7)
        best_seq = fallback_vec.tolist()
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print("\nSearch finished. Reported score =", f"{final_score:.6f}")
    return best_seq
