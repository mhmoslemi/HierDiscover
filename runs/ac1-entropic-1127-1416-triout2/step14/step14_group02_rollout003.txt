    t = 0
    lr = 0.1
    lr_decay = 0.97
    while time.time() < deadline:
        conv = full_convolution(a)
        err = conv - target
        grad = correlation_grad(a, err, size_fft)
        t += 1
        m_t = 0.9 * m_t + 0.1 * grad
        v_t = 0.999 * v_t + 0.001 * (grad ** 2)
        m_hat = m_t / (1 - 0.9 ** t)
        v_hat = v_t / (1 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step
        a = project_to_simplex(a)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        if t % 2000 == 0:

            a = a + jitter
            a = project_to_simplex(a)
        # adjust lr
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)
    return best_a

"""
search.propose_candidate
-------------------------

A multi‑phase optimiser for the “max‑convolution” problem.
The objective is               2·n·max(conv(a,a)) / (sum(a)^2)
which is invariant under scaling, so we keep a on the probability simplex
(sum = 1, a_i ≥ 0).

Phases
------
1. **Broad baseline** – cheap analytic families (power‑law, exponential,
   beta, Gaussian, sine‑power, arcsine, triangular) and many Dirichlet
   draws.  The best few candidates are retained.
2. **L2‑flattening Adam** – minimise the squared‑error of the full
   convolution to the uniform target (the term that drives the convolution
   to be as flat as possible).  This is the Adam optimiser on the exact
   L2 loss (λ = 1).  Several starts are allocated a share of the remaining
   time budget.
3. **Projected sub‑gradient polish** – direct descent on the true
   max‑objective using the sub‑gradient of the maximum convolution entry.
4. **Simulated annealing** – a final Metropolis hill‑climb that can accept
   worsening moves and performs tiny mass transfers between coordinates.

All steps respect the global 1000 s wall‑clock limit and constantly keep the
best vector found (both the exact score and the official `evaluate_sequence`
value are reported).  The final result is returned as a plain list of
non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (real FFT)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of ½·‖a⋆vec‖² w.r.t. a.
    It equals 2·(a ⋆ vec) where ⋆ denotes cross‑correlation.
    `vec` must be the same length as the full convolution (2·n‑1).
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Exact score (identical to evaluate_sequence for sum‑1 vectors)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective value for a (may be un‑normalised)."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families (used in the baseline search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_arcsine(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_triangular(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = np.minimum(i, n - i + 1.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global baseline exploration (cheap random + analytic families)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 20000,
) -> list[np.ndarray]:
    """Generate a cheap pool of promising vectors."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.1, 5.0, 61):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.5, 80):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta grid
    al_vals = np.linspace(0.05, 2.0, 21)
    be_vals = np.linspace(0.05, 2.0, 21)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # Gaussian widths
    for sigma in np.linspace(0.1 * n, 0.45 * n, 25):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # sine‑power family
    for p in np.linspace(0.1, 8.0, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # arcsine distribution
    try_candidate(generate_arcsine(n))

    # triangular distribution
    try_candidate(generate_triangular(n))

    # random mixtures of families
    for _ in range(150):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam_mix = np.random.rand()
        try_candidate(mixture(a, b, lam_mix))

    # Dirichlet draws (various concentrations)
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on the L2 flattening loss (λ = 1)
# ----------------------------------------------------------------------


def adam_l2(
    a0: np.ndarray,
    deadline: float,
    lr_start: float = 0.12,
    lr_decay: float = 0.96,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser for minimizing ‖conv(a) – uniform‖².
    No smooth‑max term (λ = 1), so the objective is purely the
    squared‑error of the convolution to the constant target 1/(2n‑1).
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m
    size_fft = 1 << ((3 * n - 3).bit_length())  # safe padding for cross‑correlation

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    lr = lr_start

    while time.time() < deadline:
        conv = full_convolution(a)
        err = conv - target
        grad = correlation_grad(a, err, size_fft)

        t += 1
        # Adam moments
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)

        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape flat regions
        if t % 2200 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

        # learning‑rate schedule
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgradient_polish(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.34,
) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv) (exact objective)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # build sub‑gradient (sum of contributions from all active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2800 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny mass swaps)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-7,
    cooling_factor: float = 0.9995,
    step_scale_init: float = 0.05,
    step_scale_min: float = 5e-6,
) -> np.ndarray:
    """Metropolis search that swaps a small amount of mass between two coordinates."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 7000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a = a + jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin (harness will terminate if we exceed budget_s)
    deadline = start_time + budget_s - 5.0

    # problem size – larger n gives a tighter theoretical bound but costs more.
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a known construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] evaluate_sequence score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Baseline exploration (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 150.0)
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(
            n, baseline_deadline, max_keep=15, random_draws=20000
        )
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val = sc
                best_seq = vec.tolist()
        print(f"[baseline] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam L2‑flattening (multiple starts)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    # keep a modest reserve for polishing / annealing
    reserve = 200.0
    adam_budget = max(0.0, remaining - reserve)

    if best_seq is not None and adam_budget > 40.0:
        starts = [
            np.array(best_seq, dtype=float),
            np.full(n, 1.0 / n),  # uniform
        ]
        # a few random simplex points
        for _ in range(3):
            starts.append(project_to_simplex(np.random.rand(n)))

        per_budget = adam_budget / len(starts)
        for idx, svec in enumerate(starts):
            local_deadline = time.time() + per_budget
            a_opt = adam_l2(svec, local_deadline, lr_start=0.12)
            val = compute_score(a_opt)
            print(f"[adam_l2 {idx}] score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_opt.tolist()
        print(f"[adam_l2] best after phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.33)
        val = compute_score(a_polished)
        print(f"[polish] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (final ~100 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(
            a_start,
            anneal_deadline,
            init_temp=2e-3,
            final_temp=1e-7,
            cooling_factor=0.9995,
            step_scale_init=0.05,
        )
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()

    # -----------------------------------------------------------------
    # 5. Safeguard – uniform vector if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final reporting using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. evaluate_sequence score = {final_score:.6f}")
    return best_seq
