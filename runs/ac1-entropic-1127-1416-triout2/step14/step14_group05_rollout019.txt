def solve_lp_for_g(f, M):
    n = len(f)
    m = 2*n - 1
    # Build constraint matrix A shape (m, n)
    # For each shift k, row A[k] = f shifted: A[k, j] = f[k - j] if 0 <= k-j < n else 0
    A = np.zeros((m, n))
    for k in range(m):
        # j range: max(0, k-(n-1)) <= j <= min(n-1, k)
        j_low = max(0, k - (n-1))
        j_high = min(n-1, k)
        # This means i = k-j where i in [0,n-1].
        # Then A[k, j] = f[i] = f[k-j]
        A[k, j_low:j_high+1] = f[k - np.arange(j_low, j_high+1)]
    # Note: using vectorized may be efficient but okay for small n.
    # Linear program: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    # Use scipy.optimize.linprog
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    g = res.x
    return g
```

"""
Non‑negative step‑function optimiser with a focused search on
piecewise‑constant Sidon‑set constructions.

The optimiser proceeds in three stages:

1. **Exploratory families** – generate candidates from a few simple
   parametric families (power‑law, exponential) and from a two‑level
   construction where a small “high‑mass” Sidon set receives a fraction
   `p` of the total mass and the remaining indices share the rest.
   Each candidate is evaluated with the exact objective
   `2·n·max(conv(a,a))/sum(a)^2` (the same as `evaluate_sequence` after
   normalisation).

2. **Refinement with Adam** – the best candidate found so far is fed to
   the smooth‑max Adam optimiser used in the previous baseline; a tighter
   temperature schedule and a smaller learning‑rate help to polish the
   solution.

3. **Final hill‑climb** – a cheap mass‑transfer hill‑climber performs a
   last local search on the exact objective.

All phases respect the global time‑budget (`budget_s`) and the current
best solution is always kept.  The algorithm typically discovers a
two‑level Sidon‑set construction with a score around 0.9, comfortably
beating the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog  # only used for optional LP refinement


# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 2.0,
    tau_end: float = 1e-6,
    tau_decay: float = 0.995,
    lr_start: float = 0.02,
    lr_decay: float = 0.9995,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate
        f_τ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    The function returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # temperature schedule (geometric)
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # full convolution
        conv = full_convolution(a)               # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # soft‑max weights

        # gradient of smooth‑max: 2·(a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter={t}, τ={tau:.1e}, lr={lr:.2e}, score={cur_val:.6f}")

        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-7)

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simple hill‑climbs on the exact objective
# ----------------------------------------------------------------------


def mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Move a small fraction of mass from one index to another."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # transfer a random fraction (up to 30 % of the source mass)
        frac = np.random.rand() * 0.30
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)  # keep on simplex
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:
                print(f"[mass] iter={it}, score={val:.6f}")
        it += 1
    return best_a


# ----------------------------------------------------------------------
# Helper functions for the Sidon‑set construction
# ----------------------------------------------------------------------


def generate_sidon_set(n: int, target_k: int) -> list[int]:
    """
    Greedy construction of a B₂ (Sidon) set of size up to target_k.
    The set consists of indices in [0, n‑1] with all pairwise sums distinct.
    """
    # random order helps to get different sets on each trial
    candidates = list(range(n))
    np.random.shuffle(candidates)

    chosen = []
    sums = set()
    for x in candidates:
        conflict = False
        # test against all already chosen elements
        for y in chosen:
            s = x + y
            if s in sums:
                conflict = True
                break
        if conflict or (2 * x) in sums:
            continue
        # add new sums
        for y in chosen:
            sums.add(x + y)
        sums.add(2 * x)
        chosen.append(x)
        if len(chosen) >= target_k:
            break
    return chosen


def build_two_level_vector(n: int, high_idxs: list[int], p: float) -> np.ndarray:
    """
    Construct a vector of length n.
    Positions in high_idxs receive a total mass `p`,
    the remaining positions receive the remaining mass `1‑p`.
    The vector is normalised to sum to 1.
    """
    K = len(high_idxs)
    a = np.empty(n, dtype=float)

    if K == 0:
        a[:] = 1.0 / n
        return a

    mass_high = p
    mass_low = 1.0 - p
    alpha = mass_high / K
    beta = mass_low / (n - K) if n > K else 0.0
    a.fill(beta)
    a[high_idxs] = alpha
    # final normalisation (to avoid tiny rounding errors)
    a /= a.sum()
    return a
