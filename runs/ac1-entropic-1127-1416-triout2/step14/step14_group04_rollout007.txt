"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a))/(sum a)²).

Main ideas
==========

1. **Broad baseline** – a very wide random sampling that now also
   explores extreme Dirichlet concentrations, tiny‑beta beta‑distributions
   (U‑shapes) and classic window families (Hann, Hamming, Blackman).

2. **Fine 1‑D sweeps** – dense scans of the exponential, power‑law and
   beta families on expanded ranges (λ∈[5e‑4,2e‑3], p∈[0.05,4],
   α,β∈[0.01,20]).

3. **Projected sub‑gradient descent** – direct optimisation of the
   *exact* objective (the nonsmooth max of the convolution) on the
   probability simplex.  The gradient of the active convolution entry
   is obtained by a single FFT‑based convolution, then the iterate is
   projected onto the simplex with the O(n) algorithm of
   Duchi/Shalev‑Shwartz.

4. **Adam on a soft‑max** – kept as a secondary polish (the original
   implementation) but started from the best sub‑gradient result.

5. **Short simulated‑annealing polish** – final hill‑climb that can
   escape shallow plateaus.

All phases stop automatically when the wall‑clock budget (≈ 1000 s)
is exhausted.  The returned list is a plain Python list of floats;
the harness will call ``evaluate_sequence`` on it.

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Helper – projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, sum x = 1}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    Works for α,β > 0 (including values < 1 to obtain U‑shapes).
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_hann(n: int) -> np.ndarray:
    """Standard Hann window, shifted to be non‑negative."""
    i = np.arange(n, dtype=float)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def generate_hamming(n: int) -> np.ndarray:
    """Standard Hamming window."""
    i = np.arange(n, dtype=float)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s

    """Standard Blackman window."""
    i = np.arange(n, dtype=float)
    a = (
        0.42

    )
    s = a.sum()
    return a / s


def generate_linear_ramp(n: int) -> np.ndarray:
    """Increasing linear ramp a_i ∝ i+1."""
    a = np.arange(1, n + 1, dtype=float)
    s = a.sum()
    return a / s


def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape (peak in the centre)."""
    half = (n + 1) // 2
    asc = np.arange(1, half + 1, dtype=float)
    desc = asc[::-1] if n % 2 == 0 else asc[:-1][::-1]
    a = np.concatenate([asc, desc])
    s = a.sum()
    return a / s

def projected_subgradient_optimize(a0: np.ndarray, deadline: float,
                                   step0: float = 0.04,
                                   decay: float = 0.9998) -> np.ndarray:
    """
    Simple projected sub‑gradient method on the exact objective
    f(a)=2·n·max(conv(a,a)).  ``a`` lives on the simplex.
    Returns the best vector spotted before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step0
    it = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        k = int(conv.argmax())                     # active lag
        # gradient of conv[k] wrt a (see derivation in analysis)
        a_rev = a[::-1]
        w = np.zeros_like(conv)
        w[k] = 1.0
        grad_full = _linear_convolution(a_rev, w)   # length = 3·n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]     # shape (n,)

        a_candidate = _project_onto_simplex(a - step * grad)
        a = a_candidate
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[subgrad] it {it:6d} step {step:.2e} score {sc:.6f}")

        step *= decay
        if step < 1e-12:                # re‑inject a tiny step if we stall
            step = step0

    return best_a

# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.08,
                    init_tau_factor: float = 5.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9997) -> np.ndarray:
    """
    Adam on the soft‑max variables ``x`` (so that a = softmax(x)).
    The soft max uses a temperature ``tau`` that is gradually annealed.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- soft‑max weighting ----------
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient wrt a (2·∑_k w_k·a_{k−i}) ----------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the logits ----------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature ----------
        tau = max(min_tau, tau * tau_decay)

        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # -----------------------------------------------------------------
    # Problem size – 2048 is a good trade‑off between expressivity
    # and speed for the 1000 s budget.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")               # based on ``compute_score`` (sum=1)

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] official score = {val:.6f}")
            sc = compute_score(np.array(fb) / np.sum(fb))
            if sc < best_val:
                best_val, best_seq = sc, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Broad random baseline (≈ 180 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Focused 1‑D and 2‑D sweeps (≈ 180 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    if time.time() < sweep_deadline:
        # exponential sweep
        sc, vec = fine_exponential_search(n, sweep_deadline)
        if sc is not None and sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[exp‑sweep] new best = {sc:.6f}")

        # power‑law sweep
        if time.time() < sweep_deadline:
            sc, vec = fine_powerlaw_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[power‑sweep] new best = {sc:.6f}")

        # beta fine sweep
        if time.time() < sweep_deadline:
            sc, vec = fine_beta_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[beta‑sweep] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 3. Projected sub‑gradient descent (≈ 200 s)
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 150.0
    if best_seq is not None and time.time() < subgrad_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = projected_subgradient_optimize(a0, subgrad_deadline)
        sc = compute_score(refined)
        print(f"[subgrad] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

        # a few additional restarts while time permits
        while time.time() < subgrad_deadline:
            remaining = subgrad_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            # start from a fresh Dirichlet draw with moderate concentration
            a_start = np.random.dirichlet(np.full(n, 0.7))
            refined = projected_subgradient_optimize(a_start, sub_deadline)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[subgrad‑restart] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Smoothed Adam optimisation (≈ 150 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 30.0
    if best_seq is not None and time.time() < smooth_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, smooth_deadline)
        sc = compute_score(refined)
        print(f"[smooth‑opt] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (≈ 30 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(final)
        print(f"[anneal] final polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 7. Official evaluation (the harness will call ``evaluate_sequence``)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return best_seq
