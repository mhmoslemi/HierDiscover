"""
Search for a non‑negative coefficient sequence that minimises
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a))/sum(a)^2).

Algorithm
---------
1. **Broad random baseline** – Dirichlet draws and simple monotone vectors.
2. **LP‑refinement** – starting from the best baseline vector,
   solve a linear program that maximises the total mass of a
   candidate vector ``b`` under the constraint ``conv(f,b) ≤ M``,
   where ``M`` is the current worst convolution value.
   After normalising ``b`` we move the current solution ``f`` a
   small convex step towards ``b``.  Repeating this step drives the
   maximum convolution down (the Frank‑Wolfe/Greedy‑LP method from
   the paper description).
3. **Smooth Adam optimisation** – a final soft‑max (smooth‑max)
   optimisation of the max‑convolution objective to polish the result.
4. Return the best sequence found before the time budget expires.

The LP step is the main novelty compared with the previous
implementation and is expected to push the bound below the
previous 1.5103 value.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (real FFT – O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv(a,a)). ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors that sum to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.isfinite(a).all():
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Simple bookkeeping utilities
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best):
    """Update the (score, vector) list ``best`` if ``vec`` improves it."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        return
    vec = vec / s
    sc = compute_score(vec)
    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Broad random baseline
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float):
    best = []   # (score, vector) – smallest scores first

    # Dirichlet draws with several concentration parameters
    alphas = [0.1, 0.3, 0.7, 1.0, 2.0, 4.0]
    draws_per_alpha = 1500
    for conc in alphas:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    # Random monotone vectors
    for _ in range(2500):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # Coarse beta sweep (helps discover asymmetric shapes)
    al_vals = np.linspace(0.2, 8.0, 10)
    be_vals = np.linspace(0.2, 8.0, 10)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best)

    # Power‑law sweep (covers heavy‑tail shapes)
    p_vals = np.linspace(0.05, 3.0, 1200)
    for p in p_vals:
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, p)
        _try_candidate(a, best)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Linear‑programming refinement (Frank‑Wolfe / LP step)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A @ b = conv(f,b).
    Uses the Toeplitz structure of the convolution matrix.
    """
    n = f.shape[0]
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n - 1, dtype=f.dtype)])
    return toeplitz(col, row)


def lp_refine(f: np.ndarray, max_iters: int, deadline: float):
    """
    Perform up‑to ``max_iters`` LP‑based refinement steps.
    Each step solves  max Σ b_i  s.t.  b_i ≥ 0  and  conv(f,b) ≤ M,
    where M = max(conv(f,f)).  The solution b is normalised to sum 1
    and the current vector is moved a convex fraction towards it.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = compute_score(best_f)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current worst convolution value
        M = full_convolution(f).max()

        # Build constraint matrix A for conv(f, b) ≤ M
        A = build_conv_matrix(f)

        # Linear program: maximise sum(b)  → minimise -sum(b)
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None)] * n
        b_ub = np.full(A.shape[0], M, dtype=float)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': deadline - time.time()})

        if not res.success or res.x is None:
            # LP failed – abort the refinement loop
            break

        b = res.x
        # Normalise to the simplex (sum = 1)
        if b.sum() == 0.0:
            break
        g = b / b.sum()

        # Convex combination – a modest step size works best
        step = 0.35
        f_new = (1.0 - step) * f + step * g
        # Re‑normalise (numerical drift)
        f_new /= f_new.sum()

        new_score = compute_score(f_new)
        if new_score < best_score:
            best_score, best_f = new_score, f_new.copy()
            f = f_new
            print(f"[LP] iter {it:3d}  score {new_score:.6f}  M {M:.6e}")
        else:
            # If no improvement, shrink step size a little and keep trying
            step *= 0.7
            if step < 0.02:
                break
            f = f_new  # still move a bit to escape flat regions

    return best_f


# ----------------------------------------------------------------------
# Smoothed Adam optimisation (final polish)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.07,
                    init_tau_factor: float = 4.0,
                    min_tau: float = 1e-10,
                    tau_decay: float = 0.9995) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max to obtain probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = full_convolution(a)
        max_c = conv.max()

        # soft‑max weighting of the max (smooth max)
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a  (2·∑_k w_k·a_{k−i})
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        tau = max(min_tau, tau * tau_decay)

        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # problem size – 2048 gives a good balance between expressivity and speed
    n = 2048
    sum_target = 1.0               # we keep the simplex normalisation

    best_seq = None
    best_val = float("inf")        # based on ``compute_score`` (sum = 1)

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            fb = np.array(fb, dtype=float)
            fb = np.maximum(fb, 0.0)
            if fb.sum() == 0:
                raise ValueError
            fb = fb / fb.sum()
            sc = compute_score(fb)
            if sc < best_val:
                best_val, best_seq = sc, fb.tolist()
                print(f"[fallback] score = {sc:.6f}")
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Broad random baseline (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0   # leave 10 min for refinement
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. LP‑based greedy refinement (≈ 250 s)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 300.0
    if best_seq is not None and time.time() < lp_deadline:
        f = np.array(best_seq, dtype=float)
        # guarantee a tiny floor to avoid zero entries that would break the LP
        f = np.maximum(f, 1e-12)
        f /= f.sum()
        max_lp_iters = 30
        refined = lp_refine(f, max_lp_iters, lp_deadline)
        sc = compute_score(refined)
        print(f"[lp‑refine] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

    # -----------------------------------------------------------------
    # 3. Final smooth Adam polish (≈ remaining time)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a0 = np.array(best_seq, dtype=float)
        final = smooth_optimize(a0, deadline)
        sc = compute_score(final)
        print(f"[final‑smooth] polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 4. Very last fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))
        print("[fallback‑uniform] using uniform distribution")

    # -----------------------------------------------------------------
    # 5. Official evaluation (the harness will call ``evaluate_sequence``)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return best_seq
