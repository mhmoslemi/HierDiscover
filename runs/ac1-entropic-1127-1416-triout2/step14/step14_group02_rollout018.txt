"""
Improved search for the max‑convolution step‑function problem.

Approach
--------
1. Rich baseline exploration – many analytic families (power‑law,
   exponentials, Beta, Gaussian, sine‑power, two‑Gaussian mixtures,
   U‑shapes, inverse‑powers) plus a large pool of Dirichlet draws.
   The best few candidates are kept.

2. Gradient descent on a smooth‑max surrogate (log‑sum‑exp) with
   Adam – λ=0 so we minimise the max directly.  Temperature τ is
   annealed aggressively.

3. Projected sub‑gradient polishing – uses the exact max‑objective
   and a diminishing step‑size to squeeze out the remaining slack.

4. Final simulated‑annealing hill‑climb with a generous temperature
   schedule to escape tiny local minima.

All phases respect the 1000 s wall‑clock limit and constantly keep the
best vector found so far.  The returned list is a plain Python list of
non‑negative floats whose sum is 1 (scaling does not affect the score).
"""

import time
import numpy as np
from scipy import sparse  # required for the optional LP (not used directly)


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using the real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of a⋅vec (where vec = soft‑max weights or error term).
    Returns 2·(a ⋆ vec) where ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Scoring (exact objective for a vector summed to 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) (vector assumed to sum to 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_two_gaussians(
    n: int,
    mu1: float,
    sigma1: float,
    mu2: float,
    sigma2: float,
    lam: float,
) -> np.ndarray:
    x = np.arange(n, dtype=float)
    g1 = np.exp(-0.5 * ((x - mu1) / sigma1) ** 2)
    g2 = np.exp(-0.5 * ((x - mu2) / sigma2) ** 2)
    a = lam * g1 + (1.0 - lam) * g2
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_U_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution: big at the ends, small in the middle."""
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / (i * (n - i + 1)) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Baseline exploration (cheap, many random draws)
# ----------------------------------------------------------------------
def baseline_search_extended(
    n: int,
    deadline: float,
    max_keep: int = 20,
    random_draws: int = 50000,
) -> list[np.ndarray]:
    """Explore a rich collection of families and Dirichlet draws."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- power‑law sweep ----
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ---- exponential sweep ----
    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ---- beta grid ----
    al_vals = np.linspace(0.02, 3.0, 35)
    be_vals = np.linspace(0.02, 3.0, 35)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ---- random Beta draws (wide range) ----
    for _ in range(500):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-4, 0.5)
        beta = 10 ** np.random.uniform(-4, 0.5)
        try_candidate(generate_beta(n, alpha, beta))

    # ---- Gaussian sweep ----
    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ---- sine‑power sweep ----
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # ---- two‑Gaussian mixtures ----
    for _ in range(300):
        if time.time() >= deadline:
            break
        mu1 = np.random.uniform(0, n - 1)
        mu2 = np.random.uniform(0, n - 1)
        sigma1 = np.random.uniform(0.05 * n, 0.4 * n)
        sigma2 = np.random.uniform(0.05 * n, 0.4 * n)
        lam = np.random.rand()
        try_candidate(
            generate_two_gaussians(n, mu1, sigma1, mu2, sigma2, lam)
        )

    # ---- U‑shape families ----
    for p in np.linspace(0.05, 4.0, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_U_shape(n, p))

    # ---- random mixtures of three families ----
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-3, 0.5),
            10 ** np.random.uniform(-3, 0.5),
        )
        c = generate_gaussian(n, np.random.uniform(0.1 * n, 0.4 * n))
        lam1 = np.random.rand()
        lam2 = np.random.rand() * (1 - lam1)
        mixed = lam1 * a + lam2 * b + (1 - lam1 - lam2) * c
        try_candidate(mixed)

    # ---- Dirichlet draws (various concentration) ----
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Adam optimisation on smooth‑max (λ = 0 → pure max)
# ----------------------------------------------------------------------
def max_only_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.40,
    tau_end: float = 1e-8,
    tau_decay: float = 0.90,
    lr_start: float = 0.20,
    lr_decay: float = 0.95,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate:
        L(a) = τ·log Σ exp(conv/τ)
    No flattening term (λ=0).  The temperature τ is annealed.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    size_fft = 1 << ((3 * n - 3).bit_length())

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # ---- schedules ----
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # ---- convolution and smooth‑max weights ----
        conv = full_convolution(a)
        shift = conv.max()
        # stable log‑sum‑exp
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()  # soft‑max distribution

        # ---- gradient of smooth‑max ----
        grad = correlation_grad(a, w, size_fft)

        # ---- Adam update ----
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t + 1))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max objective)
# ----------------------------------------------------------------------
def subgradient_refine(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.55,
) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # use a diminishing step; we start a bit larger than before
        step = step0 / (t ** 0.7)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to shake out stagnation
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t + 1))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 5e-3,
    final_temp: float = 1e-9,
    cooling_factor: float = 0.9997,
    step_scale_init: float = 0.08,
    step_scale_min: float = 1e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a
