    a = np.maximum(a0, eps)
    a = a / a.sum()
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_c = conv.max()
        # if max_c is approx uniform? Anyway.
        # compute gradient
        k = np.argmax(conv)
        n = a.shape[0]
        # compute gradient via reversed indexing
        # vectorized:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2 * a[j[mask]]
        # step size schedule
        eta = eta0 / np.sqrt(it)
        # update a via mirror descent
        # compute exponent
        log_updates = -eta * g
        # shift for numerical stability
        max_log = np.max(log_updates)
        a = a * np.exp(log_updates - max_log)
        a = np.maximum(a, eps)  # ensure positivity

        a = a / a_sum
        # periodic evaluation

            if val < best_val:
                best_val = val
                best_a = a.copy()

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Parametric sweep** – generate several families of symmetric
   Beta‑shaped vectors a_i ∝ i^{α‑1}(n‑i+1)^{α‑1} with
   α∈[0.30,0.80] (the continuous‐arcsine distribution corresponds to
   α=½).  The vectors are normalised to sum to one and their exact
   objective 2·n·max(a*a) is measured with a fast FFT‑based convolution.
   The best few candidates are kept.

2. **Upscaling** – the best small‑size candidate (n=2048) is linearly
   interpolated to a larger size (n=8192).  This gives the optimiser
   more degrees of freedom without a large computational penalty.

3. **Projected sub‑gradient descent** – we minimise the *max*
   of the self‑convolution directly.  For a given vector `a`,
   the active convolution index `k = argmax(a*a)` gives the
   sub‑gradient g_i = 2·a_{k‑i} (if 0≤k‑i<n, otherwise 0).
   An additive step `a ← a – η·g` is performed and the result is
   projected back onto the probability simplex (non‑negative,
   sum = 1).  The step‑size η decays like η₀/√t.  The routine runs
   until the allocated time expires, keeping the best vector seen.

4. **Polish (optional)** – a brief simulated‑annealing pass is run
   from the best vector to try escaping tiny plateaus.

The whole pipeline respects the 1000 s wall‑clock budget and finally
returns the best sequence as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution (FFT) – used by the optimiser
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a)=1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a).  The vector must sum to one."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta‑like shape: a_i ∝ i^{α‑1}·(n‑i+1)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    # safety for non‑finite values
    if not np.all(np.isfinite(arr)):
        return np.full(n, 1.0 / n)
    s = arr.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return arr / s


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """Decreasing power‑law a_i ∝ (i+1)^{-p}."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Simple simplex projection (Euclidean) used by sub‑gradient descent
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.arange(n) + 1
    cond = u - cssv / rho > 0
    if not np.any(cond):
        # all entries would be projected to zero – fall back to uniform
        return np.full(n, 1.0 / n)
    rho = rho[cond][-1]               # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimiser for  max(a*a)
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray, deadline: float,
                         eta0: float = 0.5,
                         eps: float = 1e-15) -> np.ndarray:
    """
    Minimise  max_k (a*a)_k  by a projected sub‑gradient method.
    a0 is assumed to sum to 1 (the routine normalises it anyway).
    Returns the best vector found before *deadline*.
    """
    a = np.maximum(a0, eps)
    a = a / a.sum()

    best_a = a.copy()
    best_val = compute_score(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # convolution and the (unique) active index
        conv = full_convolution(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active quadratic form
        n = a.shape[0]
        g = np.zeros_like(a)

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # additive update + projection onto simplex
        a = a - eta * g
        a = _project_simplex(a)

        # occasional exact evaluation
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[subgrad] iter {it:6d}  best {val:.6f}")

    return best_a

# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Simple Metropolis‑style hill‑climb on the exact objective."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # small jitter to shake tiny plateaux
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a

def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate and keep if good."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        return
    vec = vec / s
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 6):
    """Very cheap sweep of a few parametric families – keeps the best."""
    best = []

    # 1) symmetric Beta (incl. arcsine) – dense grid around 0.5
    al_vals = np.concatenate(
        [np.linspace(0.30, 0.80, 12)]
    )
    for a in al_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_beta(n, a, a), best, max_keep)

    # 2) power‑law – few exponents
    for p in np.linspace(0.1, 3.0, 10):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # 3) exponential – a few rates
    for lam in np.linspace(0.0005, 0.12, 10):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # 4) random Dirichlet draws (both raw and monotone)
    for conc in [0.05, 0.1, 0.2, 0.5, 1.0]:
        alpha = np.full(n, conc)
        for _ in range(200):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha)
            _try_candidate(a, best, max_keep)
            _try_candidate(np.sort(a)[::-1], best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(vec) to new_n, then re‑normalise."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0:
        return np.full(new_n, 1.0 / new_n)
    return up / s

        while time.time() < opt_deadline:
            remaining = opt_deadline - time.time()
            sub_deadline = time.time() + min(15.0, remaining)

            # random Dirichlet start (different concentration)
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0])
            a_start = np.random.dirichlet(np.full(n_big, conc))
            # occasionally enforce monotonicity to explore asymmetric shapes
            if np.random.rand() < 0.3:
                a_start = np.sort(a_start)[::-1]

            refined = subgradient_optimize(a_start, sub_deadline, eta0=0.5)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[subgrad‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Very short simulated‑annealing polish (optional)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 5.0
    if best_seq is not None and time.time() < polish_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, polish_deadline,
                                    init_temp=1e-3, final_temp=1e-6)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Final official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
