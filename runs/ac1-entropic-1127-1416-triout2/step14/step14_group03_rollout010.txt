"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence`` within a 1000 s budget.

Algorithm
---------
1. **Broad baseline** – sweep several parametric families
   (power‑law, exponential, beta, Gaussian) with much wider
   parameter ranges than before, and supplement with many Dirichlet
   draws (including monotone versions).  Keep the best few candidates.

2. **Upscaling** – linearly interpolate the current best vector to a
   larger dimension (8192).  A larger problem size gives more freedom
   to flatten the convolution peak.

3. **Smooth max optimisation** – minimise a smooth approximation of
   ``max(a*a)``.  The smooth max is realised by a soft‑max (temperature
   τ) over the convolution values.  We optimise the log‑softmax
   parameters with Adam and a slowly decaying temperature.  Several
   short random restarts are performed while time permits.

4. **Long simulated‑annealing polish** – a Metropolis hill‑climb with
   a higher initial temperature and a longer schedule, capable of
   escaping tiny plateaux left by the gradient stage.

All stages are timed; the remaining budget after a stage is handed to
the next one.  The best vector found is returned as a plain Python
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenience wrapper for a*a."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a)=1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Return the exact evaluation ``2·n·max(a*a)``.
    ``a`` is assumed to be non‑negative and normalised to sum 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all vectors are normalised to sum 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 1.0 / n)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the probability simplex.
    """
    v = np.maximum(v, 0.0)
    if v.sum() == 0:
        return np.full_like(v, 1.0 / v.size)
    # standard projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        tau = 0.0
    else:
        rho = ind[cond][-1]
        tau = cssv[cond][-1] / rho
    w = np.maximum(v - tau, 0.0)
    return w


def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate and keep it if good."""
    if not np.isfinite(vec).all():
        return
    vec = _project_onto_simplex(vec)
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])

# Baseline exploration (very broad)

def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """
    Sweep a broad family of parametric shapes and many Dirichlet draws.
    Return up to ``max_keep`` best vectors found before ``deadline``.
    """
    best = []   # list of (score, vector)

    # ----- 1) power‑law (exponents up to 12) -----
    for p in np.linspace(0.1, 12.0, 350):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # ----- 2) exponential (rates up to 0.5) -----
    for lam in np.linspace(0.0001, 0.5, 350):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # ----- 3) beta (allowing U‑shapes) -----
    al_vals = np.concatenate([np.linspace(0.05, 0.9, 12),
                              np.linspace(1.0, 8.0, 12),
                              np.linspace(9.0, 20.0, 6)])
    be_vals = np.concatenate([np.linspace(0.05, 0.9, 12),
                              np.linspace(1.0, 8.0, 12),
                              np.linspace(9.0, 20.0, 6)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # ----- 4) Gaussian (wide sigma range) -----
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # ----- 5) Dirichlet draws (incl. monotone) -----
    # concentrations spanning several orders of magnitude
    concs = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            # monotone version – sort descending (large‑to‑small)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation + simplex projection)
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``vec`` to length ``new_n`` and project onto simplex."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    return _project_onto_simplex(up)

# Smoothed max optimisation (Adam on log‑softmax parameters)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.15,
                    init_tau_factor: float = 50.0,
                    min_tau: float = 1e-12,
                    tau_decay: float = 0.9999,
                    print_every: int = 40000) -> np.ndarray:
    """
    Minimise the smooth approximation of ``max(a*a)`` using Adam.
    Returns the best vector encountered before ``deadline``.
    """
    a = _project_onto_simplex(a0.copy())
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)                     # log‑softmax parameters

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature (smoothness)
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- soft‑max mapping ----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---- convolution & max ----
        conv = full_convolution(a)
        max_c = conv.max()

        # ---- smooth‑max weights ----
        if tau < 1e-14:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ---- gradient w.r.t. a (g = 2·conv_w * a) ----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n-2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ---- chain rule for log‑softmax (x‑space) ----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature decay ----
        tau = max(min_tau, tau * tau_decay)

        # ---- occasional exact evaluation ----
        if it % print_every == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """
    Metropolis‑style hill‑climb with a slowly cooling temperature.
    Returns the best vector seen before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        # Metropolis acceptance
        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to escape tiny plateaux
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = _project_onto_simplex(a)
            cur_val = compute_score(a)

    return best_a

            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 180 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 660.0          # leave ~340 s for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n_small, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            # evaluate via the exact score (fast, no clipping)
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Upsample the current best to the larger dimension
    # -----------------------------------------------------------------
    if best_seq is None:
        # safe start: uniform distribution
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)

    a0_big = upsample(a0_small, n_big)

    # -----------------------------------------------------------------
    # 3. Smoothed optimisation (≈ 250 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 300.0            # reserve ~300 s for SA
    if time.time() < smooth_deadline:
        # primary run
        refined = smooth_optimize(a0_big, smooth_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # a few cheap restarts while time permits
        while time.time() < smooth_deadline:
            remaining = smooth_deadline - time.time()
            # each restart gets at most 20 s
            sub_deadline = time.time() + min(20.0, remaining)

            # random Dirichlet start (wide concentration range)
            conc = np.random.choice([0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0])
            a_start = np.random.dirichlet(np.full(n_big, conc))
            # 40 % of restarts enforce monotonicity
            if np.random.rand() < 0.4:
                a_start = np.sort(a_start)[::-1]

            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (remaining time)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
