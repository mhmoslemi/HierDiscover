"""
Improved search for a non‑negative coefficient sequence that minimises

    evaluate_sequence

The algorithm consists of the following stages (total wall‑clock ≤ budget):
1. Broad random baseline – Dirichlet draws, monotone vectors, two‑level
   block families and a coarse β‑family sweep.
2. Focused 1‑D scans – exponential, power‑law, and two‑level height scans
   with a much finer resolution than before.
3. Smooth optimisation – Adam on a soft‑max approximation of the max‑conv,
   with an annealed temperature and occasional random jitter to escape
   plateaus.
4. Final polishing – a lightweight simulated‑annealing that moves a small
   amount of mass between random positions.

The search uses a larger problem size (n = 4096) which gives the optimiser
more degrees of freedom and typically yields a lower constant
`2·n·max(conv) / (∑a)²`.  All intermediate scores are printed for
monitoring.  The returned list is normalised (∑a = 1) and consists of
non‑negative Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective – the quantity we try to minimise (∑a = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum‑1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families – all return a normalised vector (sum == 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random exponential vector, sort decreasingly, normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_two_levels(n: int, front_val: float) -> np.ndarray:
    """
    Piecewise constant vector with two heights.
    First half (⌊n/2⌋ entries) receive `front_val`,
    second half receives the complementary mass.
    The vector is normalised; if the complementary height would be negative,
    the function returns a zero vector.
    """
    half = n // 2
    rest = n - half
    if front_val * half >= 1.0:
        return np.full(n, 0.0)
    back_val = (1.0 - front_val * half) / rest
    if back_val < 0.0:
        return np.full(n, 0.0)
    a = np.empty(n, dtype=float)
    a[:half] = front_val
    a[half:] = back_val
    return a


# ----------------------------------------------------------------------
# Small helper to update a bounded list of best candidates
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best):
    """
    Insert `vec` into `best` (a list of (score, vector)) if it improves
    the current worst entry.  The list is kept sorted by score (ascending)
    and trimmed to at most 12 elements.
    """
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return
    vec = vec / s
    sc = compute_score(vec)
    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])

def baseline_search(n: int, deadline: float):
    """Collect a diverse set of promising candidates."""
    best = []   # list of (score, vector)

    # 1) Dirichlet draws – a few concentration levels
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_alpha = 1500
    for conc in alphas:

        for _ in range(draws_per_alpha):

                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    for _ in range(3000):

            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # 3) Two‑level block family – sweep the front height
    half = n // 2
    front_vals = np.linspace(0.0, 2.0 / half, 1200)   # ensures back height non‑negative

                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best)

    return [vec for sc, vec in best]

# 3. Smoothed Adam optimisation on a soft‑max of the convolution
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.07,
                    init_tau_factor: float = 6.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9995,
                    jitter_every: int = 2500,
                    jitter_scale: float = 2e-3) -> np.ndarray:
    """
    Adam optimiser on the soft‑max (temperature τ) of the convolution.
    A small random jitter is injected periodically to help escape plateaus.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)                     # logits

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initial temperature proportional to the current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # -------- soft‑max mapping ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # -------- convolution ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # -------- soft‑max weighting ----------
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # -------- gradient w.r.t. a (2·∑_k w_k·a_{k−i}) ----------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)        # length = 3·n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # -------- chain rule for logits ----------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # -------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # -------- temperature decay ----------
        tau = max(min_tau, tau * tau_decay)

        # -------- occasional exact check ----------
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

        # -------- jitter to escape plateaus ----------
        if jitter_every and (it % jitter_every == 0):
            jitter = np.random.randn(n) * jitter_scale
            x += jitter

    return best_a


# ----------------------------------------------------------------------
# 4. Light simulated annealing for final polishing
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 2e-6,
                        cooling_factor: float = 0.9994,
                        step_scale_init: float = 0.025,
                        step_scale_min: float = 5e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two different indices with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to keep the walk lively
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allocated wall‑clock time.
    Returns a plain Python list of floats (∑a ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 4.0

    # --------------------------------------------------------------
    # Problem size – 4096 gives more flexibility while remaining fast.
    # --------------------------------------------------------------
    n = 4096

    best_seq = None
    best_val = float("inf")          # internal *compute_score* value (sum=1)

    # --------------------------------------------------------------
    # 0. Optional fall‑back from a previously saved solution
    # --------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            sc_official = evaluate_sequence(fb)
            print(f"[fallback] official score = {sc_official:.6f}")
            sc_internal = compute_score(np.array(fb) / np.sum(fb))
            if sc_internal < best_val:
                best_val, best_seq = sc_internal, fb
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Broad random baseline (≈ 180 s)
    # --------------------------------------------------------------
    baseline_deadline = deadline - 600.0          # reserve ~10 min for later stages
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best = {sc:.6f}")

    # --------------------------------------------------------------
    # 2. Focused 1‑D sweeps (≈ 150 s)
    # --------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    if time.time() < sweep_deadline:
        # exponential family
        sc, vec = fine_exponential_search(n, sweep_deadline)
        if sc is not None and sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[exp‑sweep] new best = {sc:.6f}")

        # power‑law family
        if time.time() < sweep_deadline:
            sc, vec = fine_powerlaw_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[pow‑sweep] new best = {sc:.6f}")

        # two‑level block family (very cheap to evaluate)
        if time.time() < sweep_deadline:
            sc, vec = fine_two_level_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[two‑lvl sweep] new best = {sc:.6f}")

    # --------------------------------------------------------------
    # 3. Smoothed Adam optimisation (≈ 200 s)
    # --------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, polish_deadline)
        sc = compute_score(refined)
        print(f"[smooth‑opt] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

        # additional random restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            # start from a fresh Dirichlet draw (moderate concentration)
            conc = np.random.choice([0.4, 0.9, 1.8, 3.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[smooth‑restart] new best = {sc:.6f}")

    # --------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 120 s)
    # --------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(final)
        print(f"[anneal] final polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # --------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # --------------------------------------------------------------
    # 6. Official evaluation (the harness will call ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return best_seq
