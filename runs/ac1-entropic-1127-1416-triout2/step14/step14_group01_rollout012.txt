def smooth_descent(a0, sub_deadline, step0=0.5, tau_factor=0.1, min_tau=1e-8):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    tau_factor_cur = tau_factor
    n = len(a)
    while time.time() < sub_deadline:
        it += 1
        conv = full_convolution(a)  # length 2n-1
        max_conv = conv.max()
        # compute temperature tau
        tau = max(max_conv * tau_factor_cur, min_tau)
        # compute soft weights
        # shift conv by max convex to maintain numerical stability
        # Use stable softmax: subtract max_conv
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            # all weights underflow; use uniform over active near max
            w = np.zeros_like(conv)
            w[np.isclose(conv, max_conv, atol=1e-12)] = 1.0
            w_sum = w.sum()
        w = w / w_sum
        # compute gradient via correlation
        # correlation = Σ_i a_i w_{i+j}
        # compute via FFT
        L = 1 << ((n + len(w) - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=L)
        W = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(A.conj() * W, n=L)[:n]
        grad = 2.0 * corr
        # step size schedule
        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

"""
Search for a non‑negative step‑height sequence that minimises
``evaluate_sequence`` (2·n·max(conv(a,a))/sum(a)^2).

Method
------
1. Use a few handcrafted families (uniform, exponential, triangular) as
   cheap baselines.
2. Perform many random restarts.  Each restart starts from a Dirichlet‑type
   random vector and runs a *smooth* gradient descent on a soft‑max
   approximation of the max‑convolution:
        Lτ(a) = log∑_k exp(conv_k/τ)
   The gradient of Lτ is computed via an FFT‑based correlation.
   The step size decays as 1/√t and the temperature τ shrinks slowly.
   After each iteration the iterate is projected onto the probability simplex.
3. Keep the best vector seen across all restarts.
4. Finally run a short projected sub‑gradient descent on the exact
   max‑convolution (as in the previous baseline) to polish the best result.
5. Return the sequence as a plain list of Python floats.

The whole routine respects the 1000 s time budget, prints progress,
and falls back to a uniform vector if nothing else succeeds.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power‑of‑two for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth “soft‑max” gradient descent
# ----------------------------------------------------------------------


def smooth_descent(start_a: np.ndarray,
                   deadline: float,
                   step0: float = 0.5,
                   tau_factor: float = 0.15,
                   min_tau: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Gradient descent on Lτ(a)=log∑exp(conv_k/τ).  Returns the best vector seen
    and its (exact) score.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    cur_tau_factor = tau_factor

    # length of the weight vector (full convolution)
    w_len = 2 * n - 1
    # size for correlation (linear, not circular)
    L = 1 << ((n + w_len - 1 - 1).bit_length())

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)                # length 2n‑1
        max_conv = conv.max()

        # temperature – a fraction of the current peak
        tau = max(max_conv * cur_tau_factor, min_tau)

        # soft‑max weights (stable: shift by max)
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:                     # numerical under‑flow – fall back
            w = np.zeros_like(conv)
            w[np.isclose(conv, max_conv, atol=1e-12)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2 * correlation(a, w)   (g_j = 2 Σ_i a_i w_{i+j})
        A = np.fft.rfft(a, n=L)
        W = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(A.conj() * W, n=L)[:n]   # linear correlation
        grad = 2.0 * corr

        # step size decays like 1/√t
        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_val = compute_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()

        a = a_new
        cur_tau_factor *= 0.995        # slowly tighten the soft‑max

        # occasional jitter to escape shallow basins
        if it % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(it))
            a = project_to_simplex(a + jitter)

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray,
                       deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # a tiny random perturbation every few thousand iterations
        if it % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(it))
            a = project_to_simplex(a + jitter)

    return best_a
