"""
Propose a step‑height sequence with a low evaluation score.

The algorithm works in three stages:

1. **Generate a pool of analytic windows** (uniform, triangular, Hann,
   Hamming, Blackman, exponential, arcsine, and a few beta‑shapes).  All
   candidates are normalised to a probability vector (sum = 1) and the
   best‐scoring one is kept as a seed.

2. **Smooth‑max gradient descent** – we minimise the exact objective
   `2·n·max(conv(a,a))` by approximating the max with a log‑sum‑exp
   smoothing.  For a current vector `a` (on the simplex) we compute

        λ_k = exp(β·c_k) / Σ_j exp(β·c_j)      (c = a∗a)

   where `β` grows gradually (1 → 80).  The gradient of the smoothed
   objective is

        g = 4·n·conv_valid(λ, a_rev)

   (`a_rev` is `a` reversed).  After normalising the gradient we take a
   diminishing step and project back onto the probability simplex.
   Several restarts (different initial windows) are performed until the
   time budget is exhausted.

3. **Projected sub‑gradient polishing** – with the remaining time we
   run a classic sub‑gradient descent on the true max‑objective, using
   the set of active convolution indices as the sub‑gradient direction.
   This refines the solution found by the smooth descent.

All heavy numerical work uses FFT‑based convolutions (via NumPy) and
the code is fully self‑contained, requiring only NumPy (which is always
available in the evaluation environment).  The final sequence is a
list of non‑negative floats; its score is returned by
`evaluate_sequence`.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Probability‑simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    # Quick path for already feasible vectors
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    # Shift by the mean of the positive part
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Normalise to exactly sum = 1 (to avoid numerical drift)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(w, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# FFT based convolutions (full and valid)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Small helper to find the next power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def fft_convolve_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (a,b are real)."""
    n = a.shape[0]
    m = b.shape[0]
    L = n + m - 1
    n_fft = _next_pow_two(L)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:L]
    return conv


def fft_convolve_valid(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Return the *valid* part of the convolution (length = len(a)-len(b)+1
    when len(a) ≥ len(b)).
    """
    n = a.shape[0]
    m = b.shape[0]
    L = n + m - 1
    n_fft = _next_pow_two(L)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv_full = np.fft.irfft(C, n=n_fft)[:L]

    # valid range: start = m-1, end = start + (n-m+1)
    start = m - 1
    end = start + (n - m + 1)
    return conv_full[start:end]


# ----------------------------------------------------------------------
# Objective utilities (scale invariant – sum = 1)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a full self‑convolution."""
    return fft_convolve_full(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def logsumexp(x: np.ndarray) -> float:
    """Numerically stable log‑sum‑exp."""
    mx = np.max(x)
    return mx + np.log(np.sum(np.exp(x - mx)))


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------
def smooth_max_descent(start_vec: np.ndarray,
                       deadline: float,
                       betas: list[float],
                       base_lr: float = 0.4,
                       max_total_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Perform gradient descent on the smooth approximation of the max.
    Returns the best vector encountered and its objective value.
    """
    n = start_vec.shape[0]
    a = start_vec.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    eps = 1e-12

    # main loop
    for beta in betas:
        while time.time() < deadline and it < max_total_iter:
            # ----- convolution and smooth weights -----
            conv = full_convolution(a)                     # length = 2n-1
            # smooth max weights λ_k = exp(β·conv_k) / Σ exp(...)
            exp_beta = beta * conv
            log_z = logsumexp(exp_beta)
            lam = np.exp(exp_beta - log_z)                # sums to 1

            # ----- gradient (4·n·conv_valid(lam, a_rev)) -----
            a_rev = a[::-1]
            grad = 4.0 * n * fft_convolve_valid(lam, a_rev)

            # normalise the gradient direction
            gnorm = np.linalg.norm(grad) + eps
            direction = grad / gnorm

            # diminishing step size
            step = base_lr / math.sqrt(it + 1.0)
            a = project_to_simplex(a - step * direction)

            # evaluate (cheap, uses FFT internally)
            cur_val = compute_score(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[smooth] β={beta:.1f} iter={it} best={best_val:.6f}")

            it += 1

            # occasional jitter to escape shallow basins
            if it % 800 == 0:
                jitter = np.random.randn(n) * (base_lr * 0.002 / math.sqrt(it))
                a = project_to_simplex(a + jitter)

            # stop inner loop if improvement stalls
            if it % 4000 == 0 and time.time() >= deadline:
                break
        if time.time() >= deadline:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient polishing on the exact max‑objective
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray,
                       deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Starts from a0 (already on the simplex) and runs until the deadline.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active set (indices attaining the max up to a tiny tolerance)
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]
        # build a uniform indicator on the active set (its sum = 1)
        lam = np.zeros_like(conv)
        if active.size > 0:
            lam[active] = 1.0 / active.size
        # sub‑gradient for the max‑objective (see derivation above)
        grad = 4.0 * n * fft_convolve_valid(lam, a[::-1])

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * grad)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[polish] iter={t} best={best_val:.6f}")

        # occasional random perturbation (helps escape plateaus)
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Candidate window generators (all return probability vectors)
# ----------------------------------------------------------------------
def generate_candidates(n: int):
    """Yield a sequence of analytically‑motivated start vectors."""
    # 1. uniform
    uniform = np.full(n, 1.0 / n)
    yield ("uniform", uniform)

    # 2. triangular (peak in the middle)
    tri = np.arange(1, n // 2 + 1)
    if n % 2 == 0:
        tri = np.concatenate([tri, tri[::-1]])
    else:
        tri = np.concatenate([tri, [n // 2 + 1], tri[::-1]])
    tri = tri.astype(float)
    yield ("triangular", tri / tri.sum())

    # 3. Hann window (zero at ends)
    i = np.arange(n)
    hann = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    yield ("hann", hann / hann.sum())

    # 4. Hamming window
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    yield ("hamming", hamming / hamming.sum())

    # 5. Blackman window
    blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
                + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    yield ("blackman", blackman / blackman.sum())

    # 6. Exponential tails (symmetric)
    for lam in [0.001, 0.003, 0.006, 0.01, 0.015]:
        centre = (n - 1) / 2.0
        exp_vec = np.exp(-lam * np.abs(i - centre))
        yield (f"exp_{lam:.3f}", exp_vec / exp_vec.sum())

    # 7. Arcsine‑type (β = 0.5 of the beta family)
    arcs = 1.0 / np.sqrt((i + 1.0) * (n - i))
    yield ("arcsine", arcs / arcs.sum())

    # 8. Beta‑family shapes (α in (0,1))
    for alpha in [0.2, 0.3, 0.4, 0.6, 0.8]:
        beta_shape = ((i + 1.0) ** (alpha - 1.0)) * ((n - i) ** (alpha - 1.0))
        # mask zeros that can appear for very small α
        beta_shape = np.where(beta_shape > 0, beta_shape, 0.0)
        if beta_shape.sum() == 0:
            continue
        yield (f"beta_{alpha:.1f}", beta_shape / beta_shape.sum())
