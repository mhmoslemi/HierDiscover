def local_search(a0, deadline):
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_val = compute_score(a)
    step_scale = 0.01  # initial relative step size relative to a[i] magnitude
    while time.time() < deadline:
        # pick i and j
        i = np.random.randint(n)
        if a[i] <= 0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # propose moving delta = step_scale * a[i]
        delta = step_scale * a[i]  # ensure not exceed a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # clamp small negative
        cand = np.maximum(cand, 0.0)
        # re-project onto simplex: ensures sum=1
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
            a = cand
            # possibly increase step scale modestly
            step_scale = min(step_scale * 1.02, 0.2)
        else:
            # maybe accept occasionally to escape local minima
            if np.random.rand() < 0.001:
                a = cand
            # decrease step scale gradually
            step_scale = max(step_scale * 0.998, 1e-5)
        # occasionally jitter globally
        if np.random.rand() < 0.001:
            jitter = np.random.randn(n) * step_scale * 0.5
            a = np.maximum(a + jitter, 0.0)
            a = project_to_simplex(a)
    return best
```

"""
Hybrid optimizer for the max‑convolution step‑function problem.

The algorithm proceeds in stages:
1. **Broad baseline search** – many random draws from analytic families
   (power‑law, exponential, beta, Gaussian, sine‑power) and Dirichlet
   distributions with a wide range of concentration parameters.  The best
   few candidates are retained.
2. **Smooth‑max Adam** – a custom Adam optimiser that minimises a
   log‑sum‑exp surrogate of the maximum convolution value.  No uniformity
   regulariser is used (λ=0) so the optimiser can concentrate mass where
   it helps to lower the true max.
3. **Targeted hill‑climbing** – local moves that transfer a small
   fraction of mass between two coordinates.  Moves are accepted only
   when they improve the exact objective (or very rarely to escape plateaus).
4. **Projected sub‑gradient polish** – a classic sub‑gradient descent on
   the exact max‑objective finishes the refinement.
5. **Final simulated‑annealing** – a short Metropolis anneal to
   possibly discover a tiny further improvement.

All vectors are kept on the probability simplex (sum = 1) via Euclidean
projection, guaranteeing that the official `evaluate_sequence` computes the
same value as the fast internal `compute_score`.  The whole routine respects
the 1000 s wall‑clock limit and always returns a feasible sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (linear, not circular)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real‑FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Correlation‑gradient utilities
# ----------------------------------------------------------------------


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of ½·∑_k (⟨a, rev(vec)⟩_k) where `vec` has length 2·len(a)-1.
    Implements 2·(a ⋆ vec) via FFT.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Exact objective used for internal comparisons (sum = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used in the baseline phase)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Broad baseline exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 12, random_draws: int = 20000):
    """Explore many analytic families and a large pool of Dirichlet draws."""
    best = []  # list of (score, vector)

    def consider(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- power‑law sweep ----
    for p in np.linspace(0.1, 5.0, 91):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # ---- exponential sweep ----
    for lam in np.linspace(0.0001, 0.5, 130):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # ---- beta grid ----
    al_vals = np.linspace(0.02, 2.0, 35)
    be_vals = np.linspace(0.02, 2.0, 35)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    # ---- random beta draws ----
    for _ in range(400):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        consider(generate_beta(n, alpha, beta))

    # ---- Gaussian sweep ----
    for sigma in np.linspace(0.1 * n, 0.45 * n, 40):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # ---- sine‑power sweep ----
    for p in np.linspace(0.1, 8.0, 45):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # ---- mixtures of families ----
    for _ in range(300):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        consider(mixture(a, b, lam))

    # ---- Dirichlet draws with varied concentration ----
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            consider(np.random.dirichlet([conc] * n))

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (λ = 0)
# ----------------------------------------------------------------------


def smooth_adam(a0: np.ndarray,
                deadline: float,
                tau_start: float = 0.5,
                tau_end: float = 1e-7,
                tau_decay: float = 0.94,
                lr_start: float = 0.35,
                lr_decay: float = 0.97,
                beta1: float = 0.90,
                beta2: float = 0.999,
                eps: float = 1e-8) -> np.ndarray:
    """Adam optimiser on the smooth‑max surrogate τ·log‑sum‑exp(conv/τ)."""
    a = a0.copy()
    n = a.shape[0]
    size_fft = 1 << ((3 * n - 3).bit_length())
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)  # stable log‑sum‑exp
        w = exp_vals / exp_vals.sum()

        grad = correlation_grad(a, w, size_fft)

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # schedules
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # occasional jitter to escape plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.33) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Targeted hill‑climbing (only accepting improvements)
# ----------------------------------------------------------------------


def hill_climb(a0: np.ndarray, deadline: float,
               init_step: float = 0.02, min_step: float = 1e-6) -> np.ndarray:
    """Local mass‑transfer search; moves are accepted only if they improve."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    step_scale = init_step

    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        # choose a distinct target index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * a[i]
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.02, 0.25)
        else:
            # very occasional uphill move to break ties
            if np.random.rand() < 0.001:
                a = cand
            step_scale = max(step_scale * 0.997, min_step)

        # occasional global jitter
        if np.random.rand() < 0.004:
            jitter = np.random.randn(n) * step_scale * 0.5
            a = np.maximum(a + jitter, 0.0)
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing final pass
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    """Metropolis hill‑climb with slowly decreasing temperature."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

# Main entry point

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin
    deadline = start_time + budget_s - 5.0

    # problem size – larger n gives a tighter theoretical bound,
    # but also higher per‑iteration cost.
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad baseline exploration (≈ 180 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 180.0)
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=12, random_draws=40000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Repeated smooth‑max Adam + hill‑climb cycles
    # -----------------------------------------------------------------
    # allocate the bulk of the remaining budget, keeping ~250 s for polishing/annealing
    remaining = deadline - time.time() - 250.0
    if remaining > 30.0:
        # generate a few diverse starting points
        start_vectors = []
        if best_seq is not None:
            start_vectors.append(np.array(best_seq, dtype=float))
        start_vectors.append(np.full(n, 1.0 / n))
        for _ in range(5):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        # time slice each start
        per_start = max(30.0, remaining / len(start_vectors))
        for idx, start_vec in enumerate(start_vectors):
            if time.time() >= deadline - 250.0:
                break
            local_deadline = time.time() + per_start
            # smooth‑max Adam
            a_adam = smooth_adam(start_vec, local_deadline,
                                 tau_start=0.5, tau_end=1e-7,
                                 lr_start=0.35, lr_decay=0.97)
            val_adam = compute_score(a_adam)
            print(f"[smooth‑adam {idx}] score = {val_adam:.6f}")
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.tolist()

            # immediate hill‑climb refinement
            hill_deadline = time.time() + max(15.0, (deadline - time.time() - 250.0) * 0.2)
            a_hill = hill_climb(a_adam, hill_deadline)
            val_hill = compute_score(a_hill)
            print(f"[hill‑climb {idx}] score = {val_hill:.6f}")
            if val_hill < best_val:
                best_val, best_seq = val_hill, a_hill.tolist()

    print(f"[after Adam+Hill] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Projected sub‑gradient polish (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.33)
        val = compute_score(a_polished)
        print(f"[polish] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing pass (≈ 100 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline,
                                      init_temp=2e-3, final_temp=1e-7,
                                      cooling_factor=0.9995,
                                      step_scale_init=0.04)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")

    return best_seq
