"""
Improved step‑function optimiser.

Approach
--------
1. **Rich baseline** – exhaustive sweep of analytic families (power‑law,
   exponential, beta, Gaussian, linear ramps) plus many Dirichlet draws.
   The best few candidates are kept.

2. **Pure max‑reduction Adam** – a smooth‑max surrogate (τ·log‑sum‑exp)
   of the convolution maximum.  No flatten‑term is used (λ=0) so the
   optimiser focuses entirely on lowering the peak value.  Adam steps
   are projected onto the probability simplex.

3. **Projected sub‑gradient polishing** – exact sub‑gradient descent on
   the true max‑objective with a decreasing step‑size.

4. **Slow‑cooling simulated annealing** – a final Metropolis hill‑climb
   with a long cooling schedule that makes tiny mass transfers between
   coordinates, allowing the algorithm to escape shallow plateaus and
   fine‑tune the solution.

The whole search respects the 1000 s wall‑clock limit, constantly
keeps the best vector seen and finally returns it as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sums to 1)."""
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D vectors")
    n = v.shape[0]
    # shift by mean to avoid all‑zero case
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # should never happen because we shift by -1 above
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (real FFT, returns length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of ½·∑ (a⋆vec) where ‘⋆’ is cross‑correlation.
    Returns 2·(a ⋆ vec).
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]

# ----------------------------------------------------------------------
# Exact objective (sum of a is 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective used by the optimiser: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Analytic families (used for baseline)
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return a / a.sum()

def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a / a.sum()

def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        a = np.zeros_like(i)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 1.0 / n)

def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return a / a.sum()

def generate_linear_decreasing(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=float)
    return a / a.sum()

def generate_linear_increasing(n: int) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float)
    return a / a.sum()

def mixture(v1: np.ndarray, v2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * v1 + (1.0 - lam) * v2
    return project_to_simplex(mixed)

# ----------------------------------------------------------------------
# Baseline exploration (cheap random / analytic draws)
# ----------------------------------------------------------------------
def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 40000,
) -> list[np.ndarray]:
    """Explore many families and Dirichlet draws, keeping the best `max_keep`."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = project_to_simplex(np.maximum(vec, 0.0))
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # analytic sweeps ----------------------------------------------------
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline: break
        try_candidate(generate_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline: break
        try_candidate(generate_exponential(n, lam))

    al_vals = np.linspace(0.02, 2.0, 21)
    be_vals = np.linspace(0.02, 2.0, 21)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline: break
            try_candidate(generate_beta(n, alpha, beta))

    for sigma in np.linspace(0.1 * n, 0.45 * n, 25):
        if time.time() >= deadline: break
        try_candidate(generate_gaussian(n, sigma))

    # linear ramps -------------------------------------------------------
    if time.time() < deadline:
        try_candidate(generate_linear_decreasing(n))
    if time.time() < deadline:
        try_candidate(generate_linear_increasing(n))

    # random mixtures ----------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline: break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(n,
                          10 ** np.random.uniform(-2, 0.3),
                          10 ** np.random.uniform(-2, 0.3))
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # Dirichlet draws -----------------------------------------------------
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline: break
            vec = np.random.dirichlet([conc] * n)
            try_candidate(vec)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]

# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max (λ = 0)
# ----------------------------------------------------------------------
def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.5,
    tau_end: float = 1e-8,
    tau_decay: float = 0.97,
    lr_start: float = 0.5,
    lr_decay: float = 0.98,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser of the smooth‑max surrogate of max(conv(a,a)).
    λ = 0, i.e. only the max‑term is present.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    size_fft = 1 << ((3 * n - 3).bit_length())

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # update schedules
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # smooth‑max weights
        conv = full_convolution(a)
        shift = conv.max()          # for numerical stability
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()  # soft‑max

        # gradient of smooth max = 2·(a ⋆ w)
        grad = correlation_grad(a, w, size_fft)

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur = compute_score(a)
        if cur < best_val:
            best_val, best_a = cur, a.copy()

        # occasional tiny jitter to shake out of plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / math.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a

# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.5) -> np.ndarray:
    """Exact sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: for each active shift k, add 2·a shifted
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur = compute_score(a)
        if cur < best_val:
            best_val, best_a = cur, a.copy()

        # occasional jitter
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a

# ----------------------------------------------------------------------
# Simulated annealing (slow cooling) for final fine‑tuning
# ----------------------------------------------------------------------
def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-2,
    final_temp: float = 1e-9,
    cooling_factor: float = 0.99995,
    step_scale_init: float = 0.2,
    step_scale_min: float = 1e-7,
) -> np.ndarray:
    """Metropolis hill‑climb with a long cooling schedule."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # propose moving a fraction of mass from i to j
        delta = step_scale * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < math.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if val < best_val:
                best_val, best_a = val, cand.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        # cool
        T = max(final_temp, T * cooling_factor)

        # occasional global jitter to avoid stagnation
        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = project_to_simplex(a + jitter)
            cur_val = compute_score(a)

    return best_a

# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence minimising
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    # problem dimension – larger n gives more flexibility
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. (optional) fallback – use the previously known construction if
    #    it exists in the global namespace.
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = compute_score(np.array(fallback))
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 130 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 130.0)
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=12, random_draws=50000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Pure max‑reduction Adam optimisation (≈ 400 s)
    # -----------------------------------------------------------------
    # Reserve ~200 s for later polishing / annealing
    adam_deadline = deadline - 200.0
    if best_seq is not None and time.time() < adam_deadline:
        start_vectors = [
            np.array(best_seq, dtype=float),                # current best
            np.full(n, 1.0 / n),                            # uniform
        ]
        # a few random starts
        for _ in range(3):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        for idx, sv in enumerate(start_vectors):
            if time.time() >= adam_deadline:
                break
            local_deadline = time.time() + min(40.0,
                                               (adam_deadline - time.time()) / len(start_vectors))
            refined = smooth_max_adam(sv, local_deadline)
            val = compute_score(refined)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
        print(f"[adam] best after this phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 180 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 70.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        polished = subgradient_polish(a0, polish_deadline, step0=0.5)
        val = compute_score(polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, polished.tolist()

    # -----------------------------------------------------------------
    # 4. Final simulated annealing (remaining time)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(
            a_start,
            anneal_deadline,
            init_temp=2e-2,
            final_temp=1e-9,
            cooling_factor=0.99995,
            step_scale_init=0.2,
            step_scale_min=1e-7,
        )
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
