    start_time = time.time()
    deadline = start_time + budget_s - 4.0

    best_seq = None
    best_val = float('inf')

    try:
        fallback = [float(x) for x in height_sequence_1]
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_seq, best_val = fallback, val
    except Exception:
        pass

    remaining = deadline - time.time()
    n = 4096 if remaining > 250 else 2048   # maybe use larger if enough time.

    # Parametric sweep
    sweep_deadline = deadline - 250.0  # allocate up to 250 seconds for sweeping
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val_raw = compute_score(a_initial)
            val = evaluate_sequence(a_initial.tolist())
            print(f"[parametric sweep] raw score = {val_raw:.6f}, official = {val:.6f}")
            if val < best_val:
                best_seq, best_val = a_initial.tolist(), val

    # Smoothing gradient descent
    gd_deadline = deadline - 80.0
    if best_seq is not None and time.time() < gd_deadline:
        a_start = np.array(best_seq, dtype=float)
        # multiple restarts with jitter
        while time.time() < gd_deadline - 5.0:
            # allocate a slice of time
            sub_deadline = min(gd_deadline, time.time() + 30.0)
            # jitter start

            a0 = a_start + jitter

            a_opt = smoothed_max_opt(a0, sub_deadline)
            val = evaluate_sequence(a_opt.tolist())
            print(f"[smoothed_opt] score = {val:.6f}")
            if val < best_val:
                best_seq, best_val = a_opt.tolist(), val
            # keep best as start for next restarts

"""
Hybrid optimiser for the step‑function objective.

Algorithm
---------
1. **Parametric sweep** – explores a large family of analytically‑defined
   shapes (reverse power‑law, beta, geometric, mixed with uniform).  This
   quickly gives a decent starting vector.

2. **Smoothed‑max gradient descent** – treats the objective
   `max(conv(a,a))` via a smooth approximation
   `log‑sum‑exp(λ·conv)/λ`.  The gradient of this surrogate is a
   convolution of the soft‑max weights with the current vector.
   An Adam optimiser (projected onto the probability simplex) is used.
   The temperature λ is increased during the run so the surrogate
   gradually approaches the true max.

3. **Final stochastic hill‑climb** – a cheap greedy local search that
   moves a small fraction of mass between two coordinates and keeps the
   best solution.  This refines the result obtained by the gradient run.

The whole search respects the 1000 s budget, adapts the vector length
(2048 or 4096) to the available time and always returns a feasible
non‑negative sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law), p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    Convex combination   mix * reverse_power_law(p) + (1‑mix) * uniform,
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Explore simple families and keep the best vector discovered."""
    best_vec = None
    best_val = float('inf')
    t0 = time.time()

    # ---- reverse power‑law -------------------------------------------------
    for p in np.arange(0.5, 12.0 + 1e-12, 0.2):

            if a is None:

            a = mixed_reverse_uniform(n, mix, p)
            if a is None:

            val = compute_score(a)
            if val < best_val:

    # ---- random convex combos of a few basic shapes ----------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    beta0  = beta_shape(n, 0.0, 0.0)       # another copy of uniform
    basics = [uniform, rp2, geom, beta0]

        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:

    # ----------------------------------------------------------------------
    print(f"[sweep] best raw score = {best_val:.6f} after {time.time()-t0:.1f}s")
    # ----------------------------------------------------------------------
    return best_vec


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (Adam, simplex projection)
# ----------------------------------------------------------------------
def smoothed_max_opt(a0: np.ndarray, deadline: float,
                     step0: float = 0.03,
                     lam_start: float = 0.5,
                     lam_end: float = 30.0,
                     lam_ramp: int = 2500) -> np.ndarray:
    """
    Minimise max(conv(a,a)) by optimising the smooth surrogate
    fλ(a) = log‑sum‑exp(λ·conv(a,a))/λ.
    Uses Adam with projection onto the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        t += 1
        # temperature λ (linear ramp)
        lam = lam_start + (lam_end - lam_start) * min(1.0, t / lam_ramp)

        # convolution a*a
        conv = full_convolution(a)
        max_conv = conv.max()

        # soft‑max weights over the 2n‑1 entries
        w = np.exp(lam * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient = 2 * cross‑correlation of w with a
        a_rev = a[::-1]
        conv_len = len(w) + len(a_rev) - 1
        nfft = 1 << ((conv_len - 1).bit_length())
        W = np.fft.rfft(w, n=nfft)
        A_rev = np.fft.rfft(a_rev, n=nfft)
        corr = np.fft.irfft(W * A_rev, n=nfft)[:conv_len]
        # slice gives n entries (indices n‑1 … n‑1+n‑1)
        g = 2.0 * corr[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # occasional evaluation
        if t % 400 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (keeps the best solution)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 150_000) -> np.ndarray:
    """Greedy move of a small fraction of mass between two coordinates."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]               # move a fraction of mass at i
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            # fresh restart from the current best
            jitter = np.random.randn(n) * 0.0007
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
