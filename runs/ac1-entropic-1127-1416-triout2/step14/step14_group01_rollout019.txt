def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    n = 2048

    # initial simple candidate: uniform
    best_seq = np.full(n, 1.0/n)
    best_score = compute_score(best_seq)
    print(f"[init] uniform score = {2*n*max_conv?} etc.

    # Search sigma for Gaussian shape
    idx = np.arange(n)
    mu = (n-1)/2.0
    # coarse grid
    sigmas = np.linspace(0.2*n, n, num=200)
    for sigma in sigmas:
        a = np.exp(-((idx - mu)**2)/(2*sigma*sigma))
        a /= a.sum()
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_seq = a
            best_sigma = sigma
        if time.time() > deadline:
            break

    # refine around best_sigma using golden-section
    if time.time() < deadline:
        lo = max(0.1*n, best_sigma*0.7)
        hi = min(n, best_sigma*1.3)
        phi = (np.sqrt(5)-1)/2
        for _ in range(30):
            sigma1 = lo + (1 - phi)*(hi - lo)
            sigma2 = lo + phi*(hi - lo)
            a1 = np.exp(-((idx - mu)**2)/(2*sigma1*sigma1))
            a1 /= a1.sum()
            s1 = compute_score(a1)
            a2 = np.exp(-((idx - mu)**2)/(2*sigma2*sigma2))
            a2 /= a2.sum()
            s2 = compute_score(a2)
            if s1 < s2:
                hi = sigma2
                if s1 < best_score:
                    best_score, best_seq = s1, a1
            else:
                lo = sigma1
                if s2 < best_score:
                    best_score, best_seq = s2, a2
            if time.time() > deadline:
               break

    # Polishing with subgradient descent for remaining time
    if time.time() < deadline:
        a0 = best_seq.copy()
        a_polished = subgradient_polish(a0, deadline, step0=0.20)
        val = compute_score(a_polished)
        if val < best_score:
            best_score, best_seq = val, a_polished

    # Ensure list
    best_seq = best_seq.tolist()
    final_score = evaluate_sequence(best_seq) # for printing
    print(f"Final reported score={final_score}")
    return best_seq
```

"""
Optimisation of a non‑negative step‑height sequence for the
``evaluate_sequence`` objective.

The objective is

    f(a) = 2·n·max(conv(a,a)) / (∑a)²     (n = len(a))

which is scale invariant, therefore we keep the vector on the probability
simplex (∑a = 1).  The algorithm works in three stages:

1. **Geometric warm‑start** –  a_i ∝ r^i  (r∈(0,1)) is a one‑parameter family.
   We scan the parameter r on a fine grid, keep the best vector (usually
   much better than the uniform start) and use it as the initial point.

2. **Adam optimisation on a smooth approximation of the max** –  
   we replace the nonsmooth max by a soft‑max
        Lβ(a)= (1/β)·log∑_k exp(β·(a*a)[k]),
   whose gradient can be written as a convolution and is obtained
   efficiently with FFTs.  Adam (first‑order momentum) together with a
   projection onto the simplex yields fast, stable descent.  The
   temperature β is increased during the run so that the approximation
   becomes progressively tighter.

3. **Optional polishing** – a short projected sub‑gradient descent
   (the same routine used in the reference implementation) is run for
   the last few seconds to squeeze a little more out of the result.

The whole search respects the 1000 s budget; if the time runs out early,
the best vector found so far is returned.  All numbers are non‑negative,
the sequence length is 2048 (the same size used by the reference code)
and the final list is fed to ``evaluate_sequence`` before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT based convolutions (linear, not circular)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * b using FFT.
    Returned length = len(a) + len(b) - 1 .
    """
    n = a.shape[0] + b.shape[0] - 1
    L = _next_pow2(n)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    return np.fft.irfft(C, n=L)[:n]


def autocorr_max(a: np.ndarray) -> float:
    """Exact value of  max(conv(a,a)).  a is assumed to sum to 1."""
    conv = full_convolution(a, a)
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a probability vector (∑a=1)."""
    n = a.shape[0]
    return 2.0 * n * autocorr_max(a)


# ----------------------------------------------------------------------
# Geometric warm‑start
# ----------------------------------------------------------------------


def geometric_vector(n: int, r: float) -> np.ndarray:
    """Return a probability vector a_i ∝ r**i , i=0..n‑1."""
    exps = r ** np.arange(n, dtype=np.float64)
    a = exps / exps.sum()
    return a


def best_geometric(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan the one‑parameter geometric family and return the best vector.
    The scan is coarse (500 points) and finishes early if time runs out.
    """
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = compute_score(best_a)

    # Scan r in (0.9, 0.9999) – the optimum is very close to 1.
    r_vals = np.linspace(0.90, 0.9999, 500)
    for r in r_vals:
        if time.time() >= deadline:
            break
        a = geometric_vector(n, r)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimisation on a smooth max (soft‑max) objective
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  beta_start: float = 1.0,
                  beta_end: float = 500.0,
                  lr: float = 0.05) -> np.ndarray:
    """
    Adam optimisation of the smooth max objective

        Lβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k])

    The temperature β is increased linearly from ``beta_start`` to
    ``beta_end``.  After every update the vector is projected onto the
    probability simplex.  The best vector (according to the true
    objective) is returned.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0                         # iteration counter
    best_a = a.copy()
    best_score = compute_score(a)

    # lengths needed for convolutions
    len_w = 2 * n - 1
    L = _next_pow2(3 * n - 2)     # size for FFT that fits a * w_rev

    while time.time() < deadline:
        t += 1
        # temperature schedule (linear)
        beta = beta_start + (beta_end - beta_start) * min(1.0, (t / 20000.0))

        # 1) compute full autocorrelation of the current a
        conv = full_convolution(a, a)          # length 2n-1

        # 2) soft‑max weights (stable formulation)
        max_c = conv.max()
        w = np.exp(beta * (conv - max_c))
        w /= w.sum()

        # 3) gradient of Lβ :  ∇Lβ = 2·a * w_rev   (linear convolution)
        w_rev = w[::-1]
        A = np.fft.rfft(a, n=L)
        W = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(A * W, n=L)[: (n + len_w - 1)]
        grad = 2.0 * grad_full[2 * n - 2 - np.arange(n)]

        # 4) Adam update (bias‑corrected)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # 5) project back onto the simplex (keeps a ≥ 0 , Σa = 1)
        a = project_to_simplex(a)

        # 6) keep the best vector seen so far
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] iter {t:6d} – new best {best_score:.6f}")

        # safety – stop early if improvements become negligible
        if t % 5000 == 0:
            if best_score < 1.55:          # we already beat the target
                # allow a little more time for polishing
                pass

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (identical to the previous baseline)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a, a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[Polish] iter {t:6d} – new best {best_val:.6f}")

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a
