"""
Propose a non‑negative step‑function (a probability vector) that
minimises the black‑box ``evaluate_sequence`` objective.

Strategy
--------
* **Baseline exploration** – sweep several analytic families
  (power‑law, exponential, beta, Hann, Tukey, sinusoidal powers) and
  a large number of Dirichlet draws on a modest size (n=4096).  Keep
  the best few candidates.
* **Upscale** – linearly interpolate the best candidate to a larger
  dimension (n=8192).  A larger support gives the optimizer more
  freedom to flatten the autocorrelation peak.
* **Smooth optimisation** – minimise a soft‑max approximation of the
  convolution peak using Adam on the log‑softmax parametrisation.
  Multiple random restarts are performed while time permits.
* **Exact sub‑gradient descent** – switch to the true objective
  (max‑convolution) and perform projected sub‑gradient steps on the
  probability simplex.  This removes the bias introduced by the
  smoothing.
* **Final simulated‑annealing polish** – a lightweight Metropolis
  hill‑climb to escape tiny plateaus.

All heavy work (convolutions, gradients) uses FFT‑based
implementations (O(n log n)).  The code respects the 1000 s wall‑clock
budget and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real inputs)
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b via FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a * a (full length)."""
    return _linear_convolution(a, a)

# ----------------------------------------------------------------------
# Exact objective (assuming sum(a)=1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """2·n·max(a*a)  (exact evaluate_sequence with sum(a)==1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Parametric families (all return a probability vector)
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s

def generate_exponential(n: int, lam: float) -> np.ndarray:
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s

def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        arr = np.zeros_like(arr)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)

def generate_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    arr = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = arr.sum()
    return arr / s

def generate_tukey(n: int, alpha: float) -> np.ndarray:
    """Tukey window with 0<=alpha<=1 (alpha=0 -> rectangular, alpha=1 -> Hann)."""
    if alpha <= 0:
        return np.full(n, 1.0 / n)
    if alpha >= 1:
        return generate_hann(n)
    i = np.arange(n, dtype=float) / (n - 1)
    w = np.empty_like(i)
    # first taper
    taper = 0.5 * (1 + np.cos(np.pi * (2 * i / alpha - 1)))
    w[i < alpha / 2] = taper[i < alpha / 2]
    # middle (flat) part
    w[(i >= alpha / 2) & (i <= 1 - alpha / 2)] = 1.0
    # second taper
    taper2 = 0.5 * (1 + np.cos(np.pi * (2 * i / alpha - 2 / alpha + 1)))
    w[i > 1 - alpha / 2] = taper2[i > 1 - alpha / 2]
    s = w.sum()
    return w / s

def generate_sin_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    # half‑sine shape (zero at edges, max at centre)
    arr = np.sin(np.pi * (i + 0.5) / n) ** p
    s = arr.sum()
    return arr / s

def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s

# ----------------------------------------------------------------------
# Baseline search (cheap, n_small)
# ----------------------------------------------------------------------
def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise a vector, evaluate it and keep it if it belongs to the elite."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return
    vec = vec / s
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
    else:
        if sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
    best.sort(key=lambda x: x[0])

def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """Explore many analytic families + Dirichlet draws on size n."""
    best = []   # (score, vector)

    # 1) Power‑law
    for p in np.linspace(0.1, 6.0, 250):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # 2) Exponential
    for lam in np.linspace(0.0001, 0.25, 250):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # 3) Beta
    al_vals = np.linspace(0.2, 8.0, 25)
    be_vals = np.linspace(0.2, 8.0, 25)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, a, b), best, max_keep)

    # 4) Hann & Tukey
    for sigma in np.linspace(n / 64.0, n / 2.0, 20):
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    _try_candidate(generate_hann(n), best, max_keep)

    for alpha in np.linspace(0.1, 1.0, 15):
        if time.time() >= deadline:
            break
        _try_candidate(generate_tukey(n, alpha), best, max_keep)

    # 5) Sin‑power family
    for p in np.linspace(0.5, 8.0, 20):
        if time.time() >= deadline:
            break
        _try_candidate(generate_sin_power(n, p), best, max_keep)

    # 6) Dirichlet draws (both raw and sorted)
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per = 500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            # monotone version (descending)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# ----------------------------------------------------------------------
# Upsampling (linear interpolation + simplex projection)
# ----------------------------------------------------------------------
def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Resize vec (length n) to new_n using linear interpolation and simplex projection."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return up / s

# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on soft‑max of convolution)
# ----------------------------------------------------------------------
def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.15,
                    init_tau_factor: float = 8.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9997) -> np.ndarray:
    """
    Minimise a smooth approximation of max(conv) using Adam on the log‑softmax
    parametrisation (a = softmax(x)).  Returns the best vector seen before deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise log‑softmax parameters
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature proportional to current max
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max → probability vector on simplex
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution and smooth weights
        conv = full_convolution(a)
        max_c = conv.max()

        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            ws = w.sum()
            if ws == 0 or not np.isfinite(ws):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= ws

        # gradient w.r.t a (g = 2 * conv of a_rev with w)
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

        # chain rule for x (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # occasional precise evaluation
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray, deadline: float,
                         init_step: float = 0.1) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective 2·n·max(conv).
    Uses one‑hot weight on the current maximal convolution index.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    step = init_step
    it = 0
    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        max_val = conv.max()
        # randomly pick one of the maximal indices (break ties)
        max_idxs = np.flatnonzero(conv == max_val)
        idx = max_idxs[np.random.randint(len(max_idxs))]

        w = np.zeros_like(conv)
        w[idx] = 1.0

        # gradient wrt a
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # gradient step and projection
        a = a - step * grad
        a = project_simplex(a)

        # evaluate
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # if we are improving, keep the step size; otherwise slowly decay it
        if it % 500 == 0:
            step = max(step * 0.9, 1e-6)   # gentle decay

        # optional debug output
        if it % 2000 == 0:
            print(f"[subgrad] it {it:7d}  score {val:.6f}")

    return best_a

# ----------------------------------------------------------------------
# Light simulated annealing polish (final tweak)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """
    Metropolis‐style hill‑climb allowing occasional uphill moves.
    Operates on the simplex (mass is conserved).
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)

        if a[i] <= 0.0:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # tiny random jitter to shake out plateaus
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the given wall‑clock budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem sizes
    n_small = 4096
    n_big = 8192

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback: any previously saved solution (if present)
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Baseline exploration on the smaller dimension
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 700.0   # reserve most time for fine optimisation
    if time.time() < baseline_deadline:
        cand_vecs = baseline_search(n_small, baseline_deadline, max_keep=8)
        for vec in cand_vecs:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Upsample best (or uniform) to the larger dimension
    # -----------------------------------------------------------------
    if best_seq is None:
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)

    a0_big = upsample(a0_small, n_big)

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation on the large problem
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if time.time() < smooth_deadline:
        # primary run from the up‑sampled vector
        refined = smooth_optimize(a0_big, smooth_deadline)
        sc = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

        # additional short restarts while time permits
        while time.time() < smooth_deadline:
            remaining = smooth_deadline - time.time()
            sub_dead = time.time() + min(30.0, remaining)

            # random Dirichlet start with varying concentration
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n_big, conc))
            # occasionally enforce monotonicity (descending)
            if np.random.rand() < 0.35:
                a_start = np.sort(a_start)[::-1]

            refined = smooth_optimize(a_start, sub_dead)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Exact sub‑gradient descent (sharpens the solution)
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 80.0
    if time.time() < subgrad_deadline and best_seq is not None:
        a_start = np.array(best_seq, dtype=float)
        refined = subgradient_optimize(a_start, subgrad_deadline, init_step=0.12)
        sc = compute_score(refined)
        print(f"[subgradient] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(final)
        print(f"[anneal] final score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 6. Safety fallback (uniform) – should never be needed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Report official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
