import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    K = toeplitz(np.concatenate([f, np.zeros(n-1)]), np.concatenate([f[:1], np.zeros(n-1)]))
    g = cp.Variable(n)
    constraints = [g >= 0, K @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    if prob.status not in ("optimal","optimal_inaccurate"):
        return None
    return g.value
```

"""
Improved optimiser for ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Broad parametric sweep** (≈300 s) over several analytical families
   (reverse‑power, multi‑power, sigmoid, geometric, mixed‑plateau,
   beta, power‑law) to obtain a good starting vector of length *n*.
2. **LP‑based refinement** (≈80 s).  Given a current vector `f`,
   we solve a linear program that finds the most mass we can add
   (as a vector `g`) without violating the current worst convolution
   bound `M = max(conv(f,f))`.  A tiny convex combination
   `f ← (1‑t)·f + t·g` is taken; the process is repeated a few times.
   This pushes mass from the high‑peak front to the low‑peak tail
   and typically lowers `M` substantially.
3. **Projected sub‑gradient polishing** (≈150 s).  We minimise
   `max(conv(a,a))` on the simplex using the exact sub‑gradient of the
   `max` functional and a decreasing step‑size.
4. **Stochastic hill‑climb** (the remaining time).  A simple but
   effective greedy random walk with adaptive step‑size refines the
   solution further.
5. **Fallback** – if everything fails, a simple linear decreasing
   shape is returned.

All helpers are top‑level; the mandatory entry point is
``propose_candidate``.  The code respects the 1000 s budget and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear algebra helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2*n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised (sum = 1) vector)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray | None:
    """a_i ∝ (n-i)^p, p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def power_law(n: int, p: float) -> np.ndarray | None:
    """a_i ∝ (i+1)^(-p), p ≥ 0."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def geometric(n: int, q: float) -> np.ndarray | None:
    """a_i ∝ q**i, 0 < q < 1."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray | None:
    """Front plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray | None:
    """a_i ∝ x_i^α (1‑x_i)^β, where x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray | None:
    """a_i ∝ (n‑i)^p · (i+1)^q."""
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray | None:
    """a_i ∝ 1/(1+exp((i‑c)/σ))."""
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a

# Broad parametric sweep
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray | None:
    """Explore a variety of families and keep the best vector discovered."""
    best_vec = None
    best_val = float('inf')

    # --- 1. reverse power law (coarse & fine) -------------------------
    print("[sweep] reverse_power_law (coarse)", flush=True)
    for p in np.linspace(0.1, 8.0, 81):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, p)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # fine sweep around best p (if we have a candidate)
    if best_vec is not None:
        print("[sweep] reverse_power_law (fine)", flush=True)
        # use a small neighbourhood around the best p seen so far
        # (here we just reuse the full fine grid)
        for p in np.linspace(0.05, 8.0, 161):
            if time.time() >= deadline:
                return best_vec
            a = reverse_power_law(n, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # --- 2. multi‑power -------------------------------------------------
    print("[sweep] multi_power", flush=True)
    for p in np.linspace(0.2, 4.0, 19):
        for q in np.linspace(0.0, 2.0, 21):
            if time.time() >= deadline:
                return best_vec
            a = multi_power(n, p, q)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # --- 3. sigmoid -----------------------------------------------------
    print("[sweep] sigmoid_shape", flush=True)
    for centre_frac in np.linspace(0.2, 0.8, 13):
        for sigma_frac in np.linspace(0.01, 0.15, 15):
            if time.time() >= deadline:
                return best_vec
            a = sigmoid_shape(n, centre_frac, sigma_frac)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # --- 4. original families (power, geometric, plateau, beta) ------
    for p in np.linspace(0.10, 4.00, 391):
        if time.time() >= deadline:
            return best_vec
        a = power_law(n, p)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # geometric – a huge range of q close to 1
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 110))):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    alphas = np.linspace(-0.8, 5.0, 25)
    betas = np.linspace(-0.8, 5.0, 25)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # --- 5. random Dirichlet samples -----------------------------------
    print("[sweep] random Dirichlet samples", flush=True)
    for _ in range(1500):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # --- 6. random mixtures of a few strong bases ----------------------
    print("[sweep] random mixtures", flush=True)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rev2 = reverse_power_law(n, 2.0)
    sig_mid = sigmoid_shape(n, 0.5, 0.05)
    basics = [uniform, rev2, sig_mid]
    for _ in range(1200):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.14) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 1400 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (same spirit as the previous version)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size and occasional restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]            # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming refinement (the new ingredient)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    from scipy.linalg import toeplitz
    HAVE_LP = True
except Exception:          # pragma: no cover
    HAVE_LP = False


def conv_matrix(f: np.ndarray) -> np.ndarray:
    """Dense convolution matrix K (size (2n‑1)×n) satisfying K @ g = conv(f,g)."""
    n = f.shape[0]
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    first_row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    return toeplitz(first_col, first_row)


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """Linear program: maximise Σ g_j   subject to  g≥0  and  f*g ≤ M."""
    if not HAVE_LP:
        return None
    n = f.shape[0]
    K = conv_matrix(f)                     # (2n‑1) × n
    g = cp.Variable(n)
    constraints = [g >= 0, K @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    except Exception:                      # pragma: no cover
        return None
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)


def lp_refinement(f_start: np.ndarray, deadline: float,
                  max_outer: int = 20, mix_init: float = 0.15) -> np.ndarray:
    """
    Iteratively improve ``f`` using the LP recipe from the paper.
    ``mix_init`` is the initial convex‑combination weight (t).
    """
    f = f_start.copy()
    t = mix_init
    it = 0
    while time.time() < deadline and it < max_outer:
        conv_f = full_convolution(f)
        M = conv_f.max()
        g_raw = solve_lp_max_sum(f, M)
        if g_raw is None:
            break
        sum_g = g_raw.sum()
        if sum_g <= 1e-12:
            break
        g = g_raw / sum_g                        # renormalise to sum = 1
        # try a convex update
        f_new = (1.0 - t) * f + t * g
        # evaluate new bound
        M_new = full_convolution(f_new).max()
        if M_new < M:
            f = f_new
            t = min(t * 1.04, 0.5)                # be a bit more aggressive
        else:
            t *= 0.5                               # step too large → shrink
            if t < 1e-4:
                break
        it += 1
    return f

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # 5 s safety margin

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # (0) Optional fallback from a previously discovered vector
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # (1) Parametric sweep – bulk of the time (≈300 s)
    # -------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val_proxy = compute_score(a_initial)
            val_real = evaluate_sequence(a_initial.tolist())
            print(f"[sweep] proxy = {val_proxy:.6f}, official = {val_real:.6f}")
            if val_real < best_val:
                best_val, best_seq = val_real, a_initial.tolist()

    # -------------------------------------------------------------
    # (2) LP‑based refinement (≈80 s)
    # -------------------------------------------------------------
    lp_deadline = deadline - 80.0
    if best_seq is not None and HAVE_LP and time.time() < lp_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_lp = lp_refinement(a0, lp_deadline, max_outer=25, mix_init=0.15)
        val = evaluate_sequence(a_lp.tolist())
        print(f"[LP‑refine] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_lp.tolist()

    # -------------------------------------------------------------
    # (3) Projected sub‑gradient polishing – uses remaining time
    # -------------------------------------------------------------
    polish_deadline = deadline - 30.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.14)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # (4) Final stochastic hill‑climb
    # -------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # (5) Safeguard fallback
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.0)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] using linear shape, score = {best_val:.6f}")

    # -------------------------------------------------------------
    # Final report
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
