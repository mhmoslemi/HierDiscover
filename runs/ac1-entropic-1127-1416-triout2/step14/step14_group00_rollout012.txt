"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

The procedure is a lightweight multi‑stage heuristic:
1. Initialise with the discrete arcsine distribution (Beta(½,½)), which is
   already close to the theoretical optimum.
2. Perform a fast random‑mixture search: repeatedly draw a random Beta(α,β)
   shape and mix it with the current best vector.  Any mixture that improves
   the objective replaces the incumbent.
3. Refine the incumbent with a projected sub‑gradient descent on the true
   objective 2·n·max(conv).  The step‑size decays as 1/√t and the iterate is
   always projected onto the probability simplex.
4. Finish with a short simulated‑annealing hill‑climb to escape shallow
   minima.

All stages respect the 1000 s wall‑clock budget.  The returned list contains
plain Python floats; ``evaluate_sequence`` will perform its own clamping.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape: a_i ∝ (i+1)^{α−1}·(n−i)^{β−1} (α,β>0)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s <= 0:
        return np.full(n, 0.0)
    return a / s


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape:
    a_i ∝ 1 / √((i+½)*(n‑½‑i)), i = 0 … n‑1.
    """
    i = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n - 0.5 - i))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (projected sub‑gradient descent)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # gradient of the max‑entry (sum of the gradients of the active entries)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to shake the iterate out of tiny plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb (keeps an upper bound)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Random mixture search (cheap global exploration)
# ----------------------------------------------------------------------


def random_mixture_search(a_best: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly generate a random Beta(α,β) shape and mix it with the current
    best vector.  Any mixture that improves the score is kept.  The loop runs
    until the supplied deadline expires.
    """
    n = a_best.shape[0]
    best_seq = a_best.copy()
    best_score = compute_score(best_seq)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # draw random exponents (log‑uniform helps to explore both ends)
        alpha = np.exp(rng.uniform(np.log(0.05), np.log(1.5)))
        beta = np.exp(rng.uniform(np.log(0.05), np.log(1.5)))
        a_beta = generate_beta(n, alpha, beta)

        lam = rng.random()   # mixing weight for the random shape
        cand = lam * a_beta + (1.0 - lam) * best_seq
        # numerical drift – re‑normalise to the simplex
        cand = cand / cand.sum()
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the 1000 s wall‑clock limit
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety cushion for the final evaluate_sequence call
    deadline = start + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem dimension – a moderate size gives a good trade‑off
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (if ``height_sequence_1`` is defined)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Deterministic strong start: arcsine distribution
    # -----------------------------------------------------------------
    a = generate_arcsine(n)
    sc = compute_score(a)
    print(f"[arcsine] initial score = {sc:.6f}")
    if sc < best_score:
        best_score, best_seq = sc, a.copy()

    # -----------------------------------------------------------------
    # 2. Random mixture search (global exploration)
    # -----------------------------------------------------------------
    mix_time = min(180.0, max(0.0, deadline - time.time() - 500.0))
    if mix_time > 0:
        mix_deadline = time.time() + mix_time
        print(f"[mix‑search] running for ~{mix_time:.0f}s")
        a_mixed = random_mixture_search(best_seq, mix_deadline)
        sc = compute_score(a_mixed)
        print(f"[mix‑search] score = {sc:.6f}")
        if sc < best_score:

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (majority of the remaining budget)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

            a_polished = subgradient_polish(np.array(best_seq, dtype=float),
                                            polish_deadline,

            if sc < best_score:
                best_score, best_seq = sc, a_polished.copy()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (final fine‑tuning)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

        a_final = simulated_annealing(np.array(best_seq, dtype=float),

    # -----------------------------------------------------------------

        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation (inside evaluate_sequence the clamping etc.
    # is performed).  The value printed here is the *reported* score.
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq.tolist()
