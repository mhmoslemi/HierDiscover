def adam_optimize(a0, deadline, max_iters=100000):
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    while time.time()<deadline and t < max_iters:
        t+=1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        # compute gradient g of M
        n = a.shape[0]
        idx = np.arange(n)
        g = np.zeros(n)
        for k in active:
            # j = k - idx
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2 * a[j[mask]]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.05 / (np.sqrt(v_hat) + epsilon)  # adapt per coordinate
        a = a - step * m_hat
        a = project_to_simplex(a)
        # check improvement
        conv_new = full_convolution(a)
        M_new = conv_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    return best_a
```

"""
propose_candidate – heuristic optimiser for `evaluate_sequence`.

The optimiser combines:
  * a broad parametric sweep over several analytically‑motivated families
    (reverse power‑law, forward power‑law, geometric, beta, and a
    reverse‑law/uniform mixture);
  * a large random‑convex‑combination search that mixes the above bases
    with arbitrary weights;
  * a projected sub‑gradient polish (the same idea as the reference);
  * a small‑scale Adam optimiser that directly follows the sub‑gradient
    of the max‑convolution;
  * a final stochastic hill‑climb that moves tiny amounts of mass
    between two coordinates.

The search respects the 1000 s budget, dynamically picks the vector length
(4096 when enough time is left, otherwise 2048), and always returns a list
of non‑negative floats.  Intermediate scores are printed for diagnostics.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    Scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law), p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p} (decreasing forward power‑law)."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    convex combination   mix * reverse_power_law(p) + (1‑mix) * uniform,
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Initial broad parametric sweep + random composites
# ----------------------------------------------------------------------
def random_composite_search(n: int, deadline: float,
                            max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Search for a good shape by convexly mixing a few base families."""
    best_vec = None
    best_val = float('inf')
    # base families (each expects n and a parameter)
    base_families = [
        lambda n: np.full(n, 1.0 / n),                     # uniform
        lambda n, p: reverse_power_law(n, p),
        lambda n, p: power_law(n, p),
        lambda n, q: geometric(n, q)
    ]

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # choose number of components (2 or 3)
        ncomp = np.random.choice([2, 3])
        components = []
        for _ in range(ncomp):
            fam = np.random.choice(['uniform', 'rev', 'pow', 'geom'])
            if fam == 'uniform':
                comp = np.full(n, 1.0 / n, dtype=np.float64)
            elif fam == 'rev':
                p = 10 ** np.random.uniform(-2, 2)       # log‑uniform over [0.01,100]
                comp = reverse_power_law(n, p)
            elif fam == 'pow':
                p = 10 ** np.random.uniform(-2, 2)
                comp = power_law(n, p)
            else:   # geom
                q = np.random.uniform(0.80, 0.9999)
                comp = geometric(n, q)
            components.append(comp)

        # random convex weights via Dirichlet
        w = np.random.dirichlet(np.ones(ncomp))
        a = sum(c * wi for c, wi in zip(components, w))
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec, best_val


def initial_search(n: int, deadline: float) -> np.ndarray:
    """Perform a fast parametric sweep and fall back to a random composite search."""
    best_vec = None
    best_val = float('inf')

    # ---- reverse power‑law -------------------------------------------------
    for p in np.arange(0.5, 12.0 + 1e-12, 0.2):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, float(p))
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- forward power‑law -------------------------------------------------
    for p in np.arange(0.5, 12.0 + 1e-12, 0.2):
        if time.time() >= deadline:
            break
        a = power_law(n, float(p))
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- geometric ---------------------------------------------------------
    for q in np.arange(0.80, 0.999 + 1e-12, 0.02):
        if time.time() >= deadline:
            break
        a = geometric(n, float(q))
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- mixed reverse‑law + uniform ---------------------------------------
    for mix in np.linspace(0.2, 0.98, 20):
        for p in np.linspace(0.5, 6.0, 30):
            if time.time() >= deadline:
                break
            a = mixed_reverse_uniform(n, float(mix), float(p))
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- beta shapes (coarse grid) -----------------------------------------
    alphas = np.linspace(0.0, 10.0, 11)
    betas = np.linspace(0.0, 10.0, 11)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = beta_shape(n, float(alpha), float(beta))
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- random convex combinations of basic shapes ------------------------
    # Give this stage the majority of the remaining time.
    remaining = deadline - time.time()
    if remaining > 5.0:
        vec, val = random_composite_search(n, deadline)
        if vec is not None and val < best_val:
            best_val, best_vec = val, vec

    return best_vec


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (as before, but a bit more aggressive)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # gradient of the max‑convolution (see discussion in analysis)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape shallow plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Adam‑style optimisation on the max‑convolution
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray, deadline: float,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam optimiser that follows the sub‑gradient of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (same idea as the reference)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 200_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size and restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                 # move a fraction of the source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
