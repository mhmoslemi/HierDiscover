def interleaved_permutation(a):
    # sorted descending values but keep sum
    sorted_vals = np.sort(a)[::-1]
    n = len(a)
    result = np.empty_like(a)
    left = 0
    right = n-1
    toggle = True
    for v in sorted_vals:
        if toggle:
            result[left] = v
            left += 1
        else:
            result[right] = v
            right -= 1
        toggle = not toggle
    return result
```

def simulated_annealing_hill_climb(a_start, deadline, max_iter=2_000_000):
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_M = compute_score(a)  # since compute_score = 2 n max(conv)
    # compute M (max conv) consistent: M = best_M/(2*n)
    # We'll use score directly.
    best_score = best_M
    # temperature schedule: start high, gradually reduce.
    T0 = 0.01 * best_score   # maybe 1% of score
    t = 0
    while time.time() < deadline and t < max_iter:
        # propose move:
        if np.random.rand() < 0.2:  # swap move
            i, j = np.random.randint(n, size=2)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = 0.02 * a[i]  # move fraction
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.clip(cand, 0, None)
        cand = project_to_simplex(cand)
        score = compute_score(cand)
        # accept if better or with probability
        if score < best_score:
            best_score, best_a = score, cand.copy()
            a = cand
        else:
            # temperature schedule
            T = T0 * math.exp(-t / (max_iter/10))  # decays
            if np.random.rand() < math.exp(-(score - best_score) / max(T, 1e-12)):
                a = cand
        t += 1
        # occasional jitter
        if t % 50000 == 0:
            jitter = np.random.randn(n) * 1e-4
            a = a + jitter
            a = np.clip(a, 0, None)
            a = project_to_simplex(a)
    return best_a
```

"""
search_step_function_v4.py

A multi‑phase optimiser for ``evaluate_sequence``.
The pipeline is:

1.  Broad parametric sweep over many analytic families (power‑law,
    reverse‑power, multi‑power, sigmoid, beta, geometric, exponential).
2.  Large random Dirichlet sampling and random convex mixtures of strong
    bases.
3.  Re‑ordering (permutation) of the best vector – monotone vectors are
    deliberately shuffled (random permutations and a deterministic
    “interleaved” high‑low ordering) to break harmful correlations.
4.  Projected sub‑gradient polishing (fast diminishing step‑size).
5.  Simulated‑annealing hill‑climb that mixes tiny mass transfers and
    full swaps, accepting uphill moves with a temperature schedule.
All helpers are top‑level (no closures).  The search respects the 1000 s
budget by checking the deadline after each cheap operation.
"""

# ----------------------------------------------------------------------
# Linear‐algebra utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # uniform simplex if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    # Find the rho where the threshold condition holds
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def reverse_power_law(n: int, p: float) -> np.ndarray | None:
    """a_i ∝ (n‑i)^p,  p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray | None:
    """a_i ∝ (i+1)^(‑p).  p ≥ 0."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray | None:
    """a_i ∝ q^i, 0 < q < 1."""
    if not (0.0 < q < 1.0):
        return None
    a = q ** np.arange(n, dtype=np.float64)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray | None:
    """Front plateau of length L then a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray | None:
    """a_i ∝ x_i^α (1‑x_i)^β,  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray | None:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.
    Both exponents ≥ 0 give a front‑loaded shape.
    """
    left = np.arange(n, 0, -1, dtype=np.float64) ** p      # (n‑i)
    right = np.arange(1, n + 1, dtype=np.float64) ** q    # (i+1)
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray | None:
    """
    a_i ∝ 1/(1+exp((i−c)/σ)).
    centre_frac ∈ (0,1) is the location of the midpoint,
    sigma_frac controls the steepness (σ = sigma_frac·n).
    """
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def exponential_decay(n: int, lam: float) -> np.ndarray | None:
    """a_i ∝ exp(-lam·i), lam > 0."""
    if lam <= 0.0:
        return None
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a

    # All entries should be non‑negative already; we simply normalise.
    s = vec.sum()
    if s <= 0:
        return float('inf')
    a = vec / s
    return compute_score(a)

def random_sampling(n: int, deadline: float,
                    best_vec: np.ndarray | None,
                    best_val: float) -> tuple[np.ndarray | None, float]:
    """Generate a large number of random candidates and keep improvements."""
    # Random Dirichlet draws
    print("[random] Dirichlet sampling", flush=True)
    while time.time() < deadline:
        a = np.random.dirichlet(np.ones(n))
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a
        # limit number to a few thousand (break condition inside while)
        if np.random.rand() < 1e-4:   # occasionally exit early
            break

    # Random convex mixtures of a few strong bases
    print("[random] convex mixtures", flush=True)
    # Prepare a few strong bases (some from the sweep if available)
    bases = []
    if best_vec is not None:
        bases.append(best_vec)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    bases.append(uniform)
    rev2 = reverse_power_law(n, 2.0)
    if rev2 is not None:
        bases.append(rev2)
    sig_mid = sigmoid_shape(n, 0.5, 0.07)
    if sig_mid is not None:
        bases.append(sig_mid)
    # Mixture loop
    while time.time() < deadline:
        coeffs = np.random.rand(len(bases))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, bases))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a
        # break condition to avoid infinite loop if time left is tiny
        if np.random.rand() < 1e-4:
            break

    return best_vec, best_val


# ----------------------------------------------------------------------
# 3. Permutation refinement (deterministic + random shuffles)
# ----------------------------------------------------------------------
def interleaved_permutation(a: np.ndarray) -> np.ndarray:
    """
    Deterministic high‑low interleaving:
    sort descending and place the largest, then the second largest at the far end,
    then the third largest next to the leftmost, etc.
    """
    sorted_vals = np.sort(a)[::-1]           # descending
    n = len(a)
    result = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for v in sorted_vals:
        if toggle:
            result[left] = v
            left += 1
        else:
            result[right] = v
            right -= 1
        toggle = not toggle
    return result


def permutation_search(a_base: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Try a few deterministic permutations and many random shuffles.
    Keeps the best vector discovered.
    """
    best_vec = a_base.copy()
    # deterministic interleaved ordering
    cand = interleaved_permutation(best_vec)
    val = compute_score(cand)
    if val < best_val:
        best_val, best_vec = val, cand

    # another deterministic variant: reverse then interleave
    cand2 = interleaved_permutation(best_vec[::-1])
    val2 = compute_score(cand2)
    if val2 < best_val:
        best_val, best_vec = val2, cand2

    # random shuffles – limited by remaining time
    print("[perm] random shuffles", flush=True)
    while time.time() < deadline:
        perm = np.random.permutation(best_vec)
        val = compute_score(perm)
        if val < best_val:
            best_val, best_vec = val, perm
        # avoid spending the last second on a permutation
        if time.time() > deadline - 0.5:
            break

    return best_vec, best_val

# 4. Projected sub‑gradient polishing (fast step‑size schedule)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a:
        g = np.zeros_like(a)
        for k in active:
            # indices that contribute to conv[k]
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# 5. Simulated‑annealing hill‑climb (mass transfers + swaps)
# ----------------------------------------------------------------------
def simulated_annealing_hill_climb(a_start: np.ndarray, deadline: float,
                                   max_iter: int = 2_000_000) -> np.ndarray:
    """
    Hill‑climb that mixes tiny mass transfers and full swaps,
    accepting uphill moves according to a temperature schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    t = 0

    # initial temperature ~ 1 % of current score
    T0 = max(best_score * 0.01, 1e-12)

    while time.time() < deadline and t < max_iter:
        t += 1

        if np.random.rand() < 0.15:                     # swap move
            i, j = np.random.randint(n, size=2)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                                           # small mass transfer
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = 0.02 * a[i]                         # move 2 % of source mass
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.clip(cand, 0.0, None)

        cand = project_to_simplex(cand)
        score = compute_score(cand)

        if score < best_score:
            best_score, best_a = score, cand.copy()
            a = cand
            # warm up: if we found improvement, slightly increase step‑size
            continue
        else:
            # temperature decays roughly linearly in log‑time
            T = T0 * math.exp(-t / (max_iter / 10))
            prob = math.exp(-(score - best_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = cand

        # occasional jitter to escape plateaus
        if t % 50000 == 0:
            jitter = np.random.randn(n) * 8e-4
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = project_to_simplex(a)

    return best_a

        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing_hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()
        print(f"[phase‑5] final best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # (6) Safeguard – use a simple analytic candidate if we failed completely
    # -----------------------------------------------------------------

    # Final report (official evaluation)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")
    return best_seq
