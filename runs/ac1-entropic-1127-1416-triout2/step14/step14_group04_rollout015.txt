def golden_section_optimize(func, a, b, tol=1e-7, max_iter=100):
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate 0.618
    c = b - phi*(b - a)
    d = a + phi*(b - a)
    fc = func(c)
    fd = func(d)
    while (b - a) > tol and iter < max_iter and time < deadline:
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi*(b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi*(b - a)
            fd = func(d)
    return (a+b)/2, best_val

"""
Propose a low‑value step‑function using a multi‑stage search.

The pipeline (within the 1000 s wall‑clock budget) is:

1. **Broad random baseline** – Dirichlet draws, monotone vectors,
   coarse beta/power‑law sweeps.  Keeps a short list of promising candidates.
2. **Fine 1‑D optimisation** – golden‑section search for the best
   exponential decay λ and the best power‑law exponent p.
3. **Mixture families** – random search over mixtures
   (exp + exp, exp + uniform).  These low‑dimensional families can
   express shapes that improve the max‑convolution bound beyond the pure
   families.
4. **Smoothed Adam optimisation** – optimise the soft‑max of the
   convolution (temperature annealed to ~0) starting from the best
   vector found so far.  Uses the same projected‑softmax parametrisation
   as the reference code but with a tighter learning‑rate schedule.
5. **Final polishing** – lightweight simulated‑annealing hill‑climb.

All candidates are normalised to sum ≈ 1 before scoring; the official
score is obtained with ``evaluate_sequence`` at the end.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv(a,a)). ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random vector, sort it decreasingly, and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]               # decreasing
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_mix_exponential(n: int, lam1: float, lam2: float, w: float) -> np.ndarray:
    """Mixture w·exp(-lam1·i) + (1‑w)·exp(-lam2·i), then normalised."""
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_mix_exp_uniform(n: int, lam: float, w: float) -> np.ndarray:
    """Mixture w·uniform + (1‑w)·exp(-lam·i), then normalised."""
    i = np.arange(n, dtype=float)
    a = w * np.ones(n) + (1.0 - w) * np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Helper to keep the best candidates (small list)
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best):
    """Update list ``best`` (sorted by score, smallest first) with ``vec``."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        return
    vec = vec / s
    sc = compute_score(vec)
    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Golden‑section helper (scalar optimisation)
# ----------------------------------------------------------------------


def golden_section_optimize(func, a, b, deadline, tol=1e-8, max_iter=80):
    """Minimise ``func`` on [a,b] (convex‑like) using golden‑section."""
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    it = 0
    best_x = c if fc < fd else d
    best_val = min(fc, fd)
    while (b - a) > tol and it < max_iter and time.time() < deadline:
        it += 1
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - phi * (b - a)
            fc = func(c)
            if fc < best_val:
                best_val, best_x = fc, c
        else:
            a, c, fc = c, d, fd
            d = a + phi * (b - a)
            fd = func(d)
            if fd < best_val:
                best_val, best_x = fd, d
    return best_x, best_val

def baseline_search(n: int, deadline: float):
    best = []   # (score, vector), smallest first

    # Dirichlet draws with a few concentration levels
    alphas = [0.1, 0.3, 0.7, 1.0, 2.0, 4.0]
    draws_per_alpha = 1500
    for conc in alphas:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    # Random monotone vectors
    for _ in range(3000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # Coarse beta sweep
    al_vals = np.linspace(0.2, 8.0, 10)
    be_vals = np.linspace(0.2, 8.0, 10)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best)

    # Power‑law sweep (coarse)
    p_vals = np.linspace(0.05, 4.0, 30)
    for p in p_vals:
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, p)
        _try_candidate(a, best)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]

# Fine 1‑D sweeps using golden‑section

def fine_exponential_search(n: int, deadline: float):
    """Golden‑section search for the exponential decay λ."""
    # Wide interval based on earlier experiments
    lo, hi = 5e-4, 2.5e-3

    def f(lam):
        a = generate_exponential(n, lam)
        return compute_score(a)

    lam_opt, sc_opt = golden_section_optimize(f, lo, hi, deadline)
    a_opt = generate_exponential(n, lam_opt)
    return sc_opt, a_opt


def fine_powerlaw_search(n: int, deadline: float):
    """Golden‑section search on the power‑law exponent p."""
    lo, hi = 0.05, 4.0

    def f(p):
        a = generate_powerlaw(n, p)
        return compute_score(a)

    p_opt, sc_opt = golden_section_optimize(f, lo, hi, deadline)
    a_opt = generate_powerlaw(n, p_opt)
    return sc_opt, a_opt

def mixture_search(n: int, deadline: float):
    """Random search over two mixture families, returning a list of vectors."""
    best = []   # (score, vector)

    i = 0
    while time.time() < deadline and i < 6000:
        i += 1
        # 1) mix of two exponentials
        lam1 = np.random.uniform(5e-4, 1.5e-3)
        lam2 = np.random.uniform(lam1, 4e-3)   # lam2 ≥ lam1
        w = np.random.rand()
        a = generate_mix_exponential(n, lam1, lam2, w)
        _try_candidate(a, best)

        # 2) mix of exponential and uniform
        lam = np.random.uniform(5e-4, 2e-3)
        w2 = np.random.rand()
        a2 = generate_mix_exp_uniform(n, lam, w2)
        _try_candidate(a2, best)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.06,
                    init_tau_factor: float = 4.0,
                    min_tau: float = 1e-11,
                    tau_decay: float = 0.9998) -> np.ndarray:
    """
    Adam on soft‑max formulation (a = softmax(x)).
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits safely
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise temperature
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution
        conv = full_convolution(a)
        max_c = conv.max()

        # soft‑max weighting of the max
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t a
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3 n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for logits
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # occasional exact check
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            # tiny jitter to avoid stagnation
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a
