A more aggressive optimiser for ``evaluate_sequence``.
Key ideas
---------

  simplex.

# ----------------------------------------------------------------------
# Helper utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    # length of FFT – next power of two ≥ m
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law), p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def triangular(n: int, support_frac: float) -> np.ndarray:
    """
    Linearly decreasing shape that is zero after ``support_frac`` of the length.
    support_frac ∈ (0,1].
    """
    if not (0.0 < support_frac <= 1.0):
        return None
    m = int(np.ceil(n * support_frac))
    if m == 0:
        return None
    # values: m, m-1, … ,1
    values = np.arange(m, 0, -1, dtype=np.float64)
    a = np.concatenate([values, np.zeros(n - m, dtype=np.float64)])
    a /= a.sum()
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    Convex combination   mix * reverse_power_law(p) + (1‑mix) * uniform,
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

def adam_optimize(start_vec: np.ndarray,
                  time_limit: float,
                  base_lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the objective M(a)=max_k (a*a)[k].
    Returns the best vector seen and its corresponding max‑convolution value.
    """
    a = start_vec.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    n = a.shape[0]

    while time.time() < time_limit:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # Adam update (on the sub‑gradient)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # keep the best visited point
        if M < best_M:
            best_M = M
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (base_lr * 0.002 / math.sqrt(t))
            a = best_a + jitter
            a = project_to_simplex(a)

    return best_a, best_M


# ----------------------------------------------------------------------
# Stochastic hill‑climb (fine‑grained local search)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_no_improve: int = 250_000) -> np.ndarray:
    """
    Greedy stochastic moves with adaptive step‑size and restarts.
    Works on the full vector (no hidden parameters) and returns the
    best vector found before the deadline.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of the source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            # restart from the best found so far with a fresh jitter
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point required by the harness

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0      # safety margin for the harness

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0.  Optional fallback (if a previous construction exists)
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1.  Choose vector length
    # -------------------------------------------------------------
    # Use the longer vector if we have sufficient time left.
    remaining = deadline - time.time()
    n = 4096 if remaining > 250 else 2048
    print(f"Working with n = {n}")

    # -------------------------------------------------------------
    # 2.  Broad parametric sweep (≈ 200 s)
    # -------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    a_initial = None
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep_extended(n, sweep_deadline)
        if a_initial is not None:
            val = evaluate_sequence(a_initial.tolist())
            print(f"[parametric sweep] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # Use a uniform vector as a safe fallback for the optimiser.
    if a_initial is None:
        a_initial = np.full(n, 1.0 / n, dtype=np.float64)

    # -------------------------------------------------------------
    # 3.  Adam‑style optimisation with several random restarts
    # -------------------------------------------------------------
    opt_deadline = deadline - 120.0   # leave ~2 min for polishing & hill‑climb
    best_a = np.array(best_seq, dtype=float) if best_seq is not None else a_initial.copy()
    best_M = full_convolution(best_a).max()
    print(f"[adam] initial max‑conv = {best_M:.8f}")

    # basic shapes that we will combine to generate diverse starts
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rp_base = reverse_power_law(n, 1.5)
    geom_base = geometric(n, 0.97)

    while time.time() < opt_deadline:
        # construct a fresh start vector
        if np.random.rand() < 0.35:
            start = best_a + np.random.randn(n) * 0.001   # jitter around current best
            start = project_to_simplex(start)
        else:
            # random convex mix of a few elementary shapes
            weights = np.random.rand(3)
            weights /= weights.sum()
            start = (weights[0] * uniform +
                     weights[1] * rp_base +
                     weights[2] * geom_base)
            start = project_to_simplex(start)

        slice_time = min(35.0, opt_deadline - time.time())
        if slice_time < 5.0:
            break

        a_opt, M_opt = adam_optimize(start, time.time() + slice_time,
                                     base_lr=0.07)
        if M_opt < best_M:
            best_M, best_a = M_opt, a_opt.copy()
            val = evaluate_sequence(best_a.tolist())
            print(f"[adam] improved max‑conv = {best_M:.8f}  official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, best_a.tolist()

    # -------------------------------------------------------------
    # 4.  Final stochastic hill‑climb (fine tuning)
    # -------------------------------------------------------------
    hill_deadline = deadline - 4.0
    if time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float) if best_seq is not None else best_a
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # 5.  Safeguard – simple mixed shape if nothing was found
    # -------------------------------------------------------------
    if best_seq is None:
        fallback_vec = mixed_reverse_uniform(n, mix=0.85, p=2.0)
        best_seq = fallback_vec.tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # 6.  Final report
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
