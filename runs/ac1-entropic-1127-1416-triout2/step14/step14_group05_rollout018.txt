"""Minimum‑evaluation step‑function optimiser.

The optimiser works in three stages:
  1. Fast random Dirichlet seeding.
  2. A long Adam run on the smooth‑max surrogate of the max‑convolution.
  3. An aggressive mass‑transfer hill‑climb that focusses on the current
     max‑convolution index and repeatedly moves a tiny amount of probability
     from “heavy’’ positions to “light’’ ones.

All vectors are kept on the probability simplex (∑a = 1, a_i ≥ 0) so the
final call to ``evaluate_sequence`` needs no additional scaling.  The code
is time‑aware: each phase stops before the global deadline and the current
best vector is always retained.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v = 1, v ≥ 0)."""
    if np.all(v <= 0):
        # return the centre of the simplex if v is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used during the search (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 2.0,
    tau_end: float = 1e-8,
    tau_decay: float = 0.99,
    lr_start: float = 0.12,
    lr_decay: float = 0.9995,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """Adam optimiser on   f_τ(a) = τ·log Σ exp(conv/τ)   with a simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # ---- temperature schedule ------------------------------------------------
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # ---- smooth‑max computation ---------------------------------------------
        conv = full_convolution(a)                  # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)     # stable formulation
        w = exp_vals / exp_vals.sum()               # soft‑max weights (shape m, sum=1)

        # ---- gradient of the smooth‑max (2·a⋆w) ----------------------------------
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]

        # indices that map the convolution back onto the original vector
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # ---- Adam update ---------------------------------------------------------
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # ---- bookkeeping ---------------------------------------------------------
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter={t}, τ={tau:.1e}, lr={lr:.3e}, score={cur_val:.6f}")

        # learning‑rate decay
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-7)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.005 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def aggressive_mass_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Repeatedly move a tiny fraction of probability from heavy to light sites.
    The move direction is guided by the current max‑convolution index.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    # pre‑compute a list of indices; will be shuffled lazily
    while time.time() < deadline:
        # recompute the convolution only when necessary
        conv = full_convolution(a)
        k_max = int(conv.argmax())
        # the gradient of the max‑convolution term is 2·a_rev shifted by k_max
        rev_a = a[::-1]
        offset = n - 1 - k_max
        # indices that actually influence the max entry
        if offset >= 0:
            src_candidates = np.arange(max(0, -offset), min(n, n - offset))
        else:
            src_candidates = np.arange(max(0, offset), min(n, n + offset))

        # pick a heavy source among the candidates
        # (randomly sample a few and keep the heaviest)
        if src_candidates.size == 0:
            # fallback to any index if the above is empty
            i = np.random.randint(n)
        else:
            sample = np.random.choice(src_candidates,
                                      size=min(5, src_candidates.size),
                                      replace=False)
            i = sample[np.argmax(a[sample])]

        if a[i] < 1e-12:
            # nothing to move – pick a random heavy index instead
            i = np.random.choice(np.where(a > 1e-12)[0])

        # pick a light destination (uniformly among the smallest 10% entries)
        light_thresh = np.percentile(a, 10)
        light_idxs = np.where(a <= light_thresh)[0]
        if light_idxs.size == 0:
            j = np.random.randint(n)
        else:
            j = np.random.choice(light_idxs)

        if i == j:
            continue

        # transfer a small fraction (adaptive)
        frac = np.random.rand() * 0.08  # ≤ 8 % of the source mass
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:
                print(f"[mass] iter={it}, score={val:.6f}")

        it += 1

    return best_a

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 5 s safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Initialise – try to use the fallback if it exists, otherwise uniform.
    # ------------------------------------------------------------------
    fallback = None
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
    except Exception:
        pass

    if fallback is not None:
        a0 = np.array(fallback, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 = a0 / a0.sum()
        n = a0.shape[0]
        print(f"[init] using fallback of length {n}")
    else:
        n = 2048
        a0 = np.full(n, 1.0 / n)        # uniform start
        print("[init] no fallback – start with uniform vector")

    best_a = a0.copy()
    best_val = compute_score(best_a)
    print(f"[init] initial score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Quick random Dirichlet seeds – cheap way to get a diverse pool.
    # ------------------------------------------------------------------
    for _ in range(15):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[dirichlet] improved seed score = {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Main Adam optimisation (smooth‑max surrogate). Allocate ~50 % of time.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_time = remaining * 0.48
    if adam_time > 2.0:
        local_deadline = time.time() + adam_time
        print(f"[adam] running for roughly {adam_time:.1f}s")
        a_ref = smooth_max_adam(best_a, local_deadline)
        val = compute_score(a_ref)
        print(f"[adam] finished, score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # ------------------------------------------------------------------
    # 3. Aggressive mass‑transfer hill‑climb for the rest of the time.
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("[mass] starting aggressive mass‑transfer hill‑climb")
        a_ref = aggressive_mass_hill_climb(best_a, deadline)
        val = compute_score(a_ref)
        print(f"[mass] final hill‑climb score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # ------------------------------------------------------------------
    # 4. Final reporting (using the official evaluation function).
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq
