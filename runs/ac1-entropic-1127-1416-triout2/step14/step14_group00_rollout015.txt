"""
Propose a step‑function that minimises ``evaluate_sequence``.
--------------------------------------------------------------------
Approach
--------

1. Choose a moderate length (4096).  The objective is scale‑invariant,
   so the length only influences the discretisation error.

2. Parameterise a symmetric Beta‑type family:
        a_i ∝ ((i+δ)*(n‑δ‑i))^{-p}      (δ∈(0,1), p≈½)
   The continuous arcsine law corresponds to (δ,p)=(0.5,0.5).
   By sweeping (δ,p) on a fine grid we can compensate the
   discretisation artefacts and obtain a shape with a lower
   self‑convolution peak than the plain arcsine distribution.

3. For the best continuous‑family candidate also try a convex
   combination with the uniform vector – a cheap line‑search on λ.

4. Run a projected sub‑gradient descent with momentum on the exact
   objective 2·n·max(conv(a,a)) (the same value that
   ``evaluate_sequence`` returns for a normalised vector).  The
   sub‑gradient is the sum of the shifts of ``a`` that correspond
   to the active (peak) convolution indices.  A simple Adam‑style
   momentum is used and the iterate is projected onto the simplex
   after every step.

5. Finally a very short simulated‑annealing hill‑climb is performed
   as a last fine‑tuning stage.

All phases respect the 1000 s wall‑clock budget.  The best sequence
found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (same primitives as the reference solution)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) distribution."""
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


def generate_general_beta(n: int, delta: float, p: float) -> np.ndarray:
    """
    Symmetric Beta‑type shape:
        a_i ∝ ((i+δ)*(n‑δ‑i))^{-p},   δ∈(0,1), p>0.

    The classical arcsine law is (δ,p) = (0.5, 0.5).
    """
    i = np.arange(n, dtype=float)
    # the two factors are always positive for 0<δ<1
    base = (i + delta) * (n - delta - i)
    # avoid division by zero – clamp the base slightly above zero
    base = np.clip(base, 1e-12, None)
    a = base ** (-p)
    total = a.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Standard Beta(α,β) density on {1,…,n} (discrete analogue)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Sub‑gradient descent with momentum (projected Adam‑style)
# ----------------------------------------------------------------------


def subgradient_polish_momentum(a0: np.ndarray,
                                deadline: float,
                                step0: float = 0.25,
                                beta1: float = 0.9,
                                beta2: float = 0.999,
                                eps: float = 1e-8) -> np.ndarray:
    """
    Projected Adam‑style sub‑gradient descent on the true objective
    2·n·max(conv(a,a)).  The iterate is kept on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # identify the active set (indices within numerical tolerance)
        active = np.where(conv >= max_val - max_val * 1e-12)[0]

        # sub‑gradient of max(conv) = sum of gradients of active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # Adam update on the sub‑gradient
        step = step0 / np.sqrt(t)                 # simple decay
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional tiny jitter to escape plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb (as a final polishing step)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

    alphas = np.linspace(0.30, 0.80, 11)

    uniform = np.full(n, 1.0 / n)
