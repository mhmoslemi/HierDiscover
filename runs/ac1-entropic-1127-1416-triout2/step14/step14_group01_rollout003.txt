The goal is to find a non‑negative vector `a` (the step heights) that minimises  

```
score(a) = 2·n·max(conv(a,a)) / (sum(a)²)
```

where `conv` is the ordinary linear convolution.  
The score is invariant to scaling, therefore we keep the vector on the probability
simplex (`sum(a)=1`).  The problem is non‑convex because of the *max* operation,
but a sub‑gradient can be computed analytically:

```
conv_k = Σ_i a_i·a_{k-i}
∂max(conv)/∂a_i = Σ_{k∈Active} 2·a_{k-i}
```

`Active` is the (usually tiny) set of indices where the convolution attains its
maximum.

The optimiser consists of three stages that run inside the 1000 s time budget:

1. **Warm‑start** – use the previously known construction `height_sequence_1`
   (if present) or a uniform distribution.
2. **Exponential‑gradient descent** – a mirror‑descent step on the simplex
   (`a_i ← a_i·exp(‑η·g_i)`).  This keeps non‑negativity automatically and
   quickly drives the max‑convolution down.
3. **Adam‑style sub‑gradient polishing** – a classic Adam optimiser applied to
   the sub‑gradient, followed by a simple `max(0,·)` clipping and renormalisation
   onto the simplex.  This fine‑tunes the solution and removes remaining spikes.

The algorithm repeatedly restarts from random Dirichlet vectors (or a jittered
copy of the current best) until the time limit is reached, always keeping the
best sequence seen (as measured by the provided `evaluate_sequence`).  All heavy
operations (convolution and its maximum) are performed with a real FFT, which
is `O(n log n)` and fast enough for thousands of iterations on `n=2048`.

The code follows the required API, prints progress information and finally
returns the best list of floats found.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Propose a step‑height sequence that minimises the evaluation function.

Algorithm
---------
* Work on the probability simplex (sum = 1) – the objective is scale‑invariant.
* Warm‑start from the known construction ``height_sequence_1`` or uniform.
* Repeatedly run a two‑phase local optimisation:
    1. Exponential‑gradient (mirror descent) on the simplex.
    2. Adam‑style sub‑gradient descent with simple clipping & renormalisation.
* Several random restarts (Dirichlet‑distributed starts and jittered copies of
  the current best) are performed until the allotted time expires.
* The best sequence found is finally evaluated with the supplied
  ``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_full_convolution_fft(a).max())


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_convolution(a)


def _subgradient_maxconv(a: np.ndarray):
    """
    Return (g, max_val) where g is a sub‑gradient of max(conv(a,a))
    and max_val = max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    max_val = conv.max()
    # consider all indices that are (numerically) maximal
    eps = max_val * 1e-9
    active = np.where(conv >= max_val - eps)[0]

    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        # contribution 2·a_{k-i}
        g[mask] += 2.0 * a[j[mask]]
    return g, max_val


def _renorm_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform
        n = v.shape[0]
        return np.full_like(v, 1.0 / n)
    return v / s


# ----------------------------------------------------------------------
# Phase 1 – Exponential (mirror) gradient on the simplex
# ----------------------------------------------------------------------


def _exp_gradient_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Mirror‑descent on the simplex using an exponential update.
    a_i ← a_i·exp(‑η·g_i) , η decays with the iteration count.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        g, _ = _subgradient_maxconv(a)

        # learning rate schedule (fast decay → stable later)
        eta = 0.5 / (1.0 + t * 0.01)

        # exponential update (log‑domain to avoid overflow)
        log_a = np.log(a + 1e-30)
        log_a -= eta * g
        a = np.exp(log_a)

        # renormalise onto the simplex
        a = a / a.sum()

        # occasional tiny jitter to escape flat regions
        if t % 500 == 0:
            a += np.random.randn(n) * 1e-4
            a = _renorm_simplex(a)

    return a


# ----------------------------------------------------------------------
# Phase 2 – Adam‑style sub‑gradient polishing
# ----------------------------------------------------------------------


def _adam_subgrad_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser applied to the sub‑gradient of max(conv).
    After each step we clip negatives and renorm to the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)        # first moment
    v = np.zeros_like(a)        # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.4               # initial learning‑rate factor
    t = 0

    while time.time() < deadline:
        t += 1
        g, _ = _subgradient_maxconv(a)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size decays like 1/√t
        lr = base_lr / np.sqrt(t)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasibility
        a = _renorm_simplex(a)

        # periodic jitter to avoid stagnation
        if t % 2000 == 0:
            a += np.random.randn(n) * (base_lr * 0.003 / np.sqrt(t))
            a = _renorm_simplex(a)

    return a


# ----------------------------------------------------------------------
# Full optimisation routine for a single start vector
# ----------------------------------------------------------------------


def _optimise_start(start_a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Run the two‑phase optimisation (exp‑gradient → Adam polishing) on a
    probability vector start_a, respecting the global deadline.
    """
    now = time.time()
    remaining = deadline - now
    # split remaining time roughly 40 % / 60 %
    phase1_deadline = now + 0.38 * remaining
    phase2_deadline = deadline

    a = _exp_gradient_opt(start_a, phase1_deadline)
    a = _adam_subgrad_opt(a, phase2_deadline)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑height sequence that minimises
    ``evaluate_sequence``.  Returns a plain list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Settings
    # ------------------------------------------------------------------
    n = 2048                                     # length of the vector
    best_seq: list[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback – known construction if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore[name-defined]
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass  # no fallback available

    # ------------------------------------------------------------------
    # 1. Initialise a uniform vector (used when we have no better start)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=float)

    # ------------------------------------------------------------------
    # 2. Main search loop – many random restarts limited by the deadline
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 0.5:            # tiny slack for the last eval
        restart_counter += 1

        # Choose a start point:
        if best_seq and random.random() < 0.4:
            # jitter a copy of the current best
            a_start = np.array(best_seq, dtype=float)
            a_start += np.random.randn(n) * 0.01
        else:
            # fresh Dirichlet‑like draw (shape < 1 → sparser vectors)
            a_start = np.random.gamma(0.5, 1.0, size=n)
        a_start = _renorm_simplex(a_start)

        # Run the two‑phase optimiser on this start
        a_opt = _optimise_start(a_start, deadline)

        # Evaluate with the official scorer (ensures clipping etc.)
        candidate = a_opt.tolist()
        try:
            val = evaluate_sequence(candidate)
        except Exception:
            # Should never happen – fallback to a safe uniform vector
            candidate = uniform.tolist()
            val = evaluate_sequence(candidate)

        if val < best_score:
            best_score, best_seq = val, candidate
            print(f"[restart {restart_counter:03d}] new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Safeguard – if nothing succeeded, fall back to uniform
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform.tolist()
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 4. Final report
    # ------------------------------------------------------------------
    final = evaluate_sequence(best_seq)
    print(f"\nSearch finished after {restart_counter} restarts.")
    print(f"Best reported score = {final:.6f}")
    return best_seq
```
