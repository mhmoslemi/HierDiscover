"""
Propose a step‑function that minimises ``evaluate_sequence``.
----------------------------------------------------------------
1. Generate a discrete analogue of the continuous arcsine law
   (Beta(½,½)) **by integrating its CDF** – this is more accurate than
   the point‑wise 1/√(i·(n‑i)) formula used previously.
2. Scan a handful of lengths (1024 … 16384) and also scan a few
   mixtures with the uniform distribution; keep the best (lowest
   ``2·n·max(conv)``).
3. Refine the best candidate with a projected sub‑gradient descent on
   the true objective (the active entries of the convolution are used
   as a sub‑gradient).  The optimiser runs for the remaining time.
4. Finish with a brief simulated‑annealing hill‑climb to escape shallow
   minima.
5. Return the final vector as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score for a simplex‑normalised vector: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------


def arcsine_integrated(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) obtained by integrating its CDF."""
    # CDF of arcsine on [0,1]:  (2/π)·arcsin(√x)
    x = np.linspace(0.0, 1.0, n + 1)
    cdf = (2.0 / np.pi) * np.arcsin(np.sqrt(x))
    a = np.diff(cdf)                     # mass in each bin
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+1)^{α−1}·(n−i)^{β−1} (α,β>0)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective
    f(a)=2·n·max(conv(a,a))  (a stays on the simplex).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-10          # tolerance for the active set
        active = np.where(conv >= max_val - eps)[0]

        # gradient of the max‑convolution (sub‑gradient)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """
    Simple Metropolis hill‑climb that can accept worsening moves.
    Works on the simplex – the proposal swaps a small amount of mass
    between two random coordinates.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # periodic jitter
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # 0. Optional fallback from the previous run (if available)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Scan several problem sizes and simple families
    # ------------------------------------------------------------------
    length_candidates = [1024, 2048, 4096, 8192, 16384]
    for n in length_candidates:
        # (a) integrated arcsine
        a_arc = arcsine_integrated(n)
        sc = compute_score(a_arc)
        print(f"[n={n}] arcsine score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_arc.copy()

        # (b) mix arcsine with uniform – simple line‑search
        uniform = uniform_distribution(n)
        for lam in np.linspace(0.0, 0.25, 6):   # lam=0 → arcsine, lam>0 → uniform weight
            mixed = lam * uniform + (1.0 - lam) * a_arc
            mixed /= mixed.sum()
            sc = compute_score(mixed)
            if sc < best_score:
                best_score, best_seq = sc, mixed.copy()

        # (c) a small grid of Beta shapes around (0.5,0.5)
        for alpha in np.linspace(0.30, 0.80, 6):
            for beta in np.linspace(0.30, 0.80, 6):
                a_beta = generate_beta(n, alpha, beta)
                sc = compute_score(a_beta)
                if sc < best_score:
                    best_score, best_seq = sc, a_beta.copy()

    print(f"[grid] best score after naive scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Sub‑gradient polishing (major part of the remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polish_deadline = deadline - 2.5   # reserve a little time for anneal & final eval
        print("[polish] start")
        a_polished = subgradient_polish(np.array(best_seq, dtype=float), polish_deadline)
        sc = compute_score(a_polished)
        print(f"[polish] refined score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_polished.copy()

    # ------------------------------------------------------------------
    # 3. Final simulated‑annealing fine‑tuning (short run)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[anneal] start")
        a_final = simulated_annealing(np.array(best_seq, dtype=float), deadline)
        sc = compute_score(a_final)
        print(f"[anneal] final score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_final.copy()

    # ------------------------------------------------------------------
    # 4. Safeguard – if nothing succeeded return a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2048
        best_seq = uniform_distribution(n)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 5. Official evaluation (clamping etc. inside evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq.tolist()
