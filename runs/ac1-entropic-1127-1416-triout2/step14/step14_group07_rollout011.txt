"""
search_step_function_improved.py

Implements a multi‑phase optimiser for ``evaluate_sequence`` that
* scans several analytic families,
* refines the best candidate with a smooth approximation of the
  ``max`` objective (log‑sum‑exp) using projected gradient descent on the
  simplex,
* applies a classic sub‑gradient polish,
* finishes with a long stochastic hill‑climb.
The whole pipeline respects the 1000 s budget and returns a list of
non‑negative floats.
"""

import time
import math
import numpy as np

# ----- Linear algebra utilities ------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep ∑a = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----- Analytic families -------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n−i)^p, p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.
    Both p,q ≥ 0 give a front‑loaded shape that can be tuned by the
    exponent on the left‑hand side (p) and a gentle lift on the tail (q).
    """
    left = np.arange(n, 0, -1, dtype=np.float64) ** p      # n‑i
    right = np.arange(1, n + 1, dtype=np.float64) ** q    # i+1
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """
    a_i ∝ 1/(1 + exp((i - c) / σ)).
    ``centre_frac`` is the location of the mid‑point as a fraction of n,
    ``sigma_frac`` controls the steepness (σ = sigma_frac * n).
    """
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^α (1‑x_i)^β,  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


# ----- Parametric sweep --------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Explore several families and keep the one with the smallest proxy score."""
    best_vec = None
    best_val = float('inf')

    # 1. Reverse power‑law (coarse)
    for p in np.linspace(0.1, 8.0, 79):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # Fine sweep around best p (if any)
    if best_vec is not None:
        # Rough estimate of best p: just sweep finer grid across whole range
        for p in np.linspace(0.05, 8.0, 200):
            if time.time() >= deadline:
                return best_vec
            a = reverse_power_law(n, p)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 2. Multi‑power
    for p in np.linspace(0.2, 4.0, 20):
        for q in np.linspace(0.0, 2.0, 11):
            if time.time() >= deadline:
                return best_vec
            a = multi_power(n, p, q)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 3. Sigmoid family
    for centre_frac in np.linspace(0.2, 0.8, 13):
        for sigma_frac in np.linspace(0.01, 0.15, 15):
            if time.time() >= deadline:
                return best_vec
            a = sigmoid_shape(n, centre_frac, sigma_frac)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 4. Beta shapes – focus on the region around α≈β≈‑0.5 (U‑shaped)
    for alpha in np.linspace(-0.8, -0.2, 31):
        for beta in np.linspace(-0.8, -0.2, 31):
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 5. Random Dirichlet samples (cheap exploration)
    for _ in range(1500):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----- Smooth (log‑sum‑exp) gradient descent -----------------------------------
def smooth_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth approximation
    f_λ(a) = (1/λ) log Σ exp(λ·conv(a))[i].
    λ is increased slowly so that the final point approximates the true
    max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    lam = 8.0                 # start with a modest smoothing
    max_lam = 2500.0
    step0 = 0.6
    no_improve = 0

    while time.time() < deadline:
        it += 1
        # compute convex combination weights
        conv = full_convolution(a)                # length 2n-1
        # numerically stable exponentials
        scaled = lam * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # gradient = 2 * cross‑correlation(w, a)
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)          # length = len(w)+len(a_rev)-1 = 3n-2
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # simple diminishing step size
        step = step0 / math.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        # occasional jitter to break plateaus
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(it))
            a_new += jitter
            a_new = project_to_simplex(a_new)

        val = compute_score(a_new)
        if val < best_val:
            best_val, best_a = val, a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase λ gradually (every few thousand iterations)
        if it % 2000 == 0 and lam < max_lam:
            lam = min(lam * 1.4, max_lam)

        # if we have been stuck for a while, perturb around the best
        if no_improve > 120000:
            a_new = best_a + np.random.randn(n) * 0.0006
            a_new = project_to_simplex(a_new)
            no_improve = 0

        a = a_new

    return best_a


# ----- Classic sub‑gradient polishing -----------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a:
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----- Stochastic hill‑climb ----------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 2_500_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size and occasional restart."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]      # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        if no_improve > 250_000:
            # restart from current best with a small perturbation
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

        it += 1

    return best_a

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a)).  Scale‑invariant, we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for a quick bulk sweep)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n−i)^p,  p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p·(i+1)^q."""
    left = np.arange(n, 0, -1, dtype=np.float64) ** p      # n‑i
    right = np.arange(1, n + 1, dtype=np.float64) ** q    # i+1
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i−c)/σ))."""
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^α (1−x_i)^β,  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Broad parametric sweep – keeps the best proxy score
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    best_vec = None
    best_val = float('inf')

    # 1. Reverse power‑law (coarse)
    for p in np.linspace(0.1, 8.0, 79):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 2. Fine sweep across whole range (still cheap)
    if best_vec is not None:
        for p in np.linspace(0.05, 8.0, 200):
            if time.time() >= deadline:
                return best_vec
            a = reverse_power_law(n, p)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 3. Multi‑power
    for p in np.linspace(0.2, 4.0, 20):
        for q in np.linspace(0.0, 2.0, 11):
            if time.time() >= deadline:
                return best_vec
            a = multi_power(n, p, q)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 4. Sigmoid family
    for centre_frac in np.linspace(0.2, 0.8, 13):
        for sigma_frac in np.linspace(0.01, 0.15, 15):
            if time.time() >= deadline:
                return best_vec
            a = sigmoid_shape(n, centre_frac, sigma_frac)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 5. Beta shapes – focus on the U‑shaped region around α,β≈‑0.5
    for alpha in np.linspace(-0.8, -0.2, 31):
        for beta in np.linspace(-0.8, -0.2, 31):
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 6. Random Dirichlet samples (cheap high‑dim exploration)
    for _ in range(1500):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) gradient descent on max‑convolution
# ----------------------------------------------------------------------
def smooth_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth surrogate
        f_λ(a) = (1/λ) log Σ_i exp(λ·conv(a))[i].
    λ grows gradually, turning the surrogate into the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    lam = 8.0                # start with moderate smoothing
    max_lam = 2500.0
    step0 = 0.6
    no_improve = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)                       # length 2n‑1
        scaled = lam * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient = 2 * cross‑correlation(w, a)
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)               # length = 3n‑2
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        step = step0 / math.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        # occasional jitter to escape flat regions
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(it))
            a_new += jitter
            a_new = project_to_simplex(a_new)

        val = compute_score(a_new)
        if val < best_val:
            best_val, best_a = val, a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase λ slowly (every few thousand iterations)
        if it % 2000 == 0 and lam < max_lam:
            lam = min(lam * 1.4, max_lam)

        # if stuck for a long time, perturb around the current best
        if no_improve > 120_000:
            a_new = best_a + np.random.randn(n) * 0.0006
            a_new = project_to_simplex(a_new)
            no_improve = 0

        a = a_new

    return best_a


# ----------------------------------------------------------------------
# Classic sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 2_500_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size and restart on stagnation."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]            # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        if no_improve > 250_000:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

        it += 1

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # (0) Optional fallback from a pre‑computed global variable
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # (1) Broad parametric sweep – bulk of the budget
    # -------------------------------------------------------------
    sweep_deadline = deadline - 500.0
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            proxy = compute_score(a_initial)
            real = evaluate_sequence(a_initial.tolist())
            print(f"[sweep] proxy = {proxy:.6f}, official = {real:.6f}")
            if real < best_val:
                best_val, best_seq = real, a_initial.tolist()

    # -------------------------------------------------------------
    # (2) Smooth (log‑sum‑exp) gradient descent
    # -------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if best_seq is not None and time.time() < smooth_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_smooth = smooth_gradient_descent(a0, smooth_deadline)
        val = evaluate_sequence(a_smooth.tolist())
        print(f"[smooth] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_smooth.tolist()

    # -------------------------------------------------------------
    # (3) Sub‑gradient polishing
    # -------------------------------------------------------------
    polish_deadline = deadline - 100.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.12)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # (4) Final stochastic hill‑climb
    # -------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # (5) Safety fallback – simple linear decreasing shape
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.0)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] using linear shape, score = {best_val:.6f}")

    # -------------------------------------------------------------
    # Final report (using the official ``evaluate_sequence``)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")
    return best_seq
