"""
Non‑negative step‑function optimiser – enhanced version.

The optimiser proceeds in three stages:
  1. A long Adam optimisation on a smooth‑max surrogate of the
     true objective (this yields a good starting point, typically ≈1.51
     for the benchmark length 2048).
  2. Simulated‑annealing that performs random mass‑transfer and swap
     moves, accepting worsening moves with a temperature schedule.
     This stage is the main driver for pushing the score below the
     previous best (≈1.503).
  3. A short greedy hill‑climb (permutation + small mass transfers)
     for final polishing.

All vectors are kept on the probability simplex (∑a = 1, a_i ≥ 0).  The
code is deliberately time‑aware: each phase stops before the global
deadline and the best solution seen so far is always retained.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1).
    Returns 2·n·max(conv(a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 2.0,
    tau_end: float = 1e-8,
    tau_decay: float = 0.995,
    lr_start: float = 0.07,
    lr_decay: float = 0.999,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate
        f_τ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    The function returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # temperature schedule (geometric)
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # full convolution
        conv = full_convolution(a)               # length m
        shift = conv.max()
        # soft‑max weights (stable formulation)
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # shape (m,)

        # gradient of smooth‑max: 2·(a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # bookkeeping
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter={t}, τ={tau:.1e}, lr={lr:.3e}, score={cur_val:.6f}")

        # learning‑rate decay
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-6)

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing on the exact objective
# ----------------------------------------------------------------------


def simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing improving the exact objective.
    Moves are either a mass transfer between two positions or a swap.
    A simple exponential temperature schedule is used.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = compute_score(a)

    best_a = a.copy()
    best_score = cur_score

    # temperature schedule parameters
    T_start = 0.02
    T_end = 1e-7
    total_time = deadline - time.time()
    start_time = time.time()
    it = 0

    while time.time() < deadline:
        # exponentially decaying temperature based on elapsed fraction
        elapsed = time.time() - start_time
        frac = min(elapsed / total_time, 1.0)
        T = T_start * (T_end / T_start) ** frac

        # decide move type
        if np.random.rand() < 0.15:               # swap move
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # mass‑transfer move
            i = np.random.randint(n)
            if a[i] <= 1e-14:
                continue
            # pick a different destination
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            # transfer a random fraction of the source mass
            frac_move = np.random.rand() * 0.45   # up to 45 %
            delta = a[i] * frac_move
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta

        # the move preserves the sum (up to round‑off), but enforce positivity
        if np.any(cand < -1e-15):
            continue
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        # renormalise to stay on the simplex (tiny drift may appear)
        cand /= cand_sum

        cand_score = compute_score(cand)
        delta_score = cand_score - cur_score

        # accept if improvement or with Metropolis probability
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                if it % 2000 == 0:
                    print(f"[SA] iter={it}, T={T:.2e}, score={cand_score:.6f}")

        it += 1

    print(f"[SA] finished after {it} iterations, best score = {best_score:.6f}")
    return best_a

def permute_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Swap two entries at a time; keep the best permutation."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # perform swap
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[permute] iter={it}, score={val:.6f}")
        else:
            a[i], a[j] = a[j], a[i]      # revert
        it += 1
    return best_a


def mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Move a small fraction of mass from one index to another (greedy)."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:

        if a[i] <= 1e-14:

        j = np.random.randint(n)
        if i == j:

        cand[i] -= delta
        cand[j] += delta
        val = compute_score(cand)

            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:

    return best_a
