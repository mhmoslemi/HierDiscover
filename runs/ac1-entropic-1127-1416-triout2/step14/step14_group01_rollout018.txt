def gradient_refine(a0, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    step0 = 0.5 (maybe)
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # define tolerance as 1e-12 * max? earlier used 1e-9 relative. Keep relative.
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # Determine step size α
        alpha = step0 / np.sqrt(t)  # could also scale with some factor
        # Try backtracking: attempt up to 3 reductions
        improved = False
        for factor in [1.0, 0.5, 0.25]:
            a_candidate = a - alpha * factor * g
            a_candidate = project_to_simplex(a_candidate)
            val_cand = compute_score(a_candidate)
            if val_cand < best_val - 1e-9:
                a = a_candidate
                best_val = val_cand
                best_a = a_candidate.copy()
                improved = True
                break
        if not improved:
            # possible stuck; add jitter
            jitter = np.random.randn(n) * (step0 / np.sqrt(t)) * 0.01
            a = project_to_simplex(a + jitter)
        # early exit if converged: if step size very small and no improvement for many iters
        if t > 5000 and not improved:
            break
        # check time each iteration
        if time.time() > deadline:
            break
    return best_a, best_val

def projected_gradient_descent(a0, deadline, max_iters=100000, step0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    i=0
    while i < max_iters and time.time() < deadline:
        i+=1
        conv = full_convolution(a)
        max_val = conv.max()
        # find active shifts within tolerance
        tol = max_val * 1e-12 + 1e-15
        active = np.where(conv >= max_val - tol)[0]
        # compute gradient
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # step size schedule
        alpha = step0 / np.sqrt(i)
        a_new = a - alpha * g
        a_new = project_to_simplex(a_new)
        val_new = compute_score(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            # optionally increase step0?
        # acceptance criteria
        if val_new < best_val * 0.9999999: # but we already check
            a = a_new
        else:
            # no improvement; maybe add jitter or reduce step0
            if i % 100 == 0:
                jitter = np.random.randn(n) * (step0 / np.sqrt(i)) * 0.01
                a = project_to_simplex(a + jitter)
    return best_a, best_val

"""
Propose a non‑negative step‑height sequence that minimises
``evaluate_sequence``.

Strategy
--------
*   Normalise every candidate to a probability vector (∑a_i = 1).
    The evaluation is scale‑invariant, therefore the objective reduces to
    ``2·n·max(conv(a,a))``.
*   Generate many random initial vectors (uniform, Dirichlet with random
    concentration, a shuffled Dirichlet, etc.).
*   For each start run a **projected sub‑gradient descent** on the exact
    objective.  The sub‑gradient of the max‑convolution is a sum of
    contributions from all antidiagonals attaining the current maximum.
    After every step we project back onto the probability simplex.
    The step size is decreased slowly (∝1/√t) and a small jitter is added
    periodically to escape flat regions.
*   Keep the best vector found so far; if the time budget allows, polish the
    current best a little longer.
*   Finally return the best sequence as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a probability‑normalised vector.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient descent (projected onto the simplex)
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000,
                        step0: float = 0.8,
                        jitter_every: int = 3000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    step = step0

    while t < max_iters and time.time() < deadline:
        t += 1

        # ---------- gradient of the max‑convolution ----------
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-15
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # ---------- line‑search (simple back‑track) ----------
        improved = False
        # step size decays like 1/√t
        cur_step = step / np.sqrt(t)
        for factor in (1.0, 0.5, 0.25):
            cand = project_to_simplex(a - cur_step * factor * g)
            val = compute_score(cand)
            if val < best_val:
                a = cand
                best_a, best_val = cand, val
                improved = True
                break

        if not improved:
            # no improvement → shrink the step a little
            step *= 0.9

        # ---------- occasional jitter ----------
        if t % jitter_every == 0:
            jitter = np.random.randn(n) * cur_step * 0.02
            a = project_to_simplex(a + jitter)

    return best_a, best_val


# ----------------------------------------------------------------------
# Random initialisations
# ----------------------------------------------------------------------


def random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Sample a Dirichlet distribution with concentration ``alpha``."""
    if alpha <= 0:
        alpha = 1e-3
    g = np.random.gamma(alpha, 1.0, size=n)
    a = g / g.sum()
    return a


def shuffled_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Dirichlet sample with a random permutation of the coordinates."""
    a = random_dirichlet(n, alpha)
    np.random.shuffle(a)
    return a


def uniform_vector(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)
