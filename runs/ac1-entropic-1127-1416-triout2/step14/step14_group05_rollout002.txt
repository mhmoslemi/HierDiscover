"""
Non‑negative step‑function optimiser (enhanced).

This version builds on the previous approach but adds a dedicated
sub‑gradient descent on the exact max‑convolution objective,
more aggressive Dirichlet seeding and an extended mass‑transfer
hill‑climb.  The optimisation proceeds in several time‑aware phases:

0. Load a known construction (``height_sequence_1``) if available,
   otherwise start from the uniform distribution on the simplex.
1. Sample many Dirichlet vectors with a wide range of concentration
   parameters to obtain good seeds.
2. Run a short Adam optimiser on the smooth‑max surrogate for a few
   restarts.
3. Perform projected sub‑gradient descent on the true objective
   ``max(conv(a,a))`` (the most critical part for further improvement).
4. Finish with a long mass‑transfer hill‑climb for polishing.
5. Return the best sequence as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        # Uniform distribution is the projection of any non‑positive vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a via a real FFT.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (sum of coefficients is assumed to be 1):
        score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

    """
    Adam optimiser on the smooth‑max surrogate
        f_τ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    The function returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # temperature schedule (geometric)
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # full convolution
        conv = full_convolution(a)               # length m
        shift = conv.max()
        # soft‑max weights (stable formulation)
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # shape (m,)

        # gradient of smooth‑max: 2·(a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # bookkeeping
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter={t}, τ={tau:.1e}, lr={lr:.3e}, score={cur_val:.6f}")

        # learning‑rate decay
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-6)

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a) = max_k (a*a)_k .
    The sum of the coefficients is kept equal to 1 (simplex projection).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    t = 0

    while time.time() < deadline:
        # compute current convolution and its maximum
        conv = full_convolution(a)
        max_val = conv.max()
        # indices attaining the maximum (allow tiny numerical tolerance)
        ks = np.where(conv >= max_val - 1e-12)[0]

        # compute sub‑gradient: average of gradients for all active ks
        grad = np.zeros_like(a)
        for k in ks:
            i = np.arange(n)
            j = k - i
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        grad /= len(ks)               # average if several active constraints

        t += 1
        # step size – classic diminishing schedule
        step = 0.2 / np.sqrt(t)

        a = a - step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if t % 500 == 0:
                print(f"[subgrad] iter={t}, score={cur_val:.6f}")

        # occasional stochastic jitter to avoid stagnation
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Small mass‑transfer hill‑climb (as in the previous code)
# ----------------------------------------------------------------------


def mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        frac = np.random.rand() * 0.30          # up to 30 % of source mass
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # already on the simplex
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:
                print(f"[mass] iter={it}, score={val:.6f}")
        it += 1
    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    t0 = time.time()
    # keep a small safety margin for the harness
    deadline = t0 + budget_s - 4.0

    # --------------------------------------------------------------
    # 0. Load a known construction if it exists, otherwise start uniform.
    # --------------------------------------------------------------
    fallback = None
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
    except Exception:
        pass

    if fallback is not None:
        a0 = np.array(fallback, dtype=float)
        a0 = np.clip(a0, 0.0, 1000.0)
        if a0.sum() == 0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 = a0 / a0.sum()
        n = a0.shape[0]
        print(f"[init] using fallback of length {n}")
    else:
        n = 2048
        a0 = np.full(n, 1.0 / n)
        print("[init] no fallback – start with uniform vector")

    best_a = a0.copy()
    best_val = compute_score(best_a)
    print(f"[init] initial score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Aggressive Dirichlet seeding (different concentration parameters).
    # --------------------------------------------------------------
    while time.time() < deadline - 200:   # reserve ~200 s for later phases
        # draw concentration α from a log‑uniform distribution
        log_alpha = np.random.uniform(np.log(0.01), np.log(5.0))
        alpha = np.exp(log_alpha)
        a = np.random.dirichlet(np.full(n, alpha))
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[dirichlet] alpha={alpha:.3f} improved score = {val:.6f}")

    # --------------------------------------------------------------
    # 2. Short Adam optimisation on a few promising starts.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30:
        starts = [best_a]
        starts.append(np.full(n, 1.0 / n))            # uniform
        for _ in range(2):
            starts.append(np.random.dirichlet(np.full(n, 1.0)))
        per_start = remaining / len(starts)

        for idx, start_vec in enumerate(starts):
            if time.time() >= deadline - 10:
                break
            local_deadline = time.time() + per_start - 2.0
            if local_deadline <= time.time():
                continue
            print(f"[adam] start {idx+1}/{len(starts)} – budget ≈ {per_start:.1f}s")
            a_ref = smooth_max_adam(start_vec, local_deadline)
            val = compute_score(a_ref)
            if val < best_val:
                best_val = val
                best_a = a_ref.copy()
                print(f"[adam] new best score = {val:.6f}")

    # --------------------------------------------------------------
    # 3. Sub‑gradient descent on the exact objective.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20:
        print("[subgrad] starting sub‑gradient descent")
        sub_deadline = time.time() + remaining - 10.0
        a_ref = subgradient_descent(best_a, sub_deadline)
        val = compute_score(a_ref)
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()
            print(f"[subgrad] new best score = {val:.6f}")

    # --------------------------------------------------------------
    # 4. Final mass‑transfer hill‑climb for polishing.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5:
        print("[mass] final mass‑transfer hill‑climb")
        mass_deadline = time.time() + remaining - 2.0
        a_ref = mass_transfer_hill_climb(best_a, mass_deadline)
        val = compute_score(a_ref)
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()
            print(f"[mass] final best score = {val:.6f}")

    # --------------------------------------------------------------
    # 5. Report using the official evaluation function.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)   # the harness provides this
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq
