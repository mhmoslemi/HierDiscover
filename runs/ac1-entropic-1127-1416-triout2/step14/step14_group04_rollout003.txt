def project_simplex(v):
    # v is numpy array, may have negative values.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Improved search for a non‑negative step‑function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (∑seq)²

The algorithm runs in four phases (within a 1000 s wall‑clock budget)

1. **Broad exploration** – many random draws (Dirichlet, monotone, mixtures of
   simple basis shapes, and a dense random sweep of the Beta(α,β) family,
   especially the U‑shaped region α,β∈(0,1) that often beats the previous best).

2. **Focused 1‑D/2‑D sweeps** – fine grid on the symmetric Beta(α,α) family
   and on the exponential family.  Only the most promising candidates are
   kept.

3. **Soft‑max (Adam) refinement** – optimisation of the smooth approximation of
   the max‑convolution using a soft‑max temperature that is annealed.  Several
   restarts with different initial temperatures are performed.

4. **Simulated‑annealing polish** – a final Metropolis hill‑climb that can
   accept worsening moves and uses an adaptive step‑scale.

All candidates are stored on the probability simplex (they sum to 1).  The
best sequence found is returned as a plain Python list of floats.

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate bookkeeping
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best):
    """Update the (score, vector) list ``best`` if ``vec`` improves it."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s <= 0.0:
        return
    vec = vec / s
    sc = compute_score(vec)
    # keep a small set of the best candidates (size ≤ 15)
    if len(best) < 15:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    # Guard against overflow / NaN for extreme parameters
    left = np.power(i, alpha - 1.0, where=(i > 0), out=np.zeros_like(i))
    right = np.power(n - i + 1.0, beta - 1.0, where=(n - i + 1.0 > 0), out=np.zeros_like(i))
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random vector, sort it decreasingly, and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]  # decreasing
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_triangle(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape."""
    m = (n - 1) // 2
    left = np.arange(1, m + 2, dtype=float)          # 1 .. m+1
    right = np.arange(m, 0, -1, dtype=float)         # m .. 1
    a = np.concatenate([left, right]) if n % 2 == 1 else np.concatenate([left, right])
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_mixture(n: int, basis, coeffs):
    """Linear mixture of pre‑computed basis vectors (each already normalised)."""
    a = np.zeros(n, dtype=float)
    for w, b in zip(coeffs, basis):
        a += w * b
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Broad random baseline (wide random search)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float):
    """Collect a diverse set of promising candidates."""
    best = []  # list of (score, vector) – smallest scores first

    # 1) Dirichlet draws with a wide range of concentration parameters
    alphas = [0.03, 0.07, 0.15, 0.3, 0.7, 1.0, 2.0, 5.0]
    draws_per_alpha = 1200
    for conc in alphas:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    # 2) Random monotone vectors (decreasing)
    for _ in range(3000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # 3) Random mixtures of a few simple bases
    bases = [
        np.full(n, 1.0 / n),               # uniform
        generate_triangle(n),              # symmetric tent
        generate_exponential(n, 0.001),   # mildly decreasing exponential
        generate_powerlaw(n, 0.5),        # mild power‑law
    ]
    for _ in range(2500):
        if time.time() >= deadline:
            break
        coeffs = np.random.dirichlet(np.ones(len(bases)))
        a = generate_mixture(n, bases, coeffs)
        _try_candidate(a, best)

    # 4) Random Beta draws (focus on the U‑shaped region)
    for _ in range(5000):
        if time.time() >= deadline:
            break
        # Sample log‑uniform α,β in [0.02, 2]
        log_alpha = np.random.uniform(np.log(0.02), np.log(2.0))
        log_beta = np.random.uniform(np.log(0.02), np.log(2.0))
        alpha, beta = np.exp(log_alpha), np.exp(log_beta)
        a = generate_beta(n, alpha, beta)
        _try_candidate(a, best)

    # 5) Gaussian family (various widths)
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 20)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        a = generate_gaussian(n, sigma)
        _try_candidate(a, best)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Focused sweeps over promising parametric families
# ----------------------------------------------------------------------


def fine_exponential_search(n: int, deadline: float):
    """Dense scan of the exponential family."""
    best_sc = float("inf")
    best_vec = None
    lam_vals = np.linspace(0.0002, 0.0030, 3500)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        a = generate_exponential(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec


def fine_beta_symmetric_search(n: int, deadline: float):
    """Fine grid on the symmetric Beta(α,α) family (α∈(0.05,2))."""
    best_sc = float("inf")
    best_vec = None
    # use a log‑space grid because the optimum often lies near α≈0.3‑0.7
    al_vals = np.geomspace(0.05, 2.0, 300)
    for al in al_vals:
        if time.time() >= deadline:
            break
        a = generate_beta(n, al, al)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on a soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.15,
                    init_tau_factor: float = 80.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.99995) -> np.ndarray:
    """
    Adam on the soft‑max representation a = softmax(x).
    The temperature τ starts large (init_tau_factor·max_conv) and is slowly
    annealed.  Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    # Initialise logits safely
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Initialise temperature from the current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # --------- soft‑max mapping ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # --------- convolution ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # --------- soft‑max weighting (smooth max) ----------
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # --------- gradient w.r.t a (2·∑_k w_k·a_{k−i}) ----------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)        # length = 3·n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # --------- chain rule for the logits ----------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # --------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------- anneal temperature ----------
        tau = max(min_tau, tau * tau_decay)

        # occasional exact check (helps debugging)
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    # final projection onto simplex (numerical drift)
    best_a = _project_simplex(best_a)
    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter to escape plateaus
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 4.0

    # Problem size – 2048 gives a good trade‑off between expressivity and speed.
    n = 2048

    best_seq = None
    best_val = float("inf")  # based on compute_score (sum=1)

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            if fb:
                # quick sanity check using official evaluator
                val = evaluate_sequence(fb)
                print(f"[fallback] official score = {val:.6f}")
                sc = compute_score(np.array(fb) / np.sum(fb))
                if sc < best_val:
                    best_val, best_seq = sc, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Broad random baseline (≈ 30 % of the budget)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 450.0  # reserve time for later phases
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Focused sweeps (≈ 15 % of the budget)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    if time.time() < sweep_deadline:
        # exponential sweep
        sc, vec = fine_exponential_search(n, sweep_deadline)
        if sc is not None and sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[exp‑sweep] new best = {sc:.6f}")

        # symmetric beta sweep (U‑shaped region)
        if time.time() < sweep_deadline:
            sc, vec = fine_beta_symmetric_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[beta‑sweep] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation with multiple restarts (≈ 35 % of budget)
    # -----------------------------------------------------------------
    opt_deadline = deadline - 130.0
    if best_seq is not None and time.time() < opt_deadline:
        # keep a small pool of good starting points (the best few vectors)
        start_pool = [np.array(best_seq, dtype=float)]
        # add a couple of random Dirichlet draws for diversity
        for _ in range(3):
            a = np.random.dirichlet(np.full(n, 0.8))
            start_pool.append(a)

        # use the pool for several independent runs
        while time.time() < opt_deadline:
            remaining = opt_deadline - time.time()
            # each run gets up to 25 s (or the remaining time if smaller)
            sub_deadline = time.time() + min(25.0, remaining)
            a0 = start_pool[np.random.randint(len(start_pool))]
            refined = smooth_optimize(a0, sub_deadline)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[smooth‑restart] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 10 % of budget)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 4.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(final)
        print(f"[anneal] final polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 6. Official evaluation (the harness will call ``evaluate_sequence``)
    # -----------------------------------------------------------------
    official_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {official_score:.6f}")
    return best_seq
