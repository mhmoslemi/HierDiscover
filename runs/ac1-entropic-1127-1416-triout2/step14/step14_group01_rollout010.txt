def run_exponential_gradient(a0, deadline, max_iter=100000, eta0=0.5):
    a = np.copy(a0)
    n = len(a)
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = full_convolution(a)
        max_c = conv.max()
        if max_c == 0:
            break
        # Identify active indices within tolerance relative to max.
        eps = max_c * 1e-9
        active = np.where(conv >= max_c - eps)[0]
        # compute subgradient g
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # step size schedule:
        eta = eta0 / np.sqrt(t)
        # Exponential update
        a = a * np.exp(-eta * g)
        # avoid all zeros
        a = np.maximum(a, 1e-16)  # small epsilon to avoid zero
        a = a / a.sum()
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # Optionally print progress.
            if t % 500 == 0:
                print(f"[EG] iter {t} best {best_val:.6f}")
        # If delay too long
    return best_a, best_val

def subgradient_opt(a0, deadline, step0=0.6, tol_factor=1e-4):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # tolerance
        tol = max(max_val * tol_factor, 1e-12)
        active = np.where(conv >= max_val - tol)[0]
        # compute subgradient
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # step size
        step = step0 / np.sqrt(t)
        a = a - step * g
        # projection onto simplex
        a = project_to_simplex(a)
        cur_val = compute_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            if t % 500 == 0:
                print(f"[SGD] iter {t} best {best_val:.6f}")
        # jitter occasionally
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)
    return best_a, best_val

def propose_candidate(seed=12345,budget_s=1000,**kwargs):
    """
    Heuristic search for a non‑negative step‑height sequence that minimises
    ``evaluate_sequence``.  The algorithm builds several candidate
    initialisations (fallback, uniform, geometric decays, random Dirichlet)
    and refines each with a projected sub‑gradient / exponentiated‑gradient
    descent on the exact objective (ignoring the scale‑invariant clipping).
    The best vector seen before the deadline is returned.
    """
    import time, numpy as np, random
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # Determine length
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 2048

    best_seq = None
    best_score = float('inf')

    # Helper to evaluate using compute_score (fast) and final evaluate_sequence
    def record_candidate(arr, tag):
        nonlocal best_seq, best_score
        val = compute_score(arr)
        if val < best_score:
            best_score = val
            best_seq = arr.copy()
            print(f"[{tag}] new best score {best_score:.6f}")

    # 1. fallback if available
    try:
        fallback_arr = np.array(height_sequence_1, dtype=float)
        fallback_arr = np.maximum(fallback_arr, 0.0)
        if fallback_arr.sum() < 0.01:
            fallback_arr = fallback_arr / fallback_arr.sum()
        else:
            fallback_arr = fallback_arr / fallback_arr.sum()
        record_candidate(fallback_arr, "fallback")
    except Exception:
        pass

    # 2. uniform
    uniform = np.full(n, 1.0 / n, dtype=float)
    record_candidate(uniform, "uniform")

    # 3. geometric families
    for q in np.linspace(0.95, 0.9999, num=12):
        if time.time() >= deadline:
            break
        geom = (1.0 - q) * (q ** np.arange(n))
        geom = geom / geom.sum()
        record_candidate(geom, f"geom_{q:.4f}")

    # 4. random Dirichlet restarts
    # we will run subgradient on each
    while time.time() < deadline - 10.0:
        # generate a random init
        a0 = np.random.gamma(shape=1.0, scale=1.0, size=n)
        a0 = a0 / a0.sum()
        # allocate a slice of time for refinement
        slice_deadline = min(deadline, time.time() + 8.0)   # ~8 s per run
        # choose optimisation method randomly
        if random.random() < 0.5:
            a_opt, val = subgradient_opt(a0, slice_deadline, step0=0.6, tol_factor=1e-4)

            a_opt, val = exponential_opt(a0, slice_deadline, eta0=0.5)
            method_tag = "eg"

    # Safeguard: if nothing succeeded (unlikely) use uniform
    if best_seq is None:
        best_seq = uniform
        best_score = compute_score(best_seq)

    final_score = evaluate_sequence(best_seq.tolist())

def exponential_opt(a0, deadline, eta0=0.5):

    n = a.shape[0]
    best_a = a.copy()

"""
Heuristic search for a non‑negative step‑height sequence that minimises
``evaluate_sequence``.  The method builds a variety of initial candidates
(uniform, a geometric decay, random Dirichlet draws and the provided
fallback ``height_sequence_1``) and refines each of them with a projected
sub‑gradient / exponentiated‑gradient descent on the exact objective

    f(a) = 2·n·max(conv(a,a))    (the L1‑norm of a is kept equal to 1)

The optimisation runs until the time budget expires; the best sequence
encountered is returned.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Geometry utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (fast for n≈2k)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def subgradient_opt(a0: np.ndarray,
                    deadline: float,
                    step0: float = 0.8,
                    tol_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    ``a0`` must be a probability vector (sum=1, a_i≥0).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()

        # dynamic tolerance: start coarse, become finer as we iterate
        tol = max(max_val * max(tol_factor, 1.0 / np.sqrt(t)), 1e-12)
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = a.copy()
            if t % 500 == 0:
                print(f"[SGD] iter {t:5d} – new best {best_val:.6f}")

        # occasional small jitter to escape shallow plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_val


def exponential_opt(a0: np.ndarray,
                    deadline: float,
                    eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on the simplex) using the same
    sub‑gradient as above.  Multiplicative updates often give a smoother
    trajectory for the max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # use a very tight set of active indices (the exact max)
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)          # multiplicative update
        a = np.maximum(a, 1e-16)          # avoid exact zeros
        a = a / a.sum()                    # projection onto simplex

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        if t % 500 == 0:
            # small random kick to keep the search lively
            jitter = np.random.randn(n) * (eta0 * 0.005 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a, best_val

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The function runs for up to ``budget_s`` seconds
    and returns the best vector found (as plain Python floats).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Determine the problem size (use the fallback length if available)
    # -----------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2048

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Helper to record and keep the best candidate seen so far
    # -----------------------------------------------------------------
    def record_candidate(arr: np.ndarray, tag: str) -> None:
        nonlocal best_seq, best_score
        val = compute_score(arr)               # fast exact objective (sum=1)
        if val < best_score:
            best_score = val
            best_seq = arr.copy()
            print(f"[{tag}] new best score {best_score:.6f}")

    # -----------------------------------------------------------------
    # 0. Fallback construction, if the global variable exists
    # -----------------------------------------------------------------
    try:
        fallback = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        fallback = np.maximum(fallback, 0.0)
        # normalise to a probability vector
        if fallback.sum() < 0.01:
            fallback = fallback / fallback.sum()
        else:
            fallback = fallback / fallback.sum()
        record_candidate(fallback, "fallback")
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Uniform vector (baseline)
    # -----------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=float)
    record_candidate(uniform, "uniform")

    # -----------------------------------------------------------------
    # 2. Geometric (exponential decay) families
    # -----------------------------------------------------------------
    for q in np.linspace(0.95, 0.9999, num=12):
        if time.time() >= deadline:
            break
        geom = (1.0 - q) * (q ** np.arange(n))
        geom = geom / geom.sum()
        record_candidate(geom, f"geom_{q:.4f}")

    # -----------------------------------------------------------------
    # 3. Random Dirichlet restarts – each refined with a short optimisation
    # -----------------------------------------------------------------
    # allocate roughly the remaining time to many short runs (≈8 s each)
    while time.time() < deadline - 8.0:
        # draw a random probability vector
        a0 = np.random.gamma(shape=1.0, scale=1.0, size=n)
        a0 = a0 / a0.sum()

        # give each run a slice of the remaining budget
        slice_deadline = min(deadline, time.time() + 8.0)

        # randomly pick one of the two optimisation schemes
        if random.random() < 0.5:
            a_opt, _ = subgradient_opt(a0, slice_deadline,
                                       step0=0.8, tol_factor=1e-4)
            tag = "sgd"
        else:
            a_opt, _ = exponential_opt(a0, slice_deadline, eta0=0.5)
            tag = "eg"

        record_candidate(a_opt, tag)

    # -----------------------------------------------------------------
    # 4. Safeguard – if nothing succeeded, fall back to the uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform
        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # 5. Final validation with the provided ``evaluate_sequence`` implementation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final evaluated score = {final_score:.6f}")

    return best_seq.tolist()
