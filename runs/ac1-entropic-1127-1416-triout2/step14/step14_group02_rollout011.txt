def baseline_search(n, deadline, max_keep=10, random_draws=15000):
    half = n//2
    best = [] # list of (score, vec)
    def add_candidate(full_vec):
        full_vec = np.maximum(full_vec, 0)
        full_vec = full_vec / full_vec.sum() # normalize; or using project_to_simplex
        sc = compute_score(full_vec)
        if len(best) < max_keep:
            best.append((sc, full_vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, full_vec.copy())
            best.sort(key=lambda x: x[0])
    # families loops
    for p in np.linspace(0.1, 8.0, 81):
        if time.time()>=deadline: break
        half_vec = np.arange(1, half+1) ** (-p)  # powerlaw decreasing
        full_vec = symmetrize(half_vec)
        add_candidate(full_vec)
    # exponential
    for lam in np.linspace(0.001, 1.0, 120):
        if time.time()>=deadline: break
        half_vec = np.exp(-lam * np.arange(half))
        full_vec = symmetrize(half_vec)
        add_candidate(full_vec)
    # beta, gaussian, etc. (maybe skip for brevity)
    # Random Dirichlet draws for half or full:
    concs = [0.01, 0.02,0.05,0.1,0.2,0.5,1.0,2.0,5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time()>=deadline: break
            half_vec = np.random.dirichlet([conc]*half)
            full_vec = symmetrize(half_vec)
            add_candidate(full_vec)
    # Also random Dirichlet full length (non-symmetric) for diversity:
    for _ in range(2000):
        if time.time()>=deadline: break
        vec = np.random.dirichlet([1.0]*n)  # uniform Dirichlet
        add_candidate(vec)
    # return list of vectors
    best.sort(key=lambda x: x[0])
    return [vec for (_, vec) in best]

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0
    n = 4096
    best_seq = None
    best_val = float('inf')
    # baseline
    baseline_deadline = min(deadline, start + 150)
    cand_vecs = baseline_search(n, baseline_deadline, max_keep=20, random_draws=20000)
    for vec in cand_vecs:
        val = compute_score(vec)
        if val < best_val:
            best_val, best_seq = val, vec.tolist()
    print(f"[baseline] best score = {best_val:.6f}")
    # Adam optimization
    remaining = deadline - time.time() - 250.0
    if remaining > 30 and best_seq is not None:
        start_vecs = [np.array(best_seq), np.full(n, 1.0/n), np.random.rand(n)]
        for idx, start_vec in enumerate(start_vecs):
            if time.time() >= deadline - 250: break
            local_deadline = time.time() + min(45.0, remaining / len(start_vecs))
            # attempt symmetric and unsymmetric
            opt_vec = combined_adam(start_vec, local_deadline, lam=0.5, tau_start=1.0, tau_end=1e-9, tau_decay=0.98, lr_start=0.2)
            val = compute_score(opt_vec)
            print(f"[adam {idx}] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, opt_vec.tolist()
    # subgradient polish
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq)
        a_pol = subgradient_polish(a0, polish_deadline, step0=0.3)
        val = compute_score(a_pol)
        print(f"[polish] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_pol.tolist()
    # annealing
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a0 = np.array(best_seq)
        a_f = simulated_annealing(a0, anneal_deadline, init_temp=1e-2, final_temp=1e-10,
                                    cooling_factor=0.9999, step_scale_init=0.03,
                                    step_scale_min=1e-6)
        val = compute_score(a_f)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_f.tolist()
    # fallback if nothing
    if best_seq is None:
        best_seq = [1.0/n]*n
    final_score = evaluate_sequence(best_seq)
    print(f"Final reported score = {final_score:.6f}")
    return best_seq

"""
Advanced search for a step‑function with a low max‑convolution ratio.

Core ideas
----------
* Work with *symmetric* sequences (a[i] = a[n‑1‑i]) – the problem is
  symmetric, and the optimum can be taken symmetric, which cuts the
  dimensionality in half and makes the optimisation landscape smoother.
* An extensive cheap baseline explores analytic families (power‑law,
  exponential, Gaussian, beta) and a large pool of Dirichlet draws
  (both full‑length and half‑length for symmetry).  The best few
  candidates are kept.
* A customised Adam optimiser minimises a blended surrogate
  ``L = (1‑λ)·τ·log‑sum‑exp(conv/τ) + λ·½·‖conv–uniform‖²`` while
  enforcing symmetry after each update.
* A projected sub‑gradient polish further reduces the exact max‑objective.
* Finally a long simulated‑annealing hill‑climb with very slow cooling
  converges to a refined solution.

All steps respect the 1000 s wall‑clock limit by checking a deadline
after each iteration.  The function ``propose_candidate`` returns the
best sequence found (a plain ``list[float]``) and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum = 1, v_i >= 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Symmetry utilities
# ----------------------------------------------------------------------


def make_full_symmetric(half: np.ndarray) -> np.ndarray:
    """Given the left half (length n/2) return a full symmetric vector of length n."""
    return np.concatenate([half, half[::-1]])


def enforce_symmetry(v: np.ndarray) -> np.ndarray:
    """Force exact symmetry a[i] = a[n‑1‑i] (averaging both sides)."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, zero‑padding)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using the real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of a·(a convoluted with weight vector `vec`).
    It equals 2·(a ⋆ rev(vec)).
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Exact scoring (assumes sum of a = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (half‑length, later symmetrised)
# ----------------------------------------------------------------------


def gen_powerlaw_half(n_half: int, p: float) -> np.ndarray:
    a = np.arange(1, n_half + 1, dtype=float) ** (-p)
    return a / a.sum()


def gen_exponential_half(n_half: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n_half, dtype=float))
    return a / a.sum()


def gen_gaussian_half(n_half: int, sigma: float) -> np.ndarray:
    mid = (n_half - 1) / 2.0
    x = np.arange(n_half, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return a / a.sum()


def gen_beta_half(n_half: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n_half + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n_half - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        a = np.zeros_like(i)
    if a.sum() == 0:
        return np.full(n_half, 1.0 / n_half)
    return a / a.sum()


def gen_sine_power_half(n_half: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n_half, dtype=float) + 0.5) / n_half
    a = np.sin(angles) ** p
    if a.sum() == 0:
        return np.full(n_half, 1.0 / n_half)
    return a / a.sum()


def mixture(v1: np.ndarray, v2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * v1 + (1.0 - lam) * v2
    mixed = project_to_simplex(mixed)
    return mixed


# ----------------------------------------------------------------------
# Global baseline exploration (cheap, many random draws)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 15000,
) -> list[np.ndarray]:
    """Explore analytic families and Dirichlet draws, keep the best `max_keep`."""
    half = n // 2
    best = []  # list of (score, vector)

    def try_candidate(full_vec: np.ndarray):
        if not np.isfinite(full_vec).all():
            return
        full_vec = np.maximum(full_vec, 0.0)
        full_vec = project_to_simplex(full_vec)
        sc = compute_score(full_vec)
        if len(best) < max_keep:
            best.append((sc, full_vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, full_vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- power‑law half ----- (decreasing towards centre)
    for p in np.linspace(0.1, 8.0, 81):
        if time.time() >= deadline:
            break
        half_vec = gen_powerlaw_half(half, p)
        full_vec = make_full_symmetric(half_vec)
        try_candidate(full_vec)

    # ----- exponential half -----
    for lam in np.linspace(0.001, 1.0, 120):
        if time.time() >= deadline:
            break
        half_vec = gen_exponential_half(half, lam)
        full_vec = make_full_symmetric(half_vec)
        try_candidate(full_vec)

    # ----- Gaussian half -----
    for sigma in np.linspace(0.2 * half, 0.6 * half, 30):
        if time.time() >= deadline:
            break
        half_vec = gen_gaussian_half(half, sigma)
        full_vec = make_full_symmetric(half_vec)
        try_candidate(full_vec)

    # ----- Beta half -----
    al_vals = np.linspace(0.02, 2.0, 21)
    be_vals = np.linspace(0.02, 2.0, 21)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            half_vec = gen_beta_half(half, alpha, beta)
            full_vec = make_full_symmetric(half_vec)
            try_candidate(full_vec)

    # ----- Random beta draws (half) -----
    for _ in range(300):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        half_vec = gen_beta_half(half, alpha, beta)
        full_vec = make_full_symmetric(half_vec)
        try_candidate(full_vec)

    # ----- Sine‑power half -----
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        half_vec = gen_sine_power_half(half, p)
        full_vec = make_full_symmetric(half_vec)
        try_candidate(full_vec)

    # ----- Mixtures of families (half) -----
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = gen_powerlaw_half(half, np.random.uniform(0.1, 8.0))
        b = gen_beta_half(half,
                          10 ** np.random.uniform(-2, 0.3),
                          10 ** np.random.uniform(-2, 0.3))
        lam = np.random.rand()
        mixed_half = mixture(a, b, lam)
        full_vec = make_full_symmetric(mixed_half)
        try_candidate(full_vec)

    # ----- Dirichlet draws (full length, non‑symmetric) -----
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet([conc] * n)
            try_candidate(vec)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Combined Adam optimiser (smooth‑max + L2 flattening, symmetry enforced)
# ----------------------------------------------------------------------


def combined_adam(
    a0: np.ndarray,
    deadline: float,
    lam: float = 0.4,
    tau_start: float = 1.0,
    tau_end: float = 1e-9,
    tau_decay: float = 0.98,
    lr_start: float = 0.2,
    lr_decay: float = 0.97,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on a blended surrogate:
        L(a) = (1‑λ)·τ·log Σ exp(conv/τ)   +   λ·½·‖conv – uniform‖² .
    Symmetry is enforced after each update.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m

    # FFT size large enough for correlation of length n + m - 1
    size_fft = 1 << ((3 * n - 3).bit_length())

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # schedules
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # convolution and smooth‑max weights
        conv = full_convolution(a)                     # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)       # stable log‑sum‑exp
        w = exp_vals / exp_vals.sum()                 # soft‑max distribution

        # flattening error term
        err = conv - target

        # gradients
        grad_max = correlation_grad(a, w, size_fft)
        grad_err = correlation_grad(a, err, size_fft)
        grad = (1.0 - lam) * grad_max + lam * grad_err

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # enforce symmetry and simplex
        a = enforce_symmetry(a)
        a = project_to_simplex(a)

        # bookkeeping
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = enforce_symmetry(a)
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (exact max‑objective, symmetry enforced)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑objective with symmetry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = enforce_symmetry(a)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = enforce_symmetry(a)
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (symmetry maintained)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-12,
    cooling_factor: float = 0.9995,
    step_scale_init: float = 0.05,
    step_scale_min: float = 5e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that preserves symmetry."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)

        # re‑impose symmetry and normalise
        cand = enforce_symmetry(cand)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = enforce_symmetry(a)
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

    # -----------------------------------------------------------------

    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
