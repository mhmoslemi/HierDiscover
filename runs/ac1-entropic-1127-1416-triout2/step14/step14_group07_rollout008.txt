def directed_hill_climb(a_start, deadline, max_no_improve=5000):
    a = a_start.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    n = len(a)
    step_frac = 0.1
    no_improve = 0
    while time.time() < deadline:
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        # compute subgradient g_i
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # for diagonal where i == k - i, we double counted; adjust? but ignore small.
        # Choose i as argmax(g * a) maybe.
        # but we want i with large contribution and enough mass.
        # compute contributions = g * a (maybe)
        contrib = g * a
        # ensure we remove small contributions
        i_candidates = np.where(a > 1e-12)[0]
        if len(i_candidates) == 0:
            break
        i = i_candidates[np.argmax(contrib[i_candidates])]
        # choose j as index with minimal contribution (or minimal g)
        j = np.argmin(g)  # could also ensure a[j] not too large
        # ensure i != j
        if i == j:
            # random fallback
            i, j = np.random.choice(n, 2, replace=False)
        delta = step_frac * a[i]
        if delta < 1e-12:
            delta = step_frac * 1e-12  # ensure some move
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0, None)
        cand = project_to_simplex(cand)
        M_cand = full_convolution(cand).max()
        if M_cand < best_M - 1e-14:
            best_M = M_cand
            best = cand.copy()
            a = cand
            step_frac = min(step_frac * 1.05, 0.5)
            no_improve = 0
        else:
            step_frac *= 0.98
            no_improve += 1
        if no_improve > max_no_improve:
            # jitter and reset
            jitter = np.random.randn(n) * (step_frac * 0.001)
            a = best + jitter
            a = np.clip(a, 0, None)
            a = project_to_simplex(a)
            no_improve = 0
    return best

def plateau_exponential(n, L_frac, lam):
    L = int(L_frac*n)
    if L <=0 or L >= n: return None
    tail_len = n - L
    i = np.arange(tail_len)
    tail = np.exp(-lam * i)
    head_val = tail.max()
    head = np.full(L, head_val)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a
```

    a = a0.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_frac = 0.05
    eps = 1e-12
    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iter:
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - eps)[0]
        # compute subgradient g
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # compute direction: move from high g to low g
        # we can find i = argmax(g * a) and j = argmin(g)
        contrib = g * a
        i = np.argmax(contrib)  # position with high contribution
        j = np.argmin(g)        # lowest gradient
        if i == j:
            i, j = np.random.choice(n, 2, replace=False)
        delta = step_frac * a[i]
        if delta < 1e-12:
            delta = step_frac * 1e-12
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0, None)
        cand = project_to_simplex(cand)
        M_cand = full_convolution(cand).max()
        if M_cand < best_M - 1e-14:
            best_M = M_cand
            best = cand.copy()
            a = cand

            no_improve = 0

propose_candidate – search for a low‑value step‑function.

Algorithm
---------

* **Phase 0 – safety fallback** – try the global variable ``height_sequence_1``.
* **Phase 1 – quick parametric scan** – a few analytic families
  (reverse power‑law, two‑parameter “multi‑power”, a front‑plateau with
  exponential tail and a few Dirichlet‑type random draws).  The best
  vector is kept (all vectors are normalised to Σa = 1, the invariant
  of the objective).

* **Phase 2 – directed hill‑climb** – a gradient‑style local optimiser.
  At each step we compute the active set of the convolution (all
  indices where the convolution reaches its current maximum), form the
  sub‑gradient of ``max(conv)`` and move a small amount of mass from
  the “worst” contributor (largest sub‑gradient) to the “best’’
  contributor (smallest sub‑gradient).  The move is projected onto the
  probability simplex.  The step‑size is adapted, occasional jitter
  restarts the search when progress stalls.

* **Phase 3 – final random fine‑tuning** – a short stochastic hill‑climb
  (as in the previous code) is applied as a cleanup pass.

The whole process respects the 1000 s budget (a 5 s safety margin) and
always returns a non‑negative list of floats that minimises
``evaluate_sequence``.

# Linear‑algebra utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The scale‑invariance of the original problem allows us to keep the sum
    fixed to one.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p,   p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n … 1
    a = idx ** p
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.
    Both exponents are ≥0.
    """
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def plateau_exponential(n: int, L_frac: float, lam: float) -> np.ndarray:
    """
    Front plateau of length L = round(L_frac·n) with a constant value,
    followed by a decreasing exponential tail a_i ∝ exp(-lam·i).
    """
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = np.exp(-lam * np.arange(tail_len, dtype=np.float64))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^α (1‑x_i)^β,   x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

# Directed gradient‑style hill‑climb (Phase 2)
# ----------------------------------------------------------------------
def directed_hill_climb(a0: np.ndarray, deadline: float,
                        init_step: float = 0.04,
                        max_no_improve: int = 30000) -> np.ndarray:
    """
    Local optimisation that repeatedly moves mass from the element that
    contributes most to the current maximum of the convolution towards the
    element that contributes least.
    """
    a = a0.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step = init_step
    eps = 1e-12
    no_improve = 0
    it = 0

    while time.time() < deadline:
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - eps)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # choose source i (large contribution) and target j (small contribution)
        contrib = g * a
        i = int(np.argmax(contrib))
        j = int(np.argmin(g))

        if i == j:
            # fallback to a random pair
            i, j = np.random.choice(n, 2, replace=False)

        delta = step * a[i]
        if delta < 1e-15:
            delta = step * 1e-12

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M - 1e-14:
            best_M = M_cand
            best = cand.copy()
            a = cand
            step = min(step * 1.07, 0.35)   # become a bit bolder after improvement
            no_improve = 0
        else:
            step = max(step * 0.96, 1e-5)
            no_improve += 1

        # stagnation handling
        if no_improve > max_no_improve:
            jitter = np.random.randn(n) * (step * 0.001)
            a = best + jitter
            a = np.clip(a, 0.0, None)
            a = project_to_simplex(a)
            no_improve = 0

        it += 1
        if it % 5000 == 0 and time.time() < deadline:
            # occasional progress report
            print(f"[direct] it={it}, best_M={best_M:.6e}, step={step:.6e}", flush=True)

    return best

def random_hill_climb(a_start: np.ndarray, deadline: float,
                      max_iter: int = 800_000) -> np.ndarray:
    """Greedy hill‑climb that perturbs two coordinates at a time."""
    a = a_start.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0

        i, j = np.random.randint(n, size=2)
        if i == j:

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best = cand.copy()
            a = cand

            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            # jump‑start from the current best
            jitter = np.random.randn(n) * 0.0008
            a = best + jitter
            a = np.clip(a, 0.0, None)
            a = project_to_simplex(a)
            no_improve = 0

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # (0) simple fallback from a pre‑computed global variable
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val:.6f}", flush=True)
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # (1) parametric scan – cheap, broad exploration
    # -----------------------------------------------------------------
    n = 2048
    if time.time() < deadline - 200.0:
        vec, val = parametric_scan(n, deadline - 200.0)
        if vec is not None:
            real_val = evaluate_sequence(vec.tolist())
            print(f"[scan] best official score = {real_val:.6f}", flush=True)
            if real_val < best_val:
                best_val, best_seq = real_val, vec.tolist()

    # -----------------------------------------------------------------
    # (2) directed gradient‑style hill climb
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 50.0:
        a0 = np.array(best_seq, dtype=float)
        a_dir = directed_hill_climb(a0, deadline - 50.0)
        val = evaluate_sequence(a_dir.tolist())
        print(f"[direct] polished official score = {val:.6f}", flush=True)
        if val < best_val:
            best_val, best_seq = val, a_dir.tolist()

    # -----------------------------------------------------------------
    # (3) final random hill‑climb (fine‑tuning)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        a_final = random_hill_climb(a_start, deadline - 5.0)
        val = evaluate_sequence(a_final.tolist())
        print(f"[final] post‑random hill‑climb score = {val:.6f}", flush=True)
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # (4) safeguard – use a simple analytic shape if nothing else succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.0)   # linear decreasing shape
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] using linear shape, score = {best_val:.6f}", flush=True)

    # -----------------------------------------------------------------
    # final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}", flush=True)
    return best_seq
