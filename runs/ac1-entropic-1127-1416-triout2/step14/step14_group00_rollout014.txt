def adam_optimize(a0, deadline, init_lr=0.1, epsilon_start=..., epsilon_decay=0.9995, epsilon_min=1e-8):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps_adam = 1e-8
    t=0
    best_a = a.copy()
    best_score = compute_score(a)
    epsilon = epsilon_start
    while time.time() < deadline:
        conv = full_convolution(a)
        maxc = conv.max()
        # compute weight
        shift = maxc
        exps = np.exp((conv - shift) / epsilon)  # stable
        w = exps / np.sum(exps)
        # correlation
        corr = correlation(a, w)
        grad = 4 * n * corr
        # Adam
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = init_lr / (np.sqrt(v_hat) + eps_adam)
        a -= step
        # projection
        a = project_to_simplex(a)
        # compute exact score every, say, 10 iterations (to reduce overhead)
        if t % 10 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
        # adjust epsilon
        epsilon = max(epsilon * epsilon_decay, epsilon_min)
    return best_a

"""
search for a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
* Initialise a diverse pool of feasible sequences (arcsine, uniform,
  a few Beta‑shapes and random Dirichlet draws) – all lie on the simplex
  (sum = 1).
* For each start run a *smooth* projected Adam optimiser.
  The true objective 2·n·max(conv) is approximated by a
  log‑sum‑exp (soft‑max) of the full self‑convolution; this yields a
  smooth gradient that smoothly interpolates the sub‑gradient of the max.
  After each Adam step the iterate is projected onto the probability simplex
  to enforce non‑negativity and normalisation.
* The smoothing parameter ε is decreased geometrically in the course of
  optimisation, turning the smoothed objective into the true peak of the
  convolution.
* The best solution found so far is constantly tracked (exact score with the
  cheap ``compute_score`` routine).  When time permits a short sub‑gradient
  polish and a final simulated‑annealing hill‑climb are performed.
* The routine respects the 1000 s wall‑clock budget, returns the best
  sequence as a plain ``list`` and prints the progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def convolve_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b (real FFT, works for any lengths)."""
    n = a.shape[0]
    m = b.shape[0]
    size = n + m - 1
    n_fft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    return np.fft.irfft(C, n=n_fft)[:size]


def correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation of a (length n) with the weight vector w (length 2n‑1):
        corr[j] = Σ_i w_i * a_{i‑j},   j = 0…n‑1.
    Implemented via convolution with the reversed weight vector.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2·n‑1
    w_rev = w[::-1]
    conv = convolve_full(a, w_rev)          # length = n + m – 1 = 3·n‑2
    start = m - 1                           # index of lag 0
    # take the slice (start, start‑1, …) of length n
    return conv[start::-1][:n]


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation used during the search: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for initial guesses)
# ----------------------------------------------------------------------


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+1)^{α‑1}(n‑i)^{β‑1},  α,β>0."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape:
        a_i ∝ 1 / sqrt((i+½)*(n‑½‑i)), i = 0 … n‑1,
    normalised to the probability simplex.
    """
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (core of the new search)
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    init_lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
    eps_start_factor: float = 0.5,
    eps_decay: float = 0.9995,
    eps_min: float = 1e-9,
) -> np.ndarray:
    """
    Projected Adam optimiser on the smoothed objective
        f_ε(a) = 2·n·[ max(conv) + ε·log Σ exp((conv‑max)/ε) ].

    The smoothing parameter ε is decreased geometrically.
    The iterate is projected onto the simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initialise smoothing parameter
    conv0 = full_convolution(a)
    max0 = conv0.max()
    eps = max0 * eps_start_factor + eps_min

    t = 0
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        t += 1

        # ----- smoothed objective and its gradient -----
        conv = full_convolution(a)
        maxc = conv.max()
        # stable soft‑max weights
        exps = np.exp((conv - maxc) / eps)
        sum_exps = np.sum(exps)
        w = exps / sum_exps                     # w_i ≥ 0, Σ w_i = 1

        # correlation = Σ_i w_i * shift(a)
        corr = correlation(a, w)                # length n
        grad = 4.0 * n * corr                    # ∇ f_ε

        # ----- Adam step -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr / (np.sqrt(v_hat) + eps_adam)
        a -= step * m_hat

        # ----- projection onto the simplex -----
        a = project_to_simplex(a)

        # ----- bookkeeping -----
        if t % 20 == 0:                         # cheap periodic exact check
            cur = compute_score(a)
            if cur < best_val:
                best_val, best_a = cur, a.copy()

        # ----- decay smoothing -----
        eps = max(eps * eps_decay, eps_min)

    return best_a


# ----------------------------------------------------------------------
# Cheap sub‑gradient polish (kept from the previous pipeline)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true (non‑smooth) objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # gradient of the max‑convolution (sum of active contributions)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb (kept from the previous pipeline)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

    # if we already have a fallback, try to improve it directly
    if best_seq is not None:
        pool.append(("fallback_start", best_seq.copy()))

    # -----------------------------------------------------------------
    # 2. Main optimisation loop over the pool
    # -----------------------------------------------------------------
    for name, a0 in pool:
        if time.time() >= deadline:
            break

        # normalise onto the simplex (protect against tiny rounding errors)
        a0 = a0 / a0.sum()
        cur_score = compute_score(a0)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a0.copy()
            print(f"[{name}] initial score = {cur_score:.6f}")

        # allocate a fraction of the remaining budget to this start
        remaining = deadline - time.time()
        # keep at least 5 s for finishing steps
        if remaining < 10.0:
            break
        sub_deadline = time.time() + remaining * 0.5

        # --------- Adam optimisation with smoothing ----------
        a_opt = adam_optimize(a0, sub_deadline, init_lr=0.05)
        opt_score = compute_score(a_opt)
        print(f"[{name}] after Adam score = {opt_score:.6f}")
        if opt_score < best_score:
            best_score, best_seq = opt_score, a_opt.copy()

        # --------- optional sub‑gradient polish (if time permits) ----------
        if time.time() < deadline - 2.0:
            polished = subgradient_polish(best_seq.copy(),
                                          deadline - 2.0)
            pol_score = compute_score(polished)
            if pol_score < best_score:
                best_score, best_seq = pol_score, polished.copy()
                print(f"[{name}] polished score = {pol_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Final fine‑tuning with simulated annealing (if any time left)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("[final anneal] start")
        a_final = simulated_annealing(best_seq.copy(), deadline)
        final_score = compute_score(a_final)
        if final_score < best_score:
            best_score, best_seq = final_score, a_final.copy()
            print(f"[final anneal] final score = {final_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Safeguard – return a uniform vector if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # 5. Official evaluation (clamping etc. done inside evaluate_sequence)
    # -----------------------------------------------------------------
    final_report = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_report:.6f}")
    return best_seq.tolist()
