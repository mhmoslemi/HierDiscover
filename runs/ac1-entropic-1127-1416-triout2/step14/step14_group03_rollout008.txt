"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.

Main ideas
----------
* **Richer baseline families** – in addition to the plain power‑law,
  exponential and beta families we also explore their *symmetric*
  (U‑shaped) counterparts (large mass near the two ends).  They are
  cheap to generate and broaden the pool of good starting points.

* **Simple pairwise refinement** – after the smooth Adam stage we apply a
  deterministic “pairwise mass transfer” hill‑climb.  For the
  (currently) active convolution peak we move a small fraction of mass
  from the most responsible index to the least responsible one.
  This directly reduces the true max‑convolution value (no soft‑max
  approximation).

* **Time‑aware orchestration** – the 1000 s budget is split into
  baseline (≈150 s), smooth Adam optimisation (≈600 s), pairwise
  refinement (≈200 s) and a short simulated‑annealing polish (≈30 s).

All helper functions are top‑level, no closures or lambdas are used.
The function ``propose_candidate`` is the entry point required by the
harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via FFT (real)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b (FFT based)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a) = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (monotonically decreasing)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_powerlaw_sym(n: int, p: float) -> np.ndarray:
    """U‑shaped: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    left = np.arange(1, n + 1, dtype=float) ** (-p)
    right = np.arange(n, 0, -1, dtype=float) ** (-p)
    arr = left + right
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_exponential_sym(n: int, lam: float) -> np.ndarray:
    """U‑shaped exponential: exp(-lam·i) + exp(-lam·(n‑1‑i))."""
    left = np.exp(-lam * np.arange(n, dtype=float))
    right = np.exp(-lam * np.arange(n - 1, -1, -1, dtype=float))
    arr = left + right
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}  (standard beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 1.0 / n)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_beta_sym(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric beta‑shaped U: (i+1)^{α‑1}·(n‑i)^{β‑1} +
                              (n‑i)^{α‑1}·(i+1)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    term1 = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    term2 = np.power(i, beta - 1.0) * np.power(n - i + 1.0, alpha - 1.0)
    arr = term1 + term2
    if not np.all(np.isfinite(arr)):
        return np.full(n, 1.0 / n)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 1.0 / n)

# Helpers for baseline stage

def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate, keep if among the best."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0:
        return
    vec = vec / total
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """
    Sweep many parametric families (including symmetric U‑shapes) and
    Dirichlet draws.  Returns up to ``max_keep`` best vectors.
    """
    best = []   # list of (score, vector)

    # ---------- 1) Power‑law (plain and symmetric) ----------
    for p in np.linspace(0.05, 5.0, 120):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)
        _try_candidate(generate_powerlaw_sym(n, p), best, max_keep)

    # ---------- 2) Exponential (plain and symmetric) ----------
    for lam in np.linspace(0.0001, 0.2, 120):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)
        _try_candidate(generate_exponential_sym(n, lam), best, max_keep)

    # ---------- 3) Beta (plain and symmetric) ----------
    al_vals = np.linspace(0.05, 4.0, 30)
    be_vals = np.linspace(0.05, 4.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)
            _try_candidate(generate_beta_sym(n, alpha, beta), best, max_keep)

    # ---------- 4) Gaussian ----------
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # ---------- 5) Dirichlet draws (sorted, symmetric) ----------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 400
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)

            # descending sorted version (strongly monotone)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

            # symmetric mirror of the sorted version
            a_sym = (a_desc + a_desc[::-1]) * 0.5
            _try_candidate(a_sym, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# Upsampling helper (linear interpolation + simplex projection)
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``vec`` to length ``new_n`` and re‑project onto simplex."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.09,
                    init_tau_factor: float = 5.0,
                    min_tau: float = 1e-10,
                    tau_decay: float = 0.9993) -> np.ndarray:
    """
    Minimise the max‑convolution using a soft‑max approximation.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # log‑softmax parameters
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initialise temperature from the initial max
    init_conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * init_conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping (entropy mirror) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution of current a -----
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- smooth‑max weights -----
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient w.r.t. a (2·∑_k w_k·a_{k-i}) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the soft‑max (x‑space) -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature annealing -----
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact evaluation -----
        if it % 500 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# Pairwise deterministic refinement (mass‑transfer hill‑climb)
# ----------------------------------------------------------------------


def pairwise_optimize(a0: np.ndarray, deadline: float,
                      step_init: float = 0.05,
                      step_decay: float = 0.999,
                      step_min: float = 1e-8) -> np.ndarray:
    """
    Greedy refinement: move a small fraction of probability mass from the
    index that most increases the maximal convolution peak to the index
    that least increases it.  Works directly on the true “max”
    objective (no smoothing).  Returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    step = step_init
    it = 0
    while time.time() < deadline:
        it += 1

        # compute gradient of the current max‑convolution
        conv = full_convolution(a)
        M = conv.max()
        # weight vector for the (possibly multiple) maximal entries
        w = (conv == M).astype(float)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum

        # gradient = 2 * (a_rev * w) over the original index range
        grad_full = _linear_convolution(a[::-1], w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # indices to move mass between
        i_remove = np.argmax(grad)          # large gradient → increasing the max
        i_add = np.argmin(grad)            # small gradient → safest place to add

        if i_remove == i_add:
            break

        delta = step * a[i_remove]
        if delta < step_min:
            break

        cand = a.copy()
        cand[i_remove] -= delta
        cand[i_add] += delta
        # protect against tiny negatives caused by floating point errors
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        cur_val = compute_score(cand)
        if cur_val < best_val:
            best_val = cur_val
            best_a = cand.copy()
            step = min(step * 1.02, 0.5)   # accelerate when we improve
        else:
            step = max(step * step_decay, step_min)   # slow down otherwise

        a = cand

        if it % 2000 == 0:
            print(f"[pairwise] it {it:7d}  best {best_val:.6f}  step {step:.2e}")

        # early exit if step becomes minuscule
        if step <= step_min:
            break

    return best_a

# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter to shake out tiny plateaux
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size
    # -----------------------------------------------------------------
    n = 4096                     # a size that is large enough yet fast

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈150 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 660.0   # reserve most of the time for refinement
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. If baseline failed, start from a uniform distribution
    # -----------------------------------------------------------------
    if best_seq is None:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.array(best_seq, dtype=float)

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation on the full problem (≈600 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 250.0
    if time.time() < smooth_deadline:
        refined = smooth_optimize(a0, smooth_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # a few quick restarts from random Dirichlet starts
        while time.time() < smooth_deadline:
            remaining = smooth_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)

            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            # optionally enforce symmetry to explore U‑shapes
            if np.random.rand() < 0.4:
                a_start = (a_start + a_start[::-1]) * 0.5

            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Pairwise deterministic refinement (≈200 s)
    # -----------------------------------------------------------------
    pairwise_deadline = deadline - 35.0
    if best_seq is not None and time.time() < pairwise_deadline:
        a_start = np.array(best_seq, dtype=float)
        refined = pairwise_optimize(a_start, pairwise_deadline)
        val = compute_score(refined)
        print(f"[pairwise‑refine] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f} "
          f"(internal best {best_val:.6f})")
    return best_seq
