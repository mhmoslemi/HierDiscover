def propose_candidate(seed=..., budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    best_seq, best_score = None, float('inf')
    # list of n's to try
    candidate_ns = [128, 256, 512, 1024]
    # for each n, run baseline and local optimization
    for n in candidate_ns:
        if time.time() >= deadline: break
        # baseline: families and random draws
        # store top K candidates
        top = []
        # define try_candidate
        def try_candidate(vec):
            nonlocal top, best_score, best_seq
            if time.time() >= deadline: return
            # ensure nonneg
            vec = np.maximum(vec, 0)
            # normalize sum to 1
            s = vec.sum()
            if s < 1e-12: return
            vec = vec / s
            sc = compute_score(vec)  # compute max conv * 2n
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
            # keep top K by sc
            if len(top) < 30:
                top.append((sc, vec.copy()))
                top.sort(key=lambda x: x[0])
            elif sc < top[-1][0]:
                top[-1] = (sc, vec.copy())
                top.sort(key=lambda x: x[0])
        # family sweeps
        # powerlaw
        for p in np.linspace(0.1, 5.0, 90):
            try_candidate(generate_powerlaw(n, p))
        # exponential
        for lam in np.linspace(0.001, 0.5, 100):
            try_candidate(generate_exponential(n, lam))
        # beta
        for alpha in np.linspace(0.05, 3.0, 60):
            for beta in np.linspace(0.05, 3.0, 60):
                try_candidate(generate_beta(n, alpha, beta))
        # gaussian sigma
        for sigma in np.linspace(0.1*n, 0.5*n, 20):
            try_candidate(generate_gaussian(n, sigma))
        # sine power
        for p in np.linspace(0.1, 8.0, 40):
            try_candidate(generate_sine_power(n, p))
        # random Dirichlet draws (various concentrations)
        concs = [0.01,0.05,0.1,0.2,0.5,1,2,5]
        for conc in concs:
            for _ in range(2000 // len(concs)):
                a = np.random.dirichlet([conc]*n)
                try_candidate(a)

        # also add mixtures
        for _ in range(200):
            a1 = generate_powerlaw(n, np.random.uniform(0.1,5))
            a2 = generate_beta(n, 10**np.random.uniform(-2,0.5), 10**np.random.uniform(-2,0.5))
            lam = np.random.rand()
            try_candidate(mixture(a1, a2, lam))

        # after baseline, take top candidates and refine with Adam
        # allocate some time for this n
        n_deadline = deadline - (budget_s - (time.time() - start_time)) * (0.3) # rough ignore

        # Actually compute remaining time
        remaining = deadline - time.time()
        if remaining < 30: break
        # allocate part for Adam
        adam_time = min(remaining * 0.5, 300)   # at most 300 secs per n
        # split among top candidates
        per_candidate_deadline = time.time() + adam_time / max(1, len(top))
        for idx, (sc, vec) in enumerate(top):
            if time.time() >= deadline: break
            # run Adam starting from vec
            local_deadline = time.time() + per_candidate_deadline - time.time()
            a_ref = adam_optimize(vec, local_deadline)
            sc_ref = compute_score(a_ref)
            if sc_ref < best_score:
                best_score, best_seq = sc_ref, a_ref.tolist()
        # polish best for this n
        # allocate some time left
        if time.time() < deadline:
            a0 = np.array(best_seq, dtype=float) if best_seq is not None else np.full(n,1/n)
            a_polished = subgradient_polish(a0, deadline, step0=0.2)
            sc_pol = compute_score(a_polished)
            if sc_pol < best_score:
                best_score, best_seq = sc_pol, a_polished.tolist()
        print(f"[n={n}] best score so far = {best_score:.6f}")
    # final check: if best_seq None, fallback uniform n=128
    if best_seq is None:
        best_seq = [1.0/128]*128
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq

"""
Search for a non‑negative step‑function (probability) vector that
minimises `evaluate_sequence`.  The strategy combines:

1. A dense, time‑bounded sweep of several analytic families
   (power‑law, exponential, beta, Gaussian, sine‑power) together
   with many Dirichlet draws.  The best K candidates are retained.

2. For each retained candidate a short Adam optimisation is run on a
   smooth‑max surrogate of the true objective (no flattening term).

3. An optional FFT‑magnitude projection step (iteratively forces the
   DFT magnitude to be constant, which would make the self‑convolution
   flat) is applied to the Adam output.

4. Finally a projected sub‑gradient polish (exact max‑objective) is
   performed.

All phases respect the global 1000 s wall‑clock limit and continuously
track the best sequence seen.  The returned list is a plain Python
list of floats (non‑negative, sum ≈ 1).  The official evaluation
function `evaluate_sequence` is called only once at the end.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of ½·∑ (a*a - const)^2 or of the smooth‑max.
    It is 2·(a ⋆ vec) where ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Exact score (used internally – sum of a is assumed to be 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(conv(a,a)) (a sums to 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the global baseline search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Baseline exploration for a single length n
# ----------------------------------------------------------------------


def baseline_search_one_n(
    n: int,
    deadline: float,
    max_keep: int = 30,
    random_draws: int = 20000,
) -> list[np.ndarray]:
    """Heavily sampled baseline search; returns the best few vectors."""
    best = []   # list of (score, vector)

    def record(vec: np.ndarray):
        if time.time() >= deadline:
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- power‑law ----------------------------------------------------
    for p in np.linspace(0.1, 5.0, 81):
        record(generate_powerlaw(n, p))

    # ----- exponential --------------------------------------------------
    for lam in np.linspace(0.001, 0.5, 120):
        record(generate_exponential(n, lam))

    # ----- beta (grid) --------------------------------------------------
    al_vals = np.linspace(0.05, 3.0, 31)
    be_vals = np.linspace(0.05, 3.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            record(generate_beta(n, alpha, beta))

    # ----- random beta draws --------------------------------------------
    for _ in range(300):
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        record(generate_beta(n, alpha, beta))

    # ----- Gaussian sweep -----------------------------------------------
    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        record(generate_gaussian(n, sigma))

    # ----- sine‑power ----------------------------------------------------
    for p in np.linspace(0.1, 8.0, 35):
        record(generate_sine_power(n, p))

    # ----- mixtures -------------------------------------------------------
    for _ in range(200):
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        record(mixture(a, b, lam))

    # ----- Dirichlet draws (wide concentration range) --------------------
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            record(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate (λ = 0)
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-7,
    tau_decay: float = 0.94,
    lr_start: float = 0.15,
    lr_decay: float = 0.97,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on L(a) = τ·log ∑ exp(conv/τ).  No L2‑flattening term.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    size_fft = 1 << ((3 * n - 3).bit_length())

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # schedule updates
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        conv = full_convolution(a)
        shift = conv.max()
        # stable soft‑max
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()  # soft‑max weights, sum to 1

        # gradient of the smooth‑max term
        grad = correlation_grad(a, w, size_fft)

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to help escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Optional FFT‑magnitude projection (moves towards flat convolution)
# ----------------------------------------------------------------------


def magnitude_projection(a: np.ndarray, iters: int, deadline: float) -> np.ndarray:
    """
    Project onto the set of vectors whose DFT magnitude is (approximately) constant.
    Each iteration replaces the magnitude by its mean while preserving phase,
    then projects back onto the simplex.
    """
    n = a.shape[0]
    for _ in range(iters):
        if time.time() >= deadline:
            break
        A = np.fft.rfft(a)
        target = np.mean(np.abs(A))
        phases = np.angle(A)
        A_new = target * np.exp(1j * phases)
        a = np.fft.irfft(A_new, n=n)
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.33) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # -----------------------------------------------------------------
    # Global best tracking
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # Optional fallback – use any previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Choose a set of problem sizes – smaller sizes are quicker to optimise
    # -----------------------------------------------------------------
    # The factor 2·n in the objective means that reducing n can help,
    # but we still need enough degrees of freedom to flatten the convolution.
    size_list = [256, 512, 1024]   # tried in order of increasing difficulty

    for n in size_list:
        if time.time() >= deadline:
            break
        print(f"\n--- Searching with length n={n} ---")
        # ---- 1. Baseline exploration (≈ 150 s or remaining/3) --------------
        remaining = deadline - time.time()
        baseline_deadline = time.time() + min(150.0, remaining * 0.35)
        candidates = baseline_search_one_n(
            n,
            baseline_deadline,
            max_keep=30,
            random_draws=20000,
        )
        # evaluate all baseline candidates and keep the best globally
        for vec in candidates:
            cur_val = compute_score(vec)
            if cur_val < best_val:
                best_val, best_seq = cur_val, vec.tolist()
        print(f"[baseline] best score so far = {best_val:.6f}")

        # ---- 2. Adam refinement on the top‑K candidates --------------------
        remaining = deadline - time.time()
        if remaining <= 30.0:
            break  # not enough time for further refinement

        # allocate half of the remaining time to Adam runs
        adam_budget = remaining * 0.5
        per_candidate_budget = max(5.0, adam_budget / max(1, len(candidates)))
        for idx, vec in enumerate(candidates):
            if time.time() >= deadline - 30.0:   # reserve ~30 s for polishing
                break
            local_deadline = time.time() + per_candidate_budget
            a_ref = adam_optimize(vec, local_deadline)
            # optional magnitude‑projection step (fast, 5 iterations)
            proj_deadline = time.time() + 5.0
            a_ref = magnitude_projection(a_ref, iters=5, deadline=proj_deadline)

            cur_val = compute_score(a_ref)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a_ref.tolist()
                print(f"[adam n={n} idx={idx}] improved to {cur_val:.6f}")

        # ---- 3. Sub‑gradient polish (use whatever time remains) ----------
        remaining = deadline - time.time()
        if best_seq is not None and remaining > 20.0:
            a0 = np.array(best_seq, dtype=float)
            a_polished = subgradient_polish(a0, deadline - 5.0, step0=0.30)
            cur_val = compute_score(a_polished)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a_polished.tolist()
                print(f"[polish n={n}] new best = {cur_val:.6f}")

        print(f"[n={n}] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – if nothing discovered, fall back to a uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
