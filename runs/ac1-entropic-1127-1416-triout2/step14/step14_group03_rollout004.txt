"""
Optimisation of a non‑negative step‑function (vector of heights) to minimise
the black‑box objective ``evaluate_sequence``.

The search proceeds in four stages, each respecting the overall 1000 s
wall‑clock budget:

1. **Baseline sweep** – a cheap global exploration over several parametric
   families (power‑law, exponential, beta, Gaussian) and many random
   Dirichlet draws.  The best few candidates (projected on the simplex) are
   retained.

2. **Upscaling** – the current best candidate is linearly interpolated to a
   larger dimension (here 4096).  Upscaling gives the optimiser more freedom
   to flatten the autocorrelation peak.

3. **Smooth Adam optimisation** – we minimise a smooth approximation of
   ``max(conv(a,a))`` using a soft‑max temperature τ.  The variables are
   the log‑softmax parameters ``x`` (so that ``a = softmax(x)``), which
   automatically enforce non‑negativity and the simplex constraint.
   Adam with a decaying learning‑rate and a geometric cooling schedule for τ
   performs the main optimisation.

4. **Deterministic “flatten‑max” polish** – after the Adam stage we
   repeatedly locate the index of the current convolution maximum and move a
   small amount of probability mass from the indices that contribute most
   to that peak to indices that contribute least.  This greedy hill‑climb
   (with a simple step‑size adaptation) often reduces the maximum autocorrelation
   a little further, pushing the objective below the previous best.

The function ``propose_candidate`` returns the best sequence found as a
plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via FFT (real)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a)=1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(a*a) for a simplex‑normalised a."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return vectors normalised to sum 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Helper for the baseline stage
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate and keep it if good."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0:
        return
    vec = vec / total                     # projection onto simplex
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """
    Sweep a collection of parametric families and many Dirichlet draws.
    Return up to ``max_keep`` best vectors (as ndarrays).
    """
    best = []   # list of (score, vector)

    # 1) power‑law
    for p in np.linspace(0.1, 6.0, 300):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # 2) exponential
    for lam in np.linspace(0.0001, 0.2, 300):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # 3) beta
    al_vals = np.linspace(0.2, 8.0, 30)
    be_vals = np.linspace(0.2, 8.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # 4) Gaussian
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # 5) Dirichlet draws (both raw and sorted)
    concs = [0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            # monotone version – sort descending
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linearly interpolate ``vec`` (length n) to length ``new_n`` and project
    back onto the probability simplex.
    """
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


# ----------------------------------------------------------------------
# Smoothed Adam optimisation (soft‑max temperature)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.2,
                    init_tau_factor: float = 12.0,
                    min_tau: float = 1e-12,
                    tau_decay: float = 0.9999) -> np.ndarray:
    """
    Minimise a smoothed version of ``max(conv(a,a))`` using Adam on the
    log‑softmax parameters.  Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise log‑softmax parameters (add tiny epsilon for safety)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution -----
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- soft‑max weights over convolution entries -----
        if tau < 1e-14:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient w.r.t a (g_i = 2·∑_k w_k·a_{k-i}) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the soft‑max (x‑space) -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature -----
        tau = max(min_tau, tau * tau_decay)

        # ----- periodic exact evaluation -----
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:8d}   score {val:.6f}   τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Deterministic “flatten‑max” hill‑climb
# ----------------------------------------------------------------------


def flatten_max_convolution(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy polish that repeatedly moves a small amount of probability mass
    from indices that contribute most to the current convolution maximum
    to those that contribute least.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_frac = 0.05          # fraction of source mass moved each step
    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        k = int(conv.argmax())

        # Gradient of the (non‑smoothed) max w.r.t. a at index k:
        # g_i = 2 * a_{k-i} for valid i, otherwise 0.
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k - idx]

        # choose source (largest g) and target (smallest g)
        i_src = int(g.argmax())
        i_tgt = int(g.argmin())
        if i_src == i_tgt:
            # nothing to do – break to avoid endless loop
            break
        if a[i_src] <= 1e-15:
            # no mass to move – abort this iteration
            break

        delta = step_frac * a[i_src]
        delta = min(delta, a[i_src])

        # perform tentative move
        a[i_src] -= delta
        a[i_tgt] += delta

        # re‑normalise (should still sum to 1, but guard against numerical drift)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0:
            a = best_a.copy()
            continue
        a /= total

        val = compute_score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            # successful move – try a slightly larger step
            step_frac = min(step_frac * 1.15, 0.5)
        else:
            # revert move and shrink step
            a[i_src] += delta
            a[i_tgt] -= delta
            a = np.maximum(a, 0.0)
            a /= a.sum()
            step_frac = max(step_frac * 0.70, 1e-5)

        # occasional random jitter to escape tiny plateaux
        if it % 5000 == 0:
            jitter_idx = np.random.randint(n)
            other_idx = np.random.randint(n)
            if jitter_idx != other_idx and a[jitter_idx] > 1e-10:
                jitter_delta = step_frac * a[jitter_idx]
                a[jitter_idx] -= jitter_delta
                a[other_idx] += jitter_delta
                a = np.maximum(a, 0.0)
                a /= a.sum()
    print(f"[flatten] finished after {it} steps, best score {best_val:.6f}")
    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem dimensions
    # -----------------------------------------------------------------
    n_small = 2048
    n_big = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 740.0   # reserve most time for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n_small, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Upsample to the large dimension
    # -----------------------------------------------------------------
    if best_seq is None:
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)

    a0_big = upsample(a0_small, n_big)

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation on the large problem (≈ 600 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if time.time() < smooth_deadline:
        # primary run from the up‑sampled best
        refined = smooth_optimize(a0_big, smooth_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional random restarts while time permits
        while time.time() < smooth_deadline:
            remaining = smooth_deadline - time.time()
            if remaining < 30.0:
                break
            sub_deadline = time.time() + min(30.0, remaining)

            # random Dirichlet start
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0])
            a_start = np.random.dirichlet(np.full(n_big, conc))
            # occasionally enforce monotonicity
            if np.random.rand() < 0.4:
                a_start = np.sort(a_start)[::-1]

            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Deterministic flatten‑max polish (≈ remaining time)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 5.0
    if best_seq is not None and time.time() < polish_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = flatten_max_convolution(a_start, polish_deadline)
        val = compute_score(final)
        print(f"[flatten‑polish] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
