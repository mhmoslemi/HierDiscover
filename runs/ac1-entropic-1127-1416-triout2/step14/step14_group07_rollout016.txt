"""
search_step_function_v4.py

A fast, robust optimiser for the “evaluate_sequence” benchmark.

Algorithm outline
-----------------
1.  **Quick parametric sweep** – a few simple families (reverse power‑law,
    geometric, and a mixture of a reverse power‑law with a uniform background)
    are scanned on a fine grid.  The best vector found is kept as a seed.

2.  **Smoothed gradient descent** – the nondifferentiable “max”
    in the objective is replaced by a smooth log‑sum‑exp approximation
    with a temperature β.  The gradient of the smoothed objective is
    ∇ ≈ 2 corr(a, softmax(β·conv(a,a))) and a projected gradient step on the
    probability simplex is performed.  β is raised gradually so the method
    first moves quickly on a smooth landscape and then focuses on the true
    max.  This step dramatically reduces the peak of the auto‑convolution.

3.  **Short stochastic hill‑climb** – after the gradient phase a very
    fast greedy hill‑climb (mass transfer between two coordinates) refines the
    solution.

The whole routine respects the 1000 s time limit, keeps only the best
sequence and finally returns it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‐algebra helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT (real valued)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Small parametric families (used for the quick sweep)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n−i)^p,  p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n, … ,1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^i,   0 < q < 1."""
    a = q ** np.arange(n, dtype=np.float64)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Smoothed gradient descent on the max‑convolution objective
# ----------------------------------------------------------------------
def smoothed_descent(a0: np.ndarray, deadline: float,
                     step0: float = 0.15,
                     beta0: float = 30.0,
                     beta_max: float = 800.0,
                     beta_multiplier: float = 1.5,
                     beta_step: int = 5000) -> np.ndarray:
    """
    Projected gradient descent on the smoothed objective
        F_β(a) = (1/β)·log Σ_k exp(β·conv_k(a))
    The gradient is 2·corr(a, softmax(β·conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    beta = beta0
    while time.time() < deadline:
        # current convolution and (smooth) weights
        conv = full_convolution(a)                     # length 2n‑1
        max_conv = conv.max()
        # soft‑max weights (numerically stable)
        shifted = conv - max_conv                       # ≤ 0
        exp_term = np.exp(beta * shifted)              # max weight = 1
        w = exp_term / exp_term.sum()                  # Σ w = 1

        # gradient = 2·corr(a, w)  (valid mode → length n)
        grad = 2.0 * np.correlate(a, w, mode='valid')

        # step‑size schedule
        t += 1
        step = step0 / math.sqrt(t)

        # projected gradient step
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        # evaluate the *real* objective (no smoothing)
        new_score = compute_score(a_new)

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

        # accept only if we do not increase the true max‑convolution
        if new_score <= compute_score(a):
            a = a_new

        # raise the temperature periodically
        if t % beta_step == 0 and beta < beta_max:
            beta = min(beta * beta_multiplier, beta_max)

    return best_a


# ----------------------------------------------------------------------
# Very fast hill‑climb (mass transfer between two coordinates)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy hill‑climb; moves a small fraction of mass from i to j."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]                     # move a share of i → j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            # a little random jitter to escape a plateau
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
