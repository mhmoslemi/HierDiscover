"""
search_step_function_v4.py

A new optimiser for ``evaluate_sequence`` that expands the
parameter‐space dramatically and adds a lightweight simulated‑annealing
stage after a long sub‑gradient polishing phase.

Key ideas
---------
* Exhaustive but cheap grid searches over several low‑dimensional
  families (reverse‑power, two‑parameter “multi‑power”, power‑times‑exp,
  beta, sigmoid).  The grids are considerably finer than the previous
  version and cover a larger region (including negative exponents).
* A very long projected sub‑gradient polishing (the objective is
  ``2·n·max(conv)`` on the probability simplex).  The step‑size schedule
  is slightly more aggressive than before.
* A simple simulated‑annealing loop that repeatedly moves a small
  amount of mass between two random indices and accepts the move with
  probability ``exp(-(Δscore)/T)``.  Temperature decays exponentially,
  allowing the search to escape shallow local minima.
* Final short hill‑climb (as before) to polish the very best point.
* All helpers are top‑level; the only entry point required by the
  harness is ``propose_candidate``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("project_to_simplex expects a 1‑D array")
    # Fast implementation from https://arxiv.org/pdf/1101.6081.pdf
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a * a using an FFT (real input)."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for a fast FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Objective for a vector on the simplex (Σa = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors (all return a *normalized* np.ndarray)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p   (p may be negative)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)        # n, …, 1
    a = idx ** p
    tot = a.sum()
    if tot == 0.0:
        return None
    a /= tot
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p · (i+1)^q   (both exponents ≥ 0)."""
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    tot = a.sum()
    if tot == 0.0:
        return None
    a /= tot
    return a


def power_exp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p · exp(‑lam·i)   with lam ≥ 0."""
    i = np.arange(n, dtype=np.float64)
    left = (n - i) ** p
    a = left * np.exp(-lam * i)
    tot = a.sum()
    if tot == 0.0:
        return None
    a /= tot
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^α (1‑x_i)^β   with x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    tot = a.sum()
    if tot == 0.0:
        return None
    a /= tot
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i‑c)/σ))."""
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    tot = a.sum()
    if tot == 0.0:
        return None
    a /= tot
    return a

# Broad parametric sweep (≈300 s)

def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Explore many families and keep the best vector discovered."""
    best_vec = None
    best_val = float('inf')

    def try_candidate(a: np.ndarray):
        nonlocal best_vec, best_val
        if a is None:
            return
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ------------------------------------------------------------------
    # 1. Reverse power‑law, fine grid (including slight negatives)
    # ------------------------------------------------------------------
    print("[sweep] reverse_power_law (fine grid)", flush=True)
    for p in np.linspace(-1.0, 4.0, 501):          # step ≈0.01
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, p)
        try_candidate(a)

    # ------------------------------------------------------------------
    # 2. Multi‑power (2‑parameter)
    # ------------------------------------------------------------------
    print("[sweep] multi_power (fine grid)", flush=True)
    for p in np.linspace(0.0, 3.0, 61):
        for q in np.linspace(0.0, 3.0, 61):
            if time.time() >= deadline:
                return best_vec
            a = multi_power(n, p, q)
            try_candidate(a)

    # ------------------------------------------------------------------
    # 3. Power × exponential
    # ------------------------------------------------------------------
    print("[sweep] power_exp", flush=True)
    for p in np.linspace(0.0, 3.0, 31):
        for lam in np.linspace(0.0, 0.01, 21):
            if time.time() >= deadline:
                return best_vec
            a = power_exp(n, p, lam)
            try_candidate(a)

    # ------------------------------------------------------------------
    # 4. Sigmoid family (wider sigma)
    # ------------------------------------------------------------------
    print("[sweep] sigmoid_shape (wide)", flush=True)
    for centre_frac in np.linspace(0.2, 0.8, 13):
        for sigma_frac in np.linspace(0.005, 0.25, 25):
            if time.time() >= deadline:
                return best_vec
            a = sigmoid_shape(n, centre_frac, sigma_frac)
            try_candidate(a)

    # ------------------------------------------------------------------
    # 5. Beta family (larger range)
    # ------------------------------------------------------------------
    print("[sweep] beta_shape", flush=True)
    alphas = np.linspace(-1.0, 8.0, 46)
    betas = np.linspace(-1.0, 8.0, 46)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            try_candidate(a)

    # ------------------------------------------------------------------
    # 6. Random Dirichlet samples (cheap, high‑dim exploration)
    # ------------------------------------------------------------------
    print("[sweep] random Dirichlet", flush=True)
    for _ in range(4000):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        try_candidate(a)

    # ------------------------------------------------------------------
    # 7. Random convex mixtures of a few strong bases
    # ------------------------------------------------------------------
    print("[sweep] random mixtures", flush=True)
    base_uniform = np.full(n, 1.0 / n, dtype=np.float64)
    base_rev = reverse_power_law(n, 2.0)
    base_sig = sigmoid_shape(n, 0.5, 0.07)
    basics = [base_uniform, base_rev, base_sig]
    for _ in range(2000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        try_candidate(a)

    return best_vec


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (more aggressive step‑size schedule)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-15)[0]

        # sub‑gradient of the max‑convolution w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        # small jitter every few thousand steps
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (lightweight global search)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        T0: float = 0.02, decay: float = 0.998) -> np.ndarray:
    """Move small amounts of mass between random indices, accept with SA."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    T = T0
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a fraction of mass from i to j
        delta = T * a[i]            # magnitude scales with temperature
        cand = a.copy()
        cand[i] = max(cand[i] - delta, 0.0)
        cand[j] = cand[j] + delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand.copy()
            a = cand
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_score - best_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = cand

        T *= decay
        if T < 1e-6:
            T = T0   # occasional reheating to escape very deep wells

    return best_a


# ----------------------------------------------------------------------
# Short greedy hill‑climb (as a final polish)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.025
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            # restart from the global best with tiny noise
            jitter = np.random.randn(n) * 0.0006
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
