"""
Hybrid search for a low‑max‑convolution step function.

The algorithm consists of:
1. A fast global baseline that draws from many analytic families
   (power‑law, exponential, beta, Gaussian, sine‑power) and many
   Dirichlet samples.  The best few candidates are kept.
2. Multi‑start Adam optimisation with a smooth‑max surrogate.
   Several hyper‑parameter settings (λ, τ_start) are explored
   in parallel.  The best vector from each run is retained.
3. A projected sub‑gradient polish that directly minimises the
   true max‑convolution objective.
4. A final simulated‑annealing hill‑climb to escape possible
   local minima.

All vectors are kept on the probability simplex (sum = 1, non‑negative)
so that the final evaluation coincides with
    2·n·max(conv(a,a)).
The code respects a 1000 s wall‑clock limit and continuously
keeps the best sequence seen so far.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Convolution utilities (FFT for speed)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of ½·||a⋆vec||² w.r.t. a, i.e. 2·(a ⋆ vec).
    `vec` must have length 2·len(a)-1.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]

# ----------------------------------------------------------------------
# Scoring (exact objective for a normalised vector)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)), assuming sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Parametric families for the baseline search
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)

def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)

def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)

def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)

def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)

def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)

# ----------------------------------------------------------------------
# Global baseline exploration (cheap, many random draws)
# ----------------------------------------------------------------------
def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 24,
    random_draws: int = 30000,
) -> list[np.ndarray]:
    """
    Generate a pool of promising candidates from analytic families
    and Dirichlet draws.  The best `max_keep` vectors (by exact score)
    are returned.
    """
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- deterministic sweeps of analytic families -----
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    al_vals = np.linspace(0.02, 2.0, 31)
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    for _ in range(200):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # ----- mixtures of families -----
    for _ in range(300):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(n,
                           10 ** np.random.uniform(-2, 0.3),
                           10 ** np.random.uniform(-2, 0.3))
        lam_mix = np.random.rand()
        try_candidate(mixture(a, b, lam_mix))

    # ----- Dirichlet draws (various concentration) -----
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet([conc] * n)
            try_candidate(vec)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]

# ----------------------------------------------------------------------
# Combined Adam optimiser (smooth‑max + L2 flattening)
# ----------------------------------------------------------------------
def combined_adam(
    a0: np.ndarray,
    deadline: float,
    lam: float = 0.15,
    tau_start: float = 0.30,
    tau_end: float = 1e-7,
    tau_decay: float = 0.94,
    lr_start: float = 0.15,
    lr_decay: float = 0.97,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on the surrogate
        L = (1-λ)·τ·log∑exp(conv/τ) + λ·½·||conv - uniform||² .
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m
    size_fft = 1 << ((3 * n - 3).bit_length())

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # schedule updates
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # convolution and smooth‑max
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()          # soft‑max weights

        # flattening error
        err = conv - target

        # gradients
        grad_max = correlation_grad(a, w, size_fft)
        grad_err = correlation_grad(a, err, size_fft)

        grad = (1.0 - lam) * grad_max + lam * grad_err

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter for escaping plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max‑objective)
# ----------------------------------------------------------------------
def subgradient_polish(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.33,
) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum over all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------
def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-7,
    cooling_factor: float = 0.9995,
    step_scale_init: float = 0.05,
    step_scale_min: float = 5e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0
    n = a.shape[0]

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < math.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 5.0

    # problem size – larger n gives a tighter asymptotic bound
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback: previously discovered construction if available
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈150 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 150.0)
    if time.time() < baseline_deadline:
        baselines = baseline_search(
            n, baseline_deadline, max_keep=24, random_draws=30000
        )
        for vec in baselines:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Multi‑start Adam optimisation with hyper‑parameter sweep
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    # keep about 300 s for Adam phase, leaving room for polish & anneal
    adam_phase_deadline = time.time() + min(300.0, remaining * 0.5)
    if best_seq is not None and time.time() < adam_phase_deadline:
        start_vectors = [
            np.array(best_seq, dtype=float),
            np.full(n, 1.0 / n),
        ]
        # a few random starts (simple Dirichlet draws)
        for _ in range(4):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        # hyper‑parameter grid
        lam_grid = [0.05, 0.15, 0.30, 0.45, 0.60]
        tau_start_grid = [0.6, 0.3, 0.15]
        # allocate time roughly equally among combos
        combos = [(lam, tau) for lam in lam_grid for tau in tau_start_grid]
        combo_idx = 0
        while time.time() < adam_phase_deadline and combo_idx < len(combos):
            lam, tau_start = combos[combo_idx]
            combo_idx += 1
            # split remaining time across remaining combos
            time_per_combo = (adam_phase_deadline - time.time()) / (len(combos) - combo_idx + 1)

                    tau_decay=0.95,
                    lr_start=0.20,
                    lr_decay=0.96,
                )
                sc = compute_score(a_ref)
                if sc < best_val:
                    best_val, best_seq = sc, a_ref.tolist()
                print(f"[adam lam={lam:.2f} τ0={tau_start:.2f}] score = {sc:.6f}")

        print(f"[adam] best after phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.33)
        sc = compute_score(a_polished)
        print(f"[polish] score = {sc:.6f}")

            a_start,

        )

    return best_seq
