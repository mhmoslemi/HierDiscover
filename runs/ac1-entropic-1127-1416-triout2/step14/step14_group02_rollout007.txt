"""
Improved search for a low‑max‑convolution step‑function.

Main ideas
----------
* Richer baseline families – include symmetric power‑law and symmetric
  exponential shapes that concentrate mass at the edges (the region that
  helps to flatten the self‑convolution).
* A long “Adam on a smooth‑max + flattening” phase with a *schedule*:
      – τ (temperature) is annealed from 0.5 to ~1e‑10,
      – the weight λ on the flatten‑to‑uniform term is decayed from 0.5
        to 0, so the optimiser first flattens the convolution and then
        polishes the max‑value.
* After the Adam phase a projected sub‑gradient polish and a final
  simulated‑annealing hill‑climb are executed.
* All phases respect the global 1000 s wall‑clock limit and constantly
  keep the best vector seen so far.

The returned sequence is a list of non‑negative floats whose sum is 1,
which makes the official ``evaluate_sequence`` score equal to
``2 * n * max(conv)``.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, zero‑padding)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full) using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Compute 2·(a ⋆ vec) where ⋆ denotes cross‑correlation.
    ``vec`` length must be 2·len(a)-1.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Scoring (exact objective for a unit‑sum vector)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact score = 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (including new edge‑heavy families)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_symmetric_exponential(n: int, lam: float) -> np.ndarray:
    """High at the edges, decaying towards the centre."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)          # distance to the nearest edge
    a = np.exp(-lam * dist.astype(float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_symmetric_powerlaw(n: int, p: float) -> np.ndarray:
    idx = np.arange(n)
    dist = np.minimum(idx + 1, n - idx)        # 1‑based distance to the nearer edge
    a = dist.astype(float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global baseline exploration (cheap, many random draws)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 12000,
) -> List[np.ndarray]:
    """
    Explore many analytic families and Dirichlet draws.
    Returns the best ``max_keep`` vectors (lowest score) found.
    """
    best: List[tuple[float, np.ndarray]] = []

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- families ----------------------------------------------------
    for p in np.linspace(0.1, 5.0, 71):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.5, 80):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    for sigma in np.linspace(0.1 * n, 0.45 * n, 25):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    for p in np.linspace(0.1, 8.0, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # new edge‑heavy families
    for lam in np.linspace(0.001, 0.2, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_symmetric_exponential(n, lam))

    for p in np.linspace(0.1, 4.0, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_symmetric_powerlaw(n, p))

    # beta family (including random draws)
    al_vals = np.linspace(0.02, 2.0, 21)
    be_vals = np.linspace(0.02, 2.0, 21)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    for _ in range(200):
        if time.time() >= deadline:
            break
        a = np.random.beta(0.5, 0.5, size=n)   # a rough beta draw
        try_candidate(a)

    # mixtures of families
    for _ in range(150):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_symmetric_exponential(n, np.random.uniform(0.001, 0.2))
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # Dirichlet draws spanning a wide concentration range
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max + flattening with a decaying λ
# ----------------------------------------------------------------------


def adam_max_flatten(
    a0: np.ndarray,
    deadline: float,
    lam_start: float = 0.5,
    lam_end: float = 0.0,
    lam_decay: float = 0.9995,
    tau_start: float = 0.5,
    tau_end: float = 1e-10,
    tau_decay: float = 0.985,
    lr_start: float = 0.5,
    lr_decay: float = 0.995,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on the blended surrogate
        L = (1‑λ)·τ·log‑sum‑exp(conv/τ) + λ·½·∥conv‑target∥² .
    λ is annealed to zero so that the final regime focuses on the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m                       # uniform target for the convolution
    size_fft = 1 << ((3 * n - 3).bit_length())

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    lam = lam_start
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # -------------------------------------------------------------
        # convolution and smooth‑max distribution
        # -------------------------------------------------------------
        conv = full_convolution(a)                     # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)        # stable log‑sum‑exp
        w = exp_vals / exp_vals.sum()                  # soft‑max weights

        # -------------------------------------------------------------
        # flattening error term
        # -------------------------------------------------------------
        err = conv - target

        # -------------------------------------------------------------
        # gradients
        # -------------------------------------------------------------
        grad_max = correlation_grad(a, w, size_fft)
        grad_err = correlation_grad(a, err, size_fft)

        grad = (1.0 - lam) * grad_max + lam * grad_err

        # -------------------------------------------------------------
        # Adam update
        # -------------------------------------------------------------
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)

        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # -------------------------------------------------------------
        # bookkeeping
        # -------------------------------------------------------------
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # -------------------------------------------------------------
        # schedule updates (every few hundred iterations)
        # -------------------------------------------------------------
        if t % 200 == 0:
            if tau > tau_end:
                tau = max(tau_end, tau * tau_decay)
            if lr > 1e-6:
                lr = max(1e-6, lr * lr_decay)
        if t % 400 == 0:
            if lam > lam_end:
                lam = max(lam_end, lam * lam_decay)

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-7,
    cooling_factor: float = 0.9995,
    step_scale_init: float = 0.05,
    step_scale_min: float = 5e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin so the harness has time to collect the result
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – a larger n gives a tighter theoretical bound.
    # -----------------------------------------------------------------
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous best construction)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 120 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 120.0)
    if time.time() < baseline_deadline:
        cand_vecs = baseline_search(
            n=n,
            deadline=baseline_deadline,
            max_keep=12,
            random_draws=18000,
        )
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Long Adam optimisation (smooth‑max + flattening)
    # -----------------------------------------------------------------
    # Reserve most of the remaining budget for this phase,
    # but keep ~250 s for polishing/annealing.
    remaining = deadline - time.time()
    polish_reserve = 250.0
    adam_deadline = time.time() + max(0.0, remaining - polish_reserve)

    if adam_deadline > time.time() + 30.0 and best_seq is not None:
        # Prepare a few promising starts: the current best + a few random draws
        start_vectors = [
            np.array(best_seq, dtype=float),
            np.full(n, 1.0 / n),
        ]
        for _ in range(3):
            start_vectors.append(project_to_simplex(np.random.rand(n)))
        for idx, start in enumerate(start_vectors):
            if time.time() >= adam_deadline:
                break
            local_deadline = min(adam_deadline, time.time() + (adam_deadline - time.time()) / max(1, len(start_vectors) - idx))
            a_ref = adam_max_flatten(start, local_deadline,
                                     lam_start=0.5, lam_end=0.0,
                                     tau_start=0.5, tau_end=1e-10,
                                     lr_start=0.5)
            val = compute_score(a_ref)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_ref.tolist()
        print(f"[adam] best after this phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 110.0   # keep ~100 s for SA
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.33)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (final ≈ 100 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline,
                                      init_temp=2e-3,
                                      final_temp=1e-7,
                                      cooling_factor=0.9995,
                                      step_scale_init=0.05)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report with the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
