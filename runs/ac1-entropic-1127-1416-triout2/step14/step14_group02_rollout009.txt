"""
Step‑function max‑convolution optimiser (≈ 1000 s budget).

Main ideas
---------
* Work with a fairly large dimension (n = 8192) – the asymptotic
  constant is approached faster for longer vectors.
* Global baseline: many analytic families (power‑law, exponential,
  beta, Gaussian, sine‑power) plus a large pool of Dirichlet draws.
* Fast “smooth‑max + Adam” optimiser – several restarts with
  different λ (weight of a uniform‑flattening term) and an aggressive
  τ‑annealing schedule.
* Targeted pair‑wise mass‑transfer hill‑climb:
  – compute the exact sub‑gradient of the max‑convolution;
  – move a small fraction of probability from the most “responsible”
    index to the least responsible one;
  – Metropolis acceptance with a slowly cooling temperature.
* A short projected sub‑gradient polishing phase (as in the previous
  solution) to clean up any remaining slack.
* All phases respect the wall‑clock deadline; the best sequence seen
  is always kept and finally evaluated with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, zero‑padding)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using the real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of ½·∑_k (∑_i a_i a_{k-i} - const)^2 _or_ of the smooth‑max.
    It is 2·(a ⋆ vec) where ⋆ denotes cross‑correlation.
    `vec` length must be 2·len(a)-1.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Scoring (exact objective for a normalised vector)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) (vector assumed to sum≈1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used in the global baseline search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global baseline exploration (cheap random draws + families)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 25000,
) -> list[np.ndarray]:
    """Explore many analytic families and a large pool of Dirichlet draws."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- power‑law sweep ----
    for p in np.linspace(0.05, 5.0, 101):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ---- exponential sweep ----
    for lam in np.linspace(5e-4, 0.6, 130):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ---- beta grid ----
    al_vals = np.linspace(0.02, 2.0, 33)
    be_vals = np.linspace(0.02, 2.0, 33)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ---- random beta draws ----
    for _ in range(300):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    # ---- Gaussian sweep ----
    for sigma in np.linspace(0.07 * n, 0.45 * n, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ---- sine‑power sweep ----
    for p in np.linspace(0.1, 8.0, 45):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # ---- mixtures of families ----
    for _ in range(300):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # ---- Dirichlet draws (wide concentration range) ----
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Combined Adam optimiser (smooth‑max + optional flattening)
# ----------------------------------------------------------------------


def combined_adam(
    a0: np.ndarray,
    deadline: float,
    lam: float = 0.15,
    tau_start: float = 0.5,
    tau_end: float = 1e-8,
    tau_decay: float = 0.92,
    lr_start: float = 0.20,
    lr_decay: float = 0.96,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on a blended surrogate:
        L = (1‑λ)·τ·log‑sum‑exp(conv/τ) + λ·½·‖conv‑target‖²
    The first term drives max down, the second flattens the whole
    convolution towards a uniform target.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m

    size_fft = 1 << ((3 * n - 3).bit_length())

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # ----- schedule -----
        if t % 220 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 510 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # ----- convolution & smooth‑max weights -----
        conv = full_convolution(a)                     # length m
        shift = conv.max()
        # stable log‑sum‑exp
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()                 # soft‑max distribution

        # ----- flattening error term -----
        err = conv - target

        # ----- gradients -----
        grad_max = correlation_grad(a, w, size_fft)
        grad_err = correlation_grad(a, err, size_fft)

        grad = (1.0 - lam) * grad_max + lam * grad_err

        # ----- Adam update -----
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # ----- book‑keeping -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Targeted pair‑wise hill‑climb (sub‑gradient based)
# ----------------------------------------------------------------------


def pairwise_hill_climb(
    a0: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-7,
    cooling: float = 0.9995,
    step_init: float = 0.07,
    step_min: float = 5e-6,
) -> np.ndarray:
    """
    Metropolis‑style hill‑climb that moves probability from the index
    contributing most to the active max‑constraint to the one contributing
    least.  Acceptance follows a temperature schedule; the step size is
    adapted automatically.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    size_fft = 1 << ((3 * n - 3).bit_length())
    T = init_temp
    step_scale = step_init
    it = 0

    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-12   # active tolerance
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient for the exact max (2·(a ⋆ 1_{active}))
        active_indicator = np.zeros(2 * n - 1, dtype=float)
        active_indicator[active] = 1.0
        g = correlation_grad(a, active_indicator, size_fft)  # length n

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        # safeguard – ensure we move a non‑zero amount
        if a[i_max] <= 0:
            pos = np.where(a > 0)[0]
            if len(pos) == 0:
                break
            i_max = int(np.random.choice(pos))

        delta = step_scale * a[i_max]
        if delta <= 0:
            delta = step_min
        delta = min(delta, a[i_max])   # keep non‑negative

        cand = a.copy()
        cand[i_max] -= delta
        cand[i_min] += delta
        # numerical safety
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        dscore = cand_score - cur_score

        # Metropolis acceptance
        if dscore < 0 or np.random.rand() < np.exp(-dscore / max(T, 1e-16)):
            a = cand
            cur_score = cand_score
            # adapt step scale upwards on success
            step_scale = min(step_scale * 1.015, 0.5)
        else:
            # shrink step scale on rejection
            step_scale = max(step_scale * 0.985, step_min)

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # cooling
        T = max(final_temp, T * cooling)

        # occasional jitter to diversify search
        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_init * 0.012)
            a = a + jitter
            a = project_to_simplex(a)
            cur_score = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.33) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises the
    ``evaluate_sequence`` objective. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – larger n gives a tighter asymptotic constant.
    # -----------------------------------------------------------------
    n = 8192

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous construction, if present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈150 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 150.0)
    if time.time() < baseline_deadline:
        cand_vecs = baseline_search(
            n, baseline_deadline, max_keep=12, random_draws=25000
        )
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Combined Adam optimisation (multiple restarts, different λ)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 250.0 and best_seq is not None:
        # allocate ~60 % of the remaining budget to Adam
        adam_budget = remaining * 0.60
        start_vectors = [
            np.array(best_seq, dtype=float),
            np.full(n, 1.0 / n),
        ]
        # a handful of random starts
        for _ in range(3):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        # run Adam on each start; share the total budget equally
        per_run = adam_budget / max(1, len(start_vectors) * 2)  # two λ‑values per start
        for idx, start_vec in enumerate(start_vectors):
            # λ = 0.0 (focus purely on max)
            if time.time() + per_run > deadline:
                break
            local_dead = time.time() + per_run
            a_ref = combined_adam(start_vec, local_dead, lam=0.0)
            val = compute_score(a_ref)
            print(f"[adam λ=0.0 start {idx}] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_ref.tolist()

            # λ = 0.15 (adds a flattening regulariser)
            if time.time() + per_run > deadline:
                break
            local_dead = time.time() + per_run
            a_ref = combined_adam(start_vec, local_dead, lam=0.15)
            val = compute_score(a_ref)
            print(f"[adam λ=0.15 start {idx}] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_ref.tolist()

        print(f"[combined‑adam] best after this phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Pairwise hill‑climb (targeted mass transfer) – main refinement
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0 and best_seq is not None:
        pairwise_deadline = time.time() + remaining * 0.65
        a0 = np.array(best_seq, dtype=float)
        a_ref = pairwise_hill_climb(a0, pairwise_deadline)
        val = compute_score(a_ref)
        print(f"[pairwise] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_ref.tolist()

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing (≈ remaining time)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and best_seq is not None:
        polish_deadline = time.time() + remaining
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.33)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
