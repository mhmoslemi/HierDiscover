def exp_subgradient(a0, deadline, eta0=0.5, epsilon_rel=1e-9):
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * epsilon_rel
        active = np.where(conv >= max_val - eps)[0]
        # compute gradient
        n = a.shape[0]
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            # indices j that align
            g[mask] += 2.0 * a[j[mask]]
        # learning rate
        eta = eta0 / np.sqrt(it)
        # update multiplicatively
        a = a * np.exp(-eta * g)
        # renorm
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / a_sum
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            # maybe reduce eta for finer steps? but we keep schedule.
        # optional jitter every certain iterations
        if it % 3000 == 0:
            # jitter via lognormal
            jitter = np.exp(np.random.randn(n) * (eta0 / (np.sqrt(it) + 1e-12)))
            a = a * jitter
            a = a / a.sum()
    return best, best_score

"""
Simple multi‑phase optimizer for the step‑height problem.

Phases
------
1.  Initialise from the known construction (or uniform) and normalise to a
    probability vector (sum = 1).  The objective for a normalised vector is
    2·n·max(conv(a,a)).

2.  Exponentiated sub‑gradient descent.
    The sub‑gradient of  max(conv)  is computed analytically; an
    exponentiated (multiplicative) update keeps the vector strictly positive.
    The step size decays as 1/√t.

3.  Random multiplicative hill‑climbing.
    The current best vector is perturbed in log‑space with a decreasing
    Gaussian noise level (σ → σ_min).  Each proposal is re‑normalised onto the
    simplex; improvements are kept.

4.  Final polishing with the classic projected sub‑gradient method
    (identical to the baseline).

The whole routine respects the 1000 s budget and always returns a valid
list of non‑negative heights.  All heavy work is performed with NumPy FFTs
and a Euclidean projection onto the simplex where required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power‑of‑two for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability‑normalised vector:
        score = 2·n·max(conv(a,a))
    (the denominator sum(a)**2 equals 1).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1. Exponentiated sub‑gradient descent
# ----------------------------------------------------------------------


def exp_subgradient(a0: np.ndarray, deadline: float, eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(a)

    it = 0
    eps_rel = 1e-9
    n = a.shape[0]

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * eps_rel
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max‑convolution (same as in the baseline)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # decaying learning rate
        eta = eta0 / np.sqrt(it)

        # exponentiated update, keep positivity
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            # occasional debug output
            if it % 2000 == 0:
                print(f"[expSG] iter {it:5d} – new best {best_score:.6f}")

        # gentle jitter to help escape plateaus
        if it % 3000 == 0:
            jitter = np.exp(np.random.randn(n) * (eta0 / (np.sqrt(it) + 1e-12)))
            a = a * jitter
            a = a / a.sum()

    return best, best_score


# ----------------------------------------------------------------------
# 2. Random multiplicative hill‑climbing in log‑space
# ----------------------------------------------------------------------


def random_hill_climbing(start: np.ndarray,
                         deadline: float,
                         sigma_start: float = 0.5,
                         sigma_end: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Simple stochastic improvement: log‑space Gaussian perturbations with a linearly
    decreasing σ.  Only strictly improving proposals are kept.
    """
    best = start.copy()
    best_score = compute_score(best)
    log_best = np.log(best + 1e-30)  # protect against log(0)

    phase_start = time.time()
    total = deadline - phase_start
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - phase_start
        frac = elapsed / total
        sigma = sigma_start * (1.0 - frac) + sigma_end * frac

        cand_log = log_best + sigma * np.random.randn(best.shape[0])
        cand = np.exp(cand_log)
        cand = cand / cand.sum()

        cand_score = compute_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            log_best = np.log(best + 1e-30)
            if it % 5000 == 0:
                print(f"[RHC] iter {it:6d} – new best {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# 3. Baseline projected sub‑gradient polish (unchanged)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray,
                       deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected additive sub‑gradient descent on the exact objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    eps_rel = 1e-9
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * eps_rel
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to avoid tiny basins
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Load a sensible starting point (fallback) if available
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        print("[init] fallback sequence found")
    except Exception:
        fallback = None
        print("[init] no fallback – using uniform start")

    n = len(fallback) if fallback is not None else 2048

    if fallback is not None:
        init_vec = np.array(fallback, dtype=float)
    else:
        init_vec = np.full(n, 1.0 / n, dtype=float)

    # ensure a valid probability vector
    init_vec = np.maximum(init_vec, 0.0)
    init_vec = init_vec / init_vec.sum()

    best_vec = init_vec.copy()
    best_score = compute_score(best_vec)
    print(f"[init] initial score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 1 – exponentiated sub‑gradient descent
    # -----------------------------------------------------------------
    phase1_deadline = deadline - 300.0  # reserve ~5 min for later phases
    if time.time() < phase1_deadline:
        print("[phase1] exponentiated sub‑gradient descent")
        vec1, score1 = exp_subgradient(best_vec, phase1_deadline, eta0=0.6)
        if score1 < best_score:
            best_vec, best_score = vec1, score1
            print(f"[phase1] improved score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 2 – random multiplicative hill‑climbing
    # -----------------------------------------------------------------
    phase2_deadline = deadline - 80.0   # leave ~1 min for polishing
    if time.time() < phase2_deadline:
        print("[phase2] random hill‑climbing")
        vec2, score2 = random_hill_climbing(best_vec, phase2_deadline,
                                            sigma_start=0.5, sigma_end=0.001)
        if score2 < best_score:
            best_vec, best_score = vec2, score2
            print(f"[phase2] improved score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 3 – final additive sub‑gradient polishing (baseline)
    # -----------------------------------------------------------------
    phase3_deadline = deadline - 5.0
    if time.time() < phase3_deadline:
        print("[phase3] final polishing")
        polished = subgradient_polish(best_vec, phase3_deadline, step0=0.30)
        polished_score = compute_score(polished)
        if polished_score < best_score:
            best_vec, best_score = polished, polished_score
            print(f"[phase3] polished score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Convert to plain list and evaluate with the provided function
    # -----------------------------------------------------------------
    best_seq = best_vec.tolist()
    final_val = evaluate_sequence(best_seq)
    print(f"\nSearch finished. evaluate_sequence score = {final_val:.6f}")
    return best_seq
