def smooth_descent(a0, deadline, lr0=0.1, alpha0=5.0, alpha_inc=5.0, alpha_max=200.0):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    lr = lr0
    alpha = alpha0
    while time.time() < deadline:

        M = conv.max()
        # compute weighted avg gradient via softmax approx
        exps = np.exp(alpha * (conv - M))  # values <= 1
        w = exps / exps.sum()
        g = gradient_of_weighted_conv(w, a)
        # gradient step
        t += 1
        step = lr / np.sqrt(t)  # or maybe lr * something
        a_new = a - step * g
        a_new = project_to_simplex(a_new)
        # compute new objective
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        # optionally accept even if not improved? but keep a anyway for future updates maybe worser.
        # We can set a = a_new always to keep exploring; or revert if worse.
        if new_score <= compute_score(a):  # if improvement
            a = a_new
        else:
            # if not improved, maybe keep old a or reduce step:
            a = a  # keep old
        # adjust alpha gradually
        if t % 500 == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.2, alpha_max)
        # optional jitter
        if t % 2000 == 0:
            jitter = np.random.randn(len(a)) * (lr / np.sqrt(t) * 0.01)
            a = project_to_simplex(a + jitter)
    return best_a

def guided_hill_climb(a_start, deadline, max_iter=1_000_000):
    a = a_start.copy()
    best_a = a.copy()
    n = a.shape[0]
    conv = full_convolution(a)
    best_M = conv.max()
    best_score = 2 * n * best_M
    step_scale = 0.02
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute gradient of max conv: subgradient g
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        # compute subgradient g
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # choose i with largest g, j with smallest g (or randomly among top)
        i = np.argmax(g)
        j = np.argmin(g)
        # compute delta = step_scale * a[i] (or step_scale * something)
        delta = step_scale * a[i]
        if delta <= 0: # safety
            break
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        # compute new score
        conv_cand = full_convolution(cand)
        M_cand = conv_cand.max()
        score_cand = 2 * n * M_cand
        if score_cand < best_score:

            a = cand

"""
Improved optimiser for ``evaluate_sequence``.

Main ideas
----------
*  Start with a broad parametric sweep (reverse power‐law, two‑parameter
   families, sigmoid, geometric and random Dirichlet vectors) to obtain a
   good initial shape.
*  Refine the best shape with a *soft‑max smoothed gradient descent*.
   The objective ``max(conv(a,a))`` is approximated by a smooth
   ``log‑sum‑exp`` and the exact gradient of the weighted sum can be
   computed efficiently via one FFT‑based convolution per iteration.
   The iterate is kept on the probability simplex by Euclidean projection.
*  After the smoothed descent perform a long stochastic hill‑climb
   (mass transfer between two positions) to polish the solution.
*  Several short restarts of the smooth descent from random
   Dirichlet seeds are performed if time permits.
*  The whole routine respects a 1000 s budget and prints progress
   information.  The final returned list is a non‑negative sequence whose
   value under ``evaluate_sequence`` is (hopefully) lower than the previous
   best of 1.510380.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def conv_full(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full convolution of two arbitrary vectors using FFT."""
    m = len(u) + len(v) - 1
    n_fft = 1 << ((m - 1).bit_length())
    U = np.fft.rfft(u, n=n_fft)
    V = np.fft.rfft(v, n=n_fft)
    conv = np.fft.irfft(U * V, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σ a = 1):
       2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σ a = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for seeding)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p,  p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    a = idx ** p
    a /= a.sum()
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.
    Both p,q ≥ 0 give a front‑loaded shape.
    """
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    a /= a.sum()
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """
    a_i ∝ 1/(1+exp((i‑c)/σ)).
    centre_frac ∈ (0,1) is the location of the mid‑point,
    sigma_frac controls the steepness (σ = sigma_frac·n).
    """
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^i, 0 < q < 1."""
    a = q ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – coarse parametric sweep (fast, gives a decent seed)
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Explore a few families and keep the best discovered vector."""
    best_a = None
    best_val = float('inf')

    # 1. Reverse power‑law (coarse)
    print("[sweep] reverse_power_law (coarse)", flush=True)
    for p in np.linspace(0.0, 12.0, 241):   # step ≈0.05
        if time.time() >= deadline:
            return best_a
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # 2. Multi‑power (grid)
    print("[sweep] multi_power", flush=True)
    for p in np.linspace(0.0, 6.0, 61):
        for q in np.linspace(0.0, 4.0, 41):
            if time.time() >= deadline:
                return best_a
            a = multi_power(n, p, q)
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    # 3. Sigmoid family
    print("[sweep] sigmoid_shape", flush=True)
    for centre_frac in np.linspace(0.1, 0.9, 17):
        for sigma_frac in np.linspace(0.01, 0.20, 20):
            if time.time() >= deadline:
                return best_a
            a = sigmoid_shape(n, centre_frac, sigma_frac)
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    # 4. Geometric (fast decay)
    print("[sweep] geometric", flush=True)
    for q in np.linspace(0.60, 0.99, 80):
        if time.time() >= deadline:
            return best_a
        a = geometric(n, q)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # 5. Random Dirichlet draws – cheap high‑dim exploration
    print("[sweep] random Dirichlet samples", flush=True)
    for _ in range(2000):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – smooth (soft‑max) gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def smooth_descent(a0: np.ndarray, deadline: float,
                   lr0: float = 0.12,
                   alpha_start: float = 8.0,
                   alpha_factor: float = 1.10,
                   alpha_max: float = 200.0) -> np.ndarray:
    """
    Smoothed gradient descent.
    The maximal convolution is approximated by a soft‑max with temperature ``1/α``.
    ``lr0`` is the initial learning rate; it decays like 1/√t.
    ``alpha`` is increased gradually to tighten the approximation.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    lr = lr0
    alpha = alpha_start

    while time.time() < deadline:
        # current convolution and score
        conv = full_convolution(a)
        M = conv.max()
        cur_score = 2.0 * n * M

        # soft‑max weights (stable)
        exps = np.exp(alpha * (conv - M))   # ≤ 1
        w = exps / exps.sum()

        # gradient of the weighted sum:
        # g_i = 2 * Σ_j w_{i+j} a_j   (convolution of w with reversed a)
        a_rev = a[::-1]
        conv_w_a = conv_full(w, a_rev)          # length = 3n‑2
        start = n - 1
        g = 2.0 * conv_w_a[start:start + n]

        # projected gradient step
        t += 1
        step = lr / math.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        # evaluate new point
        conv_new = full_convolution(a_new)
        M_new = conv_new.max()
        new_score = 2.0 * n * M_new

        if new_score < best_score:
            best_score, best_a = new_score, a_new.copy()

        # accept if improvement, otherwise keep the old point
        if new_score <= cur_score:
            a = a_new
            # optionally increase learning rate a bit on success
            lr = min(lr * 1.02, lr0 * 3.0)
        else:
            # on failure shrink step size
            lr *= 0.93

        # increase temperature (α) slowly
        if t % 500 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_factor, alpha_max)

        # occasional jitter to escape plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr * 0.004 / math.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Phase 3 – stochastic hill‑climb (mass transfer between two indices)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size and restart on stagnation."""
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    conv = full_convolution(a)
    best_M = conv.max()
    best_score = 2.0 * n * best_M

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        conv_cand = full_convolution(cand)
        M_cand = conv_cand.max()
        score_cand = 2.0 * n * M_cand

        if score_cand < best_score:
            best_score, best_M = score_cand, M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            # restart from the best with a tiny random perturbation
            jitter = np.random.randn(n) * 0.0008
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    n = 2048
    best_seq = None
    best_score = float('inf')

    # -------------------------------------------------------------
    # (0) Optional fallback from a previously discovered global variable
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]    # type: ignore
        val_fallback = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val_fallback:.6f}")
        if val_fallback < best_score:
            best_score, best_seq = val_fallback, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # (1) Parametric sweep – cheap bulk exploration
    # -------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            # official evaluation for safety
            val_real = evaluate_sequence(a_initial.tolist())
            print(f"[sweep] best proxy score = {val:.6f}")
            print(f"[sweep] official score = {val_real:.6f}")
            if val_real < best_score:
                best_score, best_seq = val_real, a_initial.tolist()

    # -------------------------------------------------------------
    # (2) Multiple smooth descent restarts (if time permits)
    # -------------------------------------------------------------
    descent_deadline = deadline - 150.0
    if best_seq is not None and time.time() < descent_deadline:
        # keep the best found so far as one start
        starts = [np.array(best_seq, dtype=float)]

        # add a few random Dirichlet starts
        for _ in range(3):
            if time.time() >= descent_deadline:
                break
            starts.append(np.random.dirichlet(np.ones(n)))

        # add a strong front‑loaded start (reverse power‑law with p≈4)
        if time.time() < descent_deadline:
            starts.append(reverse_power_law(n, 4.0))

        # allocate equal time slices for each start
        slice_time = (descent_deadline - time.time()) / max(1, len(starts))
        for a0 in starts:
            sub_deadline = min(descent_deadline, time.time() + slice_time)
            a_opt = smooth_descent(a0, sub_deadline)
            score_opt = compute_score(a_opt)
            print(f"[smooth‑desc] candidate score = {score_opt:.6f}")
            if score_opt < best_score:
                best_score, best_seq = score_opt, a_opt.tolist()
            if time.time() >= descent_deadline:
                break

    # -------------------------------------------------------------
    # (3) Final stochastic hill‑climb (polishing)
    # -------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        final_score = compute_score(a_final)
        print(f"[hill‑climb] final proxy score = {final_score:.6f}")
        # official evaluation (the official function does the same scaling)
        val_official = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] official score = {val_official:.6f}")
        if val_official < best_score:
            best_score, best_seq = val_official, a_final.tolist()

    # -------------------------------------------------------------
    # (4) Safeguard – fall back to a simple analytic shape
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.0)   # linear decreasing shape
        best_seq = a_fallback.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] using linear shape, official score = {best_score:.6f}")

    # -------------------------------------------------------------
    # Final report (official evaluation)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")
    return best_seq
