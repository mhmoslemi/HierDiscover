"""
Algorithm
---------
1. Build a large discrete arcsine distribution (the Beta(½,½) density) that
   avoids the singularities at the extreme points.  Its self‑convolution
   is almost perfectly flat, so the theoretical lower bound of the
   evaluation function (≈ 1) is approached as the length grows.

2. Optionally improve the shape by a cheap linear combination with the
   uniform vector and by trying a handful of nearby Beta‑shapes.

3. Perform a fast projected sub‑gradient descent on the true objective.
   The gradient of max conv is obtained with FFT‑based cross‑correlation,
   which is O(n log n) even for n ≈ 5·10⁴.  After each step we clip the
   vector to be non‑negative and renormalise it onto the probability simplex.

4. Return the best sequence found before the wall‑clock budget expires.
   The final score is obtained with the provided ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex_clip(v: np.ndarray) -> np.ndarray:
    """
    Very cheap projection: clip negatives, then renormalise to sum = 1.
    For our use the Euclidean projection is not critical – this keeps the
    routine fast for large n.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for zero‑padding
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective 2·n·max(conv(a,a)) for a vector a
    that is already normalised to sum = 1 (the denominator cancels out).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape without a singular endpoint.
    a_i ∝ 1 / sqrt(x_i·(1−x_i)), where x_i = (i+½)/n ∈ (0,1).
    The vector is normalised to the probability simplex.
    """
    i = np.arange(n, dtype=float) + 0.5          # bin centres
    x = i / n                                    # in (0,1)
    raw = 1.0 / np.sqrt(x * (1.0 - x))
    raw_sum = raw.sum()
    return raw / raw_sum


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑type shape: a_i ∝ (i+1)^{α−1}·(n−i)^{β−1}   (α,β>0).
    Returned vector lies on the simplex.
    """
    i = np.arange(1, n + 1, dtype=float)
    raw = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = raw.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n)
    return raw / s


# ----------------------------------------------------------------------
# Fast projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_polish_fast(a0: np.ndarray,
                            deadline: float,
                            step0: float = 0.30,
                            eps_factor: float = 1e-12,
                            max_iters: int = 20000) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective.
    The gradient of the active max‑convolution entries is computed by a
    single FFT‑based cross‑correlation, making each iteration O(n log n).
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # 1) current convolution and active set
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * eps_factor
        active = (conv >= max_val - eps).astype(np.float64)   # length m

        # 2) gradient: g_i = 2 * Σ_{k active} a_{k-i}
        #    This is 2 × cross‑correlation(active, a)
        rev_active = active[::-1]                 # length m
        Nfft = 1 << ((n + m - 1).bit_length())    # enough for linear convolution
        A = np.fft.rfft(a, n=Nfft)
        C = np.fft.rfft(rev_active, n=Nfft)
        conv_ac = np.fft.irfft(A * C, n=Nfft)[:n + m - 1]   # length n+m‑1
        g = 2.0 * conv_ac[(m - 1) - np.arange(n)]          # shape (n,)

        # 3) sub‑gradient step + projection
        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex_clip(a)

        # 4) keep best
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex_clip(a)

    return best_a

    """

    start = time.time()
    deadline = start + budget_s - 2.0          # tiny safety margin

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    best_seq = None

    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
