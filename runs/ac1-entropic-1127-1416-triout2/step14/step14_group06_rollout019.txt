search_step_function_v4.py

A new optimiser that replaces the hand‑crafted sub‑gradient step with a
continuation method based on a smooth approximation of the max‑convolution.
The algorithm:

0.  Run the same cheap parametric sweep as before to obtain a decent
    starter vector (reverse‑power‑law, mixed shapes, beta‑shapes…).

1.  *Smooth‑max gradient descent* – minimise a soft‑max of the

    The “temperature” τ is slowly increased, turning the smooth objective
    into the true max.  The gradient of the smoothed objective is obtained
    analytically via a single extra convolution, so each iteration costs
    O(n log n) only.

2.  A final stochastic hill‑climb (identical to the previous version) to
    polish any remaining tiny irregularities.

The whole routine respects the 1000 s budget, falls back to a mixed

# ----------------------------------------------------------------------
# Linear algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σv = 1, v ≥ 0."""
    if np.all(v <= 0):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution x * y using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())          # next power of two ≥ n
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    C = X * Y
    conv = np.fft.irfft(C, n=n_fft)[:n]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a."""
    return linear_convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

    a = (x ** alpha) * ((1.0 - x) ** beta)

    return a

# Smooth‑max gradient descent (continuation method)
# ----------------------------------------------------------------------
def smooth_max_gradient_descent(a0: np.ndarray, deadline: float,
                                tau_start: float = 0.5,
                                tau_end: float = 30.0,
                                tau_factor: float = 1.02,
                                step0: float = 0.3,
                                jitter_interval: int = 2000) -> np.ndarray:
    """
    Minimise  max(conv(a,a))  using a smooth max (soft‑max) surrogate.
    τ (tau) starts small (smooth) and is multiplied by ``tau_factor``
    after every iteration until ``tau_end`` is reached.
    The iterate is kept on the simplex via Euclidean projection.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    t = 0
    tau = tau_start

    while time.time() < deadline:
        t += 1

        # --- convolution and smooth‑max weights ---------------------------------
        conv = full_convolution(a)                 # length 2n‑1
        max_conv = conv.max()
        # shift by max to avoid overflow
        shifted = conv - max_conv
        exp_scaled = np.exp(tau * shifted)
        w = exp_scaled / exp_scaled.sum()          # soft‑max over all shifts

        # --- gradient of the smooth objective ----------------------------------
        # correlation of a with w (no reversal)  →  b_i = Σ_j w_{i+j} a_j
        # we obtain it via a convolution with the reversed weight vector
        conv_rev = linear_convolve(a, w[::-1])      # length 3n‑2
        idx_start = (len(conv) - 1)                # = 2n‑2
        b = conv_rev[idx_start - np.arange(n)]

        # diagonal correction (k = 2*i) – only one term, not doubled
        diag = np.zeros_like(a)
        diag_mask = (2 * np.arange(n)) < len(conv)
        diag_idx = 2 * np.arange(n)[diag_mask]
        diag[diag_mask] = w[diag_idx]

        # gradient of the smoothed max (∂/∂a_i)
        g = 2.0 * b - diag * a

        # --- step & projection --------------------------------------------------
        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape flat plateaus
        if t % jitter_interval == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

        # keep the best vector seen so far (using the exact objective)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # raise the temperature slowly
        if tau < tau_end:
            tau *= tau_factor

    return best_a

def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 200_000) -> np.ndarray:
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

            best_val, best_seq = val, a_smooth.tolist()

    # -------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # -------------------------------------------------------------
    hill_deadline = deadline - 4.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # 5. Safeguard – fall back to a simple mixed shape if nothing else
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = mixed_reverse_uniform(n, mix=0.8, p=1.7)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
