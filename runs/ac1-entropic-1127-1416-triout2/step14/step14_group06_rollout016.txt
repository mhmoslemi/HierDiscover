"""
Optimiser for evaluate_sequence using
- quick parametric sweep of several families,
- smoothed‑max Adam gradient descent,
- sub‑gradient polishing,
- final stochastic hill‑climb.
"""

import time
import math
import numpy as np

# Helper utilities
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Families of candidate vectors ------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a

def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a

def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = uniform_shape(n)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

# Fast parametric sweep – limited but useful ---------------------------------
def parametric_sweep_fast(n: int, deadline: float) -> np.ndarray:
    """Very quick sweep over a few families; returns best discovered vector."""
    best_vec = None
    best_val = float('inf')

    # reverse power‑law (coarse)
    for p in np.arange(0.5, 8.0 + 1e-12, 0.2):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, float(p))
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # mixed reverse‑power‑law + uniform
    for mix in np.linspace(0.3, 0.9, 7):
        for p in np.linspace(0.5, 5.0, 12):
            if time.time() >= deadline:
                return best_vec
            a = mixed_reverse_uniform(n, mix, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # geometric family
    for q in np.linspace(0.985, 0.9999, 15):
        if time.time() >= deadline:
            break
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # random combinations of the three basic shapes
    uniform = uniform_shape(n)
    rp2 = reverse_power_law(n, 2.0)
    geom = geometric(n, 0.995)
    basics = [uniform, rp2, geom]

    for _ in range(1000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec

# Smoothed‑max Adam optimiser -------------------------------------------------
def smoothed_max_optimize(a0: np.ndarray, deadline: float,
                          max_iters: int = 20000,
                          min_alpha: float = 50.0,
                          max_alpha: float = 5000.0,
                          lr: float = 0.05) -> (np.ndarray, float):
    """
    Adam optimiser of the smoothed max of the convolution.
    Objective: f(a) = log‑sum‑exp(alpha·conv(a,a)).
    The gradient can be written as a weighted cross‑correlation.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # FFT size needed for the cross‑correlation (w * a_rev)
    conv_len = (2 * n - 1) + n - 1            # = 3n - 2
    n_fft_corr = 1 << ((conv_len - 1).bit_length())
    start_idx = n - 1                         # slice start for ‘valid’ result

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linearly increase alpha to focus the surrogate onto the true max
        alpha = min_alpha + (max_alpha - min_alpha) * (t / max_iters)

        # full self‑convolution (used for the surrogate loss)
        conv = full_convolution(a)          # length 2n‑1
        maxc = conv.max()

        # soft‑max weights
        exp_alpha = np.exp(alpha * (conv - maxc))
        w = exp_alpha / exp_alpha.sum()

        # gradient = 2·α·(w ⋆ a)   (cross‑correlation, shape n)
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=n_fft_corr)
        A_rev = np.fft.rfft(a_rev, n=n_fft_corr)
        full = np.fft.irfft(W * A_rev, n=n_fft_corr)
        g = full[start_idx:start_idx + n]           # valid part
        grad = 2.0 * alpha * g

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasibility
        a = project_to_simplex(a)

        # occasional true‑objective check
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score

# Sub‑gradient polishing (same as before, but slightly tweaked) -------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # gradient w.r.t. a is the sum of shifted copies of a over the active set
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# Hill‑climb (same as before) -----------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 200_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size and restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        if no_improve > max_no_improve:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

"""
Optimiser for evaluate_sequence.

The search consists of four stages:
1. a quick parametric sweep over a few analytic families;
2. an Adam optimiser that minimises a smooth approximation of the
   max‑convolution (log‑sum‑exp surrogate);
3. a sub‑gradient polishing phase (as in the reference code);
4. a final stochastic hill‑climb.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
so the objective 2·n·max(conv) is invariant under scaling and can be
computed cheaply with FFTs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())            # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a)) .
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used in the quick sweep)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  for 0 < q < 1."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    Convex combination   mix·reverse_power_law(p) + (1‑mix)·uniform.
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = uniform_shape(n)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

# Fast parametric sweep (≈ 60 s)
# ----------------------------------------------------------------------
def parametric_sweep_fast(n: int, deadline: float) -> np.ndarray:
    """Very quick sweep over a handful of families; returns the best vector."""
    best_vec = None
    best_val = float('inf')

    # reverse power‑law (coarse grid)
    for p in np.arange(0.5, 8.0 + 1e-12, 0.2):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, float(p))
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # mixed reverse‑power‑law + uniform
    for mix in np.linspace(0.3, 0.9, 7):
        for p in np.linspace(0.5, 5.0, 12):
            if time.time() >= deadline:
                return best_vec
            a = mixed_reverse_uniform(n, mix, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # geometric family
    for q in np.linspace(0.985, 0.9999, 15):
        if time.time() >= deadline:
            break
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    uniform = uniform_shape(n)
    rp2 = reverse_power_law(n, 2.0)
    geom = geometric(n, 0.995)
    basics = [uniform, rp2, geom]

        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on a log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def smoothed_max_optimize(a0: np.ndarray,
                          deadline: float,
                          max_iters: int = 20000,
                          min_alpha: float = 30.0,
                          max_alpha: float = 5000.0,
                          lr: float = 0.06) -> (np.ndarray, float):
    """
    Adam optimisation of the surrogate
        f(a) = log Σ_k exp(α·conv_k(a)),
    whose gradient is 2·α·(w ⋆ a) where w are the soft‑max weights.
    The surrogate becomes the true max as α → ∞.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # FFT parameters for the cross‑correlation w ⋆ a
    conv_len = (2 * n - 1) + n - 1                # = 3n‑2
    n_fft_corr = 1 << ((conv_len - 1).bit_length())
    start_idx = n - 1                             # valid slice start

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linearly increase α
        alpha = min_alpha + (max_alpha - min_alpha) * (t / max_iters)

        # self‑convolution (used for the surrogate loss)
        conv = full_convolution(a)                # length 2n‑1
        maxc = conv.max()

        # soft‑max weights, numerically stable
        exp_alpha = np.exp(alpha * (conv - maxc))
        w = exp_alpha / exp_alpha.sum()

        # gradient = 2·α·(w ⋆ a)   (cross‑correlation)
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=n_fft_corr)
        A_rev = np.fft.rfft(a_rev, n=n_fft_corr)
        full = np.fft.irfft(W * A_rev, n=n_fft_corr)
        g = full[start_idx:start_idx + n]          # valid part, length n
        grad = 2.0 * alpha * g

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = project_to_simplex(a)

        # occasional exact‑objective check
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as in the reference implementation)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # gradient for the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 200_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size and restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        if no_improve > max_no_improve:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. The routine combines a quick parametric sweep,
    an Adam optimiser on a smoothed max‑convolution surrogate, a
    sub‑gradient polishing phase and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0            # safety margin

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. optional fallback from a pre‑computed construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. choose vector length (larger = finer discretisation)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 250 else 2048
    print(f"[info] using vector length n = {n}")

    # -----------------------------------------------------------------
    # 2. quick parametric sweep (≈ 60 s, leaves plenty of time for later phases)
    # -----------------------------------------------------------------
    sweep_deadline = min(deadline - 300.0, time.time() + 60.0)
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep_fast(n, sweep_deadline)
        if a_initial is not None:
            val = evaluate_sequence(a_initial.tolist())
            print(f"[parametric] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 3. smoothed‑max Adam optimisation (uses the bulk of the remaining budget)
    # -----------------------------------------------------------------
    if best_seq is None:
        a_start = uniform_shape(n)
    else:
        a_start = np.array(best_seq, dtype=float)

    adam_deadline = deadline - 80.0
    best_vec = a_start.copy()
    best_score = compute_score(best_vec)

    # Run several independent Adam optimisations (≈30 s each) until the deadline
    while time.time() < adam_deadline:
        jitter = np.random.randn(n) * 0.001
        start_vec = best_vec + jitter
        start_vec = project_to_simplex(start_vec)

        run_deadline = min(adam_deadline, time.time() + 30.0)
        opt_vec, opt_score = smoothed_max_optimize(start_vec,
                                                   run_deadline,
                                                   max_iters=15000,
                                                   min_alpha=30.0,
                                                   max_alpha=5000.0,
                                                   lr=0.06)
        if opt_score < best_score:
            best_score = opt_score
            best_vec = opt_vec.copy()
            print(f"[adam] improved smoothed‑max score = {best_score:.6f}")

    # Record the best vector obtained so far
    candidate = best_vec.tolist()
    val = evaluate_sequence(candidate)
    if val < best_val:
        best_val, best_seq = val, candidate
        print(f"[after adam] official score = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. sub‑gradient polishing (≈ 60 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 20.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.12)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 5. final stochastic hill‑climb (last few seconds)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 4.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 6. safeguard fallback (in the very unlikely case we have nothing)
    # -----------------------------------------------------------------
    if best_seq is None:
        fallback = mixed_reverse_uniform(n, mix=0.8, p=1.7)
        best_seq = fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print("[fallback] using mixed shape, score =", best_val)

    # -----------------------------------------------------------------
    # final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
