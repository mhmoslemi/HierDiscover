"""
Non‑negative step‑function optimiser (new version).

The optimiser differs from the previous implementation in several
aspects:

* a longer single Adam run with an aggressive schedule for the smooth‑max
  temperature (τ) that ends close to zero, giving a much tighter
  approximation of the true max‑convolution objective;
* a final simulated‑annealing phase that is able to accept temporary
  deteriorations and therefore escapes the local minima that Adam can
  get stuck in;
* a more aggressive mass‑transfer hill‑climb that is run after the
  annealing while the remaining time permits;
* a much larger pool of cheap Dirichlet seeds, and a simple fallback
  to a previously discovered construction if it is available.

All vectors are always kept on the probability simplex
(∑a = 1, a_i ≥ 0).  The returned list is a plain Python list of
floats, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a*a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_max_adam(a0: np.ndarray, deadline: float,
                    tau_start: float = 2.0,
                    tau_end: float = 1e-9,
                    tau_decay: float = 0.97,
                    lr_start: float = 0.2,
                    lr_decay: float = 0.995,
                    beta1: float = 0.90,
                    beta2: float = 0.999,
                    eps: float = 1e-12) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate
        f_τ(a) = τ·log Σ_k exp(conv_k/τ)

    The routine runs until *deadline* and returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # --- temperature schedule (geometric) ---
        if t % 150 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # --- compute soft‑max over the full convolution ---
        conv = full_convolution(a)                     # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()                  # shape (m,)

        # --- gradient of the smooth‑max (2·a⋆w_rev) ---
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        # extract the n components that correspond to the gradient
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # --- Adam update ---
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # periodic learning‑rate decay
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-6)

        # bookkeeping
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter={t}, τ={tau:.2e}, lr={lr:.2e}, score={cur_val:.6f}")

        # occasional jitter to help escape plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Mass‑transfer hill‑climb (exact objective)
# ----------------------------------------------------------------------


def mass_transfer_hill_climb(a0: np.ndarray, deadline: float,
                            max_iters: int = 200_000) -> np.ndarray:
    """Move a small fraction of mass from one index to another, always
    keeping the best exact score seen."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        frac = np.random.rand() * 0.35          # up to 35 % of source mass
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑normalise to avoid tiny drift
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:
                print(f"[mass] iter={it}, score={val:.6f}")
        it += 1
    return best_a


# ----------------------------------------------------------------------
# Simulated annealing on the exact objective
# ----------------------------------------------------------------------


def simulated_annealing(a0: np.ndarray, deadline: float,
                        init_temp: float = 0.02,
                        cooling: float = 0.9995) -> np.ndarray:
    """
    Simple annealing that moves a random fraction of mass from one index
    to another.  Accepts worsening moves with probability exp(Δ/ T).
    """
    a = a0.copy()
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val
    n = a.shape[0]
    temp = init_temp
    step = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move up to 25 % of the mass at i towards j
        delta = a[i] * np.random.rand() * 0.25
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        accept = False
        if val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp((cur_val - val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                if step % 1000 == 0:
                    print(f"[anneal] step={step}, temp={temp:.2e}, best={val:.6f}")

        # cool down
        temp = max(temp * cooling, 1e-12)
        step += 1

    return best_a
