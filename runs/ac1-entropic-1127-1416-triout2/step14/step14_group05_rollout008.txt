```
def subgradient_descent(a0, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_max = np.max(full_convolution(a))
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        conv = full_convolution(a)
        K = np.argmax(conv)
        cur_max = conv[K]
        # compute gradient
        g = np.zeros_like(a)
        # using slicing for speed: indices where K-i in [0,n)
        # i from max(0, K-(n-1)) to min(n-1, K)
        i_start = max(0, K - (n-1))
        i_end = min(n-1, K)  # inclusive
        # slice:
        # for i in range(i_start, i_end+1):
        #    g[i] = a[K-i]
        # vectorized:
        idx = np.arange(i_start, i_end+1)
        g[idx] = a[K - idx]
        g *= 2.0
        # compute step size using Polyak rule
        diff = cur_max - best_max
        if diff <= 0:
            step = 0.0
        else:
            norm2 = np.dot(g, g) + 1e-12
            step = diff / norm2
        # optional limit step
        step = min(step, 1.0)  # not too large
        a = a - step * g
        # projection onto simplex
        a = project_to_simplex(a)
        # evaluate
        conv = full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[subgrad] iter={t}, max={cur_max:.6f}")
        t += 1
        # may add occasional jitter if no improvement for many steps.

    return best_a
```

```
    deadline = t_start + budget_s - 5.0

        print("[init] fallback not available; using uniform")

    best_a = a0.copy()
    best_max = full_convolution(a0).max()

    # Determine remaining time
    remaining = deadline - time.time()

    restarts = 5

    for r in range(restarts):

        start_deadline = min(deadline, time.time() + per_restart - 0.5)

        if r == 0:
            start_vec = best_a.copy()
        else:
            start_vec = np.random.dirichlet(np.ones(n))
        print(f"[subgrad] restart {r+1}/{restarts}, time budget {per_restart:.1f}s")
        a_opt = subgradient_descent(start_vec, start_deadline)

        if cur_max < best_max:
            best_max = cur_max
            best_a = a_opt.copy()
```

Subgradient descent optimiser for the step‑function bound.
Starts from several diverse initial vectors, runs a Polyak‑style

# Helper functions

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full(n, 1.0 / n)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    """Linear convolution a * a using the real FFT (fast for n≈2048)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

    """Exact objective (sum of a is assumed to be 1)."""
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())

        best_a = mass_transfer_hill_climb(best_a, mass_deadline)
        best_max = full_convolution(best_a).max()
        print(f"[mass] post‑polish max = {best_max:.6f}, score = {2*n*best_max:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the sequence (the harness will call ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq
