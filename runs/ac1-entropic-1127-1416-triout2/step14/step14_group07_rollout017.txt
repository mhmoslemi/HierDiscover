def simulated_hill_climb(a_start, deadline, max_iter=500000):
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    temp0 = 1.0
    t = 0
    while time.time() < deadline and t < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # propose moving fraction of mass from i to j
        step = np.random.rand() * 0.1 * a[i]  # random fraction up to 10%
        cand = a.copy()
        cand[i] = max(cand[i] - step, 0.0)
        cand[j] = cand[j] + step
        cand = project_to_simplex(cand)
        M_cand = full_convolution(cand).max()
        delta = M_cand - best_M
        # cooling schedule
        temperature = temp0 / (1 + t*0.00001)
        if delta < 0 or np.random.rand() < np.exp(-delta/temperature):
            a = cand
            if M_cand < best_M:
                best_M = M_cand
                best_a = cand.copy()
        t += 1
        if t % 100000 == 0:
            # jitter
            jitter = np.random.randn(n) * 1e-4
            a = best_a + jitter
            a = project_to_simplex(a)
    return best_a

    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a via an FFT (real‑valued)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p,  p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n,…,1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p).  p ≥ 0."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^i , 0<q<1."""
    a = q ** np.arange(n, dtype=np.float64)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p · (i+1)^q ."""
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta_: float) -> np.ndarray:
    """a_i ∝ x_i^α (1‑x_i)^β ,   x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta_)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i‑c)/σ))."""
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def front_loaded(n: int, front_frac: float, front_weight: float) -> np.ndarray:
    """
    A very simple family: a proportion `front_weight` of the total mass
    is placed uniformly on the first `front_frac·n` positions,
    the rest is spread uniformly over the remaining entries.
    """
    L = max(1, int(round(front_frac * n)))
    if L >= n:
        return None
    rest = n - L
    # allocate mass
    front_mass = front_weight
    tail_mass = 1.0 - front_mass
    if front_mass < 0 or tail_mass < 0:
        return None
    a = np.empty(n, dtype=np.float64)
    a[:L] = front_mass / L
    a[L:] = tail_mass / rest
    return a


# ----------------------------------------------------------------------
# Broad parametric sweep (coarse + local refinement)
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Explore many families and keep the vector with the smallest score."""
    best_vec = None
    best_val = float('inf')

    def try_candidate(a):
        nonlocal best_vec, best_val
        if a is None:
            return
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a
            # print only when we improve to avoid clutter
            print(f"[sweep] new best {best_val:.6f}", flush=True)

    # ------------------------------------------------------------------
    # 1. Reverse power‑law (coarse)
    # ------------------------------------------------------------------
    print("[sweep] reverse_power_law (coarse)", flush=True)
    for p in np.linspace(0.05, 10.0, 200):
        if time.time() >= deadline:
            return best_vec
        try_candidate(reverse_power_law(n, p))

    # ------------------------------------------------------------------
    # 2. Forward power‑law (coarse)
    # ------------------------------------------------------------------
    print("[sweep] power_law (coarse)", flush=True)
    for p in np.linspace(0.05, 10.0, 200):
        if time.time() >= deadline:
            return best_vec
        try_candidate(power_law(n, p))

    # ------------------------------------------------------------------
    # 3. Geometric (coarse)
    # ------------------------------------------------------------------
    print("[sweep] geometric (coarse)", flush=True)
    for q in np.linspace(0.80, 0.9999, 250):
        if time.time() >= deadline:
            return best_vec
        try_candidate(geometric(n, q))

    # ------------------------------------------------------------------
    # 4. Multi‑power (grid)
    # ------------------------------------------------------------------
    print("[sweep] multi_power (grid)", flush=True)
    for p in np.linspace(0.0, 5.0, 51):
        for q in np.linspace(0.0, 5.0, 51):
            if time.time() >= deadline:
                return best_vec
            try_candidate(multi_power(n, p, q))

    # ------------------------------------------------------------------
    # 5. Beta shape (grid)
    # ------------------------------------------------------------------
    print("[sweep] beta_shape (grid)", flush=True)
    alphas = np.linspace(-0.5, 5.0, 28)
    betas = np.linspace(-0.5, 5.0, 28)
    for a in alphas:
        for b in betas:
            if time.time() >= deadline:
                return best_vec
            try_candidate(beta_shape(n, a, b))

    # ------------------------------------------------------------------
    # 6. Sigmoid (grid)
    # ------------------------------------------------------------------
    print("[sweep] sigmoid_shape (grid)", flush=True)
    for centre in np.linspace(0.2, 0.8, 31):
        for sigma in np.linspace(0.005, 0.20, 40):
            if time.time() >= deadline:
                return best_vec
            try_candidate(sigmoid_shape(n, centre, sigma))

    # ------------------------------------------------------------------
    # 7. Front‑loaded family (grid)
    # ------------------------------------------------------------------
    print("[sweep] front_loaded (grid)", flush=True)
    for front_frac in np.linspace(0.01, 0.5, 20):
        for front_weight in np.linspace(0.05, 0.95, 20):
            if time.time() >= deadline:
                return best_vec
            try_candidate(front_loaded(n, front_frac, front_weight))

    # ------------------------------------------------------------------
    # 8. Local refinement around the best parameters discovered so far
    # ------------------------------------------------------------------
    if best_vec is not None:
        # try small Gaussian perturbations of the best vector itself
        print("[sweep] local refinement (random jitter)", flush=True)
        for _ in range(2000):
            if time.time() >= deadline:
                break
            jitter = np.random.randn(n) * 1e-4
            a = best_vec + jitter
            a = project_to_simplex(a)
            try_candidate(a)

    # ------------------------------------------------------------------
    # 9. Random Dirichlet draws (various concentration values)
    # ------------------------------------------------------------------
    print("[sweep] random Dirichlet draws", flush=True)
    concs = [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        for _ in range(800):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(np.full(n, conc))
            try_candidate(a)

    # ------------------------------------------------------------------
    # 10. Random convex mixtures of a few strong bases
    # ------------------------------------------------------------------
    print("[sweep] random mixtures", flush=True)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rev2 = reverse_power_law(n, 2.0)
    sig_mid = sigmoid_shape(n, 0.5, 0.05)
    basics = [uniform, rev2, sig_mid]
    for _ in range(3000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        try_candidate(a)

    return best_vec


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (more aggressive step‑size)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.20) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a:
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape plateaus
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Hill‑climb / simulated‑annealing moving a fraction of mass."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    # temperature schedule – starts relatively hot, cools slowly
    T0 = 0.02
    t = 0
    no_improve = 0

    while time.time() < deadline and t < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a random fraction of a[i] to a[j]
        eps = np.random.rand() * 0.08  # up to 8 % of the source mass
        delta = eps * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        Δ = M_cand - best_M
        temperature = T0 / (1 + 0.00001 * t)   # very slow cooling

        accept = Δ < 0 or np.random.rand() < math.exp(-Δ / max(temperature, 1e-12))
        if accept:
            a = cand
            if M_cand < best_M:
                best_M = M_cand
                best_a = cand.copy()
                step_scale = 1.02  # not used – kept for compatibility
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # if stuck for a long time, jitter from the best found so far
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.0006
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

        t += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # (0) Optional fallback from a previously stored global variable
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # (1) Parametric sweep – bulk of the time (~300 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 250.0
    n = 2048                                    # length of the vector
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)       # exact proxy
            print(f"[sweep] best proxy score = {val:.6f}")
            # official evaluation for safety
            val_real = evaluate_sequence(a_initial.tolist())
            print(f"[sweep] official score = {val_real:.6f}")
            if val_real < best_val:
                best_val, best_seq = val_real, a_initial.tolist()

    # -----------------------------------------------------------------
    # (2) Sub‑gradient polishing – terminates ~80 s before deadline
    # -----------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.20)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # (3) Final stochastic hill‑climb (simulated annealing)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # (4) Safeguard – use a simple analytic candidate if nothing else
    # -----------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.0)   # linear decreasing shape
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] using linear shape, score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Final report (official evaluation)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")
    return best_seq
