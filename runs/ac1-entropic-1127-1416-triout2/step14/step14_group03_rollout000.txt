# ----------------------------------------------------------------------
# FFT‑based linear convolution (real vectors)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a * a (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum of entries assumed to be 1, which holds throughout)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (optional)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    # Algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, len(v) + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the cheap baseline stage)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Baseline exploration (cheap, low dimension)
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate and keep if it improves."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0:
        return
    vec = vec / total
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 8) -> list[np.ndarray]:
    """Sweep several families and Dirichlet draws; return up to ``max_keep`` best."""
    best = []  # list of (score, vector)

    # 1) Power‑law
    for p in np.linspace(0.1, 6.0, 200):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # 2) Exponential
    for lam in np.linspace(0.0001, 0.2, 200):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # 3) Beta‑like
    al_vals = np.linspace(0.2, 8.0, 20)
    be_vals = np.linspace(0.2, 8.0, 20)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # 4) Gaussian
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 20)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # 5) Dirichlet draws (both raw and sorted descending)
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 400
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Upsampling (linear interpolation + simplex projection)
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``vec`` to ``new_n`` points and project onto simplex."""
    n = len(vec)
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0:
        return np.full(new_n, 1.0 / new_n)
    return up / s


# ----------------------------------------------------------------------
# Smooth Adam optimisation (soft‑max approximation of max)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.14,
                    init_tau_factor: float = 8.0,
                    min_tau: float = 1e-10,
                    tau_decay: float = 0.9996) -> np.ndarray:
    """
    Adam optimisation on the smoothed objective.
    Returns the best vector observed before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # log‑softmax parameters (tiny epsilon for safety)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature (smoothness)
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution of current a
        conv = full_convolution(a)
        max_c = conv.max()

        # smooth‑max weights
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient wrt a (g_i = 2·∑_k w_k·a_{k-i})
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for soft‑max parameters x
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature decay
        tau = max(min_tau, tau * tau_decay)

        # occasional exact evaluation
        if it % 400 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, deadline: float,
                        init_step: float = 0.08,
                        step_decay: float = 0.99995,
                        tol: float = 1e-12) -> np.ndarray:
    """
    Perform a few hundred iterations of sub‑gradient descent on the true
    max‑convolution objective.  The update is projected back onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    step = init_step
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        max_c = conv.max()
        # active set mask (allow a tiny tolerance)
        mask = conv >= max_c - tol
        if not np.any(mask):
            mask = np.array([True] + [False] * (conv.size - 1))
        # weight each active entry equally
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum

        # gradient of max (average over active points)
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # 3n‑2 length
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # gradient step
        a = a - step * grad_a
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # keep best
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[subgrad] iter {it:7d}  score {val:.6f}")

        step = max(1e-7, step * step_decay)

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9994,
                        step_scale_init: float = 0.025,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # propose a random transfer between two coordinates
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to escape tiny plateaux
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem sizes
    # -----------------------------------------------------------------
    n_small = 2048
    n_big = 8192

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (if needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        baseline_deadline = deadline - 650.0  # reserve most time for polishing
        if time.time() < baseline_deadline:
            print("[baseline] starting cheap global search")
            cand_vectors = baseline_search(n_small, baseline_deadline, max_keep=8)
            for vec in cand_vectors:
                val = compute_score(vec)
                if val < best_val:
                    best_val, best_seq = val, vec.tolist()
                    print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Upsample best (or uniform) to the large dimension
    # -----------------------------------------------------------------
    if best_seq is None:
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)

    a0_big = upsample(a0_small, n_big)

    # -----------------------------------------------------------------
    # 3. Smooth Adam optimisation (major phase)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 130.0
    if time.time() < polish_deadline:
        print("[smooth_opt] primary run on large dimension")
        refined = smooth_optimize(a0_big, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth_opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # Additional restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)

            # Random Dirichlet start (various concentrations) – sometimes monotone
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0])
            a_start = np.random.dirichlet(np.full(n_big, conc))
            if np.random.rand() < 0.35:
                a_start = np.sort(a_start)[::-1]

            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth_opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Exact sub‑gradient refinement
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 60.0
    if best_seq is not None and time.time() < subgrad_deadline:
        print("[subgrad] starting exact sub‑gradient refinement")
        a_start = np.array(best_seq, dtype=float)
        refined = subgradient_descent(a_start, subgrad_deadline)
        val = compute_score(refined)
        print(f"[subgrad] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        print("[anneal] final simulated‑annealing polish")
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Report official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
