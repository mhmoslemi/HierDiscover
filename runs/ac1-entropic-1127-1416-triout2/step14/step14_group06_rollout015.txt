        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)

def simulated_annealing(a_start, deadline):
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    n = a.shape[0]

    # max time = deadline - start
    while time.time() < deadline:

        # propose move
        i, j = np.random.randint(n, size=2)

        cand[j] += delta

        cand_score = compute_score(cand)

            if cand_score < best_score:

"""
Propose a candidate step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  The algorithm consists of three stages:

1. **Extended parametric sweep** – scans several low‑dimensional families
   (reverse power‑law, mixtures with the uniform distribution,
   geometric decay, beta‑shaped densities) on a fine grid.
   The best vector from this stage is kept as a seed.

2. **Projected sub‑gradient polishing** – starting from the seed it
   repeatedly reduces the maximum entry of the auto‑convolution by a
   projected sub‑gradient step (the gradient of the active constraints).
   This is the same idea as the reference optimiser, but with a slightly
   larger initial step and more aggressive jittering.

3. **Simulated‑annealing hill‑climb** – a stochastic move transfers a
   fraction of mass between two coordinates; moves that improve the
   objective are always accepted, otherwise they are accepted with a
   temperature‑dependent probability.  The temperature cools linearly
   to zero, allowing the search to escape shallow local minima.

All helpers are top‑level functions; the entry point required by the
harness is ``propose_candidate``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # return uniform distribution if the vector is all non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT (fast)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = rfft(a, n=n_fft)
    C = A * A
    conv = irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The original ``evaluate_sequence`` is scale‑invariant, therefore we
    keep Σa = 1 and return the same value.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law), p can be any real."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}  (decreasing algebraic)."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  with 0 < q < 1."""
    if not (0.0 < q < 1.0):
        return None
    a = q ** np.arange(n, dtype=np.float64)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """mix * reverse_power_law(p) + (1‑mix) * uniform, projected onto the simplex."""
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a


def mixed_reverse_geometric(n: int, mix: float, p: float, q: float) -> np.ndarray:
    """mix * reverse_power_law(p) + (1‑mix) * geometric(q)."""
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    geom = geometric(n, q)
    if rp is None or geom is None:
        return None
    a = mix * rp + (1.0 - mix) * geom
    a = project_to_simplex(a)
    return a

# Sub‑gradient polishing (same principle as the reference)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # active constraints (indices where conv is essentially maximal)
        active = np.where(conv >= M - 1e-12)[0]

        # gradient: sum of shifted copies of a over the active set
        g = np.zeros_like(a)
        for k in active:
            # contributions for i where (k-i) in [0, n-1]
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            # vectorised slice
            idx_i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - idx_i
            g[idx_i] += 2.0 * a[j]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        # optional jitter to escape plateaus
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb (stochastic mass transfer)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """Stochastic hill‑climb with a linear cooling schedule."""
    a = a_start.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    start = time.time()
    total_time = deadline - start
    while time.time() < deadline:
        # linear cooling (temperature in [0.02, 0])
        elapsed = time.time() - start
        frac = elapsed / total_time
        T = (1.0 - frac) * 0.02 + 1e-12

        # propose a mass transfer
        i, j = np.random.randint(n, size=2)
        if a[i] <= 0.0:
            continue
        # Move up to 50 % of the source mass (larger steps early)
        delta = np.random.rand() * 0.5 * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        # Accept if better, otherwise with Boltzmann probability
        if cand_score < cur_score or np.random.rand() < math.exp((cur_score - cand_score) / T):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin to avoid hitting the hard timeout
    deadline = start_time + budget_s - 4.0

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0. Optional fallback (if a previously discovered construction exists)
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Choose problem size (larger n gives a finer discretisation)
    # -------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 500:
        n = 8192
    elif remaining > 250:
        n = 4096
    else:
        n = 2048
    print(f"[setup] using n = {n}")

    # -------------------------------------------------------------
    # 2. Broad parametric sweep
    # -------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep_extended(n, sweep_deadline)
        if a_initial is not None:
            # official score (still scale‑invariant)
            val = evaluate_sequence(a_initial.tolist())
            print(f"[parametric sweep] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -------------------------------------------------------------
    # 3. Sub‑gradient polishing (main workhorse)
    # -------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.15)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # 4. Simulated‑annealing final refinement
    # -------------------------------------------------------------
    final_deadline = deadline - 2.0
    if best_seq is not None and time.time() < final_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, final_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[annealing] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # 5. Safeguard – fall back to a simple mixed shape if nothing else
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = mixed_reverse_uniform(n, mix=0.8, p=1.7)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final report
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
