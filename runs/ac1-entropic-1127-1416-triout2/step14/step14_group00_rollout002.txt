"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈ 100 s total):
1. Try several lengths (1024 … 16384).  For each length generate the
   discrete arcsine (Beta(½,½)) distribution – its self‑convolution is
   asymptotically uniform, so the objective 2·n·max(conv) approaches the
   theoretical lower bound 1 as n grows.
2. For each length also test a few simple convex mixtures with the uniform
   vector and a few tiny random perturbations; keep the best candidate.
3. Take the best candidate found and run a fast projected Adam optimiser
   that minimises the smoothed max‑convolution (soft‑max with a temperature
   parameter).  The Adam step is projected back onto the probability
   simplex after each iteration.
4. Finish with a short sub‑gradient polishing (identical to the previous
   baseline) to clean up any remaining peaks.
5. Return the final sequence as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape.
    a_i ∝ 1 / √((i+½)*(n‑½‑i)), i = 0 … n‑1.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=float) + 0.5                # centre of each bin
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


def generate_uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------


def weighted_gradient(a: np.ndarray, p: float) -> np.ndarray:
    """
    Returns the gradient of the soft‑max approximation
        f(a) = logsumexp(p * conv(a,a)) / p
    multiplied by the constant 2 (derivative of a_i a_j term).
    The result is a vector of length n.
    """
    n = a.shape[0]
    # full convolution a*a
    conv = full_convolution(a)
    max_c = conv.max()
    # soft‑max weights (stable)
    w = np.exp(p * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                # now w is a probability vector

    # we need g_i = 2 * sum_j a_j * w_{i+j}
    # i+j runs over the convolution length L = 2n-1
    # Compute via FFT: g = 2 * (a * w_rev)[2n-2 - i]
    w_rev = w[::-1]
    L = len(w_rev)                # = 2n-1
    n_fft = 1 << ((n + L - 1 - 1).bit_length())   # enough for length n+L-1
    Af = np.fft.rfft(a, n=n_fft)
    Wf = np.fft.rfft(w_rev, n=n_fft)
    conv_aw = np.fft.irfft(Af * Wf, n=n_fft)[: n + L - 1]   # length 3n-2
    # pick the slice that gives g_i
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * conv_aw[idx]
    return g


def adam_optimize(a0: np.ndarray, deadline: float,
                  lr: float = 0.02, p_start: float = 5.0,
                  p_end: float = 30.0,
                  beta1: float = 0.9, beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimiser on the soft‑max of the convolution.
    The temperature parameter p is increased linearly from p_start to p_end,
    tightening the approximation to the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    t = 0
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        t += 1
        # linearly increase p during the run
        p = p_start + (p_end - p_start) * (t / 200000.0)
        if p > p_end:
            p = p_end

        g = weighted_gradient(a, p)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to shake out plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (lr * 0.005)
            a += jitter
            a = project_to_simplex(a)

    return best_a

# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the exact objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # gradient of max conv = sum of gradients of the active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

        if time.time() >= deadline:
            break
        print(f"[scan] length = {n}")

        # base arcsine
        a = generate_arcsine(n)

        # a few simple mixtures with uniform to get a quick lower bound
        uniform = generate_uniform(n)
        for lam in np.linspace(0.0, 0.5, 11):          # lam = weight of uniform
            mixed = lam * uniform + (1.0 - lam) * a
            mixed = project_to_simplex(mixed)
            sc = compute_score(mixed)
            if sc < best_score:
                best_score, best_seq = sc, mixed.copy()
        print(f"   best after mixtures (score={best_score:.6f})")

        # a handful of random perturbations around the arcsine shape
        for _ in range(5):
            if time.time() >= deadline:
                break
            noise = np.random.randn(n) * 0.03
            pert = a * (1.0 + noise)
            pert = np.maximum(pert, 0.0)
            pert = project_to_simplex(pert)
            sc = compute_score(pert)
            if sc < best_score:
                best_score, best_seq = sc, pert.copy()

    # -----------------------------------------------------------------
    # 2. Adam‑based smoothing optimisation on the best candidate found
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[adam] start")
        adam_deadline = deadline - 10.0   # reserve time for polishing & eval
        a_adam = adam_optimize(best_seq, adam_deadline,
                               lr=0.02, p_start=5.0, p_end=30.0)
        sc = compute_score(a_adam)
        print(f"[adam] refined score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_adam.copy()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (final clean‑up)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[polish] start")
        polish_deadline = deadline - 2.0
        a_polished = subgradient_polish(best_seq, polish_deadline,
                                        step0=0.20)
        sc = compute_score(a_polished)
        print(f"[polish] final polished score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_polished.copy()

    # -----------------------------------------------------------------
    # 4. Safeguard – fallback to a uniform vector if for any reason we have none
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 2048
        best_seq = generate_uniform(n)
        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # 5. Official evaluation (clamping etc. done inside evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")

    return best_seq.tolist()
