"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s budget)
--------------------------------
1. **Parameter sweep** – try a fine grid of Beta‑type shapes
   (including the arcsine distribution) and keep the best one.

2. **Smooth max optimisation** – minimise a smoothed version of the
   ‑∞‑norm of the self‑convolution:
        f(a) = (1/β)·log ∑ exp(β·(a*a))
   using Adam on the simplex (projected by simple renormalisation).
   The gradient of the smoothed max can be computed by a single extra
   convolution, so each iteration costs only two FFT‑convolutions.

3. **Short simulated‑annealing fine‑tune** – a classic “move mass from
   i to j’’ Metropolis step that can escape shallow minima.

All steps respect the 1000 s wall‑clock; the best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT based convolutions)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """
    Full linear convolution a * b using real FFT.
    If b is None the function returns a*a (self‑convolution).
    """
    if b is None:
        b = a
    n = a.shape[0]
    m = b.shape[0]
    size = n + m - 1
    # next power‑of‑two for efficient FFTs
    n_fft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:size]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (without the
    clamping of the latter).  a is assumed to be normalised to sum = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)          # a * a, length 2n‑1
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple simplex projection (clamp‑and‑renormalise – fast)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1},  α,β>0."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape.
    a_i ∝ 1 / √((i+½)*(n‑½‑i)), i=0…n‑1,
    then normalised to a probability vector.
    """
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (Adam) on the simplex
# ----------------------------------------------------------------------


def smooth_max_gradient(a: np.ndarray, beta: float):
    """
    Returns (grad, smoothed_max) where grad is the gradient of the
    smooth max (log‑sum‑exp) of the self‑convolution and smoothed_max
    is the actual smooth approximation of max(conv).
    """
    n = a.shape[0]
    conv = full_convolution(a)               # length 2n‑1
    cmax = conv.max()
    # log‑sum‑exp trick for numerical stability
    exps = np.exp(beta * (conv - cmax))
    sum_exps = exps.sum()
    w = exps / sum_exps                       # weights, sum to 1

    # gradient = 2 * (w * a) convolution with reversed a
    a_rev = a[::-1]
    conv_wa = full_convolution(w, a_rev)     # length (2n‑1)+n‑1 = 3n‑2
    grad = 2.0 * conv_wa[n - 1 : n - 1 + n]   # slice to length n
    smooth_max = cmax + np.log(sum_exps) / beta
    return grad, smooth_max


def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  lr: float = 0.04,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  beta_start: float = 10.0,
                  beta_end: float = 50000.0,
                  beta_multiplier: float = 1.06) -> np.ndarray:
    """
    Adam optimiser that minimises the smooth‑max of a*a.
    The optimisation stops when the time budget expires.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0
    beta = beta_start
    best_a = a.copy()
    best_score = compute_score(a)

    while time.time() < deadline:
        t += 1
        grad, smooth_val = smooth_max_gradient(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # keep the best (exact) score seen so far
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[adam] iter {t:6d}  score {best_score:.6f}")

        # increase the smoothing parameter (approach the true max)
        if beta < beta_end:
            beta = min(beta_end, beta * beta_multiplier)

    return best_a


# ----------------------------------------------------------------------
# Small simulated‑annealing fine‑tune (optional)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling: float = 0.9994,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """A brief Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val
        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling)

    return best_a
