"""
Propose low‑autocorrelation step heights via a three‑phase optimizer.

Phase 1 – Adam‑style sub‑gradient descent.
Phase 2 – Multiplicative‑weights (exponential) update.
Phase 3 – Final polishing with a simple projected sub‑gradient method
         (identical to the baseline “polish” routine).

All iterates stay on the probability simplex (sum = 1, non‑negative),
so the evaluation score returned by ``evaluate_sequence`` is
simply 2·n·max(conv) and can be compared cheaply with ``compute_score``.
The routine respects the 1000 s budget and falls back to a uniform
distribution if nothing improves.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability‑normalised vector:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray,
                        conv: np.ndarray,
                        eps_rel: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Returns a (sub‑)gradient of f(a)=max(conv(a,a)) and the current max value.
    The sub‑gradient is the sum of the gradients of all active shifts.
    """
    n = a.shape[0]
    max_val = conv.max()
    eps = max_val * eps_rel
    active = np.where(conv >= max_val - eps)[0]

    # vectorised loop over active shifts
    idx = np.arange(n)
    g = np.zeros_like(a)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]
    return g, max_val


# ----------------------------------------------------------------------
# Adam‑style optimisation on the sub‑gradient
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  lr: float = 0.10,
                  beta1: float = 0.90,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the objective f(a)=max(conv(a,a)).
    Returns the best vector found and its score (2·n·max(conv)).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        g, _ = compute_subgradient(a, conv)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam] iter {t:6d} score {cur_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Multiplicative‑weights optimiser
# ----------------------------------------------------------------------


def multiplicative_opt(start_a: np.ndarray,
                       deadline: float,
                       eta0: float = 0.50) -> tuple[np.ndarray, float]:
    """
    Exponential‑weights update: a_i ← a_i·exp(−η·g_i) with simplex projection.
    Returns the best vector seen and its score.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        g, _ = compute_subgradient(a, conv)

        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)

        total = a.sum()
        if total <= 0 or not np.isfinite(total):
            # degenerate – reset to uniform
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / total

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[MW]   iter {t:6d} score {cur_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Final projected sub‑gradient polishing (identical to baseline)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray,
                       deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the exact objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

        print(f"[Adam] evaluated score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, seq_adam

    # --------------------------------------------------------------
    # 2. Multiplicative‑weights optimisation (≈ 30 % of remaining budget)
    # --------------------------------------------------------------
    mw_deadline = deadline - 150.0
    if time.time() < mw_deadline:
        print("[MW] start")
        a_mw, _ = multiplicative_opt(np.array(best_seq, dtype=float), mw_deadline)
        seq_mw = a_mw.tolist()
        val = evaluate_sequence(seq_mw)
        print(f"[MW] evaluated score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, seq_mw

    # --------------------------------------------------------------
    # 3. Final polishing with projected sub‑gradient descent
    # --------------------------------------------------------------
    polish_deadline = deadline - 5.0
    if time.time() < polish_deadline:
        print("[polish] start")
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[polish] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # --------------------------------------------------------------
    # 4. Safeguard – uniform vector if nothing succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
