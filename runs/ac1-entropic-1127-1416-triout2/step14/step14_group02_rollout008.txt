Implementation of LP iteration:

def lp_refine(a, t, deadline) -> a_refined:

    while time left > small:

        M = max(conv(a,a))

        # Build constraint matrix A

        n = a.shape[0]; m = 2*n-1

        A = np.zeros((m,n))

        for i in range(n):
            start = i
            end = i + n
            # rows start..end-1 correspond to conv index i + j for column i.
            A[start:end, i] = a[i]

        # Solve LP: maximize sum(b) subject to A @ b <= M (vector)
        c = -np.ones(n)  # minimize -sum(b)
        b_ub = np.full(m, M)
        bounds = [(0, None)] * n

        # solve using scipy.optimize.linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        b_norm = b / sum_b
        a = (1 - t) * a + t * b_norm
        a = project_to_simplex(a) # Actually ensures sum=1
        # track improvement
        cur_val = compute_score(a)
        # If no improvement in M (or cur_val) after some loops, break

        if time.time() > deadline:
            break

    return a

"""
Improved optimiser for the “min‑max auto‑convolution” problem.

Main ideas
----------
* Start from a cheap global search over analytic families and many Dirichlet
  draws (baseline phase).  Keep the top few candidates.
* Refine the current best candidate by the classic “LP‑alternating” step:
  – For the fixed vector f compute M = max(f∗f).
  – Solve a linear programme that maximises ∑b j subject to the linear
    constraints (f∗b) ≤ M and b j ≥ 0.
  – Normalise the LP‑solution to unit sum and mix it with f.
  Repeating this (t‑mixing) monotonically drives the maximum convolution
  down.
* Finish with a short projected sub‑gradient polish (exact max‑objective)
  and a final sanity check via the official ``evaluate_sequence`` routine.

The whole pipeline respects the 1000 s wall‑clock limit, continually prints
intermediate scores and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import i0  # for Kaiser window


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact objective under the normalisation sum(a)=1:
#    score = 2 * n * max(conv(a,a))
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective used for internal comparisons (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used inside the cheap baseline search)
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def generate_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def generate_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def generate_kaiser(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    r = np.sqrt(1.0 - (2.0 * i / (n - 1)) ** 2)
    a = i0(beta * r) / i0(beta)
    return a / a.sum()


# ----------------------------------------------------------------------
# Cheap global baseline exploration (many families + Dirichlet draws)
# ----------------------------------------------------------------------
def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 8,
    random_draws: int = 15000,
) -> list[np.ndarray]:
    """Explore analytic families and random Dirichlet draws, keep the best ones."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # Analytic families -------------------------------------------------
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.5, 100):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    al_vals = np.linspace(0.02, 2.0, 25)
    be_vals = np.linspace(0.02, 2.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    for sigma in np.linspace(0.1 * n, 0.45 * n, 20):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # Classical windows -------------------------------------------------
    for _ in range(12):
        if time.time() >= deadline:
            break
        try_candidate(generate_hann(n))
        try_candidate(generate_hamming(n))
        try_candidate(generate_blackman(n))
        try_candidate(generate_kaiser(n, beta=np.random.uniform(0.5, 10.0)))

    # Random Dirichlet draws --------------------------------------------
    conc_levels = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(conc_levels))
    for conc in conc_levels:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet([conc] * n)
            try_candidate(vec)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# LP‑based refinement (the classic “alternating LP” step)
# ----------------------------------------------------------------------
def lp_refine(a0: np.ndarray, t: float, deadline: float) -> np.ndarray:
    """Iteratively improve a using the linear‑programming mixing scheme."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Pre‑allocate index arrays for the sparse constraint matrix
    # (will be rebuilt each iteration because the coefficients a change)
    while time.time() < deadline:
        conv = full_convolution(a)
        M = conv.max()

        # Build sparse matrix A where A[k, j] = a[k-j] if 0≤k-j<n else 0
        # Column j has a[j] on rows j … j+n‑1
        row_ind = np.empty(n * n, dtype=np.int32)
        col_ind = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)
        pos = 0
        for j in range(n):
            a_j = a[j]
            if a_j == 0.0:
                # skip adding zero entries (still need to advance pos)
                pos += n
                continue
            rows = np.arange(j, j + n, dtype=np.int32)
            row_ind[pos : pos + n] = rows
            col_ind[pos : pos + n] = j
            data[pos : pos + n] = a_j
            pos += n
        A = csr_matrix((data, (row_ind, col_ind)), shape=(m, n))

        # Linear programme: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(m, M, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # if the LP fails, give up on this refinement
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Normalise the LP solution and mix
        b_norm = b / sum_b
        a = (1.0 - t) * a + t * b_norm
        a = project_to_simplex(a)

        # Early stop if the improvement stalls
        new_conv = full_convolution(a)
        new_M = new_conv.max()
        if new_M >= M - 1e-12:  # no appreciable drop
            break

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective (polish)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # Sub‑gradient: sum over active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a
