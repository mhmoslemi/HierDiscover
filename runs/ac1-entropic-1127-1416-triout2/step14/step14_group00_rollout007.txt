```
def full_convolution_fft(a,b):
    n = len(a)
    m = len(b)
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    C = A * B
    conv = np.fft.irfft(C, size)[:n+m-1]
    return conv
```

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise a large‐scale discrete arcsine distribution
   (the Beta(½,½) law).  Its self‑convolution is asymptotically flat,
   giving an objective close to the theoretical lower bound 1.

2. Perform a long “smooth‑max’’ projected gradient descent:
   - Replace the hard max in the objective by a log‑sum‑exp
     (controlled by a temperature parameter γ).
   - The gradient of this smooth objective is a weighted
     convolution that can be computed with two real FFTs per
     iteration (O(n log n)).
   - After each gradient step we project back onto the probability
     simplex (non‑negative entries summing to 1).
   - γ is increased slowly so the optimisation first moves
     globally and then focuses on the true peak.

3. Finish with a few iterations of classic sub‑gradient polishing
   (using only the truly active constraints) to squeeze out the
   remaining slack.

All steps respect a 1000 s wall‑clock limit; the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        # uniform fallback
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------


def full_convolution_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:m]
    return conv


def full_convolution_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    size = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    conv = np.fft.irfft(A * B, size)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring helper (exact objective for a simplex‑normalised vector)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a vector with sum=1."""
    n = a.shape[0]
    conv = full_convolution_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (arcsine – the continuous optimum)
# ----------------------------------------------------------------------


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape.
    a_i ∝ 1 / sqrt((i+½)*(n‑½‑i)), i = 0 … n‑1.
    The vector is normalised to lie on the simplex.
    """
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as in the reference implementation)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective 2·n·max(conv).
    Stops when the time deadline is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution_self(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # gradient of the max‑convolution = sum of gradients of active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to leave plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # small safety margin

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously stored solution
    # -----------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose a comfortable size (larger n gives a tighter lower bound)
    # -----------------------------------------------------------------
    n = 4096                                   # 4k entries – fast FFTs, good accuracy
    print(f"[setup] using n = {n}")

    # -----------------------------------------------------------------
    # 2. Deterministic strong start: the discrete arcsine distribution
    # -----------------------------------------------------------------
    a = generate_arcsine(n)
    cur_val = compute_score(a)
    print(f"[arcsine] initial score = {cur_val:.6f}")
    if cur_val < best_score:
        best_score, best_seq = cur_val, a.copy()

    # Small random jitter to break exact symmetries
    a += np.random.randn(n) * 1e-6
    a = project_to_simplex(a)

    # -----------------------------------------------------------------
    # 3. Smooth‑max projected gradient descent (main optimiser)
    # -----------------------------------------------------------------
    iteration = 0
    step0 = 0.25
    gamma = 5.0                     # start with a very smooth approximation
    gamma_max = 2000.0
    gamma_factor = 1.02             # increase slowly
    last_log = time.time()

    while time.time() < deadline:
        iteration += 1

        # ----- self‑convolution & smooth max weights -----
        conv = full_convolution_self(a)                     # length 2n‑1
        max_conv = conv.max()
        # stable log‑sum‑exp weights (w_i >= 0, sum w_i = 1)
        exp_terms = np.exp(gamma * (conv - max_conv))
        w_sum = exp_terms.sum()
        if w_sum == 0.0:
            w = np.full_like(exp_terms, 1.0 / exp_terms.size)
        else:
            w = exp_terms / w_sum

        # ----- gradient of the smooth objective -----
        a_rev = a[::-1]                                     # reversed a
        conv_w = full_convolution_two(w, a_rev)            # length 3n‑2
        # gradient = 2 * slice centred at position n‑1
        grad = 2.0 * conv_w[n - 1: n - 1 + n]

        # ----- projected gradient step -----
        step = step0 / np.sqrt(iteration)
        a -= step * grad
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        # ----- occasional logging & gamma update -----
        if iteration % 20 == 0:
            cur_val = compute_score(a)
            if cur_val < best_score:
                best_score, best_seq = cur_val, a.copy()
                print(f"[iter {iteration}] new best score = {best_score:.6f}")
            # increase the temperature parameter gradually
            if gamma < gamma_max:
                gamma = min(gamma_max, gamma * gamma_factor)

        # ----- occasional jitter to avoid stagnation -----
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(iteration))
            a += jitter
            a = project_to_simplex(a)

    # -----------------------------------------------------------------
    # 4. Final polishing with a classic sub‑gradient method
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[polish] start")
        polish_deadline = deadline - 2.0
        a_polished = subgradient_polish(np.array(best_seq, dtype=float),
                                        polish_deadline,
                                        step0=0.20)
        polished_score = compute_score(a_polished)
        print(f"[polish] refined score = {polished_score:.6f}")
        if polished_score < best_score:
            best_score, best_seq = polished_score, a_polished.copy()

    # -----------------------------------------------------------------
    # 5. Fallback safeguard – in the extremely unlikely case nothing
    #    succeeded we return a uniform vector.
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # 6. Official evaluation (the harness will apply the same checks)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq.tolist()
