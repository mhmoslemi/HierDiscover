def simulated_annealing(a0, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    cur_val = best_val
    n = a.shape[0]
    # compute initial gradient to bias moves
    while time.time() < deadline:
        # compute gradient for bias
        conv = full_convolution(a)
        M = conv.max()
        thr = max(M*1e-12, 1e-12)
        active = np.where(conv >= M - thr)[0]
        # compute gradient g same as before
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)

def annealing_hill_climb(a_start, deadline):

    best_a = a.copy()
    cur_val = compute_score(a)
    best_val = cur_val
    n = len(a)
    # initial temperature:
    T0 = 0.05 * cur_val  # maybe scale
    total_time = deadline - time.time()
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        # linear cooling:
        T = T0 * max(0.0, 1 - elapsed / total_time)
        # compute gradient for bias
        conv = full_convolution(a)
        M = conv.max()
        thr = max(M*1e-12, 1e-12)
        active = np.where(conv >= M - thr)[0]
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # convert to probabilities
        prob_i = g / (g.sum() + 1e-30)
        inv_g = 1.0 / (g + 1e-12)
        prob_j = inv_g / inv_g.sum()

        i = np.random.choice(n, p=prob_i)
        j = np.random.choice(n, p=prob_j)
        if i == j:

"""
propose_candidate
-----------------

A multi‑stage optimiser for the ``evaluate_sequence`` objective.

The algorithm consists of:

1. **Broad parametric sweep** over several families of
   analytically‑defined step‑functions (reverse power‑law,
   geometric, beta‑shape, and mixes).  The best vector from this
   sweep provides a solid starting point.

2. **Linear‑programming refinement**.
   For a down‑sampled version of the current vector we solve a
   linear programme (maximise the ℓ₁‑norm of a new vector g under the
   linear constraints (f * g)≤M, where M is the current maximal
   convolution value).  The solution is interpolated back to the
   original resolution and blended with the incumbent vector.

3. **Projected sub‑gradient polishing** – a classic descent on the
   max‑convolution using the active‑set sub‑gradient.

4. **Gradient‑biased simulated annealing / hill‑climb**.
   Moves are biased by the sub‑gradient (indices contributing most
   to the current peak are favoured for mass removal) and accepted
   according to a simple temperature schedule.

All heavy work is performed on a vector of length 4096 when the
time budget permits, otherwise 2048.  The routine respects the 1000 s
budget, falling back to a simple mixed shape if no candidate is
found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------


def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law), p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)
    a = idx ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    return a / a_sum


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def reverse_ends(n: int, p: float) -> np.ndarray:
    """Symmetric front‑loaded shape: 0.5·rev + 0.5·rev[::-1] (peaks at both ends)."""
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    a = 0.5 * rp + 0.5 * rp[::-1]
    a = project_to_simplex(a)  # guarantees sum=1
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """convex combination   mix·reverse_power_law + (1‑mix)·uniform."""
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    u = uniform_shape(n)
    a = mix * rp + (1.0 - mix) * u
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Parametric sweep – broader and finer
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Broad parametric sweep; returns the best discovered vector."""
    best_vec = None
    best_val = float("inf")

    # ----- reverse power‑law -----
    for p in np.arange(0.2, 8.0 + 1e-12, 0.2):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, float(p))
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ----- reverse‑ends (two peaks) -----
    for p in np.arange(0.2, 8.0 + 1e-12, 0.4):
        if time.time() >= deadline:
            return best_vec
        a = reverse_ends(n, float(p))
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ----- geometric -----
    for q in np.arange(0.80, 0.995 + 1e-12, 0.01):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, float(q))
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ----- mixed reverse‑uniform -----
    for mix in np.linspace(0.4, 0.95, 12):
        for p in np.linspace(0.2, 6.0, 30):
            if time.time() >= deadline:
                return best_vec
            a = mixed_reverse_uniform(n, mix, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- beta‑shape (wide exponent range) -----
    alphas = np.linspace(-1.0, 8.0, 30)
    betas = np.linspace(-1.0, 8.0, 30)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- random convex combos of a few basic shapes -----
    basic = [
        uniform_shape(n),
        reverse_power_law(n, 2.0),
        geometric(n, 0.95),
        reverse_ends(n, 2.0),
    ]
    for _ in range(1500):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basic))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basic))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Linear‑programming refinement (small size)
# ----------------------------------------------------------------------
def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_j   s.t.   (f * g)_k ≤ M  ∀k,   g ≥ 0.
    Returns the optimal g (not normalised).  Uses scipy.linprog.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Build A_ub where each row corresponds to a convolution index k.
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    # Objective: minimise -sum(g)  (equivalent to maximise sum(g))
    c = -np.ones(n, dtype=np.float64)
    b = np.full(rows, M, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if not res.success:
        return None
    return np.maximum(res.x, 0.0)


def lp_refine(f: np.ndarray, max_iter: int = 4) -> np.ndarray:
    """
    Repeatedly solve the LP described above and blend the result.
    The input and output are normalised (sum = 1).
    """
    a = f.copy()
    for _ in range(max_iter):
        M = full_convolution(a).max()
        g = solve_lp_max_sum(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum == 0.0:
            break
        g = g / g_sum                # normalise to sum = 1
        # convex blend – a small step towards the LP solution
        t = 0.12
        a = (1 - t) * a + t * g
        a = project_to_simplex(a)
    return a


def downsample(vec: np.ndarray, m: int) -> np.ndarray:
    """Linear interpolation down‑sample to length m."""
    n = len(vec)
    xs = (np.arange(n) + 0.5) / n
    xs_small = (np.arange(m) + 0.5) / m
    small = np.interp(xs_small, xs, vec)
    small = np.maximum(small, 0.0)
    small /= small.sum()
    return small


def upsample(vec_small: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation up‑sample to length n."""
    m = len(vec_small)
    xs_small = (np.arange(m) + 0.5) / m
    xs = (np.arange(n) + 0.5) / n
    big = np.interp(xs, xs_small, vec_small)
    big = np.maximum(big, 0.0)
    big /= big.sum()
    return big


# ----------------------------------------------------------------------
# Sub‑gradient polishing
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # active set (allow tiny tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient (sum of shifted copies of a over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape flat regions
        if t % 1800 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Gradient‑biased simulated annealing / hill‑climb
# ----------------------------------------------------------------------
def annealing_hill_climb(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Stochastic hill‑climb where moves are biased by the current
    sub‑gradient and accepted with a temperature schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    cur_val = compute_score(a)
    best_val = cur_val
    n = a.shape[0]

    # temperature schedule (linear cooling)
    start_time = time.time()
    total = max(deadline - start_time, 1e-6)
    T0 = 0.06 * cur_val  # initial temperature proportional to objective

    while time.time() < deadline:
        elapsed = time.time() - start_time
        T = T0 * max(0.0, 1.0 - elapsed / total)

        # compute sub‑gradient for bias
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # probability of picking a source index (high contribution → larger prob)
        if g.sum() > 0:
            prob_i = g / (g.sum() + 1e-30)
        else:
            prob_i = np.full(n, 1.0 / n)

        # probability of picking a destination index (low contribution → larger prob)
        inv_g = 1.0 / (g + 1e-12)
        prob_j = inv_g / inv_g.sum()

        i = np.random.choice(n, p=prob_i)
        j = np.random.choice(n, p=prob_j)
        if i == j:
            continue

        # propose moving a small fraction of the mass from i to j
        delta = np.random.rand() * 0.015 * a[i]  # up to ~1.5 % of source
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_val = compute_score(cand)

        # accept if improvement; otherwise with Boltzmann probability
        if cand_val < cur_val:
            a, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_a = cand_val, cand.copy()
        else:
            if np.random.rand() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
                a, cur_val = cand, cand_val

        # occasional global jitter when stuck
        if np.random.rand() < 0.001:
            jitter = np.random.randn(n) * 0.0006
            a = a + jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

            print(f"[parametric sweep] raw score = {raw:.6f}, official = {real:.6f}")
            if real < best_val:

    else:

    # -------------------------------------------------------------
