projection_onto_simplex(v):
  code from Duchi et al.:

  u = np.sort(v)[::-1]
  cssv = np.cumsum(u) - 1
  ind = np.arange(len(v)) + 1
  cond = u - cssv / ind > 0
  rho = ind[cond][-1]
  theta = cssv[cond][-1] / rho
  w = np.maximum(v - theta, 0)

Return w.

def pso_optimize(initial_vectors, best_vec, deadline):
   swarm_size = 30
   # Initialize positions: if best_vec is not None, include it.
   positions = []
   for i in range(swarm_size):
       if best_vec is not None and i==0:
           positions.append(best_vec.copy())
       else:
           a = np.random.dirichlet(np.ones(n))
           positions.append(a)
   positions = np.array(positions) # shape (swarm_size, n)
   velocities = np.zeros_like(positions)

   pbest = positions.copy()
   pbest_scores = np.array([compute_score(pos) for pos in positions])

   gbest_idx = np.argmin(pbest_scores)
   gbest = pbest[gbest_idx].copy()
   gbest_score = pbest_scores[gbest_idx]

   # PSO parameters
   w = 0.7
   c1 = c2 = 1.5

   while time.time() < deadline:
       for i in range(swarm_size):
           # update velocity
           r1 = np.random.rand()
           r2 = np.random.rand()
           velocities[i] = (w * velocities[i]
                            + c1 * r1 * (pbest[i] - positions[i])
                            + c2 * r2 * (gbest - positions[i]))
           # update position
           new_pos = positions[i] + velocities[i]
           # project onto simplex
           new_pos = project_onto_simplex(new_pos)
           positions[i] = new_pos

           # evaluate
           score = compute_score(new_pos)
           if score < pbest_scores[i]:
               pbest[i] = new_pos.copy()
               pbest_scores[i] = score

               # update global best
               if score < gbest_score:
                   gbest = new_pos.copy()
                   gbest_score = score

       # optional: shrink inertia weight over time

   return gbest, gbest_score

    # fallback
    fallback = globals().get('height_sequence_1')
    if fallback is not None:
        try:

   The particles are projected onto the simplex after each move.

3. **Smooth max optimisation** – Adam on a soft‑max approximation of the
   max‑convolution (identical to the original code but with a slower
   temperature decay and a longer horizon).

4. **Final Simulated‑Annealing polish** – a light Metropolis walk that
   moves mass between coordinates.

All stages respect the overall 1000 s wall‑clock budget.  The best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)); a is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random vector, sort it decreasingly, and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]               # decreasing
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical fix: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Particle Swarm Optimisation on the simplex
# ----------------------------------------------------------------------


def pso_optimize(start_vec: np.ndarray,
                 deadline: float,
                 swarm_size: int = 30,
                 inertia: float = 0.7,
                 c1: float = 1.5,
                 c2: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Simple PSO where every particle lives on the probability simplex.
    Returns the best vector encountered and its score.
    """
    n = start_vec.shape[0]
    # initialise positions (first particle = provided start)
    positions = np.empty((swarm_size, n), dtype=float)
    positions[0] = project_onto_simplex(start_vec.copy())
    for i in range(1, swarm_size):
        positions[i] = np.random.dirichlet(np.ones(n))

    velocities = np.zeros_like(positions)

    # personal bests
    pbest = positions.copy()
    pbest_scores = np.array([compute_score(p) for p in pbest])

    gbest_idx = int(np.argmin(pbest_scores))
    gbest = pbest[gbest_idx].copy()
    gbest_score = float(pbest_scores[gbest_idx])

    it = 0
    while time.time() < deadline:
        it += 1
        for i in range(swarm_size):
            # update velocity
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia * velocities[i]
                             + c1 * r1 * (pbest[i] - positions[i])
                             + c2 * r2 * (gbest - positions[i]))
            # move particle
            new_pos = positions[i] + velocities[i]
            # project back onto simplex
            new_pos = project_onto_simplex(new_pos)
            positions[i] = new_pos

            # evaluate
            sc = compute_score(new_pos)
            if sc < pbest_scores[i]:
                pbest[i] = new_pos.copy()
                pbest_scores[i] = sc
                if sc < gbest_score:
                    gbest = new_pos.copy()
                    gbest_score = float(sc)

        # optional linear decay of inertia to improve convergence
        # (not essential for the time budget)
        inertia = max(0.4, inertia * 0.9995)

    return gbest, gbest_score


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray,
                    deadline: float,
                    init_lr: float = 0.08,
                    init_tau_factor: float = 5.0,
                    min_tau: float = 1e-12,
                    tau_decay: float = 0.9995) -> np.ndarray:
    """
    Adam optimiser on a soft‑max approximation of the max‑convolution.
    The variables are logits ``x`` such that a = softmax(x).
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)                      # logits

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initial temperature from current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # ----- soft‑max mapping ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = full_convolution(a)
        max_c = conv.max()

        # ----- soft‑max weighting (smooth max) ----------
        if tau < 1e-15:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient wrt a (2·∑_k w_k·a_{k−i}) ----------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)    # length = 3·n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the logits ----------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature ----------
        tau = max(min_tau, tau * tau_decay)

        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                # occasional progress report
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            # occasional jitter to avoid stagnation
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 offers a good trade‑off between expressivity
    # and speed for the 1000 s budget.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")               # internal score = 2·n·max(conv)

    # -----------------------------------------------------------------
    # 0. Optional fallback (if a previous good solution is stored)
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            sc_official = evaluate_sequence(fb)
            sc_internal = compute_score(np.array(fb) / np.sum(fb))
            print(f"[fallback] official = {sc_official:.6f}, internal = {sc_internal:.6f}")
            if sc_internal < best_val:
                best_val, best_seq = sc_internal, fb
        except Exception as e:
            print("[fallback] error:", e)

    # -----------------------------------------------------------------
    # 1. Broad random baseline (≈ 150 s)
    # -----------------------------------------------------------------
    concs = [0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = 2500
    for conc in concs:
        alpha = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha)
            sc = compute_score(a)
            if sc < best_val:
                best_val, best_seq = sc, a.tolist()
    # monotone random vectors
    for _ in range(5000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        sc = compute_score(a)
        if sc < best_val:
            best_val, best_seq = sc, a.tolist()

    # parametric sweeps ------------------------------------------------
    # Exponential family (very fine scan)
    lam_vals = np.linspace(1e-4, 5e-3, 3000)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        a = generate_exponential(n, lam)
        sc = compute_score(a)
        if sc < best_val:
            best_val, best_seq = sc, a.tolist()

    # Power‑law family
    p_vals = np.linspace(0.05, 6.0, 3000)
    for p in p_vals:
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, p)
        sc = compute_score(a)
        if sc < best_val:
            best_val, best_seq = sc, a.tolist()

    # Beta family – coarser but covers asymmetric shapes
    al_vals = np.linspace(0.2, 10.0, 12)
    be_vals = np.linspace(0.2, 10.0, 12)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            sc = compute_score(a)
            if sc < best_val:
                best_val, best_seq = sc, a.tolist()

    # -----------------------------------------------------------------
    # 2. Particle Swarm Optimisation (≈ 250 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        start_vec = np.array(best_seq, dtype=float)
        gbest, gbest_sc = pso_optimize(start_vec, deadline - 250.0)
        if gbest_sc < best_val:
            best_val, best_seq = gbest_sc, gbest.tolist()
            print(f"[PSO] new best {gbest_sc:.6f}")

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation (≈ 200 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, deadline - 150.0,
                                  init_lr=0.06,
                                  init_tau_factor=4.0,
                                  min_tau=1e-12,
                                  tau_decay=0.9994)
        sc = compute_score(refined)
        print(f"[smooth‑opt] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

        # occasional random‑restart smooth optimisation while time permits
        while time.time() < deadline - 150.0:
            remaining = deadline - time.time()
            sub_deadline = time.time() + min(25.0, remaining)
            a_start = np.random.dirichlet(np.full(n, 1.0))
            refined = smooth_optimize(a_start, sub_deadline,
                                      init_lr=0.06,
                                      init_tau_factor=4.0,
                                      min_tau=1e-12,
                                      tau_decay=0.9994)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[smooth‑restart] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 100 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, deadline - 5.0)
        sc = compute_score(final)
        print(f"[anneal] final polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 6. Official evaluation (the harness will call ``evaluate_sequence``)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return best_seq
