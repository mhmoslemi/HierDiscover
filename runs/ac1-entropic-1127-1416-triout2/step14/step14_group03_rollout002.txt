"""
Improved step‑function search for the “max‑convolution” objective.

Algorithm outline
-----------------
1. **Broad baseline** – sweep a richer collection of parametric families
   (single & double exponential, single & double power‑law, beta
   distributions, plus many Dirichlet draws).  Keep the best few candidates.

2. **Upscaling** – linearly interpolate the current best vector to a larger
   dimension (8192).  A larger length gives more freedom to flatten the
   autocorrelation peak.

3. **Hard‑max exponentiated‑gradient descent** – directly minimise the
   non‑smooth objective  `max(a*a)` by following a sub‑gradient of the
   *active* convolution entry.  The update is performed in log‑space
   (mirror descent on the simplex) and is extremely cheap per iteration.

4. **Smoothed Adam** – a second phase that uses a soft‑max approximation of
   the max (temperature annealed to 1e‑9).  Adam on the log‑softmax
   parametrisation refines the solution found by the hard‑max phase.

5. **Light simulated‑annealing polish** – a short Metropolis hill‑climb
   to escape possible tiny plateaux.

The whole pipeline respects a 1000 s wall‑clock budget; intermediate best
vectors are kept and re‑initialised from random Dirichlet draws when time
allows.  The final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution (FFT based) – used throughout
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution of a with itself)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Objective (exact, assumes sum(a)=1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a) (sum(a) has been normalised to 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a simplex‑normalised vector)
# ----------------------------------------------------------------------


def _normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    return _normalise(arr)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (single‑sided exponential)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    return _normalise(arr)


def generate_double_exponential(n: int, lam: float) -> np.ndarray:
    """U‑shaped exponential: a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=float)
    dist = np.minimum(i, n - 1 - i)
    arr = np.exp(-lam * dist)
    return _normalise(arr)


def generate_powerlaw_double(n: int, p: float) -> np.ndarray:
    """Sum of a decreasing and an increasing power‑law, i.e. U‑shaped."""
    i = np.arange(1, n + 1, dtype=float)
    left = i ** (-p)
    right = (n - i + 1) ** (-p)
    arr = left + right
    return _normalise(arr)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        arr = np.zeros_like(arr)
    return _normalise(arr)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return _normalise(arr)


def generate_dirichlet(n: int, conc: float) -> np.ndarray:
    """Draw from Dirichlet(alpha=conc·1)."""
    vec = np.random.dirichlet(np.full(n, conc))
    return _normalise(vec)


# ----------------------------------------------------------------------
# Baseline search (keeps the best `max_keep` vectors)
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Evaluate a candidate and keep it if it belongs to the best list."""
    if not np.isfinite(vec).all():
        return
    vec = _normalise(vec)
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """
    Sweep a variety of parametric families and Dirichlet draws.
    Return up to `max_keep` best vectors found.
    """
    best = []  # list of (score, vector)

    # 1) power‑law
    for p in np.linspace(0.1, 6.0, 200):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # 2) exponential (single)
    for lam in np.linspace(0.0001, 0.3, 200):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # 3) double exponential (U‑shaped)
    for lam in np.linspace(0.001, 0.5, 150):
        if time.time() >= deadline:
            break
        _try_candidate(generate_double_exponential(n, lam), best, max_keep)

    # 4) double power‑law (U‑shaped)
    for p in np.linspace(0.2, 4.0, 150):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw_double(n, p), best, max_keep)

    # 5) beta distributions (including U‑shaped)
    al_vals = np.linspace(0.15, 1.5, 12)
    be_vals = np.linspace(0.15, 1.5, 12)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # 6) Gaussian families
    for sigma in np.linspace(n / 64.0, n / 2.0, 12):
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # 7) Dirichlet draws (including sorted descending version)
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per = 400
    for conc in concs:
        for _ in range(draws_per):

            _try_candidate(a, best, max_keep)
            # monotone version – sort descending (edge‑heavy)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Upsampling (linear interpolation + simplex projection)
# ----------------------------------------------------------------------


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `vec` to length `new_n` and re‑project onto simplex."""
    n = vec.size
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, vec)
    return _normalise(up)

# Optimiser 1 – hard‑max exponentiated gradient (sub‑gradient of max)
# ----------------------------------------------------------------------


def hardmax_optimize(a0: np.ndarray, deadline: float,
                     init_eta: float = 0.15,
                     eta_decay: float = 0.999995,
                     min_eta: float = 1e-6,
                     report_every: int = 1000) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution.
    Uses mirror descent on the simplex (log‑space multiplicative update).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    eta = init_eta
    eps = 1e-30

    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        k = int(conv.argmax())               # active index
        # gradient g_i = 2 * a_{k-i}   (if 0≤k-i<n)
        ks = k - np.arange(n)
        mask = (ks >= 0) & (ks < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ks[mask]]

        # log‑space update (equivalent to exponentiated gradient)
        log_a = np.log(a + eps)
        log_a -= eta * grad
        # re‑softmax
        max_log = np.max(log_a)
        exp_log = np.exp(log_a - max_log)
        a = exp_log / exp_log.sum()

        # decay step size
        eta = max(min_eta, eta * eta_decay)

        if it % report_every == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[hardmax] iter {it:7d}  score {val:.6f}")

    return best_a

def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 10.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9995,
                    report_every: int = 300) -> np.ndarray:
    """
    Adam optimiser on a soft‑max approximation of max(a*a).
    The temperature τ is annealed to `min_tau`.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)               # logits for soft‑max

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = full_convolution(a)
        max_c = conv.max()

        # soft‑max weights over convolution entries
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient wrt a  (g_i = 2·∑_k w_k·a_{k-i})
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain‑rule for logits
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        tau = max(min_tau, tau * tau_decay)

        if it % report_every == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6,
                        report_every: int = 4000) -> np.ndarray:
    """Metropolis hill‑climb with occasional jitter."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)

    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % report_every == 0:
            # tiny random jitter to jump out of plateaus
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

    baseline_deadline = start_time + 180.0
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n_small, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Upsample the current best (or fallback) to the larger dimension
    # ------------------------------------------------------------------
    if best_seq is None:
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)

    a0_big = upsample(a0_small, n_big)

    # ------------------------------------------------------------------
    # 3. Hard‑max exponentiated‑gradient descent (≈ 400 s)
    # ------------------------------------------------------------------
    hardmax_deadline = start_time + 180.0 + 400.0
    if time.time() < hardmax_deadline:
        refined = hardmax_optimize(a0_big, hardmax_deadline)
        val = compute_score(refined)
        print(f"[hardmax] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # Additional random restarts while time permits
        while time.time() < hardmax_deadline:
            # random Dirichlet start (different concentrations)
            conc = np.random.choice([0.03, 0.07, 0.15, 0.3, 0.6])
            a_start = generate_dirichlet(n_big, conc)
            # occasionally enforce symmetry (mirror)
            if np.random.rand() < 0.3:
                a_start = (a_start + a_start[::-1]) / 2.0
            refined = hardmax_optimize(a_start, hardmax_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[hardmax‑restart] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Smoothed Adam refinement (≈ 300 s)
    # ------------------------------------------------------------------
    adam_deadline = start_time + 180.0 + 400.0 + 300.0
    if time.time() < adam_deadline:
        a_start = np.array(best_seq, dtype=float) if best_seq is not None else a0_big
        refined = smooth_optimize(a_start, adam_deadline)
        val = compute_score(refined)
        print(f"[adam‑refine] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # quick restarts (short runs) while time remains
        while time.time() < adam_deadline:
            sub_deadline = min(adam_deadline, time.time() + 30.0)
            a_start = generate_dirichlet(n_big, np.random.choice([0.05, 0.1, 0.2]))
            if np.random.rand() < 0.4:
                a_start = np.sort(a_start)[::-1]  # edge‑heavy
            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[adam‑restart] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing polish (≈ remaining time)
    # ------------------------------------------------------------------
    anneal_deadline = deadline
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # ------------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7. Report the official evaluation
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
