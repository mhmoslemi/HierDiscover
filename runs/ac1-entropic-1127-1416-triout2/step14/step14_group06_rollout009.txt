def gradient_step(y, n, alpha):
    a = softmax(y)
    conv = full_convolution(a)
    max_corr = conv.max()
    # compute w smooth max
    exp_alpha = np.exp(alpha * (conv - max_corr))
    w = exp_alpha / exp_alpha.sum()
    # compute correlation
    # using fftconvolve
    s = fftconvolve(w, a[::-1], mode='valid')
    g = s   # gradient w.r.t a (scaled)
    # gradient w.r.t y
    ga = np.dot(g, a)
    grad_y = a * (g - ga)
    return grad_y, a, conv.max()

"""
search_step_function_adam.py

An alternative optimiser for the ``evaluate_sequence`` objective.
It combines a lightweight parametric sweep (to obtain a good
starting shape) with a large‑scale projected Adam optimisation
on a smoothed version of the “max‑convolution” objective.
The variables are kept on the probability simplex via a
soft‑max parametrisation, which guarantees non‑negativity
and Σa = 1 at every iteration.  The smoothed max is realised
by a soft‑max over the full convolution with a continuation
schedule for the temperature α, allowing the optimiser to
first see a smooth landscape and then focus on the true
sup‑norm.  The method runs until the time budget expires and
returns the best sequence ever seen (according to the official
``evaluate_sequence`` function)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → probability simplex)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a * a using a real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for an inexpensive coarse sweep)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """Front‑loaded decreasing power‑law: a_i ∝ (n‑i)^{p}, p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    if a.sum() == 0.0:
        return None
    return a / a.sum()


def uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Fast correlation (valid mode) – needed for the gradient
# ----------------------------------------------------------------------
def correlation_valid(w: np.ndarray, a: np.ndarray, N_fft: int) -> np.ndarray:
    """
    Returns np.convolve(w, a[::-1], mode='valid') using the FFT.
    Length of the result is len(a).
    """
    a_rev = a[::-1]
    # FFT of the two signals
    W = np.fft.rfft(w, n=N_fft)
    A = np.fft.rfft(a_rev, n=N_fft)
    C = W * A
    full = np.fft.irfft(C, n=N_fft)[:len(w) + len(a_rev) - 1]
    start = len(a_rev) - 1
    return full[start:start + len(a)]


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def adam_optimize(y0: np.ndarray,
                  n: int,
                  deadline: float,
                  init_alpha: float = 0.5,
                  max_alpha: float = 200.0,
                  lr: float = 0.03,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  eval_every: int = 500) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the objective
        f(a) = 2·n·(1/α)·log Σ exp(α·(a∗a)_k)
    where a = softmax(y).  The function returns the best vector a
    encountered (in the sense of the official ``evaluate_sequence``)
    together with its raw score 2·n·max(conv).
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # FFT size needed for the correlation (valid) step
    corr_len = (2 * n - 1) + n - 1                # = 3n‑2
    N_fft_corr = 1 << ((corr_len - 1).bit_length())

    best_a = None
    best_raw = float('inf')
    best_official = float('inf')
    t = 0
    start = time.time()

    while time.time() < deadline:
        t += 1

        # ---- soft‑max parametrisation ----
        a = softmax(y)

        # ---- convolution of a with itself ----
        conv = full_convolution(a)                # length 2n‑1
        conv_max = conv.max()

        # ---- continuation for α ----
        frac = (time.time() - start) / (deadline - start)
        alpha = init_alpha + frac * (max_alpha - init_alpha)

        # ---- smooth max (soft‑max over convolution entries) ----
        # shift by the max to avoid overflow
        shift = conv - conv_max
        exp_alpha = np.exp(alpha * shift)
        w = exp_alpha / exp_alpha.sum()           # weights, sum to 1

        # ---- gradient of f w.r.t a (up to a constant factor) ----
        # g_i = Σ_k w_k a_{k-i}  (valid correlation)
        s = correlation_valid(w, a, N_fft_corr)   # shape (n,)
        g = s                                      # proportional to ∂f/∂a

        # ---- gradient w.r.t y (soft‑max Jacobian) ----
        ga = np.dot(g, a)                          # scalar ⟨g,a⟩
        grad_y = a * (g - ga)                      # ∂f/∂y

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- periodic evaluation with the official scorer ----
        if t % eval_every == 0:
            raw_score = 2.0 * n * conv_max
            if raw_score < best_raw:
                best_raw = raw_score
                best_a = a.copy()
                # official evaluation (expensive, but cheap enough for a few dozen checks)
                try:
                    official = evaluate_sequence(a.tolist())
                except Exception:
                    official = float('inf')
                if official < best_official:
                    best_official = official
                    print(f"[adam] iter {t:6d} raw={raw_score:.6f} official={official:.6f}")

    # final fallback if no evaluation ever happened
    if best_a is None:
        best_a = softmax(y)
        best_raw = 2.0 * n * full_convolution(best_a).max()
        best_official = evaluate_sequence(best_a.tolist())

    return best_a, best_official

    """
    Very cheap sweep over reverse power‑law exponents.

    """
    best = None

    # coarse grid – enough to locate a promising region quickly
    for p in np.arange(0.5, 6.0 + 1e-12, 0.2):

    return best
