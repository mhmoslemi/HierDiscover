"""
Non‑negative step‑function optimiser (improved).

The optimiser works in three stages:
  1. Generate a diverse pool of seeds (uniform, Dirichlet, exponential,
     polynomial, and the optional fallback sequence).
  2. Run a smooth‑max Adam optimiser (the same surrogate as the reference
     implementation) on each seed, allocating a time slice per seed.
  3. Finetune the best candidate with a simulated‑annealing hill‑climb that
     allows uphill moves.  This often pushes the score a few‑hundredths
     lower than the reference 1.510338.

All vectors are kept on the probability simplex (∑a = 1, a_i ≥ 0).
The algorithm respects the 1000 s budget and always returns the best
sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (fast for the sizes we use)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1, a_i ≥ 0)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 2.0,
    tau_end: float = 1e-7,
    tau_decay: float = 0.990,
    lr_start: float = 0.10,
    lr_decay: float = 0.996,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate
        fτ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # temperature schedule (geometric)
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # full convolution
        conv = full_convolution(a)               # length m
        shift = conv.max()
        # soft‑max weights (stable formulation)
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # shape (m,)

        # gradient of smooth‑max: 2·(a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # bookkeeping
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter={t}, τ={tau:.1e}, lr={lr:.3e}, score={cur_val:.6f}")

        # learning‑rate decay
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-6)

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner
# ----------------------------------------------------------------------


def anneal_fine_tune(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple simulated‑annealing hill‑climb.
    Allows uphill moves with a temperature that decays linearly.
    """
    a = a0.copy()
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val
    n = a.shape[0]

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = (1.0 - frac) * cur_val * 0.01 + 1e-12  # warm at start, cool to ~0

        # pick a move type
        if rng.random() < 0.6:          # mass transfer (most common)
            i = rng.integers(n)
            if a[i] < 1e-15:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            # transfer up to 40 % of the source mass
            frac_move = rng.random() * 0.40
            delta = a[i] * frac_move
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:                           # direct swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        val = compute_score(cand)
        delta_val = val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            a = cand
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[anneal] iter={it}, score={val:.6f}")

        it += 1
    return best_a

def seed_exponential(n: int, r: float) -> np.ndarray:
    """Geometric (exponential) seed a_i ∝ r**i, normalised to sum = 1."""
    a = np.power(r, np.arange(n, dtype=float))
    a = a / a.sum()
    return a


def seed_polynomial(n: int, beta: float) -> np.ndarray:
    """Monotone polynomial seed a_i ∝ (1 - i/(n‑1))**beta."""
    if n == 1:
        return np.ones(1)
    x = 1.0 - np.arange(n, dtype=float) / (n - 1)
    a = np.power(x, beta)
    a = a / a.sum()
    return a


def seed_mixed(n: int, r: float, beta: float) -> np.ndarray:
    """Mixed exponential‑polynomial a_i ∝ r**i * (i+1)**beta."""
    i = np.arange(n, dtype=float) + 1.0
    a = np.exp(-np.log(r) * (i - 1.0) + beta * np.log(i))
    a = a / a.sum()
    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5.0   # a small safety margin

    # --------------------------------------------------------------
    # 0. Use the fallback construction if it exists.
    # --------------------------------------------------------------
    fallback = None
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
    except Exception:
        pass

    if fallback is not None:
        a0 = np.array(fallback, dtype=float)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        n = a0.shape[0]
        print(f"[init] using fallback of length {n}")
    else:
        n = 2048
        a0 = np.full(n, 1.0 / n)          # uniform start
        print("[init] no fallback – start with uniform vector")

    best_a = a0.copy()
    best_val = compute_score(best_a)
    print(f"[init] initial score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Build a pool of diverse seeds.
    # --------------------------------------------------------------
    seeds = [best_a]                     # start with the fallback / uniform

    # a few Dirichlet draws
    for _ in range(5):
        a = rng.dirichlet(np.ones(n))
        seeds.append(a)

    # exponential seeds with different decay rates
    for r in (0.999, 0.9985, 0.9995):
        seeds.append(seed_exponential(n, r))

    # polynomial (decreasing) seeds
    for beta in (0.5, 1.0, 1.5, 2.5):
        seeds.append(seed_polynomial(n, beta))

    # mixed seeds (exponential × power)
    for r, beta in ((0.999, -0.8), (0.999, -1.5), (0.9995, -0.3)):
        seeds.append(seed_mixed(n, r, beta))

    # --------------------------------------------------------------
    # 2. Run Adam optimisation on each seed (time‑sliced).
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    per_start = remaining / max(len(seeds), 1)

    for idx, start_vec in enumerate(seeds):
        if time.time() >= deadline - 20.0:
            break
        local_deadline = time.time() + per_start - 1.0
        if local_deadline <= time.time():
            continue
        print(f"[adam] start {idx+1}/{len(seeds)} – allocated ≈ {per_start:.1f}s")
        a_ref = smooth_max_adam(start_vec, local_deadline)
        val = compute_score(a_ref)
        print(f"[adam] terminated start {idx+1}, score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 3. Final polishing with simulated‑annealing.
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        anneal_deadline = deadline - 2.0
        print("[anneal] starting fine‑tuning")
        a_ref = anneal_fine_tune(best_a, anneal_deadline)
        val = compute_score(a_ref)
        print(f"[anneal] finished, score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 4. Return the best sequence seen (list of floats).
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq
