def gradient_smoothed_max(a, w):
    # Compute grad = 2 * conv(w, a_rev)[n-1:2n-1]
    n = a.shape[0]
    a_rev = a[::-1]
    m = len(w)  # =2n-1

"""
A fast “soft‑max + Adam” optimizer for the step‑function problem.

The objective of `evaluate_sequence` (after fixing the sum of the
coefficients to 1) is 2·n·max(conv(a,a)).  The algorithm works on the
probability simplex by parametrising the sequence with log‑weights
`z` (so that a = softmax(z)).  A smooth approximation of the max
operator is used (softmax‑weighted sum of the convolution) and its
gradient is obtained analytically via a single FFT‑based correlation.
Adam together with a slowly increasing smoothing parameter drives the
distribution towards a shape whose self‑convolution is as flat as
possible.  After the main phase a short projected sub‑gradient polish
tightens the final result.  The whole search respects the 1000 s wall‑clock
budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT.
    Returns the full (2·len(a)‑1) result.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def generate_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) shape, normalised to sum = 1."""
    i = np.arange(n, dtype=float) + 0.5               # bin centres
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


def gradient_smoothed_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum `sum_k w_k * (a*a)_k` with respect to `a`.
    `w` is a probability vector of length 2·n‑1 (the softmax weights
    over the convolution entries).  The result is
        g_i = 2 * sum_k w_k * a_{k-i}
    which is a correlation of `w` with the reversed `a`.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    m = w.shape[0]                     # = 2·n‑1
    conv_len = m + n - 1                # = 3·n‑2
    n_fft = 1 << ((conv_len - 1).bit_length())
    A_rev_fft = np.fft.rfft(a_rev, n=n_fft)
    W_fft = np.fft.rfft(w, n=n_fft)
    C_fft = A_rev_fft * W_fft
    corr_full = np.fft.irfft(C_fft, n=n_fft)[:conv_len]
    # slice the part that corresponds to indices 0 … n‑1
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient polish (simple active‑set descent)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective
    2·n·max(conv).  Runs until `deadline` and returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # gradient of max = sum of gradients of active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        # Euclidean projection onto the simplex
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        mask = u - cssv / ind > 0
        if np.any(mask):
            rho = np.where(mask)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(a - theta, 0.0)
        else:
            a = np.full_like(a, 1.0 / n)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            # re‑project
            u = np.sort(a)[::-1]
            cssv = np.cumsum(u) - 1.0
            ind = np.arange(1, n + 1)
            mask = u - cssv / ind > 0
            if np.any(mask):
                rho = np.where(mask)[0][-1]
                theta = cssv[rho] / (rho + 1.0)
                a = np.maximum(a - theta, 0.0)
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0            # keep a tiny safety margin

    # ------------------------------------------------------------------
    # 0. optional fallback (if a previous construction is present)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]     # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, np.array(fallback, dtype=float)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. start from the discrete arcsine distribution (moderate size)
    # ------------------------------------------------------------------
    n = 4096                                     # a good trade‑off for speed / quality
    a = generate_arcsine(n)
    cur_score = compute_score(a)
    print(f"[arcsine] initial score = {cur_score:.6f}")
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the smooth approximation of max(conv)
    # ------------------------------------------------------------------
    # work in log‑space to stay on the simplex automatically
    eps = 1e-12
    z = np.log(a + eps)                         # logits
    m = np.zeros(n)
    v = np.zeros(n)
    t_iter = 0
    beta1, beta2 = 0.9, 0.999
    lr = 0.10                                    # base learning‑rate for logits
    adam_eps = 1e-8

    alpha = 1.0                                 # smoothing strength
    alpha_max = 200.0
    alpha_factor = 1.005                       # geometric increase per iteration

    while time.time() < deadline:
        t_iter += 1

        # --- (re)compute a from logits (soft‑max) --------------------
        a = np.exp(z - np.max(z))
        a /= a.sum()

        # --- convolution and soft‑max weights ------------------------
        conv = full_convolution(a)                # length 2·n‑1
        max_conv = conv.max()
        # smooth approximation of the max using a (temperature‑controlled) soft‑max
        diff = conv - max_conv                    # ≤ 0
        w_exp = np.exp(alpha * diff)              # max weight = 1
        w_sum = w_exp.sum()
        if w_sum == 0.0:                         # numerical safety
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w = w_exp / w_sum

        # --- gradient of the smoothed max w.r.t. a --------------------
        grad_a = gradient_smoothed_max(a, w)      # shape (n,)

        # --- chain rule to obtain gradient w.r.t. logits -------------
        # a = softmax(z) ⇒ ∂a_i/∂z_j = a_i (δ_{ij} – a_j)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)               # shape (n,)

        # --- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        z -= step

        # --- evaluate current objective (fast) -----------------------
        cur_obj = 2.0 * n * max_conv
        if cur_obj < best_score:
            best_score, best_seq = cur_obj, a.copy()
            print(f"[iter {t_iter:5d}] new best = {best_score:.6f}")

        # --- increase smoothing slowly --------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # --- occasional jitter to help escape plateaus ---------------
        if t_iter % 500 == 0:
            z += np.random.randn(n) * (lr * 0.001 / np.sqrt(t_iter))

    # ------------------------------------------------------------------
    # 3. final polishing with a simple projected sub‑gradient method
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("[polish] start")
        polish_deadline = deadline - 1.0
        a_polished = subgradient_polish(best_seq, polish_deadline, step0=0.20)
        polished_score = compute_score(a_polished)
        print(f"[polish] refined score = {polished_score:.6f}")
        if polished_score < best_score:
            best_score, best_seq = polished_score, a_polished.copy()

    # ------------------------------------------------------------------
    # 4. fallback – uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 5. official evaluation (clamping, scaling etc. happen inside)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq.tolist()
