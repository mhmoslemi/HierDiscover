import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation based on the algorithm by Duchi et al. (2008)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors x and y using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    N = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    z = np.fft.irfft(X * Y, N)
    return z[:n]


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a (full convolution)."""
    return fft_convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def generate_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) distribution, normalised to sum = 1."""
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient polish (same as the reference implementation)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # gradient of max‑conv = sum of gradients of the active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (lightweight hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-8,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Temperature‑controlled soft‑max gradient descent
# ----------------------------------------------------------------------


def softmax_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smoothed ``max`` (soft‑max) of the auto‑convolution.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    # initialise temperature proportional to the spread of the convolution
    conv = full_convolution(a)
    maxv = conv.max()
    spread = maxv - conv.mean()
    T = max(spread * 0.3, 1e-6)      # avoid a zero temperature at start
    step0 = 0.02                    # base step‑size

    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        maxv = conv.max()
        cur_score = float(2.0 * n * maxv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights (more weight on larger convolution entries)
        # shift by maxv for numerical stability
        exp_term = np.exp((conv - maxv) / T)
        w = exp_term / exp_term.sum()

        # gradient = 2 * (w * reverse(a))_{i + (n-1)}
        a_rev = a[::-1]
        grad_full = fft_convolve(w, a_rev)          # length 3n-2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # diminishing step size
        step = step0 / np.sqrt(it)
        a -= step * grad
        a = project_to_simplex(a)

        # anneal temperature (slow exponential decay)
        T = max(T * 0.9995, 1e-12)

        # occasional random jitter to hop out of plateaus
        if it % 5000 == 0:
            a += np.random.randn(n) * (step0 * 0.001)
            a = project_to_simplex(a)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The routine follows the plan described above.
    Returns a plain Python list of floats.
    """

    # ------------------------------------------------------------------

        print(f"[fallback] evaluate_sequence score = {val:.6f}")

    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size – a moderate increase over the previous 2048
    # ------------------------------------------------------------------
    n = 4096                       # larger vector gives more freedom
    print(f"[setup] using n = {n}")

    # ------------------------------------------------------------------
    # 2. Deterministic strong start: arcsine distribution
    # ------------------------------------------------------------------
    a_start = generate_arcsine(n)
    score_start = compute_score(a_start)
    print(f"[arcsine] initial score = {score_start:.6f}")
    if score_start < best_score:
        best_score = score_start
        best_seq = a_start.copy()

    # ------------------------------------------------------------------
    # 3. Primary optimisation: smooth‑max gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = deadline - 4.0
        print("[softmax‑descent] start")
        a_opt, sc_opt = softmax_descent(a_start, gd_deadline)
        print(f"[softmax‑descent] best score = {sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

    # ------------------------------------------------------------------
    # 4. Sub‑gradient polishing (takes the remaining bulk of the budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polish_deadline = deadline - 2.0
        print("[polish] start")
        a_polished = subgradient_polish(best_seq, polish_deadline)
        sc_polish = compute_score(a_polished)
        print(f"[polish] refined score = {sc_polish:.6f}")
        if sc_polish < best_score:
            best_score = sc_polish
            best_seq = a_polished.copy()

    # ------------------------------------------------------------------
    # 5. Final short simulated‑annealing fine‑tune
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[anneal] start")
        a_final = simulated_annealing(best_seq, deadline)
        sc_final = compute_score(a_final)
        print(f"[anneal] final score = {sc_final:.6f}")
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # ------------------------------------------------------------------
    # 6. Safeguard – fall back to uniform if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 7. Evaluate with the official function and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported evaluate_sequence score = {final_score:.6f}")
    return best_seq.tolist()
