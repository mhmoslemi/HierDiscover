    n = a.shape[0]
    m = w.shape[0]  # 2n-1
    size = 1 << ((n + m - 1 - 1).bit_length())

"""
A fast “smooth‑max + Adam” optimizer for the step‑height problem.

The objective (scale‑invariant) is
    F(a) = 2·n·max_k (a * a)[k]          (a ≥ 0, Σ a = 1)

Directly minimising the non‑smooth max is hard.  We replace the max by a
softmax (log‑sum‑exp) with an annealed temperature β and run Adam on the
resulting smooth objective.  After each Adam step the iterate is projected
onto the probability simplex (non‑negative, unit sum).  The annealing
gradually sharpens the softmax so the algorithm first explores a smooth
landscape and finally concentrates on the true peak.

The routine uses three stages:

1.  Try a known fallback (``height_sequence_1``) – it is already very good.
2.  Run the Adam optimiser from the best start found so far (or a uniform
    vector if nothing is available).  During the run we keep the best
    iterate w.r.t. the exact objective.
3.  Polish the best iterate with a classic projected sub‑gradient method
    (the same routine used in the baseline) to squeeze out the last bit
    of improvement.

All heavy lifting (convolutions, gradient) is done with real FFTs, which
makes the inner loop cheap enough to afford many thousands of iterations
within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _next_pow_two(x: int) -> int:
    """Small helper – returns the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.  Returns an array of length
    len(a)+len(b)-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    size = _next_pow_two(n)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    c = np.fft.irfft(A * B, n=size)[:n]
    return c


def full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a."""
    return full_convolution(a, a)


def compute_score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv) – no clipping, no sum needed (Σa==1)."""
    return float(2.0 * n * conv.max())


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability‑vector a."""
    n = a.shape[0]
    conv = full_convolution_self(a)
    return compute_score_from_conv(conv, n)


# ----------------------------------------------------------------------
# Gradient of the smoothed max (log‑sum‑exp) objective
# ----------------------------------------------------------------------


def smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns the soft‑max weights w_k = exp(β·c_k)/Σ exp(β·c_k) where
    c_k = conv[k].  The implementation is numerically stable.
    """
    c_max = conv.max()
    # conv ≤ c_max, therefore the exponent is ≤ 0 and no overflow occurs.
    exp_terms = np.exp(beta * (conv - c_max))
    return exp_terms / exp_terms.sum()


def gradient_smooth_max(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k w_k·(a*a)[k]   w.r.t. a.
    The formula reduces to
        g = 2·(rev(a) * w)[n‑1 : n‑1+n]   (full convolution)
    where rev(a) is a reversed.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a, length n
    conv = full_convolution(a_rev, weights)   # length n + 2n‑2 = 3n‑2
    # slice that corresponds to indices n‑1 … n‑1+n‑1 (exactly n entries)
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_opt(start_a: np.ndarray,
                    deadline: float,
                    init_beta: float = 0.5,
                    beta_cap: float = 30.0,
                    beta_mul: float = 1.001,
                    lr: float = 0.05,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimizes the smoothed max of the autocorrelation.
    Returns the best vector (in terms of the exact objective) seen before
    the deadline and its exact score.
    """
    a = start_a.copy()
    n = a.shape[0]

    m = np.zeros_like(a)       # first moment
    v = np.zeros_like(a)       # second moment
    t = 0

    best_a = a.copy()
    best_score = compute_score(a)

    beta = init_beta

    while time.time() < deadline:
        t += 1

        # --- current exact score (for bookkeeping) ---------------------------------
        conv = full_convolution_self(a)
        cur_score = compute_score_from_conv(conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[adam] iter {t:5d} – new best exact score {best_score:.6f}")

        # --- soft‑max weights -------------------------------------------------------
        w = smooth_weights(conv, beta)

        # --- gradient ---------------------------------------------------------------
        g = gradient_smooth_max(a, w)

        # --- Adam update ------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # --- stay on the simplex ----------------------------------------------------
        a = project_to_simplex(a)

        # --- anneal β ----------------------------------------------------------------
        beta = min(beta_cap, beta * beta_mul)

        # safety break if the step becomes negligible
        if np.linalg.norm(step) < 1e-12:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Classical projected sub‑gradient polishing (identical to the baseline)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the exact objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution_self(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape very shallow basins
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

    # 2. Adam optimisation on the smoothed max
    # -----------------------------------------------------------------
    adam_deadline = deadline - 200.0               # reserve time for polishing
    if time.time() < adam_deadline:
        print("[adam] start optimiser")
        a_opt, a_score = adam_smooth_opt(start_vec, adam_deadline)
        seq = a_opt.tolist()
        val = evaluate_sequence(seq)
        print(f"[adam] evaluated exact score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, seq

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (spends the remaining budget)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 5.0
    if best_seq is not None and time.time() < polish_deadline:
        print("[polish] start final sub‑gradient refinement")
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[polish] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – uniform vector if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
