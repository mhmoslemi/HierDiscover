```
def subgradient_descent(a0, deadline, step0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    t=0
    while time.time() < deadline:
        t+=1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-8
        active = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        idx = np.arange(len(a))
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < len(a))
            g[mask] += 2.0 * a[j[mask]]
        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # jitter occasionally
        if t % 500 == 0:
            jitter = np.random.randn(len(a)) * (step0 * 0.01 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)
    return best_a, best_val
```

assistantfinal**Solution Overview**

The score to minimise is  

```
score(a) = 2·n·max(conv(a,a)) / (sum(a)²)
```

The objective is *scale‑invariant*: multiplying the whole vector by a constant does
not change the score (the clipping to `[0,1000]` never activates for the vectors
we generate).  
Therefore we can restrict the search to the probability simplex `∑ aᵢ = 1`,
`aᵢ ≥ 0`.  
For a given vector `a` the maximal entry of the convolution `conv(a,a)` is a
piece‑wise linear function of `a`.  Its sub‑gradient can be written in closed
form (the active shifts are the indices where the maximal value is attained).
Projecting the update onto the simplex keeps all constraints satisfied.

The algorithm consists of three stages

1. **Generate several diverse starting points** – uniform, a few exponential
   decays, Dirichlet draws with different concentration parameters and the
   historic fallback `height_sequence_1` (if it exists).  
   All are normalised to a probability vector.

2. **Intensive sub‑gradient optimisation** – for each start we run a
   projected sub‑gradient descent for as long as the time budget allows.
   The step size decays as `step₀ / √t`.  Every few hundred iterations a small
   Gaussian jitter is added to help escape shallow plateaus.

3. **Final polishing** – the best vector obtained so far is polished a little
   longer with the same sub‑gradient scheme to squeeze out extra improvement.

The whole search stops well before the 1000 s deadline, guaranteeing that a
valid sequence is always returned.  The procedure is completely self‑contained
and uses only `numpy` and the already imported `evaluate_sequence`.

```python
"""
search for a step‑height sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

The method:
  * generate a variety of initial probability vectors (uniform, exponentials,
    Dirichlet samples, optional historic fallback);
  * for each start run a projected sub‑gradient descent that minimises the
    maximal convolution entry;
  * keep the best result and finally polish it a little longer.

All vectors are kept on the simplex, therefore the objective is
scale‑invariant and the final list can be passed directly to
evaluate_sequence().
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # avoid division by zero – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * a using an FFT of suitable size."""
    n = a.shape[0]
    m = 2 * n - 1
    # length of the FFT – next power of two for speed
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective value for a probability vector (sum = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient optimisation on the simplex
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        deadline: float,
                        step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the objective
        2·n·max(conv(a,a))

    Parameters
    ----------
    a0 : np.ndarray
        Starting probability vector (sum = 1, non‑negative).
    deadline : float
        Absolute time (as returned by time.time()) after which the loop stops.
    step0 : float, optional
        Initial step size; the actual step at iteration t is step0 / sqrt(t).

    Returns
    -------
    best_a, best_val
        The best vector found and its (exact) objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # indices where the maximum is attained (within tiny tolerance)
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional random jitter to escape plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Helpers for initial candidates
# ----------------------------------------------------------------------


def make_exponential(n: int, r: float) -> np.ndarray:
    """Exponential decay a_i ∝ r**i, normalised to sum 1."""
    raw = np.power(r, np.arange(n, dtype=float))
    return raw / raw.sum()


def make_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `alpha` and normalise."""
    vec = np.random.dirichlet([alpha] * n)
    return vec.astype(float)

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # leave a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq: list[float] | None = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Gather a pool of diverse starting points
    # -----------------------------------------------------------------
    candidates: list[np.ndarray] = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=float))

    # exponential decays (r values < 1)
    for r in [0.9999, 0.9995, 0.999, 0.998, 0.995]:
        candidates.append(make_exponential(n, r))

    # Dirichlet draws with various concentrations
    for alpha in [0.3, 0.5, 0.8, 1.0, 2.0, 5.0]:
        for _ in range(2):   # a couple of samples per alpha
            candidates.append(make_dirichlet(n, alpha))

    # historic fallback, if present
    try:
        fallback = np.array(height_sequence_1, dtype=float)
        if fallback.size < n:
            fallback = np.pad(fallback, (0, n - fallback.size))
        else:
            fallback = fallback[:n]
        fallback = np.maximum(fallback, 0.0)
        fallback = fallback / fallback.sum()
        candidates.append(fallback)
    except Exception:
        pass

    # shuffle order to avoid systematic bias
    random.shuffle(candidates)

    # -----------------------------------------------------------------
    # 1. Run sub‑gradient optimisation from each start while time permits
    # -----------------------------------------------------------------
    for i, init in enumerate(candidates, 1):
        if time.time() >= deadline:
            break

        # allocate a modest slice of the remaining budget to this trial
        remaining = deadline - time.time()
        slice_time = min(30.0, remaining)     # 30 s per trial (or less)

        print(f"[trial {i:02d}/{len(candidates)}] start optimisation "
              f"({slice_time:.1f}s budget)")

        a_opt, val = subgradient_descent(init, time.time() + slice_time,
                                         step0=0.5)

        seq = a_opt.tolist()
        score = evaluate_sequence(seq)
        print(f"[trial {i:02d}] score = {score:.6f}")

        if score < best_score:
            best_score, best_seq = score, seq
            print(f"   ** new global best **")

    # -----------------------------------------------------------------
    # 2. Final polishing of the best candidate (if any time left)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[polish] final refinement")
        a0 = np.array(best_seq, dtype=float)
        a_polish, _ = subgradient_descent(a0, deadline, step0=0.3)
        score = evaluate_sequence(a_polish.tolist())
        print(f"[polish] score = {score:.6f}")
        if score < best_score:
            best_score, best_seq = score, a_polish.tolist()

    # -----------------------------------------------------------------
    # 3. Safeguard – return uniform if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best score = {best_score:.6f}")
    return best_seq
```
