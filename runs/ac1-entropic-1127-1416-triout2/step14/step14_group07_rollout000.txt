"""
search_step_function_v4.py

A multi‑stage optimiser for ``evaluate_sequence``.
The algorithm proceeds in four phases:

1. **Broad parametric sweep** – explores several analytic families
   (reverse power‑law, bi‑power, sigmoid, plateau‑exponential,
   linear front, beta) on a coarse grid and keeps the best candidate.

2. **Fine parametric optimisation** – refines the most promising
   families with cheap 1‑D golden‑section search (reverse power‑law,
   plateau‑exponential) and a tiny stochastic hill‑climb in the
   2‑D (p,q) space of the bi‑power family.

3. **Stochastic hill‑climb with simulated‑annealing** – starting from
   the best vector found so far, repeatedly transfers a random
   fraction of mass between two positions.  Moves that improve the
   objective are always accepted; worse moves are accepted with a
   probability that decays with a simple exponential temperature
   schedule.  Periodic Gaussian jitter helps escaping shallow
   plateaus.

4. **Final safety net** – if nothing was discovered, falls back to a
   simple reverse‑power shape (p = 1).

All intermediate scores are computed with a fast FFT‑based
convolution on vectors normalised to Σa = 1 (the true objective is
``2·n·max(conv)``).  The official ``evaluate_sequence`` is called only
once for the final returned sequence.

The search respects the 1000 s budget (a 5 s safety margin).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using an FFT (real)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families of candidate vectors (all return a normalised array)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p,   p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p),   p ≥ 0."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.
    Both exponents ≥ 0 produce a front‑loaded shape.
    """
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i‑c)/σ))."""
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^α (1‑x_i)^β  with x_i=(i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def plateau_exponential(n: int, L_frac: float, decay: float) -> np.ndarray:
    """
    Front plateau of length L followed by a geometric tail.
    ``L_frac`` ∈ (0,1) is the plateau length as a fraction of n,
    ``decay`` ∈ (0,1) is the exponential factor of the tail.
    """
    L = max(1, int(L_frac * n))
    if L >= n:
        return None
    tail_len = n - L
    head = np.full(L, 1.0, dtype=np.float64)
    tail = decay ** np.arange(tail_len, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def linear_front(n: int, L_frac: float) -> np.ndarray:
    """
    Linear decreasing front of length L, zero afterwards.
    ``L_frac`` ∈ (0,1] controls the support length.
    """
    L = max(1, int(L_frac * n))
    if L > n:
        return None
    front = np.arange(L, 0, -1, dtype=np.float64)  # L, L‑1, …, 1
    tail = np.zeros(n - L, dtype=np.float64)
    a = np.concatenate([front, tail])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# One‑dimensional golden‑section search (for a scalar parameter)
# ----------------------------------------------------------------------
def golden_section_search(func, a: float, b: float, tol: float = 1e-4,
                          max_iter: int = 60):
    """Return (x_opt, f_opt) approx. minimizing func on [a,b]."""
    phi = (1 + 5 ** 0.5) / 2
    resphi = 2 - phi  # = 1/phi^2
    c = b - resphi * (b - a)
    d = a + resphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - resphi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + resphi * (b - a)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Random hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------
def hill_climb_sa(a_start: np.ndarray, deadline: float,
                  init_temp_factor: float = 0.02,
                  max_iter: int = 2_000_000) -> np.ndarray:
    """
    Stochastic hill‑climb on the simplex.
    - Moves transfer a random fraction of the source mass to a random
      destination.
    - Improvements are always accepted.
    - Worse moves are accepted with probability exp(-Δ/Temp).
    - Temperature decays exponentially.
    """
    a = a_start.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    it = 0
    temp = cur_score * init_temp_factor
    temp_decay = 0.999995  # very slow exponential cooling

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # Choose a fraction of the source mass to move
        frac = np.random.rand() * 0.3  # up to 30 % of a[i]
        delta = frac * a[i]

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # Clip and re‑project to the simplex (numerical safety)
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_a = cand.copy()
                best_score = cand_score
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
                a = cand
                cur_score = cand_score

        # Temperature schedule
        temp *= temp_decay

        # Occasionally inject a tiny Gaussian jitter to avoid stagnation
        if it % 50000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-4
            a = project_to_simplex(a + jitter)
            cur_score = compute_score(a)

        it += 1

    return best_a

        p0, q0 = 1.0, 0.0
        if best_vec is not None:
            # try to infer p,q by a cheap least‑squares on log‑scale
            # (not critical – just a heuristic start)
            idx = np.arange(n, 0, -1)
            try:
                loga = np.log(best_vec + 1e-12)
                A = np.vstack([np.log(idx), np.log(n - idx + 1)]).T
                coeffs, *_ = np.linalg.lstsq(A, loga, rcond=None)
                p0, q0 = -coeffs[0], coeffs[1]
                p0 = max(0.0, p0)
                q0 = max(0.0, q0)
            except Exception:
                pass

        best_mp_score = compute_score(multi_power(n, p0, q0))
        p_cur, q_cur = p0, q0
        for _ in range(3000):
            if time.time() >= refine_deadline:
                break
            p_cand = np.clip(p_cur + np.random.randn() * 0.05, 0.0, 6.0)
            q_cand = np.clip(q_cur + np.random.randn() * 0.05, 0.0, 3.0)
            vec = multi_power(n, p_cand, q_cand)
            sc = compute_score(vec)
            if sc < best_mp_score:
                best_mp_score = sc
                p_cur, q_cur = p_cand, q_cand
                test_candidate(vec)

    # ------------------------------------------------------------------
    # (3) Stochastic hill‑climb with simulated annealing
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        print("[phase 3] stochastic hill‑climb (simulated annealing)", flush=True)
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb_sa(a_start, hill_deadline,
                                init_temp_factor=0.03,
                                max_iter=3_000_000)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final official score = {val:.6f}", flush=True)
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # (4) Safety fallback (unlikely to be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        print("[fallback] using simple reverse power p=1", flush=True)
        a_fallback = reverse_power_law(n, 1.0)
        best_seq = a_fallback.tolist()
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    final_val = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_val:.6f}", flush=True)
    return best_seq
