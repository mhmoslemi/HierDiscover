def lp_refine(a, M, blend_factor):
    n = a.shape[0]
    m = 2*n-1
    # Preallocate A
    A = np.empty((m, n), dtype=np.float64)
    # Build Toeplitz using broadcasting: each column j has a padded version.
    # We'll fill using loops for clarity (maybe slower but fine).
    # Note that each column j: rows j to j+n-1 get a[0:n].
    # Equivalent to A = np.vstack([np.pad(a, (j, m-n-j)) for j in range(n)]) ?? not.
    # Let's implement loops:
    for j in range(n):
        A[j:j+n, j] = a
    # Setup b = M * np.ones(m)
    b = np.full(m, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)  # maximize sum(g) -> minimize -sum(g)

    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        # fallback: return a (no improvement)
        return a
    g = res.x
    # Normalize g to sum 1 (if sum(g)==0, keep uniform)
    s = g.sum()
    if s == 0:
        g_norm = np.full_like(g, 1.0 / n)
    else:
        g_norm = g / s
    # Blend
    new_a = (1.0 - blend_factor) * a + blend_factor * g_norm
    # Project onto simplex (ensuring sum=1 and non-negative)
    new_a = np.maximum(new_a, 0.0)
    tot = new_a.sum()
    if tot == 0:
        new_a = np.full_like(new_a, 1.0 / n)
    else:
        new_a = new_a / tot
    # Symmetrize optionally
    new_a = (new_a + new_a[::-1]) / 2.0
    return new_a

"""
Improved search for a non‑negative step‑function minimizing
``evaluate_sequence``.

Main ideas
----------
* Global exploration (parametric families, Dirichlet draws) – same as the
  reference but a little larger.
* Upsampling to a finer grid (2048 → 4096) and a long “soft‑max Adam”
  optimisation of the exact objective (the temperature τ is driven to
  1e‑9 so we really optimise the true sup‑norm).
* A longer simulated‑annealing / hill‑climb phase (the original code
  stopped after a few seconds).
* A cheap linear‑programming refinement: for a current vector *f* we
  maximise Σ gⱼ under the linear constraints (f∗g)ₖ ≤ max(f∗f)ₖ.
  The LP gives a new direction *g* that moves mass into places where the
  current convolution is small.  The LP is solved on a down‑sampled
  representation (2048 variables → ~64 MiB matrix) and the result is
  up‑sampled and blended back into the full‑size vector.
* After every major stage we symmetrise the vector (a ← (a+a[::-1])/2)
  because the objective is invariant under reversal and symmetrisation
  usually lowers the peak of the auto‑convolution.

The whole pipeline respects the 1000 s wall‑clock limit and returns a
plain Python list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear convolution via FFT (real)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b (FFT based)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a) = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(a*a)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the baseline sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)

def _try_candidate(vec: np.ndarray,
                   best: List[Tuple[float, np.ndarray]],
                   max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate and keep it if good."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return
    vec = vec / total
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int,
                    deadline: float,
                    max_keep: int = 8) -> List[np.ndarray]:
    """Sweep a bunch of parametric families and Dirichlet draws."""
    best: List[Tuple[float, np.ndarray]] = []

    # power‑law
    for p in np.linspace(0.1, 6.0, 400):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # exponential
    for lam in np.linspace(0.0001, 0.25, 400):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # beta
    al_vals = np.linspace(0.2, 8.0, 40)
    be_vals = np.linspace(0.2, 8.0, 40)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # Gaussian
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 40)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # Dirichlet draws – both raw and sorted (monotone) versions
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1200
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``vec`` (length n) to length ``new_n``."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


def downsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Simple block‑averaging down‑sample (requires integer factor)."""
    old_n = vec.shape[0]
    factor = old_n // new_n
    if factor * new_n != old_n:
        # fallback to linear interpolation if not perfectly divisible
        return upsample(vec, new_n)
    return vec.reshape(new_n, factor).mean(axis=1)


def symmetrize(vec: np.ndarray) -> np.ndarray:
    """Enforce a↔a reversal symmetry."""
    return (vec + vec[::-1]) / 2.0


# ----------------------------------------------------------------------
# Smoothed Adam optimisation (soft‑max temperature → 0)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray,
                    deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 10.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9995) -> np.ndarray:
    """
    Adam optimiser on the log‑softmax parameters.
    Uses a soft‑max replacement of max(conv) with temperature τ that is
    annealed to the limit τ → 0 (hard max).
    """
    a = a0.copy()
    n = a.shape[0]

    # log‑softmax parametrisation
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initialise temperature
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max → current a
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution and its maximum
        conv = full_convolution(a)
        max_c = conv.max()

        # smooth‑max weights
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a (g = 2·conv_rev(a,w))
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain‑rule back to x (logits)
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # occasional exact evaluation
        if it % 400 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# Simulated annealing / hill‑climb (longer run)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 1e-2,
                        final_temp: float = 1e-8,
                        cooling_factor: float = 0.9999,
                        step_scale_init: float = 0.08,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """
    Metropolis hill‑climb with a temperature schedule.
    The optimisation works directly on the simplex (mass transfer between
    two coordinates) and runs as long as time permits.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection onto simplex
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to get out of tiny plateaux
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

# Linear‑programming refinement (tiny LP on a down‑sampled version)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:          # pragma: no cover
    _SCIPY_AVAILABLE = False


def lp_refine_small(a_small: np.ndarray,
                    M: float,
                    blend: float = 0.2) -> np.ndarray:
    """
    Solve  max Σ g_j   s.t.   (a_small * g)_k ≤ M   for all k,
    g ≥ 0.
    The LP is solved on the small dimension (here 2048) and the result
    is blended back onto the original vector.
    """
    if not _SCIPY_AVAILABLE:
        return a_small  # fall back silently

    n = a_small.shape[0]
    m = 2 * n - 1

    # Build Toeplitz convolution matrix A where A @ g = a_small * g .
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a_small

    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g ⇒ minimise −Σ g

    # High‑s performance; limit to a few seconds per solve.
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                  method='highs', options={'presolve': True, 'time_limit': 10})
    if not res.success:
        return a_small

    g = res.x
    sum_g = g.sum()
    if sum_g == 0.0:
        g_norm = np.full_like(g, 1.0 / n)
    else:
        g_norm = g / sum_g

    # Blend and project back onto the simplex
    new = (1.0 - blend) * a_small + blend * g_norm
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        new = np.full_like(new, 1.0 / n)
    else:
        new = new / s

    # Symmetrise – often helps the sup‑norm
    new = symmetrize(new)
    return new

def propose_candidate(seed: int = 12345,

    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the given wall‑clock budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem sizes
    # --------------------------------------------------------------
    n_small = 2048
    n_big = 4096

    best_seq: List[float] | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # --------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Global baseline exploration (≈ 200 s)
    # --------------------------------------------------------------
    baseline_deadline = deadline - 720.0          # reserve most time for fine‑tuning
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n_small, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # --------------------------------------------------------------
    # 2. Upsample to the larger dimension
    # --------------------------------------------------------------
    if best_seq is None:
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)
    a0_big = upsample(a0_small, n_big)
    a0_big = symmetrize(a0_big)

    # --------------------------------------------------------------
    # 3. Smoothed Adam optimisation (≈ 600 s)
    # --------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if time.time() < polish_deadline:
        refined = smooth_optimize(a0_big, polish_deadline)
        refined = symmetrize(refined)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # Additional random restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)

            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n_big, conc))
            if np.random.rand() < 0.4:
                a_start = np.sort(a_start)[::-1]

            refined = smooth_optimize(a_start, sub_deadline)
            refined = symmetrize(refined)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # --------------------------------------------------------------
    # 4. Extended simulated‑annealing polish (≈ 120 s)
    # --------------------------------------------------------------
    anneal_deadline = deadline - 30.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        final = symmetrize(final)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # --------------------------------------------------------------
    # 5. Linear‑programming refinement (optional, cheap)
    # --------------------------------------------------------------
    lp_deadline = deadline - 5.0
    if _SCIPY_AVAILABLE and best_seq is not None and time.time() < lp_deadline:
        a_cur = np.array(best_seq, dtype=float)
        # we run a handful of LP refinements (each ≈ a few seconds)
        for _ in range(4):
            if time.time() >= lp_deadline:
                break
            # work on a down‑sampled version for speed
            a_small = downsample(a_cur, n_small)
            conv_small = full_convolution(a_small)
            M_small = conv_small.max()
            refined_small = lp_refine_small(a_small, M_small, blend=0.2)
            # up‑sample back to the full size
            refined_big = upsample(refined_small, n_big)
            # blend a little with the current vector
            a_cur = (1.0 - 0.2) * a_cur + 0.2 * refined_big
            a_cur = np.maximum(a_cur, 0.0)
            s = a_cur.sum()
            a_cur = a_cur / s if s > 0.0 else np.full_like(a_cur, 1.0 / n_big)
            a_cur = symmetrize(a_cur)

            val = compute_score(a_cur)
            if val < best_val:
                best_val, best_seq = val, a_cur.tolist()
                print(f"[LP‑refine] new best = {val:.6f}")

    # --------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 7. Report the official evaluation
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
