def compute_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    n = a.shape[0]
    max_c = conv.max()
    # stable softmax
    shift = conv - max_c  # <=0
    exp_weights = np.exp(alpha * shift)
    sum_exp = exp_weights.sum()
    if sum_exp == 0:
        p = np.full_like(exp_weights, 1.0 / exp_weights.size)
    else:
        p = exp_weights / sum_exp

    # sliding sum of width n
    # Efficient via cumulative sum
    csum = np.concatenate([np.array([0.0]), np.cumsum(p)])
    S = csum[n:] - csum[:-n]  # length n

    # convolution a_rev with p
    a_rev = a[::-1]
    conv_rev = conv_fft(a_rev, p)  # length 3n-2
    start = n - 1
    S1 = conv_rev[start:start + n]

    # gradient
    grad = S1 + a * S
    return grad
```

"""
Gradient‐based search for a step‑height sequence.

The objective is
    2·n·max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant.  We keep the candidate on the probability simplex
(sum = 1, a_i ≥ 0), so the objective reduces to 2·n·max(conv(a,a)).
We minimise the (smoothed) max of the full linear convolution using an
Adam optimiser and a continuation on the smoothing parameter α.  After the
main phase a cheap sub‑gradient polishing (identical to the baseline) is
run for the remaining time.  The best vector found is returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, v_i≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based linear convolution (full, real)
# ----------------------------------------------------------------------
def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    return np.fft.irfft(Z, n=n_fft)[:m]


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return conv_fft(a, a)


# ----------------------------------------------------------------------
# Exact score (objective with sum = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient of a smoothed max (soft‑max) of the convolution
# ----------------------------------------------------------------------
def smoothed_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇_a  (  (1/α)·log Σ exp(α·conv_k) )
    using the soft‑max weights p_k.
    """
    n = a.shape[0]
    max_c = conv.max()
    # safe soft‑max
    shift = conv - max_c                # ≤ 0
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum

    # sliding sum of width n: S_j = Σ_{k=j}^{j+n-1} p_k
    # use cumulative sums for O(n)
    csum = np.concatenate(([0.0], np.cumsum(p)))
    S = csum[n:] - csum[:-n]           # length n

    # convolution of reversed a with p  →  S1_j = Σ_m a_m p_{j+m}
    a_rev = a[::-1]
    conv_rev = conv_fft(a_rev, p)       # length 3n‑2
    start = n - 1
    S1 = conv_rev[start:start + n]      # length n

    # gradient = S1 + a * S
    return S1 + a * S


# ----------------------------------------------------------------------
# Adam optimiser with continuation on the smoothing parameter
# ----------------------------------------------------------------------
def adam_optimize(
    init: np.ndarray,
    time_deadline: float,
    *,
    lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 5.0,
    alpha_inc: float = 0.02,
    alpha_max: float = 200.0,
    report_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation of the smoothed max with a slowly increasing α.
    Returns the best vector seen and its exact score.
    """
    a = init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    alpha = alpha_start

    while time.time() < time_deadline:
        it += 1

        # current convolution (full)
        conv = full_convolution(a)

        # gradient of smoothed max
        g = smoothed_gradient(a, conv, alpha)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # keep on the simplex
        a = project_to_simplex(a)

        # occasional reporting and best‑track
        if it % report_every == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Adam] iter {it:6d} → best score {best_val:.6f}")

        # increase smoothing parameter
        if alpha < alpha_max:
            alpha = min(alpha + alpha_inc, alpha_max)

    # final best evaluation (in case loop ended without a reporting step)
    final_val = compute_score(best_a)
    if final_val < best_val:
        best_val = final_val
        best_a = best_a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient polishing (identical to the baseline)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a
