def subgradient_descent(a, deadline, step=0.01, decay=0.99):
    best_a = a.copy()
    best_val = compute_score(a)
    while time.time() < deadline:
        conv = full_convolution(a)
        max_c = conv.max()
        # active set tolerance: within 1e-12 relative
        active = np.where(conv >= max_c - 1e-12)[0]
        # weight uniform
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
        grad_full = _linear_convolution(a[::-1], w)
        grad_a = 2.0 * grad_full[n-1:n-1+n]
        # gradient descent step
        a_new = a - step * grad_a
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total == 0:
            break
        a_new /= total
        val_new = compute_score(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            # optionally increase step (but keep under some bound)
            step = min(step * 1.05, 0.1)
        else:
            # reduce step
            step *= decay
            if step < 1e-8:
                break
        # occasional jitter to escape plateau
        if np.random.rand() < 0.01:
            jitter = np.random.randn(len(a)) * (step * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
    return best_a

def symmetric_simulated_annealing(a_start, deadline, init_temp=1e-3, final_temp=1e-6, cooling=0.9995, step_scale_init=0.03):
    a = a_start.copy()
    n = len(a)
    m = n // 2
    # Represent half shape
    a_half = a[:m].copy()
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val
    T = init_temp
    step_scale = step_scale_init
    while time.time() < deadline:
        i = np.random.randint(m)
        j = np.random.randint(m)
        if i == j:
            continue
        delta = step_scale * a_half[i]
        # propose new half
        a_half_new = a_half.copy()
        a_half_new[i] -= delta
        a_half_new[j] += delta
        if np.min(a_half_new) < 0:
            continue
        # create full
        a_new = np.concatenate([a_half_new, a_half_new[::-1]])
        a_new = a_new / a_new.sum()
        val_new = compute_score(a_new)
        dval = val_new - cur_val
        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a_half = a_half_new
            cur_val = val_new
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_new.copy()
        T = max(final_temp, T * cooling)
        step_scale = max(step_scale * 0.99, 1e-6)  # adjust
    return best_a

Main ideas
----------
1. **Symmetry** – the optimal autocorrelation is symmetric, therefore we
   represent the full vector by its left half (size n/2) and reflect it.
   This halves the number of optimisation variables and speeds up the
   gradient loop.

2. **Rich baseline** – sweep a few analytically‑defined symmetric families
   (power‑law, exponential, Gaussian, symmetric beta) and draw many
   symmetric Dirichlet samples.  Keep the best few candidates.

3. **Symmetric Adam on a smoothed max** – minimise a smooth surrogate of
   “max(conv)’’ using a soft‑max (temperature τ).  The parameters are the
   logits of the half‑vector (soft‑max → probabilities → half‑vector,
   multiplied by 0.5 to obtain a full probability vector of sum 1).
   Adam with decaying learning‑rate and exponentially decaying τ is used.
   The best result over several random restarts is retained.

4. **Sub‑gradient polishing** – after the smoothed phase we switch to an
   exact max‑subgradient (weights uniform on the active set) and perform
   a few gradient steps with projection onto the simplex.

5. **Final symmetric simulated annealing** – a light Metropolis hill‑climb
   (mass transfers respecting symmetry) is run for a short time to escape
   tiny plateaus.

The whole pipeline respects a 1000 s wall‑clock budget, prints progress
and finally returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real inputs)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a * a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective for a vector that sums to 1
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(a*a) (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate generators – all return *symmetric* vectors of length n
# ----------------------------------------------------------------------


def generate_sym_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (|i-center|+1)^(-p), symmetric, normalised."""
    d = np.abs(np.arange(n) - (n - 1) / 2.0) + 1.0
    arr = d ** (-p)
    return arr / arr.sum()


def generate_sym_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i-center|), symmetric."""
    d = np.abs(np.arange(n) - (n - 1) / 2.0)
    arr = np.exp(-lam * d)
    return arr / arr.sum()


def generate_sym_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return arr / arr.sum()


def generate_sym_beta(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric beta‑like shape:
        a_i ∝ (i+1)^{α-1}·(n-i)^{α-1}
    (setting β=α gives symmetry).
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, alpha - 1.0)
    arr = left * right
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_sym_dirichlet_half(m: int, conc: float) -> np.ndarray:
    """Draw a Dirichlet vector of length m, reflect to full length n=2·m."""
    half = np.random.dirichlet(np.full(m, conc))
    full = np.concatenate([half, half[::-1]])
    return full / full.sum()


# ----------------------------------------------------------------------
# Helper to keep the best few candidates
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate and keep if good."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0:
        return
    vec = vec / total
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search_sym(n: int, deadline: float, max_keep: int = 6):
    """
    Explore a handful of symmetric families and random Dirichlet draws.
    Return the up‑to ``max_keep`` best full‑length vectors.
    """
    best = []   # list of (score, vector)

    # 1) power‑law
    for p in np.linspace(0.0, 8.0, 180):
        if time.time() >= deadline:
            break
        _try_candidate(generate_sym_powerlaw(n, p), best, max_keep)

    # 2) exponential
    for lam in np.linspace(0.001, 0.2, 180):
        if time.time() >= deadline:
            break
        _try_candidate(generate_sym_exponential(n, lam), best, max_keep)

    # 3) Gaussian
    for sigma in np.linspace(n / 64.0, n / 2.0, 150):
        if time.time() >= deadline:
            break
        _try_candidate(generate_sym_gaussian(n, sigma), best, max_keep)

    # 4) symmetric beta
    for alpha in np.linspace(0.2, 8.0, 120):
        if time.time() >= deadline:
            break
        _try_candidate(generate_sym_beta(n, alpha), best, max_keep)

    # 5) Dirichlet draws (operate on half and reflect)
    m = n // 2
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per = 300
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            vec = generate_sym_dirichlet_half(m, conc)
            _try_candidate(vec, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Symmetric Adam optimisation on the soft‑max approximation of max(conv)
# ----------------------------------------------------------------------


def symmetric_smooth_optimize(a_half0: np.ndarray, deadline: float,
                              init_lr: float = 0.4,
                              init_tau_factor: float = 15.0,
                              min_tau: float = 1e-9,
                              tau_decay: float = 0.9997) -> np.ndarray:
    """
    Adam optimiser on the symmetric representation.
    a_half0 must sum to 0.5 (full vector sums to 1).
    Returns the best full‑length vector seen before *deadline*.
    """
    m = a_half0.shape[0]
    n = 2 * m

    # initialise logits from the half‑vector (add tiny epsilon)
    eps = 1e-30
    x = np.log(a_half0 + eps)

    # Adam moments
    m_adam = np.zeros_like(x)
    v_adam = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # helper to build the full vector from logits
    def build_full(x_vec):
        max_x = np.max(x_vec)
        exp_x = np.exp(x_vec - max_x)
        s = exp_x.sum()
        a_half = 0.5 * (exp_x / s)          # guarantees sum(a_half)=0.5
        a_full = np.concatenate([a_half, a_half[::-1]])
        return a_full, a_half

    # initial state
    a_full, a_half = build_full(x)
    conv = full_convolution(a_full)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a_full.copy()
    best_val = compute_score(a_full)

    it = 0
    while time.time() < deadline:
        it += 1

        # construct current full vector
        a_full, a_half = build_full(x)

        # convolution and soft‑max weights
        conv = full_convolution(a_full)
        max_c = conv.max()

        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient wrt full a
        grad_full = _linear_convolution(a_full[::-1], w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # reduce to half‑gradient (each half parameter influences two symmetric entries)
        grad_half = grad_a[:m] + grad_a[-m:][::-1]

        # chain rule for logits (soft‑max)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        s_vec = exp_x / exp_x.sum()      # standard soft‑max (sums to 1)
        dot = np.dot(s_vec, grad_half)   # ⟨s, ∇_half⟩
        grad_x = 0.5 * s_vec * (grad_half - dot)

        # Adam update
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_x
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature decay
        tau = max(min_tau, tau * tau_decay)

        # occasional exact evaluation
        if it % 300 == 0:
            val = compute_score(a_full)
            if val < best_val:
                best_val = val
                best_a = a_full.copy()
                print(f"[sym_smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact max, uniform weights on the active set)
# ----------------------------------------------------------------------


def subgradient_polish(a_start: np.ndarray, deadline: float,
                       init_step: float = 0.02,
                       step_decay: float = 0.97,
                       min_step: float = 1e-6) -> np.ndarray:
    """
    Simple projected gradient steps using the true max‑subgradient.
    Returns the best vector found before *deadline*.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    step = init_step

    while time.time() < deadline:
        conv = full_convolution(a)
        max_c = conv.max()
        # active set (indices achieving the maximum)
        act = np.where(conv >= max_c - 1e-15)[0]
        w = np.zeros_like(conv)
        w[act] = 1.0 / max(act.size, 1)   # uniform on the active set

        grad_full = _linear_convolution(a[::-1], w)
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # projected gradient descent on the simplex
        a_new = a - step * grad_a
        a_new = np.maximum(a_new, 0.0)
        sm = a_new.sum()
        if sm == 0:
            break
        a_new /= sm

        val_new = compute_score(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            # if we made progress we can be a little bold
            step = min(step * 1.05, 0.05)
        else:
            # no improvement → shrink the step
            step *= step_decay
            if step < min_step:
                break

        # occasional tiny random jitter to escape flat plateaus
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * (step * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a


# ----------------------------------------------------------------------
# Symmetric simulated‑annealing polish (lightweight Metropolis)
# ----------------------------------------------------------------------


def symmetric_simulated_annealing(a_start: np.ndarray, deadline: float,
                                 init_temp: float = 1e-3,
                                 final_temp: float = 1e-6,
                                 cooling: float = 0.9995,
                                 step_scale_init: float = 0.03) -> np.ndarray:
    """
    Metropolis hill‑climb that respects symmetry.
    Operates on the half‑vector; the full vector is reconstructed each step.
    """
    n = a_start.shape[0]
    m = n // 2
    a_half = a_start[:m].copy()
    cur_val = compute_score(a_start)
    best_half = a_half.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(m)
        j = np.random.randint(m)
        if i == j:
            continue

        delta = step_scale * a_half[i]
        cand_half = a_half.copy()
        cand_half[i] -= delta
        cand_half[j] += delta
        if np.min(cand_half) < 0.0:
            continue

        cand = np.concatenate([cand_half, cand_half[::-1]])
        cand /= cand.sum()
        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a_half = cand_half
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best_half = a_half.copy()
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            step_scale = max(step_scale * 0.98, 1e-6)

        T = max(final_temp, T * cooling)

        # occasional jitter
        if it % 4000 == 0:
            jitter = np.random.randn(m) * (step_scale * 0.01)
            a_half = a_half * np.exp(jitter)
            a_half = a_half / a_half.sum()
            cur_val = compute_score(np.concatenate([a_half, a_half[::-1]]))

    return np.concatenate([best_half, best_half[::-1]])


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – we work directly with a fairly large even length.
    # -----------------------------------------------------------------
    n = 8192                     # must be even for the symmetry trick
    m = n // 2

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 800.0   # reserve most time for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search_sym(n, baseline_deadline, max_keep=6)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Ensure we have a starting point
    # -----------------------------------------------------------------
    if best_seq is None:
        # uniform distribution (already symmetric)
        a0_full = np.full(n, 1.0 / n)
    else:
        a0_full = np.array(best_seq, dtype=float)

    a0_half = a0_full[:m]                 # by construction sum = 0.5

    # -----------------------------------------------------------------
    # 3. Primary symmetric Adam optimisation (≈ 600 s)
    # -----------------------------------------------------------------
    opt_deadline = deadline - 200.0
    if time.time() < opt_deadline:
        refined_full = symmetric_smooth_optimize(a0_half, opt_deadline)
        val = compute_score(refined_full)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined_full.tolist()

        # additional random restarts while time permits
        while time.time() < opt_deadline:
            remaining = opt_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)

            # random half‑vector initialisation (Dirichlet)
            conc = np.random.choice([0.02, 0.05, 0.1, 0.2, 0.5])
            a_start_half = np.random.dirichlet(np.full(m, conc))
            # enforce symmetry by mirroring (already done by the optimiser)
            refined_full = symmetric_smooth_optimize(a_start_half, sub_deadline)
            val = compute_score(refined_full)
            if val < best_val:
                best_val, best_seq = val, refined_full.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Exact sub‑gradient polishing (≈ 80 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 70.0
    if best_seq is not None and time.time() < polish_deadline:
        a_start = np.array(best_seq, dtype=float)
        polished = subgradient_polish(a_start, polish_deadline)
        val = compute_score(polished)
        print(f"[subgrad‑polish] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, polished.tolist()

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (≈ 70 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = symmetric_simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
