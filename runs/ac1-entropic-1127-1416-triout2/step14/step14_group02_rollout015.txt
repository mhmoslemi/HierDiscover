"""
Improved optimiser for the “max‑convolution” step‑function problem.

Main ideas
---------
* Enrich the exploratory families – in particular add a *symmetric power‑law*
  (U‑shaped) family which analytically flattens the autocorrelation.
* Run a few short “Adam‑style’’ refinements on the best baselines, trying a
  range of the flattening weight λ (more emphasis on a flat convolution).
* Follow the best refined vector with a projected sub‑gradient polish
  (exact max‑objective) and a final simulated‑annealing hill‑climb.
* All phases respect the 1000 s wall‑clock limit and constantly keep the
  best vector seen so far.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, zero‑padding)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using the real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of ½·∑ (a*a - const)² or of a smooth‑max term.
    Returns 2·(a ⋆ vec) where ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Scoring utilities (exact objective for a normalised vector)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) (vector assumed to sum to 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the global baseline search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) , left‑hand decreasing."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_symmetric_powerlaw(n: int, p: float) -> np.ndarray:
    """U‑shaped (symmetric) power law: (i+1)^(-p) + (n-i)^(-p)."""
    i = np.arange(1, n + 1, dtype=float)
    left = i ** (-p)
    right = (n - i + 1) ** (-p)
    a = left + right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture, then projected back onto the simplex."""
    mixed = lam * a + (1.0 - lam) * b
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global baseline exploration (cheap random draws + analytic families)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 15000,
) -> List[np.ndarray]:
    """
    Sample a large collection of analytic families and Dirichlet draws,
    keep the `max_keep` best vectors (lowest exact score).
    """
    best: List[Tuple[float, np.ndarray]] = []

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- analytical families ------------------------------------------------
    for p in np.linspace(0.1, 5.0, 81):                # plain power‑law
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    for p in np.linspace(0.1, 1.0, 30):                # symmetric power‑law (U‑shape)
        if time.time() >= deadline:
            break
        try_candidate(generate_symmetric_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.5, 120):         # exponential
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    al_vals = np.linspace(0.02, 2.0, 31)               # beta grid
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # more extreme beta (U‑shaped) draws
    for _ in range(200):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-3, -0.2)      # <0.1
        beta = 10 ** np.random.uniform(-3, -0.2)
        try_candidate(generate_beta(n, alpha, beta))

    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # ----- mixtures -----------------------------------------------------------
    for _ in range(300):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_symmetric_powerlaw(n, np.random.uniform(0.1, 1.0))
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # ----- Dirichlet draws (wide range of concentrations) --------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Combined Adam optimiser (smooth‑max + flattening term)
# ----------------------------------------------------------------------


def combined_adam(
    a0: np.ndarray,
    deadline: float,
    lam: float = 0.5,
    tau_start: float = 0.30,
    tau_end: float = 1e-7,
    tau_decay: float = 0.94,
    lr_start: float = 0.20,
    lr_decay: float = 0.97,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on the blended surrogate:

        L(a) = (1‑λ)·τ·log Σ exp(conv/τ)   +   λ·½·‖conv – target‖² .

    Parameters
    ----------
    a0   – initial vector (already on the simplex)
    lam  – weight of the flattening term (λ close to 1 puts strong
           emphasis on making the convolution flat)
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m

    size_fft = 1 << ((3 * n - 3).bit_length())
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # ----- schedule updates ------------------------------------------------
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # ----- forward pass ----------------------------------------------------
        conv = full_convolution(a)                     # length m
        shift = conv.max()
        # numerically stable log‑sum‑exp
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()                 # soft‑max weights

        err = conv - target

        # ----- gradient --------------------------------------------------------
        grad_max = correlation_grad(a, w, size_fft)
        grad_err = correlation_grad(a, err, size_fft)
        grad = (1.0 - lam) * grad_max + lam * grad_err

        # ----- Adam update -----------------------------------------------------
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # ----- bookkeeping -------------------------------------------------------
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape flat plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-7,
    cooling_factor: float = 0.9995,
    step_scale_init: float = 0.05,
    step_scale_min: float = 5e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size (a larger n gives more flexibility)
    # --------------------------------------------------------------
    n = 4096

    best_seq: List[float] | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0. Fallback – previous construction, if available
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Global baseline exploration (≈ 150 s)
    # --------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 150.0)
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=12, random_draws=25000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Multi‑start Adam refinements (≈ 400 s)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 300.0 and best_seq is not None:
        # allocate ~300 s to Adam, keep ~100 s for polishing/annealing
        adam_budget = min(300.0, remaining - 100.0)
        start_vectors: List[np.ndarray] = [
            np.array(best_seq, dtype=float),
            np.full(n, 1.0 / n),
        ]
        # a few random starts
        for _ in range(3):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        # try several λ values – more flattening tends to lower the max‑conv
        lam_values = [0.6, 0.8, 0.95]

        for lam in lam_values:
            for idx, start_vec in enumerate(start_vectors):
                if time.time() >= deadline - 100.0:
                    break
                per_vec_deadline = time.time() + min(30.0, adam_budget / (len(lam_values) * len(start_vectors)))
                refined = combined_adam(start_vec, per_vec_deadline, lam=lam)
                val = compute_score(refined)
                print(f"[adam λ={lam:.2f} start={idx}] score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, refined.tolist()

        print(f"[adam] best after this phase = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Sub‑gradient polish (≈ 150 s)
    # --------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        polished = subgradient_polish(a0, polish_deadline, step0=0.33)
        val = compute_score(polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, polished.tolist()

    # --------------------------------------------------------------
    # 4. Simulated annealing (final ~80 s)
    # --------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final_vec = simulated_annealing(
            a_start,
            anneal_deadline,
            init_temp=2e-3,
            final_temp=1e-7,
            cooling_factor=0.9995,
            step_scale_init=0.05,
        )
        val = compute_score(final_vec)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final_vec.tolist()

    # --------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final report using the official evaluation function
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
