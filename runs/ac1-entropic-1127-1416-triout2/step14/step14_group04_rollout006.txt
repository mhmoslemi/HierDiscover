"""
Heuristic search for a non‑negative step‑function sequence that minimizes the
max‑auto‑convolution bound used in ``evaluate_sequence``.  The algorithm proceeds
in four phases within the allotted wall‑clock budget:

1. **Broad stochastic sampling** – Dirichlet draws with several concentration
   parameters, random monotone vectors and a coarse grid over the beta family.
2. **Systematic parametric sweeps** – dense scans of the exponential,
   power‑law, logistic, gamma and Gaussian families (all vectors are
   normalised to unit sum).  The best few candidates are retained.
3. **Smooth Adam optimisation** – each retained vector is refined by Adam on
   a soft‑max (log‑sum‑exp) surrogate of the max‑convolution.  The temperature
   is annealed and the optimiser is run for as long as time permits, with
   occasional random restarts.
4. **Final simulated‑annealing polish** – a Metropolis walk that moves mass
   between coordinates; this can escape shallow local minima and polish the
   solution.

The search uses a larger problem size ``n = 4096`` (instead of 2048) to give
the optimiser more degrees of freedom; the objective scales as
`2·n·max(conv(a,a))` (the denominator ``sum(a)**2`` cancels after normalising).
All vectors are kept non‑negative and are re‑normalised after every operation.
The returned list is ready for direct use with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (the scale of the vector does not matter – we assume sum=1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------
def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.zeros(n, dtype=float)
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_logistic(n: int, shift: float, scale: float) -> np.ndarray:
    """a_i ∝ 1 / (1 + exp((i - shift)/scale)).  monotone decreasing."""
    i = np.arange(n, dtype=float)
    # avoid overflow
    z = (i - shift) / max(scale, 1e-12)
    a = 1.0 / (1.0 + np.exp(z))
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_gamma(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·exp(-β·i).   (discrete Gamma‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.power(i, alpha - 1.0) * np.exp(-beta * (i - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random vector, sort it decreasingly, and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]
    s = a.sum()
    return a / s if s > 0 else np.zeros(n, dtype=float)

# Book‑keeping helpers
# ----------------------------------------------------------------------
def _try_candidate(vec: np.ndarray, best):
    """
    Normalise ``vec`` (clip negatives, force sum=1), compute its score and
    insert it into the sorted list ``best`` if it improves the current
    worst entry.  ``best`` holds at most 12 elements (score, vector).
    """
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        return
    vec = vec / s
    sc = compute_score(vec)
    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])

def baseline_search(n: int, deadline: float):
    """Collect a diverse set of promising candidates with cheap random draws."""
    best = []   # list of (score, vector), smallest scores first

    # 1) Dirichlet draws with several concentration values
    alphas = [0.05, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_alpha = 800
    for conc in alphas:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    # 2) Random monotone vectors
    for _ in range(2000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # 3) Coarse sweep of the beta family (asymmetric shapes)
    al_vals = np.linspace(0.2, 12.0, 10)
    be_vals = np.linspace(0.2, 12.0, 10)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best)

    # 4) Gaussian widths (broad to narrow)
    sigma_vals = np.linspace(n / 64.0, n / 2.0, 20)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        a = generate_gaussian(n, sigma)
        _try_candidate(a, best)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# Phase 3 – smooth Adam optimisation on a soft‑max surrogate
# ----------------------------------------------------------------------
def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.08,
                    init_tau_factor: float = 6.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9995) -> np.ndarray:
    """
    Adam optimiser on the logits of a softmax parametrisation.
    ``a`` is always normalised (sum = 1).  The temperature ``tau`` governs the
    smooth‑max approximation of ``max(conv)`` and is annealed during optimisation.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits safely (avoid log(0))
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise temperature from current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- logits → probabilities ----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---- convolution of a with itself ----
        conv = full_convolution(a)
        max_c = conv.max()

        # ---- soft‑max weighting ----
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ---- gradient wrt a (2·∑_k w_k·a_{k−i}) ----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)  # length = 3·n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ---- chain rule for logits (softmax) ----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- anneal temperature ----
        tau = max(min_tau, tau * tau_decay)

        # ---- occasional exact check ----
        if it % 400 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# Phase 4 – simulated‑annealing polish
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 5e-7,
                        cooling_factor: float = 0.9993,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter to escape shallow plateaux
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

            # choose a random concentration for a fresh Dirichlet start
            conc = np.random.choice([0.2, 0.6, 1.2, 3.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[smooth‑restart] new best = {sc:.6f}")

    # ---- 4. Final simulated‑annealing polish ------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(final)
        print(f"[anneal] final polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # ---- 5. Fallback – uniform distribution (should never be needed) ----
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # ---- 6. Official evaluation (the harness will call evaluate_sequence) ----
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return best_seq
