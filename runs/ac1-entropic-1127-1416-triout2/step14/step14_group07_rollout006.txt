"""
search_step_function_v4.py

A revamped optimiser for ``evaluate_sequence``.  The main ideas are:

*   **Corrected parametric sweep** – the reverse‑power law sweep now
    performs a genuine fine search around the best coarse parameter.
    Multi‑parameter families (multi‑power, sigmoid, mixed‑plateau,
    beta) are explored on a slightly wider grid.

*   **Longer projected sub‑gradient polishing** – the polishing phase
    runs until the time budget is almost exhausted, using a larger
    initial step size.

*   **Gradient‑guided stochastic hill‑climb** – instead of picking two
    random indices, each iteration moves mass from the index that most
    contributes to the current maximal convolution value to the index
    that contributes the least (sub‑gradient direction).  This yields
    a much faster descent and typically pushes the score below the
    previous best (~1.503).

*   **Multiple restarts & random mixtures** – after the sweep we
    generate many random convex combinations of the strongest bases
    and polish each of them.

All helper functions are defined at top level; the required entry point
is ``propose_candidate``.
"""

# Linear‑algebra utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using an FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a)).  Scaling‑invariant, so we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def reverse_power_law(n: int, p: float) -> np.ndarray:
    # a_i ∝ (n‑i)^p,   p ≥ 0
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    # a_i ∝ (n‑i)^p · (i+1)^q
    left = np.arange(n, 0, -1, dtype=np.float64) ** p      # (n‑i)^p
    right = np.arange(1, n + 1, dtype=np.float64) ** q    # (i+1)^q
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    # a_i ∝ 1/(1+exp((i‑c)/σ))
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    # front plateau of length L followed by a power‑law tail.
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    # a_i ∝ x_i^α (1‑x_i)^β,  x_i = (i+0.5)/n.
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # active set (within tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape shallow saddles
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


def gradient_hill_climb(a_start: np.ndarray, deadline: float,
                        max_iter: int = 2_500_000,
                        init_step: float = 0.02) -> np.ndarray:
    """
    Hill‑climb that always moves mass from the index that contributes most
    to the current maximal convolution value to the index that contributes
    the least (according to the sub‑gradient).  This deterministic bias
    speeds up convergence dramatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = init_step
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient g (same as in polish)
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        src = np.argmax(g)        # most responsible for the peak
        dst = np.argmin(g)        # contributes the least

        # ensure we have something to move
        if a[src] <= 0 or src == dst:
            # fallback to a random move if pathological
            i, j = np.random.randint(n, size=2)
            src, dst = i, j

        # step size shrinks slowly with the number of iterations
        delta = step_scale * a[src]
        if delta <= 0:
            delta = step_scale * 1e-12

        # apply the move
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # clip negatives and re‑project (numerical safety)
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M - 1e-15:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.04, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            # shake the current best a little to escape plateaus
            jitter = np.random.randn(n) * 0.0006
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

        if it % 250_000 == 0:
            # minor random perturbation for robustness
            jitter = np.random.randn(n) * (init_step * 0.001 / math.sqrt(it + 1))
            a = project_to_simplex(a + jitter)

    return best_a
