def simulated_annealing(a0, deadline, initial_temp=0.5, final_temp=0.01, max_iter=10_000_000):

    a = a0.copy()

    best_a = a.copy()

    best_score = compute_score(a)

    n = len(a)

    start = time.time()

    while time.time() < deadline:

        # compute temperature schedule based on elapsed fraction

        elapsed = time.time() - start

        frac = elapsed / (deadline - start)

        T = initial_temp * (1 - frac) + final_temp * frac  # linear cooling

        # propose move

        i, j = np.random.randint(n, size=2)

        if i == j:

            continue

        # propose delta: fraction of a[i] * random uniform scaled by temperature factor

        max_move = a[i] * 0.5  # ensure not too large

        delta = np.random.rand() * max_move * (T / initial_temp)  # scale delta with temperature

        if delta == 0:

            continue

        cand = a.copy()

        cand[i] -= delta

        cand[j] += delta

        # ensure nonnegativity no need to clip because delta <= a[i]

        # remain sum constant

        # compute score via compute_score

        cand_score = compute_score(cand)

        cur_score = compute_score(a)  # we could keep current score but compute again or track

        # Actually track current_score outside

        # accept if better or with probability exp(-(cand_score-current_score)/T)

        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):

            a = cand

            cur_score = cand_score

            if cand_score < best_score:

                best_score = cand_score

                best_a = cand.copy()

        # optional periodic jitter

    return best_a

    a = a0.copy()

    while time.time() < deadline:
        t += 1

        active = np.where(conv >= M * active_ratio)[0]
        if active.size == 0:
            active = np.array([conv.argmax()])

            src = k - idx_i

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)
        M_new = full_convolution(a).max()
        if M_new < best_M:

            best_a = a.copy()

    # 6. mixed plateau‑power
    print("[sweep] mixed_plateau_power", flush=True)
    for L_frac in np.linspace(0.02, 0.4, 12):
        for p in np.linspace(0.3, 0.9, 13):
            if time.time() >= deadline:
                return best_vec
            test_candidate(mixed_plateau_power(n, L_frac, p))

    # 7. geometric
    print("[sweep] geometric", flush=True)
    for q in np.linspace(0.90, 0.9999, 120):
        if time.time() >= deadline:
            return best_vec
        test_candidate(geometric(n, q))

    # 8. random Dirichlet draws
    print("[sweep] random Dirichlet", flush=True)
    for _ in range(1500):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        test_candidate(a)

    # 9. random convex mixtures of strong bases
    print("[sweep] random mixtures", flush=True)
    bases = [
        reverse_power_law(n, 2.0),
        sigmoid_shape(n, 0.5, 0.05),
        geometric(n, 0.98)
    ]
    bases = [b for b in bases if b is not None]
    for _ in range(1000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(bases))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, bases))
        a = project_to_simplex(a)
        test_candidate(a)

    return best_vec

# ----------------------------------------------------------------------
# Aggressive sub‑gradient descent (multiple active peaks)
# ----------------------------------------------------------------------
def subgradient_multi(a0, deadline, step0=0.30, active_ratio=0.999999):
    """Projected sub‑gradient descent using all near‑max convolution entries."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # indices whose value is essentially the maximum
        active = np.where(conv >= M * active_ratio)[0]
        if active.size == 0:
            active = np.array([conv.argmax()])

        # accumulate shifted copies of a
        shift_sum = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            src = k - idx
            shift_sum[lo:hi + 1] += a[src]

        # sub‑gradient of max(conv) w.r.t. a
        g = shift_sum + active.size * a

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# ----------------------------------------------------------------------
# Simulated annealing on the full vector
# ----------------------------------------------------------------------
def anneal(a0, deadline):
    """Mass‑transfer annealing that accepts occasional uphill moves."""
    a = a0.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    start = time.time()
    T0, T_min = 1.0, 0.001
    total = deadline - start

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        T = max(T_min, T0 * (1.0 - frac))

        i, j = np.random.randint(n, size=2)
        if i == j or a[i] <= 0:
            continue

        max_move = a[i] * 0.4               # move at most 40 % of source mass
        delta = np.random.rand() * max_move * (T / T0)
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = compute_score(cand)

        if cand_score < cur_score:
            a, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
        else:
            # Metropolis criterion
            if np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
                a, cur_score = cand, cand_score

    return best_a

# ----------------------------------------------------------------------
# Final greedy hill‑climb for polishing
# ----------------------------------------------------------------------
def hill_climb(a_start, deadline, max_iter=800_000):
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.0006
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`.  The algorithm proceeds through four stages
    (sweep → sub‑gradient → annealing → hill‑climb) and returns the best
    sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # safety margin

    best_seq = None
    best_val = float('inf')
    n = 2048

    # -----------------------------------------------------------------
    # (0) fallback to a known good construction if available
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # (1) parametric sweep
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 350.0
    if time.time() < sweep_deadline:
        print("[stage] parametric sweep", flush=True)
        a0 = parametric_sweep(n, sweep_deadline)
        if a0 is not None:
            val_proxy = compute_score(a0)
            val_real = evaluate_sequence(a0.tolist())
            print(f"[sweep] proxy={val_proxy:.6f}, official={val_real:.6f}")
            if val_real < best_val:
                best_val, best_seq = val_real, a0.tolist()

    # -----------------------------------------------------------------
    # (2) sub‑gradient polishing
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 150.0
    if best_seq is not None and time.time() < subgrad_deadline:
        print("[stage] sub‑gradient polishing", flush=True)
        a_start = np.array(best_seq, dtype=float)
        a_polished = subgradient_multi(a_start, subgrad_deadline, step0=0.30)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # (3) simulated annealing
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 50.0
    if best_seq is not None and time.time() < anneal_deadline:
        print("[stage] annealing", flush=True)
        a_start = np.array(best_seq, dtype=float)
        a_annealed = anneal(a_start, anneal_deadline)
        val = evaluate_sequence(a_annealed.tolist())
        print(f"[anneal] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_annealed.tolist()

    # -----------------------------------------------------------------
    # (4) final hill‑climb
    # -----------------------------------------------------------------
    final_deadline = deadline - 2.0
    if best_seq is not None and time.time() < final_deadline:
        print("[stage] final hill‑climb", flush=True)
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, final_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[final hill‑climb] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # (5) safety fallback
    # -----------------------------------------------------------------
    if best_seq is None:
        print("[fallback] using simple linear shape", flush=True)
        a_fallback = reverse_power_law(n, 1.0)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] score = {best_val:.6f}")

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
