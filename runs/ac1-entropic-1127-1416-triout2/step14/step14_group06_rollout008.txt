    A = np.fft.rfft(a, n=M)

"""
optimise_step_function.py

A gradient‑based optimiser for the “evaluate_sequence” objective.
The algorithm works in three phases:

1. **Initialisation** – start from a known good construction (if the
   global ``height_sequence_1`` exists) or from a simple mixed
   reverse‑power‑law/uniform shape.

2. **Smooth max‑Adam optimisation** – minimise a differentiable
   approximation of  max(conv(a,a))  using the soft‑max smoothing
   ϕ_α(x)=log‑sum‑exp(α·x)/α.  The smoothing parameter α is increased
   gradually, turning the problem into the true max‑optimisation.
   Gradients are obtained analytically (via FFT‑based correlation) and
   the iterates are kept on the probability simplex by Euclidean
   projection.

3. **Fine stochastic hill‑climb** – a cheap mass‑transfer local search
   to polish the result after the Adam phase.

All heavy linear algebra uses real FFTs for speed (O(N log N) per
iteration).  The code respects the 1000 s time budget, prints progress
and finally returns a list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law), p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    a_sum = a.sum()
    if a_sum == 0:
        return None
    a /= a_sum
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """mix * reverse_power_law(p) + (1‑mix) * uniform, 0 ≤ mix ≤ 1."""
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (phase 2)
# ----------------------------------------------------------------------
def smooth_max_adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    minimise  max(conv(a,a))  using a smooth approximation and Adam.
    Returns the best vector discovered before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # zero‑pad length for all FFTs (linear conv of length n with length 2n‑1)
    conv_len = 2 * n - 1
    fft_len = 1 << ((3 * n - 2 - 1).bit_length())   # ≥ n + conv_len – 1

    # Adam hyper‑parameters
    lr = 0.12                # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing schedule
    alpha = 2.0                 # start smoothness
    alpha_target = 300.0        # final (very sharp)
    alpha_mult = 1.001          # multiplicative increase per iteration

    best_a = a.copy()
    best_val = float('inf')
    it = 0

    while time.time() < deadline:
        it += 1
        # ----- FFT of current a -----
        A = np.fft.rfft(a, n=fft_len)

        # ----- full convolution a*a -----
        conv_full = np.fft.irfft(A * A, n=fft_len)[:conv_len]

        # ----- current objective (exact max) -----
        cur_max = conv_full.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 200 == 0:
                print(f"[smooth Adam] iter {it:6d} – best value {best_val:.6f}")

        # ----- soft‑max weights -----
        # w_k = exp(alpha*conv_k) / Σ exp(alpha*conv)
        scaled = conv_full * alpha
        shift = scaled.max()
        exp_scaled = np.exp(scaled - shift)
        w = exp_scaled / exp_scaled.sum()

        # ----- gradient (correlation a ⋆ w) -----
        W = np.fft.rfft(w, n=fft_len)
        grad = 2.0 * np.fft.irfft(A * np.conj(W), n=fft_len)[:n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ----- stay on the simplex -----
        a = project_to_simplex(a)

        # ----- increase smoothing -----
        if alpha < alpha_target:
            alpha = min(alpha_target, alpha * alpha_mult)

    print(f"[smooth Adam] finished – best value {best_val:.6f} after {it} iters")
    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (phase 3)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 150_000) -> np.ndarray:
    """
    Greedy random mass‑transfer moves with occasional restarts.
    Returns the best vector found before the deadline.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_M = full_convolution(a).max()
    step_scale = 0.04
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a fraction of the source mass to the target
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M, best_a = M_cand, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.08, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.998
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            # restart from the current best with a tiny jitter
            jitter = np.random.randn(n) * 5e-4
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    print(f"[hill climb] finished – best max‑conv {best_M:.6e}")
    return best_a
