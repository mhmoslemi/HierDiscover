def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    # fallback using height_sequence_1
    try:

        val=evaluate_sequence(fallback)
        best_seq=fallback
        best_score=val

    # try different n values

    for n in [8192, 16384]:
        if time.time() > deadline: break

        a = generate_arcsine(n)

        a /= a.sum()
        cur_score = compute_score_fft(a)  # fast score
        print(f"[n={n}] arcsine score = {cur_score:.6f}")
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
        # mixing with uniform
        uniform = np.full(n, 1.0/n)
        # small lam range up to 0.2 perhaps
        for lam in np.linspace(0.0, 0.2, 21):
            mixed = (1-lam)*a + lam*uniform
            mixed /= mixed.sum()
            sc = compute_score_fft(mixed)
            if sc < best_score:
                best_score, best_seq = sc, mixed.copy()
        # optional short polishing

        if time.time() < deadline:

            sc = compute_score_fft(a_polished)

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

The core idea is to use the discrete *arcsine* (Beta(½,½)) distribution.
Its continuous self‑convolution is exactly uniform on the interval,
so for a sufficiently large number of points the discrete self‑convolution
is almost flat and the objective 2·n·max(conv) is extremely close to the
theoretical lower bound 2·n/(2·n‑1) ≈ 1.

The algorithm
-------------
1. Choose a moderate size (8192 – 16384).  
2. Build a corrected arcsine shape (the denominator is √(i·(n‑i)) which
   avoids the division‑by‑zero bug of the previous version).  
3. Normalise to sum = 1 (the objective is scale‑invariant).  
4. Perform a cheap line‑search mixing the arcsine vector with the uniform
   vector; keep the best mixture.  
5. (Optional) a very short projected sub‑gradient polishing pass can be
   applied if time permits.  
6. Return the best vector found as a plain list of Python floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum(x)=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution using a real FFT (linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (full length 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective (fast version): 2·n·max(conv) assuming sum(a)=1
# ----------------------------------------------------------------------
def compute_score_fft(a: np.ndarray) -> float:
    """Compute the exact objective 2·n·max(conv(a,a)) (sum(a) assumed = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Arcsine distribution (Beta(0.5,0.5)) – corrected denominator
# ----------------------------------------------------------------------
def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine shape.
    a_i ∝ 1 / sqrt( (i+0.5)·(n‑(i+0.5)) ),  i = 0 … n‑1.
    The vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5    # bin centres
    denom = np.sqrt(i * (n - i))                # avoid zero at the ends
    a = 1.0 / denom
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Very short sub‑gradient polish (optional, only if time permits)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score_fft(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9                     # active tolerance
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of max(conv) = sum of gradients of active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score_fft(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional tiny jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The method builds a (large) arcsine shape,
    optionally mixes it with the uniform vector and applies a tiny
    polishing phase.  The final sequence is returned as a plain list.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final official evaluation
    deadline = start + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0.  Optional fallback (if a previous solution exists)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, np.array(fallback, dtype=float)
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1.  Try a few problem sizes (larger n → lower theoretical bound)
    # -----------------------------------------------------------------
    for n in (8192, 16384):
        if time.time() >= deadline:
            break
        print(f"\n[search] trying n = {n}")

        # ----- 1a. arcsine distribution (corrected) -----
        a = generate_arcsine(n)
        sc = compute_score_fft(a)
        print(f"[arcsine] initial score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a.copy()

        # ----- 1b. mix with uniform (tiny convex combination) -----
        uniform = np.full(n, 1.0 / n, dtype=float)
        # restrict lam to a modest range – large mixing would destroy the flatness
        for lam in np.linspace(0.0, 0.25, 13):
            mixed = (1.0 - lam) * a + lam * uniform
            mixed = mixed / mixed.sum()
            sc = compute_score_fft(mixed)
            if sc < best_score:
                best_score, best_seq = sc, mixed.copy()
        print(f"[mix] best score after mixing = {best_score:.6f}")

        # ----- 1c. Very short polishing if time remains -----
        if time.time() < deadline - 1.0:
            print("[polish] start short projected sub‑gradient")
            polish_deadline = min(deadline, time.time() + 2.0)   # ~2 s max
            a_polished = subgradient_polish(best_seq.copy(), polish_deadline,
                                            step0=0.20)
            sc = compute_score_fft(a_polished)
            print(f"[polish] refined score = {sc:.6f}")
            if sc < best_score:
                best_score, best_seq = sc, a_polished.copy()

    # -----------------------------------------------------------------
    # 2.  Safeguard – if nothing succeeded return a uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:

        best_score = compute_score_fft(best_seq)

    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq.tolist()
