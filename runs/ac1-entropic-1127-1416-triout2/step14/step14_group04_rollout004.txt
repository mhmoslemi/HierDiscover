Search for a non‑negative step‑function (vector) that minimizes
``evaluate_sequence`` – the scaled infinity‑norm of the self‑convolution.

The algorithm proceeds in four phases:

1. **Broad sampling** – random Dirichlet draws, monotone decreasing vectors,
   and *sparse* vectors (few non‑zero entries).  All candidates are projected
   onto the probability simplex and the best several are kept.

2. **Focused 1‑D sweeps** – dense scans of the exponential and power‑law families
   (the families that have proved fruitful in earlier runs).

3. **Projected gradient descent** – Adam optimisation on a *soft‑max*
   approximation of the max‑convolution.  The temperature is annealed
   slowly, and the update respects the simplex via the soft‑max re‑parameterisation.

4. **Final simulated‑annealing polish** – a Metropolis‑style hill‑climb that
   makes mass‑transfer moves and accepts worsening steps according to a
   decreasing temperature schedule.

All phases stop when the wall‑clock budget (≈ 1000 s) is exhausted.
The best vector found (as a plain Python list) is returned.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based linear convolution, simplex projection)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = rfft(a, n=n_fft)
    B = rfft(b, n=n_fft)
    C = A * B
    conv = irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x >= 0, sum x = 1}.
    Implementation of the O(n log n) algorithm by
    Duchi, Shalev‑Shwartz, Singer & Chandra (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only defined for 1‑D arrays")
    # shift by max to improve numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Exact objective (the value returned by evaluate_sequence after normalisation)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv(a,a)) assuming ``a`` sums to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sparse_random(n: int, m_nonzero: int) -> np.ndarray:
    """
    Randomly pick ``m_nonzero`` indices, put a Dirichlet‑drawn mass on them,
    and return a full length vector summing to 1.
    """
    idx = np.random.choice(n, size=m_nonzero, replace=False)
    # Dirichlet on the active coordinates
    raw = np.random.gamma(shape=1.0, scale=1.0, size=m_nonzero)
    raw /= raw.sum()
    a = np.zeros(n, dtype=float)
    a[idx] = raw
    return a


# ----------------------------------------------------------------------
# Simple bookkeeping utilities
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best):
    """Update ``best`` (a list of (score, vector)) with ``vec`` if it improves."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        return
    vec = vec / s
    sc = compute_score(vec)
    # keep only the top‑12 candidates
    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Phase 0 – optional fallback from a previously saved solution
# ----------------------------------------------------------------------


def _fallback_candidate(best, deadline):
    fallback = globals().get("height_sequence_1")
    if fallback is None:
        return
    try:
        fb = [float(x) for x in fallback]
        sc = compute_score(np.array(fb) / np.sum(fb))
        _try_candidate(np.array(fb) / np.sum(fb), best)
        print(f"[fallback] score = {sc:.6f}")
    except Exception:
        pass


# ----------------------------------------------------------------------
# Phase 1 – broad random baseline (Dirichlet + monotone + sparse)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float):
    """Collect a diverse pool of promising candidates."""
    best = []

    # 1) Dirichlet draws – a wide range of concentration parameters
    concs = [0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = 3000
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    # 2) Monotone decreasing vectors (sorted exponentials)
    for _ in range(8000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # 3) Sparse vectors – explore many support sizes
    sparse_sizes = list(range(30, 301, 30))  # 30, 60, …, 300
    draws_per_size = 1200
    for m in sparse_sizes:
        for _ in range(draws_per_size):
            if time.time() >= deadline:
                break
            a = generate_sparse_random(n, m)
            _try_candidate(a, best)

    # Return just the vectors (sorted by score)
    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Phase 2 – fine sweeps over exponential and power‑law families
# ----------------------------------------------------------------------


def fine_exponential_search(n: int, deadline: float):
    best_sc = float("inf")
    best_vec = None
    lam_vals = np.linspace(0.0003, 0.0040, 5000)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        a = generate_exponential(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec


def fine_powerlaw_search(n: int, deadline: float):
    best_sc = float("inf")
    best_vec = None
    p_vals = np.linspace(0.05, 5.0, 4000)
    for p in p_vals:
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec


# ----------------------------------------------------------------------
# Phase 3 – Adam optimisation on a soft‑max of the convolution
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.07,
                    init_tau_factor: float = 6.0,
                    min_tau: float = 1e-12,
                    tau_decay: float = 0.9993,
                    top_k: int = 12) -> np.ndarray:
    """
    Adam on the logits ``x`` with a soft‑max mapping a = softmax(x).
    The smooth max uses a temperature ``tau`` that slowly anneals.
    ``top_k`` limits the gradient to the largest convolution entries,
    which focuses optimisation on the true bottleneck.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits safely
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise temperature from the current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---------- soft‑max ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---------- convolution ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # ---------- focus on top‑k entries ----------
        if top_k >= conv.size:
            weights = np.exp((conv - max_c) / tau)
        else:
            # find indices of the largest entries
            idx_top = np.argpartition(conv, -top_k)[-top_k:]
            # build a mask that is zero elsewhere
            mask = np.zeros_like(conv)
            mask[idx_top] = 1.0
            # apply soft‑max only on the selected indices
            weights = np.zeros_like(conv)
            sub = conv[idx_top]
            max_sub = sub.max()
            w_sub = np.exp((sub - max_sub) / tau)
            w_sub /= w_sub.sum()
            weights[idx_top] = w_sub

        # ---------- gradient wrt a ----------
        # conv = a * a  → ∂max/∂a ≈ 2 * (a reversed * weights)
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, weights)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ---------- chain rule for logits ----------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- anneal temperature ----------
        tau = max(min_tau, tau * tau_decay)

        # occasional exact check
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Phase 4 – lightweight simulated annealing polish
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9994,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            # occasional jitter to prevent stagnation
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = proj_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the harness (its own bookkeeping)
    deadline = start_time + budget_s - 5.0

    # Problem size – 2048 offers a good balance of expressivity & speed.
    n = 2048

    best_val = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # 0. Optional fallback (previous best, if present)
    # --------------------------------------------------------------
    _fallback_candidate([], deadline)  # updates global best via side‑effects

    # --------------------------------------------------------------
    # 1. Wide‑range random baseline (≈ 250 s)
    # --------------------------------------------------------------
    baseline_deadline = deadline - 600.0
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best = {sc:.6f}")

    # --------------------------------------------------------------
    # 2. Fine 1‑D sweeps (≈ 180 s)
    # --------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    if time.time() < sweep_deadline:
        sc, vec = fine_exponential_search(n, sweep_deadline)
        if sc is not None and sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[exp‑sweep] new best = {sc:.6f}")

        if time.time() < sweep_deadline:
            sc, vec = fine_powerlaw_search(n, sweep_deadline)
            if sc is not None and sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[power‑sweep] new best = {sc:.6f}")

    # --------------------------------------------------------------
    # 3. Adam‑based smooth optimisation (≈ 250 s)
    # --------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, polish_deadline)
        sc = compute_score(refined)
        print(f"[smooth‑opt] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()

        # extra random restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            # start from a fresh sparse draw (moderate support)
            m = np.random.randint(40, 200)
            a_start = generate_sparse_random(n, m)
            refined = smooth_optimize(a_start, sub_deadline)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[smooth‑restart] new best = {sc:.6f}")

    # --------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 100 s)
    # --------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(final)
        print(f"[anneal] final polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # --------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # --------------------------------------------------------------
    # 6. Official evaluation (the harness will call ``evaluate_sequence``)
    # --------------------------------------------------------------
    official_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {official_score:.6f}")
    return best_seq
