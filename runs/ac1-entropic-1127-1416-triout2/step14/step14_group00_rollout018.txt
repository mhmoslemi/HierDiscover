def gradient_descent_to_flat(a, deadline, step0=0.1, momentum=0.9):
    a_vec = a.astype(float)
    v = np.zeros_like(a_vec)
    best_a = a_vec.copy()
    best_val = compute_score(a_vec)
    t = 0
    while time.time() < deadline:
        t += 1
        # compute conv and gradient
        conv = full_convolution(a_vec)   # O(N log N)
        avg = 1.0 / (2 * len(a_vec) - 1)
        diff = conv - avg
        # gradient
        rev_a = a_vec[::-1]
        grad_full = conv_fft(diff, rev_a)
        grad = 4.0 * grad_full[len(a_vec)-1 : len(a_vec)-1 + len(a_vec)]

        # Nesterov
        v = momentum * v - step0 / np.sqrt(t) * grad
        a_new = a_vec + v
        a_new = project_to_simplex(a_new)
        a_vec = a_new

        cur_val = compute_score(a_vec)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_vec.copy()
        # optional jitter
        if t % 1000 == 0:
            # reduce step size adaptively maybe
            step0 *= 0.99
    return best_a

def gradient_to_flat(a0, deadline, step0=0.5, momentum=0.9):
    a = a0.copy()
    v = np.zeros_like(a)
    n = len(a)
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        avg = 1.0 / (2 * n - 1)
        diff = conv - avg
        rev_a = a[::-1]
        grad_full = conv_fft(diff, rev_a)
        grad = 4.0 * grad_full[n-1 : n-1 + n]  # slice length n
        step = step0 / np.sqrt(t)
        v = momentum * v - step * grad
        a = a + v
        a = project_to_simplex(a)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # occasional jitter
        if t % 2000 == 0:
            jitter = np.random.randn(n) * step0 * 0.005 / np.sqrt(t)
            a += jitter
            a = project_to_simplex(a)
    return best_a

"""
search for a step‑function (non‑negative vector on the simplex) that
minimises ``evaluate_sequence``.

Strategy
--------
1.  Initialise with the discrete arcsine (Beta(½,½)) distribution – a known
    good shape whose self‑convolution is already quite flat.

2.  Refine the shape by a *smooth* gradient descent that minimises the
    squared deviation of the self‑convolution from the exact uniform
    distribution (average value = 1/(2 n‑1)).  The gradient is obtained
    analytically and evaluated with a single FFT per iteration.

3.  Switch to a sub‑gradient descent on the true objective
    ``2·n·max(conv)`` (the code used in the previous baseline) to polish
    the final peak.  A long run with a diminishing step size reduces the
    maximum convolution further.

4.  A short simulated‑annealing hill‑climb finishes the optimisation,
    allowing occasional uphill moves to escape shallow minima.

The whole search respects the 1000 s budget; intermediate best scores are
printed and the best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    n = len(u) + len(v) - 1
    n_fft = 1 << ((n - 1).bit_length())
    U = np.fft.rfft(u, n=n_fft)
    V = np.fft.rfft(v, n=n_fft)
    conv = np.fft.irfft(U * V, n=n_fft)[:n]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    return conv_fft(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective used for comparison: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for cheap initial candidates)
# ----------------------------------------------------------------------


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape: a_i ∝ (i+1)^{α−1}·(n−i)^{β−1}  (α,β>0)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) shape."""
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient descent toward a *flat* convolution
# ----------------------------------------------------------------------


def gradient_to_flat(a0: np.ndarray, deadline: float,
                    step0: float = 2.0,
                    momentum: float = 0.9) -> np.ndarray:
    """
    Projected gradient descent on the loss
        L(a) = Σ_k (conv_k(a) - μ)^2,
    where μ = 1/(2 n‑1) is the uniform target.
    The method uses a single FFT per iteration to obtain the gradient.
    """
    a = a0.copy()
    v = np.zeros_like(a)               # momentum buffer
    n = a.shape[0]
    μ = 1.0 / (2 * n - 1)              # target uniform value
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0

    while time.time() < deadline:
        t += 1

        # self‑convolution and deviation from the uniform target
        conv = full_convolution(a)
        diff = conv - μ                     # length 2 n‑1

        # gradient = 4 * (diff * reverse(a))_{center slice}
        rev_a = a[::-1]
        grad_full = conv_fft(diff, rev_a)   # length 3 n‑2
        grad = 4.0 * grad_full[n - 1: n - 1 + n]

        # Nesterov‑style momentum step
        step = step0 / np.sqrt(t)
        v = momentum * v - step * grad
        a = a + v
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional tiny jitter to avoid stagnation
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (same as in the reference implementation)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of max(conv) = sum of gradients of active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (keeps an upper bound)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # small safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Helper to update the global best using the fast ``compute_score``
    # -----------------------------------------------------------------
    def update_best(arr: np.ndarray):
        nonlocal best_seq, best_score
        sc = compute_score(arr)
        if sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"[update] new best score = {sc:.6f}")
        return sc

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]    # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_seq = np.array(fallback, dtype=float)
            best_score = val
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Try a couple of problem sizes – larger n gives a finer lower bound
    # -----------------------------------------------------------------
    for n in (2048, 4096):
        if time.time() >= deadline:
            break
        print(f"\n--- Working with n = {n} ---")

        # ----- arcsine start -------------------------------------------------
        a = generate_arcsine(n)
        sc = update_best(a)

        # ----- cheap mix with uniform (grid on λ) ---------------------------
        uniform = np.full(n, 1.0 / n)
        for lam in np.linspace(0.0, 1.0, 21):
            mixed = lam * uniform + (1.0 - lam) * a
            mixed = mixed / mixed.sum()
            sc = update_best(mixed)

        # ----- beta‑grid search (including α,β < 0.5) ------------------------
        alphas = np.linspace(0.10, 0.90, 9)
        betas = np.linspace(0.10, 0.90, 9)
        for alpha in alphas:
            for beta in betas:
                a_beta = generate_beta(n, alpha, beta)
                sc = update_best(a_beta)

        # -----------------------------------------------------------------
        # 2. Gradient descent toward a flat convolution (main workhorse)
        # -----------------------------------------------------------------
        if time.time() < deadline:
            gd_deadline = deadline - 30.0          # keep some room for polishing
            print("[gradient] start")
            a_gd = gradient_to_flat(best_seq if best_seq is not None else a,
                                    gd_deadline,
                                    step0=2.0,
                                    momentum=0.9)
            sc = update_best(a_gd)
            print(f"[gradient] refined score = {sc:.6f}")

        # -----------------------------------------------------------------
        # 3. Sub‑gradient polishing (focuses on the actual max)
        # -----------------------------------------------------------------
        if time.time() < deadline:
            print("[polish] start")
            polish_deadline = deadline - 10.0
            a_pol = subgradient_polish(best_seq, polish_deadline, step0=0.30)
            sc = update_best(a_pol)
            print(f"[polish] polished score = {sc:.6f}")

        # -----------------------------------------------------------------
        # 4. Simulated annealing for a final fine‑tune
        # -----------------------------------------------------------------
        if time.time() < deadline:
            print("[anneal] start")
            a_ann = simulated_annealing(best_seq, deadline)
            sc = update_best(a_ann)
            print(f"[anneal] final score = {sc:.6f}")

    # -----------------------------------------------------------------
    # Fallback – guarantee a valid return
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 2048
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)
        print("[fallback uniform] score =", best_score)

    # -----------------------------------------------------------------
    # Final official evaluation (the harness will apply its own clipping)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")

    return best_seq.tolist()
