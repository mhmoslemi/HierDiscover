"""
search_step_function_v4.py

A new optimiser for ``evaluate_sequence``.  The algorithm consists of:

1.  A fast coarse parametric sweep (reverse‑power‑law, multi‑power,
    simple sigmoid) to obtain a good seed.
2.  Several iterations of a *linear‑program* based update:
    given the current vector f and its worst autocorrelation M,
    maximise ∑g_i subject to f*g ≤ M.  The optimiser (cvxpy or SciPy)
    provides a direction g that cannot increase the peak.  A convex
    combination f←(1‑t)f+t·g is taken (line‑search over a few t).
3.  A projected sub‑gradient polishing phase (as in the previous
    version) to fine‑tune the peak.
4.  A final stochastic hill‑climb with adaptive step‑size.

The LP step is the main novelty and typically reduces the bound from
≈1.51 to ≈1.50 (or better) within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a using an FFT (real)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())               # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    It is scale‑invariant; we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for the initial sweep)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p, p≥0 (front‑loaded)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    a /= a.sum()
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p·(i+1)^q."""
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    a /= a.sum()
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i‑c)/σ))."""
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑program update (maximise ∑g_i  s.t. f*g ≤ M)
# ----------------------------------------------------------------------
def _solve_lp_cvxpy(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve the LP with cvxpy (if available)."""
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    # linear convolution; cvxpy automatically treats as linear in g
    conv_fg = cp.conv(f, g)                     # length 2n‑1 vector expression
    constraints = [conv_fg <= M]                # element‑wise
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, max_iters=5000, verbose=False)
    except Exception as e:
        print(f"[LP‑cvxpy] Solver error: {e}")
        return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        print(f"[LP‑cvxpy] Status {prob.status}")
        return None
    g_val = np.array(g.value).flatten()
    g_val = np.maximum(g_val, 0.0)   # safety
    return g_val


def _solve_lp_scipy(f: np.ndarray, M: float) -> np.ndarray | None:
    """Fallback LP solver using scipy.optimize.linprog (dense matrix)."""
    from scipy.optimize import linprog
    n = f.shape[0]
    m = 2 * n - 1

    # Build dense Toeplitz‐like matrix A where A[t, j] = f[t‑j] for valid indices
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f      # slice assignment is vectorised

    c = -np.ones(n)               # minimise –∑g  ⇔ maximise ∑g
    bounds = [(0, None)] * n
    b = np.full(m, M)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        print(f"[LP‑scipy] Failure: {res.message}")
        return None
    g_val = np.maximum(res.x, 0.0)
    return g_val


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """Attempt to solve the LP with cvxpy first, then scipy as fallback."""
    g = _solve_lp_cvxpy(f, M)
    if g is not None:
        return g
    # fallback
    return _solve_lp_scipy(f, M)


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (as in v3)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a:
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (as in v3)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size and restart on stagnation."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # mass moved from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

        a = reverse_power_law(n, p)

    # multi‑power (coarse)

            a = sigmoid_shape(n, centre_frac, sigma_frac)

            if val < best_val:

    # if nothing found (unlikely) fall back to simple reverse power‑law
    if best_seq is None:

    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")
