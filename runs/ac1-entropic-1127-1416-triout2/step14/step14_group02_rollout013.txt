def flatten_adam(a0, deadline, lr_start=0.5, lr_decay=0.96):
   a = a0.copy()
   n = len(a)
   target = 1.0 / (2*n - 1)
   m = 2*n - 1
   size_fft = 1 << ((3*n - 3).bit_length())
   m_t = np.zeros_like(a)
   v_t = np.zeros_like(a)
   best = a.copy()
   best_val = 2*n * np.max(full_convolution(a))  # compute max ratio (score)
   t=0
   while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        err = conv - target
        grad = 2.0 * correlation_grad(a, err, size_fft)
        # Adam update
        m_t = beta1 * m_t + (1-beta1)*grad
        v_t = beta2 * v_t + (1-beta2)*(grad**2)
        m_hat = m_t / (1-beta1**t)
        v_hat = v_t / (1-beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)
        cur_val = 2*n * np.max(full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
        # schedule modifications: decay lr per certain steps.
        if t % 1000 == 0:
            lr = max(lr*lr_decay, 1e-5)
        if t % 5000 == 0:
            # jitter
            a += np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = project_to_simplex(a)
   return best
```

"""
Two‑stage optimiser for the self‑convolution max problem.

Stage 1 – “flattening”:
    minimise the L2 distance of the full convolution to the constant
    target (the average 1/(2n‑1)).  The optimisation is performed with an
    Adam optimiser on the smooth objective  ∑(conv−target)².  This drives the
    convolution towards a flat shape and consequently reduces the L∞‑norm.

Stage 2 – “max‑reduction”:
    starting from the best flattened candidate we minimise a smooth
    approximation of the true objective 2·n·max(conv) using the
    log‑sum‑exp surrogate.  A second Adam run with only the max‑term
    (no flattening) is used, followed by an exact projected sub‑gradient
    polish and a short simulated‑annealing hill‑climb.

All vectors are kept on the probability simplex (sum = 1, non‑negative)
by Euclidean projection.  The whole pipeline respects the 1000 s wall‑clock
budget and constantly stores the best sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    # power‑of‑two size for efficiency
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_grad(a: np.ndarray, vec: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of <a, a>⋅vec (i.e. of ½∑(conv−c)² or of the smooth‑max).
    Returns 2·(a ⋆ vec) where ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]
    m = vec.shape[0]
    rev = vec[::-1]
    A = np.fft.rfft(a, n=size_fft)
    B = np.fft.rfft(rev, n=size_fft)
    full = np.fft.irfft(A * B, n=size_fft)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Exact objective for a normalised vector (sum = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for the cheap global baseline)
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_logistic(n: int, scale: float, mid: float) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mid) / scale))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(v1: np.ndarray, v2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * v1 + (1.0 - lam) * v2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global cheap baseline (many analytic families + Dirichlet draws)
# ----------------------------------------------------------------------
def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 15,
    random_draws: int = 25000,
) -> list[np.ndarray]:
    """Explore a wide set of analytic families and random Dirichlet draws."""
    best = []   # list of tuples (score, vec)

    def consider(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # --- analytic families ------------------------------------------------
    for p in np.linspace(0.05, 5.0, 101):
        if time.time() >= deadline: break
        consider(generate_powerlaw(n, p))

    for lam in np.linspace(0.00005, 0.6, 150):
        if time.time() >= deadline: break
        consider(generate_exponential(n, lam))

    al_vals = np.linspace(0.01, 2.5, 35)
    be_vals = np.linspace(0.01, 2.5, 35)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline: break
            consider(generate_beta(n, alpha, beta))

    for sigma in np.linspace(0.15 * n, 0.5 * n, 40):
        if time.time() >= deadline: break
        consider(generate_gaussian(n, sigma))

    for p in np.linspace(0.1, 10.0, 45):
        if time.time() >= deadline: break
        consider(generate_sine_power(n, p))

    # logistic family (various scales and mid‑points)
    for scale in np.linspace(0.5, 30.0, 30):
        for mid_shift in np.linspace(0.3, 0.7, 5):
            if time.time() >= deadline: break
            mid = mid_shift * n
            consider(generate_logistic(n, scale, mid))

    # mixtures of two families
    for _ in range(150):
        if time.time() >= deadline: break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(n,
                          10 ** np.random.uniform(-2, 0.4),
                          10 ** np.random.uniform(-2, 0.4))
        lam_mix = np.random.rand()
        consider(mixture(a, b, lam_mix))

    # --- Dirichlet draws --------------------------------------------------
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    per_conc = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(per_conc):
            if time.time() >= deadline: break
            vec = np.random.dirichlet([conc] * n)
            consider(vec)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser for the L2‑flattening objective
# ----------------------------------------------------------------------
def flatten_adam(
    a0: np.ndarray,
    deadline: float,
    lr_start: float = 0.55,
    lr_decay: float = 0.96,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """Adam on  ∑(conv−target)²  (pure flattening)."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    target = 1.0 / m
    size_fft = 1 << ((3 * n - 3).bit_length())

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    lr = lr_start

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        err = conv - target
        grad = 2.0 * correlation_grad(a, err, size_fft)

        # Adam update
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # schedule
        if t % 800 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.002 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max surrogate (λ = 0)
# ----------------------------------------------------------------------
def max_smooth_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-7,
    tau_decay: float = 0.94,
    lr_start: float = 0.18,
    lr_decay: float = 0.97,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """Adam minimising the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    size_fft = 1 << ((3 * n - 3).bit_length())

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # schedules
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # smooth max
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)  # stable
        w = exp_vals / exp_vals.sum()            # soft‑max weights

        grad = correlation_grad(a, w, size_fft)

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # jitter to stay out of flat regions
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------
def subgradient_polish(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.35,
) -> np.ndarray:
    """Exact sub‑gradient polish of 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from every active shift
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------
def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-7,
    cooling_factor: float = 0.9995,
    step_scale_init: float = 0.05,
    step_scale_min: float = 5e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a = a + jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    ``evaluate_sequence``. Returns a plain list of floats (sum is irrelevant;
    the internal optimisation works with vectors normalised to sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # -----------------------------------------------------------------
    # Problem size – a larger n gives a tighter theoretical bound.
    # -----------------------------------------------------------------
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previously discovered construction)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈150 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + 150.0)

        candidates = baseline_search(
            n, baseline_deadline, max_keep=15, random_draws=30000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known (exact) score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Flattening Adam phase (≈300 s)
    # -----------------------------------------------------------------
    flatten_deadline = min(deadline, time.time() + 300.0)
    if best_seq is not None and time.time() < flatten_deadline:
        start_vecs = [
            np.array(best_seq, dtype=float),
            np.full(n, 1.0 / n),
        ]
        # add a few random Dirichlet starts
        for _ in range(3):
            start_vecs.append(project_to_simplex(np.random.rand(n)))

        per_vec_time = max(30.0, (flatten_deadline - time.time()) / len(start_vecs))
        for idx, sv in enumerate(start_vecs):
            if time.time() >= flatten_deadline:
                break
            local_dead = time.time() + per_vec_time
            a_flat = flatten_adam(sv, local_dead, lr_start=0.55)
            val = compute_score(a_flat)
            print(f"[flatten‑adam {idx}] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_flat.tolist()
        print(f"[flatten‑adam] best after phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Max‑smooth Adam phase (≈300 s)
    # -----------------------------------------------------------------
    maxadam_deadline = min(deadline, time.time() + 300.0)
    if best_seq is not None and time.time() < maxadam_deadline:
        a_start = np.array(best_seq, dtype=float)

        # several short restarts from slightly perturbed versions
        for restart in range(4):
            if time.time() >= maxadam_deadline:
                break
            pert = a_start + np.random.randn(n) * 1e-3
            pert = project_to_simplex(pert)
            local_dead = time.time() + min(70.0, (maxadam_deadline - time.time()) / (4 - restart))
            a_ref = max_smooth_adam(pert, local_dead,
                                    tau_start=0.30, tau_end=1e-7,
                                    lr_start=0.18)
            val = compute_score(a_ref)
            print(f"[max‑adam {restart}] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_ref.tolist()
        print(f"[max‑adam] best after phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing (≈130 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.33)
        val = compute_score(a_polished)
        print(f"[polish] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 5. Simulated annealing (final ~100 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline,
                                      init_temp=2e-3,
                                      final_temp=1e-7,
                                      cooling_factor=0.9995,
                                      step_scale_init=0.05)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
