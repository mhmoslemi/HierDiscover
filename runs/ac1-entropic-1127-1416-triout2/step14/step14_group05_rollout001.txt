"""
Search for a non‑negative step‑function that minimizes the black‑box
``evaluate_sequence``.  The algorithm explores several vector lengths,
initialises each length with a few Dirichlet draws (and a fallback if
available), then refines the best candidate by

1. Sub‑gradient descent on the true max‑convolution objective.
   The sub‑gradient of  max_k (a*a)[k]  w.r.t. a_i is 2·a_{k*‑i},
   where k* is the index attaining the current maximum.  A decreasing
   step‑size and Euclidean projection onto the probability simplex keep
   the iterate feasible.

2. Targeted mass‑transfer hill‑climbing.  At every iteration the entry
   that contributes most to the current maximal convolution value is
   identified and a small fraction of its mass is moved to a low‑mass
   location, again projected onto the simplex.  This focuses the search
   on the true bottleneck rather than on a smooth surrogate.

The two phases are run for a time‑share of the remaining budget for each
vector length.  The best vector over all lengths is finally reported
through ``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (simplex projection, convolution, exact score)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1, a_i ≥ 0).
    Returns 2·n·max_k (a*a)[k].
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, deadline: float,

    Sub‑gradient descent for  f(a)=max_k (a*a)[k]  with Euclidean projection
    onto the simplex.  The step size decays as  α / sqrt(t+1).
    The function returns the best vector seen before *deadline*.
    """
    a = project_to_simplex(a0.copy())
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        conv = full_convolution(a)
        k = int(conv.argmax())

        # sub‑gradient g_i = 2·a_{k‑i} for valid indices
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_i = np.arange(lo, hi + 1)
        idx_j = k - idx_i
        g[idx_i] = 2.0 * a[idx_j]

        # decreasing step size
        step = initial_step / np.sqrt(t + 1.0)

        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        val = compute_score(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[sgd] iter={t:06d}, step={step:.3e}, score={val:.6f}")

        a = a_new
        t += 1

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (initial_step * 0.01 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – targeted mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def target_mass_transfer(a0: np.ndarray, deadline: float,
                         transfer_frac: float = 0.05) -> np.ndarray:
    """
    Focused hill‑climb: repeatedly move a small fraction of the mass from the
    pair (i,j) that contributes most to the current max‑convolution entry.
    The move is projected onto the simplex.  Returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        conv = full_convolution(a)
        k = int(conv.argmax())

        # indices that form the max entry
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        # pair with largest contribution
        max_idx = np.argmax(contrib)
        i_src = i_range[max_idx]
        j_src = j_range[max_idx]

        # amount to move
        delta_i = a[i_src] * transfer_frac
        delta_j = a[j_src] * transfer_frac if i_src != j_src else 0.0

        # choose destinations with minimal current mass
        dst_i = np.argmin(a)
        if dst_i == i_src or dst_i == j_src:
            dst_i = (dst_i + 1) % n
        a[i_src] -= delta_i
        a[dst_i] += delta_i

        if i_src != j_src:
            dst_j = np.argmin(a)
            if dst_j == i_src or dst_j == j_src or dst_j == dst_i:
                dst_j = (dst_j + 1) % n
            a[j_src] -= delta_j
            a[dst_j] += delta_j

        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[mass] iter={it:06d}, score={val:.6f}")

        # occasional random swap to diversify
        if np.random.rand() < 0.001:
            i, j = np.random.choice(n, 2, replace=False)
            a[i], a[j] = a[j], a[i]
            a = project_to_simplex(a)

        it += 1

    return best_a

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The routine explores several lengths, performs
    sub‑gradient descent followed by a focused mass‑transfer hill‑climb,
    and finally returns the best vector found (as a list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so we always finish before the hard limit
    deadline = start_time + budget_s - 5.0

    # possible vector lengths to explore (powers of two up to 8192)
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    # keep a global best
    best_vec = None
    best_score = float('inf')

    # try to use the known construction if it exists
    fallback_vec = None
    try:
        fallback_vec = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # allocate an equal share of the *remaining* time to the remaining lengths
        remaining = deadline - time.time()
        share = remaining / (len(lengths) - idx)
        print(f"[main] length={n}, time share≈{share:.1f}s")

        # ------------------------------------------------------------------
        # 0. generate a few starting vectors (fallback, uniform, Dirichlet)
        # ------------------------------------------------------------------
        starts = []

        if fallback_vec is not None and len(fallback_vec) == n:
            a0 = np.array(fallback_vec, dtype=float)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-12:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / s
            starts.append(a0)
            print("[init] using fallback as start")

        # uniform vector
        starts.append(np.full(n, 1.0 / n))

        # several random Dirichlet draws
        for _ in range(4):
            starts.append(np.random.dirichlet(np.ones(n)))

        # keep the best of the cheap seeds
        local_best = None
        local_best_val = float('inf')
        for a in starts:
            val = compute_score(a)
            if val < local_best_val:
                local_best_val = val
                local_best = a.copy()
        print(f"[seed] best initial score for n={n}: {local_best_val:.6f}")

        # ------------------------------------------------------------------
        # 1. sub‑gradient descent (major part of the share)
        # ------------------------------------------------------------------
        sub_deadline = time.time() + share * 0.6
        if time.time() < sub_deadline:
            a_sgd = subgradient_descent(local_best, sub_deadline)
            val_sgd = compute_score(a_sgd)
            if val_sgd < local_best_val:
                local_best_val = val_sgd
                local_best = a_sgd.copy()
            print(f"[sgd] finished for n={n}, score={val_sgd:.6f}")

        # ------------------------------------------------------------------
        # 2. targeted mass‑transfer hill‑climb (remaining share)
        # ------------------------------------------------------------------
        mass_deadline = time.time() + share * 0.3
        if time.time() < mass_deadline:
            a_mass = target_mass_transfer(local_best, mass_deadline)
            val_mass = compute_score(a_mass)
            if val_mass < local_best_val:
                local_best_val = val_mass
                local_best = a_mass.copy()
            print(f"[mass] finished for n={n}, score={val_mass:.6f}")

        # ------------------------------------------------------------------
        # 3. optional tiny random‑restart Adam phase (very short)
        # ------------------------------------------------------------------
        # Give a few seconds to a light Adam run if any time left
        if time.time() < deadline - 5.0:
            short_deadline = time.time() + min(5.0, share * 0.1)
            # reuse the smooth‑max Adam from the previous code base
            try:
                from math import log

                # a minimal Adam variant that works on the smooth‑max surrogate
                def smooth_max_adam(a0, dl, tau_start=1.0, lr_start=0.03):
                    a = a0.copy()
                    n = a.shape[0]
                    m = 2 * n - 1
                    mt = np.zeros_like(a)
                    vt = np.zeros_like(a)
                    best = a.copy()
                    bestv = compute_score(a)
                    t = 0
                    tau = tau_start
                    lr = lr_start
                    while time.time() < dl:
                        if t % 300 == 0:
                            tau = max(1e-6, tau * 0.99)
                        conv = full_convolution(a)
                        shift = conv.max()
                        w = np.exp((conv - shift) / tau)
                        w = w / w.sum()
                        rev_w = w[::-1]
                        sz = 1 << ((n + rev_w.shape[0] - 2).bit_length())
                        A = np.fft.rfft(a, n=sz)
                        B = np.fft.rfft(rev_w, n=sz)
                        grad_full = np.fft.irfft(A * B, n=sz)[: n + rev_w.shape[0] - 1]
                        idx = (m - 1) - np.arange(n)
                        grad = 2.0 * grad_full[idx]

                        t += 1
                        mt = 0.9 * mt + 0.1 * grad
                        vt = 0.999 * vt + 0.001 * (grad ** 2)
                        m_hat = mt / (1 - 0.9 ** t)
                        v_hat = vt / (1 - 0.999 ** t)
                        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)
                        a = a - step
                        a = project_to_simplex(a)

                        curv = compute_score(a)
                        if curv < bestv:
                            bestv = curv
                            best = a.copy()
                    return best

                a_adam = smooth_max_adam(local_best, short_deadline)
                v_adam = compute_score(a_adam)
                if v_adam < local_best_val:
                    local_best_val = v_adam
                    local_best = a_adam.copy()
                print(f"[adam] short run finished for n={n}, score={v_adam:.6f}")
            except Exception:
                # if anything goes wrong, just skip this tiny Adam phase
                pass

        # ------------------------------------------------------------------
        # keep the best overall solution seen so far
        # ------------------------------------------------------------------
        if local_best_val < best_score:
            best_score = local_best_val
            best_vec = local_best.copy()
            print(f"[global] new best overall score {best_score:.6f} (n={n})")

    # ----------------------------------------------------------------------
    # 5. final evaluation & return
    # ----------------------------------------------------------------------
    if best_vec is None:
        # fallback to a uniform vector of length 2048 if everything failed
        n = 2048
        best_vec = np.full(n, 1.0 / n)

    final_seq = best_vec.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq
