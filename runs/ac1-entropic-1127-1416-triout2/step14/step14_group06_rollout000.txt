"""
search_step_function_v4.py

A more aggressive optimiser for ``evaluate_sequence`` that aims at
the sub‑2 bound (≈1.5).  The search proceeds in three stages:

1. **Wide parametric sweep** over several families (reverse power‑law,
   geometric, beta‑shape, mixed reverse‑uniform).  The sweep uses a
   finer grid and a larger exponent range than earlier versions and
   stops as soon as the time budget is exhausted.

2. **Projected sub‑gradient polishing** – a classic sub‑gradient
   descent on the max‑convolution, with a decaying step‑size and a
   small random jitter to escape shallow plateaus.

3. **Simulated‑annealing hill‑climb** – stochastic moves that transfer
   a small amount of mass between two positions; moves are accepted
   according to a temperature schedule, which allows occasional uphill
   steps and therefore a better chance of escaping local minima.

The code automatically picks a vector length (2048, 4096 or 8192)
depending on the remaining budget, and it prints progress messages
so you can see how the best score evolves.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT (fast for long vectors)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Scale‑invariant score for a vector normalised to sum = 1:
        score = 2·n·max(conv(a,a))
    Lower is better.
    """
    a = a.astype(np.float64, copy=False)
    a_sum = a.sum()
    if a_sum == 0:
        return float('inf')
    a = a / a_sum
    conv = full_convolution(a)
    return float(2.0 * a.shape[0] * conv.max())

# Families of candidate vectors

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law), p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    if not (0.0 < q < 1.0):
        return None
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    Convex combination   mix * reverse_power_law(p) + (1‑mix) * uniform,
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

# Parametric sweep – more exhaustive than before
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Explore a variety of parameterised families, keep the best raw vector."""
    best_vec = None
    best_val = float('inf')
    t0 = time.time()

    # ----------- reverse power‑law (wide range) -----------------------
    # Coarse grid: p ∈ [0.2, 30] step 0.2
    for p in np.arange(0.2, 30.0 + 1e-12, 0.2):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, float(p))
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # Fine grid around the best coarse p (±0.05 step 0.001)
    if best_vec is not None:
        # Re‑discover a good centre p by a tiny refined scan
        p_center = None
        for p in np.arange(0.2, 30.0 + 1e-12, 0.05):
            if time.time() >= deadline:
                break
            a = reverse_power_law(n, float(p))
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a
                p_center = p
        if p_center is not None:
            lo = max(0.2, p_center - 0.2)
            hi = min(30.0, p_center + 0.2)
            for p in np.arange(lo, hi + 1e-12, 0.001):
                if time.time() >= deadline:
                    return best_vec
                a = reverse_power_law(n, float(p))
                if a is None:
                    continue
                val = compute_score(a)
                if val < best_val:
                    best_val, best_vec = val, a

    # ----------- geometric family ------------------------------------
    for q in np.linspace(0.90, 0.9995, 30):
        if time.time() >= deadline:
            break
        a = geometric(n, float(q))
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ----------- mixed reverse‑uniform -------------------------------
    for mix in np.linspace(0.3, 0.95, 14):
        for p in np.linspace(0.5, 6.0, 36):
            if time.time() >= deadline:
                break
            a = mixed_reverse_uniform(n, float(mix), float(p))
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----------- beta shape (broader exponent ranges) ---------------
    alphas = np.linspace(-2.0, 12.0, 30)
    betas = np.linspace(-2.0, 12.0, 30)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = beta_shape(n, float(alpha), float(beta))
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----------- random convex combinations of a few basics ----------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rp1 = reverse_power_law(n, 1.5)
    rp2 = reverse_power_law(n, 3.0)
    geom = geometric(n, 0.985)
    basics = [uniform, rp1, rp2, geom]

    for _ in range(3000):
        if time.time() >= deadline:
            break
        # Dirichlet‑type random weights
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    elapsed = time.time() - t0
    print(f"[parametric sweep] completed in {elapsed:.1f}s, best score = {best_val:.6f}")
    return best_vec

# Sub‑gradient polishing (as in the reference but with a few tweaks)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.18) -> np.ndarray:
    """Projected sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: sum of shifted copies of a for each active index
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        # occasional jitter to escape flat regions
        if t % 1800 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Stochastic hill‑climb with a simple temperature schedule.
    Moves transfer a small fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # Temperature schedule: start high, decay to near zero
    T0 = 0.05
    T_min = 1e-5
    total_iters = 0
    max_iters = 5_000_000  # safety cap

    while time.time() < deadline and total_iters < max_iters:
        # linear cooling based on elapsed fraction
        fraction = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - (deadline - time.time())))
        # Actually we don't have start time here, so approximate using iteration count
        T = max(T_min, T0 * (0.9995 ** total_iters))

        # propose a move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a tiny fraction of mass from i to j
        delta_frac = np.random.rand() * 0.02   # up to 2 %
        delta = delta_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        delta_score = cand_score - best_score

        # accept if improvement, otherwise with Metropolis probability
        if cand_score < best_score or np.random.rand() < math.exp(-delta_score / T):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        total_iters += 1

    print(f"[annealing] performed {total_iters} iterations, best score = {best_score:.6f}")
    return best_a
