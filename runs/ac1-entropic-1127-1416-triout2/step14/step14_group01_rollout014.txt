"""
Improved search for a step‐height sequence.

The evaluation function is scale‑invariant, so we keep the vector on the
probability simplex (∑aᵢ = 1) and only work with non‑negative entries.
The algorithm consists of three phases:

1. **Initialisation** – use the provided construction
   ``height_sequence_1`` if it exists, otherwise start from the uniform
   distribution.

2. **Simulated annealing / hill‑climbing** – repeatedly move a small
   amount of mass from one coordinate to another.  Moves that improve the
   true objective are always accepted; worsening moves are accepted with
   a temperature‑controlled probability.  The best solution seen so far
   is stored.

3. **Optional polishing** – a short Adam‑style gradient descent on the
   smoothed “soft‑max” of the convolution is run, again projecting onto
   the simplex after each step.

The whole search stops before the supplied time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real output)."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power‑of‑two for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def objective_raw(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a probability‑normalised
    vector (sum = 1).  This is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Numerically stable softmax with temperature 1/β."""
    # larger β → sharper approximation of max
    mx = np.max(beta * x)
    e = np.exp(beta * x - mx)
    s = np.sum(e)
    return e / s


def grad_softmax_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max (soft‑max) of the convolution a*a.
    The derivative of conv[k] w.r.t. a[i] is 2·a[k‑i] (if in range).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    p = softmax(conv, beta)                     # shape (2n‑1,)
    # We need g_i = 2 * sum_k p_k * a_{k-i}
    # This is a convolution of a with the reversed p vector.
    p_rev = p[::-1]
    # convolve a with p_rev – length 2n‑1
    tmp = np.convolve(a, p_rev, mode='full')
    # Extract the needed n entries: index (2n‑2 - i) for i=0..n‑1
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * tmp[idx]
    return grad


def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  beta_start: float = 5.0,
                  beta_end: float = 200.0,
                  lr: float = 0.05) -> np.ndarray:
    """
    Adam optimiser on the smoothed max‑convolution.  The vector is kept on
    the simplex after each update.  The temperature (beta) is increased
    linearly from ``beta_start`` to ``beta_end``.
    """
    a = a0.copy()
    n = a.shape[0]
    m_m = np.zeros_like(a)
    v_m = np.zeros_like(a)
    beta = beta_start
    beta_increment = (beta_end - beta_start) / max(1, int((deadline - time.time()) * 100))
    t = 0
    while time.time() < deadline:
        t += 1
        # update temperature
        if beta < beta_end:
            beta = min(beta_end, beta + beta_increment)

        # gradient of smooth max (ignoring the 2·n factor – it is constant)
        g = grad_softmax_max(a, beta)

        # Adam moments
        m_m = 0.9 * m_m + 0.1 * g
        v_m = 0.999 * v_m + 0.001 * (g * g)
        m_hat = m_m / (1 - 0.9 ** t)
        v_hat = v_m / (1 - 0.999 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a -= step
        a = project_to_simplex(a)   # enforce a_i ≥ 0, sum = 1

    return a


def simulated_annealing(start: np.ndarray,
                        deadline: float,
                        init_temp: float = 0.05,
                        cooling_rate: float = 0.9995,
                        max_move_frac: float = 0.02) -> np.ndarray:
    """
    Simple SA / hill‑climbing on the *exact* evaluate_sequence objective.
    Moves consist of transferring a small fraction of mass between two
    random coordinates.
    """
    cur = start.copy()
    cur_score = evaluate_sequence(cur.tolist())
    best = cur.copy()
    best_score = cur_score

    n = cur.shape[0]
    T = init_temp
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)

        # amount to move (fraction of a[i])
        delta = max_move_frac * cur[i] * np.random.rand()

        if delta <= 0:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # evaluate
        val = evaluate_sequence(new.tolist())
        if val < best_score:
            best_score = val
            best = new.copy()
            print(f"[SA] iter {it:6d} – new best  {best_score:.6f}")

        # acceptance test
        if val < cur_score or np.random.rand() < math.exp(-(val - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = val

        # temperature cooling
        T *= cooling_rate
        if T < 1e-7:
            T = 1e-7

        # occasional random restart if we are stuck
        if it % 20000 == 0:
            if np.random.rand() < 0.1:
                cur = np.random.dirichlet(np.ones(n))
                cur_score = evaluate_sequence(cur.tolist())
    return best

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑height sequence that minimises
    ``evaluate_sequence``.  The routine combines a fallback construction,
    a simulated‑annealing hill‑climb and a short Adam polish.  The best
    sequence found before the time budget expires is returned.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 0. Start from the known construction if available
    # --------------------------------------------------------------
    try:
        init_seq = np.array([float(x) for x in height_sequence_1], dtype=float)  # type: ignore[name-defined]
        print("[fallback] using provided height_sequence_1")
    except Exception:
        # uniform start if no fallback is defined
        n = 2048
        init_seq = np.full(n, 1.0 / n, dtype=float)
        print("[fallback] no external construction – using uniform start")

    # normalise to the probability simplex (sum = 1)
    init_seq = np.maximum(init_seq, 0.0)
    init_seq = init_seq / init_seq.sum()

    # keep track of the best sequence / score seen so far
    best_seq = init_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Simulated annealing / hill‑climbing
    # --------------------------------------------------------------
    sa_deadline = deadline - 30.0          # leave a little time for polishing
    if time.time() < sa_deadline:
        print("[SA] start hill‑climbing")
        candidate = simulated_annealing(best_seq, sa_deadline,
                                        init_temp=0.08,
                                        cooling_rate=0.9997,
                                        max_move_frac=0.04)
        cand_score = evaluate_sequence(candidate.tolist())
        print(f"[SA] finished – best score from SA = {cand_score:.6f}")
        if cand_score < best_score:
            best_score, best_seq = cand_score, candidate

    # --------------------------------------------------------------
    # 2. Adam‑style polishing on the smoothed max
    # --------------------------------------------------------------
    polish_deadline = deadline - 5.0
    if time.time() < polish_deadline:
        print("[polish] start Adam refinement")
        polished = adam_optimize(best_seq, polish_deadline,
                                 beta_start=10.0,
                                 beta_end=250.0,
                                 lr=0.06)
        pol_score = evaluate_sequence(polished.tolist())
        print(f"[polish] finished – polished score = {pol_score:.6f}")
        if pol_score < best_score:
            best_score, best_seq = pol_score, polished

    # --------------------------------------------------------------
    # 3. Final sanity check – ensure we have a valid answer
    # --------------------------------------------------------------
    if best_seq is None:
        # should never happen, but fall back to uniform
        n = 2048
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch completed. Final reported score = {final_score:.6f}")
    return best_seq.tolist()
