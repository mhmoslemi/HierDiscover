"""
Optimiser using a linear‑programming refinement loop.

The algorithm works as follows:

1. Choose a moderate length (default 256) to keep each optimisation fast.
2. Initialise the step‑function with a few random Dirichlet draws.
3. Repeatedly solve a linear programme:
       maximise   Σ g_j
       subject to g ≥ 0  and  (f * g)_k ≤ M   for all k,
   where M = max_k (f * f)_k for the current vector f.
   The LP tries to find a new vector g that respects the current worst
   convolution level M but carries more total mass.  After normalising g
   to sum 1 we either accept it (if it already improves the true objective)
   or mix it with f (f ← (1‑t)f + t g) and keep the better of the two.
   This “direction‑finding’’ step is the core idea from the paper
   referenced in the prompt.
4. After a few LP refinements we polish the result with a lightweight
   mass‑transfer hill‑climb (the same routine used in the previous baseline).
5. The final sequence is returned as a plain Python list; the official
   ``evaluate_sequence`` is called only once at the end to report the
   true score.

The whole search is time‑aware and stops well before the deadline,
always keeping the best candidate found so far.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Basic numerical utilities (same as in the reference implementation)
# --------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (fast for moderate n)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector normalised to unit sum (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Linear‑programming refinement (the key new ingredient)
# --------------------------------------------------------------

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A (shape (2n‑1, n)) such that
        (f * g) = A @ g
    for any vector g of the same length as f.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # Adding f[i] to column j at row i+j.
    eye = np.eye(n, dtype=np.float64)
    for i in range(n):
        A[i:i + n, :] += f[i] * eye
    return A


def lp_maximize_sum(f: np.ndarray, M: float):
    """
    Solve the LP:
        maximise   Σ g_j
        subject to g ≥ 0,  A @ g ≤ M,
    where A encodes the convolution with the current f.
    Returns the optimal g as a NumPy array (or None if the solver fails).
    """
    import cvxpy as cp

    n = f.shape[0]
    A = build_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-7, reltol=1e-7, feastol=1e-7, verbose=False)
    except Exception:          # pragma: no cover
        return None
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


# --------------------------------------------------------------
# Light‑weight hill‑climb used for final polishing
# --------------------------------------------------------------

def mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Move a small fraction of mass from one index to another.
    Accepts only improving moves; stops when time runs out.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # Transfer a random fraction (up to 30 % of the source mass)
        frac = np.random.rand() * 0.30
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # already on the simplex (mass conserved)
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:
                print(f"[mass] iter={it}, score={val:.6f}")
        it += 1
    return best_a

# Main entry point required by the harness
# --------------------------------------------------------------

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The method combines random seeding,
    an LP‑based refinement loop and a final mass‑transfer hill‑climb.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # 0. Choose a problem size that balances speed and quality.
    #    256 is a good compromise; you may increase it if time permits.
    # ------------------------------------------------------------------
    n = 256
    print(f"[init] using length n = {n}")

    # ------------------------------------------------------------------
    # 1. Generate a few random Dirichlet seeds and keep the best.
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = compute_score(best_a)
    print(f"[init] uniform start score = {best_val:.6f}")

    for _ in range(20):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a
            print(f"[seed] new best score = {val:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑based refinement loop.
    #    We perform several independent runs (different restarts) and,
    #    within each run, a handful of LP iterations.
    # ------------------------------------------------------------------
    def lp_refinement(start_vec, time_limit):
        """Run the LP refinement until time runs out; returns the best vector found."""
        a = start_vec.copy()
        best_local = a.copy()
        best_local_val = compute_score(a)
        while time.time() < time_limit:
            conv = full_convolution(a)
            M = conv.max()
            # Solve the LP to get a candidate g
            g = lp_maximize_sum(a, M)
            if g is None:
                print("[lp] solver failed – exiting refinement")
                break
            sum_g = g.sum()
            if sum_g <= 1.0 + 1e-9:      # no room for improvement
                break
            g_norm = g / sum_g           # bring g back to the simplex
            val_g = compute_score(g_norm)

            # Accept if the pure candidate is better
            if val_g < best_local_val:
                best_local_val = val_g
                best_local = g_norm.copy()
                a = g_norm
                print(f"[lp] pure improvement → {val_g:.6f}")
                continue

            # Otherwise try a convex combination (mixing)
            t_mix = 0.5
            a_mix = (1 - t_mix) * a + t_mix * g_norm
            a_mix = a_mix / a_mix.sum()
            val_mix = compute_score(a_mix)
            if val_mix < best_local_val:
                best_local_val = val_mix
                best_local = a_mix.copy()
                a = a_mix
                print(f"[lp] mixed improvement → {val_mix:.6f}")
                continue

            # If neither improves, stop this refinement run
            break
        return best_local, best_local_val

    # Run a few independent refinement restarts
    remaining_time = deadline - time.time()
    n_restarts = 5
    time_per_restart = remaining_time / max(n_restarts, 1)

    for idx in range(n_restarts):
        if time.time() >= deadline:
            break
        print(f"[lp] restart {idx + 1}/{n_restarts}, budget ≈ {time_per_restart:.1f}s")
        a_ref, val_ref = lp_refinement(best_a, time.time() + time_per_restart)
        if val_ref < best_val:
            best_val = val_ref
            best_a = a_ref
            print(f"[lp] restart {idx + 1} yielded new best = {val_ref:.6f}")

    # ------------------------------------------------------------------
    # 3. Final polish with a cheap mass‑transfer hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[mass] starting final mass‑transfer hill‑climb")
        a_ref = mass_transfer_hill_climb(best_a, deadline - 2.0)
        val = compute_score(a_ref)
        if val < best_val:
            best_val = val
            best_a = a_ref
            print(f"[mass] improved to final score = {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final evaluation using the official function.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)   # provided by the harness
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return final_seq
