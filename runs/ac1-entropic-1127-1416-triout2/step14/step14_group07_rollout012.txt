def smooth_max_gradient_descent(a0, deadline):
    a = a0.copy()
    # adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 1.0  # smoothness parameter
    while time.time() < deadline:
        t += 1
        # update beta gradually
        beta = min(500.0, 1.0 + (t * 0.001))  # adjust ramp
        # compute conv
        conv = full_convolution(a)
        # compute weights w = softmax(beta*conv)
        # but conv is length 2n-1, values small (<0.001). So safe.
        exp_beta_conv = np.exp(beta * conv - np.max(beta * conv))  # subtract max for stability
        w = exp_beta_conv / exp_beta_conv.sum()
        # compute gradient via correlation
        L = 1 << ((3*a.shape[0] - 2 - 1).bit_length())
        W_fft = np.fft.rfft(w, n=L)
        A_fft = np.fft.rfft(a, n=L)
        corr = np.fft.irfft(W_fft * np.conj(A_fft), n=L)
        g = 2.0 * corr[a.shape[0]-1 : a.shape[0]-1 + a.shape[0]]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.05 / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat
        # project to simplex
        a = project_to_simplex(a)
        # optionally check score every 1000 iterations
        if t % 2000 == 0:
            val = evaluate_sequence(a.tolist())
            # update best seq outside maybe need global.
    return a

"""
Optimiser for ``evaluate_sequence``.

The search proceeds in several stages:

1. **Fallback / simple analytic start** – use the globally supplied
   ``height_sequence_1`` (the best known sequence) or a reverse power‑law
   shape as a starting point.

2. **Fast parametric sweep** – a very small grid of simple families
   (reverse power law, multi‑power and sigmoid) is examined to possibly
   improve the starting vector.

3. **Smoothed‑max gradient descent** – minimise a smooth approximation of
   ``max(conv(a,a))`` using Adam.  The smoothing parameter (softmax
   temperature) is increased from a modest value to a large one, turning
   the problem from a smooth landscape to the original non‑smooth one.
   Multiple random restarts (jittered best or Dirichlet draws) are used.

4. **Projected sub‑gradient polishing** – a classic sub‑gradient loop
   with a more aggressive step‑size refines the best vector from stage 3.

5. **Stochastic hill‑climb** – a long greedy hill‑climb with adaptive
   step‑size finishes the search.

All helpers are top‑level functions; the entry point is ``propose_candidate``.
The algorithm respects the 1000 s budget and stays well within the limit.
"""

def renorm_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1 (a cheap simplex proj)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution ``a * a`` using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_proxy_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Small parametric families (used for a quick sweep)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p, p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p·(i+1)^q (both exponents ≥ 0)."""
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i‑c)/σ))."""
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Very fast sweep of a few families; returns the best vector found."""
    best_vec = None
    best_val = float('inf')

    # 1. reverse power law (coarse)
    for p in np.linspace(0.1, 6.0, 31):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, p)
        if a is None:
            continue
        val = compute_proxy_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 2. fine search around the best coarse p (if any)
    if best_vec is not None:
        # we could estimate p from the best candidate, but for simplicity
        # just redo a finer grid over the whole interval
        for p in np.linspace(0.05, 8.0, 81):
            if time.time() >= deadline:
                break
            a = reverse_power_law(n, p)
            val = compute_proxy_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 3. multi‑power grid
    for p in np.linspace(0.2, 3.0, 17):
        for q in np.linspace(0.0, 1.5, 16):
            if time.time() >= deadline:
                return best_vec
            a = multi_power(n, p, q)
            val = compute_proxy_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 4. sigmoid family
    for centre in np.linspace(0.2, 0.8, 13):
        for sigma in np.linspace(0.02, 0.12, 11):
            if time.time() >= deadline:
                return best_vec
            a = sigmoid_shape(n, centre, sigma)
            val = compute_proxy_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (Adam, soft‑max over the convolution)
# ----------------------------------------------------------------------
def gradient_descent_smooth_max(a0: np.ndarray, deadline: float,
                               beta_start: float = 5.0,
                               beta_end: float = 250.0,
                               lr: float = 0.02) -> np.ndarray:
    """
    Minimise a smooth approximation of ``max(conv(a,a))``.
    Uses Adam with a linearly increasing soft‑max temperature (beta).
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT size that can hold both the convolution (2n‑1) and the correlation
    L = 1 << ((3 * n - 2 - 1).bit_length())   # next power of two ≥ 3n‑2

    m = np.zeros_like(a)          # Adam first moment
    v = np.zeros_like(a)          # Adam second moment
    t = 0
    start = time.time()
    total = deadline - start if deadline > start else 0.0

    while time.time() < deadline:
        t += 1
        # linear schedule for beta (smoothness)
        elapsed = time.time() - start
        alpha = elapsed / total if total > 0 else 1.0
        beta = beta_start + alpha * (beta_end - beta_start)

        # FFT of current a (padded to length L)
        A_fft = np.fft.rfft(a, n=L)

        # --- convolution a*a (full length 2n-1) ---
        conv_full = np.fft.irfft(A_fft * A_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # --- soft‑max weights over the convolution ---
        scaled = beta * conv
        # numerical stabilisation
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()          # length 2n‑1, sum = 1

        # --- gradient of the smoothed max (correlation w ⋆ a) ---
        W_fft = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(W_fft * np.conj(A_fft), n=L)   # length L
        g = 2.0 * corr[n - 1 : n - 1 + n]                  # slice length n

        # --- Adam update ---
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        a = a - step
        a = renorm_simplex(a)        # keep a on the probability simplex

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (as in the previous version)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on ``max(conv(a,a))``."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = renorm_simplex(a)

        # occasional jitter
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = renorm_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (same as the previous implementation)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 2_500_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size and restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = renorm_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 180_000:
            # jump start from best with a tiny random perturbation
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = renorm_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    best_seq = None
    best_val = float('inf')
    n = 2048                                        # base length

    # -----------------------------------------------------------------
    # (0) Fallback from previously known good sequence
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # (1) Very quick parametric sweep (≈ 70 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 600.0   # reserve most of the budget for optimisation
    if time.time() < sweep_deadline:
        a_sweep = parametric_sweep(n, sweep_deadline)
        if a_sweep is not None:
            val = evaluate_sequence(a_sweep.tolist())
            print(f"[sweep] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_sweep.tolist()

    # -----------------------------------------------------------------
    # (2) Smoothed‑max gradient descent with several random restarts
    # -----------------------------------------------------------------
    grad_end = deadline - 250.0                     # keep time for polishing & hill‑climb
    if best_seq is not None and time.time() < grad_end:
        # convert best_seq to numpy array for the optimiser
        best_arr = np.array(best_seq, dtype=float)
        # try a handful of restarts
        while time.time() < grad_end:
            # initialise either from a jittered best or from a fresh Dirichlet draw
            if np.random.rand() < 0.65:
                a0 = best_arr + np.random.randn(n) * 0.001
                a0 = renorm_simplex(a0)
            else:
                a0 = np.random.dirichlet(np.ones(n))
            run_deadline = min(time.time() + 15.0, grad_end)   # ≤ 15 s per run
            a_opt = gradient_descent_smooth_max(a0, run_deadline)
            val = evaluate_sequence(a_opt.tolist())
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                best_arr = a_opt
                print(f"[gradient] improved score = {val:.6f}")

    # -----------------------------------------------------------------
    # (3) Sub‑gradient polishing (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.15)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # (4) Final stochastic hill‑climb (remaining time)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # (5) Safeguard – fallback to a simple reverse‑power shape
    # -----------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.0)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] using reverse‑power shape, score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")
    return best_seq
