def solve_lp_max_sum(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10.0})  # time limit for each LP maybe

    if not res.success:
        return None
    return res.x

def iterative_lp_improvement(initial, deadline, max_iter=200, t=0.2):
    f = initial.copy()
    n = len(f)
    f = np.maximum(f, 0.0)
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score(best_f)
    cur_score = best_score
    t_current = t
    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iter:
        conv_f = full_convolution(f)
        M = conv_f.max()
        # solve LP
        g = solve_lp_max_sum(f, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # rescale g to have same sum as f (which is 1)
        g = g / sum_g  # because f sum=1
        # convex combine
        new_f = (1 - t_current) * f + t_current * g
        # ensure sum=1 (project)
        new_f = np.maximum(new_f, 0.0)
        new_f = new_f / new_f.sum()
        new_score = compute_score(new_f)
        # if improvement accept
        if new_score < cur_score:
            f = new_f
            cur_score = new_score
            # optionally reduce t to slowly converge?
            # but keep
        else:
            # no improvement, reduce t
            t_current *= 0.5
            if t_current < 1e-4:
                break
        if new_score < best_score:
            best_score = new_score
            best_f = new_f.copy()
        iter_cnt += 1
        if iter_cnt % 10 == 0:
            print(f"[LP iter {iter_cnt}] score = {new_score:.6f}")
    return best_f

"""
search for a non‑negative step‑function minimizing
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases:
1. Cheap global baseline: sample a few analytic families (power‑law,
   exponential, beta, Gaussian, sine‑power) and random Dirichlet draws,
   keep the best few candidates.
2. Iterative LP improvement (the “LP‑step” described in the statement):
   for a fixed sequence f, compute M = max(conv(f,f)) and solve the linear
   program
        max   Σ g_j
        s.t.  conv(f,g) ≤ M , g ≥ 0
   (conv(f,g) is linear in g).  The optimal g is rescaled to the same
   ℓ₁‑norm as f and mixed with f by a small step‑size t.
   Repeating the LP‑step strictly reduces the objective.
3. Final polishing with a projected sub‑gradient descent that directly
   minimizes the exact max‑convolution objective.

All phases respect the 1000 s wall‑clock limit and keep the best vector
found so far.  The returned list is a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based full linear convolution a*a
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact evaluation of the optimisation target (sum is assumed = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families for the baseline stage
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Very cheap global baseline (many random draws + families)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 5,
    random_draws: int = 8000,
) -> list[np.ndarray]:
    """Explore a few analytic families and many Dirichlet draws."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law grid
    for p in np.linspace(0.1, 5.0, 41):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential grid
    for lam in np.linspace(0.0005, 0.5, 60):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta grid
    al_vals = np.linspace(0.02, 2.0, 21)
    be_vals = np.linspace(0.02, 2.0, 21)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # random beta draws
    for _ in range(150):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    # Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.45 * n, 25):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # sine‑power grid
    for p in np.linspace(0.1, 8.0, 25):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # mixtures
    for _ in range(120):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.2, 4.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # Dirichlet draws (various concentrations)
    concs = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Build convolution matrix A such that A @ g = conv(f,g) (full length)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return matrix A (2n‑1)×n with A @ g = conv(f,g)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j holds a shifted copy of f
    for j in range(n):
        A[j : j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Solve the LP: max Σ g_j  s.t. conv(f,g) ≤ M , g ≥ 0
# ----------------------------------------------------------------------


def solve_lp_for_g(f: np.ndarray, M: float, time_limit: float = 5.0) -> np.ndarray | None:
    """Return optimal g (as a 1‑D array) or None on failure."""
    n = f.shape[0]
    A = build_conv_matrix(f)                # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)       # maximise sum(g)  → minimise -Σ g
    bounds = [(0, None)] * n

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_limit},
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None
    return res.x


# ----------------------------------------------------------------------
# Iterative LP improvement (the “LP‑step”)
# ----------------------------------------------------------------------


def iterative_lp_improvement(
    init_vec: np.ndarray, deadline: float, max_iter: int = 500, t: float = 0.2
) -> np.ndarray:
    """Apply the LP‑step repeatedly, returning the best vector seen."""
    f = np.maximum(init_vec, 0.0)
    if f.sum() == 0:
        f = np.full_like(f, 1.0 / f.shape[0])
    else:
        f = f / f.sum()                     # keep ℓ₁‑norm = 1

    best_f = f.copy()
    best_score = compute_score(best_f)
    cur_score = best_score
    step = t
    iter_cnt = 0

    while time.time() < deadline and iter_cnt < max_iter:
        conv_f = full_convolution(f)
        M = conv_f.max()

        # solve LP for g
        g = solve_lp_for_g(f, M, time_limit=5.0)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # rescale g to have ℓ₁‑norm = 1 (same as f)
        g = g / sum_g

        # convex combination
        new_f = (1.0 - step) * f + step * g
        new_f = np.maximum(new_f, 0.0)
        if new_f.sum() == 0:
            new_f = np.full_like(new_f, 1.0 / new_f.shape[0])
        else:
            new_f = new_f / new_f.sum()

        new_score = compute_score(new_f)

        # accept if improvement, otherwise shrink step size
        if new_score < cur_score:
            f = new_f
            cur_score = new_score
        else:
            step *= 0.5
            if step < 1e-4:
                break

        if new_score < best_score:
            best_score = new_score
            best_f = new_f.copy()

        iter_cnt += 1
        if iter_cnt % 25 == 0:
            print(f"[LP‑iter {iter_cnt}] score = {new_score:.6f}")

    return best_f


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgradient_polish(
    a0: np.ndarray, deadline: float, step0: float = 0.33
) -> np.ndarray:
    """Projected sub‑gradient descent directly on max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            offset = k - idx
            mask = (offset >= 0) & (offset < n)
            g[mask] += 2.0 * a[offset[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a.sum()

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # -----------------------------------------------------------------
    # Determine problem size (use fallback if available)
    # -----------------------------------------------------------------
    try:
        fallback_vec = np.array([float(x) for x in height_sequence_1], dtype=float)
        n = fallback_vec.shape[0]
        print(f"[info] using fallback length n={n}")
    except Exception:
        n = 1024
        fallback_vec = None
        print(f"[info] no fallback, using default length n={n}")

    # -----------------------------------------------------------------
    # 0. Gather initial candidates
    # -----------------------------------------------------------------
    candidates = []
    if fallback_vec is not None:
        candidates.append(project_to_simplex(fallback_vec))

    # cheap global baseline (up to ~150 s)
    baseline_deadline = min(deadline, start_time + 150.0)
    if time.time() < baseline_deadline:
        baseline_cands = baseline_search(
            n, baseline_deadline, max_keep=5, random_draws=8000
        )
        candidates.extend(baseline_cands)

    # normalise all candidates
    candidates = [
        vec if np.isclose(vec.sum(), 1.0) else project_to_simplex(vec)
        for vec in candidates
    ]
    if not candidates:
        candidates = [np.full(n, 1.0 / n)]

    best_vec = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 1. LP improvement on each candidate (share remaining budget)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        per_cand_budget = remaining * 0.75 / len(candidates)
        for idx, cand in enumerate(candidates):
            cand_deadline = time.time() + per_cand_budget
            if cand_deadline > deadline:
                cand_deadline = deadline
            print(f"[candidate {idx}] LP improvement (budget ≈{per_cand_budget:.1f}s)")
            refined = iterative_lp_improvement(cand, cand_deadline, max_iter=500, t=0.2)
            sc = compute_score(refined)
            print(f"[candidate {idx}] LP result score = {sc:.6f}")
            if sc < best_score:
                best_score, best_vec = sc, refined.copy()
            if time.time() >= deadline - 20.0:
                break
    else:
        # if almost out of time, just take the best fallback
        best_vec = candidates[0].copy()
        best_score = compute_score(best_vec)

    # -----------------------------------------------------------------
    # 2. Final polishing with sub‑gradient descent
    # -----------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[final polishing] sub‑gradient descent")
        polish_deadline = deadline - 5.0
        polished = subgradient_polish(best_vec, polish_deadline, step0=0.30)
        sc = compute_score(polished)
        print(f"[final polishing] score = {sc:.6f}")
        if sc < best_score:
            best_score, best_vec = sc, polished.copy()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing else succeeded
    # -----------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # final normalisation (just in case)
    best_vec = np.maximum(best_vec, 0.0)
    if best_vec.sum() == 0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec = best_vec / best_vec.sum()

    # -----------------------------------------------------------------
    # Official evaluation (for reporting)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")

    return best_vec.tolist()
