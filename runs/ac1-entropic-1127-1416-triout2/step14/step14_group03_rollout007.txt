"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

Main ideas
---------
1. **Broad baseline** – explore several parametric families and random
   Dirichlet draws on a modest dimension (n≈1024).  Keep the best few.

2. **LP‑refinement** – given a candidate a, solve a linear program
   ``max sum b`` subject to ``a * b ≤ M`` where ``M = max(a*a)``.
   Mixing the solution back into a (with a decreasing step‑size)
   provably reduces the convolution peak.  This step is cheap because
   the LP size is O(n²) with n≈1024.

3. **Upscaling** – linearly interpolate the refined small‑n vector to a
   larger dimension (n≈4096).  The larger space gives the optimiser
   more freedom to flatten the convolution peak.

4. **Smoothed Adam optimisation** – minimise a smooth approximation of
   the maximum convolution (soft‑max with temperature τ) using Adam on
   the log‑softmax (simplex) parametrisation.

5. **Final simulated‑annealing polish** – a lightweight Metropolis walk
   to escape any remaining tiny plateaux.

The whole pipeline respects the 1000 s wall‑clock budget and returns
the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via FFT (real)
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b (FFT based)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a used in the objective."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a) = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact evaluation: 2·n·max(a*a) (sum assumed = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for the baseline stage)
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    arr = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Helpers for the baseline stage
# ----------------------------------------------------------------------
def _try_candidate(vec: np.ndarray, best: list, max_keep: int) -> None:
    """Normalise ``vec`` onto the simplex, evaluate and keep it if good."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0:
        return
    vec = vec / total
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Sweep several families and Dirichlet draws, keep the best ``max_keep``."""
    best = []   # list of (score, vector)

    # 1) power‑law
    for p in np.linspace(0.1, 6.0, 300):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # 2) exponential
    for lam in np.linspace(0.0001, 0.2, 300):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # 3) beta
    al_vals = np.linspace(0.2, 8.0, 30)
    be_vals = np.linspace(0.2, 8.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # 4) Gaussian
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # 5) Dirichlet draws (both random and monotone)
    concs = [0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)
            a_desc = np.sort(a)[::-1]
            _try_candidate(a_desc, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation + simplex projection)
# ----------------------------------------------------------------------
def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0:
        return np.full(new_n, 1.0 / new_n)
    return up / total


# ----------------------------------------------------------------------
# LP refinement (single iteration = solve max sum(b) subject to a*b ≤ M)
# ----------------------------------------------------------------------
def lp_refine(a0: np.ndarray, deadline: float, max_iter: int = 12) -> np.ndarray:
    """
    Perform a few LP‑based refinement steps.
    Each step solves
        max   sum(b)
        s.t.  (a * b)_k ≤ M   ∀k
              b ≥ 0
    and mixes the solution back into a.
    """
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception as e:
        # SciPy not available – skip refinement
        print("[lp_refine] SciPy not available, skipping LP stage.")
        return a0.copy()

    a = a0.copy()
    n = a.shape[0]

    best_score = compute_score(a)
    print(f"[lp_refine] start score = {best_score:.6f}")

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # current bound M
        conv = full_convolution(a)
        M = conv.max()

        # Build sparse convolution matrix A_ub where (A_ub @ b)_k = (a*b)_k
        # Entry A_ub[k, j] = a_{k-j} if 0 ≤ k-j < n else 0.
        # This is a collection of n constant diagonals.
        diags_data = [np.full(n, a[d]) for d in range(n)]
        offsets = np.arange(n)               # 0 … n‑1
        A_ub = sp.diags(diags_data, offsets,
                         shape=(2 * n - 1, n), format="csc")

        # Linear program: minimise -sum(b)  (i.e. maximise sum(b))
        c = -np.ones(n)
        bounds = [(0, None)] * n
        rhs = np.full(2 * n - 1, M)

        # Give the solver a tiny time budget per iteration to stay within the global deadline
        time_left = max(0.0, deadline - time.time())
        lp_options = {"presolve": True, "time_limit": time_left}
        res = linprog(c, A_ub=A_ub, b_ub=rhs, bounds=bounds,
                      method="highs", options=lp_options)

        if not res.success:
            print(f"[lp_refine] LP iteration {it} failed ({res.message}); exiting.")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            print("[lp_refine] LP returned zero vector; stopping.")
            break

        # Normalise b to the simplex (same total mass as a)
        g = b / sum_b

        # Mixing parameter – start aggressive and decay
        t = 0.5 / (it + 1)

        a = (1.0 - t) * a + t * g
        # Project back onto simplex (numerical safety)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            print(f"[lp_refine] iter {it:2d}  score {cur_score:.6f}")

    print(f"[lp_refine] final best score = {best_score:.6f}")
    return a


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------
def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 10.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9995) -> np.ndarray:
    """
    Adam optimiser on a soft‑max parametrisation, minimising a smooth
    approximation of max(conv).  Returns the best vector seen before
    *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initialise temperature from the current max
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = full_convolution(a)
        max_c = conv.max()

        # smooth weights for the max
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for log‑softmax variables
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # occasional exact evaluation
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

    # -----------------------------------------------------------------
    # 2. LP‑based refinement (still on the small problem)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 350.0
    if time.time() < lp_deadline:
        refined_small = lp_refine(a0_small, lp_deadline, max_iter=12)
        val = compute_score(refined_small)
        if val < best_val:
            best_val, best_seq = val, refined_small.tolist()
            print(f"[lp] improved score = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Upsample to a larger dimension for fine optimisation
    # -----------------------------------------------------------------
    n_big = 4096
    if best_seq is None:
        a0_big = np.full(n_big, 1.0 / n_big)
    else:
        a0_big = upsample(np.array(best_seq, dtype=float), n_big)

    # -----------------------------------------------------------------
    # 4. Smoothed Adam optimisation on the large problem
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if time.time() < polish_deadline:
        refined_big = smooth_optimize(a0_big, polish_deadline)
        val = compute_score(refined_big)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined_big.tolist()

        # Additional restarts while we have time
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)

            # random Dirichlet start (different concentrations)
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0])
            a_start = np.random.dirichlet(np.full(n_big, conc))
            if np.random.rand() < 0.4:
                a_start = np.sort(a_start)[::-1]

            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
