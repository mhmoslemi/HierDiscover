def gradient_for_max(a, k):
    n = a.shape[0]
    grad = np.zeros_like(a)
    if k < n:
        grad[:k+1] = 2 * a[k::-1]
    else:
        r = k - (n - 1)
        # slice start: n-1 down to r
        grad[r:] = 2 * a[n-1:r-1:-1]   # note that slice a[n-1:r-1:-1] includes elements n-1, n-2, ..., r
    return grad
```

def subgradient_optimize(a0, deadline, step0=0.5):
   a = np.copy(a0)
   best_a = a.copy()
   best_val = compute_score(a)
   t=1
   while time.time() < deadline:
        conv = full_convolution(a)
        max_idx = np.argmax(conv)
        grad = gradient_for_max(a, max_idx)  # O(n)
        # step size
        eta = step0 / np.sqrt(t)
        a_new = a - eta * grad
        # projection onto simplex
        a = proj_simplex(a_new)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgrad] iter {t} best {best_val:.6f}")
        t+=1
   return best_a
```

def proj_simplex(v):
    # Projects v onto simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]

    rho = ind[cond][-1]

"""
search for a non‑negative step‑function (a probability vector) that
minimises  evaluate_sequence(seq) = 2·n·max(conv(seq,seq))/(sum(seq)^2).

Algorithm
---------
1.  Broad exploration (Dirichlet draws, beta family, stretched exponentials,
    mixtures of exponentials, a simple triangular shape).  The best few
    candidates are kept.

2.  Local refinement with a *sub‑gradient* method on the exact max‑convolution.
    The gradient of  max(conv)  w.r.t. the vector is cheap to compute
    (just a shifted copy of the vector).  After each gradient step the
    vector is projected onto the probability simplex.

3.  A final short simulated‑annealing polish (like in the reference code).

All phases respect the overall wall‑clock budget (≈ 1000 s).  The function
returns a plain Python list of floats; the harness will call
``evaluate_sequence`` on it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale does not matter)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized to sum = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple optimisation helpers
# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_for_max(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of max(conv(a,a)) w.r.t. a when the maximal index is k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        # use a[0..k] reversed
        g[: k + 1] = 2.0 * a[k :: -1]      # a[k], a[k-1], ..., a[0]
    else:
        r = k - (n - 1)                     # first index that participates
        # slice a[n-1] … a[r]
        g[r:] = 2.0 * a[n - 1 : r - 1 : -1]
    return g


# ----------------------------------------------------------------------
# Families of candidate vectors (all return a sum‑normalised vector)
# ----------------------------------------------------------------------
def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_stretched_exponential(n: int, lam: float, p: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i**p)."""
    a = np.exp(-lam * (np.arange(n, dtype=float) ** p))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_mixture_exponential(
    n: int, lam1: float, lam2: float, w: float
) -> np.ndarray:
    """Mixture of two exponentials: w·exp(-lam1·i)+(1‑w)·exp(-lam2·i)."""
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape a_i ∝ (i+1) for i≤mid, else (n‑i)."""
    mid = (n - 1) // 2
    left = np.arange(1, mid + 2, dtype=float)
    right = np.arange(n - mid - 1, 0, -1, dtype=float)
    a = np.concatenate([left, right])
    a = a[:n]  # in case of even n we drop the duplicate centre
    return a / a.sum()


# ----------------------------------------------------------------------
# Candidate bookkeeping
# ----------------------------------------------------------------------
def _try_candidate(vec: np.ndarray, best):
    """Update the top‑k list ``best`` (sorted by increasing score)."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return
    vec = vec / s
    sc = compute_score(vec)
    # keep at most 12 best candidates
    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray, deadline: float,
                         step0: float = 0.5,
                         max_iters: int = 300000) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = full_convolution(a)
        k = int(conv.argmax())
        grad = gradient_for_max(a, k)
        eta = step0 / math.sqrt(it)
        a = proj_simplex(a - eta * grad)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[subgrad] iteration {it:6d}  score {val:.6f}")
    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same as reference)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Simple Metropolis hill‑climb with temperature decay."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the time limit.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the external harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem dimension – 1024 gives a good speed / expressivity trade‑off.
    # --------------------------------------------------------------
    n = 1024

    # Keep the 12 best candidates throughout the broad phase
    best_candidates = []   # list of (score, vector)

    # --------------------------------------------------------------
    # 0.  Optional fallback from a previously saved solution
    # --------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = np.array([float(x) for x in fallback])
            _try_candidate(fb, best_candidates)
            print(f"[fallback] score = {compute_score(fb):.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Broad random and parametric exploration (≈ 300 s)
    # --------------------------------------------------------------
    exploration_deadline = deadline - 600.0   # reserve ~10 min for refinements
    if time.time() < exploration_deadline:
        # 1a) Dirichlet draws with several concentration levels
        alphas = [0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
        draws_per_alpha = 2500
        for a_val in alphas:
            for _ in range(draws_per_alpha):
                if time.time() >= exploration_deadline:
                    break
                vec = np.random.dirichlet(np.full(n, a_val))
                _try_candidate(vec, best_candidates)

        # 1b) Beta family – extended range (including U‑shaped possibilities)
        al_vals = np.concatenate((
            np.logspace(-2, -0.5, 6),            # 0.01 … 0.3
            np.linspace(0.5, 8.0, 12),           # 0.5 … 8
            np.logspace(0.9, 1.4, 6)            # 8 … 25
        ))
        be_vals = al_vals.copy()
        for alpha in al_vals:
            for beta in be_vals:
                if time.time() >= exploration_deadline:
                    break
                vec = generate_beta(n, alpha, beta)
                _try_candidate(vec, best_candidates)

        # 1c) Stretched exponentials (two free parameters)
        lam_vals = np.logspace(-5, -1, 20)
        p_vals = np.linspace(0.5, 2.0, 25)
        for lam in lam_vals:
            for p in p_vals:
                if time.time() >= exploration_deadline:
                    break
                vec = generate_stretched_exponential(n, lam, p)
                _try_candidate(vec, best_candidates)

        # 1d) Mixture of two exponentials
        lam1_vals = np.logspace(-5, -2.5, 8)
        lam2_vals = np.logspace(-2.5, -0.5, 8)
        w_vals = np.linspace(0.15, 0.85, 5)
        for lam1 in lam1_vals:
            for lam2 in lam2_vals:
                for w in w_vals:
                    if time.time() >= exploration_deadline:
                        break
                    vec = generate_mixture_exponential(n, lam1, lam2, w)
                    _try_candidate(vec, best_candidates)

        # 1e) Simple triangular shape
        _try_candidate(generate_triangular(n), best_candidates)

        print(f"[exploration] best after broad search = {best_candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2.  Sub‑gradient local optimisation (≈ 200 s)
    # --------------------------------------------------------------
    subgrad_deadline = deadline - 250.0
    if time.time() < subgrad_deadline and best_candidates:
        # run the optimiser on the top 5 candidates
        for _, vec in best_candidates[:5]:
            if time.time() >= subgrad_deadline:
                break
            refined = subgradient_optimize(vec, subgrad_deadline,
                                           step0=0.6, max_iters=500000)
            _try_candidate(refined, best_candidates)
            print(f"[subgrad‑polish] refined score = {compute_score(refined):.6f}")

    # --------------------------------------------------------------
    # 3.  Final simulated‑annealing polish (≈ 150 s)
    # --------------------------------------------------------------
    final_deadline = deadline - 5.0
    if time.time() < final_deadline and best_candidates:
        best_vec = best_candidates[0][1]
        final = simulated_annealing(best_vec, final_deadline)
        _try_candidate(final, best_candidates)
        print(f"[final‑anneal] final polished score = {best_candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # 4.  Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if not best_candidates:
        best_seq = np.full(n, 1.0 / n)
    else:
        best_seq = best_candidates[0][1]

    # --------------------------------------------------------------
    # 5.  Official evaluation (the harness will call ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
    return best_seq.tolist()
