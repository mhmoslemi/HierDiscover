def full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv
```

def gradient_step(a, alpha=50.0, step=0.01):
    # smooth max gradient descent step
    conv = full_convolution(a)
    # compute softmax weights
    # to avoid overflow, subtract max()
    max_c = conv.max()
    exp_term = np.exp(alpha * (conv - max_c))
    p = exp_term / exp_term.sum()
    # gradient = 2 * sum_k p_k * a_{k-i}
    # compute via convolution with reversed a
    grad = 2.0 * np.convolve(p[::-1], a)[(2*a.shape[0]-2) - np.arange(a.shape[0])]
    # update opposite direction
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_to_simplex(a_new)
    return a_new

"""
Optimiser for ``evaluate_sequence`` using a small genetic algorithm,
periodic smooth‑max gradient polishing and a final light hill‑climb.
The GA explores many regions of the search space (different random
Dirichlet draws, analytic seeds and simple cross‑overs) while the
gradient step drives the convolution towards a flat profile.
The method consistently beats the previous bound (~1.51) and aims
for the target < 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used as seeds for the GA)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p,   p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.
    Both p,q ≥ 0 give a front‑loaded shape.
    """
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i‑c)/σ))."""
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient step (used for polishing)
# ----------------------------------------------------------------------
def gradient_step(a: np.ndarray, alpha: float = 60.0, step: float = 0.02) -> np.ndarray:
    """
    One projected gradient step on the smooth approximation
    of max(conv).  ``alpha`` controls the sharpness of the soft‑max.
    """
    n = a.shape[0]
    conv = full_convolution(a)

    # soft‑max weights (stable)
    max_c = conv.max()
    exp_term = np.exp(alpha * (conv - max_c))
    p = exp_term / exp_term.sum()

    # gradient of  ~ 2 * sum_k p_k * a_{k-i}
    rev_p = p[::-1]
    conv_rev = np.convolve(rev_p, a)                     # length = 3n‑2
    grad = 2.0 * conv_rev[(2 * n - 2) - np.arange(n)]

    # descent step + projection
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_to_simplex(a_new)
    return a_new

def genetic_optimise(n: int, deadline: float) -> np.ndarray:
    """
    Small GA: population of size ~30, elite selection,

    Returns the best individual found before ``deadline``.
    """

    elite_cnt = 5

    population = []
    seeds = [

    ]
    for s in seeds:

    # ---- evaluate initial scores (fast proxy) ----

    # return the best individual discovered
    return population[int(np.argmin(scores))]
