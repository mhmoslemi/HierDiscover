    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_conv = conv.max()
        # softmax weights
        # use grad approximation around max
        # perform stability shift
        shift = max_conv
        exp_vals = np.exp(gamma * (conv - shift))
        w = exp_vals / exp_vals.sum()
        # gradient via convolution weighting
        # compute convolution of w with reversed a
        a_rev = a[::-1]
        # convolution using FFT
        # using same full_convolution but with arrays w and a_rev
        # w length = 2n-1, a_rev length n
        # pad sizes to power of two
        m_len = w.shape[0] + a_rev.shape[0] - 1

        v = beta2 * v + (1 - beta2) * (grad * grad)

  minimises a smooth approximation of max(a∗a) (soft‑max of the convolution).
* Phase 3 – a short projected sub‑gradient polishing of the overall best.
* Return the best sequence as a plain list of floats.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to sum exactly 1
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, fast for moderate n)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def full_convolution_generic(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two arbitrary real vectors using FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    return np.fft.irfft(Z, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of 2·n·max(a∗a) for a vector that already satisfies
    sum(a)=1 and a_i≥0.  This matches evaluate_sequence when the sum is 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Parametric families
# ----------------------------------------------------------------------


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape on {0,…,n‑1}: a_i ∝ (i+1)^{α−1}·(n−i)^{β−1}."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape:
        a_i ∝ 1 / sqrt((i+0.5)·(n-0.5-i)).
    """
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


def generate_uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n)

    a0: np.ndarray,
    deadline: float,
    lr: float = 0.20,
    gamma: float = 200.0,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Projected Adam on the smooth objective
        φ(a) = softmax_{γ}(a ∗ a) = (1/γ)·log Σ_k exp(γ·(a∗a)_k).

    The gradient of the soft‑max can be written in closed form as
        ∇φ = 2· (w * a_rev)[n‑1 : n‑1+n],
    where w_k = exp(γ·c_k)/Σ_j exp(γ·c_j) and a_rev = a[::-1].

    The method runs until ``deadline`` and returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    cur_gamma = gamma

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_conv = conv.max()

        # soft‑max weights (numerically stable)
        shift = max_conv
        w = np.exp(cur_gamma * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient: 2·(w * a_rev) restricted to the n central entries
        a_rev = a[::-1]
        conv_w = full_convolution_generic(w, a_rev)
        grad = 2.0 * conv_w[n - 1 : n - 1 + n]

        # Adam update on the simplex (project after the step)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # schedule – slowly reduce the soft‑max temperature
        if t % 500 == 0:
            cur_gamma = max(5.0, cur_gamma * 0.95)

    return best_a


# ----------------------------------------------------------------------
# Random candidate generation (mixture of a few parametric families)
# ----------------------------------------------------------------------


def random_candidate(n: int) -> np.ndarray:
    """
    Produce a random mixture of arcsine, uniform and a Beta(α,β) shape.
    Mixing weights are drawn from a Dirichlet(1,1,1) distribution.
    α and β are drawn uniformly from [0.1, 2.0].
    """
    w = np.random.dirichlet([1.0, 1.0, 1.0])
    arcs = generate_arcsine(n)
    uniform = generate_uniform(n)
    alpha = np.random.uniform(0.10, 2.00)
    beta = np.random.uniform(0.10, 2.00)
    beta_vec = generate_beta(n, alpha, beta)

    cand = w[0] * arcs + w[1] * uniform + w[2] * beta_vec
    # tiny numerical drift may break the simplex condition – renormalise.
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    return cand

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` under a 1000 s wall‑clock budget.

    Strategy:
      1. Load the historic candidate (if present) as a fallback.
      2. Initialise with the classic arcsine distribution.
      3. Randomly explore mixtures of arcsine / uniform / Beta shapes.
      4. Refine the most promising candidates with a projected Adam optimiser.
      5. Finish with a short projected sub‑gradient polish.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # problem size – 2048 offers a good trade‑off between quality and speed
    n = 2048

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional historic fallback
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback = np.array(fallback, dtype=float)
        fallback = np.maximum(fallback, 0.0)
        fallback /= fallback.sum()
        val = evaluate_sequence(fallback.tolist())
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback.copy()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic start: arcsine distribution
    # ------------------------------------------------------------------
    arcs = generate_arcsine(n)
    arcs_score = compute_score(arcs)
    print(f"[arcsine] initial score = {arcs_score:.6f}")
    if arcs_score < best_score:
        best_score, best_seq = arcs_score, arcs.copy()

    # ------------------------------------------------------------------
    # 2. Random exploration of mixtures (≈20 % of the time budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + (deadline - start_time) * 0.20
    # keep the top K mixtures seen so far
    K = 5
    import heapq

    top_heap = []  # min‑heap of (-score, vector) so the smallest score is on top
    while time.time() < explore_deadline:
        cand = random_candidate(n)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
        if len(top_heap) < K:
            heapq.heappush(top_heap, (-sc, cand))
        else:
            # keep only the K best (largest negative = smallest score)
            if -sc > top_heap[0][0]:
                heapq.heapreplace(top_heap, (-sc, cand))
    print(f"[explore] best score after random mixtures = {best_score:.6f}")

    # Extract the promising candidates (sorted by score ascending)
    promising = [vec for _, vec in sorted(top_heap, key=lambda x: -x[0])]

    # ------------------------------------------------------------------
    # 3. Adam‑softmax polishing of the promising candidates
    #    (~70 % of the remaining budget, split evenly)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and promising:
        per_candidate = remaining / len(promising)
        print(f"[adam] polishing {len(promising)} candidates, ≈{per_candidate:.1f}s each")
        for idx, cand in enumerate(promising, 1):
            sub_deadline = time.time() + per_candidate
            polished = adam_softmax_polish(cand, sub_deadline)
            sc = compute_score(polished)
            print(f"    [adam #{idx}] refined score = {sc:.6f}")
            if sc < best_score:
                best_score, best_seq = sc, polished.copy()
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # 4. Final cheap projected sub‑gradient polish (≈10 % of budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polish_deadline = deadline - 2.0  # a tiny reserve for final eval
        print("[subgrad] final polishing start")
        # reuse the subgradient routine from the previous system (lightweight)
        a = best_seq.copy()
        n = a.shape[0]
        best_a = a.copy()
        best_val = compute_score(a)

        t = 0
        while time.time() < polish_deadline:
            t += 1
            conv = full_convolution(a)
            max_val = conv.max()
            eps = max_val * 1e-9
            active = np.where(conv >= max_val - eps)[0]

            # sub‑gradient of the max‑convolution (summed over active indices)
            g = np.zeros_like(a)
            idx = np.arange(n)
            for k in active:
                j = k - idx
                mask = (j >= 0) & (j < n)
                g[mask] += 2.0 * a[j[mask]]

            step = 0.30 / math.sqrt(t)
            a -= step * g
            a = project_to_simplex(a)

            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        # update global best if improved
        if best_val < best_score:
            best_score, best_seq = best_val, best_a.copy()
        print(f"[subgrad] polished best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Safeguard – fall back to uniform if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = generate_uniform(n)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")

    return best_seq.tolist()
