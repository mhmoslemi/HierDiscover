"""
propose_candidate – enhanced stochastic optimizer for the step‑function bound.

The objective is   score = 2·n·max(conv(a,a)) / (sum(a)^2) .
Because the score is scale‑invariant we keep the vector on the probability simplex
(sum = 1, a_i ≥ 0).  The optimizer consists of three phases:

1. **Broad parametric sweep** – quick search over a few analytically‑motivated
   families (reverse power‑law, mixed reverse‑law/uniform).  The best vector
   found becomes the seed for the later phases.

2. **Projected sub‑gradient polishing** – following the reference method,
   iteratively moves the current point opposite to a sub‑gradient of the
   ‑max‑convolution functional while staying on the simplex.  This reduces the
   dominant active constraints quickly.

3. **Simulated‑annealing hill‑climb** – a long stochastic search that accepts
   occasional uphill moves with a temperature schedule.  Moves consist of
   transferring a random fraction of mass between two coordinates; the
   temperature and step‑size are adapted as the run proceeds.  Periodic jitter
   and restart‑on‑stagnation help to escape local minima.

All phases respect the overall time budget (≈ 1000 s) and use only NumPy.
The final candidate is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Parametric families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p} (front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    Convex combination   mix·reverse_power_law(p) + (1‑mix)·uniform,
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    return project_to_simplex(a)

# ----------------------------------------------------------------------
# 1. Broad parametric sweep (time‑limited)
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Search a few low‑dimensional families; returns the best vector."""
    best_vec = None
    best_val = float('inf')

    # ---- reverse power‑law (coarse then fine) ----
    # coarse grid
    for p in np.arange(0.5, 8.0 + 1e-12, 0.2):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, float(p))
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # fine grid around the best coarse p (if we have one)
    if best_vec is not None:
        # locate approximate best p by a brief refinement sweep
        p_center = None
        for p in np.arange(0.5, 8.0 + 1e-12, 0.05):
            if time.time() >= deadline:
                break
            a = reverse_power_law(n, float(p))
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec, p_center = val, a, p
        if p_center is not None:
            lo = max(0.5, p_center - 0.3)
            hi = min(8.0, p_center + 0.3)
            for p in np.arange(lo, hi + 1e-12, 0.005):
                if time.time() >= deadline:
                    return best_vec
                a = reverse_power_law(n, float(p))
                val = compute_score(a)
                if val < best_val:
                    best_val, best_vec = val, a

    # ---- mixed reverse‑law + uniform ----
    for mix in np.linspace(0.3, 0.95, 14):
        for p in np.linspace(0.5, 4.0, 45):
            if time.time() >= deadline:
                break
            a = mixed_reverse_uniform(n, mix, p)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- random convex combos of a few basics (quick) ----
    # basic shapes
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rp2 = reverse_power_law(n, 2.0)
    rp4 = reverse_power_law(n, 4.0)
    basics = [uniform, rp2, rp4]

    for _ in range(1500):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec

# 2. Projected sub‑gradient polishing (reference‑style)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑convolution functional."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient (sum of shifted copies of a over active offsets)
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # periodic jitter to shake out from plateaus
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# ----------------------------------------------------------------------
# 3. Simulated‑annealing stochastic hill‑climb
# ----------------------------------------------------------------------
def annealing_optimize(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """Stochastic optimisation with temperature schedule and restarts."""
    a = a_start.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    # temperature – start with a small fraction of the current score
    init_temp = cur_score * 0.012
    start_time = time.time()
    last_improve_iter = 0
    it = 0

    while time.time() < deadline:
        it += 1
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = max(init_temp * (1.0 - frac), 1e-12)

        # propose a simple mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] <= 1e-14:
            continue
        # transfer a random fraction of the source mass
        delta = np.random.rand() * a[i] * (0.6 if np.random.rand() < 0.7 else 0.1)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        new_score = compute_score(cand)
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score < 0 or np.random.rand() < math.exp(-delta_score / T):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
                last_improve_iter = it

        # restart if stagnated for a long time
        if it - last_improve_iter > 250_000:
            jitter = np.random.randn(n) * 5e-4
            a = project_to_simplex(best_a + jitter)
            cur_score = compute_score(a)
            last_improve_iter = it

        # occasional global jitter to keep exploration alive
        if it % 80_000 == 0:
            jitter = np.random.randn(n) * 2e-4
            a = project_to_simplex(a + jitter)
            cur_score = compute_score(a)

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so we do not exceed the budget
    deadline = start_time + budget_s - 4.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback using a previously known construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # noqa: F821
        val = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose vector length (larger gives finer discretisation)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 250 else 2048
    print(f"[setup] using n = {n}")

    # -----------------------------------------------------------------
    # 2. Fast parametric sweep (≈ 60 s)
    # -----------------------------------------------------------------
    sweep_deadline = min(deadline - 200.0, time.time() + 60.0)
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            raw_score = compute_score(a_initial)
            print(f"[parametric sweep] best raw score = {raw_score:.6f}")
            official = evaluate_sequence(a_initial.tolist())
            print(f"[parametric sweep] official score = {official:.6f}")
            if official < best_val:
                best_val, best_seq = official, a_initial.tolist()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (if we have time)
    # -----------------------------------------------------------------
    polish_deadline = min(deadline - 120.0, time.time() + 120.0)
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.15)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated‑annealing hill‑climb (bulk of the remaining budget)
    # -----------------------------------------------------------------
    anneal_deadline = deadline
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = annealing_optimize(a_start, anneal_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[annealing] final official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 5. Safety net – simple mixed shape if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        a_fallback = mixed_reverse_uniform(n, mix=0.85, p=2.2)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback‑mixed] score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
