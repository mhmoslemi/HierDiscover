"""
Step‑function optimiser (hard‑max Adam).

We search for a non‑negative vector `a` (∑a = 1) that minimises
    τ(a) = 2·n·max(conv(a))               (conv = a * a, linear convolution)
which is exactly the value returned by ``evaluate_sequence`` after normalisation.

The optimiser consists of three phases:

1. **Seeding** – a few Dirichlet draws together with a possible fallback
   sequence are normalised and the best one is kept.

2. **Hard‑max Adam** – a variant of Adam that uses the *exact* sub‑gradient
   of the max‑convolution (instead of the smooth‑max surrogate used before).
   At each iteration we:
   * compute the full convolution of the current vector,
   * identify all indices attaining the maximum,
   * build a weight vector `w` that is uniform on those active indices,
   * obtain the sub‑gradient `g = 2·(a ⋆ w)` via a single FFT pair,
   * perform an Adam update and project back onto the probability simplex.
   The learning‑rate decays slowly; occasional jitter helps to escape
   plateaus.  The best vector seen during the run is retained.

3. **Polish** – a cheap mass‑transfer hill‑climb (small fractions of mass
   moved from one coordinate to another) finishes the optimisation.

The whole routine is time‑aware: every loop checks the global deadline and
always keeps the best candidate found so far.  The final list of floats is
returned; ``evaluate_sequence`` will be called by the harness for the actual
score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex ∑v = 1, v ≥ 0."""
    if np.all(v <= 0):
        # trivial case – return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # numerical safeguard
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT (output length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1, a ≥ 0):
        2·n·max(conv(a)).
    This is exactly what ``evaluate_sequence`` would return after normalisation.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Hard‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_hardmax(a0: np.ndarray, deadline: float,
                 lr_start: float = 0.08,
                 lr_decay: float = 0.995,
                 beta1: float = 0.90,
                 beta2: float = 0.999,
                 eps: float = 1e-12,
                 jitter_interval: int = 3000,
                 jitter_scale: float = 0.002) -> np.ndarray:
    """
    Adam optimisation using the exact sub‑gradient of the max‑convolution.
    Returns the best vector (on the simplex) seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    # initial best
    best_a = a.copy()
    conv = full_convolution(a)
    best_val = 2.0 * n * conv.max()

    t = 0
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # ---- compute convolution and active set ----
        # use a single FFT size large enough for both conv and gradient
        rev_len = m
        size = 1 << ((n + rev_len - 2).bit_length())   # ≥ n+rev_len-1
        A = np.fft.rfft(a, n=size)

        # full convolution a*a
        conv_full = np.fft.irfft(A * A, n=size)[:m]
        max_val = conv_full.max()
        # active indices (allow a tiny tolerance to keep the set stable)
        atol = 1e-14
        max_inds = np.where(conv_full >= max_val - atol)[0]

        # weight vector w – uniform on the active set
        w = np.zeros(m, dtype=np.float64)
        w[max_inds] = 1.0 / max_inds.size

        # gradient of the (average) max: g = 2·(a ⋆ w)
        rev_w = w[::-1]
        B = np.fft.rfft(rev_w, n=size)
        corr_full = np.fft.irfft(A * B, n=size)[: n + rev_len - 1]
        idx = (m - 1) - np.arange(n)          # pick the centred part
        grad = 2.0 * corr_full[idx]

        # ---- Adam update ----
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # occasional jitter to escape flat plateaus
        if jitter_interval and (t % jitter_interval == 0):
            jitter = np.random.randn(n) * (lr_start * jitter_scale / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

        # learning‑rate decay
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-6)

        # ---- bookkeeping ----
        # compute current score directly from max_val of the *new* a
        # (need a fresh convolution)
        conv = full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam_hardmax] iter={t}, lr={lr:.3e}, "
                  f"score={cur_val:.6f}, active={max_inds.size}")

    return best_a

def permute_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Swap two entries at a time; keep the best permutation."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # perform swap
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional log
            if it % 5000 == 0:
                print(f"[permute] iter={it}, score={val:.6f}")
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]
        it += 1
    return best_a


def mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Move a small fraction of mass from one index to another."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # transfer a random fraction (up to 25 % of the source mass)
        frac = np.random.rand() * 0.25
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:
                print(f"[mass] iter={it}, score={val:.6f}")
        it += 1
    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

        a0 = np.array(fallback, dtype=float)
        a0 = np.maximum(a0, 0.0)

        n = a0.shape[0]

        print(f"[init] no fallback – start with uniform vector of length {n}")

    best_val = compute_score(best_a)
    print(f"[init] initial score = {best_val:.6f}")

    # --------------------------------------------------------------
    seed_vectors = [best_a]
    for _ in range(12):
        if time.time() >= deadline:
            break

        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        seed_vectors.append(a)

    # --------------------------------------------------------------

    starts.append(np.full(n, 1.0 / n))            # ensure uniform is present
    # allocate time for each start (reserve a bit for the polish phases)
    remaining = deadline - time.time()
    reserve = 120.0                               # seconds for hill‑climbs + finalisation

        if time.time() >= deadline - reserve:

        a_ref = adam_hardmax(start_vec, local_deadline)
        cur_val = compute_score(a_ref)
        print(f"[adam] finished start {idx+1}, score = {cur_val:.6f}")
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 3. Permutation hill‑climb (re‑order the entries).
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        perm_deadline = deadline - 10.0
        print("[permute] starting permutation hill‑climb")
        a_ref = permute_hill_climb(best_a, perm_deadline)
        cur_val = compute_score(a_ref)
        print(f"[permute] finished, score = {cur_val:.6f}")
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 4. Small mass‑transfer hill‑climb for final polishing.
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mass_deadline = deadline - 2.0
        print("[mass] starting mass‑transfer hill‑climb")
        a_ref = mass_transfer_hill_climb(best_a, mass_deadline)
        cur_val = compute_score(a_ref)
        print(f"[mass] finished, score = {cur_val:.6f}")
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 5. Final report using the official evaluation function.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)   # type: ignore
        print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    except Exception as exc:
        # In case evaluate_sequence is not available in this environment,
        # fall back to our own exact computation.
        final_score = compute_score(best_a)
        print(f"\nSearch finished (local score) = {final_score:.6f}")

    return final_seq
