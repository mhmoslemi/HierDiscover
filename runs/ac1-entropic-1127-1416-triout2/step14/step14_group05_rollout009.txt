"""
Subgradient‐based minimiser for the step‑function attack.

The optimiser works on a probability vector a (∑a=1, a_i≥0) and
directly minimises the exact objective

    f(a) = 2·n·max_k (a * a)[k]

using a projected sub‑gradient descent on the max‑convolution.
Multiple random Dirichlet starts, a simple symmetrisation trick and
two cheap hill‑climbs (permutation and mass‑transfer) are employed.
The code is time‑aware: every phase stops before the global deadline
and the current best solution is always kept.

The final sequence is returned as a plain Python list of floats and
will be evaluated by the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (fast for n up to a few 10⁴)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def gradient_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (conv, g) where conv = a*a and g is a sub‑gradient
    of  max(conv)  w.r.t. a.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_val = conv.max()
    max_idxs = np.flatnonzero(conv == max_val)

    # sub‑gradient: average over all maximising shifts
    g = np.zeros_like(a)
    centre = n - 1
    for idx in max_idxs:
        d = idx - centre   # shift that attains the maximum
        if d >= 0:
            # a_{i+d} term
            g[: n - d] += a[d:]
            # a_{i-d} term
            g[d:] += a[: n - d]
        else:
            d = -d
            g[d:] += a[: n - d]
            g[: n - d] += a[d:]
    g *= 2.0 / len(max_idxs)          # factor 2 from the derivative of a*a
    return conv, g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        deadline: float,
                        init_lr: float = 0.8,
                        sym_step: float = 0.05,
                        sym_every: int = 30) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact objective.
    ``init_lr`` is the initial step size; it decays like 1/√t.
    Every ``sym_every`` iterations the vector is blended a little
    with its reverse (symmetrisation) – this empirically lowers the
    max‑convolution.
    Returns (best_a, best_score).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    n = a.shape[0]
    while time.time() < deadline:
        conv, g = gradient_of_max_conv(a)

        # diminishing step size
        step = init_lr / np.sqrt(t + 1.0)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional symmetrisation with the reversed vector
        if (t % sym_every) == 0 and t > 0:
            a = (1.0 - sym_step) * a + sym_step * a[::-1]
            a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # light progress report
            if t % 500 == 0:
                print(f"[subgrad] iter={t}, score={val:.6f}")

        t += 1
        # safety guard – don’t waste time on uselessly many iterations
        if t > 50000:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbs on the exact objective (same as the baseline)
# ----------------------------------------------------------------------


def permute_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Swap two entries at a time; keep the best permutation."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # perform swap
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[permute] iter={it}, score={val:.6f}")
        else:
            a[i], a[j] = a[j], a[i]      # revert
        it += 1
    return best_a


def mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Move a small fraction of mass from one index to another."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        frac = np.random.rand() * 0.30
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:
                print(f"[mass] iter={it}, score={val:.6f}")
        it += 1
    return best_a

        print(f"[init] using fallback of length {n}")
    else:
        a0 = np.full(n, 1.0 / n)
        print("[init] no fallback – start with uniform vector")

    best_a = a0.copy()
    best_val = compute_score(best_a)
    print(f"[init] initial score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Quick cheap Dirichlet seeds – keep the best.
    # --------------------------------------------------------------
    for _ in range(12):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[dirichlet] improved seed score = {val:.6f}")

    # --------------------------------------------------------------
    # 2. Projected sub‑gradient descent from several starts.
    # --------------------------------------------------------------
    starts = [best_a, np.full(n, 1.0 / n)]
    for _ in range(5):
        starts.append(np.random.dirichlet(np.ones(n)))   # random starts

    remaining = deadline - time.time()
    per_start = remaining / max(len(starts), 1)

    for idx, start_vec in enumerate(starts):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_start - 1.0
        if local_deadline <= time.time():
            continue
        print(f"[subgrad] start {idx+1}/{len(starts)} – time budget ≈ {per_start:.1f}s")
        a_ref, val = subgradient_descent(start_vec, local_deadline)
        print(f"[subgrad] finished start {idx+1}, score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 3. Permutation hill‑climb (re‑order the entries).
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        perm_deadline = deadline - 5.0
        print("[permute] starting permutation hill‑climb")
        a_ref = permute_hill_climb(best_a, perm_deadline)
        val = compute_score(a_ref)
        print(f"[permute] finished, score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 4. Small mass‑transfer hill‑climb for a final polish.
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mass_deadline = deadline - 2.0
        print("[mass] starting mass‑transfer hill‑climb")
        a_ref = mass_transfer_hill_climb(best_a, mass_deadline)
        val = compute_score(a_ref)
        print(f"[mass] finished, score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_ref.copy()

    # --------------------------------------------------------------
    # 5. Final report using the official evaluation function.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print("\nSearch finished. Final reported score =", final_score)
    return final_seq
