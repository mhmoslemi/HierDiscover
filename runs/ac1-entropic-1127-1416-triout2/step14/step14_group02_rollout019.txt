"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages:

1. **Diverse baseline generation** – quick analytic families and many
   Dirichlet draws are tried; the best vector (normalized to sum‑1)
   is kept.

2. **Linear‑programming improvement** – given the current vector `f`,
   we solve the LP

        maximise   Σ b_j
        subject to b ≥ 0,
                   (f * b)_k ≤ M   for every k,
   where `M = max(f*f)`.  The solution pushes mass to positions that
   do not increase the current worst convolution value.  A convex
   combination `f ← (1‑t) f + t·(b / Σb)` is performed and the loop is
   repeated while time permits.

3. **Projected sub‑gradient polish** – ordinary sub‑gradient descent
   on the exact max‑convolution objective (the same routine used in the
   reference implementation) to fine‑tune the result.

4. **Simulated‑annealing hill‑climb** – a cheap Metropolis walk that
   can accept worsening moves, giving a final chance to escape tiny
   plateaus.

All stages respect the wall‑clock limit (`budget_s`).  The best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    C = A * A
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring (exact objective, assumes a is a numpy array)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


# ----------------------------------------------------------------------
# Analytic families (used for the initial random search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a /= a.sum()
    return a


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        a = np.zeros_like(a)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming step (single update of the “f → g” iteration)
# ----------------------------------------------------------------------


def lp_update(f: np.ndarray, M: float, active_shifts: np.ndarray) -> np.ndarray | None:
    """
    Solve the LP:
        maximise Σ b_j
        s.t.    b_j ≥ 0,
                (f * b)_k ≤ M   for all k in active_shifts
    Returns the optimal vector b (unnormalised) or None on failure.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Build sparse matrix A where A[row, col] = f[i] with i = k - col
    for row_idx, k in enumerate(active_shifts):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # iterate over i (index in f)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])

    A = csr_matrix((data, (rows, cols)), shape=(len(active_shifts), n), dtype=np.float64)
    b = np.full(A.shape[0], M, dtype=np.float64)

    # maximise Σ b_j  <=> minimise -Σ b_j
    c = -np.ones(n, dtype=np.float64)

    # bounds 0 ≤ b_j ≤ 1 (the upper bound merely keeps the LP bounded)
    bounds = [(0.0, 1.0) for _ in range(n)]

    # use the high‑performance HiGHS solver
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (exact max‑convolution)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from every active shift
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        # projection onto the probability simplex (non‑negative, sum=1)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a.sum()

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = np.maximum(a, 0.0)
            if a.sum() == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a.sum()
    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0:
            continue
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = np.maximum(a, 0.0)
            if a.sum() == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a.sum()
            cur_val = compute_score(a)

    return best_a

    conc_levels = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = 1500 // len(conc_levels)
    for conc in conc_levels:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            try_candidate(np.random.dirichlet([conc] * n))

    # -----------------------------------------------------------------
    # 2. Linear‑programming refinement (the core new idea)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        f = np.array(best_seq, dtype=float)
        iteration = 0
        while time.time() < deadline - 5.0:
            iteration += 1
            conv = full_convolution(f)
            M = conv.max()
            # active shifts: values within a tiny tolerance of the current max
            eps = M * 1e-6 + 1e-12
            active = np.where(conv >= M - eps)[0]
            if active.size == 0:
                break
            b_vec = lp_update(f, M, active)
            if b_vec is None:
                break
            if b_vec.sum() == 0:
                break
            g = b_vec / b_vec.sum()            # normalise to sum‑1
            t = 0.3                              # step size for convex combination
            f = (1.0 - t) * f + t * g
            f = np.maximum(f, 0.0)
            if f.sum() == 0:
                f = np.full_like(f, 1.0 / n)
            else:
                f /= f.sum()
            cur_val = compute_score(f)
            if cur_val < best_val:
                best_val, best_seq = cur_val, f.copy()
                print(f"[LP iter {iteration}] improved score = {cur_val:.6f}")
            # early stop if improvement becomes negligible
            if iteration >= 30:
                # after a few iterations we hand control to the polish phase
                break

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polish (exact max‑objective)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, deadline, step0=0.33)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (final fine‑tuning)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, deadline,
                                      init_temp=2e-3, final_temp=1e-7,
                                      cooling_factor=0.9995,
                                      step_scale_init=0.05)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report via the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
