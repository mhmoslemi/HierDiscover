def project_simplex(v):
    # returns w where w >=0, sum=1, minimizing ||w - v||_2.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    # v = np.array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

(the denominator is irrelevant after normalising the vector).

The algorithm consists of:

1.   A quick random Dirichlet + monotone baseline (≈ 30 s).
2.   Fine grid sweeps of the Beta family and of a mixed
     exponential‑power‑law family (≈ 120 s).
3.   Local optimisation of a *soft‑max* parametrisation using
     L‑BFGS‑B on a smoothed max‑convolution objective.
     The temperature τ is decreased progressively (≈ 700 s).
4.   A very short simulated‑annealing polish (≈ 10 s).

All stages respect the global 1000 s wall‑clock budget.
The final sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, O(N·log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale does not matter)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised (sum=1) vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (all vectors are normalised to sum 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_mixed(n: int, lam: float, p: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)·(i+1)^{‑p}."""
    a = np.exp(-lam * np.arange(n, dtype=float)) * np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random exponential vector, sort it decreasingly, and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax (maps ℝⁿ → simplex)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_objective(x: np.ndarray, n: int, tau: float) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad) where

        f   = 2·n·smooth_max(conv(a,a))
        grad = ∂f/∂x

    a = softmax(x)  (so a ≥ 0 and Σ a = 1).
    The smooth max is implemented via log‑sum‑exp with temperature τ.
    """
    a = softmax(x)
    conv = full_convolution(a)                # length 2·n‑1
    max_c = conv.max()

    # soft‑max weights w_k = exp((conv_k‑max_c)/τ) / Σ exp(...)
    shifted = conv - max_c
    w = np.exp(shifted / tau)
    w_sum = w.sum()
    w /= w_sum                                 # now Σ w = 1

    # smooth maximum value: max_c + τ·log Σ exp(...)
    smooth_val = max_c + tau * np.log(w_sum)
    f = 2.0 * n * smooth_val

    # gradient of the weighted sum Σ w_k·conv_k w.r.t a:
    a_rev = a[::-1]
    grad_full = _linear_convolution(a_rev, w)   # length = 3·n‑2
    start = n - 1
    grad_a = 2.0 * grad_full[start:start + n]   # ∂/∂a (Σ w_k·conv_k)

    # chain rule through the soft‑max parametrisation
    dot = np.dot(grad_a, a)
    grad_x = (grad_a - dot) * a * (2.0 * n)     # ∂f/∂x
    return f, grad_x


def smooth_obj_wrapper(x: np.ndarray, n: int, tau: float) -> tuple[float, np.ndarray]:
    """Thin wrapper required by scipy.optimize.minimize (jac=True)."""
    return smooth_objective(x, n, tau)


# ----------------------------------------------------------------------
# Local optimiser based on L‑BFGS‑B with a decreasing temperature schedule
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective.
    The temperature τ is decreased progressively.
    """
    try:
        from scipy.optimize import minimize
    except Exception as exc:
        raise RuntimeError("scipy.optimize.minimize is required for lbfgs_optimize") from exc

    n = a_start.shape[0]
    eps = 1e-30
    x = np.log(a_start + eps)          # initialise logits

    # schedule of temperatures (from smooth → almost exact)
    tau_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]

    best_a = a_start.copy()
    best_val = compute_score(best_a)

    for tau in tau_schedule:
        if time.time() >= deadline:
            break
        # a single L‑BFGS run for the current τ
        res = minimize(
            fun=smooth_obj_wrapper,
            x0=x,
            args=(n, tau),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxfun": 2000,
                "maxiter": 2000,
                "ftol": 1e-12,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        x = res.x
        a = softmax(x)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[lbfgs] τ={tau:.1e}  improved score = {val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Simple random baseline (quick diversification)
# ----------------------------------------------------------------------


def baseline_random_search(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Very fast random exploration – returns the best vector found."""
    best_a = None
    best_score = float("inf")
    alphas = [0.1, 0.5, 1.0, 2.0, 5.0]        # Dirichlet concentrations
    while time.time() < deadline:
        for conc in alphas:
            a = np.random.dirichlet(np.full(n, conc))
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[rand] new best {sc:.6f}")
            if time.time() >= deadline:
                break
        # occasional monotone draw
        if time.time() < deadline:
            a = generate_monotone_random(n)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[rand‑mono] new best {sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Fine grid sweeps for the Beta family and a mixed exponential‑powerlaw family
# ----------------------------------------------------------------------


def beta_fine_sweep(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Exhaustive search over a 2‑D grid of (α,β)."""
    best_a = None
    best_score = float("inf")
    al_vals = np.geomspace(0.2, 15.0, 30)   # geometric spacing gives good coverage
    be_vals = np.geomspace(0.2, 15.0, 30)

    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                return best_a, best_score
            a = generate_beta(n, alpha, beta)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[beta] α={alpha:.2f} β={beta:.2f}  score={sc:.6f}")
    return best_a, best_score


def mixed_sweep(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search over a 2‑D grid of (λ, p) for a·exp(‑λ·i)·(i+1)^{‑p}."""
    best_a = None
    best_score = float("inf")
    lam_vals = np.linspace(3e-4, 3e-3, 35)   # more concentrated around the known sweet‑spot
    p_vals = np.linspace(0.3, 4.0, 35)

    for lam in lam_vals:
        for p in p_vals:
            if time.time() >= deadline:
                return best_a, best_score
            a = generate_mixed(n, lam, p)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[mix] λ={lam:.6f} p={p:.2f}  score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Very short simulated‑annealing polish (optional – helps escape tiny plateaus)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """Simple Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = 1e-3
    final_T = 1e-6
    cooling = 0.9997
    step_scale = 0.02

    while time.time() < deadline:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = compute_score(cand)
        dval = val - cur_val
        if dval < 0 or np.random.rand() < math.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, 1e-6)
        T = max(final_T, T * cooling)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # problem size – 2048 gives a good trade‑off between expressivity and speed
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] official score = {val:.6f}")
            sc = compute_score(np.array(fb) / np.sum(fb))
            if sc < best_val:
                best_val, best_seq = sc, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Quick random baseline (≈ 30 s)
    # -----------------------------------------------------------------
    t0 = time.time()
    baseline_deadline = deadline - 850.0          # leave ~850 s for the refined phases
    if t0 < baseline_deadline:
        cand, cand_score = baseline_random_search(n, baseline_deadline)
        if cand is not None and cand_score < best_val:
            best_val, best_seq = cand_score, cand.tolist()
            print(f"[baseline] best score = {cand_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Fine parametric sweeps (≈ 150 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 650.0
    if time.time() < sweep_deadline:
        # beta family
        cand, cand_score = beta_fine_sweep(n, sweep_deadline)
        if cand is not None and cand_score < best_val:
            best_val, best_seq = cand_score, cand.tolist()
            print(f"[beta‑sweep] improved score = {cand_score:.6f}")

        # mixed exponential‑powerlaw family
        if time.time() < sweep_deadline:
            cand, cand_score = mixed_sweep(n, sweep_deadline)
            if cand is not None and cand_score < best_val:
                best_val, best_seq = cand_score, cand.tolist()
                print(f"[mixed‑sweep] improved score = {cand_score:.6f}")

    # -----------------------------------------------------------------
    # 3. L‑BFGS‑B on a smoothed max–convolution objective (≈ 700 s)
    # -----------------------------------------------------------------
    opt_deadline = deadline - 50.0
    if best_seq is not None and time.time() < opt_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = lbfgs_optimize(a0, opt_deadline)
        sc = compute_score(refined)
        print(f"[lbfgs] refined score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()
        # a few quick random restarts while time permits
        while time.time() < opt_deadline:
            remaining = opt_deadline - time.time()
            sub_dead = time.time() + min(20.0, remaining)
            a_start = np.random.dirichlet(np.full(n, 1.0))
            refined = lbfgs_optimize(a_start, sub_dead)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()
                print(f"[lbfgs‑restart] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 10 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(final)
        print(f"[anneal] final polished score = {sc:.6f}")
        if sc < best_val:
            best_val, best_seq = sc, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 6. Official evaluation (the harness will call ``evaluate_sequence``)
    # -----------------------------------------------------------------
    official = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {official:.6f}")
    return best_seq
