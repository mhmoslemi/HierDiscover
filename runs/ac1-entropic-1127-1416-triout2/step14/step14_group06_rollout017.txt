"""
Improved optimiser for the ``evaluate_sequence`` objective.

Algorithm outline
-----------------
1. Choose a problem size `n` (2048, 4096 or 8192) depending on the
   remaining time budget – larger `n` gives a finer discretisation
   of the underlying continuous problem.

2. Generate a diverse pool of seed vectors:
   * uniform, arcsine (β(0.5,0.5)), reverse‑power‑law, geometric,
     random beta‑shapes and random mixtures of reverse‑power‑law
     and the uniform distribution.
   The best seed (according to the *exact* score 2·n·max(conv))
   becomes the starting point.

3. **Smoothed gradient descent**.
   The non‑smooth max‑convolution is replaced by a smooth
   log‑sum‑exp surrogate:
       Mτ(a) = τ·log( Σk exp(convk/τ) )
   whose gradient is a weighted cross‑correlation of `a` with the
   soft‑max weights.  The optimisation uses:
   * an exponentially decaying temperature `τ`,
   * a step size ∝ 1/√t,
   * Euclidean projection onto the probability simplex after each
     update,
   * occasional jitter to escape shallow plateaus.
   The best iterate (according to the exact max‑convolution) is
   retained.

4. **Stochastic hill‑climb** on the best vector from step 3.
   Small amounts of mass are moved between two random indices,
   the move is accepted only if the true max‑convolution improves.
   Adaptive step‑size and restarts are employed.

5. Return the best vector found (as a plain Python list).  All
   numbers are non‑negative and sum to one, therefore the
   evaluation is scale‑invariant and respects the security
   checks in ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using an FFT (real‑valued)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The evaluation function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law),  p ≥ 0.
    """
    idx = np.arange(n, 0, -1, dtype=np.float64)
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def arcsine_shape(n: int) -> np.ndarray:
    """β(0.5,0.5) – the classic arcsine density on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a /= a.sum()
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    Convex combination   mix * reverse_power_law(p) + (1‑mix) * uniform,
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

    # Random beta shapes (including edge‑heavy arcsine‑like)
    for _ in range(30):
        alpha = np.random.uniform(-0.9, 5.0)
        beta = np.random.uniform(-0.9, 5.0)
        a = beta_shape(n, alpha, beta)
        if a is not None:
            cand.append(a)

    # Random mixtures of reverse‑power‑law and uniform
    for _ in range(20):
        mix = np.random.rand()
        p = np.random.uniform(0.5, 4.0)
        a = mixed_reverse_uniform(n, mix, p)
        if a is not None:
            cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient descent
# ----------------------------------------------------------------------
def smoothed_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector observed (according to the exact max‑convolution).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    n = a.shape[0]
    # FFT length needed for the correlation a * w (size 3n‑2)
    N_corr = 1 << ((3 * n - 2 - 1).bit_length())

    t = 0
    # Temperature schedule – start relatively high, decay towards 1e‑8
    tau = None
    step0 = 0.05

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)                # length 2n‑1
        max_c = conv.max()

        # initialise / decay temperature
        if tau is None:
            tau = max(max_c * 0.2, 1e-6)
        else:
            tau = max(tau * 0.995, 1e-8)

        # soft‑max weights (log‑sum‑exp trick for stability)
        shift = conv - max_c                      # ≤ 0
        exp_shift = np.exp(shift / tau)
        w = exp_shift / exp_shift.sum()           # length 2n‑1, sums to 1

        # ------------------- gradient of the smoothed max -------------------
        # correlation r[i] = Σ_j a[j] * w[i + j]  (zero‑padded)
        A_fft = np.fft.rfft(a, N_corr)
        W_fft = np.fft.rfft(w, N_corr)
        r_full = np.fft.irfft(A_fft * np.conj(W_fft), n=N_corr)[: 3 * n - 2]
        g = 2.0 * r_full[2 * n - 2 - np.arange(n)]   # gradient of τ·log‑sum‑exp

        # --------------- projected gradient step ---------------------------
        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape flat regions
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        # keep best according to the *exact* objective
        if t % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth‑grad] iter {t}, new best raw score {val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (as in the previous version)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 200_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size and restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M, best_a = M_cand, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0            # safety margin

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0. Optional fallback (if a previous construction is present)
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Choose vector length
    # -------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n = 8192
    elif remaining > 300:
        n = 4096
    else:
        n = 2048
    print(f"[setup] using n = {n}")

    # -------------------------------------------------------------
    # 2. Generate a pool of seed vectors and pick the best one
    # -------------------------------------------------------------
    candidates = generate_initial_candidates(n)
    for a in candidates:
        if time.time() >= deadline:
            break
        val_raw = compute_score(a)
        if val_raw < best_val:
            best_val = val_raw
            best_seq = a.tolist()
            print(f"[seed] new best raw score = {val_raw:.6f}")

    # if none succeeded (should not happen) fall back to arcsine
    if best_seq is None:
        a0 = arcsine_shape(n)
        best_seq = a0.tolist()
        best_val = compute_score(a0)

    # -------------------------------------------------------------
    # 3. Smoothed gradient descent (bulk of the remaining time)
    # -------------------------------------------------------------
    gd_deadline = deadline - 120.0          # leave ~2 min for hill‑climb & final eval
    if time.time() < gd_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_polished = smoothed_gradient_descent(a_start, gd_deadline)
        val = compute_score(a_polished)
        print(f"[smooth‑grad] polished raw score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # -------------------------------------------------------------
    hill_deadline = deadline - 4.0
    if time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final raw score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # 5. Safeguard – fall back to a simple mixed shape if nothing else
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = mixed_reverse_uniform(n, mix=0.8, p=1.7)
        best_seq = a_fallback.tolist()
        best_val = compute_score(a_fallback)

    # -------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
