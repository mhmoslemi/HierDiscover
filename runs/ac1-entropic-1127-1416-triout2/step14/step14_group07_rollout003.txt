   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1=0.9; beta2=0.999; epsilon=1e-8
   for t in range(1, max_iter+1):
      if time.time() >= deadline:
          break
      # schedule beta for smooth max: exponential increase
      beta = beta_start * (beta_end/beta_start) ** ((t-1)/(max_iter-1))

      conv_rev = np.convolve(a, w_rev)  # using FFT for speed maybe

      grad = 2.0 * conv_rev[idx]
      # adam update

      a = project_to_simplex(a)
   return a

def convolve_fft(x,y):
    n = len(x) + len(y) -1
    n_fft = 1 << ((n-1).bit_length())
    X = np.fft.rfft(x, n_fft)

    c = np.fft.irfft(X*Y, n_fft)[:n]

"""
search_step_function_improved.py

An alternative optimiser for ``evaluate_sequence``.  The main ideas are:

*   Use a smooth approximation of the ``max`` in the objective (log‑sum‑exp)
    and run a projected Adam optimiser.  The smoothing parameter (β) is
    increased during optimisation so the algorithm first follows an easy
    landscape and then zooms in on the true ‑∞ norm.

*   Initialise the optimiser from a diverse set of analytic families
    (reverse‑power‑law, multi‑power, sigmoid, beta‑distribution) as well as
    a few random Dirichlet draws, to avoid getting stuck in the same local
    minimum that the previous hill‑climb tended to converge to.

*   After the Adam phase a short stochastic hill‑climb (the same routine as
    before) is applied to polish the result.

The programme respects the 1000 s budget, prints progress and finally returns
the best sequence found as a plain list of ``float``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of *x* and *y* using a real FFT."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())   # next power of two ≥ n
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    c = np.fft.irfft(X * Y, n=n_fft)[:n]
    return c


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution)."""
    return convolve_fft(a, a)


# ----------------------------------------------------------------------
# Analytic families (all return a normalised vector Σa = 1)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p,  p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, …, 1
    a = idx ** p
    s = a.sum()
    return a / s if s > 0 else None


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    s = a.sum()
    return a / s if s > 0 else None


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^i, 0<q<1."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else None


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    return a / s if s > 0 else None


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^α (1‑x_i)^β,  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    return a / s if s > 0 else None


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p · (i+1)^q."""
    left = np.arange(n, 0, -1, dtype=np.float64) ** p
    right = np.arange(1, n + 1, dtype=np.float64) ** q
    a = left * right
    s = a.sum()
    return a / s if s > 0 else None


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """
    a_i ∝ 1/(1+exp((i‑c)/σ)),
    centre_frac ∈ [0,1] selects the centre, sigma_frac controls steepness.
    """
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    s = a.sum()
    return a / s if s > 0 else None


# ----------------------------------------------------------------------
# Smoothed max and its gradient (used by Adam)
# ----------------------------------------------------------------------
def smoothed_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad) where
        smooth_max = (1/β)·log Σ exp(β·conv_i)
    and grad = ∂smooth_max/∂a (projected onto the simplex later).
    """
    n = a.shape[0]
    conv = full_convolution(a)                     # length 2n‑1
    # stable softmax
    z = beta * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    sum_exp = exp_z.sum()
    w = exp_z / sum_exp                            # softmax weights, Σ w = 1
    smooth = (z_max + np.log(sum_exp)) / beta      # value of smooth max

    # gradient: g_k = 2·Σ_l a_l·w_{k+l}
    w_rev = w[::-1]                                # length 2n‑1
    conv_rev = convolve_fft(a, w_rev)              # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)               # indices that give the needed sums
    grad = 2.0 * conv_rev[idx]
    return smooth, grad


# ----------------------------------------------------------------------
# Adam optimiser with projection onto the simplex
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  max_iter: int = 4000,
                  beta_start: float = 5.0,
                  beta_end: float = 200.0) -> np.ndarray:
    """
    Projected Adam on the smoothed‐max loss.
    ``beta`` is increased exponentially from *beta_start* to *beta_end*
    so that early iterations see a very smooth landscape.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-9

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # schedule β (smoothness) – exponential ramp
        beta = beta_start * (beta_end / beta_start) ** ((t - 1) / (max_iter - 1))

        smooth, grad = smoothed_max_and_grad(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size – slowly decreasing
        lr = 0.12 / (math.sqrt(t) + 1.0)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex (sum = 1, non‑negative)
        a = project_to_simplex(a)

    return a

# Stochastic hill‑climb (same idea as the previous version)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """Greedy hill‑climb that moves small amounts of mass between two indices."""
    a = a_start.copy()
    best_a = a.copy()
    n = a.shape[0]
    best_M = full_convolution(a).max()
    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]                # steal a fraction from i
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            # occasional restart with tiny jitter
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
