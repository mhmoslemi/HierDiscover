"""
Propose a candidate step‑function (non‑negative, sum unrestricted) that
minimises the black‑box ``evaluate_sequence``.  The algorithm works in three
phases:

1. **Broad random search** on a modest size (2048).  Many parametric families
   (power‑law, exponential, beta, cosine) and many Dirichlet draws (including
   monotone and symmetric versions) are sampled.  The best few vectors are
   kept.

2. **Upscaling + smooth optimisation**.  The current best vector is
   linearly interpolated to a larger dimension (8192) to give the optimiser
   more freedom.  A soft‑max smoothing of the max‑convolution is minimised
   with Adam on the log‑softmax parametrisation (the vector lives on the
   simplex automatically).  Several restarts are performed while time permits.

3. **Fine polishing**.  A simple sub‑gradient descent steps directly on the
   true max‑convolution, followed by a short simulated‑annealing hill‑climb,
   are applied to escape any remaining plateaus.

The whole pipeline respects the 1000 s wall‑clock budget and returns the
best sequence found as a plain Python list of floats.
"""

import time

# ----------------------------------------------------------------------
# Fast linear convolution (real FFT)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a * a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a) = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact evaluation 2·n·max(a*a) for a simplex‑normalised a."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    arr = np.arange(1, n + 1, dtype=float) ** (-p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    arr = np.exp(-lam * np.arange(n, dtype=float))
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    arr = left * right
    if not np.all(np.isfinite(arr)):
        return np.full(n, 0.0)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def generate_cosine_pow(n: int, p: float) -> np.ndarray:
    """U‑shaped cosine: ((cos πi/(n‑1) + 1)/2)^{p}."""
    i = np.arange(n, dtype=float)
    base = (np.cos(np.pi * i / (n - 1)) + 1.0) / 2.0
    arr = np.power(base, p)
    s = arr.sum()
    return arr / s if s > 0 else np.full(n, 0.0)


def symmetric_from_half(half: np.ndarray, n_total: int) -> np.ndarray:
    """Mirror a half‑vector to a full symmetric vector of length n_total (even)."""
    # half must have length n_total//2
    if n_total % 2 == 0:
        return np.concatenate([half[::-1], half])
    # odd length – insert a centre element equal to the first of half
    centre = np.array([half[0]])
    return np.concatenate([half[::-1], centre, half])

def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``vec`` to length ``new_n`` and re‑normalise."""
    n = vec.shape[0]
    if new_n == n:
        return vec.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return up / total

def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.25,
                    init_tau_factor: float = 6.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9998) -> np.ndarray:
    """
    Minimise max(conv) using a soft‑max smooth approximation.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits (log‑softmax)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature based on the initial max convolution
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping (ensures simplex) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution and smoothing weights -----
        conv = full_convolution(a)
        max_c = conv.max()

        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient wrt a (g = 2 * a_rev * w) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for logits -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature decay -----
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact evaluation -----
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

def subgradient_descent(a0: np.ndarray, deadline: float,
                        init_step: float = 0.02) -> np.ndarray:
    """
    Simple sub‑gradient method directly on the non‑smooth objective.
    Uses decreasing step size  η_t = init_step / sqrt(t+1).
    Returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        conv = full_convolution(a)
        max_idx = np.argmax(conv)

        # gradient of the active convolution entry
        grad = np.zeros_like(a)
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)
        idxs = np.arange(low, high + 1, dtype=int)
        # derivative is 2 * a_{max_idx - i}
        grad[idxs] = 2.0 * a[max_idx - idxs]

        # step size schedule
        eta = init_step / np.sqrt(t + 1.0)
        a = a - eta * grad

        # project back onto simplex (non‑negative, sum = 1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        t += 1

    return best_a

# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total == 0.0:
            continue
        cand /= total

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Baseline random exploration on modest dimension
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 500.0   # reserve bulk of time for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_random_search(n_small, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Upsample to a larger dimension for finer optimisation
    # -----------------------------------------------------------------
    if best_seq is None:
        a0_small = np.full(n_small, 1.0 / n_small)
    else:
        a0_small = np.array(best_seq, dtype=float)

    a0_big = upsample(a0_small, n_big)

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation on the large problem
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0   # leave ~2 min for final polishing
    if time.time() < polish_deadline:
        # primary run from the up‑sampled best
        refined = smooth_optimize(a0_big, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional random restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)

            # random start – either Dirichlet or parametric
            if np.random.rand() < 0.5:
                conc = np.random.choice([0.01, 0.05, 0.1, 0.2, 0.5, 1.0])
                a_start = np.random.dirichlet(np.full(n_big, conc))
                # occasional monotone forcing
                if np.random.rand() < 0.3:
                    a_start = np.sort(a_start)[::-1]
            else:
                # random beta shape (U‑shaped)
                alpha = 10 ** np.random.uniform(-2.0, 0.3)
                beta = 10 ** np.random.uniform(-2.0, 0.3)
                a_start = generate_beta(n_big, alpha, beta)
                if np.random.rand() < 0.3:
                    a_start = np.sort(a_start)[::-1]

            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Fine sub‑gradient descent (direct max‑conv minimisation)
    # -----------------------------------------------------------------
    grad_deadline = deadline - 60.0
    if best_seq is not None and time.time() < grad_deadline:
        a_start = np.array(best_seq, dtype=float)
        refined = subgradient_descent(a_start, grad_deadline, init_step=0.02)
        val = compute_score(refined)
        print(f"[subgrad] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_big] * n_big
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
