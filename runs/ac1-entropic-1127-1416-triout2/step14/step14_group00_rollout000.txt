"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  

The algorithm works in three phases:

1. **Large‑scale initialization** – build several candidate shapes on a
   relatively large grid (n = 8192).  Candidates include the discrete
   arcsine distribution, smoothed arcsine variants, a few Beta‑shapes,
   and convex mixtures of arcsine with the uniform distribution.
2. **Projected sub‑gradient polishing** – starting from the best
   candidate, run a long projected sub‑gradient descent on the true
   objective 2·n·max(conv(a,a)).  The step size is slowly decreased,
   the active set of near‑maximal convolution entries is enlarged
   (tolerance 1e‑4) to obtain a smooth descent direction, and a small
   jitter is injected periodically to escape plateaus.
3. **Brief simulated‑annealing fine‑tuning** – a short Metropolis
   walk explores the neighbourhood of the polished solution.

All operations respect the simplex constraint Σa_i = 1, a_i ≥ 0 and all
computations are performed with fast FFT‑based convolutions.  The
function returns the best sequence (as a plain list of floats) before
the 1000 s wall‑clock deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / projection utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a power‑of‑two real FFT.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective value for a vector that already lies on the simplex:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) distribution.
    a_i ∝ 1 / √((i+½)*(n‑½‑i)), i = 0 … n‑1, then normalised.
    """
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


def generate_arcsine_delta(n: int, delta: float) -> np.ndarray:
    """
    Smoothed arcsine: shift the singularities by `delta` (0 ≤ delta ≤ 0.5).
    a_i ∝ 1 / √((i+½+δ)*(n‑½‑δ‑i)).
    """
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt((i + delta) * (n - 0.5 - delta - i))
    a = 1.0 / denom
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑type shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1},  α,β > 0.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


def mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """
    Convex combination lam*Uniform + (1‑lam)*base, projected back to simplex.
    """
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n)
    mixed = lam * uniform + (1.0 - lam) * base
    # projection is cheap because the mix stays non‑negative; only normalise:
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient polishing (projected descent on the exact objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.7,
                       tol_factor: float = 1e-4) -> np.ndarray:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    A tolerance factor enlarges the active set (all entries within
    `tol_factor * max_val` of the current maximum) to obtain a smoother
    descent direction.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active indices: those whose value is within tol_factor of the max
        eps = max_val * tol_factor
        active = np.where(conv >= max_val - eps)[0]

        # gradient = sum_{k in active} 2 * shift(a, k)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # step size schedule – simple inverse‑square‑root decay
        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to shake off plateaus
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing fine‑tuning
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9993,
                        step_scale_init: float = 0.06,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """
    Metropolis hill‑climb that can occasionally accept worsening moves.
    Operates on the exact objective (no smoothing).  The routine is
    deliberately lightweight – it stops early if the deadline is
    approaching.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices, move a small amount from i to j
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        # accept if improvement or with Metropolis probability
        if dval < 0.0 or np.random.rand() < math.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.04, 0.45)   # slowly enlarge step
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.96, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to keep exploration alive
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    """

    start = time.time()
    # Reserve a few seconds for the final evaluation and safety margins
    deadline = start + budget_s - 5.0

    # Use a relatively large grid – 8192 points give a good balance
    n = 8192
    best_seq = None
    best_score = float("inf")

    # 0. Optional fallback from previous run (height_sequence_1)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Core deterministic start – arcsine and smoothed variants
    # -----------------------------------------------------------------
    candidates = []

    # pure arcsine
    cand = generate_arcsine(n)
    candidates.append(cand)

    # smoothed arcsine (delta in [0.0, 0.4] step 0.1)
    for delta in np.linspace(0.0, 0.4, 5):
        candidates.append(generate_arcsine_delta(n, delta))

    # a few Beta shapes around (0.5, 0.5)
    for alpha in np.linspace(0.3, 0.8, 6):
        for beta in np.linspace(0.3, 0.8, 6):
            candidates.append(generate_beta(n, alpha, beta))

    # -----------------------------------------------------------------
    # 2. Mix each candidate with the uniform distribution (lam in [0,1])
    # -----------------------------------------------------------------

    for base in candidates:
        for lam in mix_lams:
            mixed = lam * uniform + (1.0 - lam) * base
            mixed /= mixed.sum()
            sc = compute_score(mixed)
            if sc < best_score:
                best_score, best_seq = sc, mixed.copy()

    print(f"[initial grid] best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing – longest phase of the budget
    # -----------------------------------------------------------------

        polish_deadline = deadline - 2.0  # tiny reserve for the final eval
        print("[polish] start")
        a_polished = subgradient_polish(np.array(best_seq, dtype=float),
                                        polish_deadline,
                                        step0=0.7,
                                        tol_factor=1e-4)
        sc = compute_score(a_polished)
        print(f"[polish] refined score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_polished.copy()

    # -----------------------------------------------------------------
    # 4. Short simulated‑annealing fine‑tuning (if time permits)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[anneal] start")
        a_final = simulated_annealing(np.array(best_seq, dtype=float),
                                      deadline,
                                      init_temp=2e-3,
                                      final_temp=1e-7,
                                      cooling_factor=0.9993,
                                      step_scale_init=0.06,
                                      step_scale_min=1e-6)
        sc = compute_score(a_final)
        print(f"[anneal] final score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, a_final.copy()

    # -----------------------------------------------------------------
    # 5. Safeguard – fall back to uniform if nothing succeeded

    if best_seq is None:

        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # 6. Official evaluation (clamping etc. performed inside evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")

    return best_seq.tolist()
