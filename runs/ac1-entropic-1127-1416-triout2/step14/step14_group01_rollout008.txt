def smooth_optimize(a, deadline):
    t = 0
    best_a = a.copy()
    best_val = compute_score(a)
    tau = max(full_convolution(a)) * 0.5 + 1e-12
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_conv = conv.max()
        # compute weights schedule: w = np.exp(conv / tau)
        w = np.exp(conv / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
        # compute gradient
        # correlation: gradient = 2 * np.correlate(a, w, mode='full')[len(w)-1:len(w)-1+len(a)]
        corr = np.correlate(a, w, mode='full')
        grad = 2.0 * corr[len(w)-1:len(w)-1+len(a)]
        # step size schedule
        step = 0.2 / np.sqrt(t)  # maybe
        a = a - step * grad
        a = project_to_simplex(a)
        # update best
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
        # update tau gradually
        tau = max(tau * 0.995, max_conv * 0.01)
        # optional jitter
        if t % 1000 == 0:
            a += np.random.randn(len(a)) * (step * 0.01)
            a = project_to_simplex(a)
    return best_a, best_val

"""
search_candidate.py

A mixed optimisation procedure for the non‑negative step‑height problem.

The objective to minimise is

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant except for the clipping performed in
``evaluate_sequence``.  The algorithm works in three stages:

1. **Initialisation** – generate a few diverse simplex vectors (uniform,
   geometric decay, random Dirichlet) for several lengths.

2. **Smoothed log‑sum‑exp descent** – minimise a smooth approximation of
   ``max(conv)``:
       fτ(a) = τ·log(∑_k exp(conv_k/τ)).
   The gradient of fτ is a weighted correlation that can be computed with
   ``np.correlate``.  During optimisation τ is annealed, the step size decays
   like 1/√t and the iterates are projected onto the probability simplex.
   This quickly drives the peak of the convolution well below the uniform
   baseline.

3. **Projected sub‑gradient polishing** – once a good candidate is found,
   a classic sub‑gradient method (working on the exact max‑convolution) is
   applied for the remaining time budget.

All helper functions are top‑level; the entry point required by the harness
is ``propose_candidate`` which returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using NumPy's efficient C implementation."""
    return np.convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def generate_initial_vector(n: int, kind: str) -> np.ndarray:
    """Create a simplex vector of length *n*."""
    if kind == "uniform":
        return np.full(n, 1.0 / n, dtype=float)

    if kind == "geometric":
        # random decay factor r in (0.4, 0.9)
        r = 0.4 + 0.5 * np.random.rand()
        a = np.array([r ** i for i in range(n)], dtype=float)
        s = a.sum()
        if s == 0:
            return np.full(n, 1.0 / n)
        return a / s

    # ``random`` – Dirichlet(1) (i.e. exponential draws) → uniform on simplex
    x = np.random.exponential(scale=1.0, size=n)
    s = x.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return x / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp descent
# ----------------------------------------------------------------------


def smooth_descent(a0: np.ndarray, deadline: float,
                   max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv) using
    fτ(a)=τ·log∑exp(conv/τ).  Returns the best vector encountered and its
    (unsmoothed) score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    # initialise temperature τ from the initial convolution
    conv = full_convolution(a)
    tau = max(conv) * 0.5 + 1e-12

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = full_convolution(a)
        max_conv = conv.max()

        # softmax weights (stable formulation)
        #   w_k ∝ exp((conv_k - max_conv) / τ)
        tmp = (conv - max_conv) / tau
        w = np.exp(tmp)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * correlation(a, w) on the non‑negative lags
        # np.correlate returns length (n + len(w) - 1);
        # the portion starting at len(w)-1 corresponds to the needed indices.
        corr = np.correlate(a, w, mode="full")
        start = len(w) - 1
        grad = 2.0 * corr[start:start + n]

        # step size schedule – 0.5 · t⁻½ works well in practice
        step = 0.5 / np.sqrt(it)

        a = a - step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[smooth] iter {it:5d} – new best score {best_val:.6f}")

        # anneal τ slowly; keep it above a tiny fraction of the current max
        tau = max(tau * 0.99, max_conv * 0.01 + 1e-12)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient polishing (identical to the previous baseline)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the exact objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to escape shallow basins
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness (5 s)
    deadline = start_time + budget_s - 5.0

    best_seq: list[float] | None = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback using any known construction (height_sequence_1)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Main stochastic search (several lengths & initialisations)
    # -----------------------------------------------------------------
    lengths = [512, 1024, 2048]          # a few candidate sizes
    kinds = ["uniform", "geometric", "random"]
    attempt = 0

    while time.time() < deadline - 2.0:
        n = lengths[attempt % len(lengths)]
        kind = kinds[(attempt // len(lengths)) % len(kinds)]
        attempt += 1

        sub_deadline = min(time.time() + 45.0, deadline - 2.0)  # ~45 s per attempt
        a0 = generate_initial_vector(n, kind)

        # 1a. Smoothed log‑sum‑exp descent
        a_opt, _ = smooth_descent(a0, sub_deadline)

        # evaluate with the actual evaluator (includes clipping, denominator, etc.)
        seq = a_opt.tolist()
        try:
            val = evaluate_sequence(seq)
        except Exception:
            val = float("inf")
        print(f"[candidate] n={n:<4d} kind={kind:<9s} score={val:.6f}")

        if val < best_score:
            best_score, best_seq = val, seq
            print(f"[new best] → score {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Final polishing of the best candidate (if any time left)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("[polish] start projected sub‑gradient refinement")
        a_start = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a_start, deadline - 1.0)
        try:
            val = evaluate_sequence(a_polished.tolist())
        except Exception:
            val = float("inf")
        print(f"[polish] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Safeguard – uniform vector if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 4. Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
