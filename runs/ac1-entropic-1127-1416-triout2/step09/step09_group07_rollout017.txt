def smooth_adam_opt(start_a, deadline, max_iters, beta_start, beta_end):
   a = start_a.copy()
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   epsilon = 1e-8
   for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        # schedule beta (smoothness param)
        t_frac = t/max_iters
        beta = beta_start * (beta_end/beta_start) ** t_frac
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta*(conv-maxc))
        w = expw/expw.sum()
        a_rev = a[::-1]
        grad = 2*convolution_fft(a_rev, w)[len(a)-1:len(a)-1+len(a)]
        # Adam update (gradient descent on objective)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = 0.01 / (np.sqrt(v_hat)+epsilon) # learning rate may decrease
        a = a - step * m_hat
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum == 0: a = np.full_like(a, 1.0/a.size)
        else: a = a / a_sum
        if t % 200 == 0:
            sc = compute_score(a)

def adam_subgrad_opt(start_a, deadline, max_iters=200000, step0=0.1):
    a = start_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    beta1=0.9;beta2=0.999;eps=1e-8
    for t in range(1, max_iters+1):
        if time.time()>=deadline: break
        conv = convolution_fft(a,a)
        k = int(conv.argmax())
        # gradient at k
        g = np.zeros_like(a)
        j = k - np.arange(a.shape[0])
        mask = (j>=0)&(j<a.shape[0])
        g[mask] = 2*a[j[mask]]
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = step0 / (np.sqrt(v_hat)+eps)
        a = a - step
        a = np.maximum(a, 0)
        s=a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0/a.shape[0])
        if t%500==0:
            sc=compute_score(a)
            if sc < best_sc:
                best_sc=sc;best_a=a.copy()
    # final evaluation
    sc=compute_score(a)
    if sc<best_sc:
        best_sc=sc;best_a=a.copy()
    return best_a,best_sc

"""
Improved step‑function search.

The algorithm:
1.   A quick parametric scan (small n) over three flexible families:
     – product‑power (α‑shape),
     – sum of inverse powers (p‑shape),
     – exponential edge‑weight (c‑shape).
     Random convex combinations of the three families are also tried.
     The best parameters are kept.

2.   The chosen parameters are lifted to a larger n (the final problem size)
     and the corresponding sequence is normalised (∑a = 1).

3.   A smooth‐max exponentiated‑gradient pass (β is annealed up to a large
     value) brings the sequence close to a stationary point of the exact
     objective.

4.   An Adam‑style projected sub‑gradient optimisation on the *exact*
     max‑convolution further refines the solution.

5.   A fast stochastic hill‑climb with O(n) incremental convolution updates
     polishes the final sequence until the time budget expires.

All stages honour the global deadline; the best sequence found is returned
as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    Identical to the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector)
# ----------------------------------------------------------------------


def shape_alpha(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family a_i ∝ [(i+½)(n‑½‑i)]^{-α}."""
    i = np.arange(n, dtype=float)
    # log‑product is stable; avoid overflow/underflow with np.exp
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w = np.nan_to_num(w, nan=0.0, posinf=np.finfo(np.float64).max, neginf=0.0)
    a = w / w.sum()
    return a


def shape_pow_sum(n: int, p: float) -> np.ndarray:
    """Edge‑heavy family a_i ∝ (i+½)^{-p} + (n‑½‑i)^{-p}."""
    i = np.arange(n, dtype=float)
    w = (i + 0.5) ** (-p) + (n - 0.5 - i) ** (-p)
    a = w / w.sum()
    return a


def shape_exp_edge(n: int, c: float) -> np.ndarray:
    """Exponential edge‑weight a_i ∝ exp(c·|i‑mid|)."""
    i = np.arange(n, dtype=float)
    d = np.abs(i - (n - 1) / 2.0)
    w = np.exp(c * d)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Lightweight parametric search on a modest size (fast)
# ----------------------------------------------------------------------


def parametric_scan_small(n_small: int, deadline: float):
    """Explore the three families on a small problem and return the best
    (vector, score, descriptor)."""
    best_a = None
    best_score = float("inf")
    best_desc = None

    i = np.arange(n_small, dtype=float)

    # 1. α‑shape (product‑power)
    alphas = np.linspace(0.0, 6.0, 61)          # 0 … 6 step 0.1
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = shape_alpha(n_small, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a, best_desc = sc, a, ("alpha", alpha)

    # 2. p‑shape (inverse‑power sum)
    ps = np.linspace(0.2, 5.0, 49)             # 0.2 … 5 step ≈0.1
    for p in ps:
        if time.time() >= deadline:
            break
        a = shape_pow_sum(n_small, p)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a, best_desc = sc, a, ("p", p)

    # 3. c‑shape (exponential edge)
    cs = np.linspace(0.001, 0.03, 60)          # 0.001 … 0.03 step ≈0.0005
    for c in cs:
        if time.time() >= deadline:
            break
        a = shape_exp_edge(n_small, c)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a, best_desc = sc, a, ("c", c)

    # 4. Random convex mixtures of the three families
    rng = np.random.default_rng(123456)
    for _ in range(200):
        if time.time() >= deadline:
            break
        # draw random parameters
        alpha = rng.uniform(0.0, 6.0)
        p = rng.uniform(0.2, 5.0)
        c = rng.uniform(0.001, 0.03)
        # draw random mixing weights (simplex)
        mix = rng.random(3)
        mix = mix / mix.sum()
        a_alpha = shape_alpha(n_small, alpha)
        a_pow = shape_pow_sum(n_small, p)
        a_exp = shape_exp_edge(n_small, c)
        a = mix[0] * a_alpha + mix[1] * a_pow + mix[2] * a_exp
        a = a / a.sum()
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a, best_desc = sc, a, ("mix", (alpha, p, c, mix))

    return best_a, best_score, best_desc

def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 150_000,
                        beta_start: float = 0.5,
                        beta_end: float = 2000.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) optimisation of the smoothed
    max‑convolution.  β is annealed exponentially to a large final value;
    the step size follows η₀ / √t.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        # softmax over the (β‑scaled) shifted values – numerically stable
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smooth objective
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def adam_subgrad_opt(start_a: np.ndarray,
                    deadline: float,
                    max_iters: int = 250_000,
                    step0: float = 0.12,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the *exact* max‑convolution objective.
    The sub‑gradient at iteration t is the gradient of the currently
    maximising convolution entry.  After each update the vector is
    projected onto the probability simplex.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv_k w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # element‑wise Adam step (scalar step0 controls overall magnitude)
        step = step0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def fast_hill_climb(start_a: np.ndarray,
                    deadline: float,
                    inc_factor: float = 1.07,
                    decay_factor: float = 0.97,
                    recompute_interval: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Massive stochastic hill‑climb.
    Moves consist of a transfer of a small packet δ from a source index
    to a destination index (or a complete swap).  The convolution is
    updated incrementally in O(n) time, avoiding a full FFT at each step.
    Only strictly improving moves are accepted.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)               # exact convolution
    best_a = a.copy()
    best_sc = 2.0 * n * float(conv.max())
    step_scale = 0.12

    steps = 0
    while time.time() < deadline:
        # pick move type
        if np.random.rand() < 0.5:
            # swap the whole difference between two entries
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            src, dst = (i, j) if a[i] > a[j] else (j, i)
            delta = a[src] - a[dst]            # positive amount to move
        else:
            # transfer a small packet from src → dst
            src = np.random.randint(n)
            if a[src] == 0.0:
                continue
            delta = step_scale * a[src]         # ≤ a[src]
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1                         # ensure dst ≠ src

        # incremental update of the convolution
        conv_cand = conv.copy()
        # linear part: –2·δ·shift_src + 2·δ·shift_dst
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic correction (exact)
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        if cand_score < best_sc:
            # accept the move
            best_sc = cand_score
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            best_a = a.copy()
            step_scale = min(step_scale * inc_factor, 0.5)
        else:
            # reject – shrink the packet size a bit
            step_scale *= decay_factor

        steps += 1
        if steps % recompute_interval == 0:
            # periodic full recomputation to avoid drift
            conv = convolution_fft(a, a)

    return best_a, best_sc

    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] starting smooth exp‑gradient")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 150.0,
            max_iters=200_000,
            beta_start=0.5,
            beta_end=2000.0,
            eta0=0.6,
        )
        print(f"[{time.time():.1f}] after exp‑grad score = {sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] new best after exp‑grad")

    # -------------------------------------------------
    # 3. Adam‑style sub‑gradient optimisation (exact max)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting Adam sub‑gradient")
        a_opt, sc_opt = adam_subgrad_opt(
            np.array(best_seq, dtype=float),
            deadline - 80.0,
            max_iters=300_000,
            step0=0.14,
        )
        print(f"[{time.time():.1f}] after Adam sub‑grad score = {sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam sub‑grad")

    # -------------------------------------------------
    # 4. Fast hill‑climb polishing (uses the remaining budget)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb polishing")
        a_final, sc_final = fast_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
        )
        print(f"[{time.time():.1f}] hill‑climb final score = {sc_final:.6f}")
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] best updated after hill‑climb")

    # -------------------------------------------------
    # 5. Safeguard – if nothing succeeded fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:
        n_small_fallback = 256
        best_seq = [1.0 / n_small_fallback] * n_small_fallback
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats as required by the harness.
    return best_seq
