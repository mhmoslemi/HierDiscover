def lp_opt_step(a, M):
    n = len(a)
    rows = 2*n - 1
    # Build A
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        length = n - j
        A[j:j+length, j] = a[:length]   # slice strip

    # Solve LP to maximize sum(g)
    c = -np.ones(n, dtype=np.float64)  # minimize -sum
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a, M  # return unchanged
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, M
    g_norm = g / sum_g
    # Determine best convex combination coefficient t
    # Choose list of candidate t values
    t_vals = [0.5, 0.7, 0.9, 1.0]  # if 1.0, use g_norm directly (maybe not advisable)
    best_a = a
    best_score = 2 * len(a) * M
    for t in t_vals:
        a_candidate = (1 - t) * a + t * g_norm
        # Normalize to sum 1 (due to floating errors)
        a_candidate = a_candidate / a_candidate.sum()
        # compute max conv
        conv = convolution_fft(a_candidate, a_candidate)
        M_cand = conv.max()
        score = 2 * len(a_candidate) * M_cand
        if score < best_score:
            best_score = score
            best_a = a_candidate
            M = M_cand
    return best_a, M

"""
LP‑based iterative refinement for the max‑convolution bound.

The method starts from a classic inverse‑square‑root (central‑binomial)
shape, then repeatedly solves a linear program

    max 1ᵀg   s.t. g ≥ 0,  conv(f,g) ≤ M

where M = max(conv(f,f)).  The LP pushes mass into indices that are
“quiet’’ for the current f.  A line‑search over a convex mixing weight t
produces a new f with a lower sup‑norm of the self‑convolution.  After
several LP steps a short projected sub‑gradient polish finishes the
search.  The whole routine respects the 1000 s wall‑clock budget and
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# Helper functions ... (next_pow2, convolution_fft, compute_score,
# central_binomial_sequence, etc.)

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def convolution_fft(a, b):
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a):
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def central_binomial_sequence(n):
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * (2*i+1) / (2.0*(i+1))
    a = a / a.sum()
    return a

def max_conv(a):
    return convolution_fft(a, a).max()

def lp_mix_step(a, M, t_candidates):
    """One LP‑step followed by a line‑search over mixing weights."""
    n = a.shape[0]
    rows = 2*n - 1
    # Build the (dense) convolution matrix A so that A @ g = conv(a,g)
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contributes a[:n-j] on rows j..j+n-j-1
    for j in range(n):
        length = n - j
        A[j:j+length, j] = a[:length]

    # Linear program: maximise sum(g)  →  minimise –sum(g)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback to the unchanged vector
        return a, M

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, M
    g_norm = g / sum_g  # now sum(g_norm)=1 and constraints are still respected

    # line‑search for the best convex mixing coefficient t
    best_a = a
    best_M = M
    best_score = 2.0 * n * M

    for t in t_candidates:
        cand = (1.0 - t) * a + t * g_norm
        cand = cand / cand.sum()
        M_cand = max_conv(cand)
        score = 2.0 * n * M_cand
        if score < best_score:
            best_score = score
            best_a = cand
            best_M = M_cand
    return best_a, best_M

def subgrad_optimize(init_a, deadline, max_iter=200_000, step0=0.07):
    """Projected sub‑gradient descent on 2·n·max(conv) for a short polishing."""
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        # Euclidean projection onto the probability simplex
        # (the classic O(n) algorithm)
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        a = np.maximum(a - theta, 0.0)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def random_hill_climb(best_a, deadline, max_iter=30_000):
    """Very cheap stochastic hill‑climb (swaps / tiny transfers)."""
    n = best_a.shape[0]
    cur_a = best_a.copy()
    cur_score = compute_score(cur_a)
    step_scale = 0.08

    while time.time() < deadline and step_scale > 1e-5:
        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        # project back to simplex (fast O(n))
        u = np.sort(cand)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        cand = np.maximum(cand - theta, 0.0)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_score = cand_score
            cur_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96
    return cur_a, cur_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative sequence that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    start_time = time.time()

    best_val = float("inf")

    try:

        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose working length
    # ------------------------------------------------------------------
    # A modest length makes the LP cheap while still giving a good bound.
    n = 2048
    if deadline - time.time() > 600:
        n = 4096   # a larger problem if we have plenty of time

    # ------------------------------------------------------------------
    # 2.  Initialise with the classic central‑binomial shape
    # ------------------------------------------------------------------
    a = central_binomial_sequence(n)
    cur_score = compute_score(a)
    cur_val = evaluate_sequence(a.tolist())
    print(f"Initial central‑binomial score = {cur_val:.6f}")
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.tolist()

    # ------------------------------------------------------------------
    # 3.  Main LP‑mix loop
    # ------------------------------------------------------------------
    M = max_conv(a)
    it = 0
    t_candidates = [0.3, 0.5, 0.7, 0.9]
    while time.time() < deadline - 30 and it < 30:
        it += 1
        a, M = lp_mix_step(a, M, t_candidates)
        cur_score = 2.0 * n * M
        cur_val = evaluate_sequence(a.tolist())
        print(f"[LP {it:02d}] score = {cur_val:.6f}")
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.tolist()
            print(f" *** new best after LP step ***")
        # If improvement stalls we can shrink the set of t values
        if it % 5 == 0:
            # occasionally tighten the mixing coefficient
            t_candidates = [max(0.1, t - 0.1) for t in t_candidates]

    # ------------------------------------------------------------------
    # 4.  Sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        print("\nSub‑gradient polishing …")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline,
            max_iter=400_000,
            step0=0.07,
        )
        refined_val = evaluate_sequence(refined_a.tolist())
        print(f"Polish score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_a.tolist()

    # ------------------------------------------------------------------
    # 5.  Tiny stochastic hill‑climb for any remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        print("\nFinal hill‑climb …")
        hill_a, hill_score = random_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
        )
        hill_val = evaluate_sequence(hill_a.tolist())
        print(f"Hill‑climb final score = {hill_val:.6f}")
        if hill_val < best_val:
            best_val = hill_val
            best_seq = hill_a.tolist()

    # ------------------------------------------------------------------
    # 6.  Return the best sequence found
    # ------------------------------------------------------------------
    final_val = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_val:.6f}")
    return best_seq

"""
LP‑based iterative refinement for the max‑convolution bound.

The routine starts from the classic inverse‑square‑root (central‑binomial)
shape, then repeatedly solves a linear programme

    maximise   1ᵀ g
    subject to   g ≥ 0 ,   conv(f, g) ≤ M

where M = max(conv(f, f)).  The LP pushes mass into indices that are
“quiet’’ for the current f.  A short line‑search over a mixing weight
t produces a new f with a lower sup‑norm of the self‑convolution.
After several LP steps a projected sub‑gradient polish finishes the
search.  The whole search respects the 1000 s wall‑clock budget and
returns the best sequence (a list of non‑negative floats).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based convolution, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (efficient for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector already normalised
    to the simplex (∑ a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape (central‑binomial coefficients) normalised."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


def max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (∑a = 1)."""
    return convolution_fft(a, a).max()


# ----------------------------------------------------------------------
# Linear‑programme step + line‑search over mixing coefficient t
# ----------------------------------------------------------------------


def lp_mix_step(a: np.ndarray, M: float, t_candidates):
    """
    Given current shape a (sum = 1) and its current max‑convolution M,
    solve the LP   maximise 1ᵀg   s.t.   g ≥ 0 , conv(a,g) ≤ M,
    then return the best convex combination (1‑t)·a + t·g_norm for the
    supplied list of t values.
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the dense convolution matrix A so that A @ g = conv(a,g)
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contributes a[:n‑j] on rows j … j+n‑j‑1
    for j in range(n):
        length = n - j
        A[j:j + length, j] = a[:length]

    # Linear programme: maximise sum(g)  →  minimise –sum(g)
    c = -np.ones(n, dtype=np.float64)          # objective
    b_ub = np.full(rows, M, dtype=np.float64)  # upper bounds conv(a,g) ≤ M
    bounds = [(0, None)] * n                    # g ≥ 0

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # fall back to the unchanged vector
        return a, M

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return a, M

    # normalise g to the simplex (still respects the constraints)
    g_norm = g / sum_g

    best_a = a
    best_M = M
    best_score = 2.0 * n * M

    for t in t_candidates:
        cand = (1.0 - t) * a + t * g_norm
        cand = cand / cand.sum()                # project onto simplex
        M_cand = max_conv(cand)
        score = 2.0 * n * M_cand
        if score < best_score:
            best_score = score
            best_a = cand
            best_M = M_cand

    return best_a, best_M


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (polishing phase)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0 , Σx = 1}.
    Implementation of the O(n) algorithm from
    Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, v.shape[0] + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgrad_optimize(init_a: np.ndarray, deadline: float,
                     max_iter: int = 200_000, step0: float = 0.07):
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Very light – suitable for a final polishing after the LP phase.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry of the convolution
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climb (optional final tweak)
# ----------------------------------------------------------------------


def random_hill_climb(start_a: np.ndarray, deadline: float,
                      max_iter: int = 30_000):
    """A very cheap hill‑climb that swaps or transfers a tiny fraction of mass."""
    n = start_a.shape[0]
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    step_scale = 0.08

    while time.time() < deadline and step_scale > 1e-5:
        cand = cur_a.copy()
        if np.random.rand() < 0.5:                        # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                             # fractional transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_score = cand_score
            cur_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

    return cur_a, cur_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for the harness

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback (previously discovered) construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working length (larger gives a richer search space)
    # ------------------------------------------------------------------
    n = 2048
    if deadline - time.time() > 600:               # plenty of time → larger n
        n = 4096

    # ------------------------------------------------------------------
    # 2. Initialise with the inverse‑square‑root (central‑binomial) shape
    # ------------------------------------------------------------------
    a = central_binomial_sequence(n)
    cur_val = evaluate_sequence(a.tolist())
    print(f"Initial central‑binomial score = {cur_val:.6f}")
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.tolist()

    # ------------------------------------------------------------------
    # 3. Main LP‑mix loop
    # ------------------------------------------------------------------
    M = max_conv(a)
    iteration = 0
    t_candidates = [0.3, 0.5, 0.7, 0.9]

    while time.time() < deadline - 30 and iteration < 40:
        iteration += 1
        a, M = lp_mix_step(a, M, t_candidates)
        cur_val = evaluate_sequence(a.tolist())
        print(f"[LP {iteration:02d}] score = {cur_val:.6f}")
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.tolist()
            print(" *** new best after LP step ***")

        # Occasionally shrink the mixing candidates to force finer updates
        if iteration % 5 == 0:
            t_candidates = [max(0.1, t - 0.1) for t in t_candidates]

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient polishing (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        print("\nSub‑gradient polishing …")
        refined_a, _ = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline,
            max_iter=400_000,
            step0=0.07,
        )
        refined_val = evaluate_sequence(refined_a.tolist())
        print(f"Polish score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_a.tolist()

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climb (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        print("\nFinal hill‑climb …")
        hill_a, _ = random_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
        )
        hill_val = evaluate_sequence(hill_a.tolist())
        print(f"Hill‑climb final score = {hill_val:.6f}")
        if hill_val < best_val:
            best_val = hill_val
            best_seq = hill_a.tolist()

    # ------------------------------------------------------------------
    # 6. Return the best sequence discovered
    # ------------------------------------------------------------------
    final_val = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_val:.6f}")
    return best_seq
