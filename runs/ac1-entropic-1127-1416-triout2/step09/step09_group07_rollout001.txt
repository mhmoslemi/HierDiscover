def annealing_targeted_hill_climb(start_a, deadline, ...):
    a = start_a.copy()
    n = len(a)
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_a = a.copy()
    # annealing schedule:
    total_time = deadline - time.time()
    T0 = 0.005
    T_end = 1e-7
    start_time = time.time()
    # step scale
    step_scale = 0.1
    # occasional adapt step_scale
    step_inc = 1.02
    step_decay = 0.98

    while time.time() < deadline:
        elapsed = time.time() - start_time
        T = T0 * (T_end / T0) ** (elapsed / total_time)

        # choose move type:
        if np.random.rand() < 0.7:
            # targeted transfer from peak contributions
            # compute kmax and contributions
            kmax = int(np.argmax(conv))
            # compute contributions for i where j = kmax - i in [0,n-1]
            # we can compute array of contributions: for i in [max(0, kmax-(n-1)), min(kmax, n-1)]
            low = max(0, kmax - (n - 1))
            high = min(kmax, n - 1)
            idxs = np.arange(low, high + 1)
            j = kmax - idxs
            contrib = a[idxs] * a[j]
            # weight by contribution; ensure sum >0
            if contrib.sum() > 0:
                prob = contrib / contrib.sum()
                src = np.random.choice(idxs, p=prob)
            else:
                src = np.random.randint(n)
            # dest random different from src
            dst = np.random.randint(n-1)
            if dst >= src:
                dst += 1
            # delta from src
            if a[src] <= 0:
                continue
            delta = step_scale * a[src] * np.random.rand()
        else:
            # random transfer
            src = np.random.randint(n)
            if a[src] <= 0:
                continue
            dst = np.random.randint(n-1)
            if dst >= src:
                dst += 1
            delta = step_scale * a[src] * np.random.rand()
        # ensure delta not zero
        if delta <= 0:
            continue

        # compute candidate conv updates using incremental.
        # Build temporary copy of conv modifications: we incorporate delta contributions.
        # Rather than copy whole conv each time, we compute modified entries on the fly and compute max.

        # compute new conv for k from src to src+n-1 and dst to dst+n-1:
        # conv_k_new = conv_k + change_src_k + change_dst_k + quadratic terms at positions 2*src etc.
        # We can compute candidate max by scanning all indices with modifications.
        # For performance, we just copy conv to conv_cand (np.copy) and modify slices.

        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        # acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / T):
            # accept
            # update a: modify src and dst
            a[src] -= delta
            a[dst] += delta
            # update conv and cur_score
            conv = conv_cand
            cur_score = cand_score

            # update best
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                # optionally increase step_scale
                step_scale = min(step_scale * step_inc, 0.5)

        else:
            # reject
            step_scale = max(step_scale * step_decay, 1e-6)

        # Periodic full recompute to avoid drift due to rounding errors:
        if (np.random.rand() < 0.001):
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * conv.max()
    return best_a, best_score

"""
Improved search for the minimum of

    score(a) = 2 * n * max(a ⊛ a) / (sum(a)^2)

where a is a non‑negative step‑function (list of floats).  The function
`evaluate_sequence` implements exactly the same formula with additional
security checks, so the optimizer works only with the normalised version
(sum(a)=1) and returns a plain Python list.

The algorithm consists of three stages:

1. **Analytic start** – a one‑parameter family
       a_i(α) ∝ [(i+½)(n‑½‑i)]^{‑α}
   is scanned on a dense α‑grid (first a coarse grid, then a refined
   grid around the best α).  The best vector from this scan is kept.

2. **Targeted simulated‑annealing hill climbing** – mass is repeatedly
   moved between two indices.  Moves are biased towards the indices that
   contribute most to the current maximal convolution coefficient, but a
   fraction of moves is purely random to maintain exploration.
   The acceptance rule is the usual Metropolis criterion with a
   temperature that decays from ~2·10⁻³ to ~10⁻⁸.  The convolution is
   updated incrementally (O(n) per move) and the best ever vector is
   recorded.

3. **Final polishing** – a short Euclidean projected sub‑gradient
   descent on the exact (non‑smoothed) objective, starting from the best
   vector found by the annealing phase.

All phases respect the wall‑clock limit (default 1000 s).  The routine
returns the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This matches the value returned by `evaluate_sequence` when the input
    already satisfies the sum‑normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytic start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    # avoid log(0) – the end points never appear because i∈[0,n‑1]
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Clip possible overflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Coarse + fine search over α.  Returns the best normalised vector and its
    exact score.
    """
    best_a = None
    best_score = float("inf")

    # --- coarse phase -------------------------------------------------
    alphas_coarse = np.linspace(0.05, 1.5, 31)          # 31 points
    for alpha in alphas_coarse:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    # --- fine phase around the best coarse α -------------------------
    if best_a is not None:
        # locate best α in the coarse list
        idx_best = np.argmin(np.abs(alphas_coarse - alpha))
        alpha_center = alphas_coarse[idx_best]
        alphas_fine = np.linspace(alpha_center - 0.2, alpha_center + 0.2, 41)
        for alpha in alphas_fine:
            if time.time() >= deadline:
                break
            a = alpha_start(n, alpha)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Targeted simulated‑annealing hill climb
# ----------------------------------------------------------------------


def annealed_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        max_no_improve: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with Metropolis acceptance.
    The move generator is biased towards the indices that contribute most
    to the current maximal convolution coefficient.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * float(conv.max())

    best_a = a.copy()
    best_score = cur_score

    # Temperature schedule: from T0 to T_end
    total_time = deadline - time.time()
    T0 = 2.0e-3
    T_end = 1.0e-8

    # Step scale (fraction of source mass to move)
    step_scale = 0.2
    step_inc = 1.03
    step_decay = 0.97

    it = 0
    while time.time() < deadline and it < max_no_improve:
        it += 1
        # ---- temperature ----
        elapsed = time.time() - (deadline - total_time)
        T = T0 * (T_end / T0) ** (elapsed / total_time)

        # ---- choose move ----
        if np.random.rand() < 0.65:
            # targeted move: pick from the current maximal convolution
            k_max = int(np.argmax(conv))
            # the contributing indices are i where 0 ≤ i < n and 0 ≤ j = k_max-i < n
            low = max(0, k_max - (n - 1))
            high = min(k_max, n - 1)
            idxs = np.arange(low, high + 1, dtype=int)
            js = k_max - idxs
            contrib = a[idxs] * a[js]                     # product contributions
            if contrib.sum() > 0:
                prob = contrib / contrib.sum()
                src = np.random.choice(idxs, p=prob)
            else:
                src = np.random.randint(n)
            # destination is a random different index
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
        else:
            # pure random move
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

        # make sure we can move something
        if a[src] <= 0.0:
            continue

        # delta = fraction of source mass (scaled by step_scale and a uniform factor)
        delta = step_scale * a[src] * np.random.rand()
        if delta <= 0.0:
            continue

        # ---- incremental convolution update (copy → modify) ----
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        # ---- Metropolis acceptance ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability ∝ exp(−Δ/T)
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_score = cand_score

            # record best ever
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                step_scale = min(step_scale * step_inc, 0.5)   # increase step if we improve
        else:
            # rejected – shrink step scale slowly
            step_scale = max(step_scale * step_decay, 1e-6)

        # Occasionally recompute the full convolution to avoid numerical drift
        if it % 250_000 == 0:
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())
            # also keep the best even if recompute changed only a tiny amount
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient refinement (short polish)
# ----------------------------------------------------------------------


def subgrad_polish(start_a: np.ndarray,
                   deadline: float,
                   max_iters: int = 120_000,
                   step0: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Classic Euclidean projected sub‑gradient descent on the exact
    max‑convolution objective.  Only a relatively small number of
    iterations are performed (the bulk of the work is done by the
    annealed hill‑climb).
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t a:
        # g_i = 2 * a_{k-i} for indices where 0 ≤ k-i < n
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

    start_time = time.time()

    best_seq = None
