analysisThe problem is a "challenge" for AI system: The objective evaluate_sequence defined above computes:

- Preprocess: check for list, non empty, each element numeric, not nan/inf.
- Convert each element to float.
- Clip negatives to 0 (max(0,x)) and limit to 1000 (min(1000,x)).
- Compute convolution of the sequence with itself (full length n+n-1). Compute max_b = maximum of convolution.
- Compute sum_a = sum of sequence.
- If sum_a < 0.01: return infinite.
- Return 2 * n * max_b / (sum_a**2).

Thus it's a problem of minimizing 2 n max(conv(a,a)) / (sum(a)^2). Equivalent to maximizing sum(a) given fixed max(conv) or shaping a to reduce max convolution relative to the sum.

def solve_lp_max_sum(A, M):
    n = A.shape[1]
    c = -np.ones(n)  # maximize sum(b) -> minimize -sum
    b_ub = np.full(A.shape[0], M)
    bounds = [(0, None)] * n
    # SciPy expects A_ub @ x <= b_ub
    # Use method='highs'.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        return None

from scipy.optimize import linprog
def solve_lp_max_sum(A, M):
    n = A.shape[1]
    c = -np.ones(n) # minimize negative sum => maximize sum
    # A is sparse CSR; linprog supports A_ub as dense by default, but can supply sparse
    # using method='highs' which accepts sparse.
    b_ub = np.full(A.shape[0], M, dtype=float)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10})
    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        return None
```

"""
step‑function optimiser for the ``evaluate_sequence`` challenge.

The optimiser explores several initial shapes (central‑binomial, arcsine,
sparse Dirichlet), repeatedly refines them with a cheap linear‑program
partner (SciPy’s high‑performance LP solver), upsamples to a larger
dimension, then performs three stages of continuous optimisation:

1.  Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
2.  Projected sub‑gradient descent on the true max‑convolution.
3.  Stochastic hill‑climb with simplex projection.

All stages are time‑aware and stop before the global deadline.  The best
sequence found (a plain list of non‑negative floats) is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum ≈ 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution matrix (Toeplitz) for the LP
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix A such that A @ b = conv(f, b).
    Column j contains f shifted down by j rows.
    """
    n = f.shape[0]
    diags = [f] * n
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = sp.diags(diagonals=diags, offsets=offsets,
                 shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum(A: sp.csr_matrix, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    c = -np.ones(n)                     # minimise -sum(b) == maximise sum(b)
    b_ub = np.full(A.shape[0], M, dtype=float)
    bounds = [(0, None)] * n

    # SciPy’s HiGHS interior‑point solver handles sparse A efficiently.
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None
    return np.maximum(res.x, 0.0)   # clip tiny negatives


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 300.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)   # first moment
    v1 = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing parameter β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed surrogate
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)            # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        v1 = adam_beta2 * v1 + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m1 / (1.0 - adam_beta1 ** it)
        v_hat = v1 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a = project_to_simplex(a - update)

        if it % 250 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        cur = compute_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 400_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)
