"""
Fast multi‑phase optimizer for the step‑function evaluation.

Phase 1 – Broad sampling of simple analytic families (α‑grid,
power‑law, random Dirichlet).  All candidates are projected onto the
probability simplex (∑a_i = 1) and the best exact score is kept.

Phase 2 – Optional exponentiated‑gradient (smooth max) refinement with
a log‑sum‑exp surrogate.  A large number of iterations is allowed
while the wall‑clock budget permits.

Phase 3 – Simulated‑annealing hill‑climb.  Moves transfer a small
packet of mass from one position to another (or swap two positions)
and update the convolution *incrementally* in O(n) time using the
exact formula

    Δconv = –2·δ·shift_src(a) + 2·δ·shift_dst(a)
            + δ²·e_{2·src} + δ²·e_{2·dst} – 2·δ²·e_{src+dst}.

Only improving moves are always accepted; uphill moves are accepted
with probability exp(−Δ/temperature) and the temperature decays
geometrically.  The step size adapts to the success rate.
The best sequence seen during the walk is returned.

All phases respect a total wall‑clock budget of ~1000 s.
The final sequence is a plain list of non‑negative floats (already
normalised to sum = 1, which leaves the evaluation invariant).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This is the value returned by `evaluate_sequence` when the input
    already satisfies the sum‑normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    # avoid log of zero
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def power_law_start(n: int, p: float, decreasing: bool = True) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (or its mirror) on the simplex."""
    i = np.arange(n, dtype=float) + 1.0
    if decreasing:
        w = i ** (-p)
    else:
        w = (n + 1 - i) ** (-p)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient optimisation (optional)
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 500.0,
                        eta0: float = 0.8,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimises a log‑sum‑exp surrogate of max(a⊛a) via a KL‑mirror
    (exponentiated‑gradient) update.  The schedule for β is exponential,
    η∝1/√t.  Returns the best exact score seen.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max via log‑sum‑exp
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))      # stabilised
        w = expw / expw.sum()                    # softmax weights

        # gradient of the smoothed objective
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # KL‑mirror step
        eta = eta0 / np.sqrt(it)
        a = np.maximum(a * np.exp(-eta * grad), eps)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def simulated_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         recompute_interval: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Random‑walk optimisation accepting only improvements, but also
    occasional uphill moves according to a temperature schedule.
    Incremental O(n) updates of the convolution are used.
    """
    a = start_a.copy()
    n = a.shape[0]
    a /= a.sum()                     # keep on simplex
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_score = cur_score

    # adaptive step size (absolute amount, not relative)
    step_scale = 0.5 / n            # roughly twice the average entry
    max_step = 0.5
    min_step = 1e-12

    # geometric temperature schedule
    temperature = cur_score * 0.01
    decay_temp = 0.9999

    # step‑size adaptation
    inc_factor = 1.02
    decay_factor = 0.98

    steps = 0
    while time.time() < deadline:
        src = np.random.randint(n)
        if a[src] == 0.0:
            continue

        # destination ≠ src
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # absolute packet – never larger than the source mass
        delta = min(step_scale, a[src])
        if delta <= 0.0:
            continue

        # incremental convolution update
        conv_cand = conv.copy()
        # linear part: –2·δ·shift_src + 2·δ·shift_dst
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic correction (exact)
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_score = 2.0 * n * float(conv_cand.max())
        delta_score = cand_score - cur_score

        # Metropolis acceptance
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if temperature > 0.0:
                prob = math.exp(-delta_score / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # commit move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_score = cand_score

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

            # enlarge step on success
            step_scale = min(step_scale * inc_factor, max_step)
        else:
            # shrink step on rejection
            step_scale = max(step_scale * decay_factor, min_step)

        # cool down
        temperature *= decay_temp

        steps += 1
        if steps % recompute_interval == 0:
            # periodic full recomputation to avoid numeric drift
            conv = convolution_fft(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows a three‑stage plan:
      1. Broad sampling of simple analytic families.
      2. Optional exponentiated‑gradient refinement.
      3. Simulated‑annealing hill‑climb with incremental convolution updates.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # -------------------------------------------------
    # Choose a reasonable length (larger = more freedom)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Broad analytic sampling
    # -------------------------------------------------
    candidates = []

    # α‑grid (extended)
    alphas = np.concatenate([
        np.linspace(-1.0, 0.0, 11),
        np.linspace(0.0, 5.0, 51)
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        candidates.append(a)

    # power‑law decreasing families
    p_vals = np.linspace(0.1, 4.0, 40)
    for p in p_vals:
        if time.time() >= deadline:
            break
        a = power_law_start(n, p, decreasing=True)
        candidates.append(a)
        a_rev = power_law_start(n, p, decreasing=False)
        candidates.append(a_rev)

    # few random Dirichlet samples
    for _ in range(20):
        if time.time() >= deadline:
            break
        a = np.random.rand(n)
        a /= a.sum()
        candidates.append(a)

    # -------------------------------------------------
    # Evaluate all candidates, keep the best
    # -------------------------------------------------
    for a in candidates:
        if time.time() >= deadline:
            break
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    if best_seq is not None:
        print(f"[{time.time():.1f}] best after sampling = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Optional smooth exponentiated‑gradient refinement
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] starting smooth exp‑gradient")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 100.0,
            max_iters=300_000,
            beta_start=0.5,
            beta_end=500.0,
            eta0=0.8,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Simulated‑annealing hill‑climb
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] starting hill‑climb")
        a_final, sc_final = simulated_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
