def subgrad_optimize(init_a, deadline, max_iter=200_000, step0=0.05):
   a = init_a.copy()
   n = a.shape[0]
   best_a = a.copy()
   best_score = compute_score(a)

        if time.time()>=deadline: break

def hillclimb(initial_a, deadline):
    a = np.array(initial_a)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step = 0.1 # fraction mass transferred relative to source
    while time.time() < deadline:
        # random move: either swap or transfer.
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            # swap a small fraction of mass? Actually swap values (exchanging)
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i == j: continue
            delta = step * best_a[i]
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            step = min(step*1.05, 0.5)
        else:
            step *= 0.97
        # break early when step tiny
        if step < 1e-5:
            break
    return best_a, best_score

"""
search_candidate.py
-------------------

A multi‑restart, β‑annealed L‑BFGS optimiser supplemented by a short
projected sub‑gradient polish and a final stochastic hill‑climb.
The optimiser works on the probability simplex (∑a = 1) so that the
objective returned by ``evaluate_sequence`` coincides with the exact
score ``2·n·max(conv(a,a))``.  The routine dynamically adapts the
sequence length to the remaining wall‑clock budget and tries several
initialisations (central‑binomial, Dirichlet draws and exponential
decay) to avoid being trapped in a local minimum.

The code respects the 1000 s time limit, prints progress messages and
returns a plain ``list[float]`` suitable for the competition harness.
"""

import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.  Returns a vector of length
    len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector already on the
    simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically‑stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate based on the log‑sum‑exp approximation of max
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))
    and g is the gradient of f with respect to the soft‑max logits.
    The parametrisation a = softmax(logits) guarantees a ≥ 0 and ∑a = 1.
    """
    a = softmax_logits(logits)

    # self‑convolution
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # smoothed max (log‑sum‑exp)
    exp_weights = np.exp(beta * (conv - maxc))       # centred for stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient w.r.t. a
    w = exp_weights / Z                               # (2n‑1,) soft‑max over conv entries
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)              # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # chain‑rule back to logits
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 300):
    """
    Anneal the smoothing parameter β while optimising the smooth surrogate.
    Returns the best simplex vector found and its exact score.
    """
    logits = np.log(start_a + 1e-20)                # initialise logits safely
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for beta in betas:
        if time.time() >= deadline:
            break

        # allocate a reasonable per‑step budget (≈30‑50 iterations per second)
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(20, int(time_left * 45)))

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.1f}] new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07):
    """
    Simple projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny jitter to shake off plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.008 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (swap / fractional transfer)
# ----------------------------------------------------------------------


def hillclimb(initial_a: np.ndarray, deadline: float):
    """
    Very cheap stochastic hill‑climb that keeps the iterate on the simplex.
    """
    a = initial_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07
    while time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96
        if step_scale < 1e-5:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic start – truncated central‑binomial coefficients
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape: a_i = C(2i,i)/4^i, normalised to the simplex."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Miscellaneous helpers for random starts
# ----------------------------------------------------------------------


def random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a random point on the simplex from a Dirichlet(alpha) distribution."""
    x = np.random.gamma(alpha, size=n)
    return x / x.sum()


def exponential_decay(n: int, lam: float = 0.0005) -> np.ndarray:
    """
    Exponential decay a_i ∝ exp(-lam·i), then normalised.
    The decay rate is chosen so that the distribution is reasonably spread.
    """
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    return a / a.sum()


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The algorithm proceeds by:
        * choosing a working length based on the time budget,
        * running several random restarts,
        * for each restart: β‑annealed L‑BFGS on the smooth log‑sum‑exp surrogate,
        * a short projected sub‑gradient polish,
        * a final stochastic hill‑climb.
    Returns a plain list of floats (the simplex‑normalised coefficients).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for the harness

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback to a previously known construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose a sensible working length (larger ⇒ more freedom)
    # -----------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 750:
        n = 65536
    elif time_left > 400:
        n = 32768
    elif time_left > 200:
        n = 16384
    else:
        n = 8192
    print(f"Working sequence length set to n = {n}")

    # -----------------------------------------------------------------
    # 2. Prepare a pool of initialisers
    # -----------------------------------------------------------------
    init_functions = [
        lambda: central_binomial_sequence(n),
        lambda: random_dirichlet(n, alpha=0.5),
        lambda: random_dirichlet(n, alpha=1.0),
        lambda: random_dirichlet(n, alpha=2.0),
        lambda: exponential_decay(n, lam=0.0003),
        lambda: exponential_decay(n, lam=0.0010),
    ]

    # -----------------------------------------------------------------
    # 3. Main optimisation loop (multiple restarts)
    # -----------------------------------------------------------------
    restart_idx = 0
    while time.time() < deadline - 30:   # leave a few seconds for final polishing
        init_fn = np.random.choice(init_functions)
        a0 = init_fn()
        # ensure numerical safety
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()

        # ---- Smooth L‑BFGS optimisation ----
        sub_deadline = deadline - 20.0   # reserve ~20 s for polishing / hill‑climb
        a_smooth, smooth_score = lbfgs_continuation(
            start_a=a0,
            beta_start=5.0,
            beta_end=2000.0,
            deadline=sub_deadline,
            schedule_steps=12,
            maxiter_per_step=300,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"[restart {restart_idx}] post‑smooth score = {best_val:.6f}")

        # ---- Short sub‑gradient polishing (if time permits) ----
        if time.time() < deadline - 8.0:
            polish_deadline = deadline - 8.0
            refined_a, refined_score = subgrad_optimize(
                init_a=np.array(best_seq, dtype=float),
                deadline=polish_deadline,
                max_iter=200_000,
                step0=0.07,
            )
            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_a.tolist()
                print(f"[restart {restart_idx}] sub‑grad polished score = {best_val:.6f}")

        restart_idx += 1

    # -----------------------------------------------------------------
    # 4. Final stochastic hill‑climb on the best candidate found so far
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nFinal hill‑climb phase …")
        final_deadline = deadline
        best_a, final_score = hillclimb(np.array(best_seq, dtype=float), final_deadline)
        if final_score < best_val:
            best_val = final_score
            best_seq = best_a.tolist()
            print(f"Hill‑climb improved final score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Fallback in the extremely unlikely case that nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report final result
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
