def adam_optimize(a_init, deadline, alpha_start=1.0, alpha_max=80.0, lr=0.1):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    t=0
    best_a = a.copy()
    best_score = compute_score(a)
    alpha = alpha_start
    while time.time() < deadline:
        t+=1
        # compute convolution
        conv = convolution_fft(a, a) # shape 2n-1
        # compute weights w
        b_max = conv.max()
        # stabilizing subtract max
        exps = np.exp(alpha * (conv - b_max))
        w = exps / exps.sum()
        # compute gradient via correlation
        grad = gradient_softmax(a, w) # returns length n
        # Adam updates
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        # project onto simplex
        a = project_to_simplex(a)
        # maybe enforce minimal sum tolerance? Already sum=1.
        # compute score occasionally (every maybe 10 iterations)

"""
Deterministic‑annealed Adam optimiser for the max‑convolution objective.

The code builds a probability vector a (Σ a = 1, a_i ≥ 0) of length *n*
and minimises

    f(a) = 2·n·max_k (a * a)[k]               (the exact score used by evaluate_sequence)

using a smooth approximation of the max (log‑sum‑exp) and the Adam optimiser.
After the Adam phase a short hill‑climb polish is applied.
All operations respect the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Low‑level utilities
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using the real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0

    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)

    return np.maximum(v - theta, 0.0)


# -------------------------------------------------------------
# Sequence generators (hand‑crafted seeds)
# -------------------------------------------------------------
def generate_deterministic(n: int) -> np.ndarray:
    """Sine‑squared shape, normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p .
    Positive p concentrates mass in the centre, negative p near the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Sample a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# -------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) objective
# -------------------------------------------------------------
def gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  ( (1/α)·log Σ_k exp(α·c_k) )  where w_k = softmax(α·c_k)
    The gradient reduces to 2·(a ⋆ w)[j]  (cross‑correlation).

    Parameters
    ----------
    a : (n,) ndarray   – current probability vector (Σa = 1)
    w : (2·n‑1,) ndarray – softmax weights over the full convolution

    Returns
    -------
    g : (n,) ndarray – gradient (not yet projected).
    """
    n = a.shape[0]
    len_w = w.shape[0]          # = 2·n‑1
    N = next_pow2(n + len_w - 1)   # size for linear correlation

    A = np.fft.rfft(a, n=N)
    W = np.fft.rfft(w, n=N)

    # linear cross‑correlation a ⋆ w
    corr = np.fft.irfft(A * np.conj(W), n=N)[: n + len_w - 1]

    offset = len_w - 1           # index of lag 0 in the linear result
    g = 2.0 * corr[offset: offset + n]
    return g


# -------------------------------------------------------------
# Adam optimiser on the smooth‑max objective
# -------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    alpha_start: float = 1.0,
    alpha_max: float = 80.0,
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam on the soft‑max surrogate of max(conv(a,a)).  The vector is
    kept on the probability simplex after each update.
    Returns the best vector (and its exact score) encountered before the deadline.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(best_a)

    t = 0
    alpha = alpha_start
    n = a.shape[0]

    # pre‑compute the FFT size needed for the correlation (constant)
    len_w = 2 * n - 1
    N_corr = next_pow2(n + len_w - 1)

    while time.time() < deadline:
        t += 1

        # ---------- convolution ----------
        conv = convolution_fft(a, a)          # length 2·n‑1
        conv_max = conv.max()

        # ---------- soft‑max weights ----------
        # stability: shift by the maximum before exponentiation
        exps = np.exp(alpha * (conv - conv_max))
        w = exps / exps.sum()                # length 2·n‑1, Σw = 1

        # ---------- gradient ----------
        # use the pre‑computed size N_corr for the correlation FFT
        A = np.fft.rfft(a, n=N_corr)
        W = np.fft.rfft(w, n=N_corr)
        corr = np.fft.irfft(A * np.conj(W), n=N_corr)[: n + len_w - 1]
        offset = len_w - 1
        grad = 2.0 * corr[offset: offset + n]          # shape (n,)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection ----------
        a = project_to_simplex(a)

        # ---------- bookkeeping ----------
        if t % 10 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                elapsed = time.time() - (deadline - (deadline - (deadline - time.time())))  # dummy for timing
                print(f"[Adam] iter {t:6d}, best score = {best_score:.6f}")

        # gradually tighten the surrogate
        if alpha < alpha_max and t % 50 == 0:
            alpha = min(alpha_max, alpha * 1.05)

    return best_a, best_score


# -------------------------------------------------------------
# Greedy hill‑climb polishing (as in the previous baseline)
# -------------------------------------------------------------
def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local moves (swap or tiny mass transfer) that are kept only
    when they improve the exact score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.07

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score
