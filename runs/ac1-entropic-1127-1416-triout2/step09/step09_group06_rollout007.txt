def arcsine_sequence(n):
    idx = np.arange(n)
    x = (idx + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = a / a.sum()
    return a

def lp_max_sum(a, M, rel_tol=0.02, max_cuts=10):
    n = len(a)
    conv_a_a = convolution_fft(a, a)
    # active constraints: indices where conv >= M * (1 - rel_tol)
    active = set(np.where(conv_a_a >= M * (1 - rel_tol))[0].tolist())
    # always include global max index
    active.add(int(np.argmax(conv_a_a)))
    for iteration in range(max_cuts):
        idx = np.array(sorted(active), dtype=int)
        k_len = len(idx)
        A = np.zeros((k_len, n), dtype=float)
        for row_i, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # fill slice
            col_range = np.arange(lo, hi + 1)
            A[row_i, lo:hi+1] = a[k - col_range]
        b_ub = np.full(k_len, M)
        # solve
        res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            # fallback: return a (no change)
            return a
        b = res.x
        # verify all convolution constraints
        conv_ab = convolution_fft(a, b)
        violations = np.where(conv_ab > M * (1 + 1e-9))[0]
        if len(violations) == 0:
            return b
        # add new violating constraints
        added = 0
        for v in violations:
            if v not in active:
                active.add(v)
                added += 1
        if added == 0:
            # cannot add new; break
            break
    return b

def hill_climb(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    step = 0.05
    while time.time() < deadline - 1.0:
        # propose neighbor
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonnegative
        if cand[i] < 0:
            cand[i] = 0.0
        cand = cand / cand.sum()
        val = compute_score(cand)
        if val < best_score:
            best_score = val
            best_a = cand
            step = min(step * 1.1, 0.2)
        else:
            step *= 0.98
    return best_a, best_score

"""
Optimiser for the max‑convolution bound.

The programme works in three phases:

1. **Initialisation** – several deterministic and random simplex vectors
   (central‑binomial, arcsine, optional historic construction and a few
   Dirichlet draws) are built, and the best of them is kept.

2. **LP‑based refinement** – an iterative scheme that repeatedly solves a
   tiny linear programme:
        maximise   Σ b_i
        subject to   (a ∗ b)[k] ≤ M   for a selected set of indices k,
                    b_i ≥ 0,
   where *M* is the current peak of the self‑convolution *a∗a*.
   The LP pushes mass into positions that do not increase the peak.
   The resulting vector *b* is normalised, mixed with the current *a*
   (line‑search over the mixing coefficient), and the process repeats
   until the time budget is exhausted.

3. **Final hill‑climb** – a cheap stochastic local‑search (fractional mass
   transfer) finishes the optimisation, polishing the solution.

All heavy work is performed with FFT‑based convolutions ( O(n log n) ) and
the HiGHS LP solver from SciPy.  The routine respects the 1000 s wall‑clock
budget and returns a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (efficient for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector already on the
    probability simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simple deterministic seeds
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape a_i = C(2i,i)/4^i, normalised to the simplex."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


def arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine density 1/sqrt(x(1‑x)), x∈(0,1), normalised to the simplex."""
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programme step (maximise Σb under a∗b ≤ M)
# ----------------------------------------------------------------------


def lp_max_sum(a: np.ndarray, M: float, rel_tol: float = 0.02,
               max_cuts: int = 8) -> np.ndarray:
    """
    Solve the LP:
        maximise   Σ b_i
        subject to (a ∗ b)[k] ≤ M for a small active set of indices k,
                    b_i ≥ 0.
    The active set is grown iteratively until the solution satisfies the
    full constraint (or the cut‑loop limit is reached).  Returns the optimal
    vector b (not normalised).
    """
    n = a.shape[0]

    # start with constraints that are already tight (within rel_tol)
    conv_aa = convolution_fft(a, a)
    active = set(np.where(conv_aa >= M * (1.0 - rel_tol))[0].tolist())
    active.add(int(np.argmax(conv_aa)))          # always keep the max‑index

    for _ in range(max_cuts):
        idx = np.array(sorted(active), dtype=int)
        k_len = len(idx)

        # build the (k_len × n) constraint matrix A_ub
        A = np.zeros((k_len, n), dtype=float)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # slice of b that participates in (a∗b)[k]
            cols = np.arange(lo, hi + 1, dtype=int)
            A[row, lo:hi + 1] = a[k - cols]

        b_ub = np.full(k_len, M, dtype=float)

        # maximise Σb  <=> minimise –Σb
        res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            # fallback – return a copy of the current vector (no progress)
            return a.copy()

        b = res.x

        # verify all constraints (full convolution)
        conv_ab = convolution_fft(a, b)
        viol = np.where(conv_ab > M * (1.0 + 1e-9))[0]
        if viol.size == 0:
            return b
        # otherwise enlarge the active set and repeat
        before = len(active)
        active.update(viol.tolist())
        if len(active) == before:       # nothing new could be added – give up
            return b

    # reached cut‑limit; return best we have
    return b


# ----------------------------------------------------------------------
# Simple line‑search for mixing coefficient
# ----------------------------------------------------------------------


def best_mixture(a: np.ndarray, g: np.ndarray, cur_score: float,
                 max_iter: int = 6) -> (np.ndarray, float):
    """
    Return the convex combination (1‑t)·a + t·g (t∈[0,1]) that yields the
    smallest score.  A tiny deterministic search over a few t values is
    sufficient because the objective is smooth in t.
    """
    best_vec = a
    best_val = cur_score
    for t in np.linspace(0.0, 1.0, max_iter + 2)[1:-1]:  # skip 0 and 1 at ends
        cand = (1.0 - t) * a + t * g
        cand = cand / cand.sum()
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_vec = cand
    # also try the pure g (t=1)
    cand = g / g.sum()
    val = compute_score(cand)
    if val < best_val:
        best_val = val
        best_vec = cand
    return best_vec, best_val


# ----------------------------------------------------------------------
# Main LP‑based refinement loop
# ----------------------------------------------------------------------


def refine_by_lp(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Repeatedly improve a simplex vector by the LP‑based step described
    above.  The loop stops when the time budget is almost exhausted.
    """
    cur_score = compute_score(a)
    it = 0
    while time.time() < deadline - 5.0:
        it += 1
        conv = convolution_fft(a, a)
        M = conv.max()

        # LP step – try to push mass into “quiet” positions
        b = lp_max_sum(a, M, rel_tol=0.03, max_cuts=6)
        if b.sum() <= 1e-12:
            break

        # Normalise the LP solution to a simplex vector g
        g = b / b.sum()

        # Mix a and g (small line‑search)
        new_a, new_score = best_mixture(a, g, cur_score, max_iter=4)

        if new_score < cur_score - 1e-9:
            a = new_a
            cur_score = new_score
            print(f"[LP‑iter {it:3d}] improved score = {cur_score:.6f}")
        else:
            # no progress – shrink the tolerance a little to possibly expose
            # new active constraints; after several stalls we break.
            if it % 10 == 0:
                print(f"[LP‑iter {it:3d}] no improvement, stopping.")
                break
    return a, cur_score


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climb for polishing
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """Very cheap random neighbour search (fractional mass transfers)."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step = 0.07

    while time.time() < deadline - 0.5:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            cand[i] = 0.0
        cand = cand / cand.sum()
        val = compute_score(cand)
        if val < best_score:
            best_score = val
            best_a = cand
            step = min(step * 1.08, 0.5)
        else:
            step *= 0.96
    return best_a, best_score
