def solve_lp_max_sum(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    # Build convolution matrix
    # Use list comprehension
    rows = []
    data = []
    col_idx = []
    for j in range(n):
        # for each i (0..n-1) where f_i > 0
        # row index = i + j
        # coefficient = f[i]
        # So for each i
        start = j
        rows.extend(range(start, start+n))
        col_idx.extend([j]*n)
        data.extend(f.tolist())
    # Build sparse matrix
    C = sp.csr_matrix((data, (rows, col_idx)), shape=(2*n-1, n))
    constraints = [C @ b <= M * np.ones(2*n-1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    return b.value
```

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` objective.

Algorithm
---------
1. **Coarse stochastic optimisation** – many cheap Adam runs on
   a smooth log‑sum‑exp surrogate of the max‑convolution for a short
   length (n≈1024).  The best vector is retained.

2. **Scale‑up & fine Adam optimisation** – the best vector is up‑sampled
   (each entry is split into two halves) to length 2048.  A longer Adam
   run with a much larger temperature β (≈5000) further lowers the
   exact max‑convolution.

3. **Linear‑programming refinement** – for the current f we solve

        maximise   Σ b_j
        subject to  b_j ≥ 0 ,
                    (f * b)[k] ≤ M   for all k ,

   where M = max (f * f).  The solution b has the same ℓ₁‑norm as f
   after normalisation; we replace f ← (1‑α) f + α b̂ with α≈0.5.
   Repeating this step (≤15 times) provably reduces the maximal
   convolution value.

4. **Stochastic hill‑climb (simulated annealing)** – small random
   swaps / mass transfers are performed; improvements are always kept,
   while occasional worsening moves are accepted with a temperature
   schedule until the time budget expires.

All stages respect a 1000 s wall‑clock budget.  The result is a plain
Python list of floats (the harness expects this).

"""

import time
import numpy as np
import scipy.linalg as la
import cvxpy as cp

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score when Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators (different shapes)
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """Smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Pick one of the three generators at random."""
    r = np.random.rand()
    if r < 0.4:
        return random_start_dense(n)
    elif r < 0.7:
        return random_start_sparse(n)
    else:
        return random_start_gaussian(n)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a differentiable soft‑max surrogate of max(conv(a,a))
    using Adam with simplex projection.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of soft‑max via cross‑correlation trick
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Linear‑programming refinement (max‑convolution bound)
# ----------------------------------------------------------------------


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n convolution matrix C such that C @ b = f * b.
    Dense matrix – acceptable for n≈2000.
    """
    n = f.shape[0]
    first_col = np.concatenate([f, np.zeros(n - 1)])
    first_row = np.concatenate([f[:1], np.zeros(n - 1)])
    C = la.toeplitz(first_col, first_row)
    return C


def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve  maximise Σ b_j  s.t.  b≥0 , (f∗b) ≤ M.
    Returns the optimal b (as a numpy array) or None on failure.
    """
    n = f.shape[0]
    C = _build_convolution_matrix(f)           # dense (2n‑1)×n
    b = cp.Variable(n, nonneg=True)

    # vectorised inequality C @ b ≤ M (broadcasted)
    constraints = [C @ b <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9, verbose=False)
    except Exception as exc:
        print("LP solve exception:", exc)
        return None

    if b.value is None or prob.status not in {"optimal", "optimal_inaccurate"}:
        return None

    return np.maximum(b.value, 0.0)


def lp_refine(
    f: np.ndarray,
    time_deadline: float,
    max_iter: int = 15,
    mix: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Iteratively improve f by solving the LP described above and
    mixing the result in.  Returns the refined vector and its exact
    score.
    """
    n = f.shape[0]
    best_score = compute_score(f)

    for it in range(max_iter):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(f, f)
        M = conv.max()

        b = solve_lp_max_sum(f, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b < 1e-12:
            break

        b_norm = b / sum_b                  # put onto simplex
        f_new = (1.0 - mix) * f + mix * b_norm

        new_score = compute_score(f_new)

        # Accept if we made progress; otherwise shrink the mixing ratio
        if new_score < best_score - 1e-9:
            f = f_new
            best_score = new_score
        else:
            mix *= 0.7
            if mix < 1e-3:
                break

    return f, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(seq, time_deadline, max_steps: int = None):
    """
    Random local moves (swap or mass transfer) with annealed acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: list[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two
    adjacent entries a_i/2, a_i/2.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)   # safety net
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Respects the 1000 s wall‑clock budget and
    returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Coarse random‑restart optimisation (n≈1024)
    # ------------------------------------------------------------------
    n_small = 1024
    print(f"Coarse phase – vector length = {n_small}")

    # reserve time for later stages
    reserve_small = 200.0
    while time.time() < deadline - reserve_small:
        remaining = deadline - time.time() - reserve_small
        slice_sec = max(5.0, min(25.0, remaining * 0.3))
        slice_deadline = time.time() + slice_sec

        a0 = random_start(n_small)
        a_opt, _ = smooth_gradient_optimize(
            a0,
            time_deadline=slice_deadline,
            max_iters=120_000,
            beta_start=5.0,
            beta_end=5000.0,
            step0=1.0,
        )
        exact = compute_score(a_opt)
        if exact < best_val:
            best_val = exact
            best_seq = a_opt.tolist()
            print(f"  New best after coarse run: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Upsample to 2048 and refine with a longer Adam run
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time() > 300):
        print("\nUpsampling best candidate to length 2048")
        a_big = upsample_sequence(best_seq)   # length 2048
        # longer Adam run on the larger vector
        slice_deadline = time.time() + min(150.0, deadline - time.time() - 150.0)
        a_big_opt, _ = smooth_gradient_optimize(
            a_big,
            time_deadline=slice_deadline,
            max_iters=200_000,
            beta_start=5.0,
            beta_end=5000.0,
            step0=0.5,
        )
        best_val = compute_score(a_big_opt)
        best_seq = a_big_opt.tolist()
        print(f"After fine Adam stage: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Linear‑programming refinement (max‑convolution bound)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60:
        print("\nLP refinement")
        f = np.array(best_seq, dtype=float)
        f, score_lp = lp_refine(
            f,
            time_deadline=deadline - 30,
            max_iter=15,
            mix=0.5,
        )
        if score_lp < best_val:
            best_val = score_lp
            best_seq = f.tolist()
            print(f"  After LP: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic hill‑climb polishing (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("\nHill‑climb polishing (annealed)")
        polished_seq, polished_score = hill_climb_anneal(
            best_seq,
            time_deadline=deadline,
            max_steps=None,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"  Polished score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Log final official evaluation
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
