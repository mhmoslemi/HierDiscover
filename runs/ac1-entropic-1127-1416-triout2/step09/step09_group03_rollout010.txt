"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1. **Exploration on a small grid** – many fast random restarts at
   n≈256 with a few thousand Adam steps.  The best small shape is kept.
2. **Upscaling** – the best small vector is linearly interpolated to a
   larger size (2048 or 4096 depending on remaining time) and projected
   onto the probability simplex.
3. **Fine‑grained Adam optimisation** – a smooth log‑sum‑exp surrogate
   of the max‑convolution is minimised with Adam.  The smoothness
   parameter β is increased from 2 up to 500, so the algorithm first
   explores and then sharpens.
4. **Projected sub‑gradient polishing** – after the smooth phase a
   true sub‑gradient step (max‑entry) is performed with a decreasing
   step‑size.
5. **Stochastic hill‑climb** – cheap random mass‑transfer moves with
   simplex projection finish the search.

All phases respect the wall‑clock budget; the best sequence seen is
returned.  The code contains only top‑level functions (no closures) and
uses only ``numpy`` and ``time``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Fast for the moderate sizes used here.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def central_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i, normalised to sum‑1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1), projected onto the simplex."""
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(idx)
    return project_to_simplex(a)


def linear_ramp(n: int) -> np.ndarray:
    """Linearly increasing shape (i+1), projected onto the simplex."""
    a = np.arange(1, n + 1, dtype=float)
    return project_to_simplex(a)


def random_dirichlet(n: int) -> np.ndarray:
    """Random probability vector from a symmetric Dirichlet."""
    a = np.random.dirichlet(np.ones(n))
    return a.astype(float)


def init_sequence(n: int, method: str) -> np.ndarray:
    """Factory for the four elementary start‑shapes."""
    if method == "central_binomial":
        return central_binomial(n)
    if method == "inverse_sqrt":
        return inverse_sqrt(n)
    if method == "linear_ramp":
        return linear_ramp(n)
    if method == "random":
        return random_dirichlet(n)
    # fallback to uniform
    return np.full(n, 1.0 / n, dtype=float)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) Adam optimisation
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 2.0,
    beta_end: float = 500.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothness β (log‑linear schedule)
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # convolution a*a
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights over the entries of conv
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective:
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)          # length = n + (2n-1) - 1 = 3n-2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]      # slice of length n

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:  # occasional progress check
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to jump out of plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 400_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12  # magnitude of transferred mass

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:  # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                      # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.04, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)

    subgrad_deadline = min(deadline - 70.0, time.time() + remaining * 0.30)
    if best_seq is not None and time.time() < subgrad_deadline:
        a_sub, sub_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=subgrad_deadline,
            max_iter=350_000,
            step0=0.10,
        )
        if sub_score < best_val:
            best_val = sub_score
            best_seq = a_sub.tolist()
            print(f"[subgrad] improved score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 5: final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        polished_seq, hill_score = hill_climb_permutation(
            best_seq,
            time_deadline=hill_deadline,
            max_steps=500_000,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = polished_seq
            print(f"[hill‑climb] improved score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
