def lp_refine(f, time_deadline, max_iter=15):
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    while time.time() < time_deadline and iteration < max_iter:
        # compute M
        conv_f = convolution_fft(f,f)
        M = conv_f.max()
        # build constraint matrix A_ub for conv(f,g) <= M
        rows = []
        cols = []
        data = []
        # for each j variable from 0..n-1
        for j in range(n):
            # rows index range j .. j+n-1
            rows.extend(range(j, j+n))
            cols.extend([j]*n)
            data.extend(f)   # f[t] for t in 0..n-1
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        ub = np.full(2*n-1, M)
        # solve LP: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        scale = 1.0 / sum_g   # because sum(f) = 1
        g_scaled = g * scale
        # combine with f: try mixing fractions and pick best
        # simple line search over alpha values
        for alpha in [0.2, 0.5, 0.8, 1.0]:
            f_new = (1-alpha)*f + alpha*g_scaled
            # need to project onto simplex (though sum should stay 1)
            f_new = project_to_simplex(f_new)
            score = compute_score(f_new)
            if score < best_score:
                best_score = score
                best_f = f_new.copy()
        # update f to best_f (or g_scaled?) to continue refinement
        f = best_f.copy()
        iteration += 1
    return best_f, best_score
```

```python
"""
Propose a step‑function (non‑negative vector) minimizing `evaluate_sequence`
by a deterministic linear‑program refinement of the max‑convolution.
The method:
  1. initialise a feasible probability vector (sum=1) – either a known
     fallback or a random Dirichlet start.
  2. repeatedly solve the LP
        max Σ g_i   s.t. conv(f,g) ≤ M   , g ≥ 0
     where M = max(conv(f,f)).  The LP gives a direction g that does not
     increase the current worst convolution value.
  3. mix the current vector f with the scaled solution g and keep the best
     mixture (projected on the simplex).
  4. after a few LP steps optionally apply a cheap stochastic hill‑climb
     to polish the result.
All work is performed under the 1000 s wall‑clock budget.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective under Σ a = 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Simple random initialisers (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def random_start_dense(n: int) -> np.ndarray:
    a = np.random.rand(n)
    return a / a.sum()

def random_start_sparse(n: int) -> np.ndarray:
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)

def random_start_gaussian(n: int) -> np.ndarray:
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    return a / a.sum()

def random_start(n: int) -> np.ndarray:
    r = np.random.rand()
    if r < 0.4:
        return random_start_dense(n)
    elif r < 0.7:
        return random_start_sparse(n)
    else:
        return random_start_gaussian(n)

# ----------------------------------------------------------------------
# Upsampling helper (double length by splitting each entry in half)
# ----------------------------------------------------------------------
def upsample_sequence(seq):
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    return project_to_simplex(up)

# ----------------------------------------------------------------------
# Linear‑program refinement step
# ----------------------------------------------------------------------
def build_conv_matrix(f):
    """Build sparse matrix A such that (A @ g) = conv(f,g) (full length)."""
    n = f.shape[0]
    # each diagonal offset 0..n-1 contains the full vector f
    # construct a (n, n) array where each row = f, then give to diags
    data = np.tile(f, (n, 1))          # shape (n, n)
    offsets = np.arange(n, dtype=int)  # 0,1,...,n-1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A

def lp_refine(f, deadline, max_iters=12, alphas=(0.2, 0.5, 0.8, 1.0)):
    """
    Repeatedly solve the LP:
        max Σ g_i        s.t.   conv(f,g) ≤ M    ,   g ≥ 0
    where M = max(conv(f,f)).
    The feasible direction g is mixed with f; the best mixture is kept.
    Returns the best vector and its exact score.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = compute_score(best_f)

    iteration = 0
    while iteration < max_iters and time.time() < deadline:
        conv_f = convolution_fft(f, f)
        M = conv_f.max()

        # build constraint matrix A such that (A @ g) = conv(f,g)
        A = build_conv_matrix(f)

        # right‑hand side of all constraints
        ub = np.full(2 * n - 1, M, dtype=float)

        # objective: maximise sum(g)  → minimise -sum(g)
        c = -np.ones(n, dtype=float)

        # bounds of variables: g_i ≥ 0
        bounds = [(0, None)] * n

        # solve the linear program
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=ub,
                                    bounds=bounds,
                                    method='highs',
                                    options={'presolve': True})
        if not res.success:
            # LP failed – break out and fall back to sub‑gradient steps
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # scale g to the same total mass as f (which is 1)
        g_scaled = g / sum_g

        # try a few convex combinations of f and g_scaled,
        # keep the best exact score
        improved = False
        for alpha in alphas:
            candidate = (1 - alpha) * f + alpha * g_scaled
            candidate = project_to_simplex(candidate)  # safety net
            score = compute_score(candidate)
            if score < best_score:
                best_score = score
                best_f = candidate.copy()
                improved = True

        # if any mixture improved, continue from that point,
        # otherwise keep the current f (no progress)
        if improved:
            f = best_f.copy()
        else:
            # no improvement – stop early
            break

        iteration += 1

    return best_f, best_score

# ----------------------------------------------------------------------
# Stochastic hill‑climb (annealed acceptance) – used as a final polish
# ----------------------------------------------------------------------
def hill_climb_anneal(seq, time_deadline, max_steps=None):
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score

"""
Propose a step‑function (non‑negative vector) minimizing `evaluate_sequence`
by a deterministic linear‑program refinement of the max‑convolution.

Method
------
1. Initialise a probability vector (sum = 1).  Use the previously found
   construction if available, otherwise a random start.
2. Repeatedly solve the LP

        max Σ g_i   s.t.   conv(f,g) ≤ M ,  g ≥ 0

   where `M = max(conv(f,f))`.  The LP yields a feasible direction `g`
   that does not increase the current worst convolution value.
3. Mix the current vector `f` with the scaled solution `g` (projected onto
   the simplex) and keep the best mixture.
4. Optionally up‑sample to double the length and finish with a cheap
   simulated‑annealing hill‑climb for polishing.

All work respects the 1000 s wall‑clock budget and returns the best
vector found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the constraint Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators (different shapes)
# ----------------------------------------------------------------------
def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    return a / a.sum()


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """A smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    return a / a.sum()


def random_start(n: int) -> np.ndarray:
    """Pick one of the three generators at random."""
    r = np.random.rand()
    if r < 0.4:
        return random_start_dense(n)
    elif r < 0.7:
        return random_start_sparse(n)
    else:
        return random_start_gaussian(n)


# ----------------------------------------------------------------------
# Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------
def upsample_sequence(seq: list[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two adjacent
    entries a_i/2, a_i/2.  The total mass is unchanged.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    return project_to_simplex(up)


# ----------------------------------------------------------------------
# Linear‑program construction
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A such that (A @ g) = conv(f, g) (full length).
    For offset k we have   (A @ g)[k] = Σ_i f_i * g_{k-i}.
    """
    n = f.shape[0]
    # each diagonal offset 0..n‑1 contains the full vector f
    data = np.tile(f, (n, 1))          # shape (n, n)
    offsets = np.arange(n, dtype=int)  # 0,1,…,n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def lp_refine(f: np.ndarray,
              deadline: float,
              max_iters: int = 12,
              alphas: tuple[float, ...] = (0.05, 0.2, 0.5, 0.8, 1.0)
              ) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP
        max Σ g_i   s.t.   conv(f,g) ≤ M , g ≥ 0,
    where M = max(conv(f,f)).
    The feasible direction g is mixed with the current vector f;
    the best mixture (projected onto the simplex) is kept.
    Returns the best vector and its exact score.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = compute_score(best_f)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv_f = convolution_fft(f, f)
        M = conv_f.max()

        # constraint matrix for conv(f,g) ≤ M
        A = build_conv_matrix(f)
        ub = np.full(2 * n - 1, M, dtype=float)

        # maximise Σ g_i  → minimise -Σ g_i
        c = -np.ones(n, dtype=float)

        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=ub,
                                     bounds=bounds,
                                     method='highs',
                                     options={'presolve': True})
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # scale g to the same total mass as f (which is 1)
        g_scaled = g / sum_g

        improved = False
        for alpha in alphas:
            candidate = (1 - alpha) * f + alpha * g_scaled
            candidate = project_to_simplex(candidate)
            score = compute_score(candidate)
            if score < best_score:
                best_score = score
                best_f = candidate.copy()
                improved = True

        if improved:
            f = best_f.copy()
        else:
            # no improvement – stop early
            break

        it += 1

    return best_f, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------
def hill_climb_anneal(seq,
                      time_deadline,
                      max_steps: int | None = None):
    """
    Random local modifications (swap or mass transfer) with a temperature
    schedule.  Improving moves are always accepted; worsening moves are
    accepted with probability exp(-Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score
