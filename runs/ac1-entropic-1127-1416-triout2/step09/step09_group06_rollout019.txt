"""
search for a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines several ideas:

1.   Deterministic starting shapes (central‑binomial and a short
     power‑law family a_i ∝ (i+β)^‑p) are screened by a fast FFT‑based
     exact score.

2.   A small linear‑programming (LP) refinement step (inspired by the
     classic “max‑convolution” LP) repeatedly enlarges the feasible
     mass under the current bound M = max(conv(f,f)).  Mixing the
     obtained direction back into the current vector strictly reduces
     the bound.

3.   A β‑annealed L‑BFGS‑B optimisation of the smooth log‑sum‑exp
     surrogate (the logits are kept on the probability simplex via a
     soft‑max re‑parameterisation).

4.   A projected sub‑gradient descent that works on the exact
     max‑convolution objective.

5.   A very cheap stochastic hill‑climb for the remaining time.

All steps respect a global 1000 s wall‑clock budget.  The final
sequence is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast for large vectors)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact value of the objective for a vector on the simplex (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities (projection and soft‑max)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (always sums to 1)."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) + gradient w.r.t. logits
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))
    and g is the gradient w.r.t. the logits.
    The parametrisation a = softmax(logits) guarantees a ≥ 0 and ∑a = 1.
    """
    a = softmax_logits(logits)

    conv = convolution_fft(a, a)
    maxc = conv.max()

    # smooth max
    exp_weights = np.exp(beta * (conv - maxc))
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient w.r.t. a
    w = exp_weights / Z                     # soft‑max over convolution entries
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)    # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # chain rule back to logits
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# L‑BFGS‑B continuation (β‑annealing)
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 9,
                       maxiter_per_step: int = 150):
    """
    Anneal the temperature β while optimising the smooth surrogate.
    Returns the best simplex vector and its exact score.
    """
    logits = np.log(start_a + 1e-20)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # allocate a generous per‑step budget
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(20, int(time_left * 30)))   # ≈30 iters per second

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f} (step {step})] new best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 400_000,
                     step0: float = 0.07):
    """Simple projected sub‑gradient method."""
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Deterministic start – inverse‑square‑root (central‑binomial) shape
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape a_i = C(2i,i)/4^i, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step”)
# ----------------------------------------------------------------------


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max ∑b_j  subject to  b_j ≥ 0  and  (f * b)_k ≤ M  for all k.
    Returns the optimal b (not yet normalised).
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Toeplitz matrix that implements (f * b)_k = Σ_i f_i * b_{k-i}
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([[f[0]], np.zeros(n - 1)])
    A = toeplitz(col, row)                     # shape (m, n)

    # Linear program: minimise -1·b  (i.e. maximise Σb)
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None) for _ in range(n)]
    res = linprog(c,
                  A_ub=A,
                  b_ub=np.full(m, M, dtype=float),
                  bounds=bounds,
                  method='highs')
    if not res.success:
        # If the LP fails we simply return the current f (no direction)
        return np.zeros_like(f)
    return res.x


def lp_iterate_reduce(f: np.ndarray,
                      max_iters: int,
                      deadline: float):
    """
    Repeatedly apply the LP‑step and mix the obtained direction back into f.
    Returns an improved simplex vector and its max‑convolution value.
    """
    best_f = f.copy()
    best_M = np.max(convolution_fft(best_f, best_f))

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        b = solve_lp_for_g(best_f, best_M)
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        g = b / sum_b                     # normalise to a probability vector

        # try a convex combination; start with a moderate step size
        t = 0.5
        improved = False
        while t > 1e-4:
            cand = (1.0 - t) * best_f + t * g
            M_cand = np.max(convolution_fft(cand, cand))
            if M_cand < best_M - 1e-12:   # strict improvement
                best_f = cand
                best_M = M_cand
                improved = True
                print(f"[LP‑iter {it}] t={t:.3f} → new M = {best_M:.6f}")
                break
            t *= 0.5
        if not improved:
            # no improvement possible with this direction
            break

    return best_f, best_M

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock budget and returns the
    best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previously found construction)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose a working length
    # -----------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 700:
        n = 16384
    elif time_left > 400:
        n = 8192
    else:
        n = 4096

    idx = np.arange(n, dtype=float)

    # -----------------------------------------------------------------
    # 2. Quick grid search over a simple 2‑parameter family
    # -----------------------------------------------------------------
    best_grid_score = float("inf")
    best_grid_seq = None

    # p in (0.45, 0.80), β in a modest interval avoiding the singularity
    p_vals = np.linspace(0.45, 0.80, 24)
    beta_vals = np.concatenate([
        np.linspace(0.1, 0.5, 5),
        np.linspace(0.6, 3.0, 10),
        np.linspace(4.0, 12.0, 5)
    ])

    for p in p_vals:
        for beta in beta_vals:
            if time.time() >= deadline - 30:
                break
            a = (idx + beta) ** (-p)
            if np.isinf(a).any() or np.isnan(a).any():
                continue
            a /= a.sum()
            score = compute_score(a)
            if score < best_grid_score:
                best_grid_score = score
                best_grid_seq = a.copy()
                print(f"[grid] p={p:.3f}, β={beta:.2f} → score={score:.6f}")

    # -----------------------------------------------------------------
    # 3. Deterministic central‑binomial start
    # -----------------------------------------------------------------
    a_central = central_binomial_sequence(n)

    # pick the better of the two deterministic starts
    start_a = a_central
    start_score = compute_score(a_central)
    if best_grid_seq is not None and best_grid_score < start_score:
        start_a = best_grid_seq
        start_score = best_grid_score
        print(f"Chosen grid start with score {start_score:.6f}")

    best_seq = start_a.tolist()
    best_val = start_score

    # -----------------------------------------------------------------
    # 4. LP‑based refinement (few quick iterations)
    # -----------------------------------------------------------------
    lp_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.15)
    f_lp, M_lp = lp_iterate_reduce(np.array(best_seq, dtype=float),
                                   max_iters=8,
                                   deadline=lp_deadline)
    lp_score = 2.0 * n * M_lp
    if lp_score < best_val:
        best_val = lp_score
        best_seq = f_lp.tolist()
        print(f"After LP refinement: score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Smooth optimisation via L‑BFGS‑B with β‑continuation
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200:
        lbfgs_deadline = time.time() + remaining * 0.45
        beta_start = 5.0
        beta_end = 400.0
        schedule_steps = 12 if remaining > 400 else 9
        maxiter_per_step = 250 if remaining > 600 else 150

        a_smooth, smooth_score = lbfgs_continuation(
            np.array(best_seq, dtype=float),
            beta_start,
            beta_end,
            lbfgs_deadline,
            schedule_steps=schedule_steps,
            maxiter_per_step=maxiter_per_step,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"After smooth L‑BFGS: best = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Exact max‑objective polishing (projected sub‑gradient)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120:
        subgrad_deadline = time.time() + remaining * 0.40
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            subgrad_deadline,
            max_iter=800_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑gradient polishing improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 7. Very cheap stochastic hill‑climb (last few seconds)
    # -----------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(best_a)
        step_scale = 0.09

        while time.time() < deadline:
            cand = best_a.copy()
            if np.random.rand() < 0.5:               # swap two entries
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]
            else:                                     # fractional transfer
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = step_scale * best_a[i]
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0:
                        cand[i] = 0.0
            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)

            if cand_score < best_score:
                best_score = cand_score
                best_a = cand
                step_scale = min(step_scale * 1.08, 0.5)
                print(f"Hill‑climb improved to {best_score:.6f}")
            else:
                step_scale *= 0.96

        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.tolist()
            print(f"Hill‑climb final score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 8. Final evaluation and return
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
