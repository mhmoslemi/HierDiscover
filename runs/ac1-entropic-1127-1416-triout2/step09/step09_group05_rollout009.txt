"""
Propose a non‑negative sequence minimizing `evaluate_sequence`.

Algorithm
---------
1. Choose a sequence length (8192 if time permits, otherwise 4096/2048).
2. Exhaustively scan a few low‑dimensional families
   – exponential, power‑law and a simple mixture of two exponentials –
   and keep the best vector (exact objective).
3. Run a long exponentiated‑gradient descent on a smooth log‑sum‑exp
   approximation of the max‑convolution.
   * β (the smoothing parameter) is annealed exponentially from 0.5 to a
     very large value (≈2·10⁴) – this makes the smooth objective approach the
     true max.
   * Step size η decays as η₀ / √t (η₀≈0.6) and the iterate is projected
     onto the probability simplex after each multiplicative update.
   * The loop stops automatically when the overall time budget is exhausted.
4. Finish with a cheap greedy hill‑climb that repeatedly transfers a small
   packet of mass between two positions (or swaps two entries) and keeps the
   move only if the exact objective improves.  The packet size is adapted
   during the search.
5. Return the best sequence found (as a plain list of floats).

All phases respect the 1000 s wall‑clock limit; intermediate scores are
printed for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic full convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a vector with sum = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def double_exponential_shape(
    n: int, lam1: float, lam2: float, mix: float
) -> np.ndarray:
    """Weighted mixture of two exponentials, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = mix * np.exp(-lam1 * i) + (1.0 - mix) * np.exp(-lam2 * i)
    return w / w.sum()


# ----------------------------------------------------------------------
# Family scan
# ----------------------------------------------------------------------


def scan_families(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """Search simple one‑parameter families and a 2‑parameter mixture."""
    best_a = None
    best_sc = float("inf")

    # ----- exponential -------------------------------------------------
    lam_grid = np.concatenate(
        (np.array([0.0]), np.logspace(-5, -2, 150))
    )
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # ----- power law ---------------------------------------------------
    p_grid = np.linspace(0.0, 8.0, 150)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # ----- mixture of two exponentials ---------------------------------
    lam1_vals = np.logspace(-5, -3, 5)   # slower decay
    lam2_vals = np.logspace(-3, -1, 5)   # faster decay
    mix_vals = [0.25, 0.5, 0.75]
    for lam1 in lam1_vals:
        for lam2 in lam2_vals:
            if lam2 <= lam1:
                continue
            for mix in mix_vals:
                if time.time() >= deadline:
                    break
                a = double_exponential_shape(n, lam1, lam2, mix)
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc, best_a = sc, a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 2_000_000,
    beta_start: float = 0.5,
    beta_end: float = 20_000.0,
    eta0: float = 0.6,
    eps: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth log‑sum‑exp surrogate of the max‑convolution
    using a KL‑mirror (exponentiated‑gradient) update.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    eval_interval = 500  # check exact objective less often

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # annealed smoothing parameter β (exponential schedule)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution and softmax weights of the smooth max
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))          # stabilised: max term → 1
        w = expw / expw.sum()

        # gradient of the smoothed objective (KL‑mirror)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # KL‑mirror multiplicative update
        eta = eta0 / np.sqrt(it)
        a = np.maximum(a * np.exp(-eta * grad), eps)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % eval_interval == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final exact evaluation
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk:
        – with prob 0.5 swap two entries,
        – otherwise transfer a small packet of mass from one index to another.
    Only moves that improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12  # initial packet size (fraction of the source entry)

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # safety projection (keeps sum = 1 and non‑negativity)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‐function that minimises `evaluate_sequence`.
    The routine respects a 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a workable vector length
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 850:
        n = 8192
    elif remaining > 550:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score, best_seq = fb, fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Scan simple one‑parameter families (including a mixture)
    # -------------------------------------------------
    a_scan, sc_scan = scan_families(n, deadline - 30.0)
    if a_scan is not None and sc_scan < best_score:
        best_score, best_seq = sc_scan, a_scan.tolist()
        print(f"[{time.time():.1f}] after family scan best = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Long exponentiated‑gradient optimisation
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting smooth exponentiated‑gradient")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=2_000_000,
            beta_start=0.5,
            beta_end=20_000.0,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

        # --- optional second refinement with a tighter smoothing schedule ---
        if time.time() < deadline - 60.0:
            print(f"[{time.time():.1f}] second refinement (higher β)")
            a_opt2, sc_opt2 = exp_smooth_optimize(
                np.array(best_seq, dtype=float),
                deadline - 40.0,
                max_iters=800_000,
                beta_start=5.0,
                beta_end=40_000.0,
                eta0=0.5,
            )
            if sc_opt2 < best_score:
                best_score, best_seq = sc_opt2, a_opt2.tolist()
                print(f"[{time.time():.1f}] after refinement best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Final hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=2_500_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard in case nothing succeeded
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats.
    return best_seq
