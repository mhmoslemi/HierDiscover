def adam_optimize(start_a, deadline, max_iters=50000, beta_start=0.5, beta_end=500.0, lr=0.05):
    # start with u = log(start_a+eps) maybe
    u = np.log(start_a + 1e-12)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    eps = 1e-8
    best_a = start_a.copy()
    best_sc = compute_score(start_a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        a = np.exp(u)
        a_sum = a.sum()
        a /= a_sum
        # compute conv a*a
        conv = convolution_fft(a, a)
        # beta schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        # compute w
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()
        # compute grad_a as 2* conv_rev w
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n-1:n-1+n]
        # gradient wrt u:
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u**2)
        m_hat = m / (1 - 0.9**it)
        v_hat = v / (1 - 0.999**it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # occasional exact evaluation every maybe 200 iter
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    # final check
    sc = compute_score(best_a)
    if sc < best_sc:
        best_sc = sc
        best_a = best_a
    return best_a, best_sc

def annealed_hill_climb(start_a, deadline, init_temp=0.001, final_temp=1e-6, decay=0.9995):
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)
    best_a = a.copy()
    best_sc = 2.0 * n * conv.max()
    cur_sc = best_sc
    temp = init_temp
    step_scale = 0.1
    while time.time() < deadline:
        # choose move: similar to fast_hill_climb
        # random pick src and dst
        src = np.random.randint(n)
        if a[src] == 0.0:
            continue
        # small delta
        delta = step_scale * a[src]
        dst = np.random.randint(n-1)
        if dst >= src:
            dst += 1
        # compute candidate conv using incremental formula
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2*src] += delta*delta
        conv_cand[2*dst] += delta*delta
        conv_cand[src + dst] -= 2.0 * delta*delta
        cand_max = conv_cand.max()
        cand_sc = 2.0 * n * cand_max
        # acceptance
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            # compute acceptance probability
            prob = np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            # update
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a.copy()
            # maybe increase step scale
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            step_scale = max(step_scale * 0.99, 1e-5)
        # cool temperature
        temp *= decay
    return best_a, best_sc
```

"""
A fast multi‑phase optimizer for the step‑function score.

Phases
------
1. Analytic α‑grid (extended) + random Dirichlet starts.
   Keep the few best normalised vectors.

2. Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The temperature β is exponentially annealed (β_start → β_end).
   The parameters are the log‑probabilities (softmax) of the vector, which
   guarantees non‑negativity and unit‐sum.  Adam’s adaptive step‑size
   typically gives a lower final max‑convolution than the plain
   exponentiated‑gradient used previously.

3. Simulated‑annealing hill‑climb with an O(n) incremental convolution update.
   Only strictly improving moves are always accepted; uphill moves are
   accepted with probability exp(−Δ/Temp) while the temperature decays.
   This polish step can exploit very small local improvements that the
   first‑order method cannot see.

All phases respect the 1000 s wall‑clock limit and keep the best sequence
encountered.  The function returns a plain list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) assuming sum(a)=1.
    This matches the value returned by `evaluate_sequence` for a normalised input.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic family a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return the normalised vector for a given α."""
    i = np.arange(n, dtype=float)
    # log product to avoid overflow/underflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation on smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  max_iters: int = 300_000,
                  beta_start: float = 0.5,
                  beta_end: float = 600.0,
                  lr: float = 0.05,
                  eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth surrogate
        f_β(a) = (1/β)·log Σ exp(β·(a⊛a))
    on the probability simplex via the softmax parametrisation a = softmax(u).
    Returns the best exact score seen.
    """
    n = start_a.shape[0]
    # initialise log‑probabilities (softmax variables)
    u = np.log(start_a + eps)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta = beta_start
    best_a = start_a.copy()
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- get probability vector -----
        # shift for numerical stability
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()

        # ----- convolution -----
        conv = convolution_fft(a, a)

        # ----- temperature schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- smooth‑max weights -----
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))          # stabilised exponentials
        w = expw / expw.sum()

        # ----- gradient ∂f/∂a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ----- chain rule to get ∂f/∂u (softmax Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # ----- Adam update -----
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u ** 2)
        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional exact evaluation -----
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    # final check
    sc = compute_score(best_a)
    if sc < best_sc:
        best_sc = sc
        best_a = best_a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill climb with incremental convolution updates
# ----------------------------------------------------------------------


def annealed_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        init_temp: float = 0.02,
                        final_temp: float = 1e-7,
                        decay: float = 0.9995,
                        inc_factor: float = 1.02,
                        dec_factor: float = 0.98,
                        recompute_interval: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts occasional uphill moves.
    Uses an O(n) incremental update of the convolution.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)               # exact convolution
    best_a = a.copy()
    best_sc = 2.0 * n * float(conv.max())
    cur_sc = best_sc
    cur_temp = init_temp
    step_scale = 0.12
    steps = 0

    while time.time() < deadline:
        # ----- pick a move -----
        src = np.random.randint(n)
        if a[src] == 0.0:
            continue
        # take a small packet from src
        delta = step_scale * a[src]                # 0 < delta ≤ a[src]
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1                                 # guarantee dst != src

        # ----- incremental convolution update -----
        conv_cand = conv.copy()
        # linear part: –2·δ·shift_src + 2·δ·shift_dst
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic corrections
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_sc = 2.0 * n * cand_max

        # ----- acceptance test (simulated annealing) -----
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            # probability to accept a worsening move
            prob = np.exp(-(cand_sc - cur_sc) / max(cur_temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # perform the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a.copy()
            # slowly increase step size after success
            step_scale = min(step_scale * inc_factor, 0.5)
        else:
            # shrink step size after a rejection
            step_scale = max(step_scale * dec_factor, 1e-5)

        # temperature cooling
        cur_temp = max(cur_temp * decay, final_temp)

        steps += 1
        if steps % recompute_interval == 0:
            # periodic full recomputation to avoid drift
            conv = convolution_fft(a, a)
    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase search for a low‑score step‑function.
    Returns a plain list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a length; larger lengths give a tighter theoretical bound.
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 850:
        n = 8192
    elif remaining > 600:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic α‑grid (extended) + random Dirichlet starts
    # -------------------------------------------------
    candidates = []

    # extended α grid
    alphas = np.concatenate([
        np.linspace(-2.0, 0.0, 41),   # centre‑heavy
        np.linspace(0.0, 5.0, 101)    # edge‑heavy
    ])
    for alpha in alphas:
        if time.time() >= deadline - 120:       # keep some budget for later phases
            break
        a = alpha_start(n, alpha)
        candidates.append(a)

    # random Dirichlet draws (moderate concentration)
    for _ in range(12):
        if time.time() >= deadline - 120:
            break
        # concentration 0.5 gives a fairly spread random simplex point
        conc = np.full(n, 0.5)
        a = np.random.gamma(conc, 1.0)
        a /= a.sum()
        candidates.append(a)

    # uniform vector
    candidates.append(np.full(n, 1.0 / n))

    # evaluate candidates and keep the five with the lowest exact score
    scores = []
    for a in candidates:
        sc = compute_score(a)
        scores.append((sc, a))
    scores.sort(key=lambda x: x[0])
    top_k = 5
    best_ones = scores[:top_k]

    # initialise the global best with the best among the top‑k
    best_sc, best_a = best_ones[0]
    best_seq = best_a.tolist()
    best_score = best_sc
    print(f"[{time.time():.1f}] best after initial sampling = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Adam optimisation on the smooth surrogate
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200:
        print(f"[{time.time():.1f}] starting Adam optimisation")
        a_start = np.array(best_seq, dtype=float)
        a_opt, sc_opt = adam_optimize(
            a_start,
            deadline - 150.0,
            max_iters=250_000,
            beta_start=0.5,
            beta_end=600.0,
            lr=0.05,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Simulated‑annealing hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print(f"[{time.time():.1f}] hill‑climb polish")
        a_start = np.array(best_seq, dtype=float)
        a_final, sc_final = annealed_hill_climb(
            a_start,
            deadline,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: fall back to a short uniform vector if search failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
