def adam_opt(start_a, deadline, max_iters=300_000, beta_start=5.0, beta_end=5000.0, lr=0.018):
    n = start_a.size
    eps = 1e-12
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute beta
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac
        a = softmax(z) # sum 1
        # compute conv
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # w (softmax of conv)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()
        # gradient wrt a: conv_rev = convolution_fft(a[::-1], w)
        conv_rev = convolution_fft(a[::-1], w)
        grad_a = 2.0 * conv_rev[n-1:n-1+n]
        # gradient wrt z
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a
        # adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if it % 2000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_tmp = softmax(z)
    sc = compute_score(a_tmp)
    if sc < best_score:
        best_score = sc; best_a = a_tmp.copy()
    return best_a, best_score

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s)
--------------------
1. **Problem normalisation** – we enforce Σ a_i = 1, therefore the
   objective reduces to 2·n·max(a∗a).  All optimisation is performed on the
   probability simplex.
2. **Small dimension optimisation** – a short length (n≈2048) lets the
   optimiser perform many thousands of iterations while still exploring
   a rich space of shapes.
3. **Initial candidates** – a cheap grid of analytic families
   (symmetric‑product, left‑heavy power‑law, exponential, Gaussian,
   sinusoidal) and many random Dirichlet draws are evaluated; the best of
   them seeds the iterative stage.
4. **Smooth Adam optimisation** – we minimise the log‑sum‑exp surrogate
   f_β(a)= (1/β)·log Σ_k exp(β·(a∗a)_k) using the soft‑max
   parametrisation a = softmax(z).  β is increased exponentially from 5
   to 5000, the learning‑rate is 0.018, and the Adam state is kept for
   up to 300 000 iterations (or until the local time budget expires).
5. **Mirror (entropy) sub‑gradient descent** – starting from the best
   Adam result we run multiplicative updates on the true max‑objective
   with step‑size η_t = 0.6 / √t for many thousands of steps.
6. **Final stochastic hill‑climb** – a lightweight simulated‑annealing
   walk on the simplex (mass transfers, random swaps) with a slowly
   decreasing temperature refines the solution.
7. **Symmetrisation test** – many optimal vectors are (almost) symmetric;
   we therefore test the simple average with their reverse and keep the
   better one.
8. The function respects the 1000 s wall‑clock limit, prints intermediate
   scores, and finally returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when the sum of ``a`` is positive.
    The vector ``a`` does **not** need to be normalised – the function
    automatically incorporates the scaling.
    """
    total = a.sum()
    if total < 0.01:
        return float("inf")
    conv = convolution_fft(a, a)
    maxc = conv.max()
    n = a.shape[0]
    return float(2.0 * n * maxc / (total * total))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) ."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) ."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) ."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(- (i‑c)² / (2σ²) ), centred at c = (n‑1)/2."""
    c = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((i - c) / sigma) ** 2)
    a = w / w.sum()
    return a


def sinusoidal_power_family(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n) ^ power .  Positive and symmetric."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random starters
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int) -> np.ndarray:
    """Draw a dense random probability vector from a Dirichlet(1,…,1)."""
    a = np.random.exponential(1.0, size=n)
    a /= a.sum()
    return a


def random_sparse_start(n: int, sparsity: float = None) -> np.ndarray:
    """Sparse uniform probability vector (~5 % non‑zeros)."""
    if sparsity is None:
        sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) via Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 5_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially.
    """
    eps = 1e-12
    n = start_a.shape[0]

    # initialise logits from the start probability vector
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential)
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # current a (always on the simplex)
        a = softmax(z)

        # smoothed weights w_k ∝ exp(β·conv_k)
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of f_β w.r.t. a (see derivation in the analysis)
        conv_rev = convolution_fft(a[::-1], w)   # length 2n‑1
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 3000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_tmp = softmax(z)
    sc = compute_score(a_tmp)
    if sc < best_score:
        best_score = sc
        best_a = a_tmp.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.6,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for the true
    objective 2·n·max(a*a).  The step size is η_t = step0 / √t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (active terms only)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with a tiny simulated‑annealing component
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
        – with prob 0.5 swap two entries,
        – otherwise move a fraction of mass from one index to another.
    Very low temperature allows occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – start small, linearly cool
    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # simplex normalisation
        cand /= cand.sum()

        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)   # slightly enlarge moves
        else:
            step_scale *= 0.97                               # shrink after rejection

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
