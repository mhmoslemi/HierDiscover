def piecewise_two_level_optimize(n, max_K_frac=0.15, num_K=30):
    # Generate candidate K values from 1 to int(max_K_frac*n)
    max_K = max(1, int(max_K_frac * n))
    K_vals = np.unique(np.round(np.geomspace(1, max_K, num=num_K)).astype(int))
    best_score = float('inf')
    best_vec = None
    for K in K_vals:
        # compute edge indicator
        e = np.zeros(n, dtype=np.float64)
        e[:K] = 1.0
        e[-K:] = 1.0
        m_vec = np.ones(n, dtype=np.float64) - e
        # compute convs
        convEE = convolution_fft(e, e)
        convEM = convolution_fft(e, m_vec)
        convMM = convolution_fft(m_vec, m_vec)
        # coefficients
        A_max = 1.0 / (2*K)
        # Precompute coeff arrays
        b0 = 1.0 / (n - 2*K)
        b1 = 2*K / (n - 2*K)
        # Actually B = b0 - b1*A
        coeff2 = convEE - 2*b1*convEM + (b1**2)*convMM
        coeff1 = 2*b0*convEM - 2*b0*b1*convMM
        coeff0 = (b0**2) * convMM

        # objective function for given A (value of edge weight)
        def maxconv_of_A(A):
            conv_val = coeff2 * (A**2) + coeff1 * A + coeff0
            return np.max(conv_val)

        # minimize maxconv using bounded scalar optimizer
        # using golden-section or minimize_scalar
        from scipy.optimize import minimize_scalar
        res = minimize_scalar(maxconv_of_A, bounds=(0, A_max), method='bounded', options={'xatol': 1e-6})
        A_opt = res.x
        # compute B
        B_opt = b0 - b1 * A_opt
        # construct vector
        a = np.full(n, B_opt, dtype=np.float64)
        a[:K] = A_opt
        a[-K:] = A_opt
        # normalize to sum=1 (should already sum to exactly 1 by construction)
        # but due to rounding may deviate, fix
        a = normalize(a)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            print(f"[{time.time():.1f}] piecewise K={K} best score {best_score:.6f}")
    return best_vec, best_score

"""
Propose a step‑function sequence that minimises

    2·n·max(convolve(a,a)) / (sum(a)**2)

under the non‑negative simplex constraint.

Algorithm overview
------------------
1.  Choose a problem size (2048 or 4096).  For this length we
    * scan an enriched family of analytic candidates:
        – symmetric product a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
        – exponential “U” a_i ∝ exp(‑γ·min(i,n‑1‑i))
        – *mixture* of the two families (λ∈[0,1]).
    * evaluate a cheap two‑level piecewise‑constant family
      (high mass on the two edge blocks, low mass in the centre)
      and find its best parameters by a tiny scalar optimisation.
    * throw in a few random monotone / U‑shaped starts.
    The best candidate from this fast “global” phase is kept.

2.  Refine the best candidate with an Adam optimiser on the
    smooth‑max surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using the soft‑max parametrisation a = softmax(z).  The
    smoothing parameter β is ramped from 5 to 800.  At most 150 k
    iterations are performed – this fits comfortably into the time
    budget while still allowing substantial improvement.

3.  Projected sub‑gradient descent on the true max‑objective
    (step‑size ∝ 1/√t) is run for a few hundred thousand cheap
    iterations.  This often squeezes the peak a little further.

4.  Symmetrise the result (a ← (a+a[::-1])/2) – the optimum is
    frequently symmetric.

5.  A short simulated‑annealing hill‑climb (swap / mass‑transfer
    moves) finishes the search.

All phases respect the global wall‑clock budget (default 1000 s).
The function returns a plain Python list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize_scalar

# -------------------------------------------------
# Basic utilities
# -------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
    2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to sum = 1 (fallback to uniform)."""
    s = v.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# -------------------------------------------------
# Analytic families
# -------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


# -------------------------------------------------
# Random initialisers
# -------------------------------------------------


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Sample a random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# -------------------------------------------------
# Two‑level piecewise‑constant family (new)
# -------------------------------------------------


def piecewise_two_level_optimize(n: int,
                                 max_frac: float = 0.15,
                                 nK: int = 25) -> tuple[np.ndarray, float]:
    """
    Search the family
          a = A·edge_K + B·middle_K
    where `edge_K` is 1 on the first/last K entries and 0 otherwise,
    `middle_K` is the complement.  The optimum (A,B) for each K is found
    by a bounded scalar optimisation of the maximal convolution entry.
    Returns the best vector and its score.
    """
    best_score = float("inf")
    best_vec = None

    max_K = max(1, int(max_frac * n))
    # geometric spacing gives good coverage of small and medium K
    K_vals = np.unique(
        np.round(np.geomspace(1, max_K, num=nK)).astype(int)
    )
    for K in K_vals:
        # build indicator vectors
        edge = np.zeros(n, dtype=np.float64)
        edge[:K] = 1.0
        edge[-K:] = 1.0
        middle = np.ones(n, dtype=np.float64) - edge

        # pre‑compute the three convolutions (they are just counts)
        convEE = convolution_fft(edge, edge)
        convEM = convolution_fft(edge, middle)
        convMM = convolution_fft(middle, middle)

        # coefficients for the quadratic expression of the convolution
        A_max = 1.0 / (2 * K)                     # from the constraint A ≤ 1/(2K)
        b0 = 1.0 / (n - 2 * K)                    # B = b0 – b1·A
        b1 = 2.0 * K / (n - 2 * K)

        coeff2 = convEE - 2.0 * b1 * convEM + (b1 ** 2) * convMM
        coeff1 = 2.0 * b0 * convEM - 2.0 * b0 * b1 * convMM
        coeff0 = (b0 ** 2) * convMM

        def maxconv(A: float) -> float:
            """Maximum entry of the convolution for a given edge weight A."""
            conv = coeff2 * (A ** 2) + coeff1 * A + coeff0
            return float(conv.max())

        # scalar optimisation (bounded)
        res = minimize_scalar(
            maxconv,
            bounds=(0.0, A_max),
            method="bounded",
            options={"xatol": 1e-7},
        )
        A_opt = float(res.x)
        B_opt = b0 - b1 * A_opt

        vec = np.full(n, B_opt, dtype=np.float64)
        vec[:K] = A_opt
        vec[-K:] = A_opt
        vec = normalize(vec)                      # safety

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
            print(
                f"[{time.time():.1f}] piecewise (K={K}) improved to {best_score:.6f}"
            )
    return best_vec, best_score


# -------------------------------------------------
# Smoothed‑max optimisation (Adam)
# -------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 800.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (see derivation in the original code)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 1000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(
                    f"[{time.time():.1f}] Adam improved to {best_score:.6f} (it {it})"
                )

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# -------------------------------------------------
# Projected sub‑gradient descent (cheap)
# -------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for 2·n·max(a*a) with step size ∝ 1/√t.
    The iterate is projected onto the simplex after each step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f} (t {t})"
                )

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# -------------------------------------------------
# Stochastic hill‑climb (simulated annealing)
# -------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a tiny temperature schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

                if time.time() >= deadline - 30.0:
                    break
                vec = (lam * symmetric_product_family(n, alpha) +
                       (1.0 - lam) * exp_u_family(n, gamma))
                vec = normalize(vec)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(
                        f"[{time.time():.1f}] new best (mix λ={lam:.2f}, α={alpha:.2f}, γ={gamma:.2f}) = {best_score:.6f}"
                    )

    # ---- cheap two‑level piecewise constant family ----
    pw_vec, pw_sc = piecewise_two_level_optimize(n)
    if pw_sc < best_score:
        best_score = pw_sc
        best_seq = pw_vec.tolist()
        print(f"[{time.time():.1f}] new best from piecewise = {best_score:.6f}")

    # ---- a few random monotone / U‑shaped starts ----
    for _ in range(80):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    for _ in range(80):
        if time.time() >= deadline - 30.0:
            break
        vec = random_u_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑U) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam‑based smooth‑max refinement on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 60.0,
            max_iters=150_000,
            beta_start=5.0,
            beta_end=800.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient refinement (cheap per iteration)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            max_iters=2_000_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Symmetrisation (many optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = normalize(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Safe fallback if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7. Official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(
        f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}"
    )
    return best_seq
