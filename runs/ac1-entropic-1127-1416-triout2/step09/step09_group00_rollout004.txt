"""
Improved search for a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

under the non‑negative simplex constraint.

Algorithm
---------
1.  Scan several problem sizes (2048, 4096, 8192, 16384).  For each length
    generate a few analytically‑motivated seeds (uniform, symmetric power,
    left‑heavy power, random monotone).

2.  For each seed run a cheap *spectral‑flattening* iteration:
    - Zero‑pad the vector, compute its real FFT.
    - Replace every magnitude by the constant √(mean(|FFT|²)) while keeping
      the phase (this enforces a flat power spectrum → flatter autocorrelation).
    - Inverse FFT, truncate, clip negatives and Euclidean‑project onto the
      probability simplex.
    - Repeat ~4000 times, keeping the best score seen.

3.  Keep the best vector over all lengths.  Starting from this vector run a
    longer *smoothed‑max* optimisation (Adam on the soft‑max parametrisation)
    with a β schedule from 5 up to 800.  This directly reduces the true
    max‑convolution.

4.  Symmetrise the result (a ← (a+a[::-1])/2) and project back onto the simplex.

5.  A final short simulated‑annealing hill‑climb (swap / mass‑transfer)
    cleans up any remaining spikes.

All phases respect the global wall‑clock budget (default 1000 s) and the
function returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (aperiodic) convolution via real FFT.
    Returns the full convolution of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    where n = len(a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    return a / a.sum()


# ----------------------------------------------------------------------
# Spectral flattening step (alternating projection)
# ----------------------------------------------------------------------


def flatten_spectrum_step(a: np.ndarray, L: int) -> np.ndarray:
    """
    One iteration of spectral flattening:
    - Zero‑pad a to length L,
    - Replace every magnitude in the real FFT by the constant sqrt(mean(|A|²)),
    - Inverse FFT, truncate, clip negatives, and project onto the simplex.
    """
    n = a.shape[0]
    a_pad = np.zeros(L, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)                 # length L//2+1 complex numbers
    power = np.abs(A) ** 2
    avg_power = np.mean(power)
    mag_target = np.sqrt(avg_power)
    A_new = mag_target * np.exp(1j * np.angle(A))
    a_full = np.fft.irfft(A_new, n=L)
    a_new = a_full[:n]
    a_new = np.maximum(a_new, 0.0)
    a_new = project_to_simplex(a_new)
    return a_new

# Smoothed‑max optimisation (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 800.0,
                         lr: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially from beta_start to beta_end.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)              # initialise in log‑prob space

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- schedule for β (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a and its convolution -----
        a = softmax(z)
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ----- gradient w.r.t a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # central n entries

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 700_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
    – 50 % probability: swap two entries.
    – 50 % probability: transfer a fraction of mass from one index to another.
    Very small temperature allows occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling of the temperature
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level search driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the global time budget and returns a plain
    Python list of floats ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin – the harness adds its own checks
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from a previously known construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Multi‑size exploration with spectral flattening
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() >= deadline - 200.0:
            break
        print(f"[{time.time():.1f}] exploring length n = {n}")

        # Pad length for the FFT used in spectral flattening
        L = next_pow2(2 * n)

        # ----- construct a few diverse seeds -----
        seeds = [
            np.full(n, 1.0 / n, dtype=np.float64),                     # uniform
            symmetric_product_family(n, alpha=2.5),                    # symmetric power
            left_heavy_power_family(n, beta=2.5),                      # left‑heavy
        ]
        # add a handful of random monotone starts
        for _ in range(2):
            seeds.append(random_monotone_start(n))

        best_n_seq = None
        best_n_score = float("inf")

        # ----- process each seed independently -----
        for s_idx, a0 in enumerate(seeds):
            if time.time() >= deadline - 120.0:
                break
            a = project_to_simplex(a0.copy())
            # occasional evaluation of the raw seed
            sc0 = compute_score(a)
            if sc0 < best_n_score:
                best_n_score = sc0
                best_n_seq = a.copy()
                if sc0 < best_score:
                    best_score = sc0
                    best_seq = a.tolist()
                    print(f"[{time.time():.1f}] new global best (seed) = {best_score:.6f}")

            # ----- spectral flattening loop -----
            max_flat_iters = 4000
            for it in range(1, max_flat_iters + 1):
                if time.time() >= deadline - 80.0:
                    break
                a = flatten_spectrum_step(a, L)

                # check progress every 800 iterations
                if it % 800 == 0:
                    sc = compute_score(a)
                    if sc < best_n_score:
                        best_n_score = sc
                        best_n_seq = a.copy()
                    if sc < best_score:
                        best_score = sc
                        best_seq = a.tolist()
                        print(f"[{time.time():.1f}] new global best (flatten it {it}) = {best_score:.6f}")

            # final check for this seed
            sc_final = compute_score(a)
            if sc_final < best_n_score:
                best_n_score = sc_final
                best_n_seq = a.copy()
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] new global best (post‑flatten) = {best_score:.6f}")

        # after all seeds for current size, store the best of this size
        if best_n_score < best_score:
            best_score = best_n_score
            best_seq = best_n_seq.tolist()
            print(f"[{time.time():.1f}] best for n={n} improved global best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed‑max refinement (Adam) on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] starting Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 30.0,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=800.0,
            lr=0.015,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam refined best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Symmetrisation test
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final hill‑climb (simulated annealing) – short run
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=700_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: if nothing succeeded, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats.
    return best_seq
