"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1.  Choose a modest length (n≈8192).  All vectors are kept on the
    probability simplex (∑a_i = 1) – this makes the objective simply
    2·n·max(conv(a,a)) and removes the need for scaling inside the optimiser.
2.  Build a few sensible starting points:
        – the classic inverse‑square‑root (central‑binomial) shape,
        – a one‑parameter inverse‑sqrt family  a_i ∝ 1/√(i+β)  (β scanned),
        – a uniform vector,
        – a few random Dirichlet draws.
3.  For each start run a *β‑continuation* L‑BFGS‑B optimisation.
    The smooth surrogate of the max‑convolution is
        f_β(a) = max(conv) + (1/β)·log∑exp(β·(conv−max(conv))),
    which converges to the true max as β→∞.
    β is increased geometrically from 0.5 to 3000, and the optimiser
    works on the soft‑max re‑parameterisation (logits) so the simplex
    constraints are satisfied automatically.
4.  The best vector from the continuation phase is polished with a long
    projected sub‑gradient descent on the exact objective.
    The step size follows a 1/√t schedule (t = iteration count) and
    the projection is the Euclidean projection onto the simplex.
5.  A final cheap stochastic hill‑climb (swap / fractional mass transfer)
    is performed while time permits.
6.  The sequence with the smallest value of ``evaluate_sequence`` is
    returned.  The code respects the 1000 s wall‑clock budget and prints
    progress information.

The method combines a smooth surrogate optimisation (exact gradients via
FFT‑based convolution) with a large number of cheap sub‑gradient steps,
which has proven to push the bound below the previous 1.5113 level.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (efficient for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a probability vector
    (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that always returns a simplex vector."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – value and gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where

        f = max(conv) + (1/β)·log Σ exp(β·(conv – max(conv)))

    and g is the gradient with respect to the logits (soft‑max
    parametrisation).  All arithmetic is performed with a on the simplex.
    """
    a = softmax_logits(logits)
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # log‑sum‑exp centred at maxc
    exp_weights = np.exp(beta * (conv - maxc))
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient wrt a
    w = exp_weights / Z               # weights for each convolution entry
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # chain‑rule back to logits (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 300):
    """
    Anneal the smoothing parameter β while optimising the smooth surrogate.
    Returns the best simplex vector seen and its exact (non‑smooth) score.
    """
    logits = np.log(start_a + 1e-20)   # initialise logits

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for beta in betas:
        if time.time() >= deadline:
            break

        # allocate a reasonable iteration budget for the current β
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(30, int(time_left * 25)))   # ≈25 L‑BFGS iters / s

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f}] new best (smooth) score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 800_000,
                     step0: float = 0.2):
    """
    Projected sub‑gradient descent on the exact 2·n·max(conv) objective.
    Uses a 1/√t step‑size schedule.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # keep previous gradient for optional momentum (not used here)
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny perturbation to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic start – truncated central‑binomial coefficients
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape  a_i = C(2i,i)/4^i  normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Simple parametric family  a_i ∝ 1/√(i+β)
# ----------------------------------------------------------------------


def inv_sqrt_offset_sequence(n: int, beta: float) -> np.ndarray:
    """a_i ∝ 1/√(i+β)  (β>0).  Returned vector is normalised to sum = 1."""
    i = np.arange(n, dtype=float) + beta
    a = 1.0 / np.sqrt(i)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine follows the plan described in the module docstring.
    Returns a plain list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback to a previously found construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose a working length (a compromise between speed and quality)
    # -----------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 800:
        n = 16384
    else:
        n = 8192

    print(f"Working length n = {n}")

    # -----------------------------------------------------------------
    # 2. Build a list of promising initial vectors
    # -----------------------------------------------------------------
    starts = []

    # (a) deterministic inverse‑square‑root (central‑binomial)
    starts.append(central_binomial_sequence(n))

    # (b) scan the 1/√(i+β) family for a few β values
    best_beta = None
    best_beta_score = float("inf")
    for beta in np.linspace(0.5, 10.0, 20):
        cand = inv_sqrt_offset_sequence(n, beta)
        sc = compute_score(cand)
        if sc < best_beta_score:
            best_beta_score = sc
            best_beta = beta
    if best_beta is not None:
        starts.append(inv_sqrt_offset_sequence(n, best_beta))
        print(f"Best β‑offset = {best_beta:.3f}, score = {best_beta_score:.6f}")

    # (c) uniform distribution (on the simplex)
    starts.append(np.full(n, 1.0 / n))

    # (d) a few random Dirichlet draws
    for _ in range(3):
        rand = np.random.rand(n)
        rand = rand / rand.sum()
        starts.append(rand)

    # -----------------------------------------------------------------
    # 3. Continuation optimisation from each start
    # -----------------------------------------------------------------
    for idx, start_vec in enumerate(starts):
        if time.time() >= deadline - 200:
            break
        print(f"\n=== Continuation start #{idx} (β‑annealing) ===")
        a_smooth, smooth_score = lbfgs_continuation(
            start_vec,
            beta_start=0.5,
            beta_end=3000.0,
            deadline=deadline - 100,
            schedule_steps=12,
            maxiter_per_step=300,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"Best after continuation #{idx}: {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Exact max‑objective polishing (projected sub‑gradient)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120:
        print("\n--- Sub‑gradient polishing ---")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 80,
            max_iter=800_000,
            step0=0.2,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑grad improvement: {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Very cheap stochastic hill‑climb (local search)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("\n--- Final hill‑climb ---")
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(best_a)
        step_scale = 0.15

        while time.time() < deadline:
            cand = best_a.copy()

            if np.random.rand() < 0.6:
                # fractional mass transfer
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = step_scale * cand[i]
                    cand[i] -= delta
                    cand[j] += delta
            else:
                # swap two entries
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]

            # ensure feasibility
            cand = np.maximum(cand, 0.0)
            cand = project_to_simplex(cand)

            cur_score = compute_score(cand)
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                step_scale = min(step_scale * 1.08, 0.5)
            else:
                step_scale *= 0.96

        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.tolist()
            print(f"Hill‑climb final score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Fallback in the extremely unlikely case that nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Final verification (using the provided evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of Python floats.
    return [float(x) for x in best_seq]
