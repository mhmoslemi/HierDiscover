import cvxpy as cp

def lp_refine(a, max_iters=5, step=0.2):
    n = len(a)
    # Precompute convolution matrix
    # Use sparse representation to reduce memory.
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # For i in [i_min, i_max], b_{k-i} appears
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])  # f_i * b_j (i is f index)
    # Build sparse matrix
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Vector of constraints: conv_f_b <= M
    for it in range(max_iters):
        M = convolution_fft(a, a).max()
        b_var = cp.Variable(n)
        constraints = [b_var >= 0, A @ b_var <= M]
        objective = cp.Maximize(cp.sum(b_var))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True) # etc.
        if b_var.value is None:
            break
        b = np.maximum(b_var.value, 0)
        sum_b = b.sum()
        if sum_b == 0:
            break
        g = b / sum_b  # normalize to sum=1
        # update a
        a = (1-step) * a + step * g
        a = project_to_simplex(a)
    return a

"""
Improved step‑function search
================================

This implementation expands on the previous baseline in several ways:

1. **Arcsine + LP warm‑start** –  an arcsine‑shaped distribution is a
   theoretically good seed for a flat self‑convolution.  A tiny linear
   program (CVXPY) is solved a few times to push mass into positions
   that do not increase the current maximal convolution value.  This
   typically brings the ℓ∞‑norm of the self‑convolution well below the
   uniform‑vector value (≈1 / n).

2. **Multi‑scale up‑sampling** – the vector is repeatedly doubled in
   length (splitting each entry in half).  After each up‑sampling we run
   a short projected sub‑gradient optimisation to adapt to the finer
   grid.

3. **High‑temperature soft‑max Adam** – a smooth log‑sum‑exp surrogate of
   the max‑convolution is minimised with Adam for a long time; the
   temperature β is annealed up to 5000 so the surrogate becomes very
   close to the true max.

4. **Final simulated‑annealing polishing** – random local mass‑transfer
   moves are accepted with a temperature schedule that decays to zero,
   allowing the algorithm to escape shallow local minima.

All stages respect a 1000 s wall‑clock budget, keep the vector on the
probability simplex (non‑negative, unit sum) and finally return a plain
Python list ready for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT‑based convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact `evaluate_sequence` score for a vector with Σa_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)) – used by the LP stage."""
    return float(convolution_fft(a, a).max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """Smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def arcsine_start(n: int) -> np.ndarray:
    """
    Arcsine‑type seed:  a_i ∝ 1 / sqrt( x_i·(1−x_i) )  with
    x_i = (i+0.5)/n .  This concentrates mass near the endpoints,
    a shape known to flatten the self‑convolution.
    """
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Pick one of the three classic generators plus arcsine."""
    r = np.random.rand()
    if r < 0.3:
        return random_start_dense(n)
    elif r < 0.6:
        return random_start_sparse(n)
    elif r < 0.85:
        return random_start_gaussian(n)
    else:
        return arcsine_start(n)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a differentiable soft‑max surrogate of max(conv(a,a))
    using Adam with simplex projection.  Returns the best exact score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # linear annealing of β on log‑scale
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))            # stable weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of the soft‑max (cross‑correlation trick)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).
    The update direction comes from the (unique) maximising offset.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the maximal entry

        # sub‑gradient for this index (see analysis in the description)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # tiny random jitter every 2 000 iterations
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = None,
) -> tuple[list[float], float]:
    """
    Random local modifications (swap or small mass transfer) with a
    temperature schedule.  Accept improving moves always;
    accept worsening moves with probability exp(-Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: list[float]) -> np.ndarray:
    """Double the length by replacing each entry a_i with two adjacent halves."""
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)
    return up


# ----------------------------------------------------------------------
# 5. Linear‑programming refinement (optional, needs cvxpy)
# ----------------------------------------------------------------------
HAS_LP = False
try:
    import cvxpy as cp
    import scipy.sparse as sp

    HAS_LP = True
except Exception:   # pragma: no cover
    pass


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix M such that (M·b) = conv(f,b).
    Row k contains the weights f_i for all i with 0≤i<n and 0≤k‑i<n.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # for each column j (b_j) the contributing rows are j … j+n‑1
    for j in range(n):
        rows.append(np.arange(j, j + n, dtype=int))
        cols.append(np.full(n, j, dtype=int))
        data.append(f)                     # f_i corresponds to row i+j

    row_arr = np.concatenate(rows)
    col_arr = np.concatenate(cols)
    data_arr = np.concatenate(data)
    M = sp.csr_matrix((data_arr, (row_arr, col_arr)), shape=(2 * n - 1, n))
    return M


def lp_refine(
    a0: np.ndarray,
    time_deadline: float,
    max_lp_iters: int = 5,
    step: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based “Frank‑Wolfe‑like’’ updates.
    Each LP maximises Σ b_j under the constraint conv(a,b)≤M,
    where M is the current maximal self‑convolution value.
    The returned vector stays on the simplex.
    """
    if not HAS_LP:
        return a0, compute_score(a0)

    a = a0.astype(float).copy()
    n = a.shape[0]
    M_mat = build_conv_matrix(a)

    for it in range(max_lp_iters):
        if time.time() >= time_deadline:
            break

        M_val = max_convolution(a)

        # LP: maximise Σ b   s.t.   M_mat @ b ≤ M_val , b ≥ 0
        b = cp.Variable(n, nonneg=True)
        constraints = [M_mat @ b <= M_val]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)

        if b.value is None:
            break

        b_vec = np.maximum(b.value, 0.0)
        s = b_vec.sum()
        if s <= 0.0:
            break

        g = b_vec / s                      # normalise to unit sum
        a = (1.0 - step) * a + step * g
        a = project_to_simplex(a)

        # update matrix for the new a (required for next LP)
        M_mat = build_conv_matrix(a)

    return a, compute_score(a)

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the 1000 s wall‑clock
    budget and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness after we return
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Warm‑start with LP‑refined seeds (small dimension)
    # ------------------------------------------------------------------
    n_small = 256
    print(f"Warm‑start phase – length {n_small}")

    # reserve some time for later stages
    reserve_time = 300.0
    while time.time() < deadline - reserve_time:
        # pick a generator at random
        generator = np.random.choice([random_start_dense,
                                      random_start_sparse,
                                      random_start_gaussian,
                                      arcsine_start])
        a0 = generator(n_small)

        # quick LP refinement (if cvxpy is available)
        lp_deadline = time.time() + 12.0
        a_lp, score_lp = lp_refine(a0, lp_deadline, max_lp_iters=3, step=0.25)
        # a quick exact evaluation
        exact = compute_score(a_lp)

        if exact < best_val:
            best_val = exact
            best_seq = a_lp.tolist()
            print(f"  New best after LP warm‑start: score={best_val:.6f}")

        # if the remaining time is very small we break
        if time.time() > deadline - reserve_time:
            break

    # ------------------------------------------------------------------
    # 2. Gradual up‑sampling to the target length (≈2048)
    # ------------------------------------------------------------------
    target_n = 2048
    if best_seq is not None and (deadline - time.time() > 200):
        print("\nUpsampling phase")
        a = np.array(best_seq, dtype=float)
        while a.shape[0] < target_n and time.time() < deadline - 120:
            a = upsample_sequence(a.tolist())
            # brief sub‑gradient refinement after each doubling
            sub_deadline = time.time() + 30.0
            a, sub_score = subgrad_optimize(a,
                                            time_deadline=sub_deadline,
                                            max_iter=150_000,
                                            step0=0.05)
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.tolist()
                print(f"  Upsampled to {a.shape[0]} – new best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. High‑temperature soft‑max Adam optimisation
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80:
        print("\nSmooth‑max Adam optimisation")
        a_start = np.array(best_seq, dtype=float)
        a_opt, _ = smooth_gradient_optimize(
            a_start,
            time_deadline=deadline - 60,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=5000.0,
            step0=0.8,
        )
        cur_val = compute_score(a_opt)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_opt.tolist()
            print(f"  After Adam – score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("\nFinal hill‑climb polishing (simulated annealing)")
        polished_seq, polished_score = hill_climb_anneal(
            best_seq,
            time_deadline=deadline,
            max_steps=None,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"  Polished score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very last fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Official evaluation (the harness will also call `evaluate_sequence`)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return a plain Python list – the harness expects this.
    return best_seq
