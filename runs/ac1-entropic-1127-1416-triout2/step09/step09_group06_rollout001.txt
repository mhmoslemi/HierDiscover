assistantfinal**Solution Overview**

The script searches for a probability vector `a` (non‑negative, sum = 1) that minimizes  

\[
F(a)=2\,n\;\max_k\;(a*a)_k,
\]

i.e. the evaluation function `evaluate_sequence` when the vector is already normalised.  
The optimisation proceeds in three stages:

1. **Parametric warm‑start** – a short scan over a two‑parameter family  
   \[
   a_i\propto x_i^{-p}(1-x_i)^{-p}\;(1-\lambda)+\lambda\; \frac1n,
   \qquad
   x_i=\frac{i+0.5}{n},
   \]
   (a symmetric “Beta‑type’’ shape mixed with the uniform distribution).  
   The scan quickly finds a promising shape for the chosen length `n`.

2. **Smooth continuation with L‑BFGS‑B** – the non‑smooth max is replaced by a
   log‑sum‑exp surrogate with an annealing parameter β.
   The variables are the logits of a soft‑max, guaranteeing the simplex
   constraints automatically.  β is increased geometrically from a small
   value (smooth landscape) to a large value (tight approximation of the
   true max).

3. **Exact polishing** – a projected sub‑gradient method on the true
   objective further reduces the maximum convolution value.  
   Finally a cheap stochastic hill‑climb fine‑tunes the result.

The whole procedure respects the 1000 s wall‑clock budget, adapts the
number of iterations to the remaining time, and keeps the best sequence
found.  If a previous construction `height_sequence_1` is present it is used
as a fallback.

```python
"""
Propose a step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum(a))²
(the same quantity returned by evaluate_sequence).

Algorithm
---------
1.  Choose a length n depending on the remaining time.
2.  Scan a 2‑parameter family (Beta‑type shape + uniform mixture) to obtain a
    good initial vector.
3.  Run a β‑annealed L‑BFGS‑B optimisation on a log‑sum‑exp smooth
    approximation of the max–convolution.
4.  Polish the result with projected sub‑gradient descent on the exact
    objective.
5.  Perform a tiny stochastic hill‑climb while time permits.
6.  Return the best sequence (already normalised to sum = 1).

All helper functions are top‑level; the code never uses closures or lambdas.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function assuming sum(a) == 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; always sums to 1."""
    m = np.max(logits)
    shifted = np.exp(logits - m)
    return shifted / shifted.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) of the max‑convolution
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv−max(conv)))
    and g = ∂f/∂logits (gradient w.r.t. the soft‑max logits).
    The parameters a = softmax(logits) lie on the simplex.
    """
    a = softmax_logits(logits)

    # 1) self‑convolution a * a
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # 2) smooth max via log‑sum‑exp
    exp_weights = np.exp(beta * (conv - maxc))      # maxc subtracted for stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # 3) gradient w.r.t. a
    w = exp_weights / Z
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)          # length 2n‑1 + n‑1 = 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # 4) chain rule back to logits (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# L‑BFGS‑B continuation optimiser (β‑annealing)
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 400):
    """
    Anneal β while optimising the smooth surrogate with L‑BFGS‑B.
    Returns the best vector (still on the simplex) and its exact score.
    """
    logits = np.log(start_a + 1e-20)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # allocate a realistic per‑step iteration budget
        time_left = deadline - time.time()
        # each L‑BFGS iteration costs a few milliseconds → allow many if time permits
        max_iter = min(maxiter_per_step, max(20, int(time_left * 120)))

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter,
                                'disp': False,
                                'gtol': 1e-8})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.1f} step {step}] new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing (projected sub‑gradient descent)
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 150_000,
                     step0: float = 0.07):
    """
    Projected sub‑gradient descent on the exact objective 2·n·max(conv).
    Returns the best vector found and its exact score.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry (indexes i with i + j = k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape shallow plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Deterministic starting shapes
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """
    Inverse‑square‑root (central‑binomial) shape:
        a₀ = 1, a_{i+1} = a_i·(2i+1)/(2(i+1))
    Normalised to the probability simplex.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


def beta_shape(n: int, p: float) -> np.ndarray:
    """
    Symmetric Beta‑type shape with tail exponent p>0:
        a_i ∝ x_i^{-p} (1-x_i)^{-p},  x_i = (i+0.5)/n
    Normalised to sum = 1.
    """
    x = (np.arange(n) + 0.5) / n
    with np.errstate(divide='ignore'):
        a = np.power(x, -p) * np.power(1.0 - x, -p)
    a[np.isinf(a)] = 0.0
    a = a / a.sum()
    return a


def mixed_shape(n: int, p: float, lam: float) -> np.ndarray:
    """
    Mixture of the Beta‑type shape and the uniform distribution.
    lam ∈ [0,1] controls the amount of uniform mass.
    """
    a_beta = beta_shape(n, p)
    a_uni = np.full(n, 1.0 / n)
    a = (1.0 - lam) * a_beta + lam * a_uni
    a = a / a.sum()
    return a

    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  Returns a plain list of floats (already on the
    simplex).  The routine respects the 1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0. Use a previously found construction if it exists
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Choose a working length (larger n → potentially smaller bound)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 800:
        n = 32768
    elif time_left > 500:
        n = 16384
    elif time_left > 300:
        n = 8192
    else:
        n = 4096
    print(f"Chosen length n = {n}")

    # --------------------------------------------------------------
    # 2. Quick parametric scan (Beta‑type + uniform mixture)
    # --------------------------------------------------------------
    scan_start = time.time()
    p_grid = np.linspace(0.42, 0.58, 17)    # around the known 0.5 optimum
    lam_grid = np.linspace(0.0, 0.30, 7)   # up to 30 % uniform mixing
    best_local_a = None
    best_local_score = float("inf")

    for p in p_grid:
        for lam in lam_grid:
            if time.time() >= deadline - 150:
                break
            a = mixed_shape(n, p, lam)
            sc = compute_score(a)
            if sc < best_local_score:
                best_local_score = sc
                best_local_a = a
                print(f"Parametric scan: p={p:.3f} lam={lam:.2f} → score={sc:.6f}")

    # If the scan produced something better, adopt it
    if best_local_a is not None and best_local_score < best_val:
        best_val = best_local_score
        best_seq = best_local_a.tolist()
        print(f"After parametric scan best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Random restarts with L‑BFGS‑B continuation
    # --------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 120:
        restarts += 1
        # initialise a starting point
        if restarts == 1:
            a0 = central_binomial_sequence(n)
        elif restarts == 2 and best_seq is not None:
            # perturb the current best
            a0 = np.array(best_seq) + 0.02 * np.random.randn(n)
            a0 = np.clip(a0, 0, None)
            a0 = a0 / a0.sum()
        else:
            # fresh Dirichlet sample
            a0 = np.random.dirichlet(np.ones(n))

        # Smooth optimisation
        beta_start = 4.0
        beta_end = 2500.0
        a_smooth, smooth_score = lbfgs_continuation(
            a0,
            beta_start,
            beta_end,
            deadline - 70,            # leave time for polishing & hill‑climb
            schedule_steps=12,
            maxiter_per_step=400,
        )
        exact_score = compute_score(a_smooth)

        if exact_score < best_val:
            best_val = exact_score
            best_seq = a_smooth.tolist()
            print(f"[restart {restarts}] new overall best = {best_val:.6f}")

        # ----------------------------------------------------------
        # 4. Polishing with exact sub‑gradient descent (if time)
        # ----------------------------------------------------------
        if time.time() < deadline - 30:
            a_ref, ref_score = subgrad_optimize(
                np.array(best_seq, dtype=float),
                deadline - 20,
                max_iter=120_000,
                step0=0.07,
            )
            if ref_score < best_val:
                best_val = ref_score
                best_seq = a_ref.tolist()
                print(f"Polish improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Very cheap stochastic hill‑climb (final refinement)
    # --------------------------------------------------------------
    if time.time() < deadline - 5 and best_seq is not None:
        print("\nFinal stochastic hill‑climb …")
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(best_a)
        step_scale = 0.09

        while time.time() < deadline and step_scale > 1e-5:
            cand = best_a.copy()
            if np.random.rand() < 0.5:                     # simple swap
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]
            else:                                          # fractional transfer
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = step_scale * best_a[i]
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0:
                        cand[i] = 0.0
            cand = project_to_simplex(cand)
            cur_score = compute_score(cand)

            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                # successful move → enlarge step a little
                step_scale = min(step_scale * 1.08, 0.5)
            else:
                step_scale *= 0.95

        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.tolist()
            print(f"Hill‑climb final score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Fallback – extremely unlikely to be needed
    # --------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # Final sanity check (the harness will also call evaluate_sequence)
    final_val = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_val:.6f}")

    # Return a plain list of floats as required by the harness
    return best_seq
```
