"""
Propose a step‑function (list of non‑negative numbers) that minimises
`evaluate_sequence`.  

The optimiser works in three stages:

1. **Parametric seeding** – a fast sweep over several families of
   analytically‑motivated shapes (geometric decay, power‑law decay and
   symmetric Beta densities).  For every candidate the exact objective
   (2·n·max(conv)) is evaluated with an FFT based convolution.  The best
   seed is retained.

2. **Projected sub‑gradient refinement** – starting from the best seed we
   run a classic sub‑gradient method on the true max‑objective.  The
   update direction is the (unique) maximizing shift of the convolution,
   the step size decays as 1/√t and each iterate is projected onto the
   probability simplex.

3. **Hill‑climbing / simulated‑annealing polish** – a cheap stochastic
   local search that repeatedly perturbs a pair of entries (swap or a
   small mass transfer).  Improving moves are always accepted; worse
   moves are accepted with a temperature‑controlled probability that
   decays to zero.

All stages respect the overall wall‑clock budget (≈1000 s).  The
algorithm never produces negative numbers; the final vector is returned
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the constraint Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families – fast generators
# ----------------------------------------------------------------------


def geom_seq(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ (1‑r)·r**i, normalised to sum = 1."""
    if r <= 0.0:
        r = 0.0
    if r >= 1.0:
        r = 0.999999999999
    a = (1.0 - r) * (r ** np.arange(n, dtype=float))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def powerlaw_seq(n: int, beta: float) -> np.ndarray:
    """Power‑law decay a_i ∝ (i+1)^{-beta}, normalised."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-beta)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def beta_symmetric_seq(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1) sampled at mid‑points,
    then normalised.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    # avoid overflow/underflow for extreme α
    a = np.power(x, max(alpha - 1.0, 0.0)) * np.power(1.0 - x, max(alpha - 1.0, 0.0))
    # for α<1 the numerator may blow up at the ends – clip to avoid NaNs
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient optimisation (exact max‑objective)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).
    The sub‑gradient for the active shift `k` is
        g_i = 2·a_{k-i}   (if 0 ≤ k-i < n, else 0).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build sub‑gradient for this k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = None,
) -> tuple[list[float], float]:
    """
    Random local modifications (swap or small mass transfer) with a
    temperature schedule.  Improving moves are always accepted; worsening
    moves are accepted with probability exp(‑Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: list[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two adjacent
    entries a_i/2, a_i/2.  The total mass is unchanged.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)
    return up
