"""
Hybrid search for the step‑function minimising
    2·n·max(convolve(a,a)) / (sum(a)**2)

The algorithm proceeds in stages:

1.  Analytic seeding – a quick scan of simple parametric families
    (symmetric product, left‑heavy power, exponential U‑shape) for a few
    problem sizes (n≈4096,8192).  The best candidate found so far is kept.

2.  Linear‑programming refinement – given a current vector f,
    we solve
        maximise   Σ b_i
        s.t.       (f * b)_k ≤ M   for the k‑indices where the autocorrelation
                                      of f attains (or is close to) its maximum M,
                b ≥ 0
    using SciPy’s high‑performance “highs” LP solver.
    The optimiser returns a direction b; we take a convex step
        f ← (1‑t)·f + t·b   (t≈0.2) and project back onto the simplex.
    Repeating this step quickly drives the peak of the convolution down.

3.  Fine‑grained smooth optimisation – a few thousand iterations of the
    Adam optimiser on the differentiable smoothed‑max objective
    f_β(a) = (1/β)·log Σ exp(β·(a*a)_k) with an exponential schedule for β.

4.  Final stochastic hill‑climb (simulated annealing) to polish the result.

All stages respect the global 1000 s wall‑clock budget; the routine finally
returns a plain Python list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a=1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for quick seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised to unit sum."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, unit‑sum."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Convex mixture of the uniform vector and the exponential‑U shape.
    lam∈[0,1] is the weight of the uniform component.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a


# ----------------------------------------------------------------------
# Random initialisers (used if no analytic candidate beats the fallback)
# ----------------------------------------------------------------------


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Sample a random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max optimiser (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 150.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed‑max objective (see header)."""
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (used as final polish)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Long random walk on the simplex (swap / mass‑transfer moves)."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (core of the new contribution)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray,
              deadline: float,
              max_iters: int = 30,
              t: float = 0.2,
              max_constraints: int = 200) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a small LP to obtain a direction that keeps
    the peak autocorrelation ≤ current maximum M.  The new point is
    a convex combination (1‑t)·a + t·b, projected onto the simplex.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current autocorrelation and its peak
        conv = convolution_fft(a, a)
        M = conv.max()

        # pick the most offending shifts (largest values)
        if conv.size <= max_constraints:
            idxs = np.arange(conv.size)
        else:
            # keep the top `max_constraints` indices
            idxs = np.argpartition(conv, -max_constraints)[-max_constraints:]

        # build the constraint matrix A_ub * b ≤ M
        A_rows = []
        for k in idxs:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            idx = np.arange(start_i, end_i + 1)
            row = np.zeros(n, dtype=np.float64)
            row[idx] = a[k - idx]
            A_rows.append(row)

        A_ub = np.vstack(A_rows)
        b_ub = np.full(A_ub.shape[0], M)

        # LP: maximise Σ b_i   (i.e. minimise −Σ b_i)
        c = -np.ones(n, dtype=np.float64)

        try:
            res = linprog(c,
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        except Exception as e:
            print(f"[{time.time():.1f}] LP exception: {e}")
            break

        if not res.success:
            print(f"[{time.time():.1f}] LP not successful: {res.message}")
            break

        b_opt = res.x
        # normalise the direction – keep it on the simplex
        b_opt = project_to_simplex(b_opt)

        # convex update
        a = (1.0 - t) * a + t * b_opt
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iteration {it} improved to {best_score:.6f}")

    return best_a, best_score
