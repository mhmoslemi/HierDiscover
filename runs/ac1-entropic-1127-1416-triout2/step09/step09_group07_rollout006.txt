def hill_climb_sa(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    conv = convolution_fft(a,a)
    cur_score = 2*n*conv.max()
    best_a = a.copy()
    best_score = cur_score
    step_scale = 0.2      # initial transfer fraction of source mass
    inc_factor = 1.07
    decay_factor = 0.985
    recompute_interval = 50000
    steps = 0
    total_time = deadline - start_time
    while time.time() < deadline:
        # Temperature schedule:
        remaining = deadline - time.time()
        temp = max(1e-8, 0.001 * (remaining / total_time))  # linear decay
        # propose move
        if np.random.rand() < 0.5:
            src = np.random.randint(n)
            if a[src] <= 1e-12:
                continue
            dst = np.random.randint(n-1)
            if dst >= src:
                dst += 1
            # random fraction
            delta = np.random.rand() * step_scale * a[src]
            if delta <= 0:
                continue
        else:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            src, dst = (i, j) if a[i] >= a[j] else (j, i)
            delta = a[src] - a[dst]   # swap total difference
            if delta <= 0:
                continue
        # apply incremental conv update
        conv_cand = conv.copy()
        # linear part
        conv_cand[src:src+n] -= 2.0 * delta * a
        conv_cand[dst:dst+n] += 2.0 * delta * a
        # quadratic part
        conv_cand[2*src] += delta*delta
        conv_cand[2*dst] += delta*delta
        conv_cand[src+dst] -= 2.0 * delta*delta
        cand_score = 2.0 * n * float(conv_cand.max())
        # decide acceptance
        accept = False
        if cand_score < cur_score - 1e-12:
            accept = True
        else:
            # accept with probability exp(-(cand_score - cur_score)/temp)
            if temp > 0 and np.random.rand() < np.exp(-(cand_score - cur_score)/temp):
                accept = True
        if accept:
            a[src] -= delta
            a[dst] += delta
            # renorm to fix roundoff
            s = a.sum()
            if s != 0.0:
                a /= s
            conv = conv_cand
            cur_score = cand_score
            # update step size
            step_scale = min(step_scale * inc_factor, 0.5)
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            step_scale *= decay_factor
        steps += 1
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # occasional random perturbation to escape plateau

            a = np.maximum(a + noise, 0)

            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score

"""
propose_candidate – search for a non‑negative step‑function (vector) that
minimises the evaluation function

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1. **Broad analytic seeding** – a small family of analytically‑motivated
   shapes (inverse‑power, Beta‑distribution, sine‑power) is scanned on a
   modest grid; the best vector (normalised to sum = 1) is kept.

2. **Fast simulated‑annealing hill‑climb** – starting from the best seed,
   a massive random‑walk is performed.  Each move transfers a small packet
   of mass from a source index to a destination index (or swaps two
   entries).  The convolution is updated incrementally (O(n)) and the
   new objective is evaluated exactly.  Moves that improve the score are
   always accepted; worsening moves are accepted with a temperature‑controlled
   probability, allowing the search to escape local minima.  The step size
   is adapted dynamically and the full convolution is recomputed periodically
   to avoid drift.

3. **Final evaluation** – the best vector found is returned as a plain
   Python list; `evaluate_sequence` is called once at the end to obtain the
   official score.

The routine respects the 1000 s wall‑clock limit (a small safety margin is
kept) and adapts the vector length (`n`) to the available time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (fast for power‑of‑2 lengths)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This is the value returned by `evaluate_sequence` when the input already
    satisfies the sum‑normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for seeding
# ----------------------------------------------------------------------


def shape_power_inverse(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^{-α}.
    For α<0 the shape is centre‑heavy, for α>0 edge‑heavy.
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def shape_beta(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ x_i^p (1‑x_i)^q  with  x_i = (i+½)/n.
    This is the discrete analogue of a Beta(p+1,q+1) density.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = (x ** p) * ((1.0 - x) ** q)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def shape_sine_power(n: int, r: float) -> np.ndarray:
    """
    a_i ∝ sin(π·x_i)^r  with  x_i = (i+½)/n.
    Produces a smooth, centre‑peaked family.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.sin(np.pi * x) ** r
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_analytic_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a modest grid over the three analytic families and keep the best vector.
    """
    best_a = None
    best_score = float("inf")

    # 1 – power‑inverse family
    alphas = np.concatenate([np.linspace(-2.0, 0.0, 13), np.linspace(0.0, 4.0, 25)])
    for alpha in alphas:
        if time.time() >= deadline:
            return best_a, best_score
        a = shape_power_inverse(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # 2 – Beta family
    p_vals = [0.0, 0.5, 1.0, 2.0, 4.0, 6.0]
    q_vals = [0.0, 0.5, 1.0, 2.0, 4.0, 6.0]
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline:
                return best_a, best_score
            a = shape_beta(n, p, q)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # 3 – sine‑power family
    r_vals = [0.5, 1.0, 2.0, 3.0, 4.0, 6.0]
    for r in r_vals:
        if time.time() >= deadline:
            return best_a, best_score
        a = shape_sine_power(n, r)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (incremental updates)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  inc_factor: float = 1.07,
                  decay_factor: float = 0.985,
                  recompute_interval: int = 50000) -> tuple[np.ndarray, float]:
    """
    Random‑walk polishing with simulated‑annealing acceptance.
    Uses the O(n) incremental convolution update from the reference code.
    """
    a = start_a.copy()
    n = a.size

    # initial exact convolution and score
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_score = cur_score

    step_scale = 0.15  # typical packet size as fraction of source mass
    steps = 0
    start_time = time.time()
    total_time = deadline - start_time

    while time.time() < deadline:
        # temperature linearly decays from ~1e-3 to ~1e-8
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, 1.0 - elapsed / total_time)
        temperature = 1e-8 + 1e-3 * remaining_frac

        # ----- propose a move -----
        if np.random.rand() < 0.5:
            # transfer a small packet from src -> dst
            src = np.random.randint(n)
            if a[src] <= 1e-14:
                continue
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * step_scale * a[src]  # 0 ≤ δ ≤ step_scale·a_src
            if delta <= 0.0:
                continue
        else:
            # swap (full exchange of the difference)
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = a[src] - a[dst]          # positive, will make the two entries equal
            if delta <= 0.0:
                continue

        # ----- incremental convolution update -----
        conv_cand = conv.copy()
        # linear part: 2·δ·(shift_dst – shift_src)
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic correction
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_score = 2.0 * n * float(conv_cand.max())

        # ----- acceptance test -----
        accept = False
        if cand_score < cur_score - 1e-14:
            accept = True
        else:
            # Metropolis rule
            if temperature > 0.0:
                prob = np.exp(-(cand_score - cur_score) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # apply the move to the true vector
            a[src] -= delta
            a[dst] += delta
            # renormalise to combat accumulated round‑off
            s = a.sum()
            if s != 0.0:
                a /= s
            conv = conv_cand
            cur_score = cand_score

            # adapt step size
            step_scale = min(step_scale * inc_factor, 0.5)

            # record best
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            step_scale *= decay_factor

        steps += 1

        # periodic full recompute to keep the incremental representation accurate
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # occasional tiny random perturbation to shake out of flat plateaux
        if steps % (recompute_interval * 5) == 0:
            noise = (np.random.randn(n) * step_scale * 0.01)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function vector that minimises
    ``evaluate_sequence``.  The routine follows the three‑stage plan
    described in the module docstring and always respects the 1000 s wall‑clock
    limit.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so the harness can still call evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a workable length (more points → potentially lower bound)
    # ------------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic seeding – scan several families
    # ------------------------------------------------------------------
    a_start, sc_start = find_best_analytic_start(n, deadline - 30.0)
    if a_start is not None and sc_start < best_score:
        best_score = sc_start
        best_seq = a_start.tolist()
        print(f"[{time.time():.1f}] best after analytic seeding = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Hill‑climb with simulated annealing (majority of remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] starting simulated‑annealing hill‑climb")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=float),
            deadline,
            inc_factor=1.07,
            decay_factor=0.985,
            recompute_interval=50000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Safeguard: fallback to a short uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
