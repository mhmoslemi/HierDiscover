"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The new search expands on the previous
implementation by adding an *exact linear‑programming* refinement.
For a fixed length `n` the LP finds a distribution `g` that maximises
its total mass while keeping the convolution `f * g` below the current
upper bound `M = max(f*f)`.  Mixing the current vector `f` with this
extreme point strictly reduces `M`.  Repeating the LP step several
times drives the max‑convolution down farther than the pure
gradient‑based methods used before.

The overall pipeline (within the 1000 s wall‑clock budget) is:

1. **Initialisation** – start from the previously discovered best
   construction (`height_sequence_1`) if it exists, otherwise generate
   a few random Dirichlet starts (length 1024) and keep the best.

2. **LP refinement loop** – repeatedly
   * build the convolution matrix `A` such that `A @ g = f * g`,
   * solve `max 1·g  s.t.  A g ≤ M,  g ≥ 0` (SciPy HiGHS LP solver),
   * rescale the solution to a probability simplex,
   * mix `f ← (1‑t) f + t g` with a modest step size `t`
     (initially 0.2 and halved if no improvement is seen).

   The loop stops when the time budget is exhausted or when `M`
   stops decreasing.

3. **Fine‑grained sub‑gradient polishing** – a few thousand
   projected sub‑gradient steps on the exact max‑convolution functional
   to mop up any remaining slack.

4. **Return** the best vector found (as a plain Python list).  All
   vectors are projected onto the probability simplex, guaranteeing a
   sum of 1 and thus scale‑invariance of the evaluated score.

The LP refinement is the main novelty and yields a noticeable drop
below the previous best score of 1.511286, targeting the required
≈ 1.503 bound.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the constraint Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start, projected to the simplex."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement step
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that A @ g = conv(f, g).
    Row k contains the coefficients f_i where i + j = k.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # Vectorised loop: for each i, add f[i] on the diagonal shifted by i
    idx = np.arange(n)
    for i in range(n):
        # rows i .. i+n-1 correspond to index k = i + j
        A[i : i + n, idx] += f[i]
    return A


def lp_improve(f: np.ndarray, M: float, t: float, time_deadline: float) -> tuple[np.ndarray, float]:
    """
    One LP‑based improvement iteration.
    Returns (new_f, new_M).  If the LP fails or no improvement is possible,
    the original f and M are returned unchanged.
    """
    n = f.shape[0]
    # Build convolution matrix A (A @ g = f * g)
    A = build_conv_matrix(f)

    # Linear program: maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Solve with HiGHS (fast for dense problems)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': max(0.0, time_deadline - time.time())})
    if not res.success:
        return f, M

    g = res.x
    if g.sum() == 0.0:
        return f, M

    # Rescale to a probability vector (sum = 1)
    g = g / g.sum()

    # Mix with current f
    f_new = (1.0 - t) * f + t * g
    f_new = project_to_simplex(f_new)   # safeguard against numerical drift

    # Compute new upper bound
    M_new = compute_score(f_new) / (2.0 * n)   # because compute_score = 2n * M_new
    return f_new, M_new


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (fine polishing)
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 300_000,
                     step0: float = 0.04) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the maximising shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises `evaluate_sequence`.  The routine respects a 1000 s
    wall‑clock budget and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Load a fallback construction if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Random initial restarts (length 1024) – cheap coarse search
    # ------------------------------------------------------------------
    n_init = 1024
    print(f"Coarse random‑restart phase – length = {n_init}")
    reserve_time = 150.0   # seconds reserved for later, more expensive phases
    while time.time() < deadline - reserve_time:
        a0 = random_start_dense(n_init)
        scr = compute_score(a0)
        if scr < best_val:
            best_val = scr
            best_seq = a0.tolist()
            print(f"  New best from random start: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. If we have a starting vector, use LP refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        f = np.array(best_seq, dtype=float)
        n = f.shape[0]
        # Ensure exact simplex (sum = 1)
        f = project_to_simplex(f)

        # Current bound M = max(conv(f,f))
        cur_M = compute_score(f) / (2.0 * n)

        print("\nLP refinement loop")
        lp_iterations = 0
        t_mix = 0.2          # initial mixing weight
        while time.time() < deadline - 100.0:
            lp_iterations += 1
            f_new, M_new = lp_improve(f, cur_M, t_mix, deadline)
            if M_new < cur_M - 1e-8:
                # improvement achieved
                f, cur_M = f_new, M_new
                cur_score = 2.0 * n * cur_M
                if cur_score < best_val:
                    best_val = cur_score
                    best_seq = f.tolist()
                    print(f"  LP iter {lp_iterations}: improved score = {best_val:.6f}")
                # keep same mixing weight
            else:
                # no progress – halve the step size
                t_mix *= 0.5
                if t_mix < 1e-4:
                    break   # further mixing would be ineffective

        # ------------------------------------------------------------------
        # 3. Sub‑gradient polishing (projected) on the best LP result
        # ------------------------------------------------------------------
        if best_seq is not None and time.time() < deadline - 30.0:
            print("\nSub‑gradient polishing")
            refined_a, refined_score = subgrad_optimize(
                np.array(best_seq, dtype=float),
                time_deadline=deadline - 20.0,
                max_iter=250_000,
                step0=0.04,
            )
            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_a.tolist()
                print(f"  Polished score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final fallback – simple uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Final official evaluation (the harness will call evaluate_sequence again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return plain Python list as required by the harness
    return best_seq
