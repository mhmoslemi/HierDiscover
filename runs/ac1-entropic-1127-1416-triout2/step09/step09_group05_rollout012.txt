"""
Propose a step‑height sequence that minimises `evaluate_sequence`.

Algorithm
---------
1. Choose a reasonable length `n` (more points give more flexibility).
2. Generate many simple candidates:
   – exponential decay, power‑law decay, their symmetric counterparts,
   – a mixed exponential × power family,
   – a few random Dirichlet vectors, and the provided fallback.
   All candidates are normalised to sum = 1 and scored exactly.
3. Take the best candidate and run a long exponentiated‑gradient
   optimisation of the smooth log‑sum‑exp surrogate of the max‑convolution.
   The temperature β is annealed to a very large value so that the
   surrogate tightly approximates the true max.
4. Polish the result with an adaptive greedy hill‑climb / simulated‑annealing
   loop until the deadline.  The move set consists of swaps,
   small mass transfers and occasional global perturbations; the step size
   is automatically increased when a move succeeds and decreased otherwise.
5. Return the best sequence found (as a plain Python list of floats).
All phases respect the 1000 s wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (float64)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the harness for a vector `a` that satisfies
    sum(a)=1.  It returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # degenerate case – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parameterised families (all returned *already* normalised to sum = 1)
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def double_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric sum of two exponentials (left‐ and right‐hand side)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return w / w.sum()


def u_power_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped power law: weight grows towards both ends."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    return w / w.sum()


def mixed_exp_power_shape(n: int, lam: float, p: float) -> np.ndarray:
    """Product of exponential decay and power‑law decay (symmetric)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) * ((i + 1) ** (-p))
    # enforce symmetry by adding the reversed copy
    w = w + w[::-1]
    return w / w.sum()


# ----------------------------------------------------------------------
# Light‑weight grid search over all families
# ----------------------------------------------------------------------


def scan_families(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """Exhaustive scan of the elementary families and a few random vectors."""
    best_a = None
    best_sc = float("inf")

    # exponential λ grid (including λ = 0 → uniform)
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -2, 200)))
    for lam in lam_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # power‑law p grid (including p = 0 → uniform)
    p_grid = np.linspace(0.0, 6.0, 200)
    for p in p_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # symmetric double exponential
    for lam in lam_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        a = double_exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # U‑shaped power law
    for p in p_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        a = u_power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # mixed exponential × power
    for lam in np.logspace(-5, -2, 80):
        for p in np.linspace(0.0, 4.0, 80):
            if time.time() >= deadline:
                return best_a, best_sc
            a = mixed_exp_power_shape(n, lam, p)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # a few random Dirichlet draws (α=1)
    for _ in range(5):
        if time.time() >= deadline:
            break
        a = np.random.rand(n)
        a /= a.sum()
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation of the smooth max‑convolution
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 0.5,
    beta_end: float = 10_000.0,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the log‑sum‑exp surrogate.
    β is exponentially annealed from `beta_start` to `beta_end`.
    The learning‑rate η_t = η₀ / √t.
    Returns the best vector seen (exact score).
    """
    a = np.maximum(start_a.astype(float), 1e-14)
    a /= a.sum()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weighting (stable log‑sum‑exp)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smooth objective
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # step‑size
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update + projection back onto simplex
        a = np.maximum(a * np.exp(-eta * grad), 1e-14)
        a /= a.sum()

        # occasional exact evaluation
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final exact evaluation
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Adaptive greedy hill‑climb with occasional uphill moves
# ----------------------------------------------------------------------


def hill_climb_adaptive(
    start_seq: np.ndarray, deadline: float, max_no_imp: int = 200_000
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk that also accepts uphill moves with a diminishing
    temperature.  The move set consists of:
      – swapping two entries,
      – transferring a fraction of mass from i to j,
      – occasional global perturbation (Gaussian noise).
    The step scale grows when moves succeed, shrinks otherwise.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    no_improve = 0
    temperature = 0.005  # start temperature for simulated‑annealing
    temp_decay = 0.99995  # exponential decay per iteration

    while time.time() < deadline:
        # propose a candidate from the current best (not from a)
        cand = best_a.copy()

        move_type = np.random.rand()
        if move_type < 0.4:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.8:
            # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        else:
            # global perturbation (small Gaussian noise)
            noise = np.random.normal(scale=step_scale, size=n)
            cand = cand + noise
            cand = np.clip(cand, 0.0, None)

        # maintain feasibility
        cand = project_to_simplex(cand)

        sc = compute_score(cand)

        # accept if improvement, otherwise maybe accept uphill
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
            temperature = max(temperature * 0.9, 1e-6)  # cool a bit slower on success
        else:
            # simulated‑annealing acceptance
            delta = sc - best_sc
            prob = np.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                # accept the worse move as new current *but not as best*
                best_a = cand  # we allow the walk to move away
                step_scale = max(step_scale * 0.97, 1e-4)
            no_improve += 1
            temperature *= temp_decay

        # periodic global perturbation if we are stuck for a while
        if no_improve >= max_no_imp:
            # perturb the best solution and try again
            perturb = np.random.normal(scale=step_scale, size=n)
            cand = best_a + perturb
            cand = np.clip(cand, 0.0, None)
            cand = project_to_simplex(cand)
            sc = compute_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand
                step_scale = min(step_scale * 1.1, 0.5)
            no_improve = 0

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑height vector that minimises `evaluate_sequence`.
    The routine combines a thorough grid scan, exponentiated‑gradient refinement,
    and an adaptive hill‑climb.  It respects the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a workable length (larger n gives more freedom but costs more)
    # ------------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback – a previously found construction, if present
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Scan the elementary families (with a modest time reserve)
    # ------------------------------------------------------------------
    scan_deadline = deadline - 300.0  # leave ~5 minutes for the refinements
    a_scan, sc_scan = scan_families(n, scan_deadline)
    if a_scan is not None and sc_scan < best_score:
        best_score = sc_scan
        best_seq = a_scan.tolist()
        print(f"[{time.time():.1f}] best after family scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Gradient‑based refinement (smooth log‑sum‑exp)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting exponentiated‑gradient")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=600_000,
            beta_start=0.5,
            beta_end=20_000.0,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final polishing with adaptive hill‑climb / simulated annealing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb_adaptive(
            np.array(best_seq, dtype=float),
            deadline,
            max_no_imp=250_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback to a short uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
