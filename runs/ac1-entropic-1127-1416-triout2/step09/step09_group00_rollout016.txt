import scipy.sparse as sp

def lp_optimize(a, M):
    n = a.shape[0]
    # Build sparse matrix A.
    # For each offset d in 0..n-1: diag values = a[d]; add to A with offset d.
    diags = [np.full(n - d, a[d]) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    # Constraints: A @ g <= M
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs-ds', options={'presolve': True, 'time_limit': remaining_time})
    if res.success:
        g = res.x
        # Rounding issues: if sum(g) is near 0 fallback.
        if g.sum() > 1e-12:
            return g / g.sum()
    return None

"""
Improved search for the step‑function that minimises

    2·n·max(convolve(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Broad analytic & random sampling** – many parametric families
   (including a new Beta‑shaped family) are scanned over several problem
   sizes.  The best vector found so far is kept.

2. **Linear‑programming refinement** – given the current vector `f`,
   the LP

        maximise   1ᵀ g
        subject to  f * g ≤ M   (point‑wise)
                    g ≥ 0

   where `M = max(f*f)`, produces a direction `g` that fills the “holes”
   left by `f`.  A small convex combination `f←(1‑t)·f + t·g` is accepted
   only if the new peak of `f*f` does not increase.  The mixing factor
   `t` is adaptively reduced when a step would raise the maximum.

   This step exploits the exact linear nature of the convolution in the
   second argument and typically lowers the objective considerably with
   only a few LP solves (each LP has ≤ 2 n‑1 constraints).

3. **Fine‑grained smooth optimisation & stochastic hill‑climb** – a
   smoothed‑max (log‑sum‑exp) objective is minimised with Adam on the
   soft‑max parametrisation, followed by a cheap projected sub‑gradient
   descent and a final simulated‑annealing walk.

All phases respect the global 1000 s budget.  The routine returns a plain
Python list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Convex mixture of the uniform vector and the exponential U‑shape.
    lam∈[0,1] is the weight of the uniform component.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a


def beta_shape_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Generalised beta shape:
        a_i ∝ (i+½)^p · (n‑½‑i)^q .
    Allows both U‑shaped (negative exponents) and bell‑shaped (positive).
    """
    i = np.arange(n, dtype=np.float64)
    log_w = p * np.log(i + 0.5) + q * np.log(n - 0.5 - i)
    w = np.exp(log_w - np.max(log_w))  # stability
    a = w / w.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform prob. vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))

    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a

    """Sample a random exponential‑U shape."""

# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 400.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (derivative of Σ w_k (a*a)_k )
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# Projected sub‑gradient descent (cheap)

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient method for 2·n·max(a*a) with diminishing steps."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑entry (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Long random walk on the simplex (swap / mass‑transfer moves)."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the new ingredient)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n convolution matrix A such that
        (A @ g) == convolution_fft(a, g)
    for any non‑negative vector g.
    """
    n = a.shape[0]
    # each diagonal d (0…n‑1) is constant a[d]
    diags = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)          # 0,1,…,n‑1
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def lp_step(current: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀ g   subject to   current * g ≤ M,  g ≥ 0 .
    Returns a normalised solution (sum = 1) or None on failure.
    """
    n = current.shape[0]
    A = build_conv_matrix(current)                # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)  # right‑hand side

    # linprog works with minimisation; we maximise sum(g) → minimise –sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as e:
        print(f"[{time.time():.1f}] LP exception: {e}")
        return None

    if not res.success:
        # infeasibility should not happen (g = current is always feasible)
        print(f"[{time.time():.1f}] LP failed: {res.message}")
        return None

    g = res.x
    total = g.sum()
    if total < 1e-12:
        return None
    return g / total   # bring back to the probability simplex


def lp_refine(start_a: np.ndarray,
              deadline: float,
              max_iter: int = 30,
              init_t: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve a vector by the LP step described above.
    The mixing factor `t` is reduced if a trial would raise the peak.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = init_t
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        M = convolution_fft(a, a).max()
        g = lp_step(a, M)
        if g is None:
            break

        # try a small convex combination; shrink t if the move is not
        # improving the peak.
        accepted = False
        trial_t = t
        while trial_t > 1e-4:
            cand = (1.0 - trial_t) * a + trial_t * g
            cand = project_to_simplex(cand)
            M_new = convolution_fft(cand, cand).max()
            if M_new <= M * (1.0 + 1e-12):      # non‑increase
                accepted = True
                break
            trial_t *= 0.5

        if not accepted:
            # cannot improve further – stop the LP refinement.
            break

        a = cand
        # occasional symmetrisation (helps for symmetric optima)
        if it % 5 == 0:
            a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        new_score = compute_score(a)
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP‑refine iter {it}, best = {best_score:.6f}")

        # slowly decay t so we take finer steps later
        t = max(t * 0.93, 0.01)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from previous run
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families + random starts (same spirit as the reference)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]   # a few sizes – larger ones are more expensive
    for n in lengths:
        if time.time() >= deadline - 180.0:
            break
        print(f"[{time.time():.1f}] exploring length n = {n}")

        # --- symmetric product (α) ------------------------------------
        for alpha in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        # --- left‑heavy power law (β) --------------------------------
        for beta in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (β={beta:.2f}) = {best_score:.6f}")

        # --- exponential U‑shape (γ) ---------------------------------
        for gamma in np.linspace(0.0, 6.0, 31):
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (γ={gamma:.2f}) = {best_score:.6f}")

        # --- mixed uniform / U‑shape ---------------------------------
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for gamma in np.linspace(0.0, 6.0, 13):
                if time.time() >= deadline - 30.0:
                    break
                vec = mixed_uniform_u_family(n, lam, gamma)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

        # --- new beta‑shape family (p,q) -----------------------------
        for p in np.linspace(-4.0, 4.0, 41):
            for q in np.linspace(-4.0, 4.0, 41):
                if time.time() >= deadline - 30.0:
                    break
                vec = beta_shape_family(n, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (βsh p={p:.2f},q={q:.2f}) = {best_score:.6f}")

        # --- random monotone draws ------------------------------------
        for _ in range(60):
            if time.time() >= deadline - 30.0:
                break
            vec = random_monotone_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

        # --- random sparse draws --------------------------------------
        for _ in range(60):
            if time.time() >= deadline - 30.0:
                break
            vec = random_sparse_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑sparse) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Linear‑programming based refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting LP‑based refinement (length {len(best_seq)})")
        a_start = np.array(best_seq, dtype=np.float64)
        # limit number of LP iterations to keep wall‑clock budget reasonable
        a_lp, sc_lp = lp_refine(a_start, deadline - 80.0, max_iter=30, init_t=0.12)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.tolist()
            print(f"[{time.time():.1f}] after LP‑refine best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑max (Adam) fine‑tuning
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        print(f"[{time.time():.1f}] starting Adam smoothing")
        a_start = np.array(best_seq, dtype=np.float64)
        a_adam, sc_adam = adam_smooth_optimize(
            a_start,
            deadline - 60.0,
            max_iters=250_000,
            beta_start=4.0,
            beta_end=800.0,
            lr=0.016,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent (very cheap per iteration)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=2_000_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation test
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a modest uniform vector if absolutely nothing found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
