def generate_candidates(n):
    cand_list = []

    # uniform
    a = np.full(n, 1.0/n)
    cand_list.append(('uniform', a))

    # central binomial
    a = np.empty(n)
    a[0] = 1.0
    for i in range(1,n):
        a[i] = a[i-1]*(2*i-1)/(2*i)
    a = a / a.sum()  # ensure sum=1
    cand_list.append(('central', a))

    # power law
    for alpha in np.linspace(0.2,2.0,9):
        a = np.arange(1, n+1, dtype=float) ** (-alpha)
        a = a / a.sum()
        cand_list.append((f'power_{alpha:.2f}', a))

    # symmetrized
    for alpha in np.linspace(0.2,2.0,5):
        a_left = np.arange(1, n+1, dtype=float) ** (-alpha)
        a = a_left + a_left[::-1]
        a = a / a.sum()
        cand_list.append((f'sym_power_{alpha:.2f}', a))

    # exponential
    for r in [0.995, 0.997, 0.999, 0.9995]:
        a = r ** np.arange(n, dtype=float)
        a = a / a.sum()
        cand_list.append((f'exp_{r}', a))

    # random Dirichlet
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        a = np.random.dirichlet(np.full(n, conc))
        cand_list.append((f'diri_{conc}', a))

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The score is    J(a) = 2·n·max(conv(a,a)) / (sum(a)²) .
Because the objective is scale‑invariant we can restrict to vectors that
lie on the probability simplex (sum = 1).  The problem is to minimise the
∞‑norm of the self‑convolution of a probability distribution.

Strategy
--------
1. **Rich set of initial shapes** – uniform, central‑binomial,
   power‑law, symmetrised power‑law, exponential and a few Dirichlet
   samples.  All are projected onto the simplex and the best few are
   retained.

2. **Smooth optimisation** – minimise a log‑sum‑exp surrogate of the
   max‑convolution with Adam.  The temperature (β) is annealed from a
   small value to a large one, so the algorithm first smooths the
   landscape and then focuses on the true peak.

3. **Exact sub‑gradient descent** – after the surrogate phase we
   switch to the exact max‑convolution and perform a projected
   sub‑gradient descent with a diminishing steps‑size and occasional
   jitter to escape plateaus.

4. **Stochastic hill‑climb** – cheap local moves (swap or fractional
   mass transfer) with simplex projection polish the final result.

All stages respect the global wall‑clock budget (default 1000 s).  The
best sequence encountered during the whole run is returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the simplex (sum = 1).

    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        # zero vector → uniform simplex point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimiser – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.

    The smoothing uses a log‑sum‑exp with temperature β which is
    annealed from ``beta_start`` to ``beta_end``.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing parameter β exponentially
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smoothed objective
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a = project_to_simplex(a - update)

        if it % 250 == 0:   # occasional progress check
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to help escape flat regions
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 800_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score

        max_hc = int(min(800_000, remaining * 180))
        if max_hc > 0:
            final_seq, final_sc = hill_climb_permutation(
                best_seq,
                time_deadline=deadline,
                max_steps=max_hc,
            )
            if final_sc < best_score:
                best_score = final_sc
                best_seq = np.array(final_seq)
                print(f"[hill‑climb] improved to {final_sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Final safeguard – if nothing succeeded return a tiny uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(256, 1.0 / 256)
        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # 3. Final evaluation through the provided black‑box
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nFinished search. Returned score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq.tolist()
