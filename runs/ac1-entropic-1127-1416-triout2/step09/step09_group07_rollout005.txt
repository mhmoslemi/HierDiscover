"""
Improved search for a step‑function minimizing
    2·n·max(a⊛a) / (∑a)²

The algorithm works in two resolution levels.

1. **Small scale (n≈256)**
   * Scan a few analytic families (power‑law & Gaussian) and keep the
     best on the probability simplex.
   * Run a short sequence of linear‑programming refinements.
     For the current vector f we solve
         max ∑g   s.t.   (f∗g) ≤ M ,  g≥0,
     where M = max(f∗f).  The LP is tiny (≤ 2·n constraints) and
     is solved with ``scipy.optimize.linprog``.  The optimiser
     returns a new candidate g which is mixed with f.
   * A fast stochastic hill‑climb (single‑source → single‑dest transfer)
     finishes the small‑scale stage.

2. **Upscaling & polish (large n, default 8192)**
   * The small‑scale optimum is up‑sampled by simple repetition
     (each entry is repeated ``factor`` times and divided by the
     factor so that the total mass stays 1).

     convolution updates (O(n) per move) is performed until the
     time budget expires.  Both improving moves and occasional
     uphill moves (Metropolis rule) are allowed, which helps

All phases respect the 1000 s wall‑clock limit and the best

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector that already satisfies
    ``np.isclose(a.sum(), 1)``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def powerlaw_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{-α}  (the family used in the baseline)."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·(i-(n‑1)/2)²)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    w = np.exp(-gamma * (i - center) ** 2)
    a = w / w.sum()
    return a


def random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with all ones)."""
    x = rng.random(n)
    return x / x.sum()


def scan_analytic_start(n: int, deadline: float) -> Tuple[np.ndarray, float]:
    """Search a modest grid of analytic families and keep the best."""
    best_a = None
    best_sc = float("inf")
    rng = np.random.default_rng(0)

def lp_refine(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve    max ∑g   s.t. (f∗g) ≤ M , g ≥ 0.
    Return the optimiser g (not normalised).
    """
    if linprog is None:                 # pragma: no cover
        return f.copy()                 # fall‑back – do nothing

    n = f.shape[0]
    conv_len = 2 * n - 1

    # Build Toeplitz‑like matrix A where A[k, j] = f[k‑j] if in range.
    rows = np.arange(conv_len)[:, None]          # (conv_len,1)
    cols = np.arange(n)                          # (n,)
    diff = rows - cols                            # broadcast → (conv_len,n)
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((conv_len, n), dtype=float)
    A[mask] = f[diff[mask]]

    # Linear program: minimise -∑g  (equivalent to maximising ∑g)
    c = -np.ones(n, dtype=float)
    b_ub = np.full(conv_len, M, dtype=float)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={"presolve": True})
    if not res.success:
        # In the rare event the LP fails we simply return the current vector.
        return f.copy()
    return res.x


def lp_iteration_loop(start_a: np.ndarray,
                      deadline: float,
                      max_iters: int = 30) -> Tuple[np.ndarray, float]:
    """
    Perform a small number of LP‑based refinement steps.
    Each step mixes the current vector with the LP‑solution g_norm.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        M = 2.0 * a.shape[0] * float(convolution_fft(a, a).max()) / (2.0 *
                a.shape[0])  # just the current max of a⊛a, i.e. max_b
        # Actually we only need the raw max_b (not multiplied by 2·n)
        max_b = convolution_fft(a, a).max()
        M = max_b

        g = lp_refine(a, M)
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g

        # Mix – moderate step size
        tau = 0.5
        a = (1.0 - tau) * a + tau * g_norm
        # Re‑project onto simplex (numerical drift)
        a = a / a.sum()

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iteration {it}: new best = {best_sc:.6f}")

    return best_a, best_sc

            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
        else:
            # optional tiny decay of temperature (already handled by schedule)
            pass

        steps += 1
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine follows a two‑resolution
    strategy: a quick optimisation on a small problem (≈256 entries)
    with analytic starts, LP refinements and a short hill‑climb,
    followed by up‑sampling and a long simulated‑annealing hill‑climb
    on the full resolution (default 8192).  All phases respect the
    1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. Choose problem sizes based on the remaining budget
    # ------------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n_big = 8192
    elif remaining > 500:
        n_big = 4096
    else:
        n_big = 2048
    n_small = 256                     # fixed small resolution
    factor = n_big // n_small
    print(f"[{time.time():.1f}] big n = {n_big}, small n = {n_small}, factor = {factor}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Small‑scale analytic start
    # ------------------------------------------------------------------
    sub_deadline = min(deadline, start_time + 30.0)
    a_small, sc_small = scan_analytic_start(n_small, sub_deadline)
    if a_small is None:
        # fall back to uniform if nothing was found (very unlikely)
        a_small = np.full(n_small, 1.0 / n_small)
        sc_small = compute_score(a_small)
    best_seq = a_small.tolist()
    best_score = sc_small
    print(f"[{time.time():.1f}] analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. LP refinement (tiny problem)
    # ------------------------------------------------------------------
    if linprog is not None and time.time() < deadline - 200.0:
        lp_deadline = deadline - 150.0
        a_small, sc_small = lp_iteration_loop(a_small, lp_deadline, max_iters=30)
        if sc_small < best_score:
            best_score = sc_small
            best_seq = a_small.tolist()
            print(f"[{time.time():.1f}] after LP best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Small‑scale hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        hc_deadline = deadline - 130.0
        a_small, sc_small = hill_climb(a_small, hc_deadline,
                                       temperature_start=0.03,
                                       temperature_end=1e-5)
        if sc_small < best_score:
            best_score = sc_small
            best_seq = a_small.tolist()
            print(f"[{time.time():.1f}] after small hill‑climb best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Upsample to the large resolution
    # ------------------------------------------------------------------
    a_big = np.repeat(a_small, factor) / factor
    # Ensure exact normalisation (tiny float drift)
    a_big = a_big / a_big.sum()
    cur_score = compute_score(a_big)
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a_big.tolist()
        print(f"[{time.time():.1f}] after up‑sampling score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Large‑scale stochastic hill‑climb (majority of the time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        final_deadline = deadline
        a_big, sc_big = hill_climb(a_big, final_deadline,
                                   temperature_start=0.02,
                                   temperature_end=1e-6)
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.tolist()
            print(f"[{time.time():.1f}] final hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Safety net – fall back to uniform if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7. Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
