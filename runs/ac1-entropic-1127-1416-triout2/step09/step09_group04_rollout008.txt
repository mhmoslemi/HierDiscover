def build_constraint_matrix(f):
    n = len(f)
    rows = 2*n - 1
    from scipy import sparse
    data = []
    row_ind = []
    col_ind = []
    for i in range(n):
        # contributions of f_i to rows i..i+n-1 for each column j
        # For each j in 0..n-1, row = i + j, col = j, value = f[i]
        # So for each i, we add n entries.

        row_ind.extend(rows_i.tolist())
        col_ind.extend(cols_i.tolist())
    A = sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

"""
Propose a step‑function (non‑negative vector) minimising `evaluate_sequence`.

Algorithm
---------
1. Initialise the vector.
   * If a global `height_sequence_1` is available it is up‑sampled to a
     convenient length (2048) and normalised.
   * Otherwise start from a random Dirichlet vector.

2. Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The surrogate temperature β is annealed from a small to a very large value,
   making the gradient converge to the true sub‑gradient of the max‑operator.
   Each step is projected onto the probability simplex.

3. Exact sub‑gradient descent on the true objective.
   All indices attaining the maximal convolution value are used (averaged)
   to compute a robust descent direction.

4. A handful of linear‑programming “directional” moves.
   The current vector is down‑sampled by a factor two, an LP maximises the total
   mass of a new vector `b` while keeping the cross‑convolution `f*b` bounded by
   the current maximal self‑convolution.  `b` is up‑sampled back, rescaled to the
   original ℓ₁‑norm and blended with the current iterate.

5. Simulated‑annealing hill‑climbing until the wall‑clock budget expires.
   Random swaps or small mass‑transfer moves are accepted according to a
   temperature schedule; improving moves are always kept.

All phases respect a 1000 s wall‑clock budget, keep the candidate on the
probability simplex (hence the objective is scaling‑invariant) and finally
return a plain Python list of floats.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * b using real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under Σ a_i = 1.
    Returns 2·n·max(convolution(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """Smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Pick one of the three generators at random."""
    r = np.random.rand()
    if r < 0.4:
        return random_start_dense(n)
    elif r < 0.7:
        return random_start_sparse(n)
    else:
        return random_start_gaussian(n)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimiser (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a differentiable soft‑max surrogate of max(conv(a,a))
    using Adam with simplex projection.
    Returns the best exact (non‑smoothed) score seen.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # linear annealing of β in log‑space
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        # stable soft‑max weights
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of weighted sum = 2 * (a_rev * w)[centered]
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a = project_to_simplex(a - update)

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Multi‑max exact sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_multi_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method on the exact max‑convolution objective.
    Uses an averaged sub‑gradient over all maximising indices.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # indices attaining (or nearly attaining) the max
        eps = 1e-12
        K = np.where(conv >= maxc - eps)[0]
        w = np.zeros_like(conv)
        w[K] = 1.0 / max(1, len(K))

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * grad)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # tiny jitter every few thousand iterations to avoid stagnation
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

def downsample_pairwise(a: np.ndarray) -> np.ndarray:
    """
    Reduce length by a factor of two: a[i] := a[2i] + a[2i+1].
    """
    return a[0::2] + a[1::2]


def upsample_pairwise(b: np.ndarray) -> np.ndarray:
    """
    Inverse of `downsample_pairwise`.  Each entry is split evenly
    into two adjacent slots.
    """
    return np.repeat(b / 2.0, 2)


def lp_improve_on_halfsize(a_half: np.ndarray, time_deadline: float) -> tuple[np.ndarray, float]:
    """
    Solve the LP:
        maximise   Σ b_i
        subject to   (a_half * b)_k ≤ M   for all k
                     b_i ≥ 0,
    where M = max(a_half * a_half).  Return the up‑sampled improved vector.
    """
    n = a_half.shape[0]
    conv_self = convolution_fft(a_half, a_half)
    M = float(conv_self.max())

    rows = 2 * n - 1
    # Constraint matrix: each column i contributes a_half[i] to rows i … i+n‑1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i : i + n, i] = a_half[i]

    b_ub = np.full(rows, M)

    # maximise Σ b_i  →  minimise -Σ b_i
    c = -np.ones(n)

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # fall back to the original half‑size vector
        return a_half, compute_score(a_half)

    b_half = res.x
    # up‑sample back to original size
    b_full = upsample_pairwise(b_half)

    # Rescale to preserve the ℓ₁‑norm
    sum_a = a_half.sum() * 2.0  # because a_half was down‑sampled from a full‑size vector
    sum_b = b_full.sum()
    if sum_b == 0:
        return a_half, compute_score(a_half)
    g = b_full * (sum_a / sum_b)

    # Try a few convex blends
    best = a_half  # placeholder, will be overwritten below
    best_score = np.inf
    for t in (0.3, 0.6, 1.0):
        cand = (1 - t) * a_half * 2.0 + t * g  # bring back to full size (2·n entries)
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
    return best, best_score


def lp_iterative_improve(
    init_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the half‑size LP improvement a few times.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break
        # ensure even length
        if a.shape[0] % 2 != 0:
            a = np.append(a, 0.0)
        a_half = downsample_pairwise(a)
        a_new, new_score = lp_improve_on_halfsize(a_half, time_deadline)
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"LP improvement {it + 1}: new score = {best_score:.6f}")
        else:
            # no progress → stop early
            break
    return best_a, best_score

# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int | None = None,
) -> tuple[list[float], float]:
    """
    Random local modifications (swap or mass transfer) with an annealing schedule.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score

def upsample_sequence(seq: list[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two adjacent entries a_i/2.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)
    return up

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects a 1000 s wall‑clock budget and
    returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for harness overhead
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Initialise – use the known construction if available
    # --------------------------------------------------------------
    try:
        fallback_seq = np.array([float(x) for x in height_sequence_1], dtype=float)
        # Bring it to a convenient length (2048) by repeated up‑sampling if needed
        while fallback_seq.shape[0] < 2048:
            fallback_seq = upsample_sequence(fallback_seq.tolist())
        if fallback_seq.shape[0] > 2048:
            fallback_seq = fallback_seq[:2048]
        fallback_seq = project_to_simplex(fallback_seq)
        best_vec = fallback_seq.copy()
        best_score = compute_score(best_vec)
        print(f"Initial fallback score = {best_score:.6f}")
    except Exception:
        # No fallback – start from a random dense vector of length 2048
        n0 = 2048
        best_vec = random_start_dense(n0)
        best_score = compute_score(best_vec)
        print(f"Random start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Smooth Adam optimisation (log‑sum‑exp surrogate)
    # --------------------------------------------------------------
    phase_deadline = min(deadline - 600.0, time.time() + 250.0)  # allocate ~250 s
    if time.time() < phase_deadline:
        a_opt, _ = smooth_gradient_optimize(
            start_a=best_vec,
            time_deadline=phase_deadline,
            max_iters=300_000,
            beta_start=3.0,
            beta_end=2500.0,
            step0=0.8,
        )
        score_opt = compute_score(a_opt)
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            print(f"After smooth Adam optimisation: score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Exact multi‑max sub‑gradient descent
    # --------------------------------------------------------------
    phase_deadline = min(deadline - 350.0, time.time() + 300.0)  # allocate ~300 s
    if time.time() < phase_deadline:
        a_sub, _ = subgrad_multi_optimize(
            init_a=best_vec,
            time_deadline=phase_deadline,
            max_iter=500_000,
            step0=0.07,
        )
        score_sub = compute_score(a_sub)
        if score_sub < best_score:
            best_score = score_sub
            best_vec = a_sub.copy()
            print(f"After multi‑max sub‑gradient descent: score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Linear‑programming directional improvement (few passes)
    # --------------------------------------------------------------
    phase_deadline = min(deadline - 150.0, time.time() + 120.0)  # allocate ~120 s
    if time.time() < phase_deadline:
        a_lp, lp_score = lp_iterative_improve(
            init_a=best_vec,
            time_deadline=phase_deadline,
            max_iters=4,
        )
        if lp_score < best_score:
            best_score = lp_score
            best_vec = a_lp.copy()
            print(f"After LP directional improvement: score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Hill‑climb polishing until time runs out
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        polished_seq, polished_score = hill_climb_anneal(
            best_vec.tolist(),
            deadline,
            max_steps=None,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_vec = np.array(polished_seq)
            print(f"After hill‑climb polishing: score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final official evaluation (the harness will call it again,
    #    but we log it here for debugging purposes)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return the plain Python list expected by the harness
    return best_vec.tolist()
