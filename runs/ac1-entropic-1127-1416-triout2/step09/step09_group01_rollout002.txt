"""
Propose a step‑function sequence that minimizes `evaluate_sequence`.

Approach
--------
1. Choose a moderate length (n = 2048) – small enough for fast linear‑program
   solves while still providing a good degree of freedom.
2. Initialise a collection of candidate vectors:
   * Analytic families (symmetric product, left‑heavy power‑law,
     exponential decay) – dense grid of parameters.
   * Random monotone draws.
   The best of these forms the starting point `f`.
3. Iterative LP refinement (the classical “f–g” update):
   * Compute the current worst convolution value M = max((f*f)).
   * Solve the linear programme
        maximise   ∑ g_j
        subject to (f * g)_k ≤ M   for all k,
                   g_j ≥ 0.
     The constraints are linear in `g`; we build the (2n‑1)×n
     convolution matrix and call SciPy’s `linprog` (HiGHS).
   * Normalise `g` to have unit sum, try several mixing factors
        t ∈ {0.05,0.1,0.2,0.4,0.6,0.8}
     and keep the mixture `f ← (1‑t)f + t g` that yields the smallest
     true objective.  Symmetrise the result (`(f+rev(f))/2`) and repeat
     while the score improves and time permits.
4. A short mirror‑subgradient descent on the exact max‑objective gives a
   final polish.
5. Return the best vector found (plain Python list of floats).

All steps respect a global 1000 s wall‑clock budget; the code prints
intermediate scores for monitoring.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def objective(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` for a non‑negative vector."""
    total = a.sum()
    if total < 1e-12:
        return np.inf
    conv = convolution_fft(a, a)
    return 2.0 * a.size * conv.max() / (total ** 2)


# ----------------------------------------------------------------------
# Analytic families (return probability vectors – sum to 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # (i+0.5)*(n-0.5-i) is always positive inside the range
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    return w / w.sum()


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing probability vector."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a / a.sum()


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """Exponentiated sub‑gradient descent on 2·n·max(a*a)."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = objective(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # active shift

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)         # multiplicative update
        # renormalise to the simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = objective(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = objective(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score
