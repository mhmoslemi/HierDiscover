"""
Non‑negative step‑function optimiser.

Idea
----
1.  Use a 1‑parameter family of symmetric “arcsine‑like’’ shapes  

        a_i(α,p) ∝ ((i+α)*(n‑1‑i+α))^(‑p)      (0 ≤ i < n)

    with p≈½.  The parameter α smooths the divergence at the ends;
    α≈0 gives the classic arcsine shape (good for flattening the
    convolution), larger α moves the distribution towards uniform.
    By scanning α (and a few p‑values) we obtain an excellent
    starting point.

2.  Optionally mix the best shape with a uniform distribution
    (still on the simplex) – a cheap one‑dimensional search for the
    mixing weight λ.

3.  Refine the candidate by a projected sub‑gradient descent on the
    true objective  

        F(a)=2·n·max(conv(a,a))   (∑a_i=1, a_i≥0)

    The sub‑gradient of the active convolution entry k is  

        g_i = 2·a_{k‑i}  (if 0≤k‑i<n, else 0).

    After each step we enforce non‑negativity and renormalise (simple
    “clip‑and‑renorm’’ projection).  Symmetry a_i = a_{n‑1‑i} is kept
    explicitly because the optimal solution is symmetric.

4.  A tiny stochastic hill‑climb is run at the end (swap or fractional
    mass transfer) to escape possible plateaus.

All heavy work (convolution, gradient) is performed with an FFT‑based
convolution which is O(n log n) and fast enough for n≈4096.  The
routine respects the wall‑clock budget (default 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Simple clip‑and‑renorm projection onto the probability simplex.
    Guarantees v_i ≥ 0 and ∑v_i = 1.
    """
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback (should never happen)
    return np.full_like(v, 1.0 / v.shape[0])


def enforce_symmetry(v: np.ndarray) -> np.ndarray:
    """Replace v by (v+v[::-1])/2 – preserves the sum."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (projected)
# ----------------------------------------------------------------------


def subgrad_optimize_simple(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Returns the best vector seen and its score.
    """
    n = a.shape[0]
    a = enforce_symmetry(project_onto_simplex(a))
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_score = 2.0 * n * float(conv.max())

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution and its worst index
        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        k = int(conv.argmax())

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step size (decays like 1/√t)
        step = step0 / np.sqrt(t)
        a = a - step * g

        # simple projection + symmetry
        a = project_onto_simplex(a)
        a = enforce_symmetry(a)
        a = project_onto_simplex(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_onto_simplex(a)
            a = enforce_symmetry(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climb (swap / fractional transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic moves (swap or fractional transfer) with
    projection onto the simplex.  Improves the best score found so far.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    step_frac = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_frac * best_a[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_onto_simplex(cand)
        cand = enforce_symmetry(cand)
        cand = project_onto_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac *= 0.97

        max_steps -= 1

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choose a problem size – larger gives a better theoretical bound,
    # but we must stay well within the time budget.
    # ------------------------------------------------------------------
    # Roughly allocate half the time to the local optimisation phase.
    time_for_shape = (deadline - time.time()) * 0.3
    if time_for_shape > 20.0 and (deadline - start_time) > 300.0:
        n = 4096
    else:
        n = 2048

    # ------------------------------------------------------------------
    # 1)  Scan a small family of arcsine‑like shapes.
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=float)
    best_vec = None
    best_score = float("inf")

    p_vals = [0.45, 0.5, 0.55]          # exponent around the classic ½
    # coarse α grid (log‑spaced)
    alphas = np.geomspace(1e-3, 30.0, num=70)

    for p in p_vals:
        for alpha in alphas:
            prod = (idx + alpha) * (n - 1 - idx + alpha)
            # avoid division by zero – prod is always >0 because alpha>0
            a_raw = prod ** (-p)
            a = a_raw / a_raw.sum()          # already on the simplex
            a = enforce_symmetry(a)           # numerical safety
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_vec = a.copy()
        # ------------------------------------------------------------------
        # 2)  Refine α locally around the current best for this p.
        # ------------------------------------------------------------------
        # (only if we still have a reasonable chunk of time)
        if time.time() > deadline - 30.0:
            break
        # fine‑grid: ±20 % around best α on a linear scale
        # locate the α that produced best_vec for this p
        # (simple heuristic: re‑use the same α again with tighter spacing)
        # – we reuse the last successful alpha as a seed
        # (the loop above left `alpha` at the last used value)
        # A quick local search:
        fine_alphas = np.linspace(max(alpha * 0.6, 1e-3),
                                 alpha * 1.4, num=30)
        for a_f in fine_alphas:
            prod = (idx + a_f) * (n - 1 - idx + a_f)
            a_raw = prod ** (-p)
            a = a_raw / a_raw.sum()
            a = enforce_symmetry(a)
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_vec = a.copy()

    # ------------------------------------------------------------------
    # 3)  Mix the best shape with a uniform distribution (one‑dimensional
    #     search for the mixing weight λ).
    # ------------------------------------------------------------------
    if best_vec is not None:
        uniform = np.full(n, 1.0 / n)
        for lam in np.linspace(0.0, 1.0, 21):
            cand = lam * best_vec + (1.0 - lam) * uniform
            cand = project_onto_simplex(cand)
            cand = enforce_symmetry(cand)
            score = compute_score(cand)
            if score < best_score:
                best_score = score
                best_vec = cand.copy()

    # ------------------------------------------------------------------
    # 4)  Projected sub‑gradient refinement (if time permits).
    # ------------------------------------------------------------------
    sub_deadline = deadline - 30.0
    if best_vec is not None and time.time() < sub_deadline:
        refined_vec, refined_score = subgrad_optimize_simple(
            best_vec,
            deadline=sub_deadline,
            max_iter=600_000,
            step0=0.09,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_vec = refined_vec.copy()

    # ------------------------------------------------------------------
    # 5)  Final stochastic hill‑climb.
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_vec is not None and time.time() < hill_deadline:
        final_vec, final_score = hill_climb(
            best_vec,
            deadline=hill_deadline,
            max_steps=300_000,
        )
        if final_score < best_score:
            best_score = final_score
            best_vec = final_vec.copy()

    # ------------------------------------------------------------------
    # 6)  Fallback – plain uniform vector (should never be needed).
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(256, 1.0 / 256)

    # ------------------------------------------------------------------
    # 7)  Report and return.
    # ------------------------------------------------------------------
    print(f"Chosen length = {n}, final reported score = {best_score:.6f}")
    return best_vec.tolist()
