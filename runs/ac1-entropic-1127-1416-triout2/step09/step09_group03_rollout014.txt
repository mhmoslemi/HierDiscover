We search for a non‑negative step‑function `a` (the list of heights) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum a_i\bigr)^2},
\qquad
a_i\ge 0,
\]

where `*` is the linear convolution.  

The algorithm works in three stages:

1. **LP‑refinement on a modest size (≈ 1024).**  
   For a current vector `f` we build the exact Toeplitz convolution matrix `A`  
   (`A·b = f * b`).  
   We solve  

   \[
   \max\;\mathbf 1^\top b\quad\text{s.t.}\;A b\le M,\;b\ge0,
   \]

   where `M = max(f*f)`.  
   The solution `b` does **not** increase the worst convolution entry.  
   By mixing `f←(1‑t)f + t b` (small `t`) and projecting back onto the probability
   simplex we strictly decrease the objective.  Repeating this step many times
   drives the shape towards the optimum.

2. **Smooth Adam optimisation on the full length (2048).**  
   After up‑sampling the refined short vector we minimise a *log‑sum‑exp*
   surrogate of the max‑convolution with Adam.  The smoothing parameter β is
   annealed from 5 to 400, giving a very accurate approximation of the true
   max while keeping the gradients stable.

3. **Exact sub‑gradient polishing + tiny stochastic hill‑climb.**  
   We finish with a projected sub‑gradient descent on the genuine max‑objective,
   followed by a cheap mass‑transfer hill‑climb that explores the neighbourhood
   of the best solution found.

All stages respect the 1000 s wall‑clock budget; the best sequence encountered
throughout the run is returned.

```python
"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The optimiser proceeds in three phases:
1. LP‑refinement on a short vector (≈1024 entries).  The convolution matrix is
   built exactly, the LP maximises the L1‑norm of a partner vector while keeping
   the worst convolution entry ≤ the current maximum.  Mixing and simplex
   projection strictly improve the objective.
2. Smooth Adam optimisation on the final length (2048).  A log‑sum‑exp
   surrogate of the max‑convolution is minimised; the smoothing parameter β
   is annealed from 5 to 400.
3. Precise sub‑gradient descent followed by a tiny hill‑climb to polish the
   result.

The best sequence found before the time limit is returned as a plain list of
floats (already projected onto the simplex, i.e. sum = 1).
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A * B
    conv = np.fft.irfft(C, n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0).
    """
    if np.all(v <= 0):
        # uniform projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution matrix (exact Toeplitz) for the LP
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ b) = conv(f, b).
    Column j contains the vector f shifted down by j rows.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    data = []
    row_ind = []
    col_ind = []
    for col in range(n):
        # each entry a[i] contributes to rows col … col+n‑1
        for offset in range(n):
            row = col + offset
            data.append(f[offset])
            row_ind.append(row)
            col_ind.append(col)
    return sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))


def solve_lp_max_sum(A: sp.csr_matrix, M: float,
                     warm_start: np.ndarray | None = None) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    b = cp.Variable(n)
    constraints = [A @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    if warm_start is not None:
        b.value = warm_start

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-7, reltol=1e-6, feastol=1e-7)
    except Exception:
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    bv = b.value
    if bv is None:
        return None

    # clip tiny negatives that may appear due to numerical noise
    return np.maximum(bv, 0.0)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimiser – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 400.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)     # first moment
    v = np.zeros_like(a)     # second moment

    best_a = a.copy()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing parameter β (log‑linear schedule)
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # log‑sum‑exp weights (numerically stable)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smooth surrogate = 2·(rev(a) * w) restricted to n entries
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (the k‑th entry of the convolution)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)
```
