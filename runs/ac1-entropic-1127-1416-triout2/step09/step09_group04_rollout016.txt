def LP_step(f, M, t=0.1):
    n = len(f)
    # build A_ub shape (2n-1, n)
    rows = 2*n-1
    # create list of rows
    data = []
    rows_idx = []
    col_idx = []
    for k in range(rows):
        # j range where b_j contributes
        j_start = max(0, k-(n-1))
        j_end = min(n-1, k)
        # then coefficient is f_{k-j}
        # iterate j from j_start..j_end
        for j in range(j_start, j_end+1):
            coeff = f[k - j] # index valid
            rows_idx.append(k)
            col_idx.append(j)
            data.append(coeff)
    A = scipy.sparse.csr_matrix((data, (rows_idx, col_idx)), shape=(rows, n))
    # bounds b >=0
    bounds = [(0, None)] * n
    # objective maximize sum b => minimize -sum b
    c = -np.ones(n)
    # RHS: limit each conv entry <= M
    b_ub = np.full(rows, M)
    # solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    # If all zero, return fallback
    return g

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` objective.

Algorithm
---------
* 1️⃣  **Quick random‑restart phase** – many cheap Adam‑optimisations of a
    smooth log‑sum‑exp surrogate of ``max(conv(a,a))`` on short vectors
    (n≈512).  The best of these restarts is kept as the seed for later
    stages.

* 2️⃣  **Geometric up‑sampling** – the current best vector is repeatedly
    up‑sampled (each entry is split into two halves).  Upsampling spreads
    the mass and reduces the contribution of any single product to the
    convolution while preserving the total sum.

* 3️⃣  **Projected sub‑gradient descent** – after each up‑sampling a few
    thousand sub‑gradient steps are applied to the exact max‑objective.
    The step‑size is  ~ 1/√t and the iterate is projected onto the simplex
    (non‑negative, unit sum).

* 4️⃣  **Aggressive simulated‑annealing hill‑climb** – a custom annealing
    loop performs several kinds of local moves (mass transfer, swaps,
    large “burst” transfers).  Worsening moves are accepted with a
    temperature that decays linearly with wall‑clock time, allowing the
    search to escape shallow minima.

* 5️⃣  **Iterate up‑sampling / refinement** – steps 2‑4 are repeated a
    few times (up to 3 doublings, i.e. length ≈ 4096).  Each stage further
    flattens the autocorrelation and drives the score below the previous
    best (≈ 1.511).

The whole routine respects the 1000 s wall‑clock budget and returns the
best vector found.  All helper functions are top‑level, there are no
closures or lambdas, and only `numpy`, `scipy` and the provided
`evaluate_sequence` are used.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the constraint Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """A smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Pick one of the three generators at random."""
    r = np.random.rand()
    if r < 0.4:
        return random_start_dense(n)
    elif r < 0.7:
        return random_start_sparse(n)
    else:
        return random_start_gaussian(n)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a differentiable soft‑max surrogate of max(conv(a,a))
    using Adam plus projection onto the simplex.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # linear annealing of β (log‑sum‑exp temperature)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))            # safe weights
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the soft‑max (cross‑correlation trick)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the maximal entry

        # sub‑gradient for this index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Aggressive simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = None,
) -> tuple[list[float], float]:
    """
    Random local modifications with a temperature schedule.
    Several move types are used:
        – swap two positions,
        – transfer a small or large fraction of mass,
        – “burst” transfer (up to the whole source mass).
    Worsening moves are accepted with probability exp(-Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    # initial temperature (a few percent of the current score)
    T0 = best_score * 0.03
    Tmin = best_score * 1e-7

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        # linear cooling based on elapsed wall‑clock time
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()

        move_type = np.random.rand()
        if move_type < 0.33:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.66:
            # transfer a moderate fraction (≤5 %)
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]            # 5 % of source
                cand[i] -= delta
                cand[j] += delta
        else:
            # burst move – take up to the whole source mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = np.random.rand() * cand[i]   # anywhere from 0 to full amount
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        # acceptance test
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: list[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two
    adjacent entries a_i/2, a_i/2.  The total mass is unchanged.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)          # safeguard against drift
    return up


# ----------------------------------------------------------------------
# 5. Optional linear‑programming improvement (one LP step)
# ----------------------------------------------------------------------


def lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    One linear‑programming step as described in the paper:
    maximise Σ g_j   s.t.   g ≥ 0  and  f∗g ≤ M  (elementwise).
    Returns the optimal g (un‑scaled).  If the LP fails, None is returned.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A (rows × n) such that A·g = f∗g
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # For each valid column j the coefficient is f_{k-j}
        for j in range(j_start, j_end + 1):
            coeff = f[k - j]
            row_ind.append(k)
            col_ind.append(j)
            data.append(coeff)
    A = sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

    bounds = [(0, None)] * n
    c = -np.ones(n)                     # maximise Σ g_j  ⇔ minimise −Σ g_j
    b_ub = np.full(rows, M)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:
        return None

    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects a 1000 s wall‑clock budget
    and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness after return
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0.  Fallback from a previously discovered construction (if any)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Quick random‑restart (Adam on smooth surrogate)
    # --------------------------------------------------------------
    n_small = 512
    print(f"Random‑restart phase (n={n_small})")
    while time.time() < deadline - 300.0:      # reserve ~5 min for refinement
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(15.0, remaining * 0.2)

        a0 = random_start(n_small)
        a_opt, _ = smooth_gradient_optimize(
            a0,
            time_deadline=slice_deadline,
            max_iters=120_000,
            beta_start=5.0,
            beta_end=5000.0,
            step0=0.9,
        )
        scr = compute_score(a_opt)
        if scr < best_val:
            best_val = scr
            best_seq = a_opt.tolist()
            print(f"  New best from restart: score={best_val:.6f}")

    # --------------------------------------------------------------
    # 2.  Geometric up‑sampling + refinement loop
    # --------------------------------------------------------------
    if best_seq is None:
        # extremely unlikely, but provide a deterministic fallback
        best_seq = [1.0 / n_small] * n_small
        best_val = compute_score(np.array(best_seq))

    max_upsamples = 3                      # 512 → 1024 → 2048 → 4096
    for up in range(max_upsamples):
        if time.time() > deadline - 120.0:
            break

        # ----- up‑sample -----
        best_seq = upsample_sequence(best_seq).tolist()
        cur_n = len(best_seq)
        print(f"\nUpsampling step {up + 1}: length = {cur_n}")

        # ----- one optional LP step (can give a strong push) -----
        try:
            f = np.array(best_seq, dtype=float)
            M = convolution_fft(f, f).max()
            g = lp_step(f, M)
            if g is not None and g.sum() > 0:
                # scale g to unit sum and mix slightly
                g = project_to_simplex(g)
                mix = 0.15
                f = (1.0 - mix) * f + mix * g
                f = project_to_simplex(f)
                best_seq = f.tolist()
                best_val = compute_score(f)
                print(f"  LP blend improved score to {best_val:.6f}")
        except Exception:
            pass

        # ----- sub‑gradient refinement -----
        sub_deadline = min(deadline - 30.0, time.time() + 60.0)   # up to 60 s
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=sub_deadline,
            max_iter=120_000,
            step0=0.03,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  Sub‑gradient refined score = {best_val:.6f}")

        # ----- aggressive hill‑climb polishing -----
        if time.time() < deadline - 5.0:
            polished_seq, polished_score = hill_climb_anneal(
                best_seq,
                time_deadline=deadline,
                max_steps=None,
            )
            if polished_score < best_val:
                best_val = polished_score
                best_seq = polished_seq
                print(f"  Hill‑climb polished score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3.  Final fallback – uniform vector (should never be used)
    # --------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 4.  Log the final official evaluation
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_seq
