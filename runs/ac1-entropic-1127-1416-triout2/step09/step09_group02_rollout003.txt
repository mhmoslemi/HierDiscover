"""
Smooth‑max gradient descent for the step‑function problem.

The objective to minimise is   2·n·max(conv(a,a)),
with a_i ≥ 0 and Σ a_i = 1 (the scaling of a does not matter).
A soft‑max approximation of the max is used:
    Sβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
Its gradient is a weighted sum of the gradients of all convolution entries,
which can be computed at O(n log n) with FFTs.
A decreasing step‑size and a gradual increase of β turn the smooth problem
into the true L∞ problem.  After the main descent a short stochastic hill–
climb is applied for final polishing.

The routine respects a wall‑clock budget (default 1000 s) and returns a
plain list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolution_fft(a: np.ndarray, b: np.ndarray, n_fft: int) -> np.ndarray:
    """
    Full linear convolution using a real FFT of size *n_fft*.
    The caller must guarantee n_fft ≥ len(a)+len(b)-1.
    """
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector that already sums to 1.
    """
    n = a.shape[0]
    n_fft = next_pow2(2 * n - 1)
    conv = convolution_fft(a, a, n_fft)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simple families for initial seeding
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sine‑squared shape a_i = sin²(π·(i+1)/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    return a / a.sum()


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p can be negative (mass at the edges) or positive (mass at the centre).
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    return w / w.sum()


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Random probability vector drawn from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Small stochastic hill‑climb (final polish)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, deadline: float, max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Greedy local‑search that only accepts moves which strictly improve the exact score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]          # swap
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        cand /= cand.sum()                                # re‑normalise
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96
        steps += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs a smooth‑max gradient descent followed by a tiny hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Choose a reasonable length – larger n gives a lower theoretical bound.
    # ------------------------------------------------------------------
    if total_time > 800.0:
        n = 4096
    else:
        n = 2048
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 0. Build a set of seed vectors (hand‑crafted + random)
    # ------------------------------------------------------------------
    seeds = [generate_deterministic(n)]
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))
    for _ in range(4):
        seeds.append(random_dirichlet(n, conc=1.0))
    for _ in range(4):
        seeds.append(random_dirichlet(n, conc=0.4))

    # fallback from previous successful construction, if compatible
    try:
        fb = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        if len(fb) == n:
            seeds.append(np.array(fb) / np.sum(fb))
    except Exception:
        pass

    # pick the best seed according to the exact objective
    best_seq = None
    best_score = np.inf
    for s in seeds:
        s = project_to_simplex(s.astype(float))
        scr = compute_score(s)
        if scr < best_score:
            best_score = scr
            best_seq = s
    print(f"Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Smooth‑max gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n_fft = next_pow2(3 * n)          # enough for linear convolution/correlation
    step0 = 0.5                       # initial step magnitude
    beta = 0.5                        # start smoothing temperature
    beta_max = 80.0
    # increase β linearly – we aim for ~10 000 iterations before β saturates
    beta_inc = (beta_max - beta) / 12000.0

    best_a = a.copy()
    best_obj = best_score
    iteration = 0

    while time.time() < deadline - 2.0:
        iteration += 1

        # ----- FFT of current a -------------------------------------------------
        a_fft = np.fft.rfft(a, n=n_fft)

        # ----- convolution a*a (needed for soft‑max weights) --------------------
        conv_full = np.fft.irfft(a_fft * a_fft, n=n_fft)[: 2 * n - 1]

        # ----- soft‑max weights over the convolution entries --------------------
        maxc = conv_full.max()
        # stability trick – shift by the maximum before exponentiation
        exp_vals = np.exp(beta * (conv_full - maxc))
        w = exp_vals / exp_vals.sum()                # shape (2n‑1,)

        # ----- gradient of the smooth objective (weighted sum of sub‑gradients)-----
        w_fft = np.fft.rfft(w, n=n_fft)
        corr_full = np.fft.irfft(a_fft * np.conj(w_fft), n=n_fft)
        grad = 2.0 * corr_full[:n]                    # ∂/∂a

        # ----- step update -------------------------------------------------------
        step = step0 / np.sqrt(iteration)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        # ----- evaluate new objective (exact max) ---------------------------------
        a_fft_new = np.fft.rfft(a_new, n=n_fft)
        conv_new = np.fft.irfft(a_fft_new * a_fft_new, n=n_fft)[: 2 * n - 1]
        obj_new = 2.0 * n * float(conv_new.max())

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"[Iter {iteration}] new best {best_obj:.6f}")

        # accept the move (pure descent)
        a = a_new

        # ----- anneal the soft‑max temperature ------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ----- occasional perturbation to avoid stagnation ------------------------
        if iteration % 5000 == 0:
            noise = np.random.randn(n) * (step0 * 0.001 / np.sqrt(iteration))
            a = project_to_simplex(a + noise)

    print(f"After smooth‑max descent best {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2. Final hill‑climb polishing (only if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        cur_a, cur_score = hill_climb(best_a, deadline, max_steps=600_000)
        if cur_score < best_obj:
            best_obj = cur_score
            best_a = cur_a
            print(f"After hill‑climb best {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the sequence (evaluate_sequence will be called by the harness)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nSearch finished – final reported score = {final_val:.6f}")
    return final_seq
