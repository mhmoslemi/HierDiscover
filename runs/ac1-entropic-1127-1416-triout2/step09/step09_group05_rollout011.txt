"""
Improved search for a step‑function with minimal evaluation score.

Algorithm
---------
1. Choose a reasonable length n (larger n gives a finer shape, but also
   more expensive convolutions).  The choice is based on the remaining
   wall‑clock budget.
2. Scan several one‑parameter families of non‑negative sequences:
   * monotone exponential   a_i ∝ exp(-λ·i)
   * monotone power‑law     a_i ∝ (i+1)^(-p)
   * symmetric exponential  a_i ∝ exp(-λ·min(i,n‑1‑i))
   * symmetric power‑law    a_i ∝ (min(i,n‑1‑i)+1)^(-p)
   All candidates are normalised to sum = 1 and the best (lowest exact
   score) is kept.
3. Starting from the best candidate, run a long *sub‑gradient mirror
   descent*.  At each iteration we compute the exact max of the
   auto‑convolution, form a sub‑gradient (uniform weight on all active
   indices where the max is attained), and perform an exponentiated‑
   gradient (KL‑mirror) update with a 1/√t step‑size schedule.
   Momentum is used to stabilise the direction.  The best vector seen
   during the run is retained.
4. Finish with a greedy hill‑climb that repeatedly moves a small packet
   of mass between two positions (or swaps two entries).  Only improving
   moves are kept, and the step‑size adapts automatically.
5. Return the best sequence found before the hard timeout.

All phases strictly respect the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector normalised to sum = 1.
    (Same as the value returned by `evaluate_sequence` for such a vector.)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def sym_exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·min(i, n‑1‑i)), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    d = np.minimum(i, n - 1 - i)
    w = np.exp(-lam * d) if lam > 0.0 else np.ones_like(d)
    return w / w.sum()


def sym_power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (min(i,n‑1‑i)+1)^(-p), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    d = np.minimum(i, n - 1 - i) + 1.0
    w = d ** (-p) if p > 0.0 else np.ones_like(d)
    return w / w.sum()


# ----------------------------------------------------------------------
# Scan families
# ----------------------------------------------------------------------


def scan_families(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """
    Exhaustively scan the four families defined above.
    Returns the best vector and its exact score (or (None, inf) if time ran out).
    """
    best_a = None
    best_sc = float("inf")

    # grids (moderate size – fast enough)
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -2, 120)))
    p_grid = np.linspace(0.0, 6.0, 120)

    families = [
        (exponential_shape, lam_grid),
        (power_shape, p_grid),
        (sym_exponential_shape, lam_grid),
        (sym_power_shape, p_grid),
    ]

    for shape_func, param_grid in families:
        for param in param_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            a = shape_func(n, param)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient mirror descent (exact max)
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.8,
    eps: float = 1e-15,
    momentum: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the exact max‑convolution (sub‑gradient version).
    a is kept on the probability simplex via exponentiated‑gradient updates.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    # momentum accumulator
    grad_acc = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]
        if active.size == 0:   # should never happen
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient of the (smooth) max‑convolution = 2 * (rev_a * w)
        grad = 2.0 * convolution_fft(a[::-1], w)

        # momentum
        grad_acc = momentum * grad_acc + (1.0 - momentum) * grad

        # step size  η_t = η₀ / √t
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update (KL‑mirror)
        # clamp exponent to avoid overflow/underflow
        delta = -eta * grad_acc
        delta = np.clip(delta, -700, 700)
        a = np.maximum(a * np.exp(delta), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 400 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (no projection – keeps sum = 1 by construction)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy local search:
        * with prob 0.5 swap two entries,
        * otherwise move a small packet of mass from i to j.
    Only improving moves are kept.  The step‑size adapts automatically.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # initial packet size (relative to current mass at i)
    step_scale = 0.12

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j and best_a[i] > 0.0:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                # protect against tiny negatives caused by rounding
                if cand[i] < 0.0:
                    cand[i] = 0.0
        # no need for a full projection – sum is preserved and all entries stay ≥0
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc, best_a = sc, cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine combines a parameter grid scan, a sub‑gradient mirror descent,
    and a final greedy hill‑climb, all respecting the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin before the hard timeout
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a workable length
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score, best_seq = fb, fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Scan simple families (including symmetric versions)
    # -------------------------------------------------
    a_scan, sc_scan = scan_families(n, deadline - 30.0)
    if a_scan is not None and sc_scan < best_score:
        best_score, best_seq = sc_scan, a_scan.tolist()
        print(f"[{time.time():.1f}] best after scan = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Sub‑gradient mirror descent (exact max)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] starting sub‑gradient optimisation")
        a_opt, sc_opt = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 200.0,
            max_iters=3_000_000,
            eta0=0.8,
            momentum=0.85,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Final hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=6_000_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
