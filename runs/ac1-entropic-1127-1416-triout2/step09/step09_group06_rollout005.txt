"""
propose_candidate – heuristic optimizer for the max‑convolution bound

The optimiser builds on the previous L‑BFGS continuation method but adds
significant upgrades:

* 1. Larger problem sizes (up to 4096 coefficients) when the time budget
  permits – more degrees of freedom often lower the bound.

* 2. Multiple diversified initialisations (central‑binomial, exponential
  decay with random rates, sparse Dirichlet draws).

* 3. A more aggressive β‑annealing schedule (β up to 2000) and a second
  refinement phase based on the Adam optimiser, which typically finds
  better minima than L‑BFGS alone.

* 4. A prolonged projected sub‑gradient polishing stage (up to a million
  iterations) to push the exact max‑convolution objective down.

* 5. A final simulated‑annealing hill‑climb that repeatedly swaps or
  transfers a tiny amount of probability mass while accepting occasional
  uphill moves; this helps escape the remaining local plateaus.

All steps respect the 1000 s wall‑clock budget and constantly monitor the
deadline.  The routine returns a plain list of non‑negative floats that
sums to 1, which is the normalised form expected by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (efficient for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector already on the
    probability simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        # all non‑positive → return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically‑stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient (combined)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = smooth approximation of max(conv) = max + log‑sum‑exp/β
        g = gradient of f w.r.t. the logits (soft‑max parametrisation).
    """
    # a lives on the simplex
    a = softmax_logits(logits)

    # 1) self‑convolution
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # 2) smooth max (log‑sum‑exp)
    exp_weights = np.exp(beta * (conv - maxc))          # centred for stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # 3) gradient w.r.t. a
    w = exp_weights / Z                                 # soft‑max weights over 2n‑1 entries
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)                # length = 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # 4) chain rule back to logits (Jacobian of soft‑max)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 9,
                       maxiter_per_step: int = 150):
    """
    Anneal the smoothing parameter β while optimising the smooth surrogate
    with L‑BFGS‑B.  Returns the best simplex vector found and its exact score.
    """
    logits = np.log(start_a + 1e-20)          # initialise logits from the simplex point
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break
        # adapt iteration budget to remaining time
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(20, int(time_left * 30)))    # ≈30 L‑BFGS calls per second

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[L‑BFGS β={beta:.1f} step={step}] new best score = {best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------

                     beta_start: float,

    """

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 1

            if time.time() >= deadline:
                return best_a, best_score

            f, grad = smooth_obj_and_grad(logits, beta)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
            t += 1

            if (_ + 1) % 10 == 0:
                a = softmax_logits(logits)
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Adam β={beta:.1f}] new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 800_000,
                     step0: float = 0.07):
    """
    Projected sub‑gradient descent on the exact 2·n·max(conv) objective.
    The routine stops early if the time budget is exhausted.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if t % 50000 == 0:
                print(f"[Sub‑grad iter {t}] best score = {best_score:.6f}")

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(start_a: np.ndarray,
                        deadline: float,
                        init_temp: float = 0.02,
                        min_temp: float = 1e-5,
                        step_scale_start: float = 0.05):
    """
    Very cheap simulated‑annealing hill‑climb that repeatedly transfers a
    small amount of probability mass between two random indices.
    Accepts uphill moves with a temperature that decays linearly.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    start_time = time.time()
    step_scale = step_scale_start

    while time.time() < deadline:
        # linear cooling
        fraction = (deadline - time.time()) / (deadline - start_time)
        T = max(min_temp, init_temp * fraction)

        # propose a simple transfer
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] = max(cand[i] - delta, 0.0)
        cand[j] = cand[j] + delta
        # short‑circuit projection – the total mass is still 1
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        delta_score = cur_score - best_score

        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(T, 1e-12)):
            a = cand
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand.copy()

        # slowly shrink the step size
        if np.random.rand() < 0.01:
            step_scale = max(step_scale * 0.94, 1e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic start – truncated central‑binomial coefficients
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape (central binomial coefficients) normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


def exponential_sequence(n: int, rate: float) -> np.ndarray:
    """Simple exponential decay a_i ∝ exp(−rate·i), normalised to sum = 1."""
    x = np.exp(-rate * np.arange(n, dtype=float))
    return x / x.sum()


def random_dirichlet_sequence(n: int, concentration: float = 0.5) -> np.ndarray:
    """Sparse Dirichlet draw (Gamma with low concentration) normalised to sum = 1."""
    # using Gamma is numerically stable for all concentration values
    draws = np.random.gamma(concentration, 1.0, size=n)
    # guard against the extremely unlikely all‑zero case
    if draws.sum() == 0:
        draws = np.ones(n, dtype=float)
    return draws / draws.sum()

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine combines several optimisation phases, adapts the problem
    size to the remaining wall‑clock time and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback to a previously discovered construction (if available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback sequence score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose a working length (larger n if we have plenty of time)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n = 4096
    else:
        n = 2048
    print(f"Working with n = {n}")

    # -----------------------------------------------------------------
    # 2. Assemble a handful of diverse initial points
    # -----------------------------------------------------------------
    init_pool = []

    # deterministic central‑binomial start
    init_pool.append(central_binomial_sequence(n))

    # a few exponential decays with random rates
    for _ in range(2):
        rate = np.random.uniform(0.001, 0.02)
        init_pool.append(exponential_sequence(n, rate))

    # a couple of sparse Dirichlet draws
    for _ in range(2):
        conc = np.random.uniform(0.2, 0.8)   # low concentration → sparsity
        init_pool.append(random_dirichlet_sequence(n, concentration=conc))

    # -----------------------------------------------------------------
    # 3. Optimisation loop – try each initialisation while time permits
    # -----------------------------------------------------------------
    for idx, a0 in enumerate(init_pool):
        if time.time() >= deadline - 200:
            print("Approaching deadline – stop launching new runs.")
            break

        print(f"\n=== Run {idx+1}/{len(init_pool)} (n={n}) ===")

        # ---- Phase A: L‑BFGS continuation with aggressive β‑schedule ----
        a_smooth, smooth_score = lbfgs_continuation(
            a0,
            beta_start=5.0,
            beta_end=800.0,
            deadline=deadline - 150.0,
            schedule_steps=12,
            maxiter_per_step=200,
        )
        cur_best_a = a_smooth
        cur_best_score = smooth_score
        print(f"After L‑BFGS best score = {cur_best_score:.6f}")

        # ---- Phase B: Adam refinement (if we still have time) ----
        if time.time() < deadline - 130:
            a_adam, adam_score = adam_continuation(
                cur_best_a,
                beta_start=800.0,
                beta_end=2000.0,
                schedule_steps=6,
                iter_per_beta=200,
                deadline=deadline - 120.0,
                lr=0.12,
            )
            if adam_score < cur_best_score:
                cur_best_a, cur_best_score = a_adam, adam_score
                print(f"Adam refinement improved score to {cur_best_score:.6f}")

        # ---- Phase C: Exact max‑objective polishing (projected sub‑gradient) ----
        if time.time() < deadline - 100:
            print("Starting sub‑gradient polishing …")
            refined_a, refined_score = subgrad_optimize(
                cur_best_a,
                deadline - 80.0,
                max_iter=800_000,
                step0=0.07,
            )
            if refined_score < cur_best_score:
                cur_best_a, cur_best_score = refined_a, refined_score
                print(f"Sub‑gradient polishing improves to {cur_best_score:.6f}")

        # ---- Phase D: Final simulated‑annealing hill‑climb ----
        if time.time() < deadline - 5:
            print("Final simulated‑annealing hill‑climb …")
            climbed_a, climbed_score = simulated_annealing(
                cur_best_a,
                deadline - 5.0,
                init_temp=0.02,
                min_temp=1e-5,
                step_scale_start=0.05,
            )
            if climbed_score < cur_best_score:
                cur_best_a, cur_best_score = climbed_a, climbed_score
                print(f"Hill‑climb yields new best {cur_best_score:.6f}")

        # ---- Update global best -------------------------------------------------
        if cur_best_score < best_val:
            best_val = cur_best_score
            best_seq = cur_best_a.tolist()
            print(f"*** Global best updated to {best_val:.6f} ***")

    # -----------------------------------------------------------------
    # 4. Final safety fallback (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        print("No candidate produced – falling back to uniform distribution.")
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
