"""
Propose a low‑score step‑function sequence.

The search uses four distinct phases and a more aggressive local search:
1. Analytic families (power‑law, β‑shape, exponential and sinusoidal)
   are sampled on a coarse grid; the best vector on the probability simplex
   is kept as the starting point.
2. A KL‑mirror (exponentiated‑gradient) optimisation on a smooth
   log‑sum‑exp surrogate of the max‑convolution, with an annealed temperature.
3. Projected sub‑gradient descent on the exact objective.
4. A long stochastic hill‑climb that updates the full autocorrelation
   incrementally (O(n) per move).  The walk mixes pure random transfers,
   heavy‑index‑biased transfers and occasional temperature‑based uphill
   moves, with an adaptive step‑size schedule and periodic full FFT
   recomputation to avoid drift.

All phases respect the 1000 s wall‑clock budget; the best sequence seen
throughout the run is finally returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector already normalized to sum = 1.
    This matches evaluate_sequence when the input already satisfies the
    sum‑normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for the initial point
# ----------------------------------------------------------------------


def power_law_start(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]⁻ᵅ (the family used previously)."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def beta_shape_start(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+½)^p·(n‑½‑i)^q  (Beta‑like shape)."""
    i = np.arange(n, dtype=float)
    w = np.power(i + 0.5, p) * np.power(np.maximum(n - 0.5 - i, 1e-12), q)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def exp_decay_start(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·|i-(n‑1)/2|)   (exponential centre‑heavy)."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    w = np.exp(-gamma * np.abs(i - centre))
    a = w / w.sum()
    return a


def sinusoid_start(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n)^power   (smooth hump)."""
    i = np.arange(n, dtype=float)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def analytic_start_grid(n: int, deadline: float):
    """Search a modest grid over the four families, keep the best."""
    best_a = None
    best_sc = float("inf")
    # families and parameter grids
    families = [
        (power_law_start, {"alpha": np.concatenate([np.linspace(-1.0, 0.0, 11),
                                                    np.linspace(0.0, 3.0, 31)])}),
        (beta_shape_start,
         {"p": np.linspace(-0.5, 2.0, 13), "q": np.linspace(-0.5, 2.0, 13)}),
        (exp_decay_start, {"gamma": np.linspace(0.0, 5.0, 21)}),
        (sinusoid_start, {"power": np.linspace(0.5, 4.0, 18)})
    ]

    for func, grid in families:
        # generate cartesian product of parameters (keep modest)
        keys = list(grid.keys())
        values = [grid[k] for k in keys]
        # naive nested loops – the total number stays below ~200
        for combo in np.ndindex(*[len(v) for v in values]):
            if time.time() >= deadline:
                return best_a, best_sc
            params = {k: values[idx][combo[idx]] for idx, k in enumerate(keys)}
            a = func(n, **params)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exponentiated‑gradient optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 120_000,
                        beta_start: float = 0.4,
                        beta_end: float = 250.0,
                        eta0: float = 0.7,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) optimisation of the log‑sum‑exp
    surrogate of the max‑convolution.  β is exponentially annealed;
    step‑size η follows η₀/√t.  The best exact score seen is returned.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # smooth max via log‑sum‑exp
        conv = convolution_fft(a, a)                      # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))               # stabilised
        w = expw / expw.sum()                             # softmax over shifts

        # gradient of the smoothed objective (cross‑correlation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        # KL‑mirror update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 180_000,
                     step0: float = 0.09) -> tuple[np.ndarray, float]:
    """Classic Euclidean projected sub‑gradient method on 2·n·max(a⊛a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑convolution (the set of indices achieving max)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 600 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Fast stochastic hill‑climb with temperature‑based uphill moves
# ----------------------------------------------------------------------


def hill_climb_incremental(start_a: np.ndarray,
                           deadline: float,
                           init_step: float = 0.15,
                           inc_factor: float = 1.06,
                           decay_factor: float = 0.96,
                           recompute_interval: int = 150_000,
                           heavy_frac: float = 0.025,
                           temperature_start: float = 5e-4) -> tuple[np.ndarray, float]:
    """
    Random walk that transfers a fraction of mass from a source index to a
    destination index, updating the autocorrelation incrementally.
    Moves are accepted if they improve the score or with a Metropolis
    probability based on a linearly cooling temperature.
    The step size is increased after a successful move and decayed otherwise.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)          # exact convolution
    cur_sc = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_sc = cur_sc

    step_scale = init_step
    steps = 0
    last_heavy_update = 0
    heavy_cnt = max(1, int(heavy_frac * n))

    # Temperature schedule – linear cooling to zero
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # update heavy‑index list every few thousand steps
        if steps - last_heavy_update > 5000:
            heavy_idx = np.argsort(-a)[:heavy_cnt]
            last_heavy_update = steps

        # Choose source index (biased towards heavy entries)
        if np.random.rand() < 0.66 and heavy_cnt > 0:
            src = np.random.choice(heavy_idx)
        else:
            src = np.random.randint(n)

        if a[src] <= 0.0:
            continue

        # Choose a different destination uniformly
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        delta = step_scale * a[src]          # always ≤ a[src]
        if delta <= 0.0:
            continue

        # --- Incremental convolution update (copy‑and‑modify) ---
        conv_cand = conv.copy()
        # linear part: –2·δ·shift_src + 2·δ·shift_dst
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic correction
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_sc = 2.0 * n * float(cand_max)

        # Temperature for this iteration
        elapsed = time.time() - start_time
        temperature = temperature_start * max(0.0, 1.0 - elapsed / total_time)

        accept = False
        if cand_sc < cur_sc:
            accept = True
        elif temperature > 0.0:
            # Metropolis acceptance
            prob = np.exp(-(cand_sc - cur_sc) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            # accept move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a.copy()
            # enlarge step size modestly
            step_scale = min(step_scale * inc_factor, 0.5)
        else:
            # reject – shrink step size
            step_scale *= decay_factor

        steps += 1

        # Periodic full recomputation to avoid drift
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_sc = 2.0 * n * float(conv.max())

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows four stages and respects the 1000 s wall‑clock limit.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a length based on the available budget
    # -------------------------------------------------
    remaining_total = deadline - start_time
    if remaining_total > 900:
        n = 16384
    elif remaining_total > 700:
        n = 8192
    elif remaining_total > 400:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic multi‑family start (grid search)
    # -------------------------------------------------
    a_start, sc_start = analytic_start_grid(n, deadline - 30.0)
    if a_start is not None and sc_start < best_score:
        best_score = sc_start
        best_seq = a_start.tolist()
        print(f"[{time.time():.1f}] best after analytic grid = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Smooth exponentiated‑gradient optimisation (single run)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] starting exp‑gradient optimisation")
        a_smooth, sc_smooth = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 150.0,
            max_iters=120_000,
            beta_start=0.4,
            beta_end=250.0,
            eta0=0.7,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Projected sub‑gradient refinement
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 80.0,
            max_iters=180_000,
            step0=0.09,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Fast hill‑climb polishing (majority of remaining budget)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] starting stochastic hill‑climb")
        a_final, sc_final = hill_climb_incremental(
            np.array(best_seq, dtype=float),
            deadline,
            init_step=0.15,
            inc_factor=1.06,
            decay_factor=0.96,
            recompute_interval=120_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats, as required by the harness.
    return best_seq
