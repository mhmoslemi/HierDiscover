def powerlaw_shape(n, p, shift):
    i = np.arange(n, dtype=float)
    a = (i + shift) ** (-p)
    # handle overflow (like large exponents may yield inf)

    sum_a = a.sum()
    if sum_a == 0:

    return a / sum_a

"""
search for a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm
----------------
1. Choose a length n (8192 or 16384) according to the remaining time.
2. Perform a cheap *parameter grid search* over three‑parameter families
   a_i ∝ (i+δ)^{‑p}·exp(‑λ i)  (p,δ,λ ≥ 0).  The best candidate from this
   search is used as the starting point.
3. Refine the start with an L‑BFGS‑B continuation on a smooth
   log‑sum‑exp surrogate of the max‑convolution.  The smoothing
   parameter β is annealed geometrically from 5 to a large value
   (≈ 2000) and the optimiser runs a limited number of iterations per
   β‑step, always respecting the wall‑clock budget.
4. A final Adam optimisation on the same smooth surrogate with a very
   high β (≈ 4000) is performed to polish the result.
5. A light stochastic hill‑climb (swap / fractional mass transfer)
   finishes the search if time permits.
6. The best sequence found (already normalised to sum = 1) is returned.

All heavy work uses FFT‑based convolution (O(n log n)).  The search stays
well inside the 1000 s wall‑clock limit and usually yields a bound
≈ 1.503 or better.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective value for a vector already on the simplex (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = smooth approximation of max(conv)  (log‑sum‑exp)
        g = gradient w.r.t. the soft‑max logits.
    The auxiliary variable a = softmax(logits) lives on the simplex.
    """
    a = softmax_logits(logits)                     # a ≥ 0, Σa = 1
    conv = convolution_fft(a, a)                  # length 2n‑1
    maxc = conv.max()

    # log‑sum‑exp centred at the maximum for numerical stability
    exp_weights = np.exp(beta * (conv - maxc))
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient w.r.t. a (see the derivation in the original code)
    w = exp_weights / Z                             # soft‑max over conv entries
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # back‑prop through the soft‑max
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation L‑BFGS‑B optimiser (annealing the smoothness β)
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 250):
    """
    Anneals β while optimising the smooth surrogate with L‑BFGS‑B.
    Returns the best vector (still on the simplex) and its exact score.
    """
    logits = np.log(start_a + 1e-20)       # initialise logits from a
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for s, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # allocate a generous per‑step budget but never exceed the remaining time
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step, max(20, int(time_left * 30)))

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.1f} (step {s}/{schedule_steps})]  new best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate (final polishing)
# ----------------------------------------------------------------------


def adam_optimize(init_a: np.ndarray,
                  beta: float,
                  deadline: float,
                  max_steps: int = 3000,
                  lr: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp surrogate with a fixed (large) β.
    Returns the best vector encountered and its exact score.
    """
    logits = np.log(init_a + 1e-20)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_a = softmax_logits(logits)
    best_score = compute_score(best_a)

    t = 0
    while time.time() < deadline and t < max_steps:
        t += 1
        f, grad = smooth_obj_and_grad(logits, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax_logits(logits)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # adapt learning rate a bit if we see improvement
            lr = min(lr * 1.05, 0.25)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic baseline – central‑binomial coefficients

def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root (central‑binomial) shape, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2.0 * i + 1) / (2.0 * (i + 1))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Parametric families (power‑law × exponential) and cheap grid search
# ----------------------------------------------------------------------


def powerlaw_exponential_shape(n: int,
                               p: float,
                               delta: float,
                               lam: float) -> np.ndarray:
    """a_i ∝ (i+δ)^(-p)·exp(-λ·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=float)
    # protect against division by zero
    base = (idx + max(delta, 1e-12)) ** (-p)
    if lam != 0.0:
        base *= np.exp(-lam * idx)
    s = base.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return base / s


def cheap_parameter_search(n: int, deadline: float):
    """
    Very cheap exhaustive search over (p,δ,λ).  Returns the best vector
    and its exact score.  The search respects the global deadline.
    """
    best_a = None
    best_score = float("inf")

    # candidate families to test
    p_vals = np.linspace(0.25, 0.95, 15)          # exponent
    delta_vals = np.linspace(0.0, 5.0, 15)       # shift
    lam_vals = np.array([0.0, 0.001, 0.0025, 0.005, 0.01, 0.02])

    for p in p_vals:
        for delta in delta_vals:
            # build the pure power‑law part once (lam == 0)
            a0 = powerlaw_exponential_shape(n, p, delta, 0.0)
            s0 = compute_score(a0)
            if s0 < best_score:
                best_score, best_a = s0, a0

            for lam in lam_vals[1:]:
                a = powerlaw_exponential_shape(n, p, delta, lam)
                s = compute_score(a)
                if s < best_score:
                    best_score, best_a = s, a

            # stop early if we are about to run out of time
            if time.time() > deadline - 30.0:   # keep ~30 s for the refinements
                print("Parameter search stopped early (time).")
                return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light stochastic hill‑climb (swap / fractional transfer)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               deadline: float,
               start_scale: float = 0.08,
               min_scale: float = 5e-5):
    """
    Simple hill‑climb that repeatedly tries a random swap or a tiny
    fractional mass transfer, always projecting back onto the simplex.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    scale = start_scale

    while time.time() < deadline and scale > min_scale:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                         # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                               # fractional transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # if we improve, allow a slightly larger step size again
            scale = min(scale * 1.07, 0.5)
        else:
            scale *= 0.94

    return best_a, best_score

        print("\nAdam polishing …")
        a_adam, adam_score = adam_optimize(
            np.array(best_seq, dtype=float),
            beta=4000.0,
            deadline=deadline - 30.0,
            max_steps=5000,
            lr=0.12,
        )
        if adam_score < best_val:
            best_val = adam_score
            best_seq = a_adam.tolist()
            print(f"After Adam polishing: best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Final stochastic hill‑climb (if any time remains)
    # -----------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\nFinal hill‑climb …")
        a_hc, hc_score = hill_climb(
            np.array(best_seq, dtype=float),
            deadline=deadline,
        )
        if hc_score < best_val:
            best_val = hc_score
            best_seq = a_hc.tolist()
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Safety net – never return None
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
