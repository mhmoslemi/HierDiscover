"""
Propose a step‑function (probability vector) that minimises the evaluation score.

Algorithm (≈ 1000 s total)
--------------------------------
1. Choose a length n based on the time budget (larger n → more degrees of freedom).
2. Initialise many candidates:

     exponential decay, quadratic, sine‑power).  A fine grid of the
     shape parameters is explored.
   * Random monotone draws.
   The best candidate (lowest exact score) is kept.
3. Smoothed optimisation:
   minimise the log‑sum‑exp surrogate of the max‑convolution using Adam.
   The smoothing parameter β is increased exponentially from 5 up to 5000.
   Several restarts are performed while time permits.
4. Mirror (entropy‑mirror) sub‑gradient descent on the true objective.
   The sub‑gradient is the sum of the gradients for *all* active
   (≈‑max) convolution positions, not only a single arg‑max.
   Step size ηₜ = η₀/√t.
5. Symmetrisation – replace a by (a + a[::-1])/2 and project back to
   the simplex; this frequently lowers the peak of the auto‑convolution.
6. Final stochastic hill‑climb with a tiny simulated‑annealing schedule.
   The moves are either a swap of two entries or a mass transfer; the
   temperature starts at 10⁻⁴·best_score and cools linearly.
7. Return the best vector (plain Python list of floats).  All intermediate
   steps respect the global deadline; the routine prints progress markers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (aperiodic) convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) = 1):
        score = 2·n·max(conv)
    where conv = a * a (linear convolution).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^(‑α), symmetric about the centre.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against overflow for alpha=0 (w = 1)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """
    a_i ∝ (i+½)^(‑β), monotone decreasing.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp(-γ·i), monotone decreasing.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def quadratic_family(n: int, power: float) -> np.ndarray:
    """
    a_i ∝ [(i+1)*(n-i)]^power, a symmetric concave parabola.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 1) * (n - i)) ** power
    a = w / w.sum()
    return a


def sine_power_family(n: int, power: float) -> np.ndarray:
    """
    a_i ∝ sin(π·(i+½)/n)^power, symmetric and zero at the ends.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    a = w / w.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]               # decreasing
    a = a / a.sum()
    return a

# Smooth optimisation (log‑sum‑exp) via Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 5_000.0,
    lr: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·conv_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially.
    """
    eps = 1e-12
    z = np.log(start_a + eps)              # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # current a
        a = softmax(z)

        # smoothed weights w ∝ exp(β·conv)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of f_β w.r.t. a:
        # Note: conv_rev = a_rev * w (linear convolution)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length = n + (2n-1) -1 = 3n-2
        # extract the middle n entries (corresponding to derivative w.r.t a)
        start = a.shape[0] - 1
        grad_a = 2.0 * conv_rev[start : start + a.shape[0]]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy‑mirror) sub‑gradient descent on the true max objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.9,
    check_interval: int = 1500,
    rel_tol: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for the objective
        f(a) = 2·n·max(conv)
    The sub‑gradient is the sum of gradients associated with *all*
    nearly‑maximal convolution entries (within rel_tol of the maximum).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # indices whose value is within rel_tol of the maximum
        active = np.where(conv >= maxc - rel_tol * maxc)[0]

        # accumulate sub‑gradient over active indices
        g = np.zeros_like(a)
        for k in active:
            # contribution of index k: for all i where 0≤i<n and 0≤j=n where i+j=k
            # i ranges from max(0, k-(n-1)) to min(n-1, k)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k - idx_i
            # gradient w.r.t. a_i is 2 * a_j
            g[idx_i] += 2.0 * a[idx_j]

        eta = step0 / np.sqrt(t)
        # exponentiated update (keeps non‑negative)
        a = a * np.exp(-eta * g)

        # re‑normalise to the simplex (entropy projection)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            # numerical disaster – fall back to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final evaluation
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated annealing
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_300_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a tiny simulated‑annealing schedule.
    Moves:
      – swap two entries,
      – transfer a fraction of mass from i to j.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature proportional to the current best score
    T0 = best_score * init_temp_factor
    T = T0

    # step scale controls the amount of transferred mass
    step_scale = 0.08

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise (exact simplex projection is cheap here)
        total = cand.sum()
        if total <= 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / total

        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase move size when we are improving
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # shrink move size after rejection
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families + cheap random draws
    # ------------------------------------------------------------------
    # Helper to evaluate and possibly replace the current best
    def try_candidate(vec: np.ndarray, label: str):
        nonlocal best_seq, best_score
        if time.time() >= deadline - 30.0:
            return
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best ({label}) = {best_score:.6f}")

    # symmetric product α ∈ [0,5]
    alphas = np.linspace(0.0, 5.0, 81)
    for a in alphas:
        try_candidate(symmetric_product_family(n, a), f"sym‑α={a:.2f}")

    # left‑heavy power‑law β ∈ [0,5]
    betas = np.linspace(0.0, 5.0, 81)
    for b in betas:
        try_candidate(left_heavy_power_family(n, b), f"left‑β={b:.2f}")

    # exponential decay γ ∈ [0,6]
    gammas = np.linspace(0.0, 6.0, 71)
    for g in gammas:
        try_candidate(exponential_decay_family(n, g), f"exp‑γ={g:.2f}")

    # quadratic power p ∈ [0.1,5]
    powers = np.linspace(0.1, 5.0, 51)
    for p in powers:
        try_candidate(quadratic_family(n, p), f"quad‑p={p:.2f}")

    # sine‑power family
    sine_powers = np.linspace(0.1, 6.0, 61)
    for p in sine_powers:
        try_candidate(sine_power_family(n, p), f"sine‑p={p:.2f}")

    # random monotone draws
    for it in range(120):
        try_candidate(random_monotone_start(n), "rand‑mono")

    # ------------------------------------------------------------------
    # 2. Smoothed optimisation (Adam) – multiple restarts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        # initialise with a sparse random start to explore diverse regions
        start_vec = random_sparse_start(n)
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 200.0,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=5_000.0,
            lr=0.02,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam‑smooth = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Mirror (entropy) sub‑gradient descent on the exact max objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            step0=0.9,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Symmetrisation (many nice solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb with simulated annealing (tiny temperature)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_300_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: if nothing succeeded, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats.
    return best_seq
