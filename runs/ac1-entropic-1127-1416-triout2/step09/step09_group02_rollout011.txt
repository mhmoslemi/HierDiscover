def smooth_max_optimize(a, max_iters, deadline):
    tau0 = initial tau maybe 0.1
    decay_tau = appropriate schedule: tau = tau0 * (0.5 ** (iter / (max_iters/2))) but not too small.

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8

    best_seq = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute convolution
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        # compute p via softmax with temperature tau
        tau = max(tau0 * (0.5 ** (t / (max_iters/4))), 1e-6)  # decays to very low
        # subtract max for stability
        exp_vals = np.exp((conv - max_conv) / tau)
        p = exp_vals / exp_vals.sum()
        # gradient via convolution p * a
        grad = 2.0 * convolution_fft(p, a)[len(a)-1 : len(a)-1 + len(a)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.05 / (np.sqrt(v_hat) + eps)  # lr scaled by 0.05
        a_new = a - step * m_hat

        # projection to simplex
        a_new = project_to_simplex(a_new)
        a = a_new

        # evaluate actual objective every 10 iterations
        if t % 10 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                # log
                print(f"[smooth-Adam] iter {t}, score {best_score:.6f}, tau {tau:.6e}")
    return best_seq, best_score

"""
Improved search for a step‑function that minimises ``evaluate_sequence``.

The algorithm consists of three phases:

1. **Entropy‑smoothed Adam** – minimise a smooth approximation of the
   infinity‑norm  ``max(conv(a,a))``.  The temperature (soft‑max parameter)
   starts relatively large and is annealed to a tiny value, while Adam
   (first‑order moments) adapts the step size.

2. **Mirror‑descent on the true max** – after the soft objective has been
   flattened we switch to exponentiated‑gradient (a.k.a. mirror descent on
   the simplex) that directly reduces the dominant convolution entry.
   The step size decays like 1/√t.

3. **Stochastic hill‑climbing** – a final polishing phase that performs
   greedy swaps / local mass transfers, accepting them only when the
   exact score improves.

All phases respect the wall‑clock budget (default 1000 s) and keep the
best vector seen so far.  The result is returned as a plain list of
floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear convolution (real FFT, zero‑padding)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (exact objective for a vector whose sum = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – Adam on a soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def adam_softmax_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of τ·log Σ exp(conv/τ).  Returns best vector."""
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    tau0 = 0.05          # start “warm”
    tau_min = 1e-6       # end “cold”

    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        max_conv = conv.max()

        # anneal the temperature slowly
        tau = max(tau_min, tau0 * (0.5 ** (t / 2000.0)))

        # stable soft‑max (shift by the max)
        exp_vals = np.exp((conv - max_conv) / tau)
        p = exp_vals / exp_vals.sum()               # p is a probability vector

        # gradient of the smooth objective: g = 2·(p * a) (valid part)
        grad_full = convolution_fft(p, a)           # length = 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate that slowly decays
        lr = 0.12 / (np.sqrt(t) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # keep a on the simplex
        a = project_to_simplex(a)

        # occasional exact check
        if t % 100 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Adam] iter {t:5d}, best {best_score:.6f}, τ {tau:.2e}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Mirror descent (exponentiated gradient) on the true max
# ----------------------------------------------------------------------
def mirror_descent_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative‑weight updates that directly lower the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]

    eta0 = 5.0                     # initial step size
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active max entry
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)          # exponentiated‑gradient step
        a = project_to_simplex(a)

        if t % 50 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Mirror] iter {t:5d}, best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Simple greedy hill‑climb (improvements only)
# ----------------------------------------------------------------------
def hill_climb(a0: np.ndarray, deadline: float, max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """Stochastic local search that accepts only improving moves."""
    n = a0.shape[0]
    best_a = a0.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]               # swap
        else:
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand /= cand.sum()                                    # re‑normalise
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)           # become bolder
        else:
            step_scale *= 0.96                                 # become more cautious

        steps += 1
    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # small safety margin
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # Choose a manageable length – 2048 works well; use 4096 if time permits
    # --------------------------------------------------------------
    n = 2048
    if total_time > 800.0:
        n = 4096
    print(f"[Info] Using sequence length n = {n}")

    # --------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered sequence
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback = fallback[:n]                            # truncate / pad if needed
        if len(fallback) < n:
            fallback = np.pad(fallback, (0, n - len(fallback)), "constant").tolist()
        fallback_val = evaluate_sequence(fallback)
        print(f"[Info] Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Initial seed (random Dirichlet if no good fallback)
    # --------------------------------------------------------------
    if best_seq is None:
        a0 = random_dirichlet(n, conc=0.6)
    else:
        a0 = np.array(best_seq, dtype=float)

    a0 = project_to_simplex(a0)          # guard against tiny drift
    cur_score = compute_score(a0)
    if cur_score < best_val:
        best_val = cur_score
        best_seq = a0.tolist()
        print(f"[Info] Initial random seed score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2.  Phase 1 – Adam on a smooth max‑approximation
    # --------------------------------------------------------------
    phase1_deadline = start_time + total_time * 0.55
    print("\n[Phase 1] Adam on soft‑max objective")
    a_opt, score_opt = adam_softmax_optimize(np.array(best_seq, dtype=float), phase1_deadline)
    if score_opt < best_val:
        best_val = score_opt
        best_seq = a_opt.tolist()
        print(f"[Phase 1] best after Adam = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3.  Phase 2 – Mirror descent on the true max
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        phase2_deadline = deadline - 30.0
        print("\n[Phase 2] Mirror‑descent on max‑convolution")
        a_opt2, score_opt2 = mirror_descent_optimize(np.array(best_seq, dtype=float), phase2_deadline)
        if score_opt2 < best_val:
            best_val = score_opt2
            best_seq = a_opt2.tolist()
            print(f"[Phase 2] best after Mirror = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4.  Phase 3 – Greedy hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 3] Hill‑climb polishing")
        a_opt3, score_opt3 = hill_climb(np.array(best_seq, dtype=float), deadline, max_steps=1_500_000)
        if score_opt3 < best_val:
            best_val = score_opt3
            best_seq = a_opt3.tolist()
            print(f"[Phase 3] best after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5.  Final official evaluation & return
    # --------------------------------------------------------------
    final_seq = best_seq
    final_val = evaluate_sequence(final_seq)
    print(f"\n[Result] Final reported score = {final_val:.6f}")
    return final_seq
