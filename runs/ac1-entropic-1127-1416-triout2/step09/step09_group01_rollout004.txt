    # Starting point in log domain:
    z0 = np.log(start_a + eps)

"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

Algorithm
---------
1.  Choose a length `n` (≈16 k for a good trade‑off between quality and speed).
2.  Scan a few analytic families (U‑shaped, power‑law, exponential, β‑distribution,
    plus many random starts) – keep the best (exact objective computed with FFT).
3.  Refine the current best with a *smooth surrogate* of the max‑convolution.
    A log‑sum‑exp surrogate `f_β(a)= (1/β)·log∑exp(β·(a*a)_k)` is minimised by
    L‑BFGS‑B on the soft‑max parametrisation `a = softmax(z)`.  The schedule
    β={200, 800, 2000} drives the surrogate to the true max.
4.  Perform a few hundred thousand steps of *mirror (exponentiated) sub‑gradient*
    descent on the true objective `2·n·max(a*a)` (the constraint `a≥0`, `∑a=1`
    is enforced by projection onto the simplex).  The step size is
    η_t = 0.8 / √t.
5.  Symmetrise the result (`a ← (a + rev(a))/2`) – the optimum is often symmetric.
6.  A final cheap stochastic hill‑climb with a tiny simulated‑annealing temperature
    removes any remaining shallow local minima.
7.  Return the best sequence found as a plain Python list of floats.

All stages respect the global wall‑clock deadline (default 1000 s).  The code
is deliberately written without closures so it can be imported safely.
"""

import time
import numpy as np
import math

# optional: use scipy when available (for L‑BFGS).  If not present we fall back to Adam.
try:
    import scipy.optimize as _opt
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Aperiodic (linear) convolution via real FFT.
    Returns the full convolution of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        score = 2·n·max(a * a)
    This matches `evaluate_sequence` up to the normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against infinities at the ends for very large α
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑shape: a_i ∝ (i/(n‑1))^α·(1‑i/(n‑1))^α."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    # add a tiny epsilon to avoid exact zero at the boundaries
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (≈5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]  # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate optimisation (L‑BFGS‑B)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where
        f   = (1/β)·log∑_k exp[β·(a*a)_k]
        grad = ∂f/∂z   (z are the log‑soft‑max parameters)
    """
    a = softmax(z)                     # a ∈ simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    # log‑sum‑exp in a numerically stable way
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient w.r.t. a
    w = exp_terms / sum_exp            # same shape as conv
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1 : 2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0),
                  max_iter_per_beta=600) -> tuple[np.ndarray, float]:
    """
    Refine `start_a` using L‑BFGS‑B on the smoothed log‑sum‑exp surrogate.
    The optimisation runs for each β in `beta_list` sequentially (warm‑started).
    Returns the best vector and its exact score.
    """
    if not _HAS_SCIPY:          # pragma: no cover
        # If SciPy is missing we simply return the input unchanged.
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    # initialise log‑soft‑max parameters
    z = np.log(best_a + eps)

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break

        # adapt max_iter to the remaining budget (very coarse)
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 10))  # ~10 iters per sec

        def func(z_vec):
            f, g = _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])
            return f, g

        # L‑BFGS‑B optimisation
        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]                     # updated parameters
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on
        F(a) = 2·n·max(a*a)
    The iterate stays in the simplex thanks to a renormalisation step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry of the convolution
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)            # multiplicative update

        # re‑project onto the simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb (tiny simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 1_200_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
        - with prob 0.5 swap two entries,
        - otherwise move a fraction of mass from one index to another.
    Very low temperature permits occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
