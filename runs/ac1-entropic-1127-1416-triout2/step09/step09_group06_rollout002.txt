"""
A refined optimiser for the max‑convolution bound.

The algorithm proceeds in three stages:

1.  Initialise with the classic inverse‑square‑root (central‑binomial)
    shape, normalised to the probability simplex.
2.  Smooth optimisation of a log‑sum‑exp surrogate of the max
    using L‑BFGS‑B with an aggressive β‑annealing schedule (β up to
    several thousand) and many LBFGS iterations.  The variables are
    parametrised by soft‑max logits, which automatically enforce the
    non‑negativity and unit‑sum constraints.
3.  A mirror‑descent (exponentiated‑gradient) phase that directly
    reduces the exact max‑convolution, followed by a cheap stochastic
    hill‑climb.  The best sequence found is returned as a plain list of
    floats.

The code respects a 1000 s wall‑clock budget, prints intermediate
scores and always returns a valid sequence (the objective is scale‑free,
so the simplex normalisation is sufficient).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (efficient for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a probability vector
    (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient (combined)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))
    and g is the gradient of f w.r.t. the soft‑max logits.
    The parametrisation a = softmax(logits) guarantees a ≥ 0 and ∑a = 1.
    """
    a = softmax_logits(logits)

    # 1) self‑convolution
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # 2) smooth max (log‑sum‑exp)
    exp_weights = np.exp(beta * (conv - maxc))  # centred for stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # 3) gradient w.r.t. a
    w = exp_weights / Z  # (2n‑1,) – normalized weights
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)          # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]      # shape (n,)

    # 4) chain rule back to logits (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Deterministic start – truncated central‑binomial coefficients
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """
    Inverse‑square‑root shape:
        a_i = C(2i,i) / 4^i,
    normalised to the probability simplex.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        # recurrence: a_{i+1} = a_i * (2i+1) / (2(i+1))
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) phase
# ----------------------------------------------------------------------


def mirror_descent_max(a0: np.ndarray, deadline: float, step0: float = 0.15):
    """
    Projected exponentiated‑gradient descent directly on the exact
    max‑convolution objective.  Returns the best sequence seen.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (see analysis)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)

        # exponentiated update (mirror descent on the simplex)
        a = a * np.exp(-eta * g)

        # renormalise to the simplex – this is the KL‑projection
        s = a.sum()
        if s == 0.0:                         # numerical safeguard
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if t % 5000 == 0:
                print(f"[Mirror] iter {t:,} – new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(a0: np.ndarray, deadline: float, init_scale: float = 0.08):
    """
    Simple random perturbations (swap or fractional transfer) with
    acceptance only if the exact score improves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = init_scale

    while time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:               # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # fractional transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        # re‑normalise to the simplex (preserves scale‑invariance)
        cand = cand / cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s wall‑clock
    budget and returns the best sequence found as a plain list of
    non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback to a previously supplied construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose a working length (larger ⇒ finer optimisation)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n = 16384
    elif remaining > 400:
        n = 8192
    else:
        n = 4096

    print(f"Using length n = {n}")

    # -----------------------------------------------------------------
    # 2. Initialise with the central‑binomial shape (small random jitter)
    # -----------------------------------------------------------------
    a0 = central_binomial_sequence(n)
    a0 += 1e-8 * np.random.rand(n)  # break perfect symmetry
    a0 /= a0.sum()

    best_a = a0.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Smooth optimisation via L‑BFGS‑B with aggressive β‑annealing
    # -----------------------------------------------------------------
    beta_start = 2.0
    beta_end = 5000.0
    schedule_steps = 20
    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    logits = np.log(best_a + 1e-20)  # initialise logits safely

    for beta in betas:
        if time.time() >= deadline - 250:   # reserve time for later phases
            break
        time_left = deadline - time.time()
        max_iter = max(30, int(time_left * 35))   # ~35 LBFGS iters per sec
        res = minimize(
            smooth_obj_and_grad,
            logits,
            args=(beta,),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:6.1f}] new best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Exact max‑objective polishing via mirror descent
    # -----------------------------------------------------------------
    if time.time() < deadline - 100:
        print("\nMirror‑descent polishing …")
        polished_a, polished_score = mirror_descent_max(
            best_a, deadline - 90, step0=0.15
        )
        if polished_score < best_score:
            best_score = polished_score
            best_a = polished_a
            print(f"Mirror‑descent improved best to {best_score:.6f}")

    # -----------------------------------------------------------------
    # 5. Very cheap stochastic hill‑climb (final refinement)
    # -----------------------------------------------------------------
    if time.time() < deadline - 5:
        print("\nFinal hill‑climb …")
        final_a, final_score = hill_climb(best_a, deadline - 2, init_scale=0.08)
        if final_score < best_score:
            best_score = final_score
            best_a = final_a
            print(f"Hill‑climb final score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 6. Fallback if nothing was produced (extremely unlikely)
    # -----------------------------------------------------------------
    if best_a is None:
        n_fallback = 256
        best_a = np.full(n_fallback, 1.0 / n_fallback)

    # -----------------------------------------------------------------
    # 7. Report final score using the official evaluator
    # -----------------------------------------------------------------
    final_val = evaluate_sequence(best_a.tolist())
    print(f"\nSearch finished. Final reported score = {final_val:.6f}")

    return best_a.tolist()
