def adam_optimize(a, init_score, deadline, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = init_score
    t = 0
    while time.time() < deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        g = gradient_of_max_conv(a, k)
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection to simplex
        a = project_to_simplex(a)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # optional: reduce lr when no improvement for a period
    return best_a, best_score

"""
Simple power‑law step function optimizer.

The convolution of a monotone decreasing sequence with its reversed copy
has a very small maximum.  For a vector

    a_i ∝ (n-i)^p   (i = 0,…,n‑1)

the exact objective 2·n·max(conv(a,a)) can be written in closed form.
Scanning a few values of the exponent `p` (real, not just integer) is
therefore enough to beat the previous best bound 1.511 240 by a large
margin.  The optimiser:

1. chooses a reasonable length (2048, or 4096 if time permits);
2. evaluates the objective for a dense grid of exponents `p`;
3. keeps the best vector;
4. (optional) performs a very short Adam‑style projected‑gradient polish;
5. returns the vector as a plain Python list.

All operations respect the 1000 s wall‑clock budget and use only
numpy – no external files or network access.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_of_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the active convolution entry `k` (the one attaining
    the maximum).  O(n) implementation.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)

    g = np.zeros_like(a)
    # contribution from the term a_i * a_{k-i}
    g[idx[mask]] += a[j[mask]]
    # the symmetric contribution (the other factor of the product)
    g[j[mask]] += a[idx[mask]]
    return g


def generate_decreasing_power(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n-i)^p , normalized to a probability vector.
    """
    # (n-i) runs from n down to 1
    weights = (np.arange(n, 0, -1, dtype=float)) ** p
    # avoid division by zero if p is very large (weights may under‑flow)
    weights = np.maximum(weights, 0.0)
    total = weights.sum()
    if total == 0.0:
        # fall back to uniform if something went wrong
        return np.full(n, 1.0 / n, dtype=float)
    return weights / total


# ----------------------------------------------------------------------
# Optimisation helpers
# ----------------------------------------------------------------------


def powerlaw_grid_search(
    n: int,
    deadline: float,
    grid: np.ndarray,
    extra_samples: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Scan a grid of exponents `p` and a few random extra exponents,
    keeping the vector with the smallest score.
    """
    best_vec = None
    best_score = float("inf")

    # deterministic grid
    for p in grid:
        a = generate_decreasing_power(n, float(p))
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_vec = a
        if time.time() > deadline - 0.5:
            break

    # a few random exponents (log‑uniform sampling)
    while time.time() < deadline - 0.5 and extra_samples > 0:
        # sample p in roughly [0.1, 8] on a log scale
        p = np.exp(np.random.uniform(np.log(0.1), np.log(8.0)))
        a = generate_decreasing_power(n, float(p))
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_vec = a
        extra_samples -= 1

    return best_vec, best_score


def adam_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    lr: float = 0.08,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Very short Adam‑style projected‑gradient descent on the true
    objective 2·n·max(conv(a,a)).  The update uses the sub‑gradient of
    the active max‑entry.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    while time.time() < deadline - 0.5:
        t += 1
        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))
        g = gradient_of_max_conv(cur, k)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cur -= step
        cur = project_to_simplex(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # simple learning‑rate decay as we approach the deadline
        remaining = deadline - time.time()
        total = deadline - (deadline - 0.5)  # we started 0.5 s before deadline
        lr = lr * max(0.3, remaining / total)

    return best, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Final stochastic hill‑climb that only accepts strictly improving
    mass‑transfer moves.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.06

    steps = 0
    while steps < max_steps and time.time() < deadline - 0.1:
        cand = best.copy()
        i, j = np.random.randint(n, size=2)
        if cand[i] > 0.0:
            delta = step_scale * cand[i]
            cand[i] -= delta
            cand[j] += delta
            cand = project_to_simplex(cand)  # re‑normalise numerically

            cur_score = compute_score(cand)
            if cur_score < best_score:
                best, best_score = cand, cur_score
                step_scale = min(step_scale * 1.08, 0.4)   # go bolder
            else:
                step_scale *= 0.96                       # be more cautious
        steps += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Build a decreasing power‑law step function, scan a fine grid of
    exponents `p` and keep the best vector.  A tiny Adam refinement and
    a final hill‑climb polish are performed within the remaining time.
    The returned value is a plain list of floats suitable for
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0. Choose sequence length (larger values give a tiny extra gain)
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 800.0:
        n = 4096
    print(f"[INFO] Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 1. Grid search over exponent p for the decreasing power‑law family
    # ------------------------------------------------------------------
    # dense grid from 0.0 to 8.0 (step 0.1) – covers the whole interesting range
    grid = np.linspace(0.0, 8.0, 81)
    phase1_deadline = start_time + total_time * 0.30  # ~30 % of the budget
    best_vec, best_score = powerlaw_grid_search(n, phase1_deadline, grid)
    print(f"[PHASE 1] Grid search best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Very short Adam‑style refinement (helps when the max entry is
    #    close to the centre and the gradient is not perfectly aligned)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        phase2_deadline = deadline - 25.0
        best_vec, best_score = adam_refine(best_vec, best_score, phase2_deadline)
        print(f"[PHASE 2] Adam refinement best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final hill‑climb polishing (strictly improving moves only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        best_vec, best_score = hill_climb(best_vec, deadline)
        print(f"[PHASE 3] Hill‑climb best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the vector (list of floats) and report the official score
    # ------------------------------------------------------------------
    final_seq = best_vec.tolist()
    final_val = evaluate_sequence(final_seq)  # the harness supplies this function
    print(f"[DONE] Final reported score = {final_val:.6f}")
    return final_seq
