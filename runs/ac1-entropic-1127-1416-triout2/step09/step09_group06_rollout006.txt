def convolution_fft(a, b):
    """Full linear convolution using real FFT (fast for large arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv
```

```
def project_to_simplex(v):
    """Euclidean projection of v onto probability simplex."""
    n = v.shape[0]
    if np.sum(v) == 0:
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
```

Works.

def smooth_obj_and_grad(logits, beta):
    a = softmax_logits(logits)
    conv = convolution_fft(a, a)
    maxc = conv.max()
    exp_weights = np.exp(beta * (conv - maxc))
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta
    w = exp_weights / Z
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)  # length 3n-2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return f, grad_logits
```

def lbfgs_continuation(start_a, beta_start, beta_end, deadline, schedule_steps=9, maxiter_per_step=150):
    logits = np.log(start_a + 1e-20)
    best_a = start_a.copy()
    best_score = compute_score(best_a)
    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)
    for idx, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step, max(20, int(time_left * 30)))
        res = minimize(smooth_obj_and_grad, logits, args=(beta,), method='L-BFGS-B',
                       jac=True, options={'maxiter': max_iter, 'disp': False})
        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f} step {idx}] new best score = {best_score:.6f}")
    return best_a, best_score

def subgrad_optimize(init_a, deadline, max_iter=400000, step0=0.07):
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)
    final_score = compute_score(best_a)
    return best_a, final_score

Algorithm outline
-----------------
1.  **Parametric warm‑start** – Scan simple power‑law and exponential families
    for a few lengths (≈ 8k–16k) and keep the best‑scoring shape.

4.  **Random restarts** – While time permits, draw random Dirichlet vectors,

5.  **Final hill‑climb** – A cheap stochastic mass‑transfer / swap hill‑climb

sequence is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a vector on the simplex
    (∑a = 1). Returns 2·n·max(conv(a,a)).
    """
    conv = convolution_fft(a, a)
    return 2.0 * a.shape[0] * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.sum(v) == 0:
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically‑stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv−max(conv)))
    and g is the gradient of f with respect to the soft‑max logits.
    """
    a = softmax_logits(logits)

    conv = convolution_fft(a, a)
    maxc = conv.max()

    exp_weights = np.exp(beta * (conv - maxc))        # centred for stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    w = exp_weights / Z                               # soft‑max of conv entries
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)              # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # chain rule back to logits (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 10,
                       maxiter_per_step: int = 200):
    """
    Anneal the smoothing parameter β while optimising the smooth surrogate.
    Returns the best simplex vector seen and its exact (non‑smoothed) score.
    """
    logits = np.log(start_a + 1e-20)          # avoid log(0)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step_idx, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        time_left = deadline - time.time()
        # give each β‑stage a modest budget, but never exceed remaining time
        max_iter = min(maxiter_per_step, max(20, int(time_left * 25)))

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f} step {step_idx}] new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 300_000,
                     step0: float = 0.07):
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Returns the best vector seen and its exact score.
    """
    a = init_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------


def powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    raw = idx ** (-p)
    return raw / raw.sum()


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(−lam·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=float)
    raw = np.exp(-lam * idx)
    return raw / raw.sum()


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root (central‑binomial) shape, normalised to the simplex."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    return a / a.sum()


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence (step‑function heights) that minimises
    ``evaluate_sequence``.  The routine mixes parametric warm‑starts,
    β‑annealed L‑BFGS, sub‑gradient polishing, random Dirichlet restarts
    and a final stochastic hill‑climb, all bounded by the supplied time budget.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin for the harness

    best_seq = None
    best_score = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback to a previously known construction
    # -----------------------------------------------------------------
    fallback = globals().get('height_sequence_1')
    if fallback is not None:
        try:
            val = evaluate_sequence([float(x) for x in fallback])
            print(f"Fallback score = {val:.6f}")
            if val < best_score:
                best_score = val
                best_seq = [float(x) for x in fallback]
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Quick scan of simple parametric families (power‑law & exponential)
    # -----------------------------------------------------------------
    print("\nScanning parametric families …")
    for n in (8192, 16384):
        # power‑law grid
        for p in np.linspace(0.4, 2.2, 18):
            a = powerlaw_sequence(n, p)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"Power‑law n={n} p={p:.3f} → {sc:.6f}")

        # exponential grid
        for lam in np.linspace(0.0001, 0.006, 14):
            a = exponential_sequence(n, lam)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"Exponential n={n} λ={lam:.5f} → {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Refine the best candidate with β‑annealed L‑BFGS
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120:
        print("\nSmooth continuation (L‑BFGS) …")
        a0 = np.array(best_seq, dtype=float)
        a_opt, _ = lbfgs_continuation(
            start_a=a0,
            beta_start=5.0,
            beta_end=1800.0,
            deadline=deadline - 100,
            schedule_steps=12,
            maxiter_per_step=250,
        )
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"After L‑BFGS → {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Exact max‑convolution polishing (projected sub‑gradient)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80:
        print("\nSub‑gradient polishing …")
        a0 = np.array(best_seq, dtype=float)
        a_ref, sc_ref = subgrad_optimize(
            init_a=a0,
            deadline=deadline - 60,
            max_iter=350_000,
            step0=0.07,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"Polish improved → {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Random Dirichlet restarts with full pipeline
    # -----------------------------------------------------------------
    print("\nRandom Dirichlet restarts …")
    while time.time() < deadline - 30:
        remaining = deadline - time.time()
        if remaining > 300:
            n = 32768
        elif remaining > 150:
            n = 16384
        else:
            n = 8192

        # draw a moderately spiky Dirichlet vector (α≈0.3–1.0)
        alpha = 0.3 + 0.7 * np.random.rand()
        gamma = np.random.gamma(alpha, 1.0, size=n)
        a0 = gamma / gamma.sum()

        # L‑BFGS continuation
        a_opt, _ = lbfgs_continuation(
            start_a=a0,
            beta_start=5.0,
            beta_end=1200.0,
            deadline=deadline - 20,
            schedule_steps=10,
            maxiter_per_step=200,
        )
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"Restart L‑BFGS new best → {best_score:.6f}")

        # Light sub‑gradient polish (only if still time)
        if time.time() < deadline - 10:
            a_ref, sc_ref = subgrad_optimize(
                init_a=a_opt,
                deadline=deadline - 5,
                max_iter=150_000,
                step0=0.06,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.tolist()
                print(f"Restart polish new best → {best_score:.6f}")

    # -----------------------------------------------------------------
    # 5. Final cheap stochastic hill‑climb
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("\nFinal hill‑climb …")
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_sc = compute_score(best_a)
        step = 0.05

        while time.time() < deadline and step > 1e-5:
            cand = best_a.copy()
            i, j = np.random.randint(n, size=2)
            # fractional mass transfer
            delta = step * best_a[i]
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0
            cand = project_to_simplex(cand)

            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand
                step = min(step * 1.07, 0.5)
            else:
                step *= 0.96

        if best_sc < best_score:
            best_score = best_sc
            best_seq = best_a.tolist()
            print(f"Hill‑climb improved to {best_score:.6f}")

    # -----------------------------------------------------------------
    # 6. Fallback (extremely unlikely)
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 7. Final evaluation via the provided security‑checked function
    # -----------------------------------------------------------------
    final_val = evaluate_sequence([float(x) for x in best_seq])
    print(f"\nSearch finished. Final reported score = {final_val:.6f}")

    return best_seq
