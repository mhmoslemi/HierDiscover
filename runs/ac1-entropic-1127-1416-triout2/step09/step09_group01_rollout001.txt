"""
Propose a step‑function sequence that minimises the evaluation objective.

Algorithm overview
------------------
1. Choose a problem size `n` (8192‑16384) based on the time budget.
2. Generate a large, diverse set of candidate vectors:
   * Analytic families (symmetric product, left‑heavy power‑law,
     exponential decay, Beta‑skewed, Hann/Hamming windows).
   * Random draws (Dirichlet, sparse, monotone).
   The best candidate becomes the current incumbent.
3. Refine the incumbent with several optimisation phases,
   each respecting the remaining wall‑clock time.
   * **Smooth surrogate optimisation** – Adam on the soft‑max
     parametrisation of the log‑sum‑exp approximation to the max‑convolution.
     The smoothing parameter β is annealed from 5 to 5000.
   * **Mirror (exponentiated) sub‑gradient descent** on the true max‑objective.
   * **Projected (additive) sub‑gradient descent** with momentum.
   * **Stochastic hill‑climb** (simulated‑annealing style) for final
     fine‑tuning.
4. Return the incumbent as a plain Python list of floats.
All phases are time‑aware; the routine never exceeds the supplied budget
(≈ 1000 s).  The objective is scale‑invariant, so all internal vectors are
kept normalised to sum = 1 and `compute_score` (fast) is used for intermediate
evaluations; the final call uses the provided `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation for a probability vector (sum(a) == 1):
        2 * n * max( a * a )
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) – symmetric, heavy tails controlled by α."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # handle overflow for α=0 (logprod may be -inf at ends)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) – monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(‑γ·i) – geometric decay."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_skewed_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shaped vector on [0,1].  α,β>0 (asymmetry allowed)."""
    i = np.arange(n, dtype=np.float64)
    # scale to (0,1) avoiding exact 0/1
    x = (i + 0.5) / (n + 1.0)
    w = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    a = w / w.sum()
    return a


def hann_window_family(n: int) -> np.ndarray:
    """Classic Hann (raised‑cosine) window, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return w / w.sum()


def hamming_window_family(n: int) -> np.ndarray:
    """Hamming window, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def sinusoid_family(n: int) -> np.ndarray:
    """Positive sinusoid shape (half‑wave)."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 1.0) / (n + 1.0))
    w = np.maximum(w, 0.0)
    return w / w.sum()

# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Uniform sparse probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)          # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort of a power‑law of uniforms)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]               # decreasing
    return a / a.sum()


def random_dirichlet_start(n: int, concentration: float) -> np.ndarray:
    """Dirichlet draw with symmetric concentration parameter."""
    g = np.random.gamma(shape=concentration, scale=1.0, size=n)
    if g.sum() == 0.0:
        g = np.ones_like(g)
    return g / g.sum()


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp surrogate) via Adam
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 250_000,
    beta_start: float = 5.0,
    beta_end: float = 5_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially.
    """
    eps = 1e-12
    # initialise logits from the start vector (avoid log(0))
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule β exponentially
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # soft‑max weights over k

        # gradient of the smoothed objective w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam interim best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for
    the exact objective 2·n·max(a*a).  Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))               # active index

        # sub‑gradient of the max‑entry (active terms only)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)               # exponentiated update

        # renormalise to the simplex (entropy projection)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Mirror sub‑grad interim best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (additive) sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def proj_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.6,
    momentum: float = 0.9,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent on the exact max‑objective,
    Euclidean projection onto the simplex, with momentum.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        v = momentum * v - eta * g
        a = a + v
        a = project_to_simplex(a)

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Projected sub‑grad interim best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with tiny simulated‑annealing component
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
    init_temp_factor: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a low temperature schedule.
    Moves:
      – with prob 0.5 swap two entries,
      – otherwise transfer a fraction of mass from one index to another.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise (simple projection onto simplex)
        cand = cand / cand.sum()

        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase step size when we improve
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97  # shrink after rejection

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a size that balances expressiveness and per‑iteration cost.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 16384
    elif budget_s >= 500:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # Helper to update the incumbent.
    # ------------------------------------------------------------------
    best_seq = None          # stored as a NumPy array (normalised)
    best_score = float("inf")

    def consider(vec: np.ndarray):
        """Normalise, evaluate with fast surrogate and possibly update incumbent."""
        nonlocal best_seq, best_score
        if vec.ndim != 1 or vec.size != n:
            return
        if np.isnan(vec).any() or np.isinf(vec).any():
            return
        total = vec.sum()
        if total <= 0.0:
            return
        a = vec / total
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0. Fallback to a previously known construction (if available).
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = np.array(fb_vec, dtype=np.float64)
            best_seq = best_seq / best_seq.sum()   # normalise for internal use
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Sweep analytic families (fast – cheap evaluations).
    # ------------------------------------------------------------------
    # symmetric product
    for alpha in np.linspace(0.0, 5.0, 51):
        if time.time() >= deadline - 40:
            break
        consider(symmetric_product_family(n, alpha))

    # left‑heavy power law
    for beta in np.linspace(0.0, 5.0, 51):
        if time.time() >= deadline - 40:
            break
        consider(left_heavy_power_family(n, beta))

    # exponential decay
    for gamma in np.linspace(0.0, 6.0, 51):
        if time.time() >= deadline - 40:
            break
        consider(exponential_decay_family(n, gamma))

    # window families (no parameters)
    consider(hann_window_family(n))
    consider(hamming_window_family(n))
    consider(sinusoid_family(n))

    # Beta‑skewed random draws
    for _ in range(40):
        if time.time() >= deadline - 40:
            break
        alpha = 10 ** np.random.uniform(-2.0, 1.0)   # ~0.01 … 10
        beta = 10 ** np.random.uniform(-2.0, 1.0)
        consider(beta_skewed_family(n, alpha, beta))

    # Random Dirichlet draws
    for _ in range(40):
        if time.time() >= deadline - 40:
            break
        conc = np.random.uniform(0.1, 5.0)
        consider(random_dirichlet_start(n, conc))

    # Random monotone and sparse starters
    for _ in range(30):
        if time.time() >= deadline - 40:
            break
        consider(random_monotone_start(n))
        consider(random_sparse_start(n))

    # ------------------------------------------------------------------
    # 2. Smooth optimisation (Adam) – multiple restarts while time permits.
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        # initialise from a fresh sparse start (good diversity)
        start_vec = random_sparse_start(n)
        # give Adam a dedicated deadline that leaves some buffer
        sub_deadline = time.time() + 200.0
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline=sub_deadline,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=5_000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Adam restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Mirror (entropy) sub‑gradient descent on the true max‑objective.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        rem = deadline - time.time()
        sub_deadline = time.time() + min(80.0, rem - 30.0)
        print(f"[{time.time():.1f}] starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            best_seq,
            deadline=sub_deadline,
            step0=0.8,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected (additive) sub‑gradient descent with momentum.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        rem = deadline - time.time()
        sub_deadline = time.time() + min(60.0, rem - 20.0)
        print(f"[{time.time():.1f}] starting projected sub‑gradient descent")
        a_proj, sc_proj = proj_subgrad_optimize(
            best_seq,
            deadline=sub_deadline,
            step0=0.6,
            momentum=0.9,
        )
        if sc_proj < best_score:
            best_score = sc_proj
            best_seq = a_proj.copy()
            print(f"[{time.time():.1f}] after projected sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation (many optima are symmetric or near‑symmetric).
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_sym = (best_seq + best_seq[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.copy()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final hill‑climb with simulated‑annealing moves.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (simulated annealing)")
        a_final, sc_final = hill_climb_sa(
            best_seq,
            deadline=deadline,
            max_steps=1_200_000,
            init_temp_factor=5e-4,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard: if nothing succeeded, fall back to a modest uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 8. Final official evaluation (the harness will also call evaluate_sequence).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_seq.tolist()
