"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

The optimiser works in several stages:

1. **LP‑based flattening (small n).**  
   Starting from a random density on ~256 points we solve a tiny linear
   program that maximises the total mass of a new vector `g` under the
   constraint `f * g ≤ max(f*f)`.  This is the exact convex relaxation
   described in the statement (the convolution‐bound is linear in `g`).
   Repeated updates `f ← (1‑t) f + t g` (with a decaying step‑size `t`)
   flatten the self‑convolution and quickly lower the objective.

2. **Upsampling & continuous optimisation.**  
   The best vector from stage 1 is repeatedly up‑sampled (each entry is
   split into two halves) which gives a finer grid while keeping the
   convolution shape.  After each up‑sampling we run a smooth
   log‑sum‑exp surrogate optimisation (Adam) followed by a projected
   sub‑gradient descent on the exact `max(conv)`.

3. **Stochastic polishing.**  
   Finally a simulated‑annealing hill‑climber with random swaps and
   mass‑transfer moves is executed until the time budget expires.

All stages respect a total wall‑clock budget of 1000 s, use only
numpy/scipy and keep the vector on the probability simplex (sum = 1,
non‑negative) – the scaling invariance of `evaluate_sequence` guarantees
that the reported score equals the exact objective.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact score under the convention Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """Smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Pick one of the three generators at random."""
    r = np.random.rand()
    if r < 0.4:
        return random_start_dense(n)
    elif r < 0.7:
        return random_start_sparse(n)
    else:
        return random_start_gaussian(n)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 5.0,
    beta_end: float = 8000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a differentiable soft‑max surrogate of max(conv(a,a))
    using Adam momentum and projection onto the simplex.
    Returns the best exact score observed.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # annealed β (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # soft‑max over the full convolution
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))            # stable weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of the surrogate (cross‑correlation trick)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).
    The update direction comes from the maximising offset.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for this index (unique for LP‑conv)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # tiny jitter every few thousand iterations
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = None,
) -> tuple[list[float], float]:
    """
    Random local modifications (swap or mass transfer) with a
    temperature schedule.  Accept improving moves always;
    accept worsening moves with probability exp(-Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                # transfer up to 20 % of the source mass
                delta = 0.2 * cand[i] * np.random.rand()
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: list[float]) -> np.ndarray:
    """Double length by replacing each entry a_i with a_i/2, a_i/2."""
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    return project_to_simplex(up)


# ----------------------------------------------------------------------
# 5. LP‑based flattening of the convolution (small dimensions)
# ----------------------------------------------------------------------


def lp_refine(
    init_f: np.ndarray,
    time_deadline: float,
    max_iter: int = 100,
    t_initial: float = 0.25,
    t_decay: float = 0.97,
) -> np.ndarray:
    """
    Repeatedly solve the linear programme:
        maximise   Σ g_j
        subject to g ≥ 0,
                   f * g ≤ M     (pointwise),
    where M = max(f*f).  The solution g is projected onto the simplex
    and mixed back into f.
    """
    f = init_f.astype(float).copy()
    n = f.shape[0]

    # Pre‑allocate the Toeplitz matrix that encodes convolution with f.
    # It will be rebuilt each iteration because f changes.
    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # bound for the convolution
        M = np.max(convolution_fft(f, f))

        # Build the (2n‑1) × n matrix A with A[k,i] = f[k‑i] when index in range
        A = np.zeros((2 * n - 1, n), dtype=float)
        for i in range(n):
            A[i:i + n, i] = f

        c = -np.ones(n)                     # maximise Σ g  ⇔ minimise −Σ g
        b = np.full(2 * n - 1, M)           # right‑hand side of inequalities
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            # If the LP fails (numerical issues) we abort the refinement.
            break

        g = project_to_simplex(res.x)

        # convex combination with a decaying step size
        t = t_initial * (t_decay ** (it - 1))
        f = (1.0 - t) * f + t * g
        f = project_to_simplex(f)

    return f

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises `evaluate_sequence`.  The routine respects a 1000 s
    wall‑clock budget and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the scorer after the return
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. LP‑based flattening on a modest size (n≈256)
    # ------------------------------------------------------------------
    n_lp = 256
    print(f"LP‑flattening stage – vector length = {n_lp}")

    reserve_lp = 200.0   # seconds reserved for the LP stage
    while time.time() < deadline - reserve_lp:
        # allocate a small slice for a single run
        slice_deadline = min(time.time() + 25.0, deadline - reserve_lp)
        f0 = random_start(n_lp)
        f_opt = lp_refine(f0, time_deadline=slice_deadline, max_iter=80)
        scr = compute_score(f_opt)
        if scr < best_val:
            best_val = scr
            best_seq = f_opt.tolist()
            print(f"  New best after LP: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Iterative up‑sampling & continuous refinement
    # ------------------------------------------------------------------
    upsample_steps = 2               # how many times we double the length
    while best_seq is not None and time.time() < deadline - 120.0 and upsample_steps > 0:
        print("\nUpsampling current best")
        cur = upsample_sequence(best_seq)
        cur_len = cur.shape[0]
        print(f" length after up‑sampling: {cur_len}")

        # smooth log‑sum‑exp optimisation
        slice_deadline = deadline - 80.0
        cur, scr = smooth_gradient_optimize(
            cur,
            time_deadline=slice_deadline,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=12000.0,
            step0=1.0,
        )
        exact = compute_score(cur)
        if exact < best_val:
            best_val = exact
            best_seq = cur.tolist()
            print(f"  After smooth optimisation: score={best_val:.6f}")

        # sub‑gradient refinement (more aggressive step)
        sub_deadline = deadline - 30.0
        cur_ref, scr_ref = subgrad_optimize(
            cur,
            time_deadline=sub_deadline,
            max_iter=300_000,
            step0=0.06,
        )
        exact_ref = compute_score(cur_ref)
        if exact_ref < best_val:
            best_val = exact_ref
            best_seq = cur_ref.tolist()
            print(f"  After sub‑gradient refinement: score={best_val:.6f}")

        # prepare for possible another up‑sampling
        upsample_steps -= 1

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nFinal hill‑climb polishing")
        polished_seq, polished_score = hill_climb_anneal(
            best_seq,
            time_deadline=deadline,
            max_steps=None,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"  Polished final score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Official evaluation (the harness will call evaluate_sequence again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return plain python list as required by the harness.
    return best_seq
