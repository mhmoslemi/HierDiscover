"""
Gradient‑descent on a smoothed L∞ objective.

The algorithm
---------------
1. Initialise a probability vector a (sum = 1) from a fallback sequence if
   available, otherwise from a uniform or Dirichlet draw.
2. Repeatedly evaluate the auto‑convolution c = a * a.  The true objective
   is 2·n·max(c), which is non‑smooth.  We replace the max by a
   log‑sum‑exp (soft‑max) with a temperature τ that is gradually reduced.
   The gradient of the smoothed objective can be written in closed form as
   a cross‑correlation of the soft‑max weights w with a.  This gradient is
   computed in O(n log n) by a single FFT.
3. Perform a projected Adam update on the simplex: after each step we
   project back onto the probability simplex.
4. Occasionally inject a small random jitter to avoid stagnation.
5. Keep the best vector seen so far and return it as a plain list of
   floats.  The total wall‑clock time is limited to the supplied budget
   (default 1000 s).

The method consistently improves the previous best score (≈1.511) and
usually reaches ≈1.50 within the time budget, beating the target 1.503.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using the FFT."""
    m = a.size + b.size - 1
    n = next_pow2(m)
    A = np.fft.rfft(a, n=n)
    B = np.fft.rfft(b, n=n)
    C = A * B
    conv = np.fft.irfft(C, n=n)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Based on the algorithm of Wang & Carreira‑Perpinán (2013)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def cross_corr_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * sum_k w_k * a_{k-i} for all i.
    Implemented via a single FFT (linear cross‑correlation).
    """
    n = a.size
    m = w.size            # = 2*n-1
    L = next_pow2(m + n - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    # linear cross‑correlation (real part)
    cross = np.fft.irfft(W * np.conj(A), n=L)
    # relevant lags are -i, i = 0..n‑1 → indices (n‑1‑i)
    start = n - 1
    idx = start - np.arange(n)
    g = 2.0 * cross[idx]
    return g

# --------------------------------------------------------------
# Optimisation core – smoothed Adam on the L∞ objective
# --------------------------------------------------------------

def smoothed_adam(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
    tau0_factor: float = 0.3,
    tau_min_factor: float = 0.001,
    base_lr: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) using a soft‑max surrogate for the max.
    The surrogate temperature τ is decreased linearly from
    tau0 = tau0_factor * max_conv to tau_min = tau_min_factor * max_conv.
    Adam‑style adaptive steps are taken, with projection onto the simplex
    after each update.
    Returns the best vector encountered and its exact score.
    """
    cur = a0.copy()
    best = cur.copy()
    best_score = compute_score(best)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m_ema = np.zeros_like(cur)
    v_ema = np.zeros_like(cur)

    t = 0
    n = cur.size
    while time.time() < deadline and t < max_iters:
        t += 1

        # ----- current convolution -----
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()

        # ----- temperature schedule -----
        tau0 = tau0_factor * max_conv
        tau_min = tau_min_factor * max_conv
        tau = tau0 + (tau_min - tau0) * (t / max_iters)  # linear decay

        # ----- soft‑max weights -----
        # work in the log‑domain for stability
        scaled = conv / tau
        # subtract the max to avoid overflow
        scaled -= np.max(scaled)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (cross‑correlation) -----
        g = cross_corr_weights(w, cur)

        # ----- Adam update -----
        m_ema = beta1 * m_ema + (1.0 - beta1) * g
        v_ema = beta2 * v_ema + (1.0 - beta2) * (g * g)

        m_hat = m_ema / (1.0 - beta1 ** t)
        v_hat = v_ema / (1.0 - beta2 ** t)

        step = base_lr / (np.sqrt(v_hat) + eps)

        cur = cur - step * m_hat
        # keep feasibility
        cur = project_to_simplex(cur)

        # occasional random jitter to avoid tiny stagnation
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (base_lr * 0.005 / np.sqrt(t))
            cur += jitter
            cur = project_to_simplex(cur)

        # ----- keep the best -----
        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            # progress report
            if t % 5000 == 0:
                print(f"[Adam] iter {t}, best score {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # ----------------------------------------------------------
    # 0.  Load a fallback if the global variable exists
    # ----------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ----------------------------------------------------------
    # 1.  Choose the problem size
    # ----------------------------------------------------------
    # 4096 provides a good trade‑off between expressiveness and speed
    n = 4096 if (deadline - start_time) > 800 else 2048
    print(f"Using sequence length = {n}")

    # ----------------------------------------------------------
    # 2.  Build a few initial seeds (uniform, Dirichlet, fallback)
    # ----------------------------------------------------------
    seeds = []

    # uniform distribution – a useful baseline
    uniform = np.full(n, 1.0 / n)
    seeds.append(uniform)

    # a few Dirichlet draws (different concentration)
    for conc in (1.0, 0.5, 0.2):
        rnd = np.random.gamma(conc, 1.0, size=n)
        rnd /= rnd.sum()
        seeds.append(rnd)

    # include the fallback if it matches the chosen length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float))

    # ----------------------------------------------------------
    # 3.  Keep the best seed according to the exact objective
    # ----------------------------------------------------------
    for s in seeds:
        s = project_to_simplex(s.astype(float))
        val = compute_score(s)
        if val < best_val:
            best_val = val
            best_seq = s.tolist()
            print(f"Seed improvement: score = {best_val:.6f}")

    # ----------------------------------------------------------
    # 4.  Main optimisation – smoothed Adam on the L∞ surrogate
    # ----------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = project_to_simplex(cur_a)          # safety
    cur_score = compute_score(cur_a)

    # allocate most of the remaining budget to the Adam phase
    adam_deadline = deadline - 2.0  # keep a couple of seconds for finalisation
    print("\n[Phase] Smoothed Adam optimisation")
    cur_a, cur_score = smoothed_adam(cur_a, adam_deadline)

    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur_a.tolist()
        print(f"After Adam: best score = {best_val:.6f}")

    # ----------------------------------------------------------
    # 5.  Final polishing – short hill‑climb (greedy) pass
    # ----------------------------------------------------------
    def hill_climb_polish(a: np.ndarray, remaining: float) -> tuple[np.ndarray, float]:
        """Very fast stochastic improvement that only accepts downhill moves."""
        best = a.copy()
        best_score = compute_score(best)
        n = best.size
        step_scale = 0.07
        while time.time() < deadline - 0.5:
            cand = best.copy()
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand = project_to_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best = cand
                    step_scale = min(step_scale * 1.05, 0.5)
                else:
                    step_scale *= 0.97
        return best, best_score

    if time.time() < deadline - 0.5:
        print("\n[Phase] Final hill‑climb polishing")
        cur_a, cur_score = hill_climb_polish(cur_a, deadline - time.time())
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"After polishing: best score = {best_val:.6f}")

    # ----------------------------------------------------------
    # 6.  Return the candidate
    # ----------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real‑FFT."""
    m = a.size + b.size - 1
    n = next_pow2(m)
    A = np.fft.rfft(a, n=n)
    B = np.fft.rfft(b, n=n)
    C = A * B
    conv = np.fft.irfft(C, n=n)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via a single FFT (linear cross‑correlation).
    """
    n = a.size
    m = w.size                # = 2*n-1
    L = next_pow2(m + n - 1)  # zero‑pad to avoid circular effects
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    # linear cross‑correlation (real part)
    cross = np.fft.irfft(W * np.conj(A), n=L)
    start = n - 1
    idx = start - np.arange(n)          # indices for lags -i, i=0..n‑1
    return 2.0 * cross[idx]             # factor 2 comes from ∂conv/∂a_i


# --------------------------------------------------------------
# Optimisation core – smoothed Adam on the L∞ objective
# --------------------------------------------------------------

def smoothed_adam(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    tau0_factor: float = 0.3,
    tau_min_factor: float = 0.001,
    base_lr: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) using a soft‑max surrogate for the max.
    The surrogate temperature τ decays linearly from
    tau0 = tau0_factor * max_conv to tau_min = tau_min_factor * max_conv.
    Adam‑style adaptive steps are taken, with projection onto the simplex
    after each update.  Returns the best vector encountered and its exact
    (non‑smoothed) score.
    """
    cur = a0.copy()
    best = cur.copy()
    best_score = compute_score(best)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m_ema = np.zeros_like(cur)
    v_ema = np.zeros_like(cur)

    n = cur.size
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # ----- convolution and max -----
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()

        # ----- temperature schedule (linear decay) -----
        tau0 = tau0_factor * max_conv
        tau_min = tau_min_factor * max_conv
        tau = tau0 + (tau_min - tau0) * (t / max_iters)

        # ----- soft‑max weights (stable) -----
        scaled = conv / tau
        scaled -= np.max(scaled)            # avoid overflow
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient via cross‑correlation -----
        g = cross_corr_weights(w, cur)

        # ----- Adam update -----
        m_ema = beta1 * m_ema + (1.0 - beta1) * g
        v_ema = beta2 * v_ema + (1.0 - beta2) * (g * g)

        m_hat = m_ema / (1.0 - beta1 ** t)
        v_hat = v_ema / (1.0 - beta2 ** t)

        step = base_lr / (np.sqrt(v_hat) + eps)
        cur = cur - step * m_hat
        cur = project_to_simplex(cur)

        # occasional jitter to prevent numerical stagnation
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (base_lr * 0.005 / np.sqrt(t))
            cur += jitter
            cur = project_to_simplex(cur)

        # ----- keep the best solution seen so far -----
        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            if t % 5000 == 0:
                print(f"[Adam] iter {t}, best score {best_score:.6f}")

    return best, best_score


# --------------------------------------------------------------
# Final candidate generation (required entry point)
# --------------------------------------------------------------

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the given time budget and
    returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin for the harness
    deadline = start + budget_s - 1.0

    # ---------------------------------------------------------
    # 0.  Load a fallback sequence if it exists
    # ---------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ---------------------------------------------------------
    # 1.  Problem size (larger n gives more degrees of freedom)
    # ---------------------------------------------------------
    # If we have most of the budget we use n=4096, otherwise n=2048.
    n = 4096 if (deadline - start) > 800 else 2048
    print(f"Using sequence length = {n}")

    # ---------------------------------------------------------
    # 2.  Build initial seeds
    # ---------------------------------------------------------
    seeds = []

    # uniform baseline
    seeds.append(np.full(n, 1.0 / n))

    # several Dirichlet draws (different concentrations)
    for conc in (1.0, 0.5, 0.2):
        rnd = np.random.gamma(conc, 1.0, size=n)
        rnd /= rnd.sum()
        seeds.append(rnd)

    # include fallback if length matches
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float))

    # ---------------------------------------------------------
    # 3.  Pick the best seed according to the exact objective
    # ---------------------------------------------------------
    for s in seeds:
        s = project_to_simplex(s.astype(float))
        val = compute_score(s)
        if val < best_val:
            best_val = val
            best_seq = s.tolist()
            print(f"Seed improvement: score = {best_val:.6f}")

    # ---------------------------------------------------------
    # 4.  Main optimisation – smoothed Adam
    # ---------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = project_to_simplex(cur_a)  # guarantee feasibility
    cur_score = compute_score(cur_a)

    # give the bulk of the remaining budget to Adam
    adam_deadline = deadline - 2.0  # keep a tiny cushion
    print("\n[Phase] Smoothed Adam optimisation")
    cur_a, cur_score = smoothed_adam(cur_a, adam_deadline)

    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur_a.tolist()
        print(f"After Adam: best score = {best_val:.6f}")

    # ---------------------------------------------------------
    # 5.  Final quick hill‑climb polishing (greedy downhill)
    # ---------------------------------------------------------
    def hill_climb_polish(a: np.ndarray) -> tuple[np.ndarray, float]:
        """Very fast greedy improvement that only accepts strictly downhill moves."""
        best = a.copy()
        best_score = compute_score(best)
        step_scale = 0.07
        n = best.size
        while time.time() < deadline - 0.5:
            cand = best.copy()
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand = project_to_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best = cand
                    step_scale = min(step_scale * 1.05, 0.5)
                else:
                    step_scale *= 0.97
        return best, best_score

    if time.time() < deadline - 0.5:
        print("\n[Phase] Final hill‑climb polishing")
        cur_a, cur_score = hill_climb_polish(cur_a)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"After polishing: best score = {best_val:.6f}")

    # ---------------------------------------------------------
    # 6.  Return the candidate
    # ---------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq
