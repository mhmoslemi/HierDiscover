"""
Proposes a non‑negative step‑function (a list of heights) that minimises
`evaluate_sequence`.  The algorithm consists of four phases:

1. **Exploratory scan** of simple parametrised families
   (exponential, power‑law, U‑shaped and V‑shaped) on a length chosen
   according to the remaining time budget.
2. **Projected sub‑gradient descent** on the exact `max(conv)` objective.
   The sub‑gradient at the current peak `k*` is `g_i = 2·a_{k*‑i}` for
   valid indices.  An Euclidean step is taken with a step size that would
   make the linearised peak vanish, while respecting non‑negativity and the
   probability‑simplex constraint.
3. **Random restarts**: a few additional sub‑gradient runs start from
   random Dirichlet vectors to avoid local minima.
4. **Greedy hill‑climb** (swap or mass‑transfer moves) for a final polish.

All phases are time‑controlled; the routine always returns a feasible
sequence before the hard 1000 s deadline, otherwise the harness awards
negative infinity.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) for a vector with sum = 1.
    (Same as the value returned by `evaluate_sequence` for such a vector.)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametrised families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def u_shape(n: int, p: float) -> np.ndarray:
    """
    U‑shaped: a_i ∝ ((i+1)*(n‑i))^(‑p), giving larger weights near the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1) * (n - i)) ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def v_shape(n: int, p: float) -> np.ndarray:
    """
    V‑shaped (valley in the centre):
    a_i ∝ max(i, n‑1‑i)^p, i.e. heavier at the ends.
    """
    i = np.arange(n, dtype=float)
    w = np.maximum(i, n - 1 - i) ** p if p > 0.0 else np.ones_like(i)
    return w / w.sum()

# ----------------------------------------------------------------------


def scan_families(
    n: int,
    deadline: float,
) -> tuple[np.ndarray | None, float]:
    """Exhaustively scan several one‑parameter families."""
    best_a = None
    best_sc = float("inf")

    # ----- exponential ---------------------------------------------------
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -0.5, 300)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # ----- power‑law ----------------------------------------------------
    p_grid = np.linspace(0.0, 8.0, 300)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # ----- U‑shaped ------------------------------------------------------
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = u_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # ----- V‑shaped ------------------------------------------------------
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = v_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv) using a simple Euclidean sub‑gradient method.
    At each iteration the step size is chosen so that the linearised
    peak would become zero; the step is clipped to preserve non‑negativity.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # current convolution and location of the peak
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        max_val = float(conv[k])

        # sub‑gradient of the active peak
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[idx[valid]]

        # quadratic term for step sizing
        norm_sq = np.dot(g, g)
        if norm_sq <= 0.0:
            continue

        # step that would (linearly) cancel the peak
        step_opt = max_val / norm_sq

        # enforce non‑negativity: a_i - step·g_i ≥ 0
        pos = g > 1e-18
        if np.any(pos):
            max_allowed = np.min(a[pos] / g[pos])
            step_opt = min(step_opt, 0.99 * max_allowed)

        # modest damping factor
        step = 0.8 * step_opt

        # Euclidean update and renorm to the simplex
        a = a - step * g
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        # occasional exact evaluation
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple hill‑climb (unchanged apart from a higher step budget)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk:
      * with prob 0.5 swap two entries,
      * otherwise transfer a fraction of mass from one index to another.
    Only strictly improving moves are kept; the step scale adapts.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine respects the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin before the hard timeout
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a workable length based on remaining budget
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback using a previously discovered vector
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Scan the families of parametrised shapes
    # -------------------------------------------------
    scan_deadline = deadline - 30.0
    a_scan, sc_scan = scan_families(n, scan_deadline)
    if a_scan is not None and sc_scan < best_score:
        best_score = sc_scan
        best_seq = a_scan.tolist()
        print(f"[{time.time():.1f}] after family scan best = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Projected sub‑gradient descent (multiple restarts)
    # -------------------------------------------------
    if time.time() < deadline - 200.0 and best_seq is not None:
        # primary run
        print(f"[{time.time():.1f}] start sub‑gradient descent")
        a_opt, sc_opt = subgradient_descent(
            np.array(best_seq, dtype=float),
            deadline - 150.0,
            max_iters=400_000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after sub‑grad primary best = {best_score:.6f}")

        # a few random restarts
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline - 150.0:
                break
            rand_vec = np.random.rand(n)
            rand_vec /= rand_vec.sum()
            a_r, sc_r = subgradient_descent(
                rand_vec,
                deadline - 120.0,
                max_iters=250_000,
            )
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.tolist()
                print(f"[{time.time():.1f}] restart {r} improved best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Final greedy hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=6_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: fall back to a uniform short vector if nothing succeeded
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
