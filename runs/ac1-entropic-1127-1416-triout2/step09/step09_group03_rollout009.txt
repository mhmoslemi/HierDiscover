"""
Non‑negative step‑function optimiser for ``evaluate_sequence`` (budget ≈ 1000 s).

Strategy
--------
1.  Parameter sweep – generate many analytically‐motivated families
    (exponential, power‑law, quadratic taper, mixed) and keep the best
    (fast, O(n log n) convolutions).

2.  Smooth Adam optimisation – minimise a log‑sum‑exp approximation of
    the max‑convolution on the full length (n = 2048) while staying on
    the probability simplex.

3.  Small‑scale LP refining – down‑scale the current best vector to
    n_small = 512, solve a linear programme (SciPy Highs) that maximises
    the ℓ₁‑norm of a partner vector under the current worst‑convolution
    bound, mix the partner back (convex combination) and upscale again.

4.  Stochastic hill‑climbing – tiny mass‑transfer moves with simplex
    projection finish the search.

All phases respect the remaining wall‑clock budget and keep the best
sequence seen.  The returned list is already normalised (sum = 1), so
``evaluate_sequence`` will only apply its final clipping.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (∑a = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (parameter families)
# ----------------------------------------------------------------------


def gen_exponential(n: int, q: float) -> np.ndarray:
    """a_i ∝ q**i (0<q≤1)."""
    if q == 1.0:
        v = np.ones(n, dtype=float)
    else:
        v = np.power(q, np.arange(n, dtype=float))
    v_sum = v.sum()
    if v_sum == 0:
        return None
    return v / v_sum


def gen_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) (p≥0)."""
    idx = np.arange(1, n + 1, dtype=float)
    v = idx ** (-p)
    v_sum = v.sum()
    if v_sum == 0:
        return None
    return v / v_sum


def gen_sym_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·(n-i)^(-p) – symmetric heavy‑ends shape."""
    i = np.arange(n, dtype=float)
    v = (i + 1) ** (-p) * (n - i) ** (-p)
    v_sum = v.sum()
    if v_sum == 0:
        return None
    return v / v_sum


def gen_quadratic_taper(n: int, lam: float) -> np.ndarray:
    """
    a_i ∝ 1 - lam·((i-mid)/mid)², clipped to ≥0.
    lam∈[0,1). lam=0 gives uniform, lam≈1 gives ‘U‑shape’.
    """
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    v = 1.0 - lam * ((i - mid) / mid) ** 2
    v = np.clip(v, 0.0, None)
    v_sum = v.sum()
    if v_sum == 0:
        return None
    return v / v_sum


def gen_mixed_exp_power(n: int, q: float, p: float) -> np.ndarray:
    """a_i ∝ q**i · (i+1)^(-p), then normalised."""
    v = np.power(q, np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** (-p))
    v_sum = v.sum()
    if v_sum == 0:
        return None
    return v / v_sum


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimiser – Adam
# ----------------------------------------------------------------------


def smooth_adam_optimize(start_a: np.ndarray,
                         time_deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 5.0,
                         beta_end: float = 300.0,
                         step0: float = 1.0,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses the log‑sum‑exp surrogate and keeps the iterate on the simplex.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing parameter β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # full convolution a*a
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # log‑sum‑exp weights
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()

        # gradient of the surrogate (see analysis)
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)      # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # slice to size n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        a = a_new
        # occasional progress check
        if it % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
    # final check
    cur = compute_score(a)
    if cur < best_score:
        best_score, best_a = cur, a.copy()
    return best_a, best_score

# Small‑scale LP refinement (SciPy HiGHS)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix A such that A @ b = conv(f, b).
    Column j contains f shifted down by j rows.
    """
    n = f.shape[0]
    diags = [f] * n
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = sp.diags(diagonals=diags, offsets=offsets,
                 shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum(A: sp.csr_matrix, M: float) -> np.ndarray | None:
    """
    Solve:
        maximise   Σ b_j
        s.t.       A @ b ≤ M,   b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    A_dense = A.toarray()
    c = -np.ones(n)                     # minimise -Σb ⇔ maximise Σb
    bounds = [(0, None)] * n
    ub = np.full(A_dense.shape[0], M)
    try:
        res = linprog(c, A_ub=A_dense, b_ub=ub,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if not res.success:
        return None
    b = res.x
    if b is None:
        return None
    return np.maximum(b, 0.0)


def lp_refine_small_vec(a_big: np.ndarray,
                        target_len: int,
                        time_deadline: float,
                        mix_factor: float = 0.2,
                        max_iter: int = 8) -> np.ndarray:
    """
    Down‑scale a_big to target_len, perform a few LP refinements,
    then upscale back to the original length.
    """
    # downscale via linear interpolation (keeps shape)
    old_len = a_big.shape[0]
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, target_len)
    a_small = np.interp(x_new, x_old, a_big)
    a_small = project_to_simplex(a_small)

    for it in range(max_iter):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a_small, a_small)
        M = float(conv.max())

        A = build_conv_matrix(a_small)
        b_opt = solve_lp_max_sum(A, M)
        if b_opt is None:
            break
        b_opt = project_to_simplex(b_opt)

        # convex combination
        cand = (1.0 - mix_factor) * a_small + mix_factor * b_opt
        cand = project_to_simplex(cand)

        # keep improvement only
        cand_score = compute_score(cand)
        cur_score = compute_score(a_small)
        if cand_score < cur_score:
            a_small = cand
            # slowly reduce mixing factor as we approach convergence
            mix_factor = max(mix_factor * 0.9, 0.02)

    # upscale back to original length
    x_small = np.linspace(0.0, 1.0, target_len)
    a_up = np.interp(x_old, x_small, a_small)   # note order for inverse mapping
    a_up = project_to_simplex(a_up)
    return a_up

def hill_climb(seq: np.ndarray,
               time_deadline: float,
               max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """Tiny random mass‑transfer moves with simplex projection."""
    a = seq.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score, best = cur, cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1
    return best, best_score

            best_val = refined_score
            best_seq = a_refined.tolist()
            print(f"[LP] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb (remaining budget)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline:
        a_initial = np.array(best_seq, dtype=float)
        polished_seq, polished_score = hill_climb(
            a_initial,
            time_deadline=hill_deadline,
            max_steps=800_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq.tolist()
            print(f"[hill‑climb] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity check & return
    # ------------------------------------------------------------------
    if best_seq is None:
        # ultimate fallback – uniform distribution
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return plain Python list (harness expects this)
    return best_seq
