import numpy as np, time, math, random

def project_to_simplex(v):
    ... (as before)
def compute_score(a):
    n = a.shape[0]
    conv = np.fft.irfft(np.fft.rfft(a, n=2*n) * np.fft.rfft(a, n=2*n))[:2*n-1]
    return float(2.0 * n * conv.max())

def init_central_binomial(n):
    a = np.empty(n)
    a[0] = 1.0
    for i in range(1,n):
        a[i] = a[i-1] * (2*i - 1) / (2.0*i)
    a = a / a.sum()
    return a

def shape_power_p(n, p):
    a = np.arange(1, n+1, dtype=float) ** (-p)
    a = a / a.sum()
    return a

def shape_exponential(n, theta):
    a = np.exp(-theta * np.arange(n))
    a = a / a.sum()
    return a

def hill_climb(a0, deadline, max_iter=10_000_000):
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    step_scale = 0.05
    n = a.shape[0]
    while time.time() < deadline and max_iter>0:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move delta from i to j
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonnegative
        if cand[i] < 0:
            # adjust: set delta = a[i]
            delta = a[i]
            cand[i] = 0.0
            cand[j] = a[j] + delta
        # optional swap:
        if np.random.rand() < 0.1:
            cand[i], cand[j] = cand[j], cand[i]
        # compute
        cand_score = compute_score(cand)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                step_scale = min(step_scale*1.1, 0.5)
        else:
            step_scale *= 0.99
        max_iter -= 1
    return best, best_score

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Idea
----
The objective 2·n·max(conv(a,a)) (sum(a)=1) is essentially the
L∞‑norm of the auto‑convolution.  A simple monotone power‑law
distribution a_i ∝ (i+1)^(-p) yields a very flat convolution when
p is chosen a little larger than ½.  By sweeping p we can locate a
shape whose worst convolution is dramatically smaller than the
classical central‑binomial start (which corresponds to p≈½).

Algorithm
---------
1. **Parameter sweep** – generate vectors a(p) = (i+1)^{-p} (i=0…n‑1)
   for p∈[0.5,0.7].  Keep the best on the simplex (normalised sum=1).
   Refine around the best p with a finer grid.

2. **Sub‑gradient polish** – starting from the best vector, run
   projected sub‑gradient descent on the true max‑convolution
   (gradient is obtained from the active index).  Step size decays
   as 1/√t and a tiny Gaussian jitter is added every 1000 steps
   to escape plateaus.

3. **Greedy hill‑climb** – finally perform many cheap mass‑transfer
   moves (move a fraction of mass from one coordinate to another)
   with projection onto the simplex, always keeping the best
   encountered sequence.

All phases respect the remaining wall‑clock budget, and the final
list of floats is returned.  The code uses only ``numpy`` and the
standard library; no external solvers are required.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Numerical utilities
# ------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# 1. Power‑law sweep
# ------------------------------------------------------------

def best_power_law_vector(n: int, time_deadline: float) -> np.ndarray:
    """
    Scan p∈[0.5,0.7] (coarse then fine) and return the best
    normalised vector a_i ∝ (i+1)^{-p}.
    """
    best_a = None
    best_val = float("inf")
    best_p = 0.5

    # ---------- coarse grid ----------
    coarse_ps = np.linspace(0.5, 0.7, 41)  # step 0.005
    for p in coarse_ps:
        if time.time() >= time_deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a, best_p = val, a, p

    # ---------- fine grid around the best p ----------
    # search in a narrow interval centred on best_p
    if best_a is not None:
        low = max(0.5, best_p - 0.03)
        high = min(0.8, best_p + 0.03)
        fine_ps = np.linspace(low, high, 31)   # step ≈0.002
        for p in fine_ps:
            if time.time() >= time_deadline:
                break
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a, best_p = val, a, p

    return best_a


# ------------------------------------------------------------
# 2. Sub‑gradient polishing
# ------------------------------------------------------------

def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ------------------------------------------------------------
# 3. Simple greedy hill‑climb
# ------------------------------------------------------------

def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_no_improve: int = 500_000) -> np.ndarray:
    """
    Tiny stochastic moves (move a fraction of mass from i to j)
    with projection onto the simplex.  Greedy acceptance.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05          # initial fraction of mass moved
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # safeguard against tiny negative due to round‑off
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            # keep the current point (greedy) but adapt step size
            step_scale *= 0.985
            no_improve += 1

        it += 1

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the harness' own bookkeeping
    deadline = start_time + budget_s - 5.0

    n = 2048                     # final length of the step function
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Power‑law sweep – cheap global search
    # -----------------------------------------------------------------
    pll_deadline = deadline - 300.0          # reserve time for polishing
    if time.time() < pll_deadline:
        a_pow = best_power_law_vector(n, pll_deadline)
        if a_pow is not None:
            val = compute_score(a_pow)
            print(f"[power sweep] best p‑score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_pow.tolist()

    # -----------------------------------------------------------------
    # 2. Sub‑gradient polishing (if we have time)
    # -----------------------------------------------------------------
    sub_deadline = deadline - 120.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, sub_deadline)
        val = compute_score(a_refined)
        print(f"[sub‑grad] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 3. Final greedy hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
