"""
Simple but aggressive optimizer for the “max‑convolution” evaluation.

The score for a vector a (non‑negative, sum >0) is
    score = 2·n·max_k (a * a)_k / (sum a)^2
which is invariant to scaling.  We may therefore normalise
the vector to a probability distribution (sum = 1) and minimise
the peak of the self‑convolution.

The algorithm proceeds in four stages:

1. **Length choice** – a few seconds are spent deciding a workable
   vector length n (larger n gives a tighter theoretical lower bound).

2. **Exploratory scan** – a quick grid search over several simple
   one‑parameter families (exponential, power‑law and truncated
   Gaussian) supplies a decent starting point.

3. **Multi‑restart smooth optimisation** – many short runs of a
   KL‑mirror (exponentiated‑gradient) optimiser are performed.
   Each run starts from a different random initialisation
   (uniform, exponential, power, Gaussian or a Dirichlet sample)
   and uses a log‑sum‑exp surrogate for the max‑convolution.
   The temperature β is annealed from a small value up to a large
   one (≈10 000) so that the smooth objective tightly approximates
   the true max.  The best candidate over all restarts is kept.

4. **Local refinement** – a deterministic sub‑gradient descent
   on the exact max‑convolution followed by a classic stochastic
   hill‑climb (mass swaps/transfers) polish the solution.

All phases respect the 1000 s wall‑clock limit; intermediate scores
are printed for monitoring.  The returned object is a plain Python
list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector normalised to sum = 1.
    Returns the same value that `evaluate_sequence` would give.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)  (lam≥0), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (p≥0), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Truncated Gaussian centred in the middle, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    # avoid the degenerate all‑zero case
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


# ----------------------------------------------------------------------
# Simple exhaustive scan of the families
# ----------------------------------------------------------------------


def scan_families(n: int, deadline: float):
    """Exhaustively scan three simple families and keep the best vector."""
    best_a = None
    best_sc = float("inf")

    # exponential λ grid (incl. λ=0 → uniform)
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -1, 120)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # power‑law p grid (p=0 → uniform)
    p_grid = np.linspace(0.0, 8.0, 120)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # Gaussian σ grid (σ small ⇒ narrow, σ large ⇒ flat)
    sigma_grid = np.linspace(0.1, max(1.0, n / 4.0), 120)
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        a = gaussian_shape(n, sigma)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp surrogate) – KL‑mirror descent
# ----------------------------------------------------------------------


def smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 0.5,
    beta_end: float = 10_000.0,
    eta0: float = 0.6,
    eps: float = 1e-14,
):
    """
    Minimise a smooth log‑sum‑exp approximation of the max‑convolution.
    KL‑mirror (exponentiated‑gradient) updates keep the vector on the simplex.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (exponential annealing)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution and soft‑max weights over the full convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))  # stabilised
        w = expw / expw.sum()

        # gradient of the smooth objective w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # step size (classic η₀/√t)
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update
        a = np.maximum(a * np.exp(-eta * grad), eps)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def max_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    eta0: float = 0.8,
):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    The sub‑gradient is 2·a_{s_max‑i} (zero outside the support).
    """
    a = np.maximum(start_a.astype(float), 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a /= a_sum

    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        s_max = int(conv.argmax())
        # sub‑gradient of the max entry
        idx = s_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        a = a - eta * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 4_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk:
    - with prob 0.5 swap two entries,
    - otherwise transfer a fraction of mass from one index to another.
    Only improvements are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc, best_a = sc, cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine combines:
        * quick scan of three parametric families,
        * many short mirror‑gradient restarts from diverse seeds,
        * a deterministic sub‑gradient refinement,
        * a final stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin to guarantee returning before the hard timeout
    deadline = start_time + budget_s - 2.0

    remaining = deadline - start_time
    if remaining > 800:
        n = 16384
    elif remaining > 500:
        n = 8192
    elif remaining > 300:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # optional fallback (global variable `height_sequence_1`)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score, best_seq = fb, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Scan simple families
    # ------------------------------------------------------------------
    a_scan, sc_scan = scan_families(n, deadline - 30.0)
    if a_scan is not None and sc_scan < best_score:
        best_score, best_seq = sc_scan, a_scan.tolist()
        print(f"[{time.time():.1f}] after family scan best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Multi‑restart smooth optimisation
    # ------------------------------------------------------------------
    # reserve a few seconds for the final polishing phases
    final_reserve = 35.0
    restart_deadline = deadline - final_reserve
    restart_count = 0

    while time.time() < restart_deadline:
        # pick a random initialisation method
        mode = np.random.choice(["uniform", "exp", "power", "gauss", "rand"])
        if mode == "uniform":
            a0 = np.full(n, 1.0 / n)
        elif mode == "exp":
            lam = 10 ** np.random.uniform(-5, -0.5)  # λ ∈ [1e‑5, ~0.3]
            a0 = exponential_shape(n, lam)
        elif mode == "power":
            p = np.random.uniform(0.0, 8.0)
            a0 = power_shape(n, p)
        elif mode == "gauss":
            sigma = np.random.uniform(0.1, max(1.0, n / 3.0))
            a0 = gaussian_shape(n, sigma)
        else:  # random Dirichlet
            alpha = np.random.lognormal(mean=0.0, sigma=1.0)
            a0 = np.random.gamma(alpha, 1.0, size=n)
            a0 = a0 / a0.sum()

        # smooth optimisation (mirror descent)
        a_opt, sc_opt = smooth_optimize(
            a0,
            restart_deadline,
            max_iters=180_000,
            beta_start=0.5,
            beta_end=12_000.0,
            eta0=0.65,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] restart {restart_count} smooth best = {best_score:.6f}")

        # sub‑gradient refinement (exact max)
        a_ref, sc_ref = max_subgrad_optimize(
            a_opt,
            restart_deadline,
            max_iters=120_000,
            eta0=0.9,
        )
        if sc_ref < best_score:
            best_score, best_seq = sc_ref, a_ref.tolist()
            print(f"[{time.time():.1f}] restart {restart_count} sub‑grad best = {best_score:.6f}")

        restart_count += 1

    # ------------------------------------------------------------------
    # 3. Final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=5_000_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a tiny uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
