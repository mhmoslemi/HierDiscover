def double_power_family(n, alpha, beta, epsilon=0.5):
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + epsilon, -alpha) * np.power(n - 1 - i + epsilon, -beta)
    # Replace infinities
    w[np.isnan(w) | np.isinf(w)] = 0.0
    a = w / w.sum()
    return a

def smooth_objective(z, beta):
    a = softmax(z)
    conv = convolution_fft(a, a)
    maxc = conv.max()
    exp_w = np.exp(beta * (conv - maxc))
    w = exp_w / exp_w.sum()
    # objective value: maxc + (1/beta)*log(sum exp(beta*(conv - maxc))) # but that's equal to (1/beta)*log sum exp(beta*conv)
    # compute directly:
    obj = (np.log(np.exp(beta * (conv - maxc)).sum()) + beta * maxc) / beta
    # gradient w.r.t a:
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n-2
    grad_a = 2.0 * conv_rev[n-1 : n-1 + n]
    # gradient w.r.t. z:
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return obj, grad_z
```

"""
Propose a step‑function (non‑negative sequence) that minimises the
`evaluate_sequence` objective.

The search proceeds in several phases, always respecting a global
wall‑clock budget (default 1000 s):

1. **Analytic families** – quick grid searches over a few
   parameterised families (uniform, symmetric product, double‑power
   / beta‑distribution).  The best of these seeds the optimisation.

2. **L‑BFGS refinement** – a smooth surrogate of the max‑convolution
   (log‑sum‑exp) is minimised with L‑BFGS‑B (if SciPy is available;
   otherwise Adam is used as a fallback).  The surrogate temperature
   β is increased geometrically (5 → 2000) to approach the true max.

3. **Mirror (entropy) sub‑gradient descent** – a cheap multiplicative
   update on the exact max‑objective, useful for fine‑tuning.

4. **Symmetrisation** – many optimal solutions are (approximately)
   symmetric; averaging with the reversed vector often reduces the
   maximum convolution.

5. **Stochastic hill‑climb (simulated annealing)** – final
   stochastic search to escape shallow local minima.

All intermediate scores and timestamps are printed.  The returned
object is a plain Python list of floats, ready for the external
`evaluate_sequence` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, projection, softmax)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against overflow/underflow
    w[np.isinf(w) | np.isnan(w)] = 0.0
    a = w / w.sum()
    return a


def double_power_family(n: int, alpha: float, beta: float, eps: float = 0.5) -> np.ndarray:
    """
    Independent power‑law on both ends:
        a_i ∝ (i+ε)^(-α)·(n‑1‑i+ε)^(-β)
    """
    i = np.arange(n, dtype=np.float64)
    left = np.power(i + eps, -alpha)
    right = np.power(n - 1 - i + eps, -beta)
    w = left * right
    w[~np.isfinite(w)] = 0.0
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector via sorted power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(z: np.ndarray, beta: float):
    """
    Returns (value, gradient) of the smoothed max‑convolution
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    where a = softmax(z).
    """
    a = softmax(z)
    conv = convolution_fft(a, a)               # length = 2n‑1
    maxc = conv.max()
    # stable log‑sum‑exp
    exp_w = np.exp(beta * (conv - maxc))
    sum_exp = exp_w.sum()
    obj = maxc + np.log(sum_exp) / beta

    # weights for each shift, sum to 1
    w = exp_w / sum_exp

    # gradient w.r.t. a
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)       # length = 3n‑2
    n = a.shape[0]
    grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # back‑prop through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return obj, grad_z


# ----------------------------------------------------------------------
# L‑BFGS refinement (uses SciPy if available, else falls back to Adam)
# ----------------------------------------------------------------------


def l_bfgs_refine(start_a: np.ndarray, deadline: float):
    """
    Refine `start_a` by minimising the smoothed objective with an
    increasing sequence of β values.  Returns (best_vector, best_score).
    """
    try:
        from scipy.optimize import minimize
    except Exception:  # SciPy not present – use Adam fallback
        minimize = None

    # preparation
    eps = 1e-12
    z = np.log(start_a + eps)
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    # schedule of β values (geometric progression)
    betas = [5.0, 20.0, 100.0, 500.0, 2000.0]

    for beta in betas:
        if time.time() >= deadline:
            break

        max_iters = 400  # cheap L‑BFGS work per β
        if minimize is not None:
            # L‑BFGS‑B with analytic gradient
            res = minimize(
                fun=lambda zz: smooth_obj_and_grad(zz, beta),
                x0=z,
                jac=True,
                method="L-BFGS-B",
                options={"maxiter": max_iters, "disp": False},
            )
            z = res.x
        else:
            # Adam fallback (very short run)
            # Re‑use the Adam implementation from the previous code block
            # – only a few hundred iterations are needed here.
            from math import sqrt

            # tiny Adam implementation (no external dependencies)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-12
            lr = 0.03

            for it in range(1, max_iters + 1):
                a = softmax(z)
                conv = convolution_fft(a, a)
                maxc = conv.max()
                exp_w = np.exp(beta * (conv - maxc))
                w = exp_w / exp_w.sum()
                a_rev = a[::-1]
                conv_rev = convolution_fft(a_rev, w)
                grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
                dot_ag = np.dot(a, grad_a)
                grad_z = a * grad_a - dot_ag * a

                m = beta1 * m + (1.0 - beta1) * grad_z
                v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # evaluate the current candidate
        a_cur = softmax(z)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"[{time.time():.1f}] L‑BFGS/β={beta:.1f} improved score = {sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for the true objective
    2·n·max(a*a).  The step size is η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {sc:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
        print(f"[{time.time():.1f}] final mirror sub‑grad score = {sc:.6f}")

    return best_a, best_score

# Stochastic hill‑climb with simulated annealing
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a tiny simulated‑annealing schedule.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# Main driver – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the time budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # --------------------------------------------------------------
    # Choose length – 16384 gives a good trade‑off between flexibility
    # and per‑iteration cost.
    # --------------------------------------------------------------
    n = 16384 if budget_s >= 900 else 8192
    print(f"[{time.time():.1f}] using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # 0.  Uniform baseline (helps for later symmetrisation)
    uniform = np.full(n, 1.0 / n)
    sc = compute_score(uniform)
    best_score, best_seq = sc, uniform.tolist()
    print(f"[{time.time():.1f}] uniform baseline score = {sc:.6f}")

    # --------------------------------------------------------------
    # 1. Analytic families – cheap dense grid searches
    # --------------------------------------------------------------
    # (a) symmetric product family
    for a in np.linspace(0.0, 5.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={a:.2f}) = {sc:.6f}")

    # (b) double‑power (beta‑distribution) family – independent exponents
    pow_vals = [0.0, 0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
    for p in pow_vals:
        for q in pow_vals:
            if time.time() >= deadline - 30.0:
                break
            vec = double_power_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (dbl‑p={p:.1f} q={q:.1f}) = {sc:.6f}")

    # (c) left‑heavy power‑law family
    for b in np.linspace(0.0, 5.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, b)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {sc:.6f}")

    # (d) exponential decay family
    for g in np.linspace(0.0, 6.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, g)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={g:.2f}) = {sc:.6f}")

    # --------------------------------------------------------------
    # 2. Random sparse starts – provide diverse seeds
    # --------------------------------------------------------------
    for _ in range(40):
        if time.time() >= deadline - 30.0:
            break
        vec = random_sparse_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑sparse) = {sc:.6f}")

    # --------------------------------------------------------------
    # 3. L‑BFGS (or Adam) refinement – many restarts while time permits
    # --------------------------------------------------------------
    while time.time() < deadline - 250.0:
        start_vec = random_sparse_start(n)
        a_opt, sc_opt = l_bfgs_refine(start_vec, deadline - 200.0)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] new best after L‑BFGS = {sc_opt:.6f}")

    # --------------------------------------------------------------
    # 4. Mirror (entropy) sub‑gradient descent on the exact max‑objective
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            step0=0.8,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {sc_sub:.6f}")

    # --------------------------------------------------------------
    # 5. Symmetrisation – many optimal solutions are symmetric
    # --------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score, best_seq = sc_sym, a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {sc_sym:.6f}")

    # --------------------------------------------------------------
    # 6. Final hill‑climb (simulated annealing) with tiny temperature
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {sc_final:.6f}")

    # --------------------------------------------------------------
    # 7. Safeguard fallback
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 8. Final official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats
    return best_seq
