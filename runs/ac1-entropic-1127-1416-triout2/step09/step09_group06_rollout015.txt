    a0 = start_a.copy()

    n = a0.shape[0]

    logits = np.log(a0 + 1e-12)

    m = np.zeros_like(logits)

    v = np.zeros_like(logits)

    beta1 = 0.9

    beta2 = 0.999

    eps = 1e-8

    best_a = softmax_logits(logits)

    best_score = compute_score(best_a)

    t=0

    while time.time() < deadline and t < max_iters:

        t+=1

        a = softmax_logits(logits)

        conv = convolution_fft(a,a)

        k = int(np.argmax(conv))

        idx = np.arange(n)

        j = k - idx

        mask = (j>=0) & (j<n)

        g = np.zeros_like(a)

        g[mask] = 2.0 * a[j[mask]]

        # gradient w.r.t logits

        dot = np.dot(g, a)

        # Adam update

        m = beta1 * m + (1 - beta1) * grad_logits

        v = beta2 * v + (1 - beta2) * (grad_logits ** 2)

        m_hat = m / (1 - beta1 ** t)

        v_hat = v / (1 - beta2 ** t)

        lr = init_lr / (np.sqrt(t) + 1)  # decreasing lr or keep constant?

        # Actually typical Adam uses constant lr.

        logits -= init_lr * m_hat / (np.sqrt(v_hat) + eps)

        # compute score

        cur_score = compute_score(a)

        if cur_score < best_score:

            best_score = cur_score

            best_a = a.copy()

        # optionally print every certain time

    return best_a, best_score

"""
search: multiple random restarts + β‑continuation L‑BFGS +
       Adam refinement on the exact max‑convolution objective +
       cheap final hill‑climb.

The algorithm:
 1. Choose a length n based on the remaining time.
 2. Build a modest collection of starting shapes:
        – inverse‑square‑root (“central‑binomial”) shape,
        – its reversal,
        – the uniform distribution,
        – several Dirichlet random draws,
        – optionally the global fallback `height_sequence_1`.
 3. For each start run a continuation of the smooth log‑sum‑exp
    surrogate (L‑BFGS‑B) with a finer β‑schedule and more inner
    iterations.  Keep the best shape.
 4. Refine the best candidate with Adam on the exact nonsmooth
    objective (gradient = sub‑gradient of the active convolution
    entry) while staying on the simplex via the soft‑max
    parametrisation.
 5. A very light stochastic hill‑climb (swap / fractional mass
    transfer) for any remaining seconds.
The routine respects a 1000 s wall‑clock budget and returns a plain
list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large n)."""
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function when ∑a = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where

        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))

    and g = ∇_logits f.
    """
    a = softmax_logits(logits)

    conv = convolution_fft(a, a)
    maxc = conv.max()

    # centred log‑sum‑exp
    exp_weights = np.exp(beta * (conv - maxc))
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient w.r.t. a
    w = exp_weights / Z                # length 2n‑1
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # back‑prop through the soft‑max
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 300):
    """
    Anneal β while optimising the smooth surrogate with L‑BFGS‑B.
    Returns the best shape (still on the simplex) and its exact score.
    """
    logits = np.log(start_a + 1e-20)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # allocate a per‑step budget but never exceed remaining time
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(20, int(time_left * 30)))   # ≈30 L‑BFGS iters per second

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f} step {step}/{schedule_steps}] new best "
                  f"score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution objective
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  max_iters: int = 200_000,
                  init_lr: float = 0.12):
    """
    Adam optimiser on 2·n·max(conv(a,a)) with the simplex enforced
    through a soft‑max parametrisation.
    Returns the best shape (still on the simplex) and its exact score.
    """
    n = start_a.shape[0]
    logits = np.log(start_a + 1e-20)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax_logits(logits)
    best_score = compute_score(best_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = softmax_logits(logits)

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]      # sub‑gradient of the active entry

        # chain rule through soft‑max
        dot = np.dot(g, a)
        grad_logits = a * (g - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # a simple step‑size schedule (decay with sqrt(t))
        lr = init_lr / (np.sqrt(t) + 1.0)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if t % 5000 == 0:
                print(f"[Adam iter {t}] improved score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic start – truncated central‑binomial coefficients
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape a_i = C(2i,i)/4^i, normalised to sum‑1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising ``evaluate_sequence``.
    The search combines several restarts, a β‑continuation L‑BFGS phase,
    an Adam refinement on the exact max‑convolution objective, and a
    final cheap hill‑climb.
    Returns a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin for the harness

    # ----- 1. Choose a working length -------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 800:
        n = 16384
    elif time_left > 400:
        n = 8192
    else:
        n = 4096
    print(f"Chosen length n = {n}")

    # ----- 2. Assemble a collection of diverse starts ---------------------------------
    starts = []

    # deterministic inverse‑square‑root start
    starts.append(central_binomial_sequence(n))

    # uniform distribution (the obvious baseline)
    starts.append(np.full(n, 1.0 / n))

    # reversed central‑binomial (monotonically increasing)
    starts.append(central_binomial_sequence(n)[::-1])

    # a few Dirichlet draws with different concentration parameters
    for alpha in (0.5, 2.0, 5.0):
        starts.append(np.random.dirichlet(alpha=np.full(n, alpha)))

    # optional fallback from the global variable `height_sequence_1`
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.shape[0] != n:
            fallback = np.resize(fallback, n)
        fallback = fallback / fallback.sum()
        starts.append(fallback)
        print("Added fallback sequence as a start.")
    except Exception:
        pass

    best_seq = None
    best_val = float("inf")

    # ----- 3. Run a brief L‑BFGS continuation on each start -------------------------
    for i, a0 in enumerate(starts):
        if time.time() >= deadline - 150:   # keep a generous buffer for polishing
            break
        a0 = a0 / a0.sum()    # ensure sum‑one (soft‑max parametrisation assumes it)

        # allocate a per‑run window (≈150 s or whatever remains)
        run_deadline = min(deadline - 60, time.time() + 150.0)
        print(f"\n=== Restart {i+1}/{len(starts)} (n={n}) ===")

        a_smooth, score = lbfgs_continuation(
            start_a=a0,
            beta_start=5.0,
            beta_end=400.0,
            deadline=run_deadline,
            schedule_steps=12,
            maxiter_per_step=300,
        )

        if score < best_val:
            best_val = score
            best_seq = a_smooth.tolist()
            print(f"Best after restart {i+1}: score = {best_val:.6f}")

    # ----- 4. Adam refinement on the exact max‑convolution objective --------------------
    if best_seq is not None and time.time() < deadline - 30:
        print("\n--- Adam refinement ---")
        a_ref, score_ref = adam_optimize(
            start_a=np.array(best_seq, dtype=float),
            deadline=deadline - 10,
            max_iters=200_000,
            init_lr=0.12,
        )
        if score_ref < best_val:
            best_val = score_ref
            best_seq = a_ref.tolist()
            print(f"Adam improved best score to {best_val:.6f}")

    # ----- 5. Very cheap stochastic hill‑climb ---------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("\n--- Final hill‑climb ---")
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(best_a)
        step_scale = 0.1

        while time.time() < deadline:
            cand = best_a.copy()
            if np.random.rand() < 0.5:          # swap two entries
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]
            else:                               # fractional mass transfer
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = step_scale * best_a[i]
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0.0:
                        cand[i] = 0.0
            cand = project_to_simplex(cand)
            cur_score = compute_score(cand)

            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                step_scale = min(step_scale * 1.07, 0.5)
            else:
                step_scale *= 0.96

        best_seq = best_a.tolist()
        best_val = best_score
        print(f"Hill‑climb final score = {best_val:.6f}")

    # ----- 6. Fallback (should never be needed) --------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback

    # ----- 7. Final evaluation --------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
