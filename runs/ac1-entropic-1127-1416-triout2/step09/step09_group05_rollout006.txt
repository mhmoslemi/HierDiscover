def hill_climb_sa(start_seq, deadline, max_steps=3_000_000):
   a = start_seq.copy()
   n = a.shape[0]
   cur_a = a.copy()
   cur_sc = compute_score(cur_a)
   best_a = cur_a.copy()
   best_sc = cur_sc
   step_scale = 0.1
   step = 0
   T_start = 0.2
   while step < max_steps and time.time() < deadline:
       T = T_start * (1 - step / max_steps)  # linear cooling
       cand = cur_a.copy()
       # move type ...
       if np.random.rand() < 0.5:
           i, j = np.random.randint(n, size=2)
           cand[i], cand[j] = cand[j], cand[i]
       else:
           i, j = np.random.randint(n, size=2)
           if i != j:
               delta = step_scale * cur_a[i]
               cand[i] -= delta
               cand[j] += delta
       # ensure nonnegative; project onto simplex
       cand = project_to_simplex(cand)
       sc = compute_score(cand)
       if sc < cur_sc or np.random.rand() < np.exp(-(sc - cur_sc) / (T + 1e-12)):
           cur_a = cand
           cur_sc = sc
           if sc < best_sc:
               best_sc = sc
               best_a = cand.copy()
               # increase step_scale a bit
               step_scale = min(step_scale * 1.05, 0.5)
       else:
           step_scale *= 0.97
       step += 1
   return best_a, best_sc

"""
search for a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the aperiodic self‑convolution.
The algorithm combines three ideas:

1. **Diverse initialisations** – uniform, exponential, power‑law,
   a new “arcsine” shape and a few random Dirichlet draws.
2. **Smooth optimisation** – minimise a log‑sum‑exp (soft‑max)
   approximation of the max‑convolution.  The variable is the
   *logits* of a soft‑max (so the simplex constraint is always
   satisfied).  Adam on the logits together with an exponential
   β‑schedule (smooth → exact max) gives fast, stable progress.
3. **Stochastic hill‑climbing with simulated‑annealing** – after the
   gradient stage we polish the best vector by random swaps / mass
   transfers, accepting occasional worsening moves with a decreasing
   temperature.

All phases respect the global 1000 s wall‑clock budget.
"""

import time
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padded FFT)."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution of real vectors `a` and `b`
    using a real FFT (zero‑padded to the next power of two).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) for a vector already normalised to sum=1.
    (Matches the value returned by `evaluate_sequence` for such a vector.)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (used only by the hill‑climb)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negative entries that sum to one.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families (used for diverse seeds)
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)   (lam≥0, lam=0 ⇒ uniform)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)   (p≥0, p=0 ⇒ uniform)."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def arcsine_shape(n: int, eps: float = 1e-14) -> np.ndarray:
    """
    Discrete analogue of the arcsine density on {1,…,n}:
        a_i ∝ 1 / sqrt(i·(n+1‑i))
    This shape is known to flatten the self‑convolution.
    """
    i = np.arange(1, n + 1, dtype=float)
    w = 1.0 / np.sqrt(i * (n + 1 - i) + eps)
    return w / w.sum()


def random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(alpha,…,alpha)."""
    return np.random.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Smooth optimisation (soft‑max + Adam on logits)
# ----------------------------------------------------------------------


def smooth_optimize_softmax(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 0.5,
    beta_end: float = 3000.0,
    lr: float = 0.8,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-14,
) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth log‑sum‑exp approximation of the max‑convolution.
    The feasible set (non‑negative, sum = 1) is enforced by keeping the
    parameters as logits of a soft‑max (θ) and updating θ with Adam.
    """
    n = start_a.shape[0]

    # initialise logits (θ) from the starting vector
    theta = np.log(start_a + eps)

    m = np.zeros_like(theta)
    v = np.zeros_like(theta)

    best_a = start_a.copy()
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # -------------------------------------------------
        # 1. Current probability vector a (soft‑max)
        # -------------------------------------------------
        a = np.exp(theta)
        a /= a.sum()

        # -------------------------------------------------
        # 2. Smooth max of the convolution (log‑sum‑exp)
        # -------------------------------------------------
        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # soft‑max over the convolution values (stabilised)
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()                            # size 2n‑1, sums to 1

        # -------------------------------------------------
        # 3. Gradient of the smooth objective w.r.t. a
        # -------------------------------------------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]       # shape (n,)

        # -------------------------------------------------
        # 4. Convert to gradient w.r.t. logits (θ) using
        #    the Jacobian of the soft‑max.
        # -------------------------------------------------
        avg = np.dot(grad_a, a)                          # scalar = ⟨grad_a, a⟩
        grad_theta = a * (grad_a - avg)                  # shape (n,)

        # -------------------------------------------------
        # 5. Adam update on θ
        # -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        theta -= step

        # -------------------------------------------------
        # 6. Occasionally evaluate the *exact* (non‑smoothed) score
        # -------------------------------------------------
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    final_sc = compute_score(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated annealing
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
    initial_temp: float = 0.25,
) -> Tuple[np.ndarray, float]:
    """
    Greedy random walk that occasionally accepts worsening moves.
    Moves are either a swap of two entries or a small transfer of mass.
    Temperature decays linearly to zero; the acceptance probability is
    exp(−Δ/ T).  Returns the best vector seen.
    """
    a_cur = start_seq.copy()
    n = a_cur.size
    cur_sc = compute_score(a_cur)
    best_a = a_cur.copy()
    best_sc = cur_sc

    step_scale = 0.12
    step = 0

    while step < max_steps and time.time() < deadline:
        # linear temperature schedule
        T = initial_temp * (1.0 - step / max_steps)

        cand = a_cur.copy()
        if np.random.rand() < 0.5:                # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * a_cur[i]
                cand[i] -= delta
                cand[j] += delta

        # re‑project onto the simplex (keeps the sum = 1)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)

        # acceptance test (Metropolis rule)
        if sc < cur_sc or (T > 0 and np.random.rand() < np.exp(-(sc - cur_sc) / (T + 1e-12))):
            a_cur = cand
            cur_sc = sc
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                step_scale = min(step_scale * 1.07, 0.5)    # become bolder after success
        else:
            step_scale *= 0.97                               # become more cautious after a reject

        step += 1

    return best_a, best_sc
