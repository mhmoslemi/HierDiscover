"""
Multi‑start soft‑max gradient descent for minimizing
`evaluate_sequence`.

The algorithm:
1. Choose a reasonable length `n` (2048–4096) depending on the
   allocated budget.
2. Generate a diverse pool of seed vectors:
   * deterministic shapes (sine‑squared, beta‑family)
   * several Dirichlet draws (dense and sparse)
   * random sparse‑support vectors
   * an optional fallback sequence if available.
3. Keep the best few seeds (according to the exact objective).
4. For each seed run a projected gradient descent on a smooth
   approximation of the max‑convolution objective:
   - soft‑max with a temperature parameter `α` that is slowly increased
   - gradient w.r.t. the vector is obtained by a single FFT‑based
     cross‑correlation
   - after each step the vector is projected onto the probability
     simplex.
   The schedule uses a decreasing step size `η_t = η₀ / sqrt(t)`.
5. Finally polish the best vector with a short greedy hill‑climb that
   only accepts improving moves.
6. Return the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
    { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sine‑squared shape a_i = sin²(π·(i+1)/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ ((i+1)*(n‑i))**p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w = np.clip(w, 0, None)
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration `conc` (≈1 → uniform)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def generate_sparse_support(n: int, m: int) -> np.ndarray:
    """Place mass on `m` random positions (1 ≤ m ≤ n)."""
    indices = np.random.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=float)
    # assign random positive values before normalisation
    a[indices] = np.random.rand(m) + 0.1
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed gradient descent on max‑convolution
# ----------------------------------------------------------------------


def smoothed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    alpha_start: float = 0.5,
    alpha_max: float = 2000.0,
    alpha_inc: float = 0.001,
    report_interval: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of the max‑conv.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    L = 2 * n - 1
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    alpha = alpha_start
    while time.time() < deadline:
        t += 1
        # ----- convolution of a with itself (full) -----
        conv = convolution_fft(a, a)                     # length L

        # ----- increase temperature (α) slowly -----
        alpha = min(alpha_max, alpha + alpha_inc)

        # ----- soft‑max weights -----
        scaled = alpha * conv
        m = np.max(scaled)                               # for stability
        w = np.exp(scaled - m)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum                                    # shape (L,)

        # ----- gradient of the smoothed objective -----
        rev_w = w[::-1]
        conv_all = convolution_fft(a, rev_w)            # length n+L-1 = 3n-2
        # extract cross‑correlation part corresponding to indices 0..n‑1
        g = 2.0 * conv_all[L - n : L][::-1]              # shape (n,)

        # ----- step size schedule -----
        step = step0 / np.sqrt(t)

        # ----- gradient step and projection -----
        a -= step * g
        a = project_to_simplex(a)

        # ----- occasional exact evaluation -----
        if t % report_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[GD] iter {t:>7d}, best {best_score:.6f}")

    # final check before returning
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb polishing (greedy)
# ----------------------------------------------------------------------


def hill_climb(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
    step_scale_start: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Accept‑only stochastic moves that improve the exact score."""
    a = a0.copy()
    n = a.shape[0]
    best_score = compute_score(a)
    best_a = a.copy()
    step_scale = step_scale_start
    for step in range(max_steps):
        if time.time() >= deadline:
            break

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction locally/global
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand /= cand.sum()                     # re‑normalise (numerical drift)
        cur = compute_score(cand)
        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.96
    return best_a, best_score

        cur_vec = np.array(best_seq, dtype=float)
        cur_vec /= cur_vec.sum()
        cur_vec, cur_score = hill_climb(cur_vec, deadline, max_steps=800_000)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_vec.tolist()
            print(f"[Phase 2] Polished best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation through the provided wrapper
    # ------------------------------------------------------------------
    final_seq = [float(x) for x in best_seq]
    final_val = evaluate_sequence(final_seq)
    print(f"\n[Result] Final reported score = {final_val:.6f}")
    return final_seq
