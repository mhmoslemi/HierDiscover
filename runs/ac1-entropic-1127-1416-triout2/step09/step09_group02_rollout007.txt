def flatten_gradient_descent(a, target, deadline, step0=0.1):
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = compute_score(cur_a)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and gradient
        conv = convolution_fft(cur_a, cur_a)
        diff = conv - target
        # compute gradient: 2 * convolution(diff, cur_a[::-1]), slice appropriate region
        grad_full = convolution_fft(diff, cur_a[::-1])  # length 3n-2
        grad = 2.0 * grad_full[len(cur_a)-1:len(cur_a)-1+len(cur_a)]
        # step size schedule
        step = step0 / np.sqrt(it)
        cur_a = cur_a - step * grad
        # project
        cur_a = project_to_simplex(cur_a)
        # evaluate
        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
        # occasional jitter
        if it % 500 == 0:
            jitter = np.random.randn(len(cur_a)) * step0 * 0.01 / np.sqrt(it)
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)
        # maybe break after some iterations
    return best_a, best_score

"""
Propose a step‑function with a low `evaluate_sequence` value.

Algorithm
---------
1. **Initial seeding** – a handful of analytically‑motivated vectors
   (sine‑squared, symmetric beta‑families, arcsine‑law, random Dirichlet
   draws and, if available, the historic `height_sequence_1`).  The best
   seed is kept.

2. **Flattening by L2‑gradient descent** – minimise the squared error
   between the auto‑convolution `a*a` and the perfectly flat target
   `t = 1/(2n‑1)`.  The gradient of the loss can be written as a
   convolution, so each iteration costs only a pair of FFTs.  After every
   step the vector is projected onto the probability simplex
   `{a_i≥0, Σa_i=1}`.

3. **Sub‑gradient refinement of the true objective** – as in the
   reference implementation, use the active‑index sub‑gradient of
   `max(a*a)` and project back to the simplex.

4. **Final greedy hill‑climb** – stochastic local moves that are kept
   only when they improve the exact score.

The whole pipeline respects a 1000 s wall‑clock budget; intermediate
best scores are printed for monitoring.  The function returns the final
list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sine‑squared shape a_i = sin^2(pi*(i+1)/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p .
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discretised arcsine law: a_i ∝ 1/√(x_i (1‑x_i)),
    with x_i = (i+½)/n.  The distribution has heavy mass at the ends,
    which helps flatten the auto‑convolution.
    """
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Return a random probability vector drawn from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 2 – L2 flattening (gradient descent on ‖a*a‑t‖²)
# ----------------------------------------------------------------------


def flatten_by_gradient(
    a: np.ndarray,
    target: np.ndarray,
    deadline: float,
    step0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Minimise the L2 distance between the auto‑convolution of `a`
    and the flat target `target`.  After each gradient step the vector
    is projected back onto the simplex.
    Returns the best vector encountered and its exact score.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = compute_score(best)

    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution
        conv = convolution_fft(cur, cur)
        diff = conv - target                     # length 2n‑1

        # gradient = 2·(diff * rev(cur))  (slice the central n entries)
        grad_full = convolution_fft(diff, cur[::-1])       # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]          # central slice

        # simple sqrt‑decay step size
        step = step0 / np.sqrt(it)
        cur = cur - step * grad
        cur = project_to_simplex(cur)

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(it))
            cur += jitter
            cur = project_to_simplex(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

    return best, best_score


# ----------------------------------------------------------------------
# Phase 3 – Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on `2·n·max(conv(a,a))`.
    Stops at the given deadline.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))

        # gradient of the active entry (see description)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        step = step0 / np.sqrt(t)
        cur -= step * g
        cur = project_to_simplex(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur += jitter
            cur = project_to_simplex(cur)

    return best, best_score


# ----------------------------------------------------------------------
# Phase 4 – Greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local moves (swap or local mass transfer) that are kept
    only when they improve the exact score.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        cand /= cand.sum()                     # re‑normalise to avoid drift
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                         # become more conservative

        steps += 1

    return best, best_score

    The function respects the time budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------

        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a tractable length – 4096 when we have enough time,
    #    otherwise 2048.
    # ------------------------------------------------------------------
    n = 4096 if total_time > 750.0 else 2048
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Build a collection of seed vectors
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 7):
        seeds.append(generate_beta_family(n, p))
    seeds.append(generate_arcsine(n))

    # random Dirichlet draws (both dense and sparse)
    for _ in range(4):
        seeds.append(random_dirichlet(n, conc=1.0))
    for _ in range(4):
        seeds.append(random_dirichlet(n, conc=0.3))

    # include the fallback when length matches
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # ------------------------------------------------------------------
    # 3. Pick the best initial seed (exact objective)
    # ------------------------------------------------------------------
    for s in seeds:

            print(f"Initial seed improvement: score = {best_val:.6f}")

    cur_a = cur_a / cur_a.sum()

    # ------------------------------------------------------------------

    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
