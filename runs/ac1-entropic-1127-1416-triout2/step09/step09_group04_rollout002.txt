def hill_climb_anneal_v2(seq, time_deadline):
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    start = time.time()
    
    t0 = best_score * 0.02
    t_min = best_score * 1e-6
    
    while time.time() < time_deadline:
        elapsed = time.time() - start
        T = max(t_min, t0 * (1 - elapsed / (time_deadline - start)))
        # Choose move type
        move_type = np.random.rand()
        cand = a.copy()
        if move_type < 0.3:
            # Gaussian perturbation
            sigma = 0.1 * T / best_score  # scale sigma with temperature
            cand += np.random.randn(n) * sigma
        elif move_type < 0.6:
            # transfer mass between two indices
            i, j = np.random.randint(n, size=2)
            delta = 0.1 * cand[i]    # 10% of i
            cand[i] -= delta
            cand[j] += delta
        else:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
    return best_a.tolist(), best_score

"""
Multi‑level search for a non‑negative step function that minimises
`evaluate_sequence`.

Algorithm
---------
0.  Try a fallback construction if the global `height_sequence_1`
    variable is available.

1.  Coarse exploration (n≈256) – many cheap random restarts using a
    smooth log‑sum‑exp surrogate of the max‑convolution.  Adam with
    simplex projection is applied for a few seconds per restart.
    The best vector (sum = 1) is kept.

2.  Progressive up‑sampling – repeatedly duplicate each entry (mass
    halved) to double the length (256→512→1024→2048→4096 …).  After each
    up‑sampling a projected sub‑gradient descent on the exact
    objective (2·n·max(conv)) is performed for a time slice.

3.  Final polishing – a simulated‑annealing hill‑climber that applies
    several kinds of local moves (Gaussian perturbation, mass transfer,
    swaps).  Moves that improve the exact score are always accepted;
    worsening moves are accepted with probability exp(−Δ/T) with a
    temperature that cools to zero.

All phases respect the 1000 s wall‑clock budget; the best sequence
found is returned as a plain Python list.  The code uses only NumPy
and the already‑imported `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the constraint Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators (different shapes)
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on O(sqrt(n)) positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """Smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start_bimodal(n: int) -> np.ndarray:
    """Two Gaussian bumps."""
    x = np.arange(n)
    sigma = n / 12.0
    a = (np.exp(-0.5 * ((x - n / 3) / sigma) ** 2) +
         np.exp(-0.5 * ((x - 2 * n / 3) / sigma) ** 2))
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Choose one of several generators at random."""
    r = np.random.rand()
    if r < 0.35:
        return random_start_dense(n)
    elif r < 0.55:
        return random_start_sparse(n)
    elif r < 0.80:
        return random_start_gaussian(n)
    else:
        return random_start_bimodal(n)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 10.0,
    beta_end: float = 5000.0,
    step0: float = 1.2,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a differentiable soft‑max surrogate of max(conv(a,a))
    using Adam and simplex projection.
    Returns the best exact score seen during optimisation.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # geometric annealing of β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed objective
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        # stable soft‑max weights
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of the soft‑max (cross‑correlation trick)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 400 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).
    Uses decreasing steps and Euclidean projection onto the simplex.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the maximal entry

        # sub‑gradient for this index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = None,
) -> tuple[list[float], float]:
    """
    Simulated‑annealing style hill climbing.
    Random local moves: Gaussian perturbation, mass transfer, or swap.
    Accept improving moves always; worsening moves with probability
    exp(-Δ/T) where T cools linearly to a tiny value.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0

    # temperature parameters (relative to current best)
    T0 = best_score * 0.025
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        move_type = np.random.rand()
        if move_type < 0.35:
            # Gaussian perturbation of the whole vector
            sigma = 0.07 * T / best_score
            cand += np.random.randn(n) * sigma
        elif move_type < 0.70:
            # Transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.12 * cand[i]       # up to 12 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        else:
            # Simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        # Acceptance test
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq):
    """
    Double the length by replacing each entry a_i with two
    adjacent entries a_i/2, a_i/2.  The total mass stays equal to 1.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises `evaluate_sequence`.  The routine respects a 1000 s
    wall‑clock budget and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the external harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback construction (if the global variable exists)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Coarse random‑restart optimisation (n≈256)
    # ------------------------------------------------------------------
    n_coarse = 256
    print(f"Coarse phase – vector length = {n_coarse}")

    # reserve time for later stages (≈ 400 s)
    reserve_time = 400.0
    while time.time() < deadline - reserve_time:
        # each restart gets a few seconds, never exceeding the remaining slice
        remaining = deadline - time.time() - reserve_time
        slice_sec = max(4.0, min(12.0, remaining * 0.35))
        slice_deadline = time.time() + slice_sec

        a0 = random_start(n_coarse)
        a_opt, _ = smooth_optimize(
            a0,
            time_deadline=slice_deadline,
            max_iters=120_000,
            beta_start=10.0,
            beta_end=5000.0,
            step0=1.2,
        )
        exact_score = compute_score(a_opt)
        if exact_score < best_val:
            best_val = exact_score
            best_seq = a_opt.tolist()
            print(f"  New best (coarse) : score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling & sub‑gradient refinement
    # ------------------------------------------------------------------
    target_lengths = [512, 1024, 2048, 4096]
    for target_n in target_lengths:
        if best_seq is None:
            break
        # Upsample step‑by‑step until we reach the target length
        while len(best_seq) < target_n and time.time() < deadline - 80.0:
            best_seq = upsample_sequence(best_seq).tolist()
            up_score = compute_score(np.array(best_seq))
            if up_score < best_val:
                best_val = up_score
                print(f"Upsampled to {len(best_seq)} : score = {best_val:.6f}")

        # Sub‑gradient refinement on the current length
        if time.time() >= deadline - 50.0:
            break   # not enough time for a full refinement

        print(f"\nSub‑gradient refinement at length {len(best_seq)}")
        slice_deadline = time.time() + min(30.0, deadline - time.time() - 30.0)
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=slice_deadline,
            max_iter=300_000,
            step0=0.08,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  Refined score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final hill‑climb polishing (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\nHill‑climb polishing (annealed)")
        polished_seq, polished_score = hill_climb_anneal(
            best_seq,
            time_deadline=deadline,
            max_steps=None,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"  Polished score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Final official evaluation (for our own logging)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
