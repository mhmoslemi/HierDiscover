def eg_subgrad_optimize(init_a, deadline, max_iter=500000, eta0=0.3):
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iter+1):
        if time.time() >= deadline: break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute gradient
        grad = np.zeros_like(a)
        # compute contribution: for each i where j=k-i in [0,n-1]
        # use vectorized: indices i = np.arange(n); j=k-i; mask = (j>=0)&(j<n)
        idx = np.arange(len(a))
        j = k - idx
        mask = (j >= 0) & (j < len(a))
        grad[mask] = 2 * a[j[mask]]
        # step size
        eta = eta0 / np.sqrt(t)
        # exponentiated update: a = a * np.exp(-eta * grad)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0/len(a))
        else:
            a = a / a_sum
        # occasionally add jitter to avoid stagnation
        if t % 5000 == 0:
            # small random multiplicative jitter
            jitter = np.exp(0.01*np.random.randn(len(a))*eta)
            a = a * jitter
            a = a / a.sum()
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score
```

"""
Propose a low‑upper‑bound step‑function.

The algorithm proceeds as follows:
1. Choose a problem size n based on the remaining budget.
2. Build several promising initial shapes (fallback, central_binomial,
   arcsine, sinusoidal and a random Dirichlet).  All are projected onto the
   probability simplex (∑a = 1) – the evaluation function is scale‑invariant.
3. For each initial shape (until time runs out):
   a) **β‑annealed L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of
      max (a⋆a).  The smoothing parameter β is increased geometrically from a
      small to a large value (≈2000) so that the final solution is close to
      the true max‑convolution objective.
   b) **Exponentiated‑gradient sub‑gradient descent** – works directly on the
      exact max‑convolution objective using the sub‑gradient at the active
      index and a KL‑type (exponentiated) update, which is often more
      aggressive than plain Euclidean sub‑gradient steps.
   c) **Projected sub‑gradient polishing** – a short run of the classic
      Euclidean sub‑gradient method provides a final fine‑tune.
   d) **Stochastic hill‑climb** – random swaps / fractional mass transfers
      with an adaptive step‑size explore the local neighbourhood.
   The best sequence seen in any phase is kept as the current global best.
4. Return the best sequence as a plain ``list[float]``.  All helper
   functions are top‑level, use only NumPy / SciPy and respect the 1000 s
   wall‑clock budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation for a vector on the probability simplex (∑a = 1).
    Returns the quantity   2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient (logits)
# ----------------------------------------------------------------------


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically‑stable softmax that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = smooth approximation of max(conv(a,a))
        g = gradient w.r.t. the logits.
    a = softmax(logits) lives on the simplex.
    """
    a = softmax_logits(logits)

    conv = convolution_fft(a, a)
    maxc = conv.max()

    # log‑sum‑exp centred at maxc for stability
    exp_weights = np.exp(beta * (conv - maxc))
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient w.r.t. a
    w = exp_weights / Z                       # length 2n‑1
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)      # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # chain rule back to logits (softmax Jacobian)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 200):
    """
    Anneal β while optimising the smooth surrogate.
    Returns the best simplex vector found and its exact (2·n·max) score.
    """
    logits = np.log(start_a + 1e-20)                # initialise logits
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(20, int(time_left * 30)))   # ≈30 L‑BFGS iters per sec

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.1f} (step {step})] new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient optimiser (KL‑type update)
# ----------------------------------------------------------------------


def eg_subgrad_optimize(init_a: np.ndarray,
                        deadline: float,
                        max_iter: int = 200_000,
                        eta0: float = 0.5):
    """
    Sub‑gradient descent on max (conv) with an exponentiated‑gradient (KL)
    update.  Operates on the simplex, retains scale‑invariance.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # locate the active convolution index
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step‑size (decreasing)
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update and re‑normalisation
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            jitter = np.exp(0.01 * np.random.randn(n) * eta)
            a = a * jitter
            a = a / a.sum()

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient optimiser (classic)
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.05):
    """
    Classic projected sub‑gradient descent on the exact max‑convolution
    objective.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random jitter
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap stochastic hill‑climb (swap / fractional transfer)
# ----------------------------------------------------------------------


def hill_climb(start_a: np.ndarray, deadline: float):
    """
    Final random local‑search: swaps and small mass transfers with an
    adaptive step‑scale.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    step_scale = 0.08
    while time.time() < deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # fractional transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i] * np.random.rand()
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic starters
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """
    Inverse‑square‑root (central‑binomial) shape:
           a_i = C(2i,i) / 4^i   ,  normalised to ∑a = 1.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * (i + 1))
    a /= a.sum()
    return a


def arcsine_shape(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density:
           a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i))
    """
    idx = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((idx + 0.5) * (n - 0.5 - idx))
    a /= a.sum()
    return a


def sinusoidal_shape(n: int) -> np.ndarray:
    """
    Symmetric sinusoidal shape: a_i ∝ sin(π*(i+0.5)/n)
    """
    idx = np.arange(n, dtype=float)
    a = np.sin(np.pi * (idx + 0.5) / n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The pipeline combines:
        • several analytically motivated initial shapes,
        • β‑annealed L‑BFGS on a smooth surrogate,
        • exponentiated‑gradient sub‑gradient descent,
        • projected Euclidean sub‑gradient polishing,
        • a final stochastic hill‑climb.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # -----------------------------------------------------------------
    # Choose a working length (larger may give a tighter theoretical bound)
    # -----------------------------------------------------------------
    if deadline - time.time() > 800:
        n = 16384
    elif deadline - time.time() > 600:
        n = 8192
    else:
        n = 4096
    print(f"Working length n = {n}")

    # -----------------------------------------------------------------
    # Assemble a collection of initial vectors
    # -----------------------------------------------------------------
    init_vectors = []

    # 0. Optional fallback (global variable may exist)
    try:
        fb = [float(x) for x in height_sequence_1]   # type: ignore
        fb = np.array(fb, dtype=float)
        # Adjust to required length (truncate or pad with zeros)
        if len(fb) > n:
            fb = fb[:n]
        elif len(fb) < n:
            fb = np.concatenate([fb, np.zeros(n - len(fb))])
        fb = fb / fb.sum()
        init_vectors.append(("fallback", fb))
        print(f"Added fallback init (pre‑score ≈ {evaluate_sequence(fb.tolist()):.6f})")
    except Exception:
        pass

    # 1. Central‑binomial
    init_vectors.append(("central_binomial", central_binomial_sequence(n)))

    # 2. Arcsine shape
    init_vectors.append(("arcsine", arcsine_shape(n)))

    # 3. Sinusoidal shape
    init_vectors.append(("sinusoidal", sinusoidal_shape(n)))

    # 4. Random Dirichlet (one sample)
    rnd = np.random.rand(n)
    rnd = rnd / rnd.sum()
    init_vectors.append(("random", rnd))

    best_seq = None
    best_score = float("inf")
    best_name = None

    # -----------------------------------------------------------------
    # Iterate over the initial candidates (stop when time runs out)
    # -----------------------------------------------------------------
    for name, a0 in init_vectors:
        if time.time() >= deadline:
            break

        print(f"\n>>> Optimising from initial shape: {name}")

        # -------------------------------------------------------------
        # Phase 1 – β‑annealed L‑BFGS (smooth surrogate)
        # -------------------------------------------------------------
        remaining = deadline - time.time()
        phase1_deadline = time.time() + min(remaining * 0.35, 200.0)

        a_smooth, smooth_score = lbfgs_continuation(
            start_a=a0,
            beta_start=2.0,
            beta_end=2000.0,
            deadline=phase1_deadline,
            schedule_steps=12,
            maxiter_per_step=200,
        )
        print(f"   After smooth L‑BFGS: score = {smooth_score:.6f}")

        # -------------------------------------------------------------
        # Phase 2 – exponentiated‑gradient sub‑gradient descent
        # -------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 5:
            continue
        phase2_deadline = time.time() + min(remaining * 0.30, 200.0)

        a_eg, eg_score = eg_subgrad_optimize(
            init_a=a_smooth,
            deadline=phase2_deadline,
            max_iter=200_000,
            eta0=0.5,
        )
        print(f"   After EG sub‑grad:  score = {eg_score:.6f}")

        # -------------------------------------------------------------
        # Phase 3 – short Euclidean sub‑gradient polishing
        # -------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5:
            a_polished, pol_score = subgrad_optimize(
                init_a=a_eg,
                deadline=deadline,            # use all remaining time for polishing
                max_iter=200_000,
                step0=0.05,
            )
            print(f"   After Euclidean sub‑grad polish: score = {pol_score:.6f}")
        else:
            a_polished, pol_score = a_eg, eg_score

        # -------------------------------------------------------------
        # Phase 4 – final stochastic hill‑climb (if any time left)
        # -------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 2:
            a_final, final_score = hill_climb(a_polished, deadline)
            print(f"   After hill‑climb:  score = {final_score:.6f}")
        else:
            a_final, final_score = a_polished, pol_score

        # -------------------------------------------------------------
        # Register global best
        # -------------------------------------------------------------
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.tolist()
            best_name = name
            print(f"*** New overall best from {name}: {best_score:.6f}")

    # -----------------------------------------------------------------
    # Safeguard fallback (should never happen)
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence.")

    # Final verification using the provided evaluator
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Best reported score = {final_score:.6f} (source: {best_name})")
    return best_seq
