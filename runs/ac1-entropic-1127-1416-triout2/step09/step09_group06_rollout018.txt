"""
Optimisation of the step‑function bound 2·n·max(conv(a,a)).
The programme improves on the reference implementation by

* trying several initialisations (central‑binomial, power‑law, noisy perturbations)
* a more aggressive β‑annealing (up to β≈2000) with many L‑BFGS iterations
* a back‑tracking exact sub‑gradient descent (more iterations, larger step0)
* a final stochastic hill‑climb.

All heavy work respects the 1000 s wall‑clock budget, and the best sequence
found (already normalised to sum‑one) is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (efficient for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector already on the
    probability simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))   (smooth max)
        g = ∂f/∂logits  (chain‑rule through the soft‑max)
    """
    a = softmax_logits(logits)                      # a lives on the simplex
    conv = convolution_fft(a, a)                    # length 2·n‑1
    maxc = conv.max()

    # smooth max via log‑sum‑exp
    exp_weights = np.exp(beta * (conv - maxc))      # centred for stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient w.r.t. a (see derivation in the original code)
    w = exp_weights / Z
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3·n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # chain rule to logits
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 300):
    """
    Anneal the smoothing parameter β while optimising the smooth surrogate
    with L‑BFGS‑B.  Returns the best simplex vector found and its exact score.
    """
    logits = np.log(start_a + 1e-20)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # allocate iterations based on remaining wall‑clock time
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(30, int(time_left * 35)))   # ≈35 L‑BFGS iters / s

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.1f} (step {step}/{len(betas)})] "
                  f"new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing – back‑tracking projected sub‑gradient
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 400_000,
                     step0: float = 0.12):
    """
    Projected sub‑gradient descent on the exact 2·n·max(conv) objective.
    Uses a simple back‑tracking line‑search: the step size is halved until
    the objective improves.
    """
    a = init_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    step = step0
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))               # active index

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # back‑tracking line search
        improved = False
        trial_step = step
        while trial_step > 1e-9:
            cand = a - trial_step * g
            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)
            if cand_score < best_score:
                a = cand
                best_a = cand
                best_score = cand_score
                improved = True
                break
            trial_step *= 0.5

        # if we could not improve, shrink the base step size
        if not improved:
            step *= 0.7
        else:
            # slowly decay the step size (standard sub‑gradient schedule)
            step = step0 / np.sqrt(t)

        # occasional jitter to escape tiny plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        # early‑stop when the improvement per 10 k iterations is negligible
        if t % 10000 == 0:
            cur = compute_score(a)
            if best_score - cur < 1e-7:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple deterministic start – central‑binomial coefficients
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape a_i = C(2i,i)/4^i, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Alternative low‑dimensional parametrisation (power‑law)
# ----------------------------------------------------------------------


def power_law_sequence(alpha: float,
                       n: int,
                       gamma: float = 0.0,
                       lam: float = 0.0) -> np.ndarray:
    """
    a_i ∝ (i+1+γ)^(-α)·exp(-λ·i)   ,   re‑normalised to sum = 1.
    α∈(0,1)  ;  γ≥0  ;  λ≥0.
    """
    i = np.arange(n, dtype=float)
    a = (i + 1.0 + gamma) ** (-alpha)
    if lam != 0.0:
        a *= np.exp(-lam * i)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a

    for alpha in (0.48, 0.50, 0.52):
        init_pool.append(power_law_sequence(alpha, n, gamma=0.0, lam=0.0))
        init_pool.append(power_law_sequence(alpha, n, gamma=2.0, lam=0.0))
        init_pool.append(power_law_sequence(alpha, n, gamma=0.0, lam=1e-4))

    # noisy perturbations of the central shape
    base = init_pool[0]
    for _ in range(5):
        pert = base + 0.02 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert = project_to_simplex(pert)
        init_pool.append(pert)

    # -----------------------------------------------------------------
    # 3. Aggressive L‑BFGS continuation on each start
    # -----------------------------------------------------------------
    for idx, a0 in enumerate(init_pool, 1):
        if time.time() >= deadline - 200.0:
            break
        print(f"\n=== L‑BFGS run {idx}/{len(init_pool)} ===")
        a_opt, score_opt = lbfgs_continuation(
            start_a=a0,
            beta_start=5.0,
            beta_end=2000.0,
            deadline=deadline - 150.0,
            schedule_steps=12,
            maxiter_per_step=300,
        )
        if score_opt < best_val:
            best_val = score_opt
            best_seq = a_opt.tolist()
            print(f"New overall best after L‑BFGS: score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Exact max‑objective polishing (projected sub‑gradient)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print("\n--- Sub‑gradient polishing ---")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 50.0,
            max_iter=500_000,
            step0=0.12,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑grad improved overall best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Very light stochastic hill‑climb (swap / mass‑transfer moves)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Final hill‑climb ---")
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(best_a)
        step_scale = 0.07

        while time.time() < deadline and step_scale > 1e-5:
            cand = best_a.copy()
            if np.random.rand() < 0.6:   # more frequent fractional transfer
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = step_scale * best_a[i]
                    cand[i] -= delta
                    cand[j] += delta
                    cand[i] = max(cand[i], 0.0)
            else:                         # occasional pure swap
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]

            cand = project_to_simplex(cand)
            cur_score = compute_score(cand)

            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                step_scale = min(step_scale * 1.08, 0.5)
            else:
                step_scale *= 0.95

        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.tolist()
            print(f"Hill‑climb final score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Fallback – extremely unlikely that we have no candidate
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Final reporting (use the external evaluator for the last check)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
