def adam_optimize(start_a, deadline, max_iters=200000):
    a = start_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_sc = compute_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute gradient of smooth max with β schedule?
        # we could use direct subgradient of max (like subgrad_optimize) but that is nondifferentiable.
        # Use smoothed max with log-sum-exp with high β (like start 0.5 go to 1e5)
        # We can define β schedule like beta = min(1000, 0.1 * t**0.5) or exponentially increasing.
        # Provide gradient as earlier.
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # Choose β large for accurate approximation but avoid overflow. Use large β gradually.
        beta = min(500.0, 0.1 * t)  # increase linearly up to 500.
        # smoothed weights:
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()
        # Compute gradient:
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[len(a)-1:len(a)-1+len(a)]
        # Adam updates:
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.01 / (np.sqrt(v_hat) + eps)  # learning rate scaling
        a = a - step * m_hat
        # Project onto simplex and non-negative
        a = project_to_simplex(np.maximum(a, 0.0))
        # periodic best check:
        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc

def fast_hill_climb_anneal(start_a, deadline, init_step=0.2):
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a,a)
    cur_sc = 2.0*n*float(conv.max())
    best_a = a.copy()
    best_sc = cur_sc
    step_scale = init_step
    # schedule temperature: start with some relative tolerance
    temp = cur_sc * 0.001   # start tolerance
    temp_decay = 0.99995  # each iteration multiplies
    steps = 0
    while time.time() < deadline:
        # choose src, dst
        src = np.random.randint(n)
        if a[src] <= 0:
            continue
        dst = np.random.randint(n-1)
        if dst >= src:
            dst += 1
        # choose delta = step_scale * a[src]
        delta = step_scale * a[src]
        # incremental update
        conv_cand = conv.copy()
        conv_cand[src:src+n] -= 2.0*delta*a
        conv_cand[dst:dst+n] += 2.0*delta*a
        conv_cand[2*src] += delta*delta
        conv_cand[2*dst] += delta*delta
        conv_cand[src+dst] -= 2.0*delta*delta
        cand_max = conv_cand.max()
        cand_sc = 2.0*n*cand_max
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            # Prob accept
            prob = np.exp(-(cand_sc - cur_sc) / temp)
            if np.random.rand() < prob:
                accept = True
        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a.copy()
            step_scale = min(step_scale*1.02, 0.5)  # increase slowly
        else:
            step_scale *= 0.98
        temp *= temp_decay
        steps += 1
        if steps % 100000 == 0:
            # recompute conv fully to avoid drift
            conv = convolution_fft(a,a)
            cur_sc = 2.0*n*float(conv.max())
        # also optionally check occasionally for best via compute_score, but cur_sc already exact.
    return best_a, best_sc

"""
Optimising a non‑negative step‑function to minimise
    2·n·max(a⊛a) / (∑a)²
as evaluated by the provided `evaluate_sequence` function.

The search consists of four stages, each respecting the overall
wall‑clock limit (≈1000 s):

1. Analytic start – a one‑parameter family
      a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ
   is scanned on a fine grid (α∈[0,5]), then refined around the best
   value.

2. Smoothed exponential‑gradient descent – minimises a
   log‑sum‑exp (soft‑max) surrogate of the true max‑convolution.
   The “temperature’’ β is annealed, while a KL‑mirror (exponentiated‑gradient)
   step with a √t‑decaying stepsize is used.  The current best exact score
   is recorded throughout.

3. Projected sub‑gradient descent on the exact objective
   (max‑convolution) with Euclidean projection onto the probability simplex.

4. Stochastic annealed hill‑climb – mass is transferred between two
   positions; the convolution is updated incrementally in O(n) time.
   Moves that increase the objective are occasionally accepted according
   to a temperature schedule, allowing escape from local minima.
   The stepsize adapts to the recent acceptance rate.

The best sequence found in any phase is returned as a plain Python list
of floats.  A fallback sequence (`height_sequence_1` if present) is also
considered.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector a.
    The vector is first normalised to sum = 1 (the ratio is scaling‑invariant).
    """
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    a_norm = a / s
    n = a_norm.shape[0]
    conv = convolution_fft(a_norm, a_norm)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytic start (α‑grid)
# ----------------------------------------------------------------------


def shape_alpha(alpha: float, n: int) -> np.ndarray:
    """
    Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex.
    The expression is evaluated in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    # log of the product (i+0.5)*(n-0.5-i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # w = exp(-α·logprod) = ((i+½)(n‑½‑i))⁻ᵅ
    w = np.exp(-alpha * logprod)
    # Clip extreme values that may have overflowed to inf
    w = np.clip(w, 0.0, np.finfo(np.float64).max)
    a = w / w.sum()
    return a


def find_best_alpha_grid(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Coarse + refined search over α∈[0,5] to obtain a good initial shape."""
    best_a = None
    best_sc = float("inf")
    best_alpha = None

    # coarse scan (step ≈0.05)
    alphas = np.linspace(0.0, 5.0, 101)
    for α in alphas:
        if time.time() >= deadline:
            break
        a = shape_alpha(α, n)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            best_alpha = α

    # refined scan around the best coarse point (±0.2, step ≈0.005)
    if best_alpha is not None and time.time() < deadline:
        lo = max(0.0, best_alpha - 0.2)
        hi = best_alpha + 0.2
        alphas_ref = np.linspace(lo, hi, 81)
        for α in alphas_ref:
            if time.time() >= deadline:
                break
            a = shape_alpha(α, n)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                best_alpha = α

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Smooth exponential‑gradient optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 250_000,
                        beta_start: float = 0.5,
                        beta_end: float = 800.0,
                        eta0: float = 0.7,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed max‑convolution using an exponentiated‑gradient
    (KL‑mirror) update.  β (temperature) is exponentially annealed from
    `beta_start` to `beta_end`.  Step size η decays as η₀/√t.
    Returns the best exact score observed.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # annealing schedule for β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        # soft‑max weights (log‑sum‑exp gradient)
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smoothed objective (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)      # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # KL‑mirror step
        eta = eta0 / np.sqrt(it)
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent with Euclidean projection onto the simplex."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the *chosen* max‑convolution index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Fast annealed stochastic hill‑climb (incremental convolution updates)
# ----------------------------------------------------------------------


def hill_climb_anneal(start_a: np.ndarray,
                      deadline: float,
                      init_step: float = 0.12,
                      temp_factor: float = 0.9995,
                      recompute_interval: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Random walk that transfers a fraction of mass from a source index to a
    destination index.  The full convolution is updated incrementally in
    O(n) time.  Moves that increase the objective are accepted with a
    probability that decays with a temperature schedule, allowing escape
    from local minima.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)               # exact convolution
    cur_sc = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_sc = cur_sc

    step_scale = init_step
    temp = cur_sc * 0.001                       # initial temperature
    steps = 0

    while time.time() < deadline:
        # pick source and destination (must be distinct)
        src = np.random.randint(n)
        if a[src] <= 0.0:
            continue
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # random fraction of src's mass to move
        delta = step_scale * a[src] * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_sc = 2.0 * n * cand_max

        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a.copy()
            # slowly increase the step size on success
            step_scale = min(step_scale * 1.03, 0.5)
        else:
            # slightly shrink step size on rejection
            step_scale *= 0.97

        # temperature schedule (geometric decay)
        temp *= temp_factor

        steps += 1
        if steps % recompute_interval == 0:
            # full recomputation to avoid drift
            conv = convolution_fft(a, a)
            cur_sc = 2.0 * n * float(conv.max())

    return best_a, best_sc
