```
def adam_smooth_optimize(start_a, deadline, max_iters=200_000, beta_start=0.5, beta_end=5000.0, alpha=0.1):
    a = np.maximum(start_a, 1e-12)
    a /= a.sum()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_a=a.copy(); best_sc=compute_score(a)
    for it in range(1, max_iters+1):
        if time.time()>= deadline: break
        # schedule beta
        t = it / max_iters
        beta = beta_start * (beta_end/beta_start)**t
        conv = convolution_fft(a, a)
        w_exp = np.exp(beta * (conv - conv.max()))  # shift
        w = w_exp / w_exp.sum()
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = -alpha * m_hat / (np.sqrt(v_hat) + eps)  # gradient descent step
        a = a + step
        # ensure positivity and simplex
        a = np.maximum(a, 0)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a = a / s
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc; best_a = a.copy()
    # final evaluation
    sc = compute_score(a)
    if sc < best_sc: best_sc=sc; best_a=a.copy()
    return best_a, best_sc
```

```
def hill_climb_simulated_annealing(start_seq, deadline, max_steps=2_000_000):
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)
    # current temperature based on max steps
    T_start = 0.01 * best_sc  # scaling
    T_end = 1e-5
    for step in range(max_steps):
        if time.time() >= deadline:
            break
        t = step / max_steps
        T = T_start * (T_end / T_start) ** t  # exponential cooling
        # propose move
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            # swap
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                # transfer fraction
                delta = np.random.rand() * 0.1 * cand[i]  # up to 10% of mass
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        delta_sc = sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept move
            best_a = cand
            best_sc = sc
        # occasionally reduce accepted only best? We always keep best anyway
    return best_a, best_sc
```

assistantfinal**Solution Overview**

The code searches for a non‑negative sequence `a` that minimizes  

\[
F(a)=\frac{2\,n\;\max(a*a)}{\bigl(\sum_i a_i\bigr)^2},
\]

i.e. the value returned by `evaluate_sequence`.  
Because the objective is *scale invariant* we can restrict the search to vectors with
\(\sum_i a_i = 1\) and minimise \(\max(a*a)\).

The algorithm consists of several independent phases that respect the 1000 s wall‑clock
limit:

1. **Parameter‑scan** – cheap exhaustive scan of two one‑parameter families
   (exponential and power‑law) for several lengths `n`.  
2. **Adam optimisation** – minimises a smooth log‑sum‑exp approximation of the
   maximum convolution using the Adam optimiser (mirrored on the simplex).  
3. **Sub‑gradient descent** – works directly on the true max‑convolution,
   always moving opposite to a sub‑gradient of the active constraint.  
4. **Simulated‑annealing hill‑climb** – a stochastic local search that can
   escape shallow local minima.  

All phases keep the current best candidate and stop early if the global deadline
is reached.  The final sequence is returned as a plain Python list.

```python
"""Search for a non‑negative step‑function with minimal max‑convolution.

The routine follows four stages:
  1. Scan simple exponential / power‑law families (fast, cheap).
  2. Adam optimiser on a smooth log‑sum‑exp surrogate of the max.
  3. Sub‑gradient descent on the exact max‑convolution.
  4. Simulated‑annealing hill‑climb for final polishing.

All phases respect a global wall‑clock deadline (default 1000 s).  The
returned sequence is a plain list of floats ready for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) under the normalisation sum(a)=1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p <= 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


def scan_exponential_and_power(
    n: int, deadline: float
) -> tuple[np.ndarray | None, float]:
    """Exhaustive grid scan of the two families."""
    best_a = None
    best_sc = float("inf")

    # λ grid – include λ=0 → uniform
    lam_grid = np.concatenate(
        (np.array([0.0]), np.logspace(-6, -1, 300))
    )
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # p grid – include p=0 → uniform
    p_grid = np.linspace(0.0, 8.0, 300)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 0.5,
    beta_end: float = 5000.0,
    alpha: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth max:
        f_β(a) = (1/β)·log Σ_i exp(β·(a*a)[i]).
    Uses KL projection onto the simplex after each step.
    """
    a = np.maximum(start_a.astype(float), 1e-14)
    a /= a.sum()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weights (log‑sum‑exp)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smoothed objective
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # Adam update (mirror descent on the simplex)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = -alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a + step, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    eta0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient method for  max(a*a)."""
    a = np.maximum(start_a.astype(float), 1e-14)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_idx = int(conv.argmax())
        # construct the sub‑gradient g_j = 2·a[max_idx‑j] where defined
        g = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        j_idx = np.arange(lo, hi + 1, dtype=int)
        k_idx = max_idx - j_idx
        g[j_idx] = 2.0 * a[k_idx]

        eta = eta0 / np.sqrt(it)
        a = np.maximum(a - eta * g, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill climbing (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk with Metropolis acceptance.
    Moves:
        * swap two entries,
        * or transfer a random fraction of mass from i to j.
    """
    cur = start_seq.copy()
    n = cur.size
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # Temperature schedule (exponential decay)
    T_start = max(0.02 * best_sc, 1e-3)
    T_end = 1e-6

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        t = step / max_steps
        T = T_start * (T_end / T_start) ** t

        cand = cur.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.1 * cand[i] * np.random.rand()   # up to 10 % of i
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)
        delta_sc = cand_sc - cur_sc

        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand, cand_sc

        # occasional progress report (lightweight)
        if step % 250_000 == 0 and step > 0:
            print(f"[{time.time():.1f}] SA step {step:,} – best score {best_sc:.6f}")

    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for the sequence that minimises `evaluate_sequence`.
    The algorithm combines cheap family scans, Adam optimisation,
    sub‑gradient descent and a final simulated‑annealing hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to guarantee return before the hard timeout
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a working length according to time left
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n_candidates = [16384, 8192]
    elif remaining > 500:
        n_candidates = [8192, 4096]
    else:
        n_candidates = [4096, 2048]

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Fallback: try the supplied global `height_sequence_1`
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score, best_seq = fb, fallback
    except Exception:
        pass

    # -------------------------------------------------
    # Iterate over candidate lengths
    # -------------------------------------------------
    for n in n_candidates:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] ---- exploring length n={n} ----")

        # ---- 1. simple family scan ---------------------------------
        a_scan, sc_scan = scan_exponential_and_power(
            n, deadline - 30.0
        )
        if a_scan is not None and sc_scan < best_score:
            best_score = sc_scan
            best_seq = a_scan.tolist()
            print(f"[{time.time():.1f}] new best after family scan: {best_score:.6f}")

        # ---- 2. Adam optimisation (multiple restarts) -------------
        if best_seq is not None and len(best_seq) == n:
            for restart in range(2):
                if time.time() >= deadline - 50.0:
                    break
                # perturb the current best a little before each restart
                pert = np.array(best_seq, dtype=float)
                pert += 0.02 * np.random.rand(n)
                pert = project_to_simplex(pert)

                a_opt, sc_opt = adam_smooth_optimize(
                    pert,
                    deadline - 30.0,
                    max_iters=200_000,
                    beta_start=0.5,
                    beta_end=10_000.0,
                    alpha=0.07,
                )
                if sc_opt < best_score:
                    best_score, best_seq = sc_opt, a_opt.tolist()
                    print(f"[{time.time():.1f}] Adam restart {restart} improved to {best_score:.6f}")

        # ---- 3. Sub‑gradient descent --------------------------------
        if best_seq is not None and len(best_seq) == n:
            a_sub, sc_sub = subgradient_descent(
                np.array(best_seq, dtype=float),
                deadline - 30.0,
                max_iters=120_000,
                eta0=0.15,
            )
            if sc_sub < best_score:
                best_score, best_seq = sc_sub, a_sub.tolist()
                print(f"[{time.time():.1f}] sub‑gradient step improved to {best_score:.6f}")

        # ---- 4. Final simulated‑annealing hill‑climb ---------------
        if best_seq is not None and len(best_seq) == n and time.time() < deadline - 5.0:
            a_final, sc_final = hill_climb_sa(
                np.array(best_seq, dtype=float),
                deadline,
                max_steps=1_500_000,
            )
            if sc_final < best_score:
                best_score, best_seq = sc_final, a_final.tolist()
                print(f"[{time.time():.1f}] SA hill‑climb improved to {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded, fall back to a tiny uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # Return a plain Python list
    return best_seq
```
