"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

The algorithm consists of three phases:

1. **Fast parametric search (small scale)**
   – Scan a two‑parameter family  a_i ∝ ((i+δ)(n‑1‑i+δ))^{-α}  (a discrete
     beta‑type distribution) on a cheap small size (n≈2048).
   – Also scan simple left‑heavy power‑law and exponential families.
   – Keep the best parameters (α,δ) and optionally improve them by mixing
     with the uniform distribution.

2. **Large‑scale refinement**
   – Build the best shape on the full target length (≈16384) and run a
     smoothed log‑sum‑exp optimisation (Adam) with an annealed smoothing
     parameter β.  Several restarts with small random perturbations are
     performed while time permits.

3. **Fine‑tuning**
   – Mirror (entropy) sub‑gradient descent on the true max‑objective,
     optional symmetrisation, and a short simulated‑annealing hill‑climb.

All steps respect a global wall‑clock budget (default 1000 s).  The
function returns a plain Python list of floats ready for `evaluate_sequence`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.ifft(C, n=nfft).real[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Analytic / parametric families
# ----------------------------------------------------------------------


def symmetric_product_family_delta(n: int, alpha: float, delta: float = 0.5) -> np.ndarray:
    """
    a_i ∝ ((i+δ)(n‑1‑i+δ))^{-α}   (beta‑type, symmetric).
    """
    i = np.arange(n, dtype=np.float64)
    term1 = i + delta
    term2 = (n - 1 - i) + delta
    prod = term1 * term2
    w = np.exp(-alpha * np.log(prod))
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{-β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i), monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation of the log‑sum‑exp surrogate of max‑convolution
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 2.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)  # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothed weights over convolution entries
        conv = convolution_fft(a, a)
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()

        # gradient of smoothed max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# Mirror (entropy) sub‑gradient descent on the true max‑objective

def mirror_subgrad_optimize(start_a: np.ndarray,
                           deadline: float,
                           step0: float = 0.5,
                           check_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for 2·n·max(a*a).
    Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Stochastic hill‑climb with simulated annealing

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_200_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex:
    - swap two entries or transfer a fraction of mass,
    - low temperature allows occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
