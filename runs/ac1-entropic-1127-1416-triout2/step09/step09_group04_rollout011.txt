def lp_step(f, M, time_deadline):
    n = len(f)
    # Build A_ub
    # Using sparse matrix for speed
    rows = []
    cols = []
    data = []
    for j in range(n):
        # rows k = j ... j+n-1 correspond to f[i], i = 0..n-1
        row_start = j
        row_end = j + n  # exclusive
        rows.extend(range(row_start, row_end))
        cols.extend([j] * n)
        data.extend(f.tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Bounds are each row <= M
    b = np.full(2*n-1, M)
    c = -np.ones(n) # maximize sum -> minimize -sum
    # Solve LP with high efficiency
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'time_limit': max(0, time_deadline - time.time()), 'presolve': True})
    if not res.success:
        return None
    return res.x
```

"""
Improved search for a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – start from a known good construction
   (`height_sequence_1`) if it is present, otherwise from a dense random
   Dirichlet start.  The vector is normalised to sum to 1 (the objective
   is scale‑invariant).

2. **LP‑guided mass redistribution** – given the current vector `f`
   compute M = max(conv(f,f)).  Solve the linear program

        maximise   Σ g_i
        subject to  (f * g)[k] ≤ M   for every convolution index k
                    g_i ≥ 0

   (`f * g` is linear in `g`).  The solution tries to add as much extra
   mass as possible while respecting the same peak‑convolution bound.
   The LP is solved with `scipy.optimize.linprog` on a sparse Toeplitz
   matrix that represents the convolution operator.

   The raw solution `g` is renormalised to unit sum, then mixed with the
   current iterate:
        f ← (1‑t)·f + t·g ,  with a decreasing step‑size `t`.
   Only moves that strictly improve the exact score are kept (monotone
   descent).  This phase rapidly flattens the self‑convolution.

3. **Upsampling** – after the LP phase the best vector is split into
   twice as many entries (each entry is halved and duplicated).  This
   doubles the length while keeping the total mass unchanged and gives
   additional degrees of freedom.

   on the true objective `2·n·max(conv(f,f))` with Euclidean projection

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the invariant Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k

def random_start_gaussian(n: int) -> np.ndarray:

    a /= a.sum()
    return a

def lp_step(f: np.ndarray, M: float, time_deadline: float):
    """
    Solve
        maximise   Σ g_i
        subject to (f * g)[k] ≤ M  for all k
                   g_i ≥ 0
    Returns the optimal g as a dense ndarray, or None on failure.
    """
    n = f.shape[0]
    # Build the (2n‑1) × n Toeplitz matrix of the convolution operator.
    # Column j contains f shifted down by j rows.
    # Use sparse diagonals for speed.
    diags = []
    offsets = []
    for d in range(n):
        diags.append(f[: n - d])  # length n‑d
        offsets.append(d)         # diagonal offset
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n, dtype=float)          # maximize Σ g_i  →  minimize -Σ g_i
    bounds = [(0, None)] * n

    remaining = max(0.0, time_deadline - time.time())
    # `linprog` may ignore unknown options – we only pass the time limit if supported.
    opts = {"presolve": True}
    if hasattr(linprog, "options"):
        opts["time_limit"] = remaining

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options=opts)
    except Exception as e:
        print(f"linprog exception: {e}")
        return None

    if res.success:
        return np.array(res.x, dtype=float)
    else:
        # Infeasibility can happen because of numerical slop; tolerate it.
        # `res.message` often contains useful diagnostics.
        print(f"LP not successful (status {res.status}): {res.message}")
        return None

    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)
    return up


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).
    The update direction comes from the index attaining the maximum convolution.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the maximal entry

        # Sub‑gradient for this index (derived from d/d a_i of conv[k])
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional random jitter to avoid dead‑ends
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = None,
) -> tuple[list[float], float]:
    """
    Random local modifications (swap or mass transfer) with a temperature schedule.
    Accept improving moves always; otherwise accept with probability exp(-Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]   # move 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats)
    that minimises `evaluate_sequence`.  The routine respects a
    1000 s wall‑clock budget and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness after return
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. If a previous construction is available, use it as a warm start
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback = project_to_simplex(np.array(fallback, dtype=float))
        val = evaluate_sequence(fallback.tolist())
        print(f"Fallback initial score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working length (start with 1024, later upsample to 2048)
    # ------------------------------------------------------------------
    n = 1024
    if best_seq is not None and len(best_seq) == n:
        f = np.array(best_seq, dtype=float)
    else:
        f = random_start_dense(n)
    f = project_to_simplex(f)

    # ------------------------------------------------------------------
    # 2. LP‑guided refinement (monotone descent)
    # ------------------------------------------------------------------
    if _HAS_SCIPY:
        print("\n--- LP‑guided mass redistribution phase ---")
        # Reserve about 300 s for the later smoothing / hill‑climb phases
        lp_phase_end = deadline - 300.0
        it = 0
        cur_score = compute_score(f)
        while time.time() < lp_phase_end:
            # 2a. current peak of the self‑convolution
            conv = convolution_fft(f, f)
            M = float(conv.max())
            # 2b. solve LP
            g_raw = lp_step(f, M, lp_phase_end)
            if g_raw is None:
                print("LP step failed – aborting LP phase.")
                break
            sum_g = g_raw.sum()
            if sum_g <= 0.0:
                break
            # 2c. normalise g to unit sum (feasibility is preserved)
            g = g_raw / sum_g
            # 2d. mixing – decreasing step size, monotone acceptance
            t = max(0.05, 0.4 / np.sqrt(it + 1))
            new_f = (1.0 - t) * f + t * g
            new_f = project_to_simplex(new_f)
            new_score = compute_score(new_f)
            if new_score < cur_score:                # accept only improvements
                f = new_f
                cur_score = new_score
                if new_score < best_val:
                    best_val = new_score
                    best_seq = f.tolist()
                    print(f"  LP iter {it:3d}: improved score = {best_val:.6f}")
                it += 1
            else:
                # If no improvement, shrink the step size more aggressively
                t *= 0.5
                if t < 1e-3:
                    break
        print(f"LP phase finished after {it} iterations. Best score = {best_val:.6f}")

    else:
        print("\nSciPy not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # 3. Upsample to 2048 entries (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print("\n--- Upsampling to length 2048 ---")
        up = upsample_sequence(best_seq)
        up_val = compute_score(up)
        if up_val < best_val:
            best_val = up_val
            best_seq = up.tolist()
            print(f"  Upsampled score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print("\n--- Sub‑gradient refinement ---")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 90.0,
            max_iter=400_000,
            step0=0.04,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  Refined score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic hill‑climb polishing (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Hill‑climb polishing (annealed) ---")
        polished_seq, polished_score = hill_climb_anneal(
            best_seq,
            time_deadline=deadline,
            max_steps=None,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"  Polished score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final fallback – a uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7. Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
