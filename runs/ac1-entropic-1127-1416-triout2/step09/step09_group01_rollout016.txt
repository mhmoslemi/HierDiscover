from scipy.optimize import linprog

def solve_lp_max_sum(f, M):
    n = f.shape[0]
    A = build_conv_matrix(f)
    b = np.full(2*n-1, M)
    # want to maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g
```

"""
Search for best step-function heights minimizing the evaluation objective.  
Algorithm:
1. Initialise a population of candidates using analytic families and random draws.
2. Refine the best candidate with a smooth log‑sum‑exp surrogate (Adam).
3. Repeatedly solve a linear‑program that maximises the total mass that can be
   added without increasing the current maximal convolution (the classic
   “LP‑refinement” step).  After each LP we blend the current vector with the
   optimiser and renormalise.
4. Symmetrise the vector (many optima are palindromic).
5. Finetune with a cheap simulated‑annealing hill‑climb.
All stages respect the global time‑budget (default 1000 s).  The routine
returns a plain Python list of non‑negative floats.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
# SciPy imports guarded
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

# Helper utilities ---------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution using real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = rfft(a, n=nfft)
    B = rfft(b, n=nfft)
    C = A * B
    conv = irfft(C, n=nfft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective assuming sum(a) == 1: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    # If all non‑positive, return uniform
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()

# Analytic families -------------------------------------------------------

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α), symmetrised, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against overflow at the ends
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a

def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β), monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a

def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a

# Random starts -----------------------------------------------------------

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse (≈ 5 %) uniform probability vector."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)

def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector built from powered uniform draws."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]         # decreasing
    a = a / a.sum()
    return a

# Smooth surrogate optimisation (Adam) -----------------------------------

def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 120_000,
    beta_start: float = 5.0,
    beta_end: float = 3000.0,
    lr: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise from logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothed max‑convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()           # soft‑max weights over k

        # gradient w.r.t. a (derivation from chain‑rule)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam iter {it} – new best {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam finished – final best {best_score:.6f}")

    return best_a, best_score

# Linear‑program refinement ------------------------------------------------

def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """Return the (2n‑1) × n sparse matrix A with A[k,i] = f_i if 0≤k‑i<n else 0."""
    n = f.shape[0]
    i = np.arange(n, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)
    # outer sum to obtain all k = i + j
    rows = np.add.outer(i, j).ravel()
    cols = np.tile(j, n)
    data = np.repeat(f, n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_max_sum_under_M(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve max Σ g_j subject to A·g ≤ M·1, g≥0.  Return g or None."""
    if not SCIPY_AVAILABLE:
        return None
    n = f.shape[0]
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise –Σg => maximise Σg
    bounds = [(0, None)] * n
    # SciPy's 'highs' method is fast for moderate sizes.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        print(f"[{time.time():.1f}] LP failed: {res.message}")
        return None

# Simple simulated‑annealing hill‑climb ------------------------------------

def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
    init_temp_factor: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """
    Very light SA on the simplex.
    Moves: swap two entries or transfer a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the global time budget and returns a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length that balances expressiveness and LP tractability.
    # 2048 works well for the LP‑refinement step.
    # ------------------------------------------------------------------
    n = 2048 if budget_s >= 800 else 1024
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. fallback on a known construction if present
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        # maybe the known vector has different length – we truncate/pad
        if len(fb_vec) != n:
            # resample by simple interpolation / truncation
            fb_arr = np.array(fb_vec, dtype=np.float64)
            if len(fb_arr) > n:
                fb_arr = fb_arr[:n]
            else:
                # pad with zeros and renormalise
                fb_arr = np.pad(fb_arr, (0, n - len(fb_arr)), constant_values=0.0)
            fb_arr = project_to_simplex(fb_arr)
        else:
            fb_arr = project_to_simplex(np.array(fb_vec, dtype=np.float64))
        sc = compute_score(fb_arr)
        print(f"[{time.time():.1f}] fallback score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = fb_arr.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. quick scan of analytic families + random draws
    # ------------------------------------------------------------------

    # Search some parameters for symmetric product family
    for alpha in np.linspace(0.0, 5.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

    # Left‑heavy power law
    for beta in np.linspace(0.0, 5.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (lh‑β={beta:.2f}) = {best_score:.6f}")

    # Exponential decay
    for gamma in np.linspace(0.0, 6.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    # Random monotone starts
    for _ in range(80):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth surrogate optimisation (Adam) – using the current best as warm‑start
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        start_a = np.array(best_seq, dtype=np.float64)
        print(f"[{time.time():.1f}] starting Adam‑smooth optimisation")
        a_opt, sc_opt = adam_smooth_optimize(
            start_a,
            deadline - 200.0,
            max_iters=120_000,
            beta_start=5.0,
            beta_end=3000.0,
            lr=0.015,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. LP‑based refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE and best_seq is not None and time.time() < deadline - 100.0:
        f = np.array(best_seq, dtype=np.float64)
        print(f"[{time.time():.1f}] entering LP‑refinement loop")
        iteration = 0
        while time.time() < deadline - 20.0:
            iteration += 1
            M = convolution_fft(f, f).max()
            g_raw = lp_max_sum_under_M(f, M)
            if g_raw is None:
                break
            # normalise g to the simplex
            g = g_raw / g_raw.sum()
            # try a few blending factors and keep the best improvement
            improved = False
            for t in (0.15, 0.25, 0.35, 0.45):
                f_new = (1.0 - t) * f + t * g
                f_new = f_new / f_new.sum()
                sc_new = compute_score(f_new)
                if sc_new < best_score - 1e-8:
                    best_score = sc_new
                    best_seq = f_new.tolist()
                    f = f_new
                    improved = True
                    print(f"[{time.time():.1f}] LP iter {iteration}, blend t={t:.2f} → {best_score:.6f}")
                    break   # accept the first improvement
            if not improved:
                # if none of the tried blends improved, exit the loop
                print(f"[{time.time():.1f}] LP iteration {iteration} gave no improvement – stop")
                break

    # ------------------------------------------------------------------
    # 4. Symmetrisation (many optima are palindromic)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)

"""
Search for a non‑negative step‑function (height sequence) that minimises
`evaluate_sequence`.  The algorithm consists of several stages, each respecting
the global wall‑clock budget (default 1000 s).

1. **Initialisation** – generate a variety of analytic families (symmetric
   product, left‑heavy power‑law, exponential decay) and random monotone or
   sparse vectors.  Keep the best.

2. **Smooth surrogate optimisation** – minimise the log‑sum‑exp surrogate of
   the max‑convolution using Adam on the soft‑max parametrisation.  The
   smoothing parameter β is annealed from 5 up to 3000.

3. **LP‑refinement** – given the current vector `f` with maximal convolution
   `M = max(f*f)`, solve the linear programme  

        maximise   Σ g_j
        subject to A·g ≤ M·1,   g ≥ 0

   where `A` is the convolution matrix of `f`.  The optimiser `g` is
   normalised to a probability simplex and blended with `f` (`f←(1‑t)f+t g`,
   `t∈{0.15,0.25,0.35,0.45}`).  The loop stops when no improvement is found
   or the time budget expires.

4. **Symmetrisation** – averaging the vector with its reverse often reduces the
   peak of the autocorrelation.

5. **Final hill‑climb** – a tiny simulated‑annealing walk on the simplex that
   performs swaps or small mass transfers.

All intermediate scores and timestamps are printed.  The routine returns a
plain Python list of floats; `evaluate_sequence` (provided by the harness)
will be used for the final official score.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# -------------------------------------------------------------
# Optional SciPy imports (required for the LP‑refinement)
# -------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    SCIPY_AVAILABLE = False

# -------------------------------------------------------------
# Basic utilities
# -------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = rfft(a, n=nfft)
    B = rfft(b, n=nfft)
    C = A * B
    conv = irfft(C, n=nfft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective when sum(a) == 1: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()

# -------------------------------------------------------------
# Analytic families
# -------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a

def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β), monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a

def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a

# -------------------------------------------------------------
# Random initialisers
# -------------------------------------------------------------
def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)

def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector obtained from powered uniforms."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]               # decreasing
    a = a / a.sum()
    return a

# -------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp surrogate) via Adam
# -------------------------------------------------------------
def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 120_000,
    beta_start: float = 5.0,
    beta_end: float = 3000.0,
    lr: float = 0.015,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution."""
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential annealing)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothed max: log‑sum‑exp over the convolution entries
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # soft‑max weights over k

        # gradient of the surrogate w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam iter {it} – new best {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam finished – final best {best_score:.6f}")

    return best_a, best_score

# -------------------------------------------------------------
# Linear‑program (LP) refinement
# -------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n sparse matrix A with
        A[k, i] = f_i   if 0 ≤ k‑i < n   else 0 .
    Then (A @ g)[k] = (f * g)[k].
    """
    n = f.shape[0]
    i = np.arange(n, dtype=np.int64)[:, None]   # shape (n,1)
    j = np.arange(n, dtype=np.int64)[None, :]   # shape (1,n)
    rows = (i + j).ravel()
    cols = np.tile(j.ravel(), n)
    data = np.repeat(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def lp_max_sum_under_M(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   s.t.   A·g ≤ M·1,   g ≥ 0,
    where A is the convolution matrix of `f`.  Returns the optimiser g
    (unnormalised) or None if the solver fails.
    """
    if not SCIPY_AVAILABLE:               # pragma: no cover
        return None
    A = build_conv_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)   # minimise – Σg  →  maximise Σg
    bounds = [(0, None)] * f.shape[0]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        print(f"[{time.time():.1f}] LP failed: {res.message}")
        return None

# -------------------------------------------------------------
# Simulated‑annealing hill‑climb (final fine‑tuning)
# -------------------------------------------------------------
def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
    init_temp_factor: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """Simple SA on the simplex – swaps or small mass transfers."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                      # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                           # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
