"""
Improved conditional‑gradient search for the step‑function problem.

The objective is   2·n·max(conv(a,a)) / (sum(a)^2) .
Because the scaling cancels, we may always work with a probability vector
(sum(a)=1) and minimise   f(a)=2·n·max(conv(a,a)).

The algorithm consists of three stages:

1. **Seeding** – generate a handful of sensible probability vectors
   (deterministic sine‑squared, β‑polynomials and random Dirichlet draws)
   and keep the best one w.r.t. the exact objective.

2. **Conditional‑gradient (Frank‑Wolfe) iteration via a linear programme**.
   For the current vector *a* we compute M = max(conv(a,a)).
   The sub‑problem
        maximise    Σ b_i
        subject to  (a * b)[k] ≤ M   for all k,
                    b_i ≥ 0,
   is a linear programme because the convolution with the *fixed* vector *a*
   is linear in *b*.  Its optimum pushes mass onto indices where *a* is tiny,
   thereby offering a direction that can reduce the maximal autocorrelation.
   We obtain a feasible direction g = b / Σb, mix it with the current point
   (convex combination) and accept the mixture if it lowers the objective.
   A tiny safety‑epsilon guarantees that no component becomes exactly zero,
   avoiding an unbounded LP.

3. **Polishing** – a short stochastic hill‑climb that only accepts
   improving moves (swap or a small transfer of mass).  This cleans up any
   remaining irregularities before the final evaluation.

All work stops long enough before the provided deadline (the harness
adds a 1 s safety margin) so that a valid list of non‑negative floats is
always returned.  The implementation uses only NumPy and SciPy (HiGHS
LP solver) and respects the one‑CPU restriction.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_self(a: np.ndarray) -> float:
    """Return max(conv(a,a)) for a probability vector a."""
    return float(convolution_fft(a, a).max())


def objective(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv(a,a)) (sum(a)=1)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution_self(a)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sine‑squared shape a_i = sin^2(pi*(i+1)/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ ((i+1)*(n‑i))**p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Random probability vector drawn from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Conditional‑gradient step (LP sub‑problem)
# ----------------------------------------------------------------------


def solve_lp_subproblem(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ b_i   s.t. (a * b)[k] ≤ M for all k,  b_i ≥ 0.
    Returns the optimal b (may be None if the LP fails).
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the dense constraint matrix A (rows × n).
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a[i]          # a_i contributes to rows i … i+n‑1

    # Right‑hand side: every inequality bound is the current maximal value M.
    b_ub = M * np.ones(rows, dtype=float)

    # Linear objective: maximise Σ b_i  → minimise -Σ b_i.
    c = -np.ones(n, dtype=float)

    # Bounds for each variable: b_i ≥ 0.
    bounds = [(0.0, None)] * n

    # Solve using HiGHS (the default for recent SciPy versions).
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )

    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Simple hill‑climb polishing (greedy improvements only)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, deadline: float, max_steps: int = 200_000) -> np.ndarray:
    """
    Stochastic greedy local search that only keeps strictly improving moves.
    It conserves the total mass, therefore a always stays a probability vector.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = objective(best_a)

    step_scale = 0.08
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # small transfer from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # re‑normalise (numerical drift)
        cand /= cand.sum()
        cur_score = objective(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                       # become more conservative

        steps += 1
    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------

    print(f"[INFO] Using sequence length n = {n}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    seeds = []
