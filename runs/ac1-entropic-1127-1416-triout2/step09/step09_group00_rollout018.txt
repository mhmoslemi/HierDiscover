"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint (∑a = 1).  The implementation

* scans a few problem sizes (1024 … 8192) with many analytically‑motivated
  families (symmetric product, power‑law, exponential “U”, Gaussian,
  skewed exponential, mixtures with the uniform vector) and a handful of
  random Dirichlet draws,
* keeps the best vector seen so far,
* refines it with three complementary optimisation stages:
    – a linear‑programming “active‑set” step (the reference paper’s f←(1‑t)f+t g)
      that maximises the total mass of a vector g while keeping the
      convolution f∗g below the current peak,
    – a smoothed‑max Adam optimiser on the soft‑max parametrisation,
    – a cheap projected sub‑gradient descent on the true max‑objective,
* finally symmetrises the vector, runs a short simulated‑annealing hill‑climb
  and returns the best sequence found before the wall‑clock deadline.
All heavy work respects the 1000 s budget and never returns an empty list,
ensuring a finite score for the harness.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Convolution utilities (linear, aperiodic)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full 1‑D convolution via real FFT (no circular wrap‑around)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by several helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred at the middle, σ is a fraction of n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = w / w.sum()
    return a


def skewed_exponential_family(n: int, gamma_left: float, gamma_right: float) -> np.ndarray:
    """Asymmetric exponential U shape."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    dist = np.where(i <= centre,
                    i,                     # distance from left edge
                    n - 1 - i)             # distance from right edge
    w = np.where(i <= centre,
                 np.exp(-gamma_left * dist),
                 np.exp(-gamma_right * dist))
    a = w / w.sum()
    return a


def random_dirichlet_start(n: int) -> np.ndarray:
    """Draw from Dirichlet with random concentration α∈[0.05,2.0]."""
    alpha = np.random.uniform(0.05, 2.0)
    a = np.random.dirichlet([alpha] * n)
    return a

# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform prob. vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector via power‑transformed uniforms."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                # decreasing
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 400.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)           # initialise in log‑space

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        conv = convolution_fft(a, a)          # length 2n‑1
        c_max = conv.max()
        # stable soft‑max over the convolution entries
        exp_w = np.exp(beta * (conv - c_max))
        w = exp_w / exp_w.sum()               # smooth max‑weights

        # gradient of smoothed max w.r.t. a (see derivation in the paper)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad)
        grad_z = a * grad - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (very cheap per iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient method for 2·n·max(a*a) on the simplex,
    using a diminishing stepsize ∝ 1/√t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny temperature schedule)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex.
      – with prob 0.5 swap two entries,
      – otherwise transfer a fraction of mass from one index to another.
    Temperature is lowered linearly; uphill moves are occasionally accepted.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4                # tiny initial temperature
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:        # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                             # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)    # linear cooling

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “f←(1‑t)f+t g” step)
# ----------------------------------------------------------------------


def lp_refine(start_a: np.ndarray,
              deadline: float,
              max_iter: int = 8,
              t0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based active‑set refinements.
    For the current vector f we solve
        maximise   Σ_j g_j
        s.t.       g ≥ 0 ,  (f * g)_k ≤ M for all k,
    where M = max(f*f).  The solution g is then convex‑combined
    with f (step size t that decays with the iteration number) and
    re‑normalised to the simplex.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        n = a.shape[0]
        conv = convolution_fft(a, a)
        M = conv.max()

        # Build the constraint matrix (2n‑1) × n  :  (f * g) ≤ M
        rows = 2 * n - 1
        A = sp.lil_matrix((rows, n), dtype=np.float64)

        # each column j of g appears in rows i+j with weight a_i
        for i in range(n):
            ai = a[i]
            if ai == 0.0:
                continue
            start = i
            A[start:start + n, :] += ai * sp.eye(n, dtype=np.float64)

        A = A.tocsr()
        c = -np.ones(n)                         # minimise -Σ g  → maximise Σ g
        b_ub = M * np.ones(rows)
        bounds = [(0, None)] * n

        # Solve the LP
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs',
                                    options={"presolve": True})
        if not res.success:
            print(f"[{time.time():.1f}] LP failed at iter {it}: {res.message}")
            break

        g = res.x
        # convex combination with a decreasing step size
        t = t0 / (it + 0.0)          # e.g. 0.2, 0.1, 0.067, …
        a = (1.0 - t) * a + t * g
        a = a / a.sum()              # re‑normalise to the simplex

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP refine improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. fallback (if a previously known construction is available)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Multi‑size analytic + stochastic exploration
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]           # a spread of sizes
    for n in lengths:
        if time.time() >= deadline - 120.0:
            break
        print(f"[{time.time():.1f}] exploring length n = {n}")

        # ----- symmetric product family (α) -----
        for alpha in np.linspace(0.0, 12.0, 31):
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        # ----- left‑heavy power law (β) -----
        for beta in np.linspace(0.0, 12.0, 31):
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

        # ----- exponential U‑shape (γ) -----
        for gamma in np.linspace(0.0, 6.0, 19):
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

        # ----- Gaussian family (σ) -----
        for sigma_frac in (0.05, 0.10, 0.15, 0.20, 0.30):
            if time.time() >= deadline - 30.0:
                break
            sigma = sigma_frac * n
            vec = gaussian_family(n, sigma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (gauss‑σ={sigma_frac:.2f}) = {best_score:.6f}")

        # ----- Skewed exponential families (γL,γR) -----
        for gamma_left in np.linspace(0.0, 4.0, 9):
            for gamma_right in np.linspace(0.0, 4.0, 9):
                if time.time() >= deadline - 30.0:
                    break
                vec = skewed_exponential_family(n, gamma_left, gamma_right)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (skew γL={gamma_left:.2f},γR={gamma_right:.2f}) = {best_score:.6f}")

        # ----- mixtures of uniform and U‑shape -----
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for gamma in np.linspace(0.0, 6.0, 13):
                if time.time() >= deadline - 30.0:
                    break
                vec = mixed_uniform_u_family(n, lam, gamma)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

        # ----- random Dirichlet draws -----
        for _ in range(60):
            if time.time() >= deadline - 30.0:
                break
            vec = random_dirichlet_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑Dirichlet) = {best_score:.6f}")

        # ----- random monotone draws -----
        for _ in range(40):
            if time.time() >= deadline - 30.0:
                break
            vec = random_monotone_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

        # ----- random U‑shape draws -----
        for _ in range(40):
            if time.time() >= deadline - 30.0:
                break
            vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑U) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑based active‑set refinement (few heavy but very potent steps)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting LP refinement")
        a_lp, sc_lp = lp_refine(np.array(best_seq, dtype=np.float64),
                                deadline - 70.0,
                                max_iter=8,
                                t0=0.18)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.tolist()
            print(f"[{time.time():.1f}] after LP refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑max Adam optimisation (intensive local search)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] starting Adam‑smooth optimisation")
        a_opt, sc_opt = adam_smooth_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 50.0,
            max_iters=250_000,
            beta_start=4.0,
            beta_end=2000.0,        # push closer to the true max
            lr=0.015,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam‑smooth best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient fine‑tuning
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=800_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation (many optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final hill‑climb (short SA) – uses remaining time only
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard: fall back to a modest uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final verification (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
