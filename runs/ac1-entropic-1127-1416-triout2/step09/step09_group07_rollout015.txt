def hill_climb_sa(start_a, deadline, start_temp=0.01, end_temp=1e-6, inc_factor=1.05, decay_factor=0.95, recompute_interval=50000):
    a = start_a.copy()
    n = a.shape[0]
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    steps = 0
    step_scale = 0.1
    accept_cnt = 0
    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total
        temp = start_temp * (1 - progress) + end_temp * progress
        # find peak position
        k_max = int(conv.argmax())
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_range = np.arange(i_start, i_end + 1)
        # contributions for each i in range
        contrib = a[idx_range] * a[k_max - idx_range]
        if contrib.sum() > 0:
            src = np.random.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = np.random.randint(n)
        if a[src] <= 0:
            continue
        # choose dst weighted towards low mass
        # compute weights inversely related to a
        inv_weights = 1.0 / (a + 1e-12)
        inv_weights[src] = 0.0
        dst = np.random.choice(n, p=inv_weights / inv_weights.sum())
        # small transfer
        delta = step_scale * a[src]
        # build candidate conv
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic correction
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta
        cand_max = conv_cand.max()
        cand_score = 2.0 * n * float(cand_max)
        # acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(temp,1e-12)):
            # accept
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_score = cand_score
            accept_cnt += 1
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                # optional increase step_scale
                step_scale = min(step_scale * inc_factor, 0.5)
        else:
            # reject
            step_scale *= decay_factor
        steps += 1
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())
    return best_a, best_score

"""
Algorithm Overview
------------------
We attack the convex problem

    min   2·n·max_i (a ⊛ a)_i
    s.t.  a_i ≥ 0,  Σ a_i = 1

by a four‑stage pipeline that respects the 1000 s wall‑clock budget.

1. **Analytic seed** – scan a one‑parameter family
   a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on a modest α‑grid.
2. **Random seed** – draw a Dirichlet (exponential) vector and normalise it.
3. **Guided Simulated‑Annealing hill‑climb** – for each seed we run an
   SA optimiser that repeatedly transfers a fraction of the mass from the
   index that contributes most to the current maximal convolution entry to a
   random destination.  The move is accepted if it improves the objective
   or with probability exp(−Δ/Temp) where the temperature is linearly
   cooled.  Convolution updates are performed incrementally; a full FFT
   recomputation is done every few‑thousand steps to eliminate drift.
4. **Return the best sequence** that was seen by any stage.

All helper functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution using real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector that already satisfies
    Σ a = 1.  This coincides with the value returned by evaluate_sequence.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytic start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the probability simplex."""
    i = np.arange(n, dtype=float)
    # log‑product to avoid overflow for large n
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a modest α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([
        np.linspace(-1.0, 0.0, 11),      # centre‑heavy
        np.linspace(0.0, 2.5, 26)        # edge‑heavy
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Random Dirichlet seed
# ----------------------------------------------------------------------


def random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (i.i.d. exponential, then normalise)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# 3. Guided Simulated‑Annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  start_temp: float = 0.02,
                  end_temp: float = 1e-6,
                  inc_factor: float = 1.07,
                  decay_factor: float = 0.97,
                  recompute_interval: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing.
    At each step we transfer a fraction of the mass from the index that
    contributes most to the current maximal convolution entry to a random
    destination.  The move is accepted if it improves the score or with a
    probability exp(−Δ/Temp).  Convolution is updated incrementally.
    """
    a = start_a.copy()
    n = a.size

    # exact convolution and initial score
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * float(conv.max())

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    steps = 0
    step_scale = 0.12  # fraction of source mass to move each step

    while time.time() < deadline:
        # -------------------------------------------------
        # temperature (linear cooling)
        # -------------------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total
        temp = start_temp * (1.0 - progress) + end_temp * progress

        # -------------------------------------------------
        # locate the current maximal convolution index
        # -------------------------------------------------
        k_max = int(conv.argmax())
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_range = np.arange(i_start, i_end + 1)

        # contributions a_i * a_{k_max-i}
        j = k_max - idx_range
        contrib = a[idx_range] * a[j]

        # -------------------------------------------------
        # choose source – the index with the largest contribution
        # -------------------------------------------------
        if contrib.size > 0 and contrib.max() > 0:
            src = idx_range[np.argmax(contrib)]
        else:
            src = np.random.randint(n)

        if a[src] <= 0.0:
            # nothing to move – pick a fresh random source
            src = np.random.randint(n)
            if a[src] <= 0.0:
                continue

        # -------------------------------------------------
        # choose destination – uniform random different index
        # -------------------------------------------------
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1  # guarantee dst != src

        # -------------------------------------------------
        # amount to transfer (always ≤ a[src])
        # -------------------------------------------------
        delta = step_scale * a[src]

        # -------------------------------------------------
        # incremental convolution update (exact)
        # -------------------------------------------------
        conv_cand = conv.copy()
        # linear part: –2·δ·shift_src + 2·δ·shift_dst
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic correction
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * float(cand_max)

        # -------------------------------------------------
        # acceptance test (SA)
        # -------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # uphill move – accept with probability exp(−Δ/Temp)
            if temp > 0.0:
                prob = np.exp(-(cand_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # apply move
            a[src] -= delta
            a[dst] += delta
            # clamp tiny negatives that may appear because of round‑off
            a[src] = max(a[src], 0.0)
            conv = conv_cand
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                # increase step size slowly when we improve
                step_scale = min(step_scale * inc_factor, 0.5)
        else:
            # reject – shrink step size
            step_scale *= decay_factor

        steps += 1

        # occasional full recomputation to wipe accumulated numerical error
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())

    return best_a, best_score
