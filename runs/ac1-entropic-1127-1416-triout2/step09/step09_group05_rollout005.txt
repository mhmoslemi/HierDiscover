    sum_a = a.sum()
    conv = convolution_fft(a, a)
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a**2)

Now possible pitfalls:

- During exponentiated gradient, if exponentials overflow: compute exp(beta*(conv - maxc)) where conv-maxc <=0, exponent of negative values leads to values ≤1, safe even for large beta (they become super small or zero). However underflow may produce zero entries causing sum zero, dividing by zero. They performed:

```
expw = np.exp(beta * (conv - maxc))
w = expw / expw.sum()
```

If expw all zero due to underflow, sum may be zero => error. With extreme beta large, differences may be large causing underflow to 0. However conv - maxc <=0; for large beta, exp(-large) underflows to 0 leaving only the max entry (where conv - maxc = 0) gives exp(0)=1, others ~0; sum=1; okay.

If numerically everything underflows except max, still fine.

"""
Propose a non‑negative step–height sequence that (approximately) minimizes
evaluate_sequence.  The algorithm proceeds in several stages:

0.  Choose a workable length (larger lengths give a better asymptotic bound).
1.  Scan a handful of simple parametric families (exponential, power‑law,
    Gaussian, cosine) and a few random Dirichlet draws, keeping the best.
2.  Run several independent exponentiated‑gradient refinements on the
    smoothed log‑sum‑exp approximation of the max‑convolution.  The schedule
    for the smoothing parameter β is exponential (β_start → β_end) and the
    KL‑mirror step keeps the vector on the probability simplex.
3.  Polish the best candidate with a greedy stochastic hill‑climb that moves
    a small packet of mass between two positions (or swaps two entries) and
    accepts only improving moves.  The step‑size is adapted during the walk.
4.  Finally try a few random circular shifts of the best vector (zero‑padding)
    and keep the best shifted version.

All stages respect the 1000 s wall‑clock budget; frequent time‑checks abort
early if the deadline is near.  The returned object is a plain Python list of
floats, ready for the harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (real‑valued inputs)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def objective(a: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` for a vector `a` (the sum is NOT
    assumed to be 1 – we compute the correct scaling).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = convolution_fft(a, a)
    maxb = conv.max()
    return 2.0 * n * maxb / (s * s)


def fast_objective(a: np.ndarray) -> float:
    """
    Fast evaluation when `a` is already normalised to sum = 1.
    In this case the denominator is 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * conv.max()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (lam≥0)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (p≥0)."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Truncated Gaussian centred in the middle of the vector."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def cosine_shape(n: int, power: float) -> np.ndarray:
    """Raised‑cosine window: w_i = sin(π·(i+1)/(n+1))^power."""
    i = np.arange(n, dtype=float)
    w = np.sin(np.pi * (i + 1) / (n + 1))
    if power != 1.0:
        w = w ** power
    return w / w.sum()


# ----------------------------------------------------------------------
# Simple systematic scan of families
# ----------------------------------------------------------------------


def scan_families(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """Return the best vector found among a few hand‑crafted families."""
    best_vec = None
    best_sc = float("inf")

    # uniform baseline
    u = np.full(n, 1.0 / n)
    sc = fast_objective(u)
    if sc < best_sc:
        best_sc, best_vec = sc, u.copy()

    # exponential
    lam_grid = np.concatenate(
        (np.array([0.0]), np.logspace(-5, -1.5, 250))  # λ from 0 to ~0.03
    )
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = fast_objective(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # power‑law
    p_grid = np.linspace(0.0, 8.0, 250)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = fast_objective(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # Gaussian windows
    sigma_grid = np.linspace(0.1 * n, 0.5 * n, 200)
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        a = gaussian_shape(n, sigma)
        sc = fast_objective(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # Raised‑cosine windows
    power_grid = np.linspace(0.2, 6.0, 200)
    for p in power_grid:
        if time.time() >= deadline:
            break
        a = cosine_shape(n, p)
        sc = fast_objective(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # a few random Dirichlet draws (different concentration)
    for _ in range(15):
        if time.time() >= deadline:
            break
        alpha = np.random.choice([0.2, 0.5, 1.0, 2.0])
        a = np.random.dirichlet(np.full(n, alpha))
        sc = fast_objective(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Exponentiated‑gradient (KL‑mirror) optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 600_000,
    beta_start: float = 0.5,
    beta_end: float = 5000.0,
    eta0: float = 0.6,
    eps: float = 1e-15,
) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp approximation of max (a*a) using a KL‑mirror
    (exponentiated‑gradient) update.  The smoothing parameter β is increased
    exponentially from `beta_start` to `beta_end`.  The step size follows
    η₀/√t.  Returns the best vector (and its exact objective) seen so far.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = fast_objective(best_a)

    for it in range(1, max_iters + 1):
        # check time every 1000 iterations (cheap)
        if (it & 1023) == 0 and time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weights (log‑sum‑exp of the convolution)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stabilise the exponentials
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smooth objective
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # step size
        eta = eta0 / math.sqrt(it)

        # KL‑mirror update and renormalisation
        a = np.maximum(a * np.exp(-eta * grad), eps)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # occasional exact evaluation
        if it % 500 == 0:
            sc = fast_objective(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final exact check
    sc = fast_objective(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb (greedy)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 3_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy random walk:
    - with 50 % probability swap two entries,
    - otherwise transfer a fraction of mass from one index to another.
    Only improvements are kept; the move size is adapted automatically.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = fast_objective(best_a)

    step_scale = 0.12
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        sc = fast_objective(cand)

        if sc < best_sc:
            best_sc, best_a = sc, cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1
    return best_a, best_sc


# ----------------------------------------------------------------------
# Small utility: evaluate a shifted version (zero‑padded)
# ----------------------------------------------------------------------


def shift_vector(a: np.ndarray, k: int, direction: str = "right") -> np.ndarray:
    """Shift `a` by `k` positions, inserting zeros at the vacated slots."""
    n = a.shape[0]
    if k == 0:
        return a.copy()
    if direction == "right":
        return np.concatenate([np.zeros(k), a[: n - k]])
    else:  # left
        return np.concatenate([a[k:], np.zeros(k)])
