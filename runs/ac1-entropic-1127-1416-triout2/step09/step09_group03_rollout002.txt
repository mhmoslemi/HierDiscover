"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1. **Quick parametric sweep** – generate many simple families
   (central‑binomial, power‑law, shifted power‑law, exponential,
   power‑law × exponential, sin‑shapes).  All vectors are projected onto the
   probability simplex (∑a_i = 1) and the exact objective
   `2·n·max(conv(a,a))` is evaluated.  The best among them is kept as a
   starting point.

2. **Smoothed Adam optimisation** – minimise a smooth surrogate of the
   max‑convolution:
        L_β(a) = (1/β)·log∑_k exp(β·(a*a)_k) .
   The gradient of L_β is obtained analytically via a soft‑max weighting.
   An Adam optimiser with an annealing schedule β∈[5,200] runs until the
   allocated time runs out, always projecting back onto the simplex.
   The best iterate seen during the run is retained.

3. **Projected sub‑gradient polishing** – switch to the true nonsmooth
   objective and perform a long sub‑gradient descent with a 1/√t step‑size.

4. **Final stochastic hill‑climb** – cheap mass‑transfer moves (swap or
   fractional shift) with simplex projection finish the search.

All stages respect the global wall‑clock deadline (default 1000 s) and
periodically report the current best bound.  The function returns a plain
list of non‑negative floats; the harness will call ``evaluate_sequence`` on
it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (∑a = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def smooth_adam_optimize(start_a: np.ndarray,
                         time_deadline: float,
                         beta_start: float = 5.0,
                         beta_end: float = 200.0,
                         step0: float = 0.9,
                         adam_beta1: float = 0.9,
                         adam_beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the log‑sum‑exp surrogate of the max‑convolution.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    start = time.time()
    total = max(time_deadline - start, 1e-6)

    it = 0
    while time.time() < time_deadline:
        it += 1

        # anneal smoothing parameter β
        t_frac = (time.time() - start) / total
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # full self‑convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (log‑sum‑exp trick)
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient = 2 * (a_rev ∗ w) restricted to n entries
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional progress check
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Smooth] it={it:,d}  score={best_score:.6f}")

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


def subgradient_optimize(init_a: np.ndarray,
                         time_deadline: float,
                         step0: float = 0.06,
                         max_iter: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (k‑th entry of the convolution)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()
            print(f"[Subgrad] it={it:,d}  score={best_score:.6f}")

        a = a_new

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


def hill_climb(seq: np.ndarray,
               time_deadline: float,
               max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic moves (swap or fractional mass transfer) with simplex
    projection.  Returns the best sequence encountered.
    """
    a = seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    step = 0
    while step < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # fractional transfer i -> j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        step += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Candidate generation (parametric families)
# ----------------------------------------------------------------------


def generate_candidate_pool(n: int) -> list[np.ndarray]:
    """Return a list of plausible initialization vectors (already normalized)."""
    candidates = []

    # 0. Central binomial coefficients (the classic starter)
    a = np.ones(n, dtype=float)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    candidates.append(a)

    # 1. Pure power‑law a_i ∝ (i+1)^(-p)
    for p in np.linspace(0.2, 1.0, 17):  # 0.2 … 1.0 step 0.05
        v = np.arange(1, n + 1, dtype=float) ** (-p)
        v /= v.sum()
        candidates.append(v)

    # 2. Shifted power‑law a_i ∝ (i+shift)^(-p)
    for p in (0.45, 0.5, 0.55):
        for shift in (0.1, 0.5, 1.0, 2.0):
            v = (np.arange(shift, shift + n, dtype=float)) ** (-p)
            v /= v.sum()
            candidates.append(v)

    # 3. Simple exponential decay a_i ∝ exp(-λ i)
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005):
        v = np.exp(-lam * np.arange(n))
        v /= v.sum()
        candidates.append(v)

    # 4. Product of power‑law and exponential
    for p in (0.4, 0.5, 0.6):
        for lam in (0.0005, 0.001, 0.002):
            v = (np.arange(1, n + 1, dtype=float) ** (-p)) * np.exp(-lam * np.arange(n))
            v /= v.sum()
            candidates.append(v)

    # 5. Sine‑shaped families a_i ∝ sin(π (i+0.5)/(2n))^β
    x = np.linspace(0.0, 1.0, n, endpoint=False) + 0.5 / n
    for beta in (0.5, 1.0, 2.0, 3.0):
        v = np.sin(np.pi * x / 2) ** beta
        v /= v.sum()
        candidates.append(v)

    return candidates


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float("inf")
    n = 2048

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"[Fallback] score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Parametric sweep (very cheap)
    # ------------------------------------------------------------------
    pool = generate_candidate_pool(n)
    print(f"[Sweep] evaluating {len(pool)} candidate families...")
    for idx, cand in enumerate(pool, 1):
        score = compute_score(cand)
        if score < best_val:
            best_val = score
            best_seq = cand.tolist()
            print(f"[Sweep] new best (#{idx})  score = {best_val:.6f}")

        # early‑exit if we run out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2. Smoothed Adam optimisation (major workhorse)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        a0 = np.array(best_seq, dtype=float)
        smooth_deadline = deadline - 100.0
        print("[Smooth] starting Adam optimisation...")
        a_smooth, smooth_score = smooth_adam_optimize(
            a0,
            time_deadline=smooth_deadline,
            beta_start=5.0,
            beta_end=250.0,
            step0=0.9,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"[Smooth] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 50.0:
        a0 = np.array(best_seq, dtype=float)
        subgrad_deadline = deadline - 30.0
        print("[Subgrad] polishing with sub‑gradient descent...")
        a_sub, sub_score = subgradient_optimize(
            a0,
            time_deadline=subgrad_deadline,
            step0=0.06,
            max_iter=800_000,
        )
        if sub_score < best_val:
            best_val = sub_score
            best_seq = a_sub.tolist()
            print(f"[Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a0 = np.array(best_seq, dtype=float)
        hill_deadline = deadline - 2.0
        print("[Hill] final stochastic hill‑climb...")
        a_hill, hill_score = hill_climb(
            a0,
            time_deadline=hill_deadline,
            max_steps=600_000,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = a_hill.tolist()
            print(f"[Hill] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Safety net – tiny uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final verification (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[Done] Final reported score = {final_score:.6f}")
    return best_seq
