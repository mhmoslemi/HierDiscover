"""
Propose a step‑function sequence that minimises the given evaluation objective.

Approach
--------
1. Work first on a modest size (n_small = 2048) where many
   inexpensive iterations are possible.
   * Scan a few analytic families (uniform, Hanning, Blackman,
     symmetric product, power‑law, exponential, triangular).
   * Random monotone / sparse starts.
   * Refine the best candidates with Adam on the log‑sum‑exp
     smooth approximation of the max‑convolution.
2. Upsample the best small‑size vector to a larger size (n_large,
   up to 16384) by linear interpolation and renormalise.
3. On the large vector run:
   * Mirror (entropy) sub‑gradient descent on the exact max‑objective.
   * A second, longer Adam smoothing phase with a very large
     smoothing parameter (β up to 1e4) to approach the true max.
   * Symmetrisation (averaging with the reversed vector) and projection.
   * A final stochastic hill‑climb with a tiny simulated‑annealing
     schedule.
All stages respect the global time budget (≈ 1000 s).  The routine
prints intermediate scores for monitoring and finally returns a plain
Python list of floats that can be fed to `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # optional, not used in the final pipeline

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (real inputs)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score_fast(a: np.ndarray) -> float:
    """
    Fast exact objective for a probability vector (sum(a) = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_conv = float(conv.max())
    return 2.0 * n * max_conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    """Uniform distribution."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^(‑α) (edge‑heavy for α>0,
    centre‑heavy for α<0).  Symmetric, normalised to sum=1.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # numerical safety
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def hanning_family(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return w / w.sum()


def blackman_family(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return w / w.sum()


def triangular_family(n: int) -> np.ndarray:
    """Symmetric triangle (peak in the middle, zero at ends)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = 1.0 - np.abs(i - mid) / mid
    w[w < 0] = 0.0
    return w / w.sum()


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]  # decreasing
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) via Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 120_000,
    beta_start: float = 5.0,
    beta_end: float = 5_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β grows exponentially from ``beta_start`` to
    ``beta_end``.  Returns the best vector found (as a probability vector)
    and its exact objective value.
    """
    eps = 1e-12
    # initialise log‑softmax parameters
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score_fast(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- schedule for β (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- current a (soft‑max) ----
        a = softmax(z)

        # ---- smoothed weights over convolution entries ----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # softmax over conv entries

        # ---- gradient of the smoothed surrogate w.r.t. a ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ---- back‑prop through soft‑max ----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional exact evaluation ----
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score_fast(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score_fast(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for the true
    objective 2·n·max(a*a).  The step size is η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score_fast(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # renormalise to the simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score_fast(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score_fast(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with a tiny simulated‑annealing component
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_000_000,
    init_temp_factor: float = 5e-4,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
    – with prob 0.5 swap two entries,
    – otherwise move a fraction of mass from one index to another.
    Very low temperature allows occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score_fast(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – start tiny, linearly cool
    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise (simple projection onto simplex)
        cand = cand / cand.sum()

        cand_score = compute_score_fast(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)  # enlarge moves slowly
        else:
            step_scale *= 0.97  # shrink after rejection

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the global time budget and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Lengths
    # ------------------------------------------------------------------
    # Small size for fast exploration, large size for final refinement.
    n_small = 2048
    n_large = 16384 if budget_s >= 900 else 8192
    print(f"[{time.time():.1f}] n_small = {n_small}, n_large = {n_large}")

    # ------------------------------------------------------------------
    # 1. Initial cheap scans on the small size
    # ------------------------------------------------------------------
    best_vec_small = None
    best_score_small = math.inf

    # ----- analytic families -------------------------------------------------
    families = []

    # uniform (baseline)
    families.append(("uniform", lambda n, _: uniform_family(n)))

    # Hanning window
    families.append(("hanning", lambda n, _: hanning_family(n)))

    # Blackman window
    families.append(("blackman", lambda n, _: blackman_family(n)))

    # Triangular
    families.append(("triangular", lambda n, _: triangular_family(n)))

    # Symmetric product family (scan α ∈ [-2, 5])
    alphas = np.linspace(-2.0, 5.0, 71)
    for a in alphas:
        families.append(("sym_prod", lambda n, α=a: symmetric_product_family(n, α)))

    # Left‑heavy power‑law (β ∈ [0,5])
    betas = np.linspace(0.0, 5.0, 71)
    for b in betas:
        families.append(("left_power", lambda n, β=b: left_heavy_power_family(n, β)))

    # Exponential decay (γ ∈ [0,6])
    gammas = np.linspace(0.0, 6.0, 61)
    for g in gammas:
        families.append(("exp_decay", lambda n, γ=g: exponential_decay_family(n, γ)))

    # evaluate families (respect early exit if time runs out)
    for name, maker in families:
        if time.time() >= deadline - 30.0:
            break
        vec = maker(n_small, None)
        sc = compute_score_fast(vec)
        if sc < best_score_small:
            best_score_small = sc
            best_vec_small = vec.copy()
            print(f"[{time.time():.1f}] new best {name} score = {best_score_small:.6f}")

    # ----- random starts ----------------------------------------------------
    for _ in range(150):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n_small)
        sc = compute_score_fast(vec)
        if sc < best_score_small:
            best_score_small = sc
            best_vec_small = vec.copy()
            print(f"[{time.time():.1f}] new best (rand‑mono) score = {best_score_small:.6f}")

    # ------------------------------------------------------------------
    # 2. Refine the best small‑size candidate with Adam (many restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        # random sparse start is a good diverse initialiser
        start_vec = random_sparse_start(n_small)
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 200.0,
            max_iters=120_000,
            beta_start=5.0,
            beta_end=5_000.0,
            lr=0.018,
        )
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_vec_small = a_opt.copy()
            print(f"[{time.time():.1f}] new best after Adam‑smooth (small) = {best_score_small:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample to the large size
    # ------------------------------------------------------------------
    if best_vec_small is None:
        best_vec_small = uniform_family(n_small)

    x_small = np.arange(n_small, dtype=np.float64)
    x_large = np.linspace(0, n_small - 1, n_large, dtype=np.float64)
    best_vec_large = np.interp(x_large, x_small, best_vec_small)
    best_vec_large = project_to_simplex(best_vec_large)
    best_score_large = compute_score_fast(best_vec_large)
    print(f"[{time.time():.1f}] upsampled score (large) = {best_score_large:.6f}")

            best_vec_large,
            deadline - 100.0,
            step0=0.9,
            check_interval=2000,
        )
        if sc_sub < best_score_large:
            best_score_large = sc_sub
            best_vec_large = a_sub.copy()
            print(f"[{time.time():.1f}] after mirror sub‑grad (large) = {best_score_large:.6f}")

    # ------------------------------------------------------------------
    # 5. Second Adam smoothing with a very large β (approaching the true max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] start second Adam smoothing (large)")
        a_smooth, sc_smooth = adam_smooth_optimize(
            best_vec_large,
            deadline - 40.0,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=20_000.0,
            lr=0.02,
        )
        if sc_smooth < best_score_large:
            best_score_large = sc_smooth
            best_vec_large = a_smooth.copy()
            print(f"[{time.time():.1f}] after second Adam (large) = {best_score_large:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (many optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_vec_large is not None:
        a_sym = (best_vec_large + best_vec_large[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score_fast(a_sym)
        if sc_sym < best_score_large:
            best_score_large = sc_sym
            best_vec_large = a_sym.copy()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score_large:.6f}")

    # ------------------------------------------------------------------
    # 7. Final hill‑climb with simulated annealing
    # ------------------------------------------------------------------
    if best_vec_large is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_hc, sc_hc = hill_climb_sa(
            best_vec_large,
            deadline,
            max_steps=1_200_000,
            init_temp_factor=5e-4,
        )
        if sc_hc < best_score_large:
            best_score_large = sc_hc
            best_vec_large = a_hc.copy()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score_large:.6f}")

    # ------------------------------------------------------------------
    # 8. Fallback safety (uniform vector) – should never be needed
    # ------------------------------------------------------------------
    if best_vec_large is None:
        best_vec_large = np.full(256, 1.0 / 256)
        best_score_large = compute_score_fast(best_vec_large)

    # ------------------------------------------------------------------
    # 9. Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_vec_large.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return plain Python list
    return final_seq
