def subgradient_maxconv(a):
    conv = convolution_fft(a, a)
    maxc = conv.max()
    ks = np.where(conv == maxc)[0]
    k = ks[0]  # pick first
    # gradient = 2 * a_rev shifted.
    # Let shift = k - (n-1). Actually need to map.
    # Using relation: conv[k] = sum_{i} a[i] a[k-i]
    # For each j from max(0,k-(n-1)) to min(n-1,k):
    # g[j] = 2 * a[k-j]
    # Implementation:
    start = max(0, k-(n-1))
    end = min(n-1, k)
    # indices j = start..end inclusive, then k-j in range.
    g[start:end+1] = 2 * a[k - np.arange(start, end+1)]

"""
Propose a step‑function (non‑negative list) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the aperiodic convolution.  The algorithm consists of
three phases that respect a 1000 s wall‑clock budget:

1. **Broad random exploration** – generate many random vectors on the
   probability simplex (Dirichlet draws with several concentration
   parameters) and keep the best few.

2. **Smooth‑max exponentiated‑gradient refinement** – minimise a
   log‑sum‑exp approximation of the maximum convolution value with a
   KL‑mirror (multiplicative) update.  The smoothing parameter β is
   annealed to a very large value (≈10⁵) so that the surrogate becomes
   an accurate proxy for the true max.  The update is performed until
   only a few seconds remain.

3. **Greedy stochastic hill‑climb** – from the best vector obtained
   above repeatedly transfer a small packet of probability mass between
   two coordinates (or swap two coordinates).  Only improving moves are
   kept; the step‑size is adapted dynamically.  The procedure runs to
   the end of the wall‑clock budget.

All helper routines are top‑level, use only `numpy` and respect the
time limit.  The final list of floats is returned; the harness will
call `evaluate_sequence` on it for the official score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast for large n)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective `2·n·max(conv(a,a))` for a vector normalised to sum=1.
    This matches the value returned by `evaluate_sequence` for such a
    normalised vector.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random‑direction exploration
# ----------------------------------------------------------------------


def random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a point from Dirichlet(alpha,…,alpha) and return as float array."""
    if alpha <= 0.0:
        # degenerate uniform limit
        return np.full(n, 1.0 / n)
    x = np.random.gamma(shape=alpha, scale=1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return x / s


# ----------------------------------------------------------------------
# Sub‑gradient step for the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_maxconv(a: np.ndarray):
    """
    Return a sub‑gradient of the pointwise maximum of a*a and the
    corresponding maximum value.  The gradient w.r.t. a is
    `g_j = 2 * a_{k_max-j}` for all j where the index is inside the
    support, otherwise 0.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    maxc = float(conv.max())
    # pick the first index attaining the maximum (any works for a sub‑gradient)
    k = int(np.argmax(conv))
    g = np.zeros_like(a)
    # feasible range for i where both i and k-i lie in [0,n)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    # j = i, contribution 2 * a[k-i]
    idx = np.arange(i_min, i_max + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g, maxc


def gradient_descent_maxconv(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    The step size follows η₀ / sqrt(t).  Returns the best vector seen.
    """
    a = np.maximum(start_a, 0.0)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        g, _ = subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)

        # projected gradient update
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new  # keep the improvement
        else:
            # occasionally accept a non‑improving step to shake out of flat plateaus
            if np.random.rand() < 0.001:
                a = a_new

        # occasional logging
        if it % 5000 == 0:
            print(f"[{time.time():.1f}] GD iter {it}, best score {best_sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max exponentiated‑gradient optimisation (existing idea, but
# with a more aggressive schedule and longer run)
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 800_000,
    beta_start: float = 0.5,
    beta_end: float = 150_000.0,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Minimise a log‑sum‑exp approximation of the max‑convolution using a
    KL‑mirror (exponentiated‑gradient) update.  The smoothing parameter
    β is annealed exponentially from `beta_start` to `beta_end`, ending
    with a very large value so the surrogate is practically the true
    max.  The routine runs until the time budget expires.
    """
    a = np.maximum(start_a.astype(float), 1e-14)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- β schedule (exponential) ---
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # --- smooth max via log‑sum‑exp ---
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # the exponentials are ≤ 1 because we subtract maxc, safe from overflow
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()  # a probability distribution over the 2n‑1 positions

        # gradient of the smoothed objective (cf. original code)
        a_rev = a[::-1]
        # convolution of the reversed a with w gives the gradient components
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # step size (AdaGrad‑like decay)
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update
        a = np.maximum(a * np.exp(-eta * grad), 1e-14)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] exp‑grad iter {it}, new best {best_sc:.6f}")

    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print(f"[{time.time():.1f}] final exp‑grad best {best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb with adaptive step size
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 8_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic walk:
        * with prob 0.5 swap two entries,
        * otherwise transfer a fraction of the mass from one index to another.
    Only moves that improve the exact score are kept; the step‑scale
    adapts dynamically (increases after success, decays otherwise).
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12  # initial fraction of mass to move

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # swap move
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                         # mass‑transfer move
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder after a win
        else:
            step_scale *= 0.97                         # become more cautious

        steps += 1
        if steps % 500_000 == 0:
            print(f"[{time.time():.1f}] hill‑climb step {steps}, best {best_sc:.6f}")

    return best_a, best_sc

    remaining = deadline - start_time
