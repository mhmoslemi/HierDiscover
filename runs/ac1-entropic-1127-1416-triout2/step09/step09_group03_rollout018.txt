"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.
-------------------------------------------------------------
Goal  : minimise 2·n·max(conv(a,a))  (a is forced onto the simplex).
Strategy
---------
0.  Initialise with a cheap but expressive family of analytic shapes
    (power‑law, exponential, beta‑like).  Scan a modest grid of
    parameters – each candidate is projected to the simplex and
    evaluated with an FFT‑based convolution.
1.  Take the best analytic shape and polish it with a smooth
    log‑sum‑exp surrogate of the max‑convolution using Adam.
    The smoothing parameter β is annealed from 5 to 1000, the step‑size
    follows a 1/√t schedule and each iterate is Euclidean‑projected onto
    the probability simplex.
2.  Switch to exact projected sub‑gradient descent on the true
    max‑convolution objective.  The gradient is non‑zero only for the
    active (max) entry, which makes each iteration cheap.
3.  Finish with a tiny stochastic hill‑climb (fractional mass transfer
    and swaps) to escape possible plateaus.
All stages respect the remaining wall‑clock budget and keep the best
vector seen.  The final return value is a plain list of non‑negative
floats (already on the simplex).
"""

import time
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (∑a = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    beta_start: float = 5.0,
    beta_end: float = 1000.0,
    step0: float = 0.9,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)           # first moment
    v = np.zeros_like(a)           # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # anneal smoothing parameter β
        frac = (time.time() - (deadline - (deadline - start_time))) / (deadline - start_time)
        beta = beta_start * (beta_end / beta_start) ** min(frac, 1.0)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights
        w = np.exp(beta * (conv - maxc))
        w = w / w.sum()

        # gradient of the smooth max: 2 * (a_rev ∗ w) restricted to n entries
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = step0 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_to_simplex(a)

        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact projected sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgradient_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.15,
    max_iter: int = 400_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = start_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active (max) entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a = a - step * g
        a = project_to_simplex(a)

        cur = compute_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    init_seq,
    deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(init_seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Analytic families – cheap grid search
# ----------------------------------------------------------------------


def generate_power_law(n, gamma):
    """a_i ∝ (i+1)^{-gamma}."""
    i = np.arange(1, n + 1)
    a = i ** (-gamma)
    return project_to_simplex(a)


def generate_exponential(n, lam):
    """a_i ∝ exp(-lam*i)."""
    i = np.arange(n)
    a = np.exp(-lam * i)
    return project_to_simplex(a)


def generate_beta_symmetric(n, p):
    """a_i ∝ (x)^{p-1}(1-x)^{p-1}, x = (i+0.5)/n."""
    x = (np.arange(n) + 0.5) / n
    a = (x ** (p - 1)) * ((1.0 - x) ** (p - 1))
    return project_to_simplex(a)


def generate_mixed_power_exp(n, gamma, lam):
    """a_i ∝ (i+1)^{-gamma} * exp(-lam*i)."""
    i = np.arange(n)
    a = (i + 1) ** (-gamma) * np.exp(-lam * i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # Reserve a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float("inf")
    n = 2048                        # final length

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]          # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"[fallback] score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Quick grid‑search over a few analytic families
    # -----------------------------------------------------------------
    grid_deadline = deadline - 300.0   # keep ~300 s for refinement phases
    families_evaluated = 0

    def try_candidate(a):
        nonlocal best_seq, best_val, families_evaluated
        families_evaluated += 1
        score = compute_score(a)
        if score < best_val:
            best_val = score
            best_seq = a.tolist()
            print(f"[grid] new best {best_val:.6f} after {families_evaluated} families")

    # power‑law
    for gamma in np.linspace(0.15, 0.90, 16):
        if time.time() >= grid_deadline:
            break
        a = generate_power_law(n, gamma)
        try_candidate(a)

    # exponential
    for lam in np.linspace(0.0002, 0.015, 16):
        if time.time() >= grid_deadline:
            break
        a = generate_exponential(n, lam)
        try_candidate(a)

    # beta‑symmetric
    for p in np.linspace(0.7, 3.5, 14):
        if time.time() >= grid_deadline:
            break
        a = generate_beta_symmetric(n, p)
        try_candidate(a)

    # mixed power–exponential
    for gamma in np.linspace(0.3, 0.8, 6):
        for lam in np.linspace(0.001, 0.008, 5):
            if time.time() >= grid_deadline:
                break
            a = generate_mixed_power_exp(n, gamma, lam)
            try_candidate(a)

    # a few random Dirichlet draws (they sometimes discover unexpected shapes)
    for _ in range(120):
        if time.time() >= grid_deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        a = project_to_simplex(a)
        try_candidate(a)

    if best_seq is None:
        # fallback uniform (should never happen)
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))
        print(f"[fallback uniform] score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smooth Adam optimisation (log‑sum‑exp surrogate)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 150.0
    if time.time() < smooth_deadline:
        print("[smooth] starting Adam optimisation")
        a_start = np.array(best_seq, dtype=float)
        a_smooth, smooth_score = smooth_adam_optimize(
            a_start,
            deadline=smooth_deadline,
            beta_start=5.0,
            beta_end=1000.0,
            step0=0.9,
            max_iters=800_000,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"[smooth] improved to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Projected sub‑gradient polishing
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 30.0
    if time.time() < subgrad_deadline:
        print("[subgrad] starting projected sub‑gradient descent")
        a_start = np.array(best_seq, dtype=float)
        a_refined, subgrad_score = subgradient_optimize(
            a_start,
            deadline=subgrad_deadline,
            step0=0.15,
            max_iter=400_000,
        )
        if subgrad_score < best_val:
            best_val = subgrad_score
            best_seq = a_refined.tolist()
            print(f"[subgrad] improved to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        print("[hill‑climb] final polishing")
        polished_seq, hill_score = hill_climb(
            best_seq,
            deadline=hill_deadline,
            max_steps=500_000,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = polished_seq
            print(f"[hill‑climb] improved to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Report final result (the harness will call evaluate_sequence itself)
    # -----------------------------------------------------------------
    final_report = evaluate_sequence(best_seq)
    print("\n--- search finished ---")
    print(f"Best value reported by evaluate_sequence : {final_report:.6f}")
    print(f"Vector length = {len(best_seq)}")
    return best_seq
