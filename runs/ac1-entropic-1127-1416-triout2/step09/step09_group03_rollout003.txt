"""
Non‑negative step‑function optimiser (≈ 1000 s).

Strategy
--------
1. **Initial shape** – central‑binomial coefficients (≈ i⁻¹ᐟ²) on a modest
   length (n=512).  The vector is projected onto the simplex (sum = 1).

2. **Iterated LP refinement** – given a current vector `a`,
   compute M = max(conv(a,a)).  Solve the linear programme

        maximise   Σ b_j
        subject to  conv(a,b) ≤ M      (2n‑1 constraints)
                    b ≥ 0

   (variables = the partner vector `b`).  The LP is solved with
   `scipy.optimize.linprog` (HiGHS) on a scaled constraint matrix for
   numerical stability.

   The optimal partner is projected back onto the simplex and mixed with
   `a`.  The mixing weight `α∈[0,1]` is chosen by a cheap ternary search on
   the exact objective `2·n·max(conv((1‑α)a+αb,(1‑α)a+αb))`.  Because the
   objective is convex in α the search is guaranteed to find the best
   convex combination.  Several dozen LP iterations are performed while
   time permits.

3. **Upscaling** – the refined short vector is linearly interpolated to
   the final length (`n=2048`) and projected onto the simplex.

4. **Smooth Adam optimisation** – a log‑sum‑exp surrogate of the
   `max`‑convolution is minimised with Adam (temperature β annealed from
   5 to 200).  This gives fine‑grained adjustments that are difficult for
   the LP.

5. **Projected sub‑gradient polishing** – after the smooth phase we
   perform classical sub‑gradient descent on the true non‑smooth
   objective.

6. **Stochastic hill‑climb** – a final cheap random‑move phase (swap or
   fractional mass transfer) with simplex projection to escape any tiny
   plateaus.

All stages respect the remaining wall‑clock budget and keep the best
sequence seen.  The returned list is a plain Python list of non‑negative
floats (its sum is 1, so `evaluate_sequence` will only apply the final
clipping and scaling).
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution matrix (Toeplitz) for the LP
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix A such that A @ b = conv(f, b).
    Column i contains the vector f shifted down by i rows.
    """
    n = f.shape[0]
    diags = [f] * n          # each column is the same vector f
    offsets = np.arange(n)   # shifts 0 … n‑1
    A = sp.diags(diagonals=diags, offsets=offsets,
                 shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum_scipy(A: sp.csr_matrix, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    # Scale constraints so that the right‑hand side is 1.0 – improves
    # numerical stability because M is typically ≈10⁻³.
    A_scaled = A.multiply(1.0 / M)
    m = A_scaled.shape[0]
    n = A_scaled.shape[1]
    ub = np.ones(m)               # all constraints become ≤ 1
    c = -np.ones(n)               # maximise sum(b) → minimise -sum(b)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_scaled, b_ub=ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
    except Exception:
        return None

    if not res.success:
        return None
    return np.maximum(res.x, 0.0)


# ----------------------------------------------------------------------
# Generators
# ----------------------------------------------------------------------


def central_binomial_distribution(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i  (≈ i⁻¹ᐟ²) for i=0,…,n‑1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)   # first moment
    v1 = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < time_deadline and it < max_iters:
        it += 1

        # anneal smoothing parameter β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # log‑sum‑exp trick
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the soft‑max version:
        # ∂/∂a_i ( Σ_k w_k (a*a)_k ) = 2 * (reverse(a) * w)_i (restricted)
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]        # keep n entries

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        v1 = adam_beta2 * v1 + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m1 / (1.0 - adam_beta1 ** it)
        v_hat = v1 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # occasional progress report
                print(f"[smooth] it={it:,}, score={cur_score:.6f}")

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to help escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        if t % 5000 == 0:
            print(f"[subgrad] it={t:,}, best={best_score:.6f}")

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 800_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

        if max_steps % 50000 == 0:
            print(f"[hill] steps left={max_steps:,}, best={best_score:.6f}")

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"[fallback] initial score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Coarse LP refinement on a small problem (n≈512)
    # -----------------------------------------------------------------
    n_small = 512
    a = central_binomial_distribution(n_small)
    a = project_to_simplex(a)

    best_small_score = compute_score(a)
    if best_small_score < best_val:
        best_val = best_small_score
        best_seq = a.tolist()
        print(f"[init] small n={n_small}, score={best_small_score:.6f}")

    # LP loop parameters
    lp_iters = 0
    max_lp_iters = 60               # generous – will stop earlier if time runs out
    lp_deadline = deadline - 300.0  # reserve most of the budget for later stages

    while time.time() < lp_deadline and lp_iters < max_lp_iters:
        lp_iters += 1

        # current bound M
        conv_a = convolution_fft(a, a)
        M = float(conv_a.max())
        cur_score = 2.0 * n_small * M
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a.tolist()
            print(f"[LP {lp_iters}] new best (score={cur_score:.6f})")

        # build Toeplitz matrix for conv(a,·)
        A = build_convolution_matrix(a)

        # solve LP to obtain partner b (maximises Σ b_j under conv(a,b) ≤ M)
        b = solve_lp_max_sum_scipy(A, M)
        if b is None:
            print("[LP] linear programme failed – exiting LP loop.")
            break

        # normalise partner to the simplex
        b = project_to_simplex(b)

        # ----- choose mixing weight α by ternary search (convex) -----
        lo, hi = 0.0, 1.0
        for _ in range(15):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            cand1 = (1 - m1) * a + m1 * b
            cand2 = (1 - m2) * a + m2 * b
            s1 = compute_score(cand1)
            s2 = compute_score(cand2)
            if s1 < s2:
                hi = m2
            else:
                lo = m1
        alpha = (lo + hi) / 2.0
        a_new = (1 - alpha) * a + alpha * b
        a_new = project_to_simplex(a_new)

        new_score = compute_score(a_new)

        # accept if we made a noticeable improvement
        if new_score < cur_score * 0.9999:
            a = a_new
            print(f"[LP {lp_iters}] improvement: α={alpha:.3f}, score={new_score:.6f}")
        else:
            # no clear improvement – shrink the mixing aggressiveness
            if alpha < 1e-3:
                print("[LP] mixing weight too small, terminating LP phase.")
                break
            # keep current a, but try a lighter mix in the next iteration
            a = (1 - 0.05) * a + 0.05 * b
            a = project_to_simplex(a)
            print(f"[LP {lp_iters}] modest mix applied, score unchanged ({cur_score:.6f})")

    print(f"[LP] finished after {lp_iters} iterations, best small score={best_small_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Upscale to the final length (2048) for fine‑grained optimisation
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    n_big = 2048 if remaining > 350 else 1024
    a_big = upscale_vector(a, n_big)
    print(f"[upscale] interpolated to n={n_big}")

    # -----------------------------------------------------------------
    # 3. Smooth Adam optimisation (log‑sum‑exp surrogate)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 140.0
    if time.time() < smooth_deadline:
        a_smooth, smooth_score = smooth_gradient_optimize(
            a_big,
            time_deadline=smooth_deadline,
            max_iters=600_000,
            beta_start=6.0,
            beta_end=200.0,
            step0=0.9,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"[smooth] improved best to {smooth_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Projected sub‑gradient polishing
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 60.0
    if time.time() < subgrad_deadline and best_seq is not None:
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=subgrad_deadline,
            max_iter=800_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"[subgrad] improved best to {refined_score:.6f}")

    # -----------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline and best_seq is not None:
        polished_seq, polished_score = hill_climb_permutation(
            best_seq,
            time_deadline=hill_deadline,
            max_steps=900_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"[hill] improved best to {polished_score:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a tiny uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[final] Finished search. Reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
