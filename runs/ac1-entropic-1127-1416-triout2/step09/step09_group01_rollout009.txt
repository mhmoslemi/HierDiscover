"""
Optimise a non‑negative step‑function (a probability vector) to minimise

    evaluate_sequence(a) = 2·n·max(a*a) / (sum(a)**2)

The algorithm works in three phases inside the global 1000 s budget:

1. **Quick analytic start** – evaluate a few parameterised families
   (symmetric product, left‑heavy power‑law, exponential decay) and keep
   the best one.

2. **Smooth log‑sum‑exp optimisation** – run a long Adam optimisation on
   the smoothed surrogate
       f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
   with β annealed exponentially from 5 up to 5 000.  This gives a very
   accurate approximation of the true max‑objective.

3. **Exponentiated‑gradient sub‑gradient descent** – starting from the
   result of phase 2, perform many mirror‑descent steps on the *exact*
   max‑objective:
        a ← a ⊙ exp(‑η·g)   (⊙ = element‑wise product)
   where g is the sub‑gradient of the active convolution entry.
   η decays as 1/√t.  The loop runs until the time budget is exhausted,
   periodically recording the best exact score.

A final tiny local “mass‑transfer” refinement is performed (if time
remains) before returning the best vector found as a plain Python list.

All heavy work uses FFT‑based convolutions (O(n log n)) and respects the
wall‑clock deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the harness for a *probability* vector
    (i.e. sum(a) == 1).  The returned value is exactly what
    evaluate_sequence() would give after its internal scaling.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    # max of aperiodic convolution
    max_c = float(conv.max())
    return 2.0 * n * max_c


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the cheap initialisation)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 5_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- schedule for β (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a -----
        a = softmax(z)

        # ----- smoothed weights w_k ∝ exp(β·conv_k) -----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ----- gradient of f_β w.r.t. a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length ~3n‑2
        # the relevant slice corresponds to indices n‑1 … 2n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 2000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam iter {it}: new best {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 2.0,
    check_interval: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for the exact objective
    2·n·max(a*a).  The step size is η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                 # active shift
        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)                # decreasing stepsize
        a = a * np.exp(-eta * g)                # exponentiated update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s                               # renormalise to simplex

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror step {t}: improved best = {best_score:.6f}")
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Very fine local mass‑transfer refinement
# ----------------------------------------------------------------------


def local_refine(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 300_000,
    frac: float = 0.003,
) -> tuple[np.ndarray, float]:
    """
    Small deterministic mass‑transfer moves aimed at the dominant
    convolution entry.  Each step moves a tiny fraction of mass from the
    most offending index to the least contributing one, followed by a
    simplex projection.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contributions of each coefficient to this maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        # most harmful index (largest contribution) and a harmless one
        i_big = int(np.argmax(contrib))
        i_small = int(np.argmin(a))     # index with smallest value overall

        if a[i_big] <= 0.0:
            continue

        delta = frac * a[i_big]        # amount to relocate
        cand = a.copy()
        cand[i_big] -= delta
        cand[i_small] += delta

        # project back to simplex (positivity + normalisation)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand /= cand_sum

        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand                     # accept improvement immediately
        else:
            # occasional uphill acceptance (very low prob) to escape plateaus
            if np.random.rand() < 0.001:
                a = cand

    return best_a, best_score
