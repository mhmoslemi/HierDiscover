    best_a = a.copy()
    best_score = start_score
    beta_start = 5.0
    beta_end = 2000.0

"""
Advanced search for a step‑function that minimizes
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three phases

1. **Seeding** – a collection of hand‑crafted shapes (sine‑squared,
   Beta‑family, sparse spikes) and random Dirichlet draws.  The best
   seed (exact objective) is kept.

2. **Soft‑max gradient descent** – we minimise a smooth surrogate of
   the max‑convolution:
        Lβ(a) = log Σ exp(β·conv(a,a))
   whose gradient is
        ∇Lβ = 2β·correlate(softmaxβ(conv), a) .
   The temperature β is annealed from a small value to a large one,
   while a projected gradient step (onto the probability simplex) is
   performed.  The best iterate (according to the true objective) is
   retained.

3. **Final polishing** – a few thousand steps of the classic
   sub‑gradient descent followed by a greedy hill‑climb, exactly as in
   the previous state‑of‑the‑art code.

All work respects the 1000 s wall‑clock budget.  The returned value is a
plain ``list[float]`` of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin^2 shape a_i = sin^2(pi·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 clusters mass in the centre, p<0 pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 2 – soft‑max gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step_coef: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate of the L∞‑norm via projected gradient
    descent.  The surrogate is Lβ(a)=log Σ exp(β·conv(a,a)).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    start = time.time()
    it = 0
    while time.time() < deadline:
        # ----- exact objective (for bookkeeping) -----
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # ----- schedule temperature β -----
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # ----- soft‑max weights -----
        shifted = conv - max_conv                # max(conv) becomes 0
        exp_vals = np.exp(beta * shifted)       # stable, all ≤1
        w = exp_vals / exp_vals.sum()           # soft‑max distribution over 2n‑1 points

        # ----- gradient of the surrogate -----
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # ----- step size (decays with √iteration and with β) -----
        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        # ----- projected update -----
        nxt = cur_a - step * grad
        nxt = project_to_simplex(nxt)

        cur_a = nxt

        # occasional progress report
        if it % 5000 == 0:
            print(
                f"[GD] it {it:6d} | time {elapsed:6.1f}s | best {best_score:.6f} | β {beta:.1f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – sub‑gradient refinement (identical to previous implementation)
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """Stochastic moves that are kept only when they improve the exact score."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise (numerical drift)
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score

    best_seq = None
    best_val = float("inf")
    try:

        fb = evaluate_sequence(fallback)

        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Choose a moderate length – 2048 works well in 1000 s

    n = 2048

        n = 4096
    print(f"Using sequence length = {n}")

    # --------------------------------------------------------------
    # 2. Generate a diverse set of seeds and keep the best one
    # --------------------------------------------------------------
    seeds = []

    # deterministic shapes

        seeds.append(generate_beta_family(n, p))

    # random Dirichlet (various concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # very sparse spikes – often a good starting point for max‑conv reduction
    for k in (30, 50, 80):
        seeds.append(random_spike(n, k=k))

    # include fallback if it matches the current length

        if score < best_val:

    # --------------------------------------------------------------

        phase2_deadline = deadline - 30.0

    final_seq = cur_a.tolist()

    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
