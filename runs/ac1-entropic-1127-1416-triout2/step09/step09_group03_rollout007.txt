    m = len(a)+len(b)-1

"""
Non‑negative step‑function optimizer for ``evaluate_sequence``.

Approach
--------
1. **Parametric sweep** – explore several simple families of
   monotone / valley shapes (power‑law, geometric, symmetric power,
   valley = ((i+1)*(n‑i))⁻ᵖ).  The best candidate from this cheap search
   already beats the previous 1.511 value.

2. **Smoothed Adam** – minimise a log‑sum‑exp surrogate of the
   max‑convolution.  β is annealed from a modest to a very large value,
   the step‑size follows 1/√t and the iterate is projected onto the
   probability simplex.

3. **Projected sub‑gradient** – polish the result w.r.t. the true
   max‑convolution using a decreasing 1/√t step‑size.

4. **Stochastic hill‑climb** – tiny random mass‑transfer moves with
   simplex projection finish the search.

All phases respect the remaining wall‑clock budget and keep the
best sequence seen.  The returned list is a plain Python list of
non‑negative floats (its ℓ₁‑norm is 1, so ``evaluate_sequence`` will
only apply the final clipping and scaling).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for typical sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (∑a = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0).
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 1_000_000,
    beta_start: float = 5.0,
    beta_end: float = 300.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)  # first moment
    v1 = np.zeros_like(a)  # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal the smoothing parameter β geometrically
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # convolution a*a
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # log‑sum‑exp weights (soft‑max)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient = 2 * (a_rev ∗ w) restricted to n entries
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)          # length = n + (2n‑1)‑1 = 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        v1 = adam_beta2 * v1 + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m1 / (1.0 - adam_beta1 ** it)
        v_hat = v1 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional progress check
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Smooth] iter {it}, best score = {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 800_000,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (the k‑th entry of the convolution)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[Subgrad] iter {t}, improved score = {best_score:.6f}")

        a = a_new

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_optimize(
    seq: np.ndarray,
    deadline: float,
    max_steps: int = 600_000,
) -> tuple[np.ndarray, float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                # enforce non‑negativity (project_to_simplex will also fix it)
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_score
