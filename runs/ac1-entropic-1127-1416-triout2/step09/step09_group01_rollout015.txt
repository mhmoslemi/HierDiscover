def build_sym_block_seq(p, n):
    """
    Build symmetric sequence of length n from block weights p.
    p is a 1‑D numpy array of non‑negative block intensities.
    The left half (including middle) is filled by repeating each block value.
    The right half is the mirror of the left.
    Returns a normalized probability vector (sum=1).
    """
    half = (n + 1) // 2
    K = p.shape[0]
    # Determine block lengths: allocate as evenly as possible
    base_len = half // K
    rem = half - base_len * K
    lengths = np.full(K, base_len, dtype=int)
    # distribute remainder among first rem blocks
    lengths[:rem] += 1
    # Build left side
    left_blocks = [np.full(l, p[i], dtype=np.float64) for i, l in enumerate(lengths)]
    left = np.concatenate(left_blocks)[:half]  # just in case rounding
    # Build full symmetric sequence
    if n % 2 == 0:
        full = np.concatenate([left, left[::-1]])
    else:
        # duplicate middle element is already included, remove it from mirror
        full = np.concatenate([left, left[-2::-1]])
    # Clip negative values (should not happen)
    full = np.maximum(full, 0.0)
    # Normalise
    total = full.sum()
    if total == 0.0:
        # rare case: all zeros -> uniform fallback
        full = np.full_like(full, 1.0 / n)
    else:
        full = full / total
    return full

"""
Propose a step‑function sequence that minimises the given evaluation objective.

Algorithm
---------
1. Choose a modest length (default 8192).  Larger n gives a larger
   multiplicative factor 2·n, but the optimum max‑convolution scales roughly
   as 1/n, so a medium size is a good trade‑off for rapid experimentation.
2. Initialise a baseline from the analytic families used previously.
3. **Block‑search** – represent a symmetric distribution by K
   constant blocks on the left half of the vector and mirror it.
   Random Dirichlet samples for the K block heights are generated
   (different concentration parameters to explore sparse and dense
   shapes).  The best block‑vector is kept.
4. **Local refinement** – from the best block vector perform a cheap
   stochastic hill‑climb that moves a small amount of mass between two
   blocks (mass‑conserving, non‑negative).  Accept moves that improve the
   exact objective.
5. **Fine‑tune with Adam** – run a few thousand Adam steps on the full
   (non‑blocked) vector using the log‑sum‑exp surrogate of the max‑convolution.
   The smoothing parameter β is increased exponentially, driving the surrogate
   to the true max.
6. **Mirror sub‑gradient descent** – a few passes of the entropy‑mirrored
   sub‑gradient method (as in the reference implementation) give a final push.
7. **Symmetrisation** – averaging the vector with its reverse often reduces the peak.
8. Return the best sequence found (plain ``list[float]``).

All stages respect the global wall‑clock deadline (default 1000 s).  The routine
prints intermediate scores and timestamps for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT convolution, scoring, simplex projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Block representation utilities
# ----------------------------------------------------------------------


def build_symmetric_from_blocks(blocks: np.ndarray, n: int) -> np.ndarray:
    """
    Construct a symmetric non‑negative vector of length ``n`` from the
    left‑half block intensities ``blocks`` (len(blocks) = K).
    Each block is replicated a (nearly) equal number of entries.
    The resulting vector is normalised to sum to 1.
    """
    half = (n + 1) // 2               # size of the left side (includes centre if n odd)
    K = blocks.shape[0]

    # allocate block lengths as evenly as possible
    base_len = half // K
    rem = half - base_len * K
    lengths = np.full(K, base_len, dtype=int)
    lengths[:rem] += 1                 # first ``rem`` blocks get one extra entry

    # build the left half
    left_parts = [np.full(l, blocks[i], dtype=np.float64) for i, l in enumerate(lengths)]
    left = np.concatenate(left_parts)[:half]

    # mirror to obtain the full symmetric vector
    if n % 2 == 0:
        full = np.concatenate([left, left[::-1]])
    else:
        full = np.concatenate([left, left[-2::-1]])   # omit duplicated centre element

    full = np.maximum(full, 0.0)
    s = full.sum()
    if s == 0.0:
        # fallback – uniform distribution
        full = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        full = full / s
    return full


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 120_000,
    beta_start: float = 5.0,
    beta_end: float = 4000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """Minimise the smoothed max‑convolution using Adam on the soft‑max layer."""
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # geometric schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # soft‑max over convolution entries

        # gradient of the smoothed objective w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # back‑prop through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent (same as reference)
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """Exponentiated sub‑gradient descent on the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))  # index of the active diagonal

        # sub‑gradient (active terms only)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb on the simplex (tiny simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing style local search."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()            # project back onto simplex
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level search driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the global time budget and returns a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length choice – a compromise between flexibility and speed.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 8192          # fairly large but still cheap to evaluate many candidates
    else:
        n = 4096
    print(f"[{time.time():.1f}] chosen length n = {n}")

    # ------------------------------------------------------------------
    # Fallback: use a previously known construction if provided.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families (cheap dense grid)
    # ------------------------------------------------------------------
    # symmetric product family
    for a in np.linspace(0.0, 8.0, 161):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={a:.2f}) = {best_score:.6f}")

    # left‑heavy power law
    for b in np.linspace(0.0, 8.0, 161):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, b)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {best_score:.6f}")

    # exponential decay family
    for g in np.linspace(0.0, 10.0, 201):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, g)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={g:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Block‑search (low‑dimensional global exploration)
    # ------------------------------------------------------------------
    K = 20                         # number of blocks on the left half
    half_len = (n + 1) // 2
    print(f"[{time.time():.1f}] starting block‑search with K={K}")

    # Helper to evaluate a block vector
    def eval_blocks(p_vec: np.ndarray) -> float:
        a_full = build_symmetric_from_blocks(p_vec, n)
        return compute_score(a_full)

    # Random Dirichlet samples – mix different concentration parameters
    while time.time() < deadline - 120.0:
        conc = np.random.uniform(0.1, 2.0)        # sparsity control
        p = np.random.dirichlet(np.full(K, conc))
        sc = eval_blocks(p)
        if sc < best_score:
            best_score = sc
            best_seq = build_symmetric_from_blocks(p, n).tolist()
            print(f"[{time.time():.1f}] block‑search new best (rand dir, conc={conc:.2f}) = {best_score:.6f}")

        # occasional finer grid around current best (if we already have one)
        if best_seq is not None and np.random.rand() < 0.02:
            # extract block representation of the best sequence (inverse operation)
            # simply reuse the same best blocks by normalising the left half
            best_arr = np.array(best_seq, dtype=np.float64)
            left = best_arr[:half_len]
            # compute average value per block (inverse of build_symmetric_from_blocks)
            base_len = half_len // K
            rem = half_len - base_len * K
            lengths = np.full(K, base_len, dtype=int)
            lengths[:rem] += 1
            idx = 0
            p_guess = []
            for l in lengths:
                p_guess.append(left[idx:idx + l].mean())
                idx += l
            p_guess = np.array(p_guess)
            p_guess = np.maximum(p_guess, 0.0)
            p_guess = p_guess / p_guess.sum()
            # jitter a little
            noise = np.random.randn(K) * 0.05
            p_new = p_guess * (1.0 + noise)
            p_new = np.maximum(p_new, 0.0)
            if p_new.sum() == 0.0:
                continue
            p_new = p_new / p_new.sum()
            sc = eval_blocks(p_new)
            if sc < best_score:
                best_score = sc
                best_seq = build_symmetric_from_blocks(p_new, n).tolist()
                print(f"[{time.time():.1f}] block‑search refinement improved = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local hill‑climb on the block representation (mass re‑allocation)
    # ------------------------------------------------------------------
    if best_seq is not None:
        blk_best = None
        # derive current block vector from best_seq for a coherent start
        best_arr = np.array(best_seq, dtype=np.float64)
        left = best_arr[:half_len]
        base_len = half_len // K
        rem = half_len - base_len * K
        lengths = np.full(K, base_len, dtype=int)
        lengths[:rem] += 1
        idx = 0
        p_curr = []
        for l in lengths:
            p_curr.append(left[idx:idx + l].mean())
            idx += l
        p_curr = np.array(p_curr)
        p_curr = np.maximum(p_curr, 0.0)
        if p_curr.sum() == 0.0:
            p_curr = np.full(K, 1.0 / K)
        p_curr = p_curr / p_curr.sum()
        blk_best = p_curr.copy()
        blk_best_score = compute_score(build_symmetric_from_blocks(blk_best, n))

        # hill‑climb loop
        while time.time() < deadline - 80.0:
            # propose a small mass transfer between two random blocks
            i, j = np.random.choice(K, size=2, replace=False)
            if blk_best[i] == 0.0:
                continue
            delta = blk_best[i] * np.random.uniform(1e-4, 0.05)
            new_p = blk_best.copy()
            new_p[i] = max(new_p[i] - delta, 0.0)
            new_p[j] += delta
            # re‑normalise (already preserves total mass)
            new_p = new_p / new_p.sum()
            sc = eval_blocks(new_p)
            if sc < blk_best_score:
                blk_best = new_p
                blk_best_score = sc
                # keep the globally best sequence as well
                if sc < best_score:
                    best_score = sc
                    best_seq = build_symmetric_from_blocks(new_p, n).tolist()
                    print(f"[{time.time():.1f}] block‑hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Fine‑tune the full vector with Adam on the smoothed surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 45.0:
        print(f"[{time.time():.1f}] starting Adam smooth optimisation")
        a_opt, sc_opt = adam_smooth_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=120_000,
            beta_start=5.0,
            beta_end=4000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Mirror sub‑gradient descent (entropy mirror) on the exact objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 10.0,
            step0=0.8,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (averaging with reverse)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final stochastic hill‑climb (tiny SA) for any last gains
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – fall back to a modest uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq
