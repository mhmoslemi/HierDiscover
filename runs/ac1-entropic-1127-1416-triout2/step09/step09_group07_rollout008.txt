def simulated_annealing_optimize(start_a, deadline, max_moves=None):
    a = start_a.copy()
    n = a.shape[0]
    conv = convolution_fft(a,a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # parameters
    total_time = deadline - time.time()
    T0 = cur_score * 0.1   # initial temperature
    T_end = cur_score * 1e-5
    step_scale = 0.1
    inc_factor = 1.02
    dec_factor = 0.995
    moves = 0
    recompute_interval = 5000

    while time.time() < deadline:
        # compute fraction of time elapsed
        elapsed = time.time() - start_time
        t_frac = elapsed / (deadline - start_time)
        T = T0 * (1 - t_frac) + T_end * t_frac

        # choose src weighted by a (mass)
        src = np.random.choice(n, p=a)   # ensures a[src] >0 mostly
        # choose dst distinct
        dst = np.random.randint(n-1)
        if dst >= src:
            dst += 1

        # compute delta
        delta = step_scale * a[src]
        if delta <= 0:
            continue
        # candidate conv update
        conv_cand = conv.copy()
        # linear updates
        conv_cand[src:src+n] -= 2.0 * delta * a
        conv_cand[dst:dst+n] += 2.0 * delta * a
        # quadratic
        conv_cand[2*src] += delta * delta
        conv_cand[2*dst] += delta * delta
        conv_cand[src+dst] -= 2.0 * delta * delta

        cand_score = 2.0 * n * conv_cand.max()

        # accept?
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability acceptance
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # update state
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_score = cand_score
            # update step_scale
            step_scale = min(step_scale * inc_factor, 0.5)
            # symmetrize?
            # optional: a = (a + a[::-1])/2; recompute conv later
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            step_scale *= dec_factor

        moves += 1
        # periodic full recompute
        if moves % recompute_interval == 0:
            conv = convolution_fft(a,a)
            cur_score = 2.0 * n * conv.max()

        # optionally restart if no improvement for many moves? maybe skip.

    return best_a, best_score

"""
Simulated‑annealing + sub‑gradient refinement for the step‑function
upper‑bound problem.

The algorithm works in three phases:

1. **Analytic seeding** – scan a one‑parameter family
   a_i(α) ∝ [(i+½)(n‑½‑i)]^{‑α} (and a few random Dirichlet starts)
   on a modest grid and keep the best vector (normalized to sum = 1).

2. **Stochastic annealing** – a temperature‑controlled random walk that
   moves a packet of mass from a source index to a destination index.
   The convolution is updated incrementally (O(n) per move) using the
   exact formula for a mass transfer, allowing millions of cheap
   proposals.  Moves that improve the exact objective are always kept;
   uphill moves are accepted with probability exp(‑Δ/T).  The step‑size
   adapts based on recent acceptance rate.

3. **Projected sub‑gradient descent** – a short Euclidean‑projected
   sub‑gradient run from the best annealed point for fine polishing.

All phases respect the global 1000 s wall‑clock limit; the best vector
found is returned as a plain list of non‑negative floats (already
normalized, so the final call to ``evaluate_sequence`` will compute the
correct score).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector already normalized to sum = 1.
    This matches the value returned by `evaluate_sequence` for the same
    vector (the denominator Σa² is 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for seeding
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the probability simplex."""
    i = np.arange(n, dtype=float)
    # log of the product (i+½)*(n-½-i) – stay in log‑space for stability
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow/underflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0
    return w / w.sum()


def explore_alpha_grid(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a moderate α‑grid and return the best vector and its score."""
    best_vec = None
    best_sc = float("inf")
    # a slightly broader grid than the original implementation
    alphas = np.concatenate([
        np.linspace(-2.0, -0.2, 19),   # strongly centre‑heavy
        np.linspace(-0.2, 0.6, 17),    # around flat
        np.linspace(0.6, 3.0, 25),     # edge‑heavy
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        vec = alpha_start(n, alpha)
        sc = compute_score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    return best_vec, best_sc


def random_dirichlet_start(n: int, conc: float = 0.5) -> np.ndarray:
    """Generate a random point on the simplex using a Dirichlet distribution."""
    sample = np.random.gamma(conc, 1.0, size=n)
    if sample.sum() == 0.0:
        sample[:] = 1.0
    return sample / sample.sum()


# ----------------------------------------------------------------------
# Simulated annealing optimizer (high‑volume cheap moves)
# ----------------------------------------------------------------------


def simulated_annealing_optimize(start_vec: np.ndarray,
                                 deadline: float,
                                 min_moves: int = 0) -> tuple[np.ndarray, float]:
    """
    Random‑walk with Metropolis acceptance that updates the convolution
    incrementally.  Returns the best vector (sum = 1) seen before ``deadline``.
    """
    a = start_vec.astype(float).copy()
    n = a.shape[0]
    conv = convolution_fft(a, a)          # full convolution, length 2n‑1
    cur_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_sc = cur_score

    start_time = time.time()
    total_time = deadline - start_time
    # temperature schedule (linear annealing)
    T_start = cur_score * 0.10
    T_end = cur_score * 1e-6

    # step‑size adaption parameters
    step_scale = 0.12      # initial fraction of the source mass to move
    inc_factor = 1.02
    dec_factor = 0.95

    moves = 0
    recompute_interval = max(5000, n // 2)
    symmetrise_interval = max(10000, n)

    while time.time() < deadline or moves < min_moves:
        # ---- temperature (linear decay) ----
        elapsed = time.time() - start_time
        t_frac = min(1.0, elapsed / total_time)
        T = T_start * (1.0 - t_frac) + T_end * t_frac

        # ---- choose source and destination ----
        # source: weighted by current mass (more likely to move mass from a large bucket)
        src = np.random.choice(n, p=a)
        # destination: uniform index different from src
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        if a[src] == 0.0:
            continue
        # transfer a positive packet
        delta = step_scale * a[src]          # 0 < delta ≤ a[src]
        if delta <= 0.0:
            continue

        # ---- candidate convolution (incremental update) ----
        conv_cand = conv.copy()
        # linear contributions
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic corrections
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_score = 2.0 * n * float(conv_cand.max())

        # ---- Metropolis acceptance ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability ≈ exp(‑Δ/T)
            dE = cand_score - cur_score
            if dE / T < 50.0:               # avoid overflow
                if np.random.rand() < np.exp(-dE / T):
                    accept = True

        if accept:
            # apply the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_score = cand_score
            # adapt step size upwards on success
            step_scale = min(step_scale * inc_factor, 0.5)

            if cur_score < best_sc:
                best_sc = cur_score
                best_a = a.copy()
        else:
            # shrink step size on rejection
            step_scale *= dec_factor

        moves += 1

        # ---- periodic full recomputation to avoid drift ----
        if moves % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())

        # ---- enforce symmetry every once in a while (helps the shape) ----
        if moves % symmetrise_interval == 0:
            a = (a + a[::-1]) * 0.5
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())
            if cur_score < best_sc:
                best_sc = cur_score
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient refinement (optional final polish)
# ----------------------------------------------------------------------


def subgrad_refine(start_vec: np.ndarray,
                   deadline: float,
                   max_iters: int = 200_000,
                   step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Euclidean projected sub‑gradient descent on the exact objective."""
    a = start_vec.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    start = time.time()
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑convolution at the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function vector that minimises
    `evaluate_sequence`.  The routine follows three stages described above
    and respects the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # -------------------------------------------------
    # Choose a problem size (larger n gives a stronger theoretical bound
    # but also multiplies the objective by n).  We adapt to the remaining
    # time budget.
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback if a previous construction exists
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic seeding (α‑grid + a few random Dirichlet starts)
    # -------------------------------------------------
    vec_alpha, sc_alpha = explore_alpha_grid(n, deadline - 30.0)
    if vec_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = vec_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # a couple of random Dirichlet starts for diversity
    for _ in range(3):
        if time.time() >= deadline - 30.0:
            break
        cand = random_dirichlet_start(n, conc=0.4)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] improved by random start = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Simulated annealing (majority of the remaining budget)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting simulated annealing")
        a_start = np.array(best_seq, dtype=float)
        # enforce a tiny epsilon to avoid exact zeros (helps weighted sampling)
        a_start = np.maximum(a_start, 1e-15)
        a_start /= a_start.sum()
        a_sa, sc_sa = simulated_annealing_optimize(
            a_start,
            deadline - 30.0,
            min_moves=0,
        )
        if sc_sa < best_score:
            best_score = sc_sa
            best_seq = a_sa.tolist()
            print(f"[{time.time():.1f}] after annealing best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Short projected sub‑gradient refinement (fine polishing)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] sub‑gradient polishing")
        a_sub, sc_sub = subgrad_refine(
            np.array(best_seq, dtype=float),
            deadline,
            max_iters=200_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: fall back to a uniform vector if nothing succeeded
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats
    return best_seq
