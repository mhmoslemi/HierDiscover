"""
Propose a non‑negative step‑function (sequence) that minimizes the
``evaluate_sequence`` objective.

The algorithm combines three complementary optimisation stages:

1. **Exploration of simple parametric families** – generate a few
   deterministic candidates (central‑binomial, symmetric power‑law
   shapes with various exponents) and keep the best one as a seed.

2. **Smooth surrogate optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution using L‑BFGS‑B while
   annealing the smoothing parameter β (continuation).  This yields a
   high‑quality basin on the probability simplex.

3. **Exact polishing** – a projected sub‑gradient descent on the true
   2·n·max(conv) objective, followed by a light greedy hill‑climb that
   repeatedly transfers a small amount of mass between two coordinates.
   All steps respect the simplex constraints and the 1000 s wall‑clock
   budget.

The code is fully self‑contained, uses only NumPy/SciPy, and returns a
plain list of floats suitable for the competition harness.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (efficient for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector already on the
    probability simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically‑stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient (combined)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))
    and g is the gradient of f with respect to the soft‑max logits.
    The parametrisation a = softmax(logits) guarantees a ≥ 0 and ∑a = 1.
    """
    a = softmax_logits(logits)

    # self‑convolution
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # smooth max (log‑sum‑exp)
    exp_weights = np.exp(beta * (conv - maxc))  # centred for stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient w.r.t. a
    w = exp_weights / Z                         # (2n‑1,) weights that sum to 1
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # chain rule back to logits (softmax Jacobian)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 300):
    """
    Anneal the smoothing parameter β while optimising the smooth surrogate.
    Returns the best sequence (still on the simplex) and its exact score.
    """
    logits = np.log(start_a + 1e-30)  # initialise logits safely

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # allocate a per‑step iteration budget respecting the global deadline
        time_left = deadline - time.time()
        max_iter = min(
            maxiter_per_step,
            max(20, int(time_left * 25))  # aim for ≈25 L‑BFGS iters / sec
        )

        res = minimize(
            smooth_obj_and_grad,
            logits,
            args=(beta,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'disp': False}
        )

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f} step {step}/{len(betas)}] new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 400_000,
                     step0: float = 0.09):
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Simple, robust polishing after the smooth phase.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (shift k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape shallow plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Deterministic start – truncated central‑binomial coefficients
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape: a_i = C(2i,i)/4^i, normalised to the simplex."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


def symmetric_power_sequence(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law shape:
        a_i ∝ ((i+0.5)*(n‑0.5‑i))^{‑p/2}
    for 0 ≤ i < n, then normalised to the simplex.
    For p≈0.5 this resembles the classical inverse‑sqrt shape;
    larger p concentrates mass near the ends.
    """
    idx = np.arange(n, dtype=float)
    d = (idx + 0.5) * (n - 0.5 - idx)          # product of distances to the two boundaries
    # avoid zero (never occurs because of the +0.5 offset)
    a = np.power(d, -p / 2.0)
    # numerical safety – replace possible NaNs/Infs produced for extreme p
    a = np.where(np.isfinite(a), a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The procedure consists of:
        • a cheap sweep over a few parametric families,
        • β‑annealed L‑BFGS‑B optimisation of a smooth surrogate,
        • projected sub‑gradient polishing,
        • a greedy mass‑transfer hill‑climb.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # -----------------------------------------------------------------
    # 0. Choose a working length (larger n ⇒ better asymptotic bound)
    # -----------------------------------------------------------------
    time_for_init = deadline - time.time()
    if time_for_init > 800:
        n = 16384
    elif time_for_init > 400:
        n = 8192
    else:
        n = 4096
    print(f"Working with sequence length n = {n}")

    # -----------------------------------------------------------------
    # 1. Assemble a few deterministic seeds
    # -----------------------------------------------------------------
    candidates = []

    # fallback construction if a previous result is present
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.size == n:
            fallback = fallback / fallback.sum()
            candidates.append(('fallback', fallback))
            print("Using provided height_sequence_1 as a seed.")
    except Exception:
        pass

    # classic central‑binomial shape
    candidates.append(('central_binomial', central_binomial_sequence(n)))

    # symmetric power‑law family (explore several exponents)
    for p in (0.2, 0.35, 0.5, 0.65, 0.8):
        a = symmetric_power_sequence(n, p)
        candidates.append((f'p{p:.2f}', a))

    # initialise best
    best_a = None
    best_score = float('inf')
    for name, a in candidates:
        s = compute_score(a)
        print(f"Initial candidate {name:>15} → score {s:.6f}")
        if s < best_score:
            best_score = s
            best_a = a.copy()

    # -----------------------------------------------------------------
    # 2. Smooth optimisation via continuation L‑BFGS‑B
    # -----------------------------------------------------------------
    if time.time() < deadline - 150:
        phase1_deadline = time.time() + max(0.45 * (deadline - start_time), 70)
        print("\n--- Phase 1: smooth L‑BFGS‑B optimisation ---")
        a_smooth, smooth_score = lbfgs_continuation(
            best_a,
            beta_start=1.0,
            beta_end=800.0,
            deadline=phase1_deadline,
            schedule_steps=12,
            maxiter_per_step=300,
        )
        if smooth_score < best_score:
            best_score = smooth_score
            best_a = a_smooth.copy()
            print(f"Best after smooth phase: {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Exact max‑objective polishing (projected sub‑gradient)
    # -----------------------------------------------------------------
    if time.time() < deadline - 60:
        phase2_deadline = deadline - 30
        print("\n--- Phase 2: sub‑gradient polishing ---")
        a_polished, polished_score = subgrad_optimize(
            best_a,
            deadline=phase2_deadline,
            max_iter=600_000,
            step0=0.09,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_a = a_polished.copy()
            print(f"Best after sub‑grad: {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Greedy hill‑climb (small random mass transfers)
    # -----------------------------------------------------------------
    if time.time() < deadline - 5:
        print("\n--- Phase 3: greedy hill‑climb ---")
        cur_a = best_a.copy()
        cur_score = best_score
        best_local_a = cur_a.copy()
        best_local_score = cur_score
        n = cur_a.shape[0]

        # simple heuristic for the number of iterations based on remaining time
        while time.time() < deadline - 0.5:
            cand = cur_a.copy()

            # mass‑transfer move (always preserves the sum)
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 1e-12:
                # transfer a random fraction (up to 10 %) of the mass at i to j
                frac = np.random.rand() * 0.10
                delta = frac * cand[i]
                cand[i] -= delta
                cand[j] += delta

                # occasional swap move (helps explore different permutations)
                if np.random.rand() < 0.07:
                    ii, jj = np.random.randint(n, size=2)
                    cand[ii], cand[jj] = cand[jj], cand[ii]

                # because of floating‑point round‑off we project very rarely
                if (cand < 0).any() or abs(cand.sum() - 1.0) > 1e-12:
                    cand = project_to_simplex(cand)

                cand_score = compute_score(cand)

                if cand_score < cur_score:
                    cur_a, cur_score = cand, cand_score
                    if cand_score < best_local_score:
                        best_local_score = cand_score
                        best_local_a = cand.copy()
                # no uphill acceptance – greedy hill‑climb

        if best_local_score < best_score:
            best_score = best_local_score
            best_a = best_local_a.copy()
            print(f"Hill‑climb improved best to {best_score:.6f}")

    # -----------------------------------------------------------------
    # 5. Final evaluation and return
    # -----------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # external, provided by the harness
    print(f"\nSearch finished. Final reported score = {final_val:.6f}")
    print(f"Length of returned construction = {len(final_seq)}")
    return final_seq
