def simulated_annealing_hill_climb(start_a, deadline,
                                   init_temp=0.01, final_temp=1e-5,
                                   step_scale_start=0.08,
                                   step_scale_max=0.5,
                                   decay_factor=0.99,
                                   recompute_interval=50000):
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_score = cur_score

    step_scale = step_scale_start
    it = 0
    while time.time() < deadline:
        # compute temperature
        t = (deadline - time.time()) / (deadline - start_time_global)  # fraction left
        T = init_temp * (t) + final_temp * (1 - t)   # linear schedule decreasing
        # move selection
        src = np.random.randint(n)
        if a[src] <= 1e-12:
            continue
        dst = np.random.randint(n-1)
        if dst >= src:
            dst += 1
        delta = step_scale * a[src]
        # optional random factor
        # delta = np.random.rand() * step_scale * a[src] ?

        # candidate conv update
        conv_cand = conv.copy()
        conv_cand[src:src+n] -= 2.0 * delta * a
        conv_cand[dst:dst+n] += 2.0 * delta * a
        conv_cand[2*src] += delta * delta
        conv_cand[2*dst] += delta * delta
        conv_cand[src+dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        delta_score = cand_score - cur_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            # acceptance probability
            if T > 0:
                prob = np.exp(-delta_score / T)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            # update state
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                # possibly increase step_scale
                step_scale = min(step_scale * 1.05, step_scale_max)
        else:
            # decay step_scale
            step_scale *= decay_factor

        it += 1
        if it % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())
    return best_a, best_score
```

"""
Advanced step‑function optimizer.

The routine builds a non‑negative sequence a=(a₀,…,aₙ₋₁) with ∑aᵢ=1
and minimises 2·n·max(a⊛a), which is exactly the value returned by
`evaluate_sequence` after the built‑in normalisation checks.

Main ideas
----------
1. Choose a problem size n based on the time budget (larger n gives a
   lower theoretical bound).
2. Generate several deterministic / random starting points:
   • a one‑parameter family a_i(α) ∝ [(i+½)(n‑½‑i)]⁻α      (α‑grid)
   • a two‑parameter “Beta’’ family a_i(p,q) ∝ (i/(n‑1))^p·(1‑i/(n‑1))^q
   • a handful of Dirichlet draws.
   The best among them is taken forward.
3. Long‐run optimisation on the smooth log‑sum‑exp surrogate of the
   max‑convolution, using an exponentiated‑gradient (KL‑mirror) update
   and an aggressive β‑schedule (β→1000).  The iterate is kept on the
   probability simplex.
4. Projected sub‑gradient descent on the exact objective
   (max‑convolution) for further refinement.
5. Simulated‑annealing hill‑climb with O(n) incremental convolution
   updates.  Moves transfer a small packet of mass from a source index
   to a destination; worse moves are sometimes accepted according to a
   temperature schedule, allowing the search to escape shallow local
   minima.

All stages respect the supplied wall‑clock limit (default 1000 s) and
the best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic full convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]⁻α, then normalised."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow/underflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def beta_start(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑like family a_i ∝ (i/(n‑1))^p·(1‑i/(n‑1))^q, normalised."""
    if n == 1:
        return np.array([1.0])
    x = np.linspace(0.0, 1.0, n)
    w = np.power(x, p) * np.power(1.0 - x, q)
    # avoid the exact zeros at the ends when p/q = 0
    w = np.where(w == 0.0, 1e-30, w)
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Grid‑search over α and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([
        np.linspace(-1.0, 0.0, 11),          # centre‑heavy
        np.linspace(0.0, 2.5, 26)            # edge‑heavy
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def find_best_beta_grid(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Grid‑search over (p,q) ∈ [0,5]² (step 0.25) for the Beta‑family."""
    best_a = None
    best_score = float("inf")
    # coarse grid first, then optionally a finer one if time permits
    p_vals = np.linspace(0.0, 5.0, 21)   # 0,0.25,…,5
    q_vals = p_vals
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline:
                return best_a, best_score
            a = beta_start(n, p, q)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score


def random_dirichlet_candidates(n: int, count: int, deadline: float) \
        -> list[tuple[np.ndarray, float]]:
    """
    Generate `count` Dirichlet(1) samples and evaluate their scores.
    Returns a list of (a, score) tuples sorted by score (best first).
    """
    cand = []
    for _ in range(count):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        sc = compute_score(a)
        cand.append((a, sc))
    cand.sort(key=lambda x: x[1])
    return cand


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp optimisation (exponentiated gradient)
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 400_000,
                        beta_start: float = 0.5,
                        beta_end: float = 1000.0,
                        eta0: float = 0.8,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Mirrors KL‑gradient descent on the smoothed objective
        f_β(a) = 1/β·log Σ_k exp(β·(a⊛a)_k)

    β is exponentially annealed from `beta_start` to `beta_end`.
    Returns the best exact score observed.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smooth max via log‑sum‑exp
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))   # stabilised
        w = expw / expw.sum()                 # soft‑max weights (sum to 1)

        # gradient of the smoothed max (cross‑correlation a_rev * w)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # KL‑mirror step
        eta = eta0 / np.sqrt(it)
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for 2·n·max(a⊛a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def simulated_annealing_hill_climb(start_a: np.ndarray,
                                   deadline: float,
                                   init_temp: float = 0.008,
                                   final_temp: float = 1e-7,
                                   step_scale_start: float = 0.08,
                                   step_scale_max: float = 0.5,
                                   inc_factor: float = 1.05,
                                   decay_factor: float = 0.96,
                                   recompute_interval: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Random‑walk polishing where worse moves are occasionally accepted.
    Uses the O(n) incremental convolution update described in the original code.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_score = cur_score

    step_scale = step_scale_start
    total_hc_time = deadline - time.time()  # total time allocated for this phase
    it = 0

    while time.time() < deadline:
        # temperature linearly decreasing with the *remaining* fraction of time
        remaining = max(deadline - time.time(), 0.0)
        frac = remaining / total_hc_time if total_hc_time > 0 else 0.0
        T = final_temp + (init_temp - final_temp) * frac

        # pick a source with positive mass
        src = np.random.randint(n)
        if a[src] <= 1e-15:
            continue
        # pick a different destination uniformly
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # packet size (always ≤ a[src])
        delta = step_scale * a[src]
        # extreme safeguard
        if delta <= 0.0:
            continue

        # incremental convolution update (copy then modify)
        conv_cand = conv.copy()
        # linear part
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic corrections
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max
        delta_score = cand_score - cur_score

        accept = False
        if delta_score < 0.0:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0.0:
                prob = np.exp(-delta_score / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # apply move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_score = cand_score
            # record best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                step_scale = min(step_scale * inc_factor, step_scale_max)
        else:
            # reject – shrink step size gently
            step_scale *= decay_factor

        it += 1
        if it % recompute_interval == 0:
            # full recomputation prevents drift from rounding errors
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows a multi‑stage search and respects the 1000 s wall‑clock limit.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a workable length (larger n → lower theoretical bound)
    # ------------------------------------------------------------------
    remaining_total = deadline - start_time
    if remaining_total > 800:
        n = 16384
    elif remaining_total > 500:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    stage_deadline = deadline - 180.0  # reserve time for later stages
    a_alpha, sc_alpha = find_best_alpha_start(n, stage_deadline)

    a_beta, sc_beta = find_best_beta_grid(n, stage_deadline)
    if a_beta is not None and sc_beta < best_score:

        best_seq = a_beta.tolist()

    if best_seq is not None and time.time() < deadline - 80.0:

    # ------------------------------------------------------------------

    if best_seq is not None and time.time() < deadline - 30.0:

    return best_seq
