    v = np.log(p_start + 1e-12)
    best_p = p_start.copy()
    best_score = compute_score(p_start)
    betas = [1.0, 10.0, 100.0, 1000.0, 10000.0]
    for beta in betas:
        if time.time() >= deadline:
            break
        def func(v):
            # compute p from v
            ex = np.exp(v - np.max(v))  # shift to avoid overflow
            p = ex / ex.sum()
            # compute smooth max
            conv = convolution_fft(p, p)
            maxc = conv.max()
            # stable logsumexp
            expw = np.exp(beta * (conv - maxc))
            sumexp = expw.sum()
            f = maxc + np.log(sumexp) / beta
            return f

        def grad(v):
            ex = np.exp(v - np.max(v))
            p = ex / ex.sum()
            conv = convolution_fft(p, p)
            maxc = conv.max()
            expw = np.exp(beta * (conv - maxc))

            # gradient wrt p
            rev = p[::-1]
            conv_rev = convolution_fft(rev, w)

            # chain rule to v
            dot = np.dot(p, grad_p)
            grad_v = p * (grad_p - dot)

            best_score = sc
            best_p = p.copy()
            print(f"[{time.time():.1f}] L-BFGS beta {beta} improved score {best_score:.6f}")

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(convolution(a,a)) / (sum(a))**2

The algorithm consists of three stages:

1. **Broad parametric scan** – generate many candidates from simple
   families (exponential, power, truncated‑Gaussian, Hann, Kaiser) and
   keep the best exact score.

2. **Smooth‑max L‑BFGS refinement** – minimise a log‑sum‑exp
   approximation of the max‑convolution.  The optimisation variable is
   the unconstrained log‑probabilities `v` (so the simplex constraint is
   satisfied automatically).  A schedule of increasing β values
   (1 → 10 → 100 → 1 000 → 10 000) drives the smooth objective to the
   true max.  Several random restarts are tried.

3. **Greedy stochastic hill‑climb** – a simple random walk that moves a
   small packet of mass between two positions (or swaps two entries)
   and keeps only improving moves.  It runs only for the remaining time.

All stages respect the global 1000 s wall‑clock budget.  The final
sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import i0

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector a that is already normalised
    to sum = 1.  (Same as `evaluate_sequence` for such a vector.)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Parametric families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def gaussian_shape(n: int, sigma_frac: float) -> np.ndarray:
    """truncated Gaussian centred in the middle, σ = sigma_frac·n."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    sigma = max(sigma_frac, 1e-12) * n
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def hann_shape(n: int, alpha: float) -> np.ndarray:
    """Raised‑cosine (Hann) window raised to a power α."""
    i = np.arange(n, dtype=float)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    w = np.power(w, alpha)
    return w / w.sum()


def kaiser_shape(n: int, beta: float) -> np.ndarray:
    """Kaiser window with shape parameter β."""
    i = np.arange(n, dtype=float)
    x = 2.0 * i / (n - 1) - 1.0            # in [-1,1]
    w = i0(beta * np.sqrt(1.0 - x * x))
    w /= i0(beta)
    return w / w.sum()

    best_a = None

# Smooth‑max L‑BFGS optimisation

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max conversion from unconstrained log‑weights."""
    shift = np.max(v)
    ex = np.exp(v - shift)
    return ex / ex.sum()


def smooth_max_value(v: np.ndarray, beta: float, n: int) -> float:
    """Smooth approximation of max(conv(p,p)) where p = softmax(v)."""
    p = _softmax(v)
    conv = convolution_fft(p, p)
    maxc = conv.max()
    # stable log‑sum‑exp based on maxc
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    return maxc + np.log(sumexp) / beta


def smooth_max_grad(v: np.ndarray, beta: float, n: int) -> np.ndarray:
    """Gradient of the smooth‑max w.r.t. the log‑weights v."""
    p = _softmax(v)
    conv = convolution_fft(p, p)
    maxc = conv.max()
    expw = np.exp(beta * (conv - maxc))
    w = expw / expw.sum()                     # soft weights over convolution indices

    # gradient w.r.t. p (the same expression used in the original code)
    rev = p[::-1]
    conv_rev = convolution_fft(rev, w)
    grad_p = 2.0 * conv_rev[n - 1:n - 1 + n]

    # chain rule: ∂f/∂v = (∂f/∂p)·(∂p/∂v) ; the Jacobian of soft‑max is
    # diag(p) - p p^T, so the result simplifies to the expression below.
    dot = np.dot(p, grad_p)
    grad_v = p * (grad_p - dot)
    return grad_v


def lbfgs_refine(start_p: np.ndarray,
                 deadline: float,
                 betas: list[float] | None = None,
                 max_iter_per_beta: int = 200) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by successive L‑BFGS runs on the smooth‑max.
    Returns the best vector (still summing to 1) and its exact score.
    """
    if betas is None:
        betas = [1.0, 10.0, 100.0, 1_000.0, 10_000.0]

    n = start_p.shape[0]
    # initialise the unconstrained variable
    v = np.log(start_p + 1e-15)

    best_a = start_p.copy()
    best_sc = compute_score(best_a)

    for beta in betas:
        if time.time() >= deadline:
            break
        args = (beta, n)
        res = minimize(smooth_max_value,
                       v,
                       args=args,
                       method='L-BFGS-B',
                       jac=smooth_max_grad,
                       options={'maxiter': max_iter_per_beta,
                                'disp': False})
        v = res.x                     # use the solution as start for the next β
        p = _softmax(v)
        sc = compute_score(p)
        if sc < best_sc:
            best_sc = sc
            best_a = p.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:g} => score {best_sc:.6f}")

    return best_a, best_sc

# Greedy stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simple random walk that only accepts improving moves.
    Two move types:
        * swap two entries,
        * transfer a fraction of mass from i to j.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small packet
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97
        steps += 1
    return best_a, best_sc

    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting L‑BFGS refinement")
        start_vec = np.array(best_seq, dtype=float)

        # first run on the pure best vector
        a_opt, sc_opt = lbfgs_refine(start_vec, deadline - 120.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] L‑BFGS first pass score = {best_score:.6f}")

        # a few randomised restarts to escape possible local minima
        for restart in range(2):
            if time.time() >= deadline - 120.0:
                break
            pert = start_vec * (1.0 + 0.02 * np.random.randn(n))
            pert = np.clip(pert, 0.0, None)
            if pert.sum() == 0.0:
                pert = np.full(n, 1.0 / n)
            else:
                pert /= pert.sum()
            a_opt2, sc_opt2 = lbfgs_refine(pert, deadline - 120.0)
            if sc_opt2 < best_score:
                best_score = sc_opt2
                best_seq = a_opt2.tolist()
                print(f"[{time.time():.1f}] restart {restart} improved score = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Final hill‑climb polishing (uses remaining time)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(np.array(best_seq, dtype=float),
                                       deadline,
                                       max_steps=2_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a short uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
