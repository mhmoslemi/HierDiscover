def simulated_annealing(start_a, deadline, max_iters=None, ...):

    n = len(start_a)
    a = start_a.copy()
    conv = convolution_fft(a, a) # compute full conv.

    # compute current score
    current_max = conv.max()
    current_score = 2.0*n*current_max

    best_a = a.copy()
    best_score = current_score

    step_scale = initial_step (maybe 0.1) controlling delta = step_scale * a[src].

    T0 = (current_score - 1) (some) maybe first high.

    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # propose move
        src = np.random.randint(n)
        if a[src] <= 0:
            continue
        # choose move type: transfer or swap
        if np.random.rand() < 0.5:
            delta = step_scale * a[src]
        else:
            # choose a fraction random like uniform(0, step_scale)*a[src]
            delta = np.random.rand()*step_scale*a[src]

        dst = np.random.randint(n-1)
        if dst >= src:
            dst += 1

        # compute candidate conv update
        # conv_cand = conv copy
        conv_cand = conv.copy()
        # shift contributions
        conv_cand[src:src+n] -= 2*delta*a
        conv_cand[dst:dst+n] += 2*delta*a
        # quadratic corrections
        conv_cand[2*src] += delta*delta
        conv_cand[2*dst] += delta*delta
        conv_cand[src+dst] -= 2*delta*delta

        cand_max = conv_cand.max()
        cand_score = 2.0*n*cand_max

        # Acceptance
        delta_score = cand_score - current_score
        # Temperature schedule decays over iteration
        if max_iters is None:
            # reduce gradually
            T = T0 * max(0.01, 1 - iteration/ (deadline - start_time) * 0.999)
        else:
            T = T0 * (0.99 ** (iteration/1000))

        accept = False
        if delta_score < 0:
            accept = True
        else:
            if T > 0:
                if np.random.rand() < np.exp(-delta_score / T):
                    accept = True
        if accept:
            # accept move
            a[src] -= delta
            a[dst] += delta
            # Ensure small negatives are zeroed
            if a[src] < 0:
                a[src] = 0.0
            conv = conv_cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                # Optionally adapt step_scale
                step_scale = min(step_scale * 1.05, 0.5)  # slowly increase step scale after improvement

        else:
            # reject
            # Possibly adjust step scale downward if many rejections
            step_scale *= 0.9995

        # Periodically recompute full conv and reset T0 maybe
        if iteration % 200000 == 0:
            conv = convolution_fft(a, a)
            current_score = 2.0 * n * conv.max()
            # maybe adapt T0
            T0 = max(T0 * 0.9, 1e-12)

        # Possibly print progress every certain iterations.
        if iteration % 50000 == 0:
            print(f"[{time.time():.1f}] SA iter {iteration}, best score {best_score:.6f}, cur {current_score:.6f}, step {step_scale:.4f}")

    return best_a, best_score

"""
Optimisation of a non‑negative step‑function sequence for the
objective 2·n·max(a⊛a) / (∑a)² (the value returned by `evaluate_sequence`).

Strategy
--------
1. Choose a modest length `n` (≈2048‑4096) so that millions of cheap
   incremental moves fit into the 1000 s budget.
2. Initialise the search with several analytic families:
   * power‑law shapes a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
   * Gaussian shapes a_i ∝ exp(−(i−(n‑1)/2)²/(2σ²))
   The best among them (according to an exact FFT evaluation) becomes
   the start point.
3. Refine the start point with a simulated‑annealing (SA) hill‑climber.
   A move transfers a random fraction of the mass from a source index
   to a destination index.  The convolution is updated **incrementally**
   (O(n) vectorised operations) using the exact formula

        Δconv = –2·δ·shift_src(a) + 2·δ·shift_dst(a)
                + δ²·e_{2·src} + δ²·e_{2·dst} – 2·δ²·e_{src+dst} ,

   where `shift_k(a)` is the vector `a` shifted by `k` positions.
   The move is accepted if it reduces the objective or with a
   Metropolis probability `exp(−Δ/T)`.  Temperature `T` decays linearly
   with the remaining wall‑clock time, allowing occasional uphill
   steps early on and pure descent later.
4. If time permits a second SA run starts from a lightly perturbed
   version of the best solution found so far.
5. The best sequence discovered is returned as a plain list of floats.

All heavy work is performed with NumPy (FFT for the occasional full
re‑evaluation, vectorised updates for the incremental steps).  The
algorithm monitors the deadline and stops well before the 1000 s
limit, guaranteeing a valid return value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def exact_score(a: np.ndarray) -> float:
    """
    Exact objective value 2·n·max(a⊛a) for a vector with L1‑norm 1.
    If the vector is not normalised the returned value is still the
    same as ``evaluate_sequence`` (the scaling cancels out).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families for initialisation
# ----------------------------------------------------------------------


def alpha_shape(n: int, alpha: float) -> np.ndarray:
    """
    a_i(α) ∝ [(i+½)(n‑½‑i)]^{‑α}  (symmetric, power‑law shape).
    Returned vector is normalised to have sum = 1.
    """
    i = np.arange(n, dtype=float)
    # work in log‑space to avoid overflow/underflow
    logw = -alpha * (np.log(i + 0.5) + np.log(n - 0.5 - i))
    w = np.exp(logw)
    w[np.isinf(w)] = 0.0
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return w / s


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """
    a_i ∝ exp(−(i−(n‑1)/2)²/(2σ²)), normalised to sum = 1.
    """
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    w = np.exp(-0.5 * (i / sigma) ** 2)
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return w / s


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (incremental convolution updates)
# ----------------------------------------------------------------------


def simulated_annealing_optimize(start_a: np.ndarray,
                                 deadline: float,
                                 recompute_interval: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    * Moves transfer a random fraction of the mass from a source index to a
      destination index.
    * The convolution is updated incrementally (no FFT per move).
    * Temperature decays linearly with the remaining wall‑clock time.
    Returns the best vector seen and its exact score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    # full convolution for the current state
    conv = convolution_fft(a, a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best_a = a.copy()
    best_score = cur_score

    # initialise temperature as a fraction of the current excess over the
    # theoretical lower bound 1 (the best possible objective would be 2,
    # i.e. score 2*0.5 = 1; we use a more pragmatic value)
    T0 = max(1e-6, cur_score * 0.10)

    start_time = time.time()
    total_time = deadline - start_time
    step_scale = 0.12          # typical fraction of source mass to move
    min_step = 1e-5
    max_step = 0.5

    iteration = 0
    accept_cnt = 0
    reject_cnt = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        iteration += 1

        # pick a source with positive mass
        src = np.random.randint(n)
        if a[src] <= 0.0:
            continue

        # pick a different destination
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # random fraction of the source mass – bounded by step_scale
        frac = np.random.rand() * step_scale
        delta = frac * a[src]                # 0 ≤ delta ≤ step_scale·a[src] ≤ a[src]

        # Build a candidate convolution via the exact incremental formula
        conv_cand = conv.copy()
        # linear part: –2·δ·shift_src(a) + 2·δ·shift_dst(a)
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic correction terms
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max
        delta_score = cand_score - cur_score

        # temperature proportional to remaining time
        rem = max(1e-9, deadline - now)
        T = T0 * (rem / total_time)

        # Metropolis acceptance
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if T > 0 and np.random.rand() < np.exp(-delta_score / T):
                accept = True

        if accept:
            # apply the move permanently
            a[src] -= delta
            a[dst] += delta
            # numerical clean‑up
            if a[src] < 0.0:
                a[src] = 0.0
            conv = conv_cand
            cur_score = cand_score
            cur_max = cand_max
            accept_cnt += 1

            # enlarge step size a little after a successful move
            step_scale = min(step_scale * 1.02, max_step)

            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
        else:
            reject_cnt += 1
            # shrink step size slowly after rejections
            step_scale = max(step_scale * 0.995, min_step)

        # occasional full recomputation to avoid accumulated drift
        if iteration % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            # re‑anchor temperature a bit
            T0 = max(T0 * 0.9, 1e-6)

        # light progress reporting
        if iteration % 250_000 == 0:
            print(f"[{time.time():.1f}] SA it {iteration:,} "
                  f"best {best_score:.6f} cur {cur_score:.6f} step {step_scale:.4f}")

    return best_a, best_score
