   a = start_a.copy()
   n = len(a)
   best_a = a.copy()

            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        conv = convolution_fft(a, a)
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))

        # renormalize
        a_sum = a.sum()
        if a_sum == 0:
            # restore uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
   # final return best_a, best_score

def hill_climb_anneal(seq, time_deadline, max_steps=None):
    a = np.array(seq)
    n = a.shape[0]
    # compute best score
    best_a = a.copy()
    best_score = compute_score(best_a)
    cur_a = a.copy()
    cur_score = best_score
    start = time.time()
    T0 = best_score * 0.1   # bigger temperature maybe
    Tmin = best_score * 1e-6
    steps = 0
    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))
        # propose move
        cand = cur_a.copy()
        # Choose move type
        r = np.random.rand()
        if r < 0.3:
            # swap two
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.7:
            # transfer small fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = np.random.rand() * 0.2 * cand[i]  # up to 20% transfer
                cand[i] -= delta
                cand[j] += delta
        else:
            # reallocate mass among three indices
            idx = np.random.choice(n, size=3, replace=False)
            total_mass = cand[idx].sum()
            # draw dirichlet (alpha=1) random splits
            new_split = np.random.rand(3)
            new_split /= new_split.sum()
            cand[idx] = total_mass * new_split
        # ensure nonnegative (clipping)
        cand = np.clip(cand, 0, None)
        # renorm to simplex
        cand_sum = cand.sum()
        if cand_sum == 0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / cand_sum
        cand_score = compute_score(cand)
        # acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T,1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        steps += 1
    return best_a.tolist(), best_score

"""
Improved search for a step‑function that minimises `evaluate_sequence`.

The algorithm consists of four phases:

1. **Noisy exponentiated‑gradient optimisation** on a smooth
   log‑sum‑exp surrogate of the max‑convolution.  This is a mirror‑descent
   (multiplicative‑weights) method on the probability simplex; a small
   Gaussian jitter is added to escape shallow minima.

2. **Upsampling** – the current best vector is split into twice as many
   entries (each half‑mass placed in two adjacent slots).  The extra
   degrees of freedom give the optimiser more flexibility.

3. **Projected sub‑gradient descent** on the exact max‑objective.
   The step size is the classic ηₖ = η₀ / √k and the update is projected
   onto the simplex with the Euclidean projector.

4. **Simulated‑annealing hill climbing** with richer local moves
   (swap, transfer, three‑way re‑allocation) to polish the solution.

All phases respect a global 1000 s wall‑clock budget and continuously
track the best sequence (its exact score is `2·n·max(conv(a,a))`,
identical to the value returned by `evaluate_sequence` when the vector
sums to 1).  The code falls back to a previously discovered construction
if the global variable `height_sequence_1` is present.
"""
import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using a real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the constraint Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators (different shapes)
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start (dense)."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """Smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Pick one of the three generators at random."""
    r = np.random.rand()
    if r < 0.4:
        return random_start_dense(n)
    elif r < 0.7:
        return random_start_sparse(n)
    else:
        return random_start_gaussian(n)


# ----------------------------------------------------------------------
# 1. Noisy exponentiated‑gradient optimisation (smooth surrogate)
# ----------------------------------------------------------------------


def exp_grad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    eta0: float = 0.8,
    noise0: float = 0.05,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative‑weights) optimisation of the
    log‑sum‑exp surrogate of max(conv(a,a)).  A small Gaussian jitter
    with decreasing variance is added to aid exploration.
    Returns the best vector (as a NumPy array) and its exact score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---- annealing schedule for the smoothness parameter β ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smooth max via log‑sum‑exp ----
        conv = convolution_fft(a, a)
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))          # stable weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.shape[0])
        else:
            w /= w_sum

        # ---- gradient of the soft‑max (cross‑correlation trick) ----
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # ---- exponentiated (multiplicative) update ----
        eta = eta0 / math.sqrt(it)
        step = -eta * grad
        # clip to avoid overflow in exp()
        step = np.clip(step, -12.0, 12.0)
        a = a * np.exp(step)

        # ---- add a tiny relative Gaussian jitter (annealed) ----
        noise_scale = noise0 / math.sqrt(it)
        if noise_scale > 0:
            a += a * np.random.randn(n) * noise_scale

        # ---- projection back onto the simplex ----
        a = np.maximum(a, 1e-16)          # protect against exact zeros
        a_sum = a.sum()
        if a_sum == 0.0:
            a.fill(1.0 / n)
        else:
            a /= a_sum

        # ---- occasional exact evaluation ----
        if it % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.04,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).
    The update direction comes from the (unique) maximising offset.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the maximal entry

        # sub‑gradient for this index (see analysis)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # tiny random jitter every 2000 iterations
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq: List[float],
    time_deadline: float,
    max_steps: int = None,
) -> Tuple[List[float], float]:
    """
    Random local modifications (swap, transfer, three‑way re‑allocation)
    with a temperature schedule.  Improving moves are always accepted;
    worsening moves are accepted with probability exp(-Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    # initial temperature as a modest fraction of the current score
    T0 = best_score * 0.05
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        r = np.random.rand()
        if r < 0.35:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.7:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.1 * cand[i] * np.random.rand()   # up to 10 % of source
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        else:
            # three‑way re‑allocation of the total mass on three indices
            idx = np.random.choice(n, size=3, replace=False)
            total_mass = cand[idx].sum()
            # random Dirichlet split (α=1)
            splits = np.random.rand(3)
            splits /= splits.sum()
            cand[idx] = total_mass * splits

        # projection onto simplex (keeps sum = 1, non‑negative)
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        # acceptance test
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score

# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: List[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two adjacent
    entries a_i/2, a_i/2.  The total mass is unchanged.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)          # safety against numerical drift
    return up
