def smooth_max_optim(a, n, deadline):
    gamma = 1.0
    step = 0.1
    # schedule gamma: multiply by factor each epoch
    while time.time() < deadline:
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # increase gamma gradually
        # compute w using scaled conv
        scaled = (conv - maxc) * gamma   # shift for numeric stability
        w = np.exp(scaled - np.max(scaled))
        w /= w.sum()
        # compute gradient
        w_rev = w[::-1]
        conv_full = convolution_fft(a, w_rev)
        slice_mid = conv_full[n-1:2*n-1]
        grad = 2.0 * slice_mid[::-1]  # shape n
        # update
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)
        # compute score
        new_score = compute_score(a_new)
        # accept if improvement else maybe step reduction
        if new_score < compute_score(a):
            a = a_new
            best = a
        else:
            step *= 0.9  # reduce step
        # adapt gamma
        gamma = min(gamma * 1.01, 500)   # increase slowly
        # break if step small
        if step < 1e-9:
            break

def smoothed_gradient_opt(a0, deadline, n):
    a = a0.copy()
    best = a.copy()
    best_sc = compute_score(a)
    # hyperparameters
    step = 0.1
    gamma = 1.0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        # smoothing weight
        maxc = conv.max()
        # scaling to improve numeric stability
        scaled = (conv - maxc) * gamma
        w = np.exp(scaled - np.max(scaled))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/len(w))
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        conv_full = convolution_fft(a, w_rev)
        slice_mid = conv_full[n-1:2*n-1]
        grad = 2.0 * slice_mid[::-1]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)
        # evaluate
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
        # adapt step
        step = max(step * 0.999, 1e-6)  # slowly decay
        # maybe increase gamma
        gamma = min(gamma * 1.001, 500.0)
        # occasional random jitter
        if t % 500 == 0:
            jitter = np.random.randn(n) * step * 0.1
            a += jitter
            a = project_to_simplex(a)
    return best, best_sc
```

def exponentiated_gradient_opt(a0, deadline):
    a = a0.copy()
    best = a.copy()
    best_sc = compute_score(a)
    eta0 = 0.1
    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # find indices where conv close to max
        eps = 1e-12  # value small
        active = np.where(conv >= maxc - 1e-12)[0]  # maybe 1 only
        # compute gradient as average over active
        grad = np.zeros_like(a)
        for k in active:
            # For each i, add 2 * a_{k - i} if in range
            # implement by using vectorized: idx = k - np.arange(n); mask range
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2 * a[idx[mask]]
        grad /= len(active)
        # exponentiated gradient step
        eta = eta0 / np.sqrt(t)   # decaying
        a = a * np.exp(-eta * grad)
        a /= a.sum()
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
            # maybe reduce eta0 when improvement found
        # optional jitter
        if t % 2000 == 0:
            a += np.random.randn(n) * (eta0 * 0.001 / np.sqrt(t))
            a = np.clip(a, 1e-12, None)
            a /= a.sum()
    return best, best_sc

def smoothed_adam_opt(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.1  # initial learning rate (per iteration)
    gamma = 5.0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        # weight vector
        conv_max = conv.max()
        scaled = gamma * (conv - conv_max)
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum
        # gradient
        rev_w = w[::-1]
        conv_full = convolution_fft(a, rev_w)
        g_slice = conv_full[n-1:2*n-1]
        grad = 2.0 * g_slice[::-1]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # schedule updates
        if t % 200 == 0:
            # decay learning rate and increase gamma slowly
            lr *= 0.97
            gamma = min(gamma + 0.5, 500.0)
            # jitter occasional
            jitter = np.random.randn(n) * (lr * 0.1)
            a += jitter
            a = project_to_simplex(a)
    return best_a, best_score

"""
search for a low‑autoconvolution step‑function.

Algorithm
---------
1.  Initialise a probability vector (non‑negative, sum = 1).  Several quick
    hand‑crafted shapes (uniform, sine‑squared, exponential) and a handful
    of random Dirichlet draws are examined – the best one (by the exact
    objective 2·n·max(conv)) is kept.

2.  Smoothed exponentiated‑gradient descent.
    The non‑smooth objective  max_k (a*a)[k] is approximated by a
    log‑sum‑exp proxy:
        f(a) = (1/γ)·log Σ_k exp(γ·conv_k) .
    The gradient of the proxy can be written in closed form using a single
    FFT, and the update a ← a·exp(−η·∇f) is performed, followed by a Euclidean
    projection onto the probability simplex.  The temperature γ is slowly
    increased while the step size η decays – this yields a smooth global
    search that automatically breaks symmetry.

3.  Targeted “max‑reduction’’ moves.
    After the gradient phase the current worst convolution index k\* is
    identified.  The pair (i, j) with i + j = k\* and maximal product
    a_i·a_j is selected and a small amount of probability mass is taken
    away from both and redistributed to two random locations.  Moves are
    accepted only if the exact score improves; otherwise they are reverted.
    This local, problem‑specific hill‑climb removes the remaining
    sub‑optimal spikes in the autocorrelation.

4.  The best sequence found is returned as a plain list of floats.
    All numbers are non‑negative; the evaluation routine normalises the
    vector internally, so the exact scaling of the output does not matter.

The whole search respects the supplied time‑budget (default 1000 s) and
uses only a single CPU.  No external files or network calls are performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution (np.convolve) using a real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the competition:
        2·n·max(conv(a, a))
    The vector a is assumed to sum to 1 (any scaling leaves the
    value unchanged because the expression is homogeneous).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sine‑squared shape a_i = sin²(π·(i+1)/(n+1))."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def exponential_shape(n: int, r: float) -> np.ndarray:
    """Exponential decay a_i ∝ r**i, 0 < r < 1, normalised to sum 1."""
    i = np.arange(n, dtype=float)
    a = r ** i
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------


def exp_gradient_opt(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution using an exponentiated‑gradient
    update and projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    # hyper‑parameters (will be adapted on the fly)
    eta = 0.12          # initial learning rate for the exponential update
    gamma = 5.0         # smoothing parameter for the log‑sum‑exp
    t = 0

    while time.time() < deadline:
        t += 1

        # --- 1. convolution and soft‑max weights ---
        conv = convolution_fft(a, a)                 # length 2n‑1
        c_max = conv.max()
        # log‑sum‑exp with numerical stability
        scaled = gamma * (conv - c_max)
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # --- 2. gradient of the smoothed objective ---
        # grad_i = 2 * Σ_j w_{i+j} * a_j
        rev_w = w[::-1]                               # length 2n‑1
        conv_full = convolution_fft(a, rev_w)         # length 3n‑2
        # slice that corresponds to the required sum
        slice_mid = conv_full[n - 1 : 2 * n - 1]        # n elements
        grad = 2.0 * slice_mid[::-1]                  # reverse to obtain g_i

        # --- 3. exponentiated‑gradient step and projection ---
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        # --- 4. bookkeeping ---
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # adaptive schedules (every few hundred iterations)
        if t % 250 == 0:
            # slowly decay η and increase γ (sharpen the max)
            eta = max(eta * 0.97, 1e-6)
            gamma = min(gamma + 0.6, 200.0)

            # occasional tiny random jitter to escape plateaus
            jitter = np.random.randn(n) * (eta * 0.3)
            a = a * np.exp(jitter)
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 2 – targeted max‑reduction hill‑climb
# ----------------------------------------------------------------------


def targeted_max_reduce_opt(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Identify the current worst convolution index and move a tiny amount of
    mass away from the pair that contributes most to it.  Only improving
    moves are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    # step size expressed as a fraction of the smaller partner's mass
    step_frac = 0.015

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contributions a_i * a_{k-i}
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[j[mask]]

        i0 = int(np.argmax(contrib))
        j0 = k - i0

        # pick two distinct random receivers (different from i0, j0)
        r, s = np.random.choice(n, 2, replace=False)
        while r == i0 or r == j0:
            r = np.random.randint(n)
        while s == i0 or s == j0 or s == r:
            s = np.random.randint(n)

        # amount of probability mass to relocate
        if i0 == j0:
            delta = step_frac * a[i0]
            if delta <= 0:
                continue
            a_candidate = a.copy()
            a_candidate[i0] -= delta
            a_candidate[r] += delta / 2.0
            a_candidate[s] += delta / 2.0
        else:
            delta = step_frac * min(a[i0], a[j0])
            if delta <= 0:
                continue
            a_candidate = a.copy()
            a_candidate[i0] -= delta
            a_candidate[j0] -= delta
            a_candidate[r] += delta / 2.0
            a_candidate[s] += delta / 2.0

        # re‑normalise (projection also eliminates tiny negatives)
        a_candidate = project_to_simplex(a_candidate)

        new_sc = compute_score(a_candidate)
        if new_sc < best_sc:
            # accept the improvement
            best_sc = new_sc
            best_a = a_candidate.copy()
            a = a_candidate
            # slowly increase aggressiveness after a success
            step_frac = min(step_frac * 1.07, 0.05)
        else:
            # reject – become more conservative
            step_frac = max(step_frac * 0.94, 0.001)

    return best_a, best_sc

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The function respects the given time budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"[fallback] score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a manageable length – 2048 works well within 1000 s
    # ------------------------------------------------------------------
    n = 2048
    if budget_s > 800:
        n = 4096
    print(f"[info] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2. Generate a handful of seed vectors (hand‑crafted + random)
    # ------------------------------------------------------------------
    seeds = []

    # a) uniform distribution (the trivial baseline)
    seeds.append(np.full(n, 1.0 / n))

    # b) sine‑squared shape – a classic smooth start
    seeds.append(generate_deterministic(n))

    # c) a few exponential decays (asymmetric, break the symmetry)
    for r in (0.85, 0.90, 0.95):
        seeds.append(exponential_shape(n, r))

    # d) random Dirichlet draws with different concentrations
    for conc in (0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc))

    # e) fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # ------------------------------------------------------------------
    # 3. Evaluate each seed with the exact objective and keep the best.
    # ------------------------------------------------------------------
    for s in seeds:
        s = project_to_simplex(s.astype(float))
        sc = compute_score(s)
        if sc < best_val:
            best_val = sc
            best_seq = s.tolist()
            print(f"[seed] improvement → score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation – Phase 1 (smoothed exponentiated‑gradient)
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    phase1_deadline = start_time + (deadline - start_time) * 0.65
    print("\n[phase 1] smoothed exponentiated‑gradient descent")
    cur_a, cur_score = exp_gradient_opt(cur_a, phase1_deadline)

    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur_a.tolist()
        print(f"[phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Phase 2 – targeted max‑reduction hill‑climb (purely improvement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[phase 2] targeted max‑reduction climb")
        cur_a, cur_score = targeted_max_reduce_opt(cur_a, deadline)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"[phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\n[final] reported score = {final_val:.6f}")
    return final_seq
