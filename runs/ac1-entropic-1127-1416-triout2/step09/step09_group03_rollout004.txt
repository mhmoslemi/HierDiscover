def solve_lp_max_sum_scipy(A, M):
    n = A.shape[1]
    c = -np.ones(n)  # maximize sum -> minimize -sum
    b_ub = np.full(A.shape[0], M, dtype=float)
    bounds = [(0, None)]*n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        return None

"""
Non‑negative step‑function optimizer for evaluate_sequence().
-----------------------------------------------------------

Algorithm outline
-----------------
1. **Start from a central‑binomial (Catalan) tail** – a proven good
   shape for the problem.  The vector is projected onto the probability
   simplex (sum = 1).

2. **Iterative LP boost** – for a moderate size (n≈1024) we solve the
   linear program
       maximise   ∑ b_j
       subject to  (a * b)_k ≤ M   (all k),   b ≥ 0
   where M = max(a * a).  The LP is solved with SciPy’s high‑performance
   “highs” interior‑point method.  The feasible partner b has a
   larger ℓ¹‑norm; after normalising it back to the simplex we convex‑mix
   it with the current a.  Repeating a few times strictly decreases the
   worst convolution entry.

3. **Upscaling** – the refined short vector is linearly interpolated to a
   larger dimension (2048 or 4096 depending on the remaining time) and
   projected again onto the simplex.

4. **Projected sub‑gradient descent** – we minimise the exact objective
   2·n·max(conv(a,a)) using sub‑gradients of the active convolution
   entry.  Step size follows a 1/√t schedule; each iterate is projected
   onto the simplex.

5. **Random‑shape restarts** – exponential‑decay and power‑law shapes are
   generated, each undergoes a short sub‑gradient phase, and any improvement
   is kept.

6. **Final stochastic hill‑climb** – cheap moves (swap or fractional
   mass transfer) with simplex projection polish the best sequence found.

All stages respect the global time budget (≈1000 s).  The function returns
a plain list of non‑negative floats; the harness will call
``evaluate_sequence`` on it.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation of the algorithm by Wang & Carreira‑Perpiñán (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution matrix needed for the LP
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix A such that (A @ b)[k] = conv(f, b)[k].
    Column i contains f shifted down by i rows.
    """
    n = f.shape[0]
    # each column i has f placed starting at row i
    diags = [f] * n
    offsets = np.arange(n)  # 0 … n‑1
    A = sp.diags(diagonals=diags, offsets=offsets,
                 shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum_scipy(A: sp.csr_matrix, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    c = -np.ones(n)                     # minimise -∑b  → maximise ∑b
    b_ub = np.full(A.shape[0], M, dtype=float)
    bounds = [(0, None)] * n

    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                     bounds=bounds, method='highs',
                                     options={'presolve': True, 'time_limit': 20.0})
    except Exception:
        return None

    if not res.success:
        return None
    return np.maximum(res.x, 0.0)        # clip tiny negatives


# ----------------------------------------------------------------------
# Sub‑gradient optimiser on the exact max‑convolution objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for 2·n·max(conv(a,a)).
    Returns the best vector seen and its score.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (swap / fractional transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq,
                           time_deadline,
                           max_steps: int = 400_000) -> tuple[list[float], float]:
    """Tiny stochastic moves with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10   # fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # fractional transfer i → j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (linear interpolation + simplex projection)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)
