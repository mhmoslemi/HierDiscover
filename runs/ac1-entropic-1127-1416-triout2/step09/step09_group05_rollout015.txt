    a = start_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time()>=deadline: break
        conv = convolution_fft(a, a)
        max_idx = int(conv.argmax())
        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        # Using slice
        idx = np.arange(i_start, i_end+1)
        g[idx] = 2 * a[max_idx - idx]
        # step size
        eta = eta0 / np.sqrt(it)
        # exponentiated update
        a = np.maximum(a * np.exp(-eta * g), 1e-15)
        # project to simplex (normalize)
        a = a / a.sum()
        # optional check
        if it % 1000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc; best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_sc: best_sc = sc; best_a = a.copy()
    return best_a, best_sc

    for _ in range(3):

        a_rand = a_rand / a_rand.sum()
        sc = compute_score(a_rand)
        if sc < best_score:
            best_seq = a_rand.tolist()
            best_score = sc

    # Smooth gradient if possible
    if time.time() < deadline - 150:
        a_current = np.array(best_seq)

"""
A compact optimisation routine for the “step‑function” problem.

The algorithm proceeds in several phases that respect a 1000 s wall‑clock budget:

1. **Problem size** – choose a length `n` (larger is better if time permits).
2. **Quick grid search** over several 1‑parameter families  
   (exponential, power‑law, and a discrete Beta distribution) and a few
   random Dirichlet vectors.  The best candidate is kept.
3. **Smooth max‑descent** – minimise a log‑sum‑exp approximation of the
   maximum convolution using an exponentiated‑gradient (KL‑mirror) update.
4. **Exact max‑descent** – work directly with the true maximum (sub‑gradient
   of the max‑convolution) and continue the KL‑mirror updates.
5. **Targeted hill‑climb** – repeatedly move a small packet of mass away
   from the index that contributes most to the current maximal convolution
   entry; moves are accepted only if they improve the exact objective.
6. Return the best sequence seen.  All operations keep the vector in the
   probability simplex (non‑negative, sum = 1), which makes the evaluation
   invariant under scaling and matches the behaviour of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that already
    satisfies the simplex constraint (sum = 1).  It returns
        2·n·max(a*a)
    where the convolution is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ( (i+0.5)/n )^(a‑1) * (1‑(i+0.5)/n)^(b‑1)
    The offset 0.5 prevents division‑by‑zero for a,b<1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()

# ----------------------------------------------------------------------


def scan_exponential_and_power(n: int, deadline: float):
    """Exhaustive scan of exponential and power‑law families."""
    best_a, best_sc = None, float("inf")

    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -2, 200)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    p_grid = np.linspace(0.0, 6.0, 200)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):

    # grids concentrate around 0.2–3.0 (edges vs centre emphasis)

            a_vec = beta_shape(n, a, b)
            sc = compute_score(a_vec)
            if sc < best_sc:

    return best_a, best_sc

def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 3000.0,
                        eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp (smooth max) of the convolution using a KL‑mirror
    step.  β grows exponentially from `beta_start` to `beta_end`.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable soft‑max weights
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smooth max (see derivation in the description)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * grad), 1e-15)
        a /= a.sum()

        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Direct sub‑gradient descent on the true max‑convolution (no smoothing).
    Uses the same KL‑mirror update as `exp_smooth_optimize`.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # build sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * g), 1e-15)
        a /= a.sum()

        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def targeted_hill_climb(start_seq: np.ndarray,
                        deadline: float,
                        max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Greedy walk that always extracts mass from the index that contributes most
    to the current maximal convolution entry and redistributes it elsewhere.
    Accepts only improving moves.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        # contributions to the active sum
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]

        # pick the index with the largest contribution
        i_max = idx[np.argmax(contrib)]

        # pick a random destination (avoid no‑op)
        j = np.random.randint(n)
        if j == i_max:
            continue

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimises `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # -------------------------------------------------
    # Choose a sequence length (larger gives more freedom)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 16384
    elif remaining > 500:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback from a previously discovered vector
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Grid search over simple families
    # -------------------------------------------------
    a_scan, sc_scan = scan_exponential_and_power(n, deadline - 150.0)
    if a_scan is not None and sc_scan < best_score:
        best_score, best_seq = sc_scan, a_scan.tolist()
        print(f"[{time.time():.1f}] after exp/power scan = {best_score:.6f}")

    a_beta, sc_beta = scan_beta_family(n, deadline - 120.0)
    if a_beta is not None and sc_beta < best_score:
        best_score, best_seq = sc_beta, a_beta.tolist()
        print(f"[{time.time():.1f}] after beta scan = {best_score:.6f}")

    # -------------------------------------------------
    # 2. A few random Dirichlet starts (cheap diversification)
    # -------------------------------------------------
    for _ in range(3):
        if time.time() >= deadline - 110.0:
            break
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        sc = compute_score(a_rand)
        if sc < best_score:
            best_score, best_seq = sc, a_rand.tolist()
            print(f"[{time.time():.1f}] random start improved to {best_score:.6f}")

    # -------------------------------------------------
    # 3. Smooth max optimisation (log‑sum‑exp surrogate)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting smooth exponentiated‑gradient")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 50.0,
            max_iters=400_000,
            beta_end=5000.0,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after smooth phase best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Direct max‑descent (sub‑gradient on the true max)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        print(f"[{time.time():.1f}] starting exact max‑descent")
        a_opt2, sc_opt2 = max_descent_optimize(
            np.array(best_seq, dtype=float),
            deadline - 30.0,
            max_iters=300_000,
            eta0=0.5,
        )
        if sc_opt2 < best_score:
            best_score, best_seq = sc_opt2, a_opt2.tolist()
            print(f"[{time.time():.1f}] after max‑descent best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Targeted hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final targeted hill‑climb")
        a_final, sc_final = targeted_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=500_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: tiny uniform vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
