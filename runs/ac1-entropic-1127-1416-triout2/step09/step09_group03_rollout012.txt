"""
Improved non‑negative step‑function optimiser for ``evaluate_sequence``.

The algorithm consists of the following stages:

1. **Initialisation** – a central‑binomial distribution (C(2i,i)/4^i)
   gives a strong starting shape.  Several short random restarts are
   explored with cheap sub‑gradient polishing.

2. **Linear‑programming refinement** – repeatedly solve the LP
   maximise Σb_j  s.t.  conv(a,b) ≤ M (M = max conv(a,a)), using a robust
   GLPK fallback.  The partner vector b is mixed with a (convexly) and
   projected back onto the simplex; each successful mix reduces the
   worst convolution entry.

3. **Upscaling** – the refined short vector is interpolated to a large
   dimension (up to 8192) and again projected onto the simplex.

4. **Smooth Adam optimisation** – minimise a log‑sum‑exp surrogate of
   the max‑convolution with a long, annealed β schedule (β = 5 … 800)
   and a standard Adam step‑size schedule.

5. **Exact sub‑gradient polishing** – projected sub‑gradient descent on
   the true max‑objective (2 n · ‖a*a‖_∞) with a very slow step‑size
   decay.

6. **Simulated annealing hill‑climb** – many tiny random mass‑transfer
   moves are attempted; uphill moves are occasionally accepted with a
   temperature that decays to zero, allowing escape from shallow local
   minima.

All time‑budget checks respect the 1000 s limit; the best sequence found
so far is returned as a plain list of floats (already normalised to sum
≈ 1, which makes ``evaluate_sequence`` invariant to the exact scaling).
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum≈1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Build convolution matrix for the LP (Toeplitz)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Returns the (2n‑1)×n matrix A such that A @ b = conv(f, b).
    Column i of A contains the vector f shifted down by i rows.
    """
    n = f.shape[0]
    diags = [f] * n
    offsets = np.arange(n)
    A = sp.diags(diagonal=diags, offsets=offsets,
                 shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum(A: sp.csr_matrix, M: float,
                     warm_start: np.ndarray | None = None) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Tries a fast LP solver (GLPK) and falls back to ECOS if needed.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # Try GLPK first (linear programming, no warm‑start support)
    try:
        prob.solve(solver=cp.GLPK, warm_start=False, max_iters=20000,
                   glpk={ 'msg_lev': cp.GLPK.MSG_OFF })
    except Exception:
        pass

    # If GLPK failed or gave no solution, fall back to ECOS
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=5000, abstol=1e-7, reltol=1e-6, feastol=1e-7)
        except Exception:
            return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    bv = b.value
    if bv is None:
        return None
    return np.maximum(bv, 0.0)   # clip tiny negatives


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 2_000_000,
    beta_start: float = 5.0,
    beta_end: float = 800.0,
    step0: float = 0.7,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    v1 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing parameter β exponentially
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # convolution a*a
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient = 2 * (rev(a) ∗ w) restricted to n entries
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        v1 = adam_beta2 * v1 + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m1 / (1.0 - adam_beta1 ** it)
        v_hat = v1 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:          # periodic progress check
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final clean‑up
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 800_000,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional small jitter to avoid plateaus
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny random moves)
# ----------------------------------------------------------------------


def simulated_annealing(
    start_a: np.ndarray,
    time_deadline: float,
    init_temp: float = 0.01,
    final_temp: float = 1e-5,
    move_scale: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic mass‑transfer moves with a decreasing temperature.
    Returns the best sequence visited.
    """
    a = start_a.copy()
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    # exponential temperature schedule
    total_time = time_deadline - time.time()
    while time.time() < time_deadline:
        elapsed = time.time() - (time_deadline - total_time)
        frac = elapsed / total_time
        T = init_temp * (final_temp / init_temp) ** frac

        cand = a.copy()
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta = move_scale * cand[i]            # move a fraction of mass from i
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta_score = cand_score - cur_score
        if delta_score < 0 or np.exp(-delta_score / max(T, 1e-12)) > np.random.rand():
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Central‑binomial starter (truncated)
# ----------------------------------------------------------------------


def central_binomial_vector(n: int) -> np.ndarray:
    """Generate the first n central‑binomial coefficients, normalised to sum=1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"[fallback] score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Small‑scale exploration (n≈256) with random restarts + LP
    # -----------------------------------------------------------------
    n_small = 256
    a_small = central_binomial_vector(n_small)

    best_small = a_small.copy()
    best_small_score = compute_score(best_small)
    print(f"[init] small start score = {best_small_score:.6f}")

    # a few cheap random restarts with sub‑gradient polishing
    restart_deadline = deadline - 300.0   # reserve ~5 min for later phases
    for rs in range(4):
        if time.time() >= restart_deadline:
            break
        a_rand = np.random.rand(n_small)
        a_rand = project_to_simplex(a_rand)
        a_opt, sc = subgrad_optimize(a_rand,
                                      time_deadline=min(deadline - 20,
                                                        time.time() + 30))
        if sc < best_small_score:
            best_small_score = sc
            best_small = a_opt
            print(f"[restart {rs}] improved score = {best_small_score:.6f}")

    # ---------------------------------------------------------------
    # 2. LP refinement (robust solver fallback)
    # ---------------------------------------------------------------
    lp_deadline = deadline - 250.0
    t_mix = 0.18
    warm_b = None
    lp_iter = 0
    while time.time() < lp_deadline and lp_iter < 25:
        lp_iter += 1
        conv_f = convolution_fft(best_small, best_small)
        M = float(conv_f.max())
        A = build_convolution_matrix(best_small)
        b_opt = solve_lp_max_sum(A, M, warm_start=warm_b)
        if b_opt is None:
            print(f"[LP {lp_iter}] solver failed – exiting LP loop.")
            break
        warm_b = b_opt
        b_opt = project_to_simplex(b_opt)

        candidate = (1.0 - t_mix) * best_small + t_mix * b_opt
        candidate = project_to_simplex(candidate)
        cand_score = compute_score(candidate)

        if cand_score < best_small_score * 0.9995:
            best_small_score = cand_score
            best_small = candidate
            print(f"[LP {lp_iter}] improved score = {best_small_score:.6f}")
        else:
            t_mix = max(t_mix * 0.85, 0.03)

    print(f"[LP phase] finished after {lp_iter} iterations. "
          f"Best small score = {best_small_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Upscale to a large dimension (as much as remaining time permits)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    # decide final length – larger lengths give finer density approximation
    if remaining > 420:
        n_big = 8192
    elif remaining > 250:
        n_big = 4096
    else:
        n_big = 2048

    a_big = upscale_vector(best_small, n_big)
    print(f"[upscale] using length {n_big}")

    # -----------------------------------------------------------------
    # 4. Smooth Adam optimisation (log‑sum‑exp surrogate)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 150.0
    if time.time() < smooth_deadline:
        a_smooth, smooth_score = smooth_gradient_optimize(
            a_big,
            time_deadline=smooth_deadline,
            max_iters=1_200_000,
            beta_start=5.0,
            beta_end=800.0,
            step0=0.7,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"[smooth] improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Exact sub‑gradient polishing
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 80.0
    if time.time() < subgrad_deadline:
        start_vec = np.array(best_seq if best_seq is not None else a_big,
                             dtype=float)
        refined_a, refined_score = subgrad_optimize(
            start_vec,
            time_deadline=subgrad_deadline,
            max_iter=800_000,
            step0=0.045,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"[subgrad] improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Simulated‑annealing hill‑climb (final polishing)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 4.0
    if time.time() < anneal_deadline and best_seq is not None:
        start_vec = np.array(best_seq, dtype=float)
        polished_a, polished_score = simulated_annealing(
            start_vec,
            time_deadline=anneal_deadline,
            init_temp=0.012,
            final_temp=1e-6,
            move_scale=0.10,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_a.tolist()
            print(f"[anneal] improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – if nothing succeeded, fall back to a modest uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[final] reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
