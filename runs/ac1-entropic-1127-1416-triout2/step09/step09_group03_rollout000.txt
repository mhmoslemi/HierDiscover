def hill_climb(a, deadline):
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    step_scale = 0.05
    n = len(a)
    while time.time() < deadline:
        cand = best.copy()
        # pick two indices i, j
        i, j = np.random.randint(n, size=2)
        # move fraction step_scale * best[i] from i to j
        delta = step_scale * best[i]
        cand[i] -= delta
        cand[j] += delta
        # ensure non-negativity
        cand = np.clip(cand, 0, None)
        # re-project to simplex (normalize)
        cand = cand / cand.sum()
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97
    return best, best_score
```

"""Generate a near‑optimal step function using the continuous arcsine distribution.

The arcsine law (Beta(1/2,1/2)) has the remarkable property that its
self‑convolution is exactly uniform on [0,2].  Discretising the density
and then normalising to a probability simplex yields a vector whose
discrete self‑convolution is almost constant, i.e. the value
`2·n·max(conv(a,a))` is close to the theoretical lower bound `≈1`.  The
routine builds the ar‑sine shape for several lengths, keeps the best,
and optionally performs a very cheap hill‑climb that only accepts
improvements.  The whole search finishes well within the 1000 s budget.
"""

import time
import numpy as np

def arcsine_vector(n: int) -> np.ndarray:
    """Return a length‑n non‑negative vector on the simplex approximating
    the arcsine density f(x)=1/(π√(x(1−x))) on [0,1]."""
    # x in (0,1) avoiding the singular endpoints
    x = (np.arange(n, dtype=float) + 0.5) / n
    # unnormalised density (π factor omitted – cancelled by later normalisation)
    v = 1.0 / np.sqrt(x * (1.0 - x))
    # Normalise to sum = 1 (all entries already ≥ 0)
    v /= v.sum()
    return v

def hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Simple mass‑transfer hill‑climb that only keeps improvements.
    Returns the best vector found and its score."""
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    n = best.shape[0]
    step_scale = 0.10  # start moving 10 % of the source mass

    while time.time() < deadline:
        cand = best.copy()
        i, j = np.random.randint(n, size=2)
        # transfer a fraction of the mass from i to j
        delta = step_scale * cand[i]
        cand[i] -= delta
        cand[j] += delta
        # protect against tiny negatives caused by rounding
        if cand[i] < 0:
            cand[i] = 0.0
        # re‑normalise – this is the Euclidean projection onto the simplex
        cand /= cand.sum()
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97
    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    supplied ``evaluate_sequence`` function.

    The core idea is to start from the discrete arcsine distribution,
    which is provably close to optimal, and then perform a tiny
    hill‑climb that only accepts strictly better candidates.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the surrounding harness
    deadline = start + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # Try a few lengths – larger n reduces discretisation error.
    # The loop stops early if we run out of time.
    for n in (1024, 2048, 4096, 8192):

        a = arcsine_vector(n)
        # evaluate the raw candidate
        score = evaluate_sequence(a.tolist())
        print(f"Arcsine candidate n={n}: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a

    # If we have enough time left, do a cheap hill‑climb starting from the

    if best_seq is not None and time.time() < deadline - 0.5:
        print(f"Starting hill‑climb (n={best_seq.shape[0]})")
