"""
Propose a step‑function sequence that minimises

    2·n·max(convolve(a,a)) / (sum(a)**2)

under the non‑negative simplex.  

The new search routine expands the original approach in several ways:

1. **Richer analytic families** – we add a symmetric power‑law family,
   a mixture of uniform / power‑law / exponential‑U shapes, and extend
   the parameter ranges (α,β up to 30, γ up to 15) which were previously
   limited.

2. **More exhaustive random sampling** – for each tested length we draw
   many random monotone, sparse, U‑shaped and mixed‑family vectors.
   This uncovers promising regions that the grid search may miss.

3. **Longer lengths** – we also consider 4096 and 8192 entries (the
   original search stopped at 2048).  The objective is essentially
   scale‑invariant, so a larger n gives the optimizer more freedom.

4. **Improved refinement** – after the coarse search we perform a
   smoothed‑max optimisation (Adam on the soft‑max parametrisation) with
   a higher β‑schedule and more iterations, followed by a projected
   sub‑gradient descent whose iteration budget is scaled with the
   problem size.

5. **Final local search** – a simulated‑annealing hill‑climb is run
   on the remaining time budget.

All stages respect a global 1000 s wall‑clock budget.  The function
returns a plain Python list ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector that sums to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} , symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} , monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def symmetric_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} + (n‑½‑i)^{‑β} , symmetric high‑edge shape."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5)) + np.exp(-beta * np.log(n - 0.5 - i))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Convex mixture of the uniform vector and the exponential‑U shape.
    lam∈[0,1] weighs the uniform component.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def mixed_three_family(
    n: int, lam_u: float, lam_p: float, beta: float, gamma: float
) -> np.ndarray:
    """
    Convex mixture of three components:
        – uniform (weight lam_u)
        – left‑heavy power law (weight lam_p)
        – exponential‑U shape (remaining weight)
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    power = left_heavy_power_family(n, beta)
    u_shape = exp_u_family(n, gamma)
    remaining = max(0.0, 1.0 - lam_u - lam_p)
    a = lam_u * uniform + lam_p * power + remaining * u_shape
    return a / a.sum()

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform prob. vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Sample a random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 12.0)

    lam_p = np.random.rand() * (1.0 - lam_u)

# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 250_000,
    beta_start: float = 5.0,
    beta_end: float = 400.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimises the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential increase)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        # smooth max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 2n‑1
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact check
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheap per iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 1_000_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for 2·n·max(a*a) on the simplex.
    The step size decays like 1/√t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature based on current best
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]         # swap
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine respects the global time budget
    and returns a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a couple of seconds margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from previously known construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Coarse search over multiple lengths and families
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]   # larger sequences give more freedom

    for n in lengths:
        if time.time() >= deadline - 180.0:
            break
        print(f"[{time.time():.1f}] --- exploring length n = {n} ---")

        # ---- systematic grids for the classic families ----
        # symmetric product (α)
        for alpha in np.linspace(0.0, 30.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        # left‑heavy power law (β)
        for beta in np.linspace(0.0, 30.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

        # symmetric power‑law (both edges) (β)
        for beta in np.linspace(0.0, 30.0, 31):
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑β={beta:.2f}) = {best_score:.6f}")

        # exponential U‑shape (γ)
        for gamma in np.linspace(0.0, 15.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

        # convex mixtures of uniform and U‑shape
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for gamma in np.linspace(0.0, 15.0, 31):
                if time.time() >= deadline - 30.0:
                    break
                vec = mixed_uniform_u_family(n, lam, gamma)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

        # random draws (monotone, sparse, U‑shape)
        for _ in range(120):
            if time.time() >= deadline - 30.0:
                break
            choice = np.random.randint(3)
            if choice == 0:
                vec = random_monotone_start(n)
            elif choice == 1:
                vec = random_sparse_start(n)
            else:
                vec = random_u_start(n)

            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑draw) = {best_score:.6f}")

        # random three‑component mixtures
        for _ in range(120):
            if time.time() >= deadline - 30.0:
                break
            vec = random_mixed_three_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑mix3) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed‑max refinement (Adam) on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting Adam‑based smooth refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        # allocate the bulk of the remaining budget to this phase
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 60.0,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=800.0,
            lr=0.015,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam‑smooth best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient refinement (budget scaled with n)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 50.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_best = np.array(best_seq, dtype=np.float64)
        n = a_best.shape[0]
        # scale iteration budget inversely with n (larger n ⇒ fewer cheap steps)
        max_iters = int(2_000_000 * 2048 / n)
        a_sub, sc_sub = subgrad_optimize(
            a_best,
            deadline - 20.0,
            max_iters=max_iters,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional symmetrisation (keep whichever is better)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb (simulated annealing) on the remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard – if nothing succeeded, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7. Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats.
    return best_seq
