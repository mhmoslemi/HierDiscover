"""Step‑function optimizer.

This version augments the previous approach with a constructive “Sidon‑set”
initialisation.  A large Sidon (B₂) set of indices is placed in the vector,
receives half of the total mass (so each of its entries has weight
α/N with α≈0.5).  The remaining half of the mass is spread uniformly over all
other positions.  Because the Sidon set guarantees that all sums of two
large indices are distinct, the peak of the auto‑convolution is driven down
towards the theoretical lower bound 1/(2·n).  The construction is used as a
starting point for a smoothed‑max exponentiated‑gradient refinement and a final
stochastic hill‑climb.

The routine respects a 1000 s wall‑clock limit and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector normalised to sum = 1.
    (Same as the value returned by `evaluate_sequence` for such a vector.)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simple families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p <= 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


# ----------------------------------------------------------------------
# Sidon set construction
# ----------------------------------------------------------------------


def is_prime(x: int) -> bool:
    """Very small deterministic primality test (x ≤ 2000)."""
    if x < 2:
        return False
    if x % 2 == 0:
        return x == 2
    r = int(math.isqrt(x))
    for d in range(3, r + 1, 2):
        if x % d == 0:
            return False
    return True


def largest_prime_le(limit: int) -> int:
    """Return the largest prime ≤ limit (limit ≥ 2)."""
    for cand in range(limit, 1, -1):
        if is_prime(cand):
            return cand
    return 2


def sidon_set_quadratic(p: int) -> list[int]:
    """
    Classical quadratic Sidon (B₂) construction:
        S = { i·p + (i*i mod p) | i = 0,…,p‑1 }.
    Guarantees all pairwise sums i+j (i ≤ j) are distinct.
    """
    return [i * p + (i * i) % p for i in range(p)]


def build_sidon_vector(n: int, alpha: float = 0.5) -> np.ndarray:
    """
    Build a vector of length n using a Sidon set.
    `alpha` of the total L1 mass is placed uniformly on the Sidon indices,
    the remaining (1‑alpha) is spread uniformly over the other positions.
    """
    # choose p ≈ sqrt(n) (largest prime ≤ sqrt(n))
    p = largest_prime_le(int(math.isqrt(n)))
    sidon = sidon_set_quadratic(p)
    # keep only those inside the range [0, n)
    sidon = [i for i in sidon if i < n]
    N = len(sidon)
    if N == 0:
        raise RuntimeError("Sidon construction failed for n={}".format(n))

    # weight for the large (Sidon) entries
    w = alpha / N
    # weight for the remaining entries
    rest_len = n - N
    s = (1.0 - alpha) / rest_len if rest_len > 0 else 0.0

    a = np.full(n, s, dtype=float)
    a[sidon] = w
    # due to rounding, renormalise to exactly sum‑to‑1
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation of the smoothed max
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 0.5,
    beta_end: float = 3000.0,
    eta0: float = 0.6,
    eps: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed log‑sum‑exp approximation of the max‑convolution
    using a KL‑mirror (exponentiated‑gradient) step.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weights (log‑sum‑exp)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))  # stabilised
        w = expw / expw.sum()

        # gradient of the smoothed objective
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # step size (diminishing)
        eta = eta0 / np.sqrt(it)

        # mirror‑KL update, then project back to simplex
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk:
      – with prob 0.5 swap two entries,
      – otherwise transfer a fraction of mass from one index to another.
    Only moves that improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # initial step magnitude (relative to the source entry)
    step_scale = 0.12

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        # re‑normalise onto the simplex (cheap projection)
        total = cand.sum()
        if total > 0:
            cand /= total
        else:
            cand = np.full_like(cand, 1.0 / n)

        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The pipeline is:
      * quick scan of exponential / power‑law families,
      * a Sidon‑set based construction,
      * exponentiated‑gradient refinement,
      * final hill‑climb polishing.
    All phases respect the 1000 s wall‑clock limit.
    """

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # safety margin

    # -------------------------------------------------
    # Choose a workable length
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_score = float("inf")
    best_seq = None

    # -------------------------------------------------
    # 0. Fallback: uniform vector (guaranteed finite)
    # -------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    sc_uni = compute_score(uniform)
    best_score, best_seq = sc_uni, uniform.tolist()
    print(f"[{time.time():.1f}] uniform baseline score = {sc_uni:.6f}")

    # -------------------------------------------------
    # 1. Scan simple families (coarse grids, cheap)
    # -------------------------------------------------
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -2, 120)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] new best after exponential λ={lam:.3g} → {sc:.6f}")

    p_grid = np.linspace(0.0, 5.0, 120)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] new best after power p={p:.3g} → {sc:.6f}")

    # -------------------------------------------------
    # 2. Sidon‑set based construction (half‑mass on a Sidon set)
    # -------------------------------------------------
    if time.time() < deadline - 30.0:
        try:
            a_sidon = build_sidon_vector(n, alpha=0.5)
            sc_sid = compute_score(a_sidon)
            print(f"[{time.time():.1f}] Sidon candidate score = {sc_sid:.6f}")
            if sc_sid < best_score:
                best_score, best_seq = sc_sid, a_sidon.tolist()
                print(f"[{time.time():.1f}] new best after Sidon → {sc_sid:.6f}")
        except Exception as e:
            print(f"[{time.time():.1f}] Sidon construction failed: {e}")

    # -------------------------------------------------
    # 3. Long exponentiated‑gradient optimisation
    # -------------------------------------------------
    if time.time() < deadline - 120.0 and best_seq is not None:
        print(f"[{time.time():.1f}] starting smooth exponentiated‑gradient")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 80.0,
            max_iters=500_000,
            beta_start=0.5,
            beta_end=5000.0,      # push much closer to the true max
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {sc_opt:.6f}")

    # -------------------------------------------------
    # 4. Final hill‑climb polishing
    # -------------------------------------------------
    if time.time() < deadline - 5.0 and best_seq is not None:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=6_000_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {sc_final:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a short uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
