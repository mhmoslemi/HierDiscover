def lp_step(a, M):
    n = len(a)
    # build sparse matrix A_ub shape (2*n-1, n)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # For each i, g_i coefficient is a_{k-i} (which = a_{k-i}) -> j = k-i
        # So column i corresponds to g_i coefficient a_{k-i}
        # We need to iterate i from i_start..i_end
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(i)
            data.append(a[j])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # set bound vector b = M * ones
    b = np.full(2*n-1, M)
    # Solve LP: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        g0 = res.x
        return g0
    else:
        return None

"""
Propose a step‑function sequence that minimises

    2·n·max(convolve(a,a)) / (sum(a)**2)

using a multi‑stage search:

0.  Load the previous best (fallback) if available.
1.  Scan a few analytic families (including new “quadratic” and
    sinusoidal shapes) on several lengths and keep the best vector.
2.  If the best vector is large, compress it to a moderate size
    (≈1024) – this speeds up the next stage without hurting the bound.
3.  Iteratively improve the vector by solving a linear program:
    given the current a, maximise Σ g_j subject to the linear
    constraints a∗g ≤ M (where M=‖a∗a‖∞).  The solution is normalised
    and mixed into a (t=0.5).  Repeating this 10–15 times drives the
    peak of the auto‑convolution towards its theoretical lower bound.
4.  Refine the result with a differentiable smoothed‑max optimisation
    (Adam on the soft‑max parametrisation) and a cheap projected
    sub‑gradient descent.
5.  Symmetrise the vector (a←(a+a[::-1])/2) and run a final stochastic
    hill‑climb (simulated‑annealing style) for the remaining time.
All stages respect a global 1000 s wall‑clock budget.
The function returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a=1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (∑=1, ≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} , symmetric, normalised to a sum of 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} , monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a


def quadratic_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [i·(n‑1‑i)]^{α}  (zeros at the boundaries), normalised.
    α≥0; α=0 yields uniform.
    """
    i = np.arange(n, dtype=np.float64)
    w = (i) * (n - 1 - i)
    # avoid exact zeros when α=0 (they become ones)
    w = np.where(w > 0, w ** alpha, 0.0)
    # add a tiny epsilon so that we never end up with a completely zero vector
    eps = 1e-12
    w += eps
    a = w / w.sum()
    return a


def sin_power_family(n: int, power: float) -> np.ndarray:
    """
    a_i ∝ sin(π·(i+½)/n)^{power}, a smooth bell‑shaped curve.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    a = w / w.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform prob. vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Sample a random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)

# Smoothed‑max optimisation (Adam on soft‑max parametrisation)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 200.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max parametrisation to minimise a smoothed max‑conv."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (see derivation in original code)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient method for 2·n·max(a*a)."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Long random walk on the simplex with a tiny temperature schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def build_conv_matrix(f: np.ndarray):
    """
    Build the (2n‑1)×n matrix A with entries A[k,i] = f[k‑i] (0 if out of range).
    This matrix realises the linear map g ↦ f∗g.
    """
    n = f.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    # each diagonal d has constant value f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   max Σ g_j   s.t.   f∗g ≤ M  (element‑wise) ,  g ≥ 0.
    Returns the optimal g (as a dense ndarray) or None on failure.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)                 # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)        # maximize Σg ⇔ minimise −Σg
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs')
    if res.success:
        return res.x
    else:
        return None


def lp_iterative_refine(a: np.ndarray,
                        max_iters: int,
                        deadline: float,
                        mix: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above and mix the solution into ``a``.
    ``mix`` is the convex‑combination weight (0→no change, 1→replace).
    Returns the best vector seen and its exact score.
    """
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = convolution_fft(a, a).max()
        g0 = solve_lp_max_sum(a, M)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0                       # renormalise to sum = 1
        # convex mixture
        a = (1.0 - mix) * a + mix * g
        # keep the best
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {it:02d} improved to {best_score:.6f}")

    return best_a, best_score

def compress_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Down‑sample by simple averaging (preserves non‑negativity)."""
    src_n = v.shape[0]
    if src_n == target_n:
        return v.copy()
    factor = src_n // target_n
    if factor < 1:
        # up‑sample by repetition if original is smaller (unlikely)
        reps = int(np.ceil(target_n / src_n))
        v_up = np.tile(v, reps)[:target_n]
        return v_up / v_up.sum()
    # truncate excess elements so that src_n is a multiple of target_n
    v = v[:factor * target_n]
    reshaped = v.reshape(target_n, factor)
    compressed = reshaped.mean(axis=1)
    compressed = compressed / compressed.sum()
    return compressed

        # -- mixed uniform / U ------------------------------------------
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for gamma in np.linspace(0.0, 6.0, 13):
                if time.time() >= deadline - 30.0:
                    break
                vec = mixed_uniform_u_family(n, lam, gamma)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

        # -- quadratic product family (new) -----------------------------
        for alpha in np.linspace(0.0, 6.0, 31):
            if time.time() >= deadline - 30.0:
                break
            vec = quadratic_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (quad‑α={alpha:.2f}) = {best_score:.6f}")

        # -- sinusoidal power family (new) -----------------------------
        for power in np.linspace(0.5, 8.0, 31):
            if time.time() >= deadline - 30.0:
                break
            vec = sin_power_family(n, power)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sin‑p={power:.2f}) = {best_score:.6f}")

        # -- random monotone draws --------------------------------------
        for _ in range(40):
            if time.time() >= deadline - 30.0:
                break
            vec = random_monotone_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

        # -- random U‑shape draws ---------------------------------------
        for _ in range(40):
            if time.time() >= deadline - 30.0:
                break
            vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑U) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Ensure we have a working vector of length target_n
    # ------------------------------------------------------------------
    if best_seq is None:
        # as a safe fallback use a modest uniform vector
        best_seq = [1.0 / target_n] * target_n
        best_score = compute_score(np.array(best_seq, dtype=np.float64))
    else:
        # if the best we have is not of length target_n, compress it
        if len(best_seq) != target_n:
            a_arr = np.array(best_seq, dtype=np.float64)
            a_arr = compress_vector(a_arr, target_n)
            best_seq = a_arr.tolist()
            best_score = compute_score(a_arr)

    # ------------------------------------------------------------------
    # 3. Linear‑programming iterative refinement (core improvement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting LP‑based refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        # use a modest number of LP iterations – each is relatively cheap at n=1024
        a_refined, sc_refined = lp_iterative_refine(
            a_start,
            max_iters=12,
            deadline=deadline - 80.0,
            mix=0.5,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] after LP refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smoothed‑max (Adam) refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting Adam‑smooth refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 50.0,
            max_iters=250_000,
            beta_start=4.0,
            beta_end=400.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam‑smooth best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=2_000_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation test
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return plain Python list as required
    return best_seq
