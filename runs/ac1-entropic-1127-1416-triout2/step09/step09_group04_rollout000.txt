"""
Improved search for a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – generate a random non‑negative vector `f` on the
   probability simplex (∑ f = 1).  Length `n` is fixed (default 1024).

2. **Frank‑Wolfe LP phase** – repeatedly solve the linear program

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k
                    g_i ≥ 0 ,

   where `M = max(f * f)`.  The LP gives a direction `g` that can be added
   without increasing the current worst‑case convolution value.
   Update `f ← (1‑t)·f + t·g` with a modest step size `t`.
   The bound `M` (and therefore the objective `2·n·M`) is monotone
   decreasing; we run as many iterations as the time budget allows.

3. **Optional up‑sampling** – when the LP phase finishes we double the
   number of entries (splitting each mass into two halves).  This gives a
   finer granularity without changing the current bound.

4. **Projected sub‑gradient refinement** – classic sub‑gradient descent
   on the true max‑convolution objective (as in the previous solution).

5. **Stochastic hill‑climb with simulated‑annealing** – final polishing
   using random local moves.  Improving moves are always taken; worsening
   moves are accepted with a temperature that decays to zero.

All phases respect the 1000 s wall‑clock budget.  The best vector found
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the constraint Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Random start generators (different shapes)
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """A smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Pick one of the three generators at random."""
    r = np.random.rand()
    if r < 0.4:
        return random_start_dense(n)
    elif r < 0.7:
        return random_start_sparse(n)
    else:
        return random_start_gaussian(n)

# Frank‑Wolfe LP sub‑routine

def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve:
        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for all k = 0 … 2n‑2
                    g_i ≥ 0
    Returns the optimal g (may be all zeros if the LP fails).
    """
    from scipy.optimize import linprog

    n = f.shape[0]
    # Build A_ub where row k, column j has coefficient f_{k‑j} if valid
    k_idx = np.arange(2 * n - 1)[:, None]   # shape (2n‑1,1)
    j_idx = np.arange(n)                    # shape (n,)
    indices = k_idx - j_idx                 # shape (2n‑1, n)
    mask = (indices >= 0) & (indices < n)

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # vectorised fill
    A[mask] = f[indices[mask]]

    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximise sum => minimise -sum
    bounds = [(0, None)] * n

    # Use HiGHS (default) – fast and reliable for this size
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)
    return g


def frank_wolfe_optimize(
    init_f: np.ndarray,
    deadline: float,
    step_size: float = 0.15,
    max_iters: int = 10_000,
):
    """
    Frank‑Wolfe style optimisation that repeatedly solves the LP
    (see `solve_lp_max_sum`).  Returns the best vector encountered.
    """
    f = init_f.astype(float).copy()
    best_f = f.copy()
    best_score = compute_score(f)

    n = f.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # current bound
        conv = convolution_fft(f, f)
        M = float(conv.max())

        # solve LP for a promising direction
        g0 = solve_lp_max_sum(f, M)
        if g0 is None:
            print("LP failed – aborting Frank‑Wolfe.")
            break
        s = g0.sum()
        if s <= 1e-12:
            # nothing to add
            break
        g = g0 / s                     # renormalise to sum = 1

        # convex combination
        f_new = (1.0 - step_size) * f + step_size * g

        # evaluate
        sc_new = compute_score(f_new)
        if sc_new < best_score:
            best_score = sc_new
            best_f = f_new.copy()
            print(f"  FW iteration {it:4d}: improved score = {best_score:.6f}")

        f = f_new
        # optional: slowly decay step size to improve stability
        if it % 200 == 0:
            step_size = max(0.02, step_size * 0.95)

    final_sc = compute_score(f)
    if final_sc < best_score:
        best_score = final_sc
        best_f = f.copy()
        print(f"  FW final improvement: score = {best_score:.6f}")

    return best_f, best_score

def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # occasional logging
            if t % 5000 == 0:
                print(f"    Subgrad iter {t:6d}: score = {best_score:.6f}")

        a = a_new

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated‑annealing (same as before)
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = None,
) -> tuple[list[float], float]:
    """Random local modifications with temperature‑controlled acceptance."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def upsample_sequence(seq: list[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two adjacent entries a_i/2.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.size, dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects a 1000 s wall‑clock budget
    and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for harness bookkeeping
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Frank‑Wolfe LP optimisation (core phase)
    # ------------------------------------------------------------------
    n = 1024
    print(f"\n--- Frank‑Wolfe LP phase (n={n}) ---")
    # We devote the first ~600 s to this phase, leaving room for polishing.
    fw_deadline = min(deadline - 300.0, start_time + 600.0)

    # start from a random shape
    f0 = random_start(n)

    f_opt, fw_score = frank_wolfe_optimize(f0, fw_deadline, step_size=0.15, max_iters=10_000)
    # store if better
    if fw_score * (2 * n) < best_val:   # convert to same scale as evaluate_sequence
        best_val = 2 * n * (fw_score / (2 * n))  # actually fw_score already equals objective
    if fw_score < best_val:
        best_val = fw_score
        best_seq = f_opt.tolist()
        print(f"After Frank‑Wolfe: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Optional up‑sampling to gain finer granularity
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time() > 180.0):
        print("\nUpsampling best candidate (doubling length)")
        up_seq = upsample_sequence(best_seq)
        up_score = compute_score(up_seq)
        if up_score < best_val:
            best_val = up_score
            best_seq = up_seq.tolist()
            print(f"  Upsampled score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        n_cur = len(best_seq)
        print(f"\nSub‑gradient refinement (n={n_cur})")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 90.0,
            max_iter=400_000,
            step0=0.04,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  Refined score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic hill‑climb polishing (annealed)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nHill‑climb polishing (annealed)")
        polished_seq, polished_score = hill_climb_anneal(
            best_seq,
            time_deadline=deadline,
            max_steps=None,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"  Polished score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Final official evaluation (the harness will call it again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
