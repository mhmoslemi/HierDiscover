"""
Searcher for step‑function heights that minimizes `evaluate_sequence`.

The algorithm:
1. Build a diverse set of seed vectors (hand‑crafted shapes, asymmetric
   beta‑family, random Dirichlet draws) and keep the best few.
2. Simulated‑annealing on the simplex to explore the neighbourhood.
3. Mirror‑descent (exponential‑gradient) refinement that directly targets the
   active maximal convolution entry.
4. Short stochastic hill‑climb polishing.
All phases respect the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """
    Sine‑squared shape a_i = sin²(π·(i+1)/(n+1)), normalised to sum 1.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p .
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_asymmetric_beta(n: int, p: float, q: float) -> np.ndarray:
    """
    Asymmetric polynomial shape a_i ∝ (i+1)**p * (n-i)**q .
    Parameters p and q may be positive or negative.
    """
    i = np.arange(n, dtype=float)
    w = (i + 1.0) ** p * (n - i) ** q
    w[w < 0] = 0.0
    total = w.sum()
    if total == 0.0:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    a = w / total
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Return a random probability vector drawn from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a

def simulated_annealing(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    total_time: float,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk on the simplex.
    Returns the best vector encountered and its exact score.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = max(1e-9, cur_score * 0.04)  # initial temperature
    step_initial = 0.35
    min_step = 5e-6

    smooth_kernel = np.ones(5) / 5.0

    it = 0
    while time.time() < deadline - 2.0:
        it += 1
        elapsed = time.time() - (deadline - total_time)
        frac = elapsed / total_time
        T = max(1e-12, T0 * (1.0 - frac))
        step_scale = max(min_step, step_initial * (1.0 - frac))

        r = np.random.rand()
        if r < 0.65:
            # local mass transfer (to a neighbour)
            i = np.random.randint(n)
            if cur_a[i] <= 0.0:
                continue
            j = (i + 1) % n if np.random.rand() < 0.5 else (i - 1) % n
            delta = step_scale * np.random.rand() * cur_a[i]
        elif r < 0.85:
            # global mass transfer
            i = np.random.randint(n)
            if cur_a[i] <= 0.0:
                continue
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = step_scale * np.random.rand() * cur_a[i]
        else:
            # smoothing move (rare)
            a_smooth = np.convolve(cur_a, smooth_kernel, mode="same")
            s = a_smooth.sum()
            if s == 0.0:
                continue
            a_smooth /= s
            new_score = compute_score(a_smooth)
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T):
                cur_a[:] = a_smooth
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = cur_a.copy()
                continue
            else:
                continue

        # apply the transfer
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = compute_score(cur_a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            cur_a[i] += delta
            cur_a[j] -= delta

        if it % 5000 == 0:
            print(f"[SA] elapsed {elapsed:.1f}s, best {best_score:.6f}")

    return best_a, best_score

def counterpart_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Returns the vector c where c[i] = a[k-i] if 0 ≤ k-i < len(a), else 0.
    """
    n = a.shape[0]
    c = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        c[idx] = a[k - idx]
    return c


def mirror_descent_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 250_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) on the active maximal convolution
    entry.  The update is a_i ← a_i·exp(‑η·g_i) with g_i = 2·a_{k‑i},
    where k is the index achieving the current maximum.
    After each step the vector is re‑normalised onto the simplex.
    """

    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

        if time.time() >= deadline:
            break

        k = int(np.argmax(conv))

        # gradient for the active max entry
        counterpart = counterpart_for_shift(cur_a, k)
        g = 2.0 * counterpart  # non‑negative

        # decreasing step size
        eta = eta0 / np.sqrt(t)

        # exponential update
        cur_a = cur_a * np.exp(-eta * g)

        # renormalise onto simplex (positivity already guaranteed)
        total = cur_a.sum()
        if total == 0.0:
            # rebuild uniform to stay feasible
            cur_a = np.full(n, 1.0 / n)
        else:
            cur_a /= total

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # occasional progress report
        if t % 50000 == 0:
            print(f"[Mirror] iter {t}, best {best_score:.6f}")

    return best_a, best_score

def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,

    """
    Stochastic moves (swap or small transfer) that are kept only when they
    improve the exact score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise for numerical safety
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                       # become more conservative

        steps += 1

"""

`evaluate_sequence` objective.

Algorithm Overview
------------------
1. **Seed Generation** – Create a diverse set of initial vectors:
   * sine‑squared shape,
   * symmetric and *asymmetric* beta families,
   * several Dirichlet‑drawn random vectors,
   * optional fallback from a previously known solution.
   All seeds are normalised to sum = 1 and projected onto the simplex.

2. **Per‑seed Optimisation**
   * **Simulated annealing** – a temperature‑controlled random walk on the
     simplex (≈ 55 % of the time allocated to the seed).  Moves include
     local neighbour transfers, global transfers and occasional smoothing.
   * **Mirror‑descent refinement** – exponential‑gradient updates that
     directly target the active maximal convolution entry (≈ 35 % of the
     per‑seed budget).  After each update the vector is renormalised onto
     the simplex.
   * **Hill‑climb polishing** – a short stochastic local‑search that
     accepts only improving swaps or tiny mass transfers (≈ remaining time).

   The best result over all seeds is kept.

3. **Return** – The best sequence (list of floats) is evaluated with the
   official `evaluate_sequence` and returned.

All phases respect the 1000 s wall‑clock limit; intermediate scores are

# ----------------------------------------------------------------------
# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def generate_deterministic(n: int) -> np.ndarray:
    """
    Sine‑squared shape a_i = sin²(π·(i+1)/(n+1)), normalised to sum = 1.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p .
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_asymmetric_beta(n: int, p: float, q: float) -> np.ndarray:
    """
    Asymmetric polynomial shape a_i ∝ (i+1)**p * (n-i)**q .
    Parameters p and q may be positive or negative.
    """
    i = np.arange(n, dtype=float)
    w = (i + 1.0) ** p * (n - i) ** q
    w[w < 0] = 0.0
    total = w.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    a = w / total
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Return a random probability vector drawn from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a

def simulated_annealing(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    total_time: float,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk on the simplex.
    Returns the best vector encountered and its exact score.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = max(1e-9, cur_score * 0.04)      # initial temperature
    step_initial = 0.35
    min_step = 5e-6

    smooth_kernel = np.ones(5) / 5.0

    it = 0
    while time.time() < deadline - 2.0:
        it += 1
        elapsed = time.time() - (deadline - total_time)
        frac = elapsed / total_time
        T = max(1e-12, T0 * (1.0 - frac))
        step_scale = max(min_step, step_initial * (1.0 - frac))

        r = np.random.rand()
        if r < 0.65:
            # local neighbour transfer
            i = np.random.randint(n)
            if cur_a[i] <= 0.0:
                continue
            j = (i + 1) % n if np.random.rand() < 0.5 else (i - 1) % n
            delta = step_scale * np.random.rand() * cur_a[i]
        elif r < 0.85:
            # global transfer
            i = np.random.randint(n)
            if cur_a[i] <= 0.0:
                continue
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = step_scale * np.random.rand() * cur_a[i]
        else:
            # smoothing (rare)
            a_smooth = np.convolve(cur_a, smooth_kernel, mode="same")
            s = a_smooth.sum()
            if s == 0.0:
                continue
            a_smooth /= s
            new_score = compute_score(a_smooth)
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T):
                cur_a[:] = a_smooth
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = cur_a.copy()
                continue
            else:
                continue

        # apply the transfer
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = compute_score(cur_a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            cur_a[i] += delta
            cur_a[j] -= delta

        if it % 5000 == 0:
            print(f"[SA] elapsed {elapsed:.1f}s, best {best_score:.6f}")

    return best_a, best_score

        best_overall_score = compute_score(best_overall_a)

    # ------------------------------------------------------------------
    # 5. Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq
