"""
Optimisation of a non‑negative step‑function for the ``evaluate_sequence`` objective.

The program performs a multi‑stage search:

1. **Random‑restart smoothed optimisation** – many cheap Adam runs on a
   log‑sum‑exp surrogate of the max‑convolution.  Restarts use a variety of
   initial shapes (dense, sparse, edge‑biased, power‑law) and are confined to
   a user‑specified time slice.

2. **Iterative up‑sampling** – the best candidate is repeatedly doubled in
   length by splitting each entry in half.  This reduces self‑squared terms
   and gives finer granularity, allowing the next stage to improve the score.

3. **Projected sub‑gradient descent** – an exact sub‑gradient method on the
   true max‑convolution objective refines the up‑sampled vector.

4. **Stochastic hill‑climb with simulated‑annealing acceptance** – local
   moves (swap, mass transfer, far‑transfer) are accepted greedily or with a
   temperature‑controlled probability, polishing the solution until the
   global time budget expires.

The whole search respects a wall‑clock budget of 1000 s (a small safety
margin is kept for the harness).  The returned Python list is the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact ``evaluate_sequence`` score under the constraint Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """Smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start_edge(n: int) -> np.ndarray:
    """U‑shaped start – more mass near the ends."""
    # random exponent between 0.5 and 2.0
    p = np.random.uniform(0.5, 2.0)
    i = np.arange(n, dtype=float)
    w = (i + 1) ** (-p) + (n - i) ** (-p)
    a = w / w.sum()
    return a


def random_start_power(n: int) -> np.ndarray:
    """Power‑law shape (flexible)."""
    # random exponent between -2 and 2
    p = np.random.uniform(-2.0, 2.0)
    i = np.arange(n, dtype=float) / (n - 1)
    w = (i + 1e-12) ** p * ((1 - i) + 1e-12) ** p
    # avoid NaNs when p<0
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    a = w / w.sum()
    return a


def random_start_combination(n: int) -> np.ndarray:
    """Pick one of the generators at random."""
    r = np.random.rand()
    if r < 0.25:
        return random_start_dense(n)
    elif r < 0.5:
        return random_start_sparse(n)
    elif r < 0.75:
        return random_start_edge(n)
    else:
        return random_start_power(n)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 1.0,
    beta_end: float = 3000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a differentiable soft‑max surrogate of max(conv(a,a))
    using Adam with simplex projection.  Returns the best exact
    (non‑smoothed) score seen.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # linear (log) annealing of β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # forward convolution
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        # stable soft‑max weights
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of the soft‑max (cross‑correlation trick)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).
    The sub‑gradient is taken from the (any) maximising offset.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the maximal entry

        # sub‑gradient for this index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = None,
) -> tuple[list[float], float]:
    """
    Random local modifications (swap, near‑transfer, far‑transfer) with a
    temperature schedule.  Improving moves are always kept; worsening moves
    are accepted with probability exp(-Δ/T).  Runs until the deadline.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        move_type = np.random.rand()
        if move_type < 0.33:
            # swap two random positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.66:
            # local transfer (5% of source)
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]
                cand[i] -= delta
                cand[j] += delta
        else:
            # far transfer – move a fraction to a far index
            i = np.random.randint(n)
            # pick a far index at distance >= n//4
            offset = np.random.choice([-1, 1]) * np.random.randint(n // 4, n // 2)
            j = (i + offset) % n
            delta = 0.08 * cand[i]   # a slightly larger chunk
            cand[i] -= delta
            cand[j] += delta

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: list[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two
    adjacent entries a_i/2, a_i/2.  The total mass is unchanged.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)           # safety net against numerical drift
    return up
