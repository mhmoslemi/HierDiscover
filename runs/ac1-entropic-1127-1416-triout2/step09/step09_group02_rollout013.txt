Hybrid optimiser for the step‑function evaluation problem.

The algorithm proceeds in four stages:

   yields a direction `b` that can increase the total mass without
   violating the current peak bound.  After scaling `b` to a probability
   vector we either replace `a` by `b` (if this reduces the peak) or
   move towards it by a short convex combination.  The LP is solved
   with `scipy.optimize.linprog` (HiGHS).  This step is very inexpensive
   (a few seconds per iteration) and quickly drives the peak lower.

3. **Projected sub‑gradient descent** – once the LP stops improving,
   we optimise the true objective `2·n·max(conv(a,a))` directly.
   The gradient of the active max–entry is
        g_i = 2·a_{k*‑i}   (zero if the index is out of range).
   After each step we project back onto the probability simplex.

4. **Greedy hill‑climb polishing** – a final stochastic local search
   that only accepts moves improving the exact score.  Small mass
   transfers and swaps are tried repeatedly.

All phases respect the overall wall‑clock budget (default 1000 s) and
the final sequence is returned as a plain Python list of non‑negative
floats.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (fast for the lengths we use).
    Result length = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum == 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_sine_squared(n: int) -> np.ndarray:
    """a_i = sin^2(pi*(i+1)/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the Toeplitz matrix T_f (size (2n‑1)×n) such that
    T_f @ b = conv(f, b) for any vector b of length n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contains f shifted by +j
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(a: np.ndarray, deadline: float, max_outer: int = 15) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based refinement described above.
    Returns the improved vector and its exact score.
    """
    n = a.shape[0]
    a = a.copy()
    best_score = compute_score(a)
    best_M = best_score / (2.0 * n)

    it = 0
    while time.time() < deadline - 1.0 and it < max_outer:
        it += 1
        M = best_M

        # --- build the linear constraints conv(a, b) ≤ M -----------------
        A = build_conv_matrix(a)                 # (2n‑1)×n dense matrix
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # --- solve maximise sum(b)  (i.e. minimise -sum(b)) -------------
        res = scipy.optimize.linprog(
            c=-np.ones(n, dtype=np.float64),   # maximise sum(b)
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={"presolve": True, "time_limit": max(0, deadline - time.time())},
        )
        if not res.success:
            print(f"[LP] iteration {it}: linprog failed ({res.message})")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print(f"[LP] iteration {it}: sum(b) non‑positive, aborting.")
            break

        # Scale to a probability vector
        b_scaled = b / sum_b

        # Try a direct replacement
        new_score = compute_score(b_scaled)
        if new_score < best_score:
            a = b_scaled
            best_score = new_score
            best_M = new_score / (2.0 * n)
            print(f"[LP] iteration {it}: direct replace → score {best_score:.6f}")
            continue

        # Otherwise attempt a short convex combination
        improved = False
        for tau in (0.5, 0.3, 0.1):
            cand = (1.0 - tau) * a + tau * b_scaled
            cand = project_to_simplex(cand)      # keep on simplex
            cand_score = compute_score(cand)
            if cand_score < best_score:
                a = cand
                best_score = cand_score
                best_M = cand_score / (2.0 * n)
                improved = True
                print(f"[LP] iteration {it}: mix τ={tau:.2f} → score {best_score:.6f}")
                break
        if not improved:
            print(f"[LP] iteration {it}: no improvement, stop.")
            break

    return a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient refinement (same as before, slightly tuned)
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 800_000,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on `2·n·max(conv(a,a))`.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        # step size with a slower decay (helps fine‑tuning)
        step = step0 / (1.0 + 0.001 * t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # occasional tiny jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002)
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb polishing (same as before)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic improvement that only keeps moves decreasing the exact score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a tiny fraction
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise (round‑off may drift)
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                       # become more conservative

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # small safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0. Optional fallback sequence (if the global variable exists)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"[init] fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a length (2048 works well; enlarge if we have a lot of time)
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 800.0:
        n = 4096
    print(f"[init] using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Generate a variety of seed vectors
    # ------------------------------------------------------------------
    seeds = []

    # deterministic shapes
    seeds.append(generate_sine_squared(n))
    for p in np.linspace(-2.5, 2.5, 7):
        seeds.append(generate_beta_family(n, p))

    # random Dirichlet draws
    for _ in range(5):
        seeds.append(random_dirichlet(n, conc=1.0))
    for _ in range(5):
        seeds.append(random_dirichlet(n, conc=0.3))

    # include fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # pick the best seed (exact score)
    for s in seeds:
        s = project_to_simplex(s.astype(float))
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[init] new best seed score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Prepare the current vector
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = project_to_simplex(cur_a)   # safeguard
    cur_score = compute_score(cur_a)
    print(f"[init] initial exact score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Linear‑programming refinement (the strongest driver)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        lp_deadline = time.time() + remaining * 0.45   # use ~45 % of the rest
        print("\n[Phase 1] Linear‑programming refinement")
        cur_a, cur_score = lp_refine(cur_a, lp_deadline)
        best_val = cur_score
        print(f"[Phase 1] best score after LP = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient refinement (fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline - 40.0:
        print("\n[Phase 2] Sub‑gradient refinement")
        sub_deadline = deadline - 30.0
        cur_a, cur_score = subgradient_refine(cur_a, cur_score, sub_deadline)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[Phase 2] improved score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 3] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[Phase 3] final polished score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Return the result
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\n[done] final reported score = {final_val:.6f}")
    return final_seq
