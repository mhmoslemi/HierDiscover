"""
Propose candidate sequence to minimise ``evaluate_sequence``.

Algorithm (≈ 1000 s budget)

1. Choose a problem size `n` (larger `n` gives a tighter theoretical bound,
   but we keep the FFT size manageable).
2. Build a small pool of initial simplex vectors:
     • the classic inverse‑square‑root (central‑binomial) shape,
     • a few random monotone‑decreasing vectors generated from power‑law or
       Dirichlet samples.
3. For each start run a *continuation* of L‑BFGS‑B on a smooth log‑sum‑exp
   surrogate of the max‑convolution.  The smoothing parameter β is
   increased geometrically from 5 up to 2000 (more steps than before) and
   the optimiser is given a generous per‑step iteration budget.
4. Keep the best simplex vector obtained so far.
5. Polish the best candidate with a projected sub‑gradient descent on the
   exact objective `2·n·max(conv(a,a))`.  The step size decays as
   `step0/√t` and we add occasional jitter to escape plateaus.
6. A tiny stochastic hill‑climb (swap / fractional transfer) uses the
   remaining time to squeeze out a final improvement.
7. The final list of floats (still on the simplex) is returned; the outer
   harness will call `evaluate_sequence` on it.

All helper routines are top‑level, there are no closures or lambdas,
and the code respects the 1 CPU and 1000 s limits.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (efficient for large arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector already on the
    probability simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically‑stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient (combined)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where

        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))

    and g is the gradient of f with respect to the soft‑max logits.
    The parametrisation a = softmax(logits) guarantees a ≥ 0 and ∑a = 1.
    """
    a = softmax_logits(logits)

    # 1) self‑convolution
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # 2) smooth max using log‑sum‑exp
    exp_weights = np.exp(beta * (conv - maxc))  # centred for numerical stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # 3) gradient w.r.t. a
    w = exp_weights / Z                         # (2n‑1,) probabilities
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # 4) chain rule to logits
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 300):
    """
    Anneal the smoothing parameter β while optimising the smooth surrogate
    with L‑BFGS‑B.  Returns the best simplex vector found and its exact score.
    """
    logits = np.log(start_a + 1e-20)  # initialise logits from simplex point
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # budget for this β – try to use as many iterations as time allows
        time_left = deadline - time.time()
        max_iter = min(
            maxiter_per_step,
            max(30, int(time_left * 20))  # roughly 20 iterations per second
        )

        # Run L‑BFGS‑B on the smooth surrogate
        res = minimize(
            fun=smooth_obj_and_grad,
            x0=logits,
            args=(beta,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'disp': False}
        )
        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.1f} (step {step}/{schedule_steps})] "
                  f"new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 500_000,
                     step0: float = 0.07):
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Simple but robust polishing after the smooth phase.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape shallow plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Deterministic start – truncated central‑binomial coefficients
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root shape: a_i = C(2i,i)/4^i, normalised to the simplex."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Miscellaneous helpers for random starts
# ----------------------------------------------------------------------


def random_powerlaw_simplex(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), projected onto the simplex."""
    x = np.arange(1, n + 1, dtype=float) ** (-p)
    x = x / x.sum()
    return project_to_simplex(x)


def random_monotone_simplex(n: int) -> np.ndarray:
    """Generate a random decreasing vector and project onto simplex."""
    v = np.random.rand(n)
    a = np.sort(v)[::-1]          # descending order (edge‑heavy)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The algorithm uses multiple random restarts, a β‑annealed L‑BFGS‑B
    optimisation of a smooth surrogate, a sub‑gradient polishing phase and,
    if time permits, a tiny stochastic hill‑climb.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 1. Choose a working length (larger ⇒ tighter theoretical bound)
    # -----------------------------------------------------------------
    time_left_initial = deadline - time.time()
    if time_left_initial > 800:
        n = 32768
    elif time_left_initial > 500:
        n = 16384
    else:
        n = 8192
    print(f"Chosen sequence length n = {n}")

    # -----------------------------------------------------------------
    # 2. Build a pool of initial simplex vectors
    # -----------------------------------------------------------------
    initial_vectors = []

    # deterministic central‑binomial start
    initial_vectors.append(central_binomial_sequence(n))

    # a few random power‑law shapes with varying exponents
    for p in [0.55, 0.65, 0.75]:
        if time.time() >= deadline:
            break
        initial_vectors.append(random_powerlaw_simplex(n, p))

    # a few purely random monotone vectors
    for _ in range(3):
        if time.time() >= deadline:
            break
        initial_vectors.append(random_monotone_simplex(n))

    # -----------------------------------------------------------------
    # 3. Main optimisation loop (multiple restarts)
    # -----------------------------------------------------------------
    for start_idx, a0 in enumerate(initial_vectors, 1):
        if time.time() >= deadline:
            break

        a0 = project_to_simplex(a0)  # just to be safe
        cur_start_score = compute_score(a0)
        print(f"\nRestart {start_idx}/{len(initial_vectors)} – "
              f"initial score = {cur_start_score:.6f}")

        # ---- 3a. Smooth optimisation via L‑BFGS‑B with β‑continuation ----
        beta_start = 5.0
        beta_end = 2000.0
        a_smooth, smooth_score = lbfgs_continuation(
            start_a=a0,
            beta_start=beta_start,
            beta_end=beta_end,
            deadline=deadline - 150.0,          # reserve time for polishing
            schedule_steps=12,
            maxiter_per_step=300
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"  After smooth phase: new global best = {best_val:.6f}")

        # ---- 3b. Exact max‑objective polishing (sub‑gradient) ----
        if time.time() < deadline - 70.0:
            print("  Sub‑gradient polishing …")
            polished_a, polished_score = subgrad_optimize(
                init_a=a_smooth,
                deadline=deadline - 60.0,
                max_iter=500_000,
                step0=0.07
            )
            if polished_score < best_val:
                best_val = polished_score
                best_seq = polished_a.tolist()
                print(f"  Polishing improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Final stochastic hill‑climb (very cheap) if time permits
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nFinal hill‑climb …")
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(best_a)
        step_scale = 0.08

        while time.time() < deadline and step_scale > 1e-5:
            cand = best_a.copy()
            if np.random.rand() < 0.5:        # simple swap
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]
            else:                             # fractional transfer
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = step_scale * best_a[i]
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0.0:
                        cand[i] = 0.0
            cand = project_to_simplex(cand)
            cur_score = compute_score(cand)

            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                # when we improve we allow the step size to grow a bit
                step_scale = min(step_scale * 1.07, 0.5)
            else:
                step_scale *= 0.96

        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.tolist()
            print(f"Hill‑climb final score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Fallback in the extremely unlikely case that nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report final score with the official evaluator
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
