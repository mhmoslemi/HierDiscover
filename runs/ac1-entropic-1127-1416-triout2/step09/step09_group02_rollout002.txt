"""
Soft‑max‑smoothed gradient descent on the max‑convolution objective.

We seek a non‑negative vector a (∑a = 1) that minimises
    F(a) = 2·n·max(conv(a,a))
where conv(a,a) is the linear convolution of a with itself.
The max can be approximated by a smooth log‑sum‑exp (soft‑max) which
yields a dense gradient that can be followed by projected gradient steps.
The algorithm proceeds in three stages:

1. **Multi‑start seeding** – a collection of hand‑crafted shapes
   (sine‑power, symmetric beta‑polynomials, Gaussians) and random Dirichlet draws.
   The best seed (lowest exact score) is retained.

2. **Soft‑max gradient descent** – for the chosen seed we repeatedly:
        * evaluate conv(a,a)
        * form weights w_k ∝ exp(λ·conv_k) (λ grows from 1 to ≈70)
        * compute the sub‑gradient g = 2·conv(w, a[::-1]) on the active window
        * take a step a ← a – η·g with a diminishing stepsize η_t = η₀/√t
        * project back onto the probability simplex.
   The best iterate seen during the run is kept.

3. **Hill‑climb polishing** – a fast greedy local search that only accepts
   improving mass‑transfer or swap moves, again respecting the simplex.

The whole search respects the wall‑clock budget (≈1000 s) and finally returns
a plain list of floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def sine_power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p(π·(i+1)/(n+1))."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def beta_shape(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def gaussian_shape(n: int, sigma_factor: float) -> np.ndarray:
    """a_i ∝ exp(-0.5((i‑center)/σ)^2), σ = sigma_factor·n."""
    sigma = sigma_factor * n
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a_start: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.2,
    lambda0: float = 1.0,
    lambda_max: float = 70.0,
) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent on 2·n·max(conv(a,a)).
    Uses a log‑sum‑exp approximation of the max with a temperature
    parameter λ that grows linearly from λ₀ to λ_max.
    Returns the best vector encountered and its exact score.
    """
    a = a_start.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # convolution of a with itself
        conv = convolution_fft(a, a)                      # length 2n‑1
        max_c = conv.max()

        # progressive soft‑max temperature
        lam = lambda0 + (lambda_max - lambda0) * (t / max_iter)

        # soft‑max weights (stable by subtracting max_c)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient: g_i = 2·∑_k w_k·a_{k‑i}
        rev_a = a[::-1]
        grad_full = convolution_fft(w, rev_a)            # length 2n‑1
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]        # slice length n

        # diminishing step size
        eta = step0 / np.sqrt(t)

        a = a - eta * grad
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # occasional progress report
        if t % 5000 == 0:
            print(f"[Softmax] iter {t:>6d}, best {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic local search that only accepts moves which improve
    the exact objective.  Small mass transfers and swaps are tried.
    """
    a = a_start.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.05
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                # numerical safety
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                         # become more conservative

        steps += 1

    return best, best_score

        for _ in range(3):
            seeds.append(random_dirichlet(n, conc))

    # e) reuse fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # ------------------------------------------------------------------
    # 2. Evaluate seeds on the exact objective, keep the best one
    # ------------------------------------------------------------------
    for s in seeds:
        s = project_to_simplex(s.astype(float))
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[Seed] improved score = {best_val:.6f}")

    # Prepare the current best vector
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------------------------------------------------------
    # 3. Soft‑max gradient descent (major part of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + total_time * 0.70
    print("\n[Phase 1] Soft‑max gradient descent")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a,
        phase1_deadline,
        max_iter=600_000,
        step0=0.25,
        lambda0=1.0,
        lambda_max=80.0,
    )
    best_val = cur_score
    print(f"[Phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final hill‑climb polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 2] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=2_000_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[Phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the final sequence
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported value = {final_val:.6f}")
    return final_seq
