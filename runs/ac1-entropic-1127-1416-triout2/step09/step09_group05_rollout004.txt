def hill_climb_sa(start_seq, deadline, max_steps=2_000_000, init_temp=1e-3):
   a = start_seq.copy()
   best_a = a.copy()
   best_sc = compute_score(best_a)
   temp = init_temp
   step_scale = 0.1
   step = 0
   while step < max_steps and time.time()<deadline:
       # generate candidate
       cand = best_a.copy()
       if np.random.rand() < 0.5:  # swap
           i, j = np.random.randint(n, size=2)
           cand[i], cand[j] = cand[j], cand[i]
       else:
           i, j = np.random.randint(n, size=2)
           if i!=j:
               delta = step_scale * best_a[i]
               cand[i] -= delta
               cand[j] += delta
       cand = project_to_simplex(cand)
       sc = compute_score(cand)
       if sc < best_sc:
           best_sc = sc
           best_a = cand
           # possibly increase step_scale
           step_scale = min(step_scale * 1.05, 0.5)
       else:
           # accept with probability
           accept_prob = np.exp(-(sc - best_sc)/ (temp + 1e-12))
           if np.random.rand() < accept_prob:
               best_a = cand
               best_sc = sc
           # reduce step_scale slowly
           step_scale *= 0.999

       # decay temperature
       temp *= 0.99999
       step += 1

def upsample_vector(a, target_n):
    if len(a) == target_n:
        return a
    old_x = np.linspace(0, 1, len(a))
    new_x = np.linspace(0, 1, target_n)
    new_a = np.interp(new_x, old_x, a)
    # ensure nonneg and normalize
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()
    return new_a
```

"""
Propose a low‑autocorrelation non‑negative step function.

Algorithm
---------
1. Choose a sequence length `n` that fits the remaining wall‑clock budget
   (larger `n` gives a finer discretisation of the continuous problem).
2. Scan two simple one‑parameter families (exponential‑decay and
   power‑law) on a fine grid and keep the best vector.
3. Optionally up‑sample the best vector to the chosen length (linear
   interpolation) and renormalise it to a probability distribution.
4. Perform a long exponentiated‑gradient optimisation of the smooth
   log‑sum‑exp approximation of the max‑convolution.  The smoothing
   parameter β is annealed from 0.5 to a very large value (default 5 000)
   so that the objective approaches the true max.  The KL‑mirror update
   (multiplicative weights) automatically respects the simplex constraint.
5. Finish with a stochastic hill‑climb that uses simulated‑annealing style
   acceptance; this can escape shallow local minima while still keeping
   only improving moves most of the time.
6. Return the best sequence found (plain Python list of floats).

All steps respect the 1000 s wall‑clock limit; intermediate scores are
printed for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector normalised to sum = 1.
    Matches the value returned by `evaluate_sequence` for such a vector.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation based on the algorithm of Wang & Carreira‑Perpinán (2013)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def upsample_vector(a: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation to a larger length, then renormalise to sum = 1."""
    if a.shape[0] == target_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, a.shape[0])
    new_x = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()
    return new_a


# ----------------------------------------------------------------------
# Parameterised families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p <= 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


def mixed_shape(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp(-lam * i**gamma), normalised to sum = 1.
    Gives a richer family that interpolates between exponential (γ=1)
    and faster‑decaying shapes (γ>1) or slower (γ<1).
    """
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * (i ** gamma))
    return w / w.sum()


# ----------------------------------------------------------------------
# Simple grid scans
# ----------------------------------------------------------------------


def scan_exponential_and_power(n: int, deadline: float):
    """Exhaustive scan of exponential and power‑law families."""
    best_a = None
    best_sc = float("inf")

    # exponential λ grid (including λ=0 → uniform)
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -2, 300)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # power‑law p grid
    p_grid = np.linspace(0.0, 6.0, 300)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # mixed (lam,gamma) grid – very cheap (few points)
    lam_grid2 = np.logspace(-4, -1, 80)
    gamma_grid = np.linspace(0.5, 2.0, 30)
    for lam in lam_grid2:
        for gamma in gamma_grid:
            if time.time() >= deadline:
                break
            a = mixed_shape(n, lam, gamma)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (smooth max of convolution)
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 600_000,
    beta_start: float = 0.5,
    beta_end: float = 5000.0,
    eta0: float = 0.5,
    eps: float = 1e-16,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (KL‑projection) on the log‑sum‑exp smooth approximation
    of the max‑convolution.  β is increased exponentially from `beta_start`
    to `beta_end`.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential in iteration fraction)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weights (log‑sum‑exp)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stabilised exponentials
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smooth max (convolution of a_rev with w)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # step size (standard 1/√t decay)
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
    init_temp: float = 1e-3,
    temp_decay: float = 0.99995,
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk with occasional uphill moves (simulated annealing).
    Moves:
      * swap two entries,
      * transfer a fraction of mass from one index to another.
    The current candidate is always projected back onto the simplex.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    temp = init_temp
    step_scale = 0.12
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        # choose move type
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # simulated‑annealing acceptance
            prob = np.exp(-(sc - best_sc) / (temp + 1e-18))
            if np.random.rand() < prob:
                best_a = cand
                best_sc = sc
                step_scale = max(step_scale * 0.97, 0.02)  # small step when uphill
            else:
                step_scale *= 0.97

        temp *= temp_decay
        steps += 1

    return best_a, best_sc
