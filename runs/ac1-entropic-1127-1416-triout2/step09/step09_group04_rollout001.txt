"""
A fast LP‑based refinement of a step‑function.

The evaluation function is scale‑invariant, therefore we keep all
candidates on the probability simplex (non‑negative, sum = 1).  The
algorithm proceeds as follows:

1. Initialise the candidate.  If the global variable
   ``height_sequence_1`` (the best construction found by a previous run)
   exists we start from it, otherwise we fall back to a random start.

2. Repeatedly solve a linear programme that, for the current vector f,
   finds the largest possible extra mass b (b ≥ 0) such that the
   convolution f * b never exceeds the current peak M = max(f * f).
   Adding this mass and re‑normalising never increases the bound M,
   and typically reduces the objective.

   The LP is
        maximise   Σ b_i
        subject to conv(f,b) ≤ M  (element‑wise)
                  b_i ≥ 0.

   For a vector of length n the constraint matrix is a (2n‑1) × n
   Toeplitz‑like matrix; it is built once per iteration using a sparse
   representation and solved with SciPy’s HiGHS interior‑point solver.

3. When the LP stops improving (or the time budget is low) we optionally
   up‑sample the vector (replace each entry a_i by a_i/2, a_i/2 in two
   adjacent slots) – this spreads the mass and often reduces the peak
   of the self‑convolution.

4. Finally a very cheap stochastic hill–climb with simulated‑annealing
   acceptance polishes the result.

All steps respect the 1000 s wall‑clock budget and the routine returns
the best list of heights found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT version is much faster than plain np.convolve)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return convolution_fft(a, a).max()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective under the normalization Σ a_i = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    return 2.0 * n * float(max_convolution(a))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (needed after random moves)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σ x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Build the convolution constraint matrix for the LP (sparse)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray):
    """
    Build a sparse (2n‑1) × n matrix A such that
        (A @ b)[k] = (f * b)[k]   for k = 0 … 2n‑2.
    Column j contains the vector f shifted down by j rows.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = lil_matrix((m, n), dtype=float)
    # each column j has f in rows j … j+n‑1
    for j in range(n):
        rows = np.arange(j, j + n)
        A[rows, j] = f
    return A.tocsr()


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def solve_lp_for_b(f: np.ndarray, M: float):
    """
    Solve
        maximise   Σ b_i
        subject to conv(f,b) ≤ M
                  b_i ≥ 0
    Returns the optimal b (numpy array) or None on failure.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)               # minimise -Σb → maximise Σb
    bounds = [(0, None)] * n

    # HiGHS is the default method for SciPy ≥1.6; it handles sparse A_ub.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def lp_refine(f: np.ndarray, deadline: float, max_iters: int = 12):
    """
    Repeatedly enlarge the current vector by solving the LP above and
    re‑normalising.  Stops when the time budget is exhausted or when no
    further mass can be added.
    Returns the best vector and its exact score.
    """
    best_f = f.copy()
    best_score = compute_score(best_f)
    iteration = 0

    while time.time() < deadline and iteration < max_iters:
        M = max_convolution(f)
        b = solve_lp_for_b(f, M)
        if b is None:
            print("  LP failed – terminating refinement.")
            break
        sum_b = b.sum()
        if sum_b < 1e-12:               # nothing useful to add
            break

        # convex combination (f + b) / (1 + sum_b) keeps Σ = 1
        f_new = (f + b) / (1.0 + sum_b)

        score_new = compute_score(f_new)
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
            print(f"  LP iteration {iteration+1}: improved score = {best_score:.6f}")

        f = f_new
        iteration += 1

    return best_f, best_score


# ----------------------------------------------------------------------
# Upsampling helper (splits each entry into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq):
    """Double the length by replacing a_i with a_i/2, a_i/2."""
    a = np.asarray(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    # enforce simplex (tiny numerical drift)
    up = project_to_simplex(up)
    return up


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(seq, deadline, max_steps=None):
    """Local random modifications; improves exact score."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    # initial temperature modest fraction of the current score
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]      # move 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The routine respects a ~1000 s wall‑clock budget and returns the best
    list of heights it discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Global fallback (previous best construction, if available)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]     # type: ignore
        fallback_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_score:.6f}")
        if fallback_score < best_score:
            best_score = fallback_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. If no fallback, start from a simple random vector (length 1024)
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 1024
        a0 = np.random.rand(n0)
        a0 /= a0.sum()
        best_seq = a0.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"Random start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Convert to numpy array on the simplex (sum = 1)
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)
    cur = cur / cur.sum()          # normalise (scale invariant)

    # ------------------------------------------------------------------
    # 3. LP‑based refinement (may be repeated after up‑sampling)
    # ------------------------------------------------------------------
    phase = 1
    while time.time() < deadline - 10.0:
        print(f"\n--- LP refinement phase {phase} (len={len(cur)}) ---")
        cur, phase_score = lp_refine(cur, deadline, max_iters=12)
        eval_score = evaluate_sequence(cur.tolist())
        print(f"Phase {phase} evaluated score = {eval_score:.6f}")

        if eval_score < best_score:
            best_score = eval_score
            best_seq = cur.tolist()
            print(f"  New overall best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 4. Optional up‑sampling to give the optimiser finer granularity.
        #    Only performed once when enough time remains.
        # --------------------------------------------------------------
        if time.time() < deadline - 30.0:
            up = upsample_sequence(cur)
            up_score = evaluate_sequence(up.tolist())
            print(f"   Upsampled candidate score = {up_score:.6f}")
            if up_score < best_score:
                best_score = up_score
                best_seq = up.tolist()
                cur = up.copy()
                print("   Upsampled version accepted.")
        else:
            break

        phase += 1
        if phase > 4:      # a few phases are enough for the time budget
            break

    # ------------------------------------------------------------------
    # 5. Final stochastic polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\n--- Final hill‑climb polishing ---")
        polished_seq, polished_score = hill_climb_anneal(best_seq, deadline)
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq
            print(f"Polished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Log final result (the harness will call `evaluate_sequence` again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
