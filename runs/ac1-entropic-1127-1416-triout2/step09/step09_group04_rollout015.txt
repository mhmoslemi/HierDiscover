"""
enhanced step‑function optimizer
--------------------------------
A multi‑stage optimiser that works with the same objective as the
reference solution but adds a few new ingredients aimed at breaking the
~1.511 barrier:

* **Two‑level up‑sampling** – start with a cheap 512‑node search, then
  double the length twice (up to 4096) so that the algorithm has far more
  degrees of freedom to “smooth out’’ the convolution peak.

* **More aggressive soft‑max annealing** – the smooth surrogate used in the
  Adam phase runs with a final temperature β≈3000 (instead of 1500) and a
  longer schedule, which gives a tighter approximation to the true
  max‑operator.

* **Diverse random starts** – besides uniform, sparse and Gaussian starts
  we also use a sinusoidal start; this widens the explored region of the
  feasible simplex.

* **Longer sub‑gradient polishing** – after each up‑sampling we run a
  classic sub‑gradient method for many more iterations with a smaller
  initial step‑size.

* **Extended simulated‑annealing hill‑climber** – the final polishing
  stage now also allows three‑point mass‑transfer moves, which helps to
  escape shallow basins that the Adam/sub‑gradient phases cannot see.

All phases respect the global 1000 s wall‑clock budget and keep the
best vector seen (as a plain Python list) for the harness.
"""

import time
import numpy as np
import math
import sys
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the normalization Σ a_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators (different shapes)
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Put mass on roughly √n random positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """Smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start_sine(n: int) -> np.ndarray:
    """
    A half‑sine shape (non‑negative, integral = 1).  Provides a
    smooth, non‑uniform profile that is known to reduce the central
    correlation peak.
    """
    x = np.arange(n)
    a = np.sin(np.pi * (x + 0.5) / n)  # shift to avoid zeros at the edges
    a[a < 0] = 0.0
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Select a generator uniformly at random."""
    r = np.random.rand()
    if r < 0.25:
        return random_start_dense(n)
    elif r < 0.5:
        return random_start_sparse(n)
    elif r < 0.75:
        return random_start_gaussian(n)
    else:
        return random_start_sine(n)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 250_000,
    beta_start: float = 5.0,
    beta_end: float = 3000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> Tuple[np.ndarray, float]:
    """
    Minimise a differentiable soft‑max surrogate of max(conv(a,a))
    using Adam momentum and a simplex projection.
    Returns the best exact (non‑smoothed) score observed.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # linear annealing of β on a log‑scale
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of the soft‑max (cross‑correlation trick)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.03,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).  The update direction
    is the (unique) sub‑gradient given by the maximising offset.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the maximal entry

        # sub‑gradient for this index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to keep away from stagnation
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.001 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq: List[float],
    time_deadline: float,
    max_steps: int = None,
) -> Tuple[List[float], float]:
    """
    Random local modifications (swap, pair transfer or three‑point
    redistribution) with a temperature schedule.  Improves the exact
    objective; always accepts an improving move and occasionally a
    worsening one.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    # start temperature as a small fraction of the current best score
    T0 = max(best_score * 0.025, 1e-3)
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        move_type = np.random.rand()
        if move_type < 0.4:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.8:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.04 * cand[i]          # 4 % of source mass
                cand[i] -= delta
                cand[j] += delta
        else:
            # three‑point redistribution (i→j, j→k)
            i, j, k = np.random.choice(n, size=3, replace=False)
            delta = 0.03 * cand[i]
            cand[i] -= delta
            cand[j] += delta * 0.6
            cand[k] += delta * 0.4

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: List[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two adjacent
    entries a_i/2, a_i/2.  Total mass is unchanged.
    """
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)               # safety net
    return up

            max_iters=300_000,
            beta_start=5.0,
            beta_end=3000.0,
            step0=1.0,
        )
        scr_fine_exact = compute_score(a_fine)
        if scr_fine_exact < best_val:
            best_val = scr_fine_exact
            best_seq = a_fine.tolist()
            report(f"  After fine Adam: score = {best_val:.6f}")

        # another sub‑gradient pass (shorter, tighter step)
        report("Final sub‑gradient pass (4096)")
        a_final_sub, scr_final_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 40.0,
            max_iter=400_000,
            step0=0.02,
        )
        if scr_final_sub < best_val:
            best_val = scr_final_sub
            best_seq = a_final_sub.tolist()
            report(f"  After final sub‑grad: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic hill‑climb polishing (annealed)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        report("\nHill‑climb polishing (annealed)")
        polished_seq, polished_score = hill_climb_anneal(
            best_seq, time_deadline=deadline, max_steps=None
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            report(f"  Polished score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback to simple uniform vector if everything else failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Final official evaluation (the harness calls `evaluate_sequence`
    #    again, but we log it here for debugging).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    report(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
