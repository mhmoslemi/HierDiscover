"""
Propose a step‑function coefficient vector that minimises

    2·n·max(convolve(a, a)) / (sum(a)**2)

under the non‑negative simplex constraint.

Improvements over the reference implementation
------------------------------------------------
*   Added an **asymmetric power‑law** family  a_i ∝ (i+½)^‑α·(n‑½‑i)^‑β,
    giving a richer set of monotone shapes.
*   Added a **combined exponential‑power** family  a_i ∝ exp(‑γ·i)·(i+1)^‑β.
*   Performed a *coarse‑to‑fine* grid search for the new families and
    refined the best parameters locally.
*   Adam‑based smoothed‑max optimisation now runs with a larger β‑schedule
    (β_end up to 2000) and a larger iteration budget (subject to the time
    limit).  A few short Adam restarts from perturbed copies of the current
    best vector are also performed.
*   Sub‑gradient descent receives a slightly larger initial step‑size and
    is allowed to run longer when time permits.
*   Hill‑climbing (simulated annealing) is given a longer horizon and a
    slower temperature decay.
*   All stages share a single global deadline (default 1000 s) and respect
    the remaining budget aggressively.

The driver returns a plain Python list of floats (sum≈1).  The evaluation
harness will call `evaluate_sequence` on the returned list.
"""

import time
import numpy as np
from scipy import fftpack   # used only for the FFT‑based convolution

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return vectors with sum≈1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} (U‑shape, symmetric)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}  (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of the uniform vector and the exponential U‑shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    a = a / a.sum()
    return a


def asymmetric_power_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Asymmetric power‑law:
        a_i ∝ (i+½)^{-α}·(n‑½‑i)^{-β},
    which gives a wide variety of skewed monotone shapes.
    """
    i = np.arange(n, dtype=np.float64)
    left = np.log(i + 0.5)
    right = np.log(n - 0.5 - i)
    logw = -alpha * left - beta * right
    w = np.exp(logw)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def combined_exp_power_family(n: int, beta: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp(-γ·i)·(i+1)^{-β}.
    The exponential term sharpens the left‑hand tail, the power term controls
    the overall decay.
    """
    i = np.arange(n, dtype=np.float64)
    logw = -gamma * i - beta * np.log(i + 1.0)
    w = np.exp(logw)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Sample a random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 4.0,
                         beta_end: float = 800.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing β is increased exponentially from beta_start to beta_end.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)                # initialise soft‑max parameters

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- schedule for β (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a and its convolution -----
        a = softmax(z)
        conv = convolution_fft(a, a)            # length 2n‑1
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                # shape (2n-1,)

        # ----- gradient wrt a (see original code) -----
        # conv_rev = a reversed * w   (via convolution)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)    # length 2n‑1
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 1000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final exact evaluation
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# Projected sub‑gradient descent (cheap per iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method on 2·n·max(a*a) with diminishing step 1/√t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs i+j=k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex.
    – 0.5 prob: swap two entries.
    – 0.5 prob: transfer a fraction of mass from one index to another.
    A decreasing temperature permits occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature proportional to current best
    T0 = max(best_score * 1e-4, 1e-9)
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

                if sc < best_score:

                    print(f"[{time.time():.1f}] new best (asym α={a_left:.2f},β={a_right:.2f}) = {best_score:.6f}")

        # ----- combined exponential‑power family (β,γ) -----
        for beta in np.linspace(0.0, 12.0, 7):
            for gamma in np.linspace(0.0, 8.0, 5):
                if time.time() >= deadline - 30.0:
                    break
                vec = combined_exp_power_family(n, beta, gamma)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best (exp‑pow β={beta:.2f},γ={gamma:.2f}) = {best_score:.6f}")

        # ----- random starts (monotone, sparse, U) -----
        for _ in range(60):
            if time.time() >= deadline - 30.0:
                break
            vec = random_monotone_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

        for _ in range(40):
            if time.time() >= deadline - 30.0:
                break
            vec = random_sparse_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑sparse) = {best_score:.6f}")

        for _ in range(60):
            if time.time() >= deadline - 30.0:
                break
            vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑U) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Local refinement of the asymmetric family around the best parameters
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        # try a quick local refinement for the asymmetric family
        try:
            n_best = len(best_seq)
            # recover approximate parameters by fitting (simple heuristic)
            # we just run a few rounds of Adam on the current best
            a_start = np.array(best_seq, dtype=np.float64)
            a_opt, sc_opt = adam_smooth_optimize(
                a_start,
                deadline - 60.0,
                max_iters=150_000,
                beta_start=5.0,
                beta_end=1200.0,
                lr=0.015,
            )
            if sc_opt < best_score:
                best_score, best_seq = sc_opt, a_opt.tolist()
                print(f"[{time.time():.1f}] after focused Adam best = {best_score:.6f}")
        except Exception as e:
            print(f"[{time.time():.1f}] asymmetric refinement failed: {e}")

    # ------------------------------------------------------------------
    # 3. Global smoothed‑max Adam refinement (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting global Adam‑smooth refinement")
        a_start = np.array(best_seq, dtype=np.float64)

        # main long run
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 90.0,
            max_iters=250_000,
            beta_start=4.0,
            beta_end=1500.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after main Adam best = {best_score:.6f}")

        # a few short restarts from perturbed copies
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline - 30.0:
                break
            noise = np.random.normal(scale=0.02, size=a_opt.shape)
            a_pert = a_opt + noise
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = project_to_simplex(a_pert)
            a_tmp, sc_tmp = adam_smooth_optimize(
                a_pert,
                deadline - 20.0,
                max_iters=80_000,
                beta_start=5.0,
                beta_end=800.0,
                lr=0.016,
            )
            if sc_tmp < best_score:
                best_score, best_seq = sc_tmp, a_tmp.tolist()
                print(f"[{time.time():.1f}] after restart {r+1} Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient refinement (longer if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 50.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            max_iters=2_500_000,
            step0=0.08,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation test (many optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score, best_seq = sc_sym, a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final hill‑climb (simulated annealing) with a generous budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) phase")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_500_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 8. Final official evaluation (for logging)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
