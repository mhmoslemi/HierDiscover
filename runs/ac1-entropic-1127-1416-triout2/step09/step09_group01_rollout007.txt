    i = np.arange(n)
    mu = (n-1)/2
    w = np.exp(-((i - mu)**2)/(2 * sigma**2))
    w[w == 0] = 1e-16  # maintain positivity
    a = w / w.sum()
    return a

def beta_family(n, alpha, beta):
    i = np.arange(n)
    t = i/(n-1)
    w = np.power(t, alpha) * np.power(1 - t, beta)
    # avoid all zeros:
    w[w == 0] = 1e-16
    a = w / w.sum()
    return a

def semicircle_family(n, p=1.0):
    i = np.arange(n)
    mu = (n-1)/2
    radius = (n-1)/2
    x = (i - mu) / radius
    w = np.sqrt(np.maximum(0.0, 1 - x**2))
    # raise to power p
    w = np.power(w, p)
    a = w / w.sum()
    return a

The function `propose_candidate` searches for a non‑negative step‑function (a probability vector) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}\qquad
\bigl(a*a \text{ = aperiodic convolution of }a\text{ with itself}\bigr)
\]

Because the objective is scale‑invariant we restrict ourselves to the
probability simplex `∑a_i = 1`.  
The search consists of several well‑balanced phases:

1. **Initialisation** – a large set of analytically‑parametrised families
   (symmetric product, power‑law, exponential, Gaussian, beta‑distribution,
   semi‑ellipse) together with a few random draws.  The best of these
   supplies a solid starting point.

2. **Smooth optimisation (Adam on a log‑sum‑exp surrogate)** – we minimise  

   \[
   f_\beta(a)=\frac1\beta\log\!\Bigl(\sum_k\exp\bigl(\beta\,(a*a)_k\bigr)\Bigr)
   \]

   by Adam on the soft‑max parametrisation `a = softmax(z)`.  
   The smoothing parameter `β` is increased exponentially from `5` up to `5000`,
   which drives the surrogate arbitrarily close to the true max‑convolution.
   Several short restarts are performed while time permits.

3. **Mirror (entropy) sub‑gradient descent on the true objective** – a cheap
   multiplicative update `a ← a·exp(-η g)` where `g` is the sub‑gradient of the
   active convolution entry.  The stepsize `η_t = η₀ / t^α` with
   `α≈0.6` keeps the updates aggressive for a long time.  We run this phase
   for the bulk of the remaining budget, storing the best iterate.

4. **Symmetrisation** – many optimal solutions are symmetric; averaging a
   vector with its reverse often lowers the maximal convolution entry.

5. **Fine‑grained stochastic hill‑climb (simulated‑annealing)** – a last
   stochastic local search with a tiny temperature that can escape shallow
   plateaus.

All phases respect the global wall‑clock deadline.  The best sequence found

"""Propose a step‑function sequence that (approximately) minimises
   evaluate_sequence(sequence).

   The algorithm uses several stages:
   1) a wide variety of analytic families and random starts,
   2) smooth Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution,
   3) long mirror (entropy) sub‑gradient descent on the true max‑convolution,
   4) optional symmetrisation,
   5) a final stochastic hill‑climb with a tiny simulated‑annealing component.

   All work is constrained by an overall wall‑clock budget (default 1000 s).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on the grid, normalised to a probability vector."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = w / w.sum()
    return a


def beta_family(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution shape on the grid, normalised."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    # add a tiny epsilon to avoid exact zero at the boundaries
    eps = 1e-12
    w = (t + eps) ** a * (1.0 - t + eps) ** b
    a_vec = w / w.sum()
    return a_vec


def semicircle_family(n: int, power: float = 1.0) -> np.ndarray:
    """Upper half of a circle, optionally raised to a power."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    radius = (n - 1) / 2.0
    x = (i - mu) / radius
    w = np.sqrt(np.maximum(0.0, 1.0 - x ** 2))
    w = w ** power
    a = w / w.sum()
    return a


def triangular_family(n: int) -> np.ndarray:
    """Symmetric triangular distribution (linear increase then decrease)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.where(i <= mid, i + 1, n - i)
    a = w / w.sum()
    return a

# Random initialisers

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)          # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                       # decreasing
    return a / a.sum()


def perturb_around(a: np.ndarray, scale: float = 0.02) -> np.ndarray:
    """Add zero‑mean Gaussian noise (clipped) and renormalise."""
    b = a + scale * np.random.randn(*a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0.0:
        b = a.copy()
    return b / b.sum()

def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 5_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max layer a = softmax(z).
    The smoothing parameter β is increased exponentially.
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothed weights w_k ∝ exp(β·conv_k)
        conv = convolution_fft(a, a)                 # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                      # soft‑max over the convolution

        # gradient of f_β w.r.t. a (see derivation in the original code)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        # extract the part that corresponds to original length n
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    eta0: float = 1.2,
    power: float = 0.6,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror sub‑gradient descent on the true max‑objective.
    Update rule:  a ← a·exp(−η·g)  with η = η0 / t^power.
    g is the sub‑gradient of the maximal convolution entry.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # index of the active max

        # sub‑gradient: g_i = 2·a_{k−i} if (k−i) in [0,n) else 0
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / (t ** power)

        # exponentiated (multiplicative) update – stays non‑negative
        a = a * np.exp(-eta * g)

        # renormalise to the simplex (entropy projection)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing hill‑climb on the simplex.
    - With 0.5 probability a random swap is performed,
    - Otherwise a small mass transfer between two indices occurs.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – start tiny, linear cool
    T0 = best_score * init_temp_factor
    T = T0

    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                 # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        # renormalise (exact simplex projection)
        cand = cand / cand.sum()

        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)   # enlarge moves a bit
        else:
            step_scale *= 0.97                               # shrink after rejection

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Symmetrisation – many optimal solutions are symmetric
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()

    # ------------------------------------------------------------------

    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard – if everything failed fall back to a modest uniform vector

    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats, as required by the harness
    return best_seq
