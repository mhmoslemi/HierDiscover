def propose_candidate(seed=..., budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - small_margin

    n = 2048  # may increase later depending on remaining time.

    # Step 1: generate candidate seeds.
    seeds = [] # list of np arrays.

    # a) uniform distribution: all 1/n
    seeds.append(np.full(n, 1.0/n))

    # b) central binomial truncated to n.
    a_central = np.empty(n)
    a_central[0]=1.0
    for i in range(1,n):
        a_central[i]=a_central[i-1]*(2*i-1)/(2.0*i)
    a_central = project_to_simplex(a_central)
    seeds.append(a_central)

    # c) symmetric beta distributions.
    x = np.linspace(0,1,n)
    for alpha in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]:
        beta = alpha
        pdf = x**(alpha-1) * (1 - x)**(beta-1)
        pdf = pdf / pdf.sum()
        seeds.append(pdf)

    # d) symmetric exponential shape: a_i ∝ exp(-c * |i - (n-1)/2|)
    for c in [0.001, 0.002,0.004,0.008,0.016]:
        i = np.arange(n)
        pdf = np.exp(-c * np.abs(i - (n-1)/2))
        pdf = pdf / pdf.sum()
        seeds.append(pdf)

    # e) symmetric Gaussian
    for sigma in [n/8, n/6, n/4]:
        i = np.arange(n)
        pdf = np.exp(-0.5 * ((i - (n-1)/2)/sigma)**2)
        pdf = pdf / pdf.sum()
        seeds.append(pdf)

    # f) random Dirichlet seeds: generate maybe 5 random seeds.
    for _ in range(5):
        rand = np.random.rand(n)
        pdf = rand / rand.sum()
        seeds.append(pdf)

    # Remove duplicates etc.

    # Step 2: For each seed, run smoothing optimizer for given budget per seed.
    best_seq = None
    best_score = float('inf')
    t_now = time.time()
    remaining = deadline - t_now
    # Determine per-seed budget: allocate equal slices maybe.
    n_seeds = len(seeds)
    # We'll iterate seeds sequentially with a max iterations per seed but break early if time low.

    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        # allocate time per seed: maybe (remaining / (n_seeds - idx)) * 0.8
        # we can allow each seed up to a fixed number of iterations (e.g., 200k) but check time.
        seed_deadline = min(deadline, time.time() + (remaining / (n_seeds - idx)) * 0.9)
        # Run smooth optimizer.
        a_opt, score_opt = smooth_gradient_optimize(seed_vec, time_deadline=seed_deadline,

                                                    step0=1.0,
                                                    adam_beta1=0.9,
                                                    adam_beta2=0.999,
                                                    adam_eps=1e-8)
        # Compute exact score.
        cur_score = compute_score(a_opt)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.tolist()
            print(f"Seed {idx} improved best to {best_score:.6f}")
        # also try subgrad from a_opt
        sub_deadline = min(deadline, time.time() + 30)  # allocate small.
        if time.time() < sub_deadline:
            refined_a, refined_score = subgrad_optimize(a_opt, time_deadline=sub_deadline,
                                                       max_iter=200_000,
                                                       step0=0.06)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_a.tolist()
                print(f"Subgrad after seed {idx} improved best to {best_score:.6f}")

        # update remaining time
        remaining = deadline - time.time()

    # Step 3: final hill climb if time remains
    if best_seq is not None and time.time() < deadline - 5:
        hill_deadline = deadline - 5
        polished_seq, polished_score = hill_climb_permutation(best_seq,

        if polished_score < best_score:
            best_score = polished_score

            print(f"Hill climb final improved best to {best_score:.6f}")

    # fallback uniform if best_seq still None.
    if best_seq is None:
        best_seq = [1.0/n for _ in range(n)]

    print(f"Final best score {best_score:.6f}")
    return best_seq

def build_conv_matrix(f):
    n = len(f)
    # first column: f padded with zeros to length 2n-1
    col = np.concatenate([f, np.zeros(n-1)])
    # first row: f[0] followed by zeros of length n-1
    row = np.concatenate([f[0:1], np.zeros(n-1)])
    A = sp.linalg.toeplitz(col, row)  # shape (2n-1, n)
    return A

"""Search for sequence minimizing evaluate_sequence using LP refinement,
smooth Adam optimisation and final hill‑climb.  The algorithm builds a
small “seed” with a linear‑program partner vector (fixed convolution
matrix), upscales it, then performs many gradient‑descent steps on a
log‑sum‑exp surrogate of the max‑convolution, followed by a sub‑gradient
polish and stochastic local search.  All steps respect the 1000 s
budget."""
import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp
import scipy.linalg as sla

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (fast for our sizes).
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution matrix (Toeplitz) for the LP
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix A such that A @ b = conv(f, b).
    """
    n = f.shape[0]
    # first column: f followed by n‑1 zeros
    col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    # first row: f[0] followed by n‑1 zeros
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=float)])
    A_dense = sla.toeplitz(col, row)
    return sp.csr_matrix(A_dense)


def solve_lp_max_sum(A: sp.csr_matrix, M: float,
                     warm_start: np.ndarray | None = None) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns the optimal b as a numpy array, or None on failure.
    """
    n = A.shape[1]
    b = cp.Variable(n)
    constraints = [A @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    if warm_start is not None:
        b.value = warm_start

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-7, reltol=1e-6, feastol=1e-7)
    except Exception:
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    bv = b.value
    if bv is None:
        return None

    return np.maximum(bv, 0.0)   # clip tiny negatives


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)           # first moment
    m2 = np.zeros_like(a)           # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # Anneal smoothing parameter β (geometric progression)
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # Linear convolution a*a
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # Soft‑max weights (stable log‑sum‑exp)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # Gradient of the smoothed max: ∇ = 2·(rev(a) * w) restricted to n entries
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)          # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m1_hat / (np.sqrt(m2_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact score check
        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # print progress
                # print(f"[smooth] it {it} best {best_score:.6f}")

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the active entry (the k‑th entry of the convolution)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)

"""Search for a sequence minimizing ``evaluate_sequence``.

The algorithm proceeds in three stages:

1. **LP refinement on a small problem (n≈512)** – starting from a
   central‑binomial distribution we solve a linear program that
   finds a partner vector b with the same maximal convolution value.
   A convex combination (1‑t) f + t b is projected onto the simplex;
   repeating this reduces the worst convolution entry.

2. **Upscaling & smooth Adam optimisation** – the improved short vector
   is linearly interpolated to the target length (2048).  We then
   minimise a log‑sum‑exp surrogate of the max‑convolution using Adam
   (with a geometric increase of the smoothing parameter β).  The
   iterates are projected onto the simplex after each update.

3. **Polishing** – a short sub‑gradient descent on the exact max‑objective
   followed by a cheap stochastic hill‑climb (swap / fractional transfer)
   finishes the search.

All stages respect the 1000 s wall‑clock budget and keep the best
sequence found.  The returned list is already normalised (∑aᵢ=1)."""

    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution matrix (Toeplitz) for the LP
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix A such that A @ b = conv(f, b)
    (linear convolution of the fixed vector f with a variable b).
    """
    n = f.shape[0]
    col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=float)])
    A_dense = sla.toeplitz(col, row)          # shape (2n‑1, n)
    return sp.csr_matrix(A_dense)


def solve_lp_max_sum(
    A: sp.csr_matrix,
    M: float,
    warm_start: np.ndarray | None = None,
) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns the optimal b as a numpy array, or None on failure.
    """
    n = A.shape[1]
    b = cp.Variable(n)
    constraints = [A @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    if warm_start is not None:
        b.value = warm_start

    try:
        prob.solve(
            solver=cp.ECOS,
            warm_start=True,
            max_iters=5000,
            abstol=1e-7,
            reltol=1e-6,
            feastol=1e-7,
        )
    except Exception:
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    bv = b.value
    if bv is None:
        return None

    return np.maximum(bv, 0.0)   # clip tiny negatives


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)          # first moment
    m2 = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # geometric annealing of the smoothing parameter β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)          # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m1_hat / (np.sqrt(m2_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness' own bookkeeping
    deadline = start_time + budget_s - 4.0

    best_seq = None
    best_val = float("inf")
    print("Search started – total budget", budget_s, "seconds")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Small‑scale LP refinement (n≈512)
    # -----------------------------------------------------------------
    n_small = 512
    a_small = np.empty(n_small, dtype=float)
    a_small[0] = 1.0
    for i in range(1, n_small):
        a_small[i] = a_small[i - 1] * (2 * i - 1) / (2.0 * i)
    a_small = project_to_simplex(a_small)

    best_small = a_small.copy()
    best_small_score = compute_score(best_small)

    warm_b = None
    t_mix = 0.22
    lp_iter = 0
    # Reserve roughly half the wall‑clock time for later stages
    lp_deadline = deadline - 560.0

    while time.time() < lp_deadline and lp_iter < 12:
        lp_iter += 1
        conv_f = convolution_fft(best_small, best_small)
        M = float(conv_f.max())

        A = build_convolution_matrix(best_small)
        b_opt = solve_lp_max_sum(A, M, warm_start=warm_b)
        if b_opt is None:
            print("LP solve failed – exiting LP loop.")
            break
        warm_b = b_opt

        b_opt = project_to_simplex(b_opt)

        candidate = (1.0 - t_mix) * best_small + t_mix * b_opt
        candidate = project_to_simplex(candidate)

        cand_score = compute_score(candidate)

        if cand_score < best_small_score * 0.9999:
            best_small_score = cand_score
            best_small = candidate
            print(f"[LP {lp_iter}] improved score = {best_small_score:.6f}")
            t_mix = min(t_mix * 1.07, 0.35)
        else:
            t_mix = max(t_mix * 0.85, 0.05)

    print(
        f"LP phase finished after {lp_iter} iterations. Best small score = {best_small_score:.6f}"
    )

    # -----------------------------------------------------------------
    # 2. Upscale to the final length (2048) for fine‑grained optimisation
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    n_big = 2048 if remaining > 400 else 1024
    a_big = upscale_vector(best_small, n_big)

    # -----------------------------------------------------------------
    # 3. Smooth (log‑sum‑exp) Adam optimisation
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 250.0
    if time.time() < smooth_deadline:
        print("Smooth Adam optimisation started")
        a_smooth, smooth_score = smooth_gradient_optimize(
            a_big,
            time_deadline=smooth_deadline,
            max_iters=350_000,
            beta_start=4.0,
            beta_end=400.0,
            step0=1.0,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"After smooth optimisation: score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Projected sub‑gradient polishing
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 100.0
    if time.time() < subgrad_deadline and best_seq is not None:
        print("Sub‑gradient polishing started")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=subgrad_deadline,
            max_iter=500_000,
            step0=0.06,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑gradient improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline and best_seq is not None:
        print("Final hill‑climb started")
        polished_seq, polished_score = hill_climb_permutation(
            best_seq,
            time_deadline=hill_deadline,
            max_steps=800_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb further improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a modest uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
