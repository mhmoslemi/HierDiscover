def subgradient_local(a, steps, step0):
    cur = a.copy()
    for t in range(steps):
        conv = convolution_fft(cur, cur)
        k = np.argmax(conv)
        # gradient:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]
        # step: step0 / sqrt(t+1)
        step = step0 / np.sqrt(t+1)
        cur -= step * g
        cur = project_to_simplex(cur)
    return cur
```

"""
Improved search for a low‑score step function.

Overview
--------
1. **Rich seeding** – generate a large variety of candidate shapes:
   * deterministic sine‑squared, a wide range of beta‑family exponents,
   * Gaussian windows of different widths,
   * piece‑wise constant “edge‑heavy’’ patterns,
   * many random Dirichlet draws (different concentrations).

   All candidates are projected onto the probability simplex and the
   best few (default 10) are kept.

2. **Iterated sub‑gradient refinement** – from each of the kept seeds we
   run a short projected sub‑gradient descent on the exact objective
   `2·n·max(conv(p,p))`.  The step‑size is larger than in the previous
   baseline (``step0=0.2``) which encourages escaping shallow minima.
   After each run the best vector (global) is updated.

3. **Local perturbation loop** – while time permits we repeatedly
   perturb the current best vector with a small Gaussian jitter,
   re‑project onto the simplex and perform a brief sub‑gradient run.
   Occasionally a greedy hill‑climb polishing step is applied.

The whole procedure respects the wall‑clock budget (default 1000 s) and
returns a plain Python list of non‑negative heights.  The objective is
independent of scaling, therefore all vectors are normalised to sum = 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """
    Sine‑squared shape a_i = sin^2(pi*(i+1)/(n+1)), normalised.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p .
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """
    Discrete Gaussian centred in the middle.
    """
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def generate_edge_step(n: int, e: int, r: float) -> np.ndarray:
    """
    Two‑level step function.
    * `e` – number of points on each edge that receive the higher weight.
    * `r` – ratio (edge weight) / (inner weight)  (r ≥ 1).
    """
    if e * 2 >= n:
        raise ValueError("edge width too large")
    # inner weight = b, edge weight = a = r * b
    b = 1.0 / (n - 2 * e + 2 * e * r)
    a = r * b
    a_vec = np.full(n, b)
    a_vec[:e] = a
    a_vec[-e:] = a
    return a_vec


# ----------------------------------------------------------------------
# Optimisation helpers
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on `2·n·max(conv(a,a))`.
    Stops when `deadline` is reached.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy local search that only accepts moves that improve the
    exact objective.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.05

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # renormalise (numerical drift)
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                       # become more conservative

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the given time budget and
    returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0. Problem size
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 800.0:
        n = 4096
    print(f"[INFO] Using length n = {n}")

    # ------------------------------------------------------------------
    # 1. Seeding phase – generate a large pool of diverse candidates
    # ------------------------------------------------------------------
    TOP_K = 10                     # keep the ten best seeds
    top = []                       # list of (score, vector)
    best_seq = None
    best_val = float("inf")

    def register(vec: np.ndarray):
        nonlocal best_seq, best_val, top
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if sc < best_val:
            best_val = sc
            best_seq = vec.copy()
            print(f"[SEED] New global best: {best_val:.6f}")
        # maintain top‑K list
        if len(top) < TOP_K:
            top.append((sc, vec.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if sc < top[-1][0]:
                top[-1] = (sc, vec.copy())
                top.sort(key=lambda x: x[0])

    # deterministic sine‑squared
    register(generate_deterministic(n))

    # beta‑family over a wide exponent range
    for p in np.linspace(-6.0, 6.0, 33):
        register(generate_beta_family(n, p))

    # Gaussian windows
    for sigma_frac in np.linspace(0.08, 0.45, 20):
        sigma = sigma_frac * n
        register(generate_gaussian(n, sigma))

    # Edge‑heavy step families
    for frac in np.linspace(0.02, 0.25, 12):
        e = max(1, int(round(frac * n)))
        if 2 * e >= n:
            continue
        for r in np.logspace(0.0, 2.0, 15):  # r from 1 to 100
            register(generate_edge_step(n, e, r))

    # random Dirichlet draws with several concentrations
    concentrations = [0.1, 0.3, 0.6, 1.0, 2.0, 5.0]
    for conc in concentrations:
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            register(random_dirichlet(n, conc=conc))

    # ------------------------------------------------------------------
    # 2. Iterated sub‑gradient optimisation from the stored seeds
    # ------------------------------------------------------------------
    # each seed gets a short refinement; we keep the best overall.
    print("[INFO] Starting iterated sub‑gradient refinement")
    for idx, (_, seed_vec) in enumerate(top):
        if time.time() >= deadline - 5.0:
            break
        # allocate a modest slice of the remaining budget
        sub_deadline = min(deadline, time.time() + (total_time * 0.02))
        refined_vec, refined_score = subgradient_refine(
            seed_vec,
            compute_score(seed_vec),
            sub_deadline,
            max_iter=120_000,
            step0=0.2,
        )
        register(refined_vec)

    # ------------------------------------------------------------------
    # 3. Local perturbation loop – repeatedly jitter the current best
    # ------------------------------------------------------------------
    print("[INFO] Entering perturbation / polishing loop")
    while time.time() < deadline - 2.0:
        # pick a base (biased towards the best)
        if np.random.rand() < 0.6 and best_seq is not None:
            base = best_seq
        else:
            base = top[np.random.randint(len(top))][1]

        # Gaussian jitter (2 % of mass on each coordinate)
        jitter = np.random.randn(n) * 0.02
        cand = base + jitter
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        # short sub‑gradient run
        sub_deadline = min(deadline, time.time() + (total_time * 0.015))
        cand, cand_score = subgradient_refine(
            cand,
            compute_score(cand),
            sub_deadline,
            max_iter=60_000,
            step0=0.2,
        )
        register(cand)

        # occasional greedy hill‑climb polishing
        if np.random.rand() < 0.12:
            hc_deadline = min(deadline, time.time() + (total_time * 0.008))
            cand, cand_score = hill_climb(cand, hc_deadline, max_steps=30_000)
            register(cand)

    # ------------------------------------------------------------------
    # 4. Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\n[RESULT] Final returned score = {final_val:.6f}")
    return final_seq
