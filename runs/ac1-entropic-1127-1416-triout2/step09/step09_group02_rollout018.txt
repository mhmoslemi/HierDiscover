def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0

    # Choose length; may use 2048 or larger if time
    n = 2048
    if (deadline - start_time) > 300:
        n = 4096

    # initial distribution: uniform
    a = np.full(n, 1.0/n, dtype=float)

    # function to compute score (2*n * max(conv))
    def score(vec):
        conv = convolution_fft(vec, vec)
        return 2.0 * n * float(conv.max())

    best_a = a.copy()
    best_score = score(a)

    # parameters for smoothing gradient descent
    beta = 5.0
    beta_max = 200.0
    beta_rate = (beta_max / beta) ** (1.0 / 50000) # exponential increase per iteration
    eta = 0.05   # step size (additive)
    eta_decay = 0.9995

    iteration = 0
    while time.time() < deadline - 5.0:
        # compute conv
        conv = convolution_fft(a, a)   # length 2n-1
        # compute softmax weights
        max_c = conv.max()
        # stable softmax
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # compute gradient via convolution w * a_rev
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)  # length 3n-2
        grad = 2.0 * grad_full[n-1 : n-1 + n]   # slice length n
        # additive update
        a_new = a - eta * grad
        # ensure non-neg and renormalise
        a_new = np.maximum(a_new, 0.0)
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= sum_a

        cur_score = 2.0 * n * float(conv.max())
        # evaluate new score (for new a)
        new_conv = convolution_fft(a_new, a_new)
        new_score = 2.0 * n * float(new_conv.max())

        # accept if improves
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            # print progress
            print(f"Iter {iteration}: new best score {best_score:.6f}")
        # simple acceptance rule: accept if improves else keep old
        if new_score <= cur_score:
            a = a_new
        else:
            # maybe still accept with some prob? for now keep old
            pass

        # update schedules
        beta = min(beta_max, beta * beta_rate)
        eta *= eta_decay
        iteration += 1

    # final polishing: simple hill-climb using local mass moves
    cur_a = best_a.copy()
    cur_score = best_score
    # hill climb loop
    while time.time() < deadline - 0.5:
        i, j = np.random.randint(n, size=2)
        if cur_a[i] == 0:

        delta = cur_a[i] * 0.1  # move 10% of i's mass
        cand = cur_a.copy()

"""
Optimisation of a non‑negative step function for the provided 
`evaluate_sequence` score.

The score is 2·n·max(conv(a,a)) / (sum(a)²) where n=len(a) and
`conv` is the linear convolution.  Minimising the maximal entry of the
self‑convolution is equivalent to making the distribution of the sum of
two independent draws from the step‑function as flat as possible.

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1).  Uniform weights are a
   good generic start.
2. Repeatedly perform a *smooth‑max* descent:
   *   Compute the full convolution `c = a*a`.
   *   Form a soft‑max distribution `w` over the convolution entries:
       w_k ∝ exp(β·c_k).  The temperature β is increased slowly so the
       first iterations smooth the objective and later ones focus on the
       true maximum.
   *   The gradient of the smoothed objective is  
       g_i = 2·∑_j w_{i+j}·a_j .
       It can be obtained efficiently by a second FFT‑based convolution
       of `w` with the reversed vector `a[::-1]`.
   *   Update `a ← a – η·g` (additive) and re‑project onto the simplex
       (non‑negative, sums to 1).  Step‑size η decays gradually.
   *   Keep the best vector seen so far.
3. A short stochastic hill‑climb that only accepts improving
   mass‑transfer moves finishes the search.
4. Return the best vector as a plain list; the harness will call
   `evaluate_sequence` on it.

The whole routine respects a wall‑clock budget (default 1000 s) and
uses only NumPy – no external file or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using the FFT.
    Length of the result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact score used by the evaluation function (assuming a already
    respects the clipping limits).  The function works for any non‑zero
    sum; the optimisation loop keeps the sum equal to 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:          # guard – matches evaluate_sequence
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def smooth_max_descent(
    a: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 150.0,
    eta_start: float = 0.08,
    eta_end: float = 0.001,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation of the max‑convolution.
    The temperature `beta` grows from `beta_start` to `beta_end`,
    while the step size `eta` decays from `eta_start` to `eta_end`.
    The best vector encountered is returned.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- compute convolution of the current iterate ----
        conv = convolution_fft(a, a)                # length 2n-1
        max_c = conv.max()

        # ---- soft‑max weights over the convolution entries ----
        #   w_k = exp(beta * (c_k - max_c)) / Σ
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / max_iters)
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()                # length 2n-1

        # ---- gradient of the smooth objective (vectorised) ----
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)       # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # slice of length n

        # ---- step‑size schedule ----
        eta = eta_start + (eta_end - eta_start) * min(1.0, it / max_iters)

        # ---- additive update + projection onto the simplex ----
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()                        # normalisation = projection

        # ---- evaluate new point (exact score) ----
        new_score = compute_score(a_new)

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[smooth‑max] iter {it:6d}, new best {best_score:.6f}")

        # simple acceptance rule – move if we made progress,
        # otherwise stay at the current point.
        cur_score = compute_score(a)
        if new_score <= cur_score:
            a = a_new

        it += 1

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that only accepts strictly improving
    mass‑transfer moves.  Acts as a final polishing step.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions (does not change the score but adds
            # stochasticity – useful if the vector has exact ties)
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a small fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # renormalise (numerical drift)
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                         # become more conservative

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Finds a non‑negative step‑function with a low `evaluate_sequence`
    score.  The routine respects the given time budget and returns a
    plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # 1. Choose a length – 2048 gives a good trade‑off between speed
    #    and achievable bound; 4096 is used if we have ample time.
    # --------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = deadline - time.time()
    n = 2048
    if remaining > 300.0:
        n = 4096
    print(f"Using sequence length n = {n}")

    # --------------------------------------------------------------
    # 2. Initialise – start from the uniform distribution (the simplest
    #    feasible point).  Also keep a cheap deterministic shape as a
    #    fallback in case the optimisation stalls.
    # --------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=float)          # uniform probabilities

    # try a sine‑squared seed as an alternative initial point
    i = np.arange(1, n + 1, dtype=float)
    sine = np.sin(np.pi * i / (n + 1.0)) ** 2
    sine /= sine.sum()
    if compute_score(sine) < compute_score(a):
        a = sine.copy()

    best_a, best_score = a.copy(), compute_score(a)
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Smooth‑max gradient descent (major part of the time budget)
    # --------------------------------------------------------------
    phase1_deadline = deadline - 30.0               # keep ~30 s for polishing
    print("\n[Phase 1] Smooth‑max gradient descent")
    best_a, best_score = smooth_max_descent(
        best_a,
        phase1_deadline,
        beta_start=5.0,
        beta_end=150.0,
        eta_start=0.08,
        eta_end=0.001,
        max_iters=400_000,
    )
    print(f"After smooth‑max phase: best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 2] Hill‑climb polishing")
        best_a, best_score = hill_climb(best_a, deadline, max_steps=800_000)
        print(f"After polishing: best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return the best sequence found (as plain list of floats)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)        # official score
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq
