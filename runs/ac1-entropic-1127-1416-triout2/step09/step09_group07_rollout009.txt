def exp_gradient_optimize(start_a, deadline, max_iters):
    a = np.copy(start_a)
    eps = 1e-14
    a = np.maximum(a, eps)
    a /= a.sum()
    n = a.shape[0]
    best_a, best_sc = a.copy(), compute_score(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        t = it / max_iters
        beta = 0.1 * (2000.0/0.1) ** t # start at 0.1
        # compute conv
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute softmax over conv entries
        # Use stabilized exp
        exp_w = np.exp(beta * (conv - maxc))
        # softmax numerator
        w = exp_w / np.sum(exp_w)
        # compute gradient: 2 * (a reversed) * w conv
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n-1:n-1+n]
        # step size schedule; use eta0 maybe 0.5
        eta = 0.5 / np.sqrt(it)
        # update via exponentiated gradient (KL mirror)
        a = np.maximum(a * np.exp(-eta * grad), eps)
        # normalize
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a /= a_sum
        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    # final check
    final_sc = compute_score(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()
    return best_a, best_sc

def hill_climb_sa(start_a, deadline, initial_step=0.2, inc_factor=1.02, dec_factor=0.98, recompute_interval=50000):
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    cur_score = best_score
    step_scale = initial_step
    T0 = best_score * 0.01  # initial temperature relative to score maybe.
    while time.time() < deadline:
        elapsed = time.time() - start_time # need to capture original start
        # compute temperature linearly decaying:
        T = T0 * max(0.0, 1 - (elapsed / total_time))
        # choose src,dst
        src = np.random.randint(n)
        if a[src] == 0:
            continue
        dst = np.random.randint(n-1)
        if dst >= src:
            dst += 1
        # delta ratio maybe random factor * step_scale
        delta = step_scale * a[src]  # up to step_scale fraction
        # incremental conv update
        conv_cand = conv.copy()
        conv_cand[src:src+n] -= 2.0 * delta * a
        conv_cand[dst:dst+n] += 2.0 * delta * a
        conv_cand[2*src] += delta*delta
        conv_cand[2*dst] += delta*delta
        conv_cand[src+dst] -= 2.0 * delta * delta
        cand_score = 2.0 * n * float(conv_cand.max())
        diff = cand_score - cur_score
        accept = False
        if diff < 0:
            accept = True
        else:
            # accept with probability exp(-diff / T) if T > 0
            if T > 0 and np.random.rand() < np.exp(-diff / T):
                accept = True
        if accept:
            # apply move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_score = cand_score
            # adapt step_scale
            step_scale = min(step_scale * inc_factor, 0.5)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            step_scale = step_scale * dec_factor
        # periodic recompute conv to reduce drift
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())
        steps += 1
    return best_a, best_score

"""
Propose a low‑score step‑function sequence.

The search consists of three phases:

1. **Fast analytic grid** – a tiny two‑parameter family
   a_i(α,λ) = λ·edge(α) + (1‑λ)·uniform, where
   edge(α)_i ∝ [(i+½)(n‑½‑i)]⁻ᵅ.
   The grid (α∈[0,5], λ∈[0,1]) is scanned once; the best
   vector is kept as the starting point.

2. **Long exponentiated‑gradient optimisation** – minimises a
   log‑sum‑exp smooth approximation of the max‑convolution.
   β (smoothness) is annealed from a small value to a large one,
   the step‑size decays as η₀/√t and a KL‑mirror (exponentiated‑gradient)
   update keeps the vector on the probability simplex.
   The loop runs until the remaining budget is exhausted,
   constantly remembering the best exact score.

3. **Simulated‑annealing hill‑climb** – an O(n) incremental walk.
   Each move transfers a small packet of mass from a source index
   to a destination index; the convolution is updated by the
   linear formula Δconv = 2·δ·(shift_dst(a) – shift_src(a))
   plus quadratic corrections.  Moves that improve the exact
   score are always accepted; worsening moves are accepted
   with probability exp(‑Δ/ T) where a temperature T decays
   linearly to 0.  The step‑size adapts up/down on accept/reject.

All phases respect the 1000 s wall‑clock limit and the best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
import random
import sys

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) assuming sum(a)=1.
    This matches the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def analytic_edge(alpha: float, n: int) -> np.ndarray:
    """
    Edge‑heavy shape a_i ∝ [(i+½)(n‑½‑i)]⁻ᵅ, normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    # use logs to avoid overflow/underflow for extreme α
    log_term = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * log_term)
    # protect against infinities (very large α)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def grid_initial_search(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a modest two‑parameter family (α, λ) and keep the best vector.
    λ mixes the edge‑heavy shape with the uniform distribution.
    """
    best_a = None
    best_sc = float("inf")
    # uniform reference
    uniform = np.full(n, 1.0 / n, dtype=float)

    alphas = np.linspace(0.0, 5.0, 51)          # 0 … 5  (step 0.1)
    lambdas = np.linspace(0.0, 1.0, 21)         # mixture weight

    for alpha in alphas:
        if time.time() >= deadline:
            break
        edge = analytic_edge(alpha, n)
        for lam in lambdas:
            a = lam * edge + (1.0 - lam) * uniform
            # already normalised (both components sum to 1)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc


def exp_gradient_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         beta_start: float = 0.1,
                         beta_end: float = 2000.0,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) minimisation of the
    smoothed max‑convolution (log‑sum‑exp surrogate).
    The schedule for β grows exponentially from beta_start to beta_end.
    """
    a = np.maximum(start_a.astype(float), 1e-14)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smoothed max (log‑sum‑exp) ----
        conv = convolution_fft(a, a)           # length 2n‑1
        maxc = conv.max()
        # stabilised expo
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # soft‑max weights

        # ---- gradient of the smooth objective ----
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # ---- step size (vanishing) ----
        eta = eta0 / np.sqrt(it)

        # ---- KL‑mirror update (exponentiated gradient) ----
        a = np.maximum(a * np.exp(-eta * grad), 1e-14)
        s = a.sum()
        a /= s

        # ---- occasional exact evaluation ----
        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final exact evaluation
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  init_step: float = 0.2,
                  inc_factor: float = 1.03,
                  dec_factor: float = 0.97,
                  recompute_interval: int = 40_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing random walk that transfers a small packet of mass
    between two positions.  Convolution is updated incrementally.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)                     # exact convolution
    cur_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_score = cur_score

    step_scale = init_step
    steps = 0
    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best_a, best_score

    while time.time() < deadline:
        # linear temperature decay (to 0)
        elapsed = time.time() - start_time
        T = cur_score * 0.01 * max(0.0, 1.0 - elapsed / total_time)  # 1 % of current score

        # pick a source with positive mass
        src = np.random.randint(n)
        if a[src] <= 0.0:
            continue
        # pick a distinct destination
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # amount to move – a fraction of the source mass
        delta = step_scale * a[src]
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_score = 2.0 * n * float(conv_cand.max())
        diff = cand_score - cur_score

        # Metropolis acceptance
        accept = False
        if diff < 0:
            accept = True
        else:
            if T > 0.0 and random.random() < np.exp(-diff / T):
                accept = True

        if accept:
            # apply move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_score = cand_score
            step_scale = min(step_scale * inc_factor, 0.5)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            step_scale *= dec_factor

        steps += 1
        if steps % recompute_interval == 0:
            # periodic full recomputation to avoid drift
            conv = convolution_fft(a, a)
            cur_score = 2.0 * n * float(conv.max())

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows three stages (grid → exponentiated gradient → SA hill‑climb)
    and respects the 1000 s wall‑clock limit.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 1.5

    # -------------------------------------------------
    # Choose a workable length (larger n → potentially lower bound)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Fast analytic grid search
    # -------------------------------------------------
    grid_deadline = min(deadline - 250.0, time.time() + 30.0)
    a_grid, sc_grid = grid_initial_search(n, grid_deadline)
    if a_grid is not None and sc_grid < best_score:
        best_score = sc_grid
        best_seq = a_grid.tolist()
        print(f"[{time.time():.1f}] after grid search best = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Long exponentiated‑gradient optimisation
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] exponentiated‑gradient optimisation")
        a_start = np.array(best_seq, dtype=float)
        a_opt, sc_opt = exp_gradient_optimize(
            a_start,
            deadline - 120.0,
            max_iters=2_000_000,
            beta_start=0.1,
            beta_end=2000.0,
            eta0=0.5,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Simulated‑annealing hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] simulated‑annealing hill‑climb")
        a_start = np.array(best_seq, dtype=float)
        a_final, sc_final = hill_climb_sa(
            a_start,
            deadline,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return the plain Python list expected by the harness.
    return best_seq
