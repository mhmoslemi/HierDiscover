"""
Propose a non‑negative step‑function (probability vector) that minimises

    evaluate_sequence(a) = 2·n·max(a * a) / (sum(a))²

where “*” denotes aperiodic convolution.
The algorithm is a multi‑stage heuristic:

0.  Use the previously known construction (if available) as a fallback.
1.  Scan several analytic families (symmetric power‑law, left‑heavy power,
    exponential decay) and many random initialisers (Dirichlet, sparse,
    monotone). Keep the best.
2.  Refine the best candidate with a smoothed surrogate
    fβ(a)= (1/β)·log Σ exp(β·(a*a)k) using Adam on the soft‑max
    parametrisation. β is annealed up to 5 000.
3.  Perform a long mirror‑subgradient (exponentiated‑gradient) descent
    on the true max‑objective. The sub‑gradient of the active convolution
    entry is computed analytically; the update stays on the probability
    simplex (entropy mirror map). The convolution is evaluated with a
    single FFT per iteration, allowing many thousands of steps.
4.  Symmetrise the vector (a←(a+aᵀ)/2) – most good solutions are symmetric.
5.  Finish with a cheap stochastic hill‑climb (simulated annealing)
    that moves a small amount of mass between two positions.
All phases respect the global time budget (≈ 1000 s) and the best sequence
found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self(a: np.ndarray) -> np.ndarray:
    """Aperiodic auto‑convolution a*a using a single real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    C = A * A
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum(a)=1)."""
    n = a.shape[0]
    conv = conv_self(a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α), symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β), monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i), monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_dirichlet_start(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw from a Dirichlet(α) with common concentration parameter."""
    shape = np.random.gamma(conc, 1.0, size=n)
    a = shape / shape.sum()
    return a


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp surrogate) via Adam
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 5_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """Minimise the smoothed surrogate fβ using Adam on the soft‑max layer."""
    eps = 1e-12
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothed max of convolution
        conv = conv_self(a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()          # softmax over convolution entries

        # gradient of fβ w.r.t. a (see derivation in the original code)
        a_rev = a[::-1]
        conv_rev = np.fft.rfft(a_rev, n=next_pow2(3 * a.shape[0] - 2))
        w_rfft = np.fft.rfft(w, n=conv_rev.size)
        grad_a = np.fft.irfft(conv_rev * w_rfft, n=conv_rev.size)
        grad_a = grad_a[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact check
        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 2.0,
    check_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on 2·n·max(a*a)."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    conv = conv_self(a)
    best_score = 2.0 * n * float(conv.max())

    t = 0
    while time.time() < deadline:
        t += 1
        conv = conv_self(a)
        k = int(np.argmax(conv))
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional symmetrisation (helps for symmetric optima)
        if t % 1000 == 0:
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)

        # avoid too frequent exact re‑evaluation – we already have it each iter
        if t % check_interval == 0:
            pass  # the update above already handled a new best

    # final exact check
    conv = conv_self(a)
    cur_score = 2.0 * n * float(conv.max())
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (tiny simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a tiny temperature schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – start small, linearly cool
    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()            # renorm = projection onto simplex
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families + random draws (cheap, dense grid)
    # ------------------------------------------------------------------
    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    sc = compute_score(uniform)
    if sc < best_score:
        best_score = sc
        best_seq = uniform.tolist()
        print(f"[{time.time():.1f}] new best (uniform) = {best_score:.6f}")

    # symmetric product family
    for alpha in np.linspace(0.0, 5.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

    # left‑heavy power‑law family
    for beta in np.linspace(0.0, 5.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

    # exponential decay family
    for gamma in np.linspace(0.0, 6.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    # random Dirichlet draws with various concentrations
    concentrations = [0.1, 0.3, 0.5, 1.0, 2.0]
    for conc in concentrations:
        for _ in range(20):
            if time.time() >= deadline - 30.0:
                break
            vec = random_dirichlet_start(n, conc)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (Dirichlet c={conc:.1f}) = {best_score:.6f}")

    # random sparse starts
    for _ in range(30):
        if time.time() >= deadline - 30.0:
            break
        vec = random_sparse_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑sparse) = {best_score:.6f}")

    # random monotone starts
    for _ in range(30):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam smoothing (many restarts while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        # initialise from the current best with a small random perturbation
        base = np.array(best_seq, dtype=np.float64) if best_seq is not None else uniform
        # add a little Gaussian noise in the log‑space
        z0 = np.log(base + 1e-12)
        sigma = 0.2
        z0 += sigma * np.random.randn(n)
        start_vec = softmax(z0)

        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 200.0,
            max_iters=200_000,
            beta_start=5.0,
            beta_end=5_000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam‑smooth = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Prolonged mirror sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            step0=2.0,
            check_interval=200,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Symmetrisation (most optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb (tiny simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: fall back to a modest uniform vector if nothing found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
