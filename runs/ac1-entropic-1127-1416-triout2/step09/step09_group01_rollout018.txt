    n_small = len(vec_small)

    # Ensure non-negativity
    vec_target = np.maximum(vec_target, 0)
    # Normalize to sum 1
    vec_target = vec_target / vec_target.sum()

def generate_sidon_set(max_n, target_s):
    S = []
    sums = set()
    for x in range(max_n):
        conflict = False
        for y in S:
            if x + y in sums:
                conflict = True
                break
        if not conflict:
            # check also self-sum x+x
            if x + x in sums:
                conflict = True
        if not conflict:
            # add
            for y in S:
                sums.add(x + y)
            sums.add(x + x)
            S.append(x)
            if len(S) >= target_s:
                break
    return S
```

def flatten_conv_optimize(start_a, deadline, max_iters=200_000, lr=0.05,
                         lambda0=0.5, lambda_end=0.0, beta_start=2.0,
                         beta_end=2000.0):
    a = start_a.copy()
    n = len(a)
    # param logit
    eps=1e-12
    z = np.log(a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1=0.9
    beta2=0.999
    eps_adam=1e-12

    best_a = a
    best_score = compute_score(a)

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # schedule beta
        t_ratio = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_ratio
        # schedule lambda
        lambda_cur = lambda0 * (lambda_end / lambda0) ** t_ratio if lambda0 > 0 else 0.0

        a = softmax(z)

        # compute convolution
        conv = convolution_fft(a, a)
        avg = conv.mean()
        residual = conv - avg

        # gradient of variance part
        a_rev = a[::-1]
        conv_res = convolution_fft(a_rev, residual)  # length 3n-2
        grad_var = 2.0 * conv_res[n-1:n-1+n]

        # gradient of smoothed max part
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()
        conv_max = convolution_fft(a_rev, w)
        grad_max = 2.0 * conv_max[n-1:n-1+n]

        # combine
        grad_a = (1 - lambda_cur) * grad_max + lambda_cur * grad_var

        # backprop through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

Algorithm (≈ 1000 s total)
--------------------------------
1. Choose a modest “pilot” length (≈ 4096).  For this size the FFT‑based
   gradients are cheap, so we can run many iterations.
2. Optimise a *flattening* loss:
        L = (1‑β)·log‑sum‑exp(β·conv)  +  λ·∑(conv – mean(conv))²
   – the first term drives the maximum of the convolution down,
   – the second term spreads the convolution mass uniformly.
   The variables are the logits  z  of a soft‑max (a = softmax(z));
   optimisation is performed with Adam and the smoothing/λ schedule is
   annealed from very soft (β≈2) to hard (β≈2000) while λ decays to 0.
3. The best pilot vector is *upsampled* to the final length
   (8192 or 16384) by linear interpolation and projected back onto the
   probability simplex.
4. A final Adam optimisation on the pure smoothed‑max objective
   (log‑sum‑exp) refines the large‑scale vector.
5. Mirror (entropy‑mirror) sub‑gradient descent on the true max‑objective
   (the “max‑convolution”) further reduces the objective.
6. Symmetrisation (averaging with the reversed vector) is tried – many
   optimal solutions are symmetric.
7. A tiny simulated‑annealing hill‑climb finishes the search.
8. The best sequence found is returned as a plain Python list.
All stages respect the global time budget (default 1000 s) and print
progress for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) = 1):
        2·n·max(a * a)
    This is identical to evaluate_sequence up to the harmless scaling
    invariances (the harness will call evaluate_sequence on the final
    list).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for cheap initial baselines)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_dirichlet_start(n: int, conc: float = 1.0) -> np.ndarray:
    """Full‑support Dirichlet draw with concentration ‘conc’."""
    a = np.random.dirichlet(np.full(n, conc))
    return a


# ----------------------------------------------------------------------
# Optimiser that flattens the convolution (log‑sum‑exp + variance)
# ----------------------------------------------------------------------


def flatten_conv_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
    lambda_start: float = 0.6,
    beta_start: float = 2.0,
    beta_end: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the blended loss

        L = (1‑λ)·(1/β)·log Σ exp(β·conv)   +   λ·‖conv – mean(conv)‖²

    λ is linearly annealed from `lambda_start` to 0.
    The result is a vector that gives a rather flat self‑convolution,
    which is a good warm‑start for the final max‑objective.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise logits from the start vector
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- schedules ----
        t = it / max_iters                  # 0…1
        beta = beta_start * (beta_end / beta_start) ** t
        lam = lambda_start * (1.0 - t)      # linear decay to zero

        a = softmax(z)

        # ---- convolution and statistics ----
        conv = convolution_fft(a, a)        # length 2n‑1
        avg = conv.mean()
        residual = conv - avg               # for the variance part

        # ---- gradient of variance term (2·(residual * a_rev)) ----
        a_rev = a[::-1]
        conv_res = convolution_fft(a_rev, residual)
        grad_var = 2.0 * conv_res[n - 1 : n - 1 + n]

        # ---- gradient of smoothed‑max term (log‑sum‑exp) ----
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()
        conv_max = convolution_fft(a_rev, w)
        grad_max = 2.0 * conv_max[n - 1 : n - 1 + n]

        # ---- blended gradient w.r.t. a ----
        grad_a = (1.0 - lam) * grad_max + lam * grad_var

        # ---- back‑prop through soft‑max ----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional exact evaluation (cheaper than every step) ----
        if it % 800 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for the true
    objective 2·n·max(a*a).  The step size is η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(

    Very low temperature allows occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                      # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper for up‑sampling a pilot vector to the final length
# ----------------------------------------------------------------------


def upscale_vector(vec_small: np.ndarray, target_len: int) -> np.ndarray:
    """
    Linear interpolation from length len(vec_small) to `target_len`,
    followed by a projection onto the simplex.
    """
    n_small = vec_small.shape[0]
    x_small = np.arange(n_small, dtype=np.float64)
    x_target = np.linspace(0, n_small - 1, target_len, dtype=np.float64)
    vec_big = np.interp(x_target, x_small, vec_small)
    vec_big = np.maximum(vec_big, 0.0)
    vec_big = project_to_simplex(vec_big)
    return vec_big


# ----------------------------------------------------------------------
# High‑level search driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a couple of seconds for the harness to retrieve the result
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose lengths
    # --------------------------------------------------------------
    final_n = 16384 if budget_s >= 900 else 8192
    pilot_n = 4096 if final_n >= 8192 else 2048
    print(f"[{time.time():.1f}] pilot length = {pilot_n}, final length = {final_n}")

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. fallback to a known construction if available
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. cheap analytic families on the pilot length (quick baseline)
    # --------------------------------------------------------------
    for a in np.linspace(0.0, 5.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(pilot_n, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={a:.2f}) = {best_score:.6f}")

    for b in np.linspace(0.0, 5.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(pilot_n, b)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {best_score:.6f}")

    for g in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(pilot_n, g)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={g:.2f}) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Pilot optimisation – flatten the convolution
    # --------------------------------------------------------------
    pilot_deadline = start_time + budget_s * 0.38   # roughly 38 % of the budget
    while time.time() < pilot_deadline:
        # random Dirichlet start with a concentration that varies widely
        conc = np.random.uniform(0.05, 5.0)
        start_vec = random_dirichlet_start(pilot_n, conc)
        a_opt, sc_opt = flatten_conv_optimize(
            start_vec,
            pilot_deadline,
            max_iters=200_000,
            lr=0.05,
            lambda_start=0.6,
            beta_start=2.0,
            beta_end=2000.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] pilot flatten improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Upscale to the final length and perform a smooth‑max Adam run
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback uniform pilot
        best_seq = [1.0 / pilot_n] * pilot_n

    a_up = upscale_vector(np.array(best_seq, dtype=np.float64), final_n)

    # we give the remaining time (about 50 %) to this stage
    stage2_deadline = time.time() + (deadline - time.time()) * 0.45
    a_opt2, sc_opt2 = adam_smooth_optimize(
        a_up,
        stage2_deadline,
        max_iters=200_000,
        beta_start=2.0,
        beta_end=5000.0,
        lr=0.02,
    )
    if sc_opt2 < best_score:
        best_score = sc_opt2
        best_seq = a_opt2.tolist()
        print(f"[{time.time():.1f}] after smooth‑max Adam best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Mirror sub‑gradient descent on the exact max‑objective
    # --------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            step0=0.8,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Symmetrisation (many optimal solutions are symmetric)
    # --------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final tiny simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Safeguard: uniform vector if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / final_n] * final_n
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 8. Final official evaluation (the harness will also call)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return plain Python list of floats
    return best_seq
