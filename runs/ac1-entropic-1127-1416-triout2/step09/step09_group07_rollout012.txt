def simulated_hill_climb(start_a, n, deadline):
    a_cur = start_a.copy()
    conv_cur = convolution_fft(a_cur, a_cur)
    cur_sc = 2*n*conv_cur.max()
    best_a = a_cur.copy()
    best_sc = cur_sc

    # parameters
    step_scale = 0.12
    inc_factor=1.05, decay_factor=0.97
    temp = (best_sc) * 0.01   # initial temperature proportionally small.
    temp_min = 1e-8
    temp_decay = 0.9999  # per iteration maybe

    steps = 0
    while time.time() < deadline:
        # choose move
        if np.random.rand() < 0.5:
            # swap like earlier
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # we can just transfer delta from larger to smaller
            if a_cur[i] > a_cur[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = (a_cur[src] - a_cur[dst]) * step_scale  # maybe use step_scale fraction of difference
        else:
            src = np.random.randint(n)
            if a_cur[src] == 0.0:
                continue
            delta = step_scale * a_cur[src]  # fraction of src

                dst += 1

        # build candidate conv via copy and updates
        conv_cand = conv_cur.copy()

        conv_cand[dst:dst+n] += 2.0 * delta * a_cur
        conv_cand[2*src] += delta * delta
        conv_cand[2*dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

            a_cur[src] -= delta
            a_cur[dst] += delta

            best_a = a_cur.copy()

        if steps % 200000 == 0:
            conv_cur = convolution_fft(a_cur, a_cur)

"""
A multi‑stage optimizer for the step‑function evaluation problem.

1. Analytic one‑parameter family  a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ  (grid search).
2. Several runs of L‑BFGS‑B on a smooth approximation of the max‑convolution.
   The variables are the logits of a soft‑max (ensuring non‑negativity and ∑a=1).
3. A long stochastic hill‑climb with simulated‑annealing acceptance.
   The convolution is updated incrementally (O(n) per move) to keep the
   procedure fast even for large n.

The algorithm respects the 1000 s wall‑clock limit, keeps the best sequence
found and returns it as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities – must stay at top level (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector already normalised to sum = 1.
    This matches the value returned by `evaluate_sequence` after normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytic start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return the family a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow
    w = np.where(np.isfinite(w), w, np.finfo(np.float64).max)
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """Search a modest α‑grid, keep the best normalised vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([
        np.linspace(-1.0, 0.0, 11),   # centre‑heavy
        np.linspace(0.0, 2.5, 26)    # edge‑heavy
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Smooth max objective for L‑BFGS
# ----------------------------------------------------------------------


def softmax_obj_grad(v: np.ndarray, n: int, beta: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of 2·n·max(a⊛a) where a = softmax(v).
    Returns (objective, gradient w.r.t. v).
    """
    # soft‑max parametrisation (ensures positivity and sum=1)
    a = np.exp(v)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / a_sum

    # convolution a⊛a
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # soft‑max over convolution values (log‑sum‑exp)
    expw = np.exp(beta * (conv - maxc))          # stabilised
    w = expw / expw.sum()                        # weights, sum to 1

    # gradient w.r.t. a (see derivation in analysis)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)         # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

    # chain rule to v (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    # smooth max value
    smooth_max = maxc + np.log(expw.sum()) / beta
    f = 2.0 * n * smooth_max
    return f, grad_v.astype(np.float64)


def lbfgs_opt(start_a: np.ndarray,
              n: int,
              deadline: float,
              beta_seq: list[float] = [5.0, 15.0, 50.0, 150.0, 300.0]) -> tuple[np.ndarray, float]:
    """
    Run a short L‑BFGS optimisation on a sequence of beta values.
    Returns the best exact score and its vector.
    """
    eps = 1e-12
    v_cur = np.log(start_a + eps)      # logits for the soft‑max
    best_a = start_a.copy()
    best_sc = compute_score(best_a)

    for beta in beta_seq:
        if time.time() >= deadline:
            break
        # give each beta a modest iteration budget (max 150 L‑BFGS steps)
        remaining = deadline - time.time()
        maxiter = min(150, int(remaining * 20))   # crude scaling, never too large
        if maxiter <= 0:
            break

        res = minimize(softmax_obj_grad,
                       v_cur,
                       args=(n, beta),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter,
                                'disp': False,
                                'ftol': 1e-12})

        v_cur = res.x
        # evaluate the exact (non‑smoothed) objective
        a_opt = np.exp(v_cur)
        a_opt /= a_opt.sum()
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def simulated_hill_climb(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         inc_factor: float = 1.07,
                         decay_factor: float = 0.97,
                         recompute_interval: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Random walk that moves a small packet of mass between two positions.
    Moves are accepted if they improve the objective or with a temperature‑
    dependent probability (simulated annealing).  The convolution is updated
    incrementally (O(n) per move).
    """
    a_cur = start_a.copy()
    conv_cur = convolution_fft(a_cur, a_cur)               # exact convolution
    cur_sc = 2.0 * n * float(conv_cur.max())

    best_a = a_cur.copy()
    best_sc = cur_sc

    # temperature schedule – start small, decay slowly
    temp = max(1e-6, cur_sc * 0.005)
    temp_min = 1e-10
    temp_decay = 0.9997

    step_scale = 0.12
    steps = 0

    while time.time() < deadline:
        # -------------------------------------------------
        # pick a move (transfer or swap)
        # -------------------------------------------------
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            if a_cur[i] > a_cur[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = step_scale * (a_cur[src] - a_cur[dst])
            if delta <= 0.0:
                continue
        else:
            src = np.random.randint(n)
            if a_cur[src] == 0.0:
                continue
            delta = step_scale * a_cur[src]
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

        # -------------------------------------------------
        # incremental convolution update
        # -------------------------------------------------
        conv_cand = conv_cur.copy()
        # linear part
        conv_cand[src:src + n] -= 2.0 * delta * a_cur
        conv_cand[dst:dst + n] += 2.0 * delta * a_cur
        # quadratic correction
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_sc = 2.0 * n * cand_max

        # -------------------------------------------------
        # acceptance test (SA)
        # -------------------------------------------------
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # apply the move
            a_cur[src] -= delta
            a_cur[dst] += delta
            conv_cur = conv_cand
            cur_sc = cand_sc
            step_scale = min(step_scale * inc_factor, 0.5)
        else:
            step_scale *= decay_factor

        # keep the best ever seen
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_cur.copy()

        # temperature cooling
        temp = max(temp * temp_decay, temp_min)

        steps += 1
        if steps % recompute_interval == 0:
            # periodic full recomputation to avoid numerical drift
            conv_cur = convolution_fft(a_cur, a_cur)
            cur_sc = 2.0 * n * float(conv_cur.max())

    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  The routine runs a short analytic grid search,
    a few L‑BFGS refinements, and finishes with an intensive simulated‑
    annealing hill‑climb.  The best sequence found before the deadline is
    returned as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # -------------------------------------------------
    # Choose a length (larger n gives a lower theoretical bound)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 16384
    elif remaining > 500:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]      # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic start (α‑grid)
    # -------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # -------------------------------------------------
    # 2. L‑BFGS refinement (multiple seeds)
    # -------------------------------------------------
    if time.time() < deadline - 300.0:
        seeds = []
        # seed from the current best (if any)
        if best_seq is not None:
            seeds.append(np.array(best_seq, dtype=float))
        else:
            seeds.append(alpha_start(n, 0.5))

        # a few random Dirichlet seeds (sparse)
        for _ in range(3):
            if time.time() >= deadline - 300.0:
                break
            seeds.append(np.random.dirichlet(np.ones(n) * 0.3))

        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline - 300.0:
                break
            print(f"[{time.time():.1f}] L‑BFGS run {idx + 1}/{len(seeds)}")
            a_opt, sc_opt = lbfgs_opt(seed_vec, n, deadline - 250.0)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}] L‑BFGS improved best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Fast simulated‑annealing hill‑climb (majority of budget)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] simulated‑annealing hill‑climb")
        a_final, sc_final = simulated_hill_climb(
            np.array(best_seq, dtype=float),
            n,
            deadline
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a tiny uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
