"""
Simple mirror‑descent optimiser for the “max‑convolution” objective.

The algorithm works in three phases:
1.  Several random restarts are refined by a smoothed max‑convolution
    gradient using an exponentiated‑gradient (mirror‑descent) update.
    The smoothing parameter β is annealed from a small to a large value,
    so the objective gradually changes from the average convolution to the
    true max‑convolution.
2.  A short projected sub‑gradient descent phase (exact max‑convolution)
    polishes the best candidate found so far.
3.  A tiny stochastic hill‑climb (swap / fractional mass transfer) cleans
    up any remaining local irregularities.

All vectors are kept on the probability simplex (non‑negative, sum = 1),
so the “evaluate_sequence” function reduces to 2·n·max(conv(a,a)).
The implementation automatically chooses a longer sequence (n = 4096) when
the time budget permits – longer sequences give a lower theoretical bound
≈1 for the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT.
    Returns array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[subgrad] iter {t:,}  better score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb (tiny swaps / mass transfers)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline:
        a = np.array(best_seq, dtype=float)
        step_scale = 0.10
        while time.time() < hill_deadline:
            cand = a.copy()
            i, j = np.random.randint(n, size=2)
            if np.random.rand() < 0.5:
                # fractional transfer i -> j
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
            else:
                # simple swap
                cand[i], cand[j] = cand[j], cand[i]

            cand = project_to_simplex(cand)
            cur_score = compute_score(cand)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cand.tolist()
                a = cand
                step_scale = min(step_scale * 1.05, 0.5)
                print(f"[hill] improved to {best_score:.6f}")
            else:
                step_scale *= 0.97

    # -----------------------------------------------------------------
    # Final evaluation (the harness expects a plain list)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
