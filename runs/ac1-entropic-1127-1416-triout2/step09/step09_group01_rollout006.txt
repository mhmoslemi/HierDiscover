"""
Search for a non‑negative step‑function sequence that minimizes the
`evaluate_sequence` objective.

The algorithm works in several stages:

1.  Quick scan of analytically‑parameterised families (including a new
    symmetric exponential “U‑shape”) – small cost, gives a decent baseline.
2.  Intensive smoothed optimisation on a *small* problem (n_small≈2048)
    using Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Many restarts are performed while time permits.
3.  The best small‑scale solution is up‑sampled to a larger dimension
    (n_large≈16384) and refined again with Adam (higher smoothing parameter).
    Throughout Adam we enforce symmetry (a ↔ a[::-1]) which is known to be
    optimal for the min‑max autocorrelation problem.
4.  A mirror‑subgradient (entropy‑mirror) descent on the true max‑objective
    further reduces the worst convolution entry.
5.  Final stochastic hill‑climb with a tiny simulated‑annealing schedule.
6.  The best sequence found is returned as a plain list of floats.

All stages respect the global wall‑clock budget (default 1000 s).  The code
uses only NumPy and the provided `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) = 1):
        score = 2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def normalize_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Analytic families (all generate probability vectors)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shape: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay: a_i ∝ exp(-γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def symmetric_exponential_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape obtained from two opposite exponentials:
       a_i ∝ exp(-γ·i) + exp(-γ·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
    a = w / w.sum()
    return a


def valley_power_family(n: int, p: float) -> np.ndarray:
    """
    Strong U‑shape: a_i ∝ (min(i, n‑1‑i)+0.5)^(‑p).
    """
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i) + 0.5
    w = np.exp(-p * np.log(d))
    a = w / w.sum()
    return a

# Random initialisers

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_gaussian_start(n: int) -> np.ndarray:
    """Draw Gaussian samples, make them non‑negative and project."""
    a = np.abs(np.random.randn(n))
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth optimisation via Adam on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    lr: float = 0.018,
    enforce_sym: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    If enforce_sym is True the iterate is projected onto the symmetric
    simplex (a ↔ a[::-1]) at each step.
    """
    eps = 1e-12
    # initialise z from log‑softmax of start_a
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    if enforce_sym:
        best_a = (best_a + best_a[::-1]) * 0.5
        best_a = normalize_to_simplex(best_a)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- schedule for β (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a (softmax) -----
        a = softmax(z)
        if enforce_sym:
            a = (a + a[::-1]) * 0.5
            a = normalize_to_simplex(a)

        # ----- smoothed weights w_k ∝ exp(β·conv_k) -----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ----- gradient of f_β w.r.t. a -----
        a_rev = a[::-1]
        # convolution of reversed a with w yields needed derivatives
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 400 == 0:
            a_tmp = softmax(z)
            if enforce_sym:
                a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
                a_tmp = normalize_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    if enforce_sym:
        a_fin = (a_fin + a_fin[::-1]) * 0.5
        a_fin = normalize_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy‑mirror) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    enforce_sym: bool = True,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for
    the true objective 2·n·max(a*a).  The iterate is kept on the simplex.
    If enforce_sym is True we project onto the symmetric simplex each
    iteration (a ↔ a[::-1]).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (active term only)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)          # multiplicative update

        # simplex projection
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # optional symmetry projection
        if enforce_sym:
            a = (a + a[::-1]) * 0.5
            a = normalize_to_simplex(a)

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Stochastic hill‑climb with simulated annealing
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a tiny temperature schedule.
    Moves are either a swap of two entries or a small mass transfer.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – start small, linearly cool
    T0 = best_score * init_temp_factor
    T = T0

    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise (simple simplex projection)
        cand = cand / cand.sum()

        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)  # enlarge moves slowly
        else:
            step_scale *= 0.97  # shrink after rejection

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def upsample_sequence(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation from a_small (length m) to length n_large,
    followed by simplex normalisation.
    """
    m = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, m)
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.maximum(a_large, 0.0)
    return normalize_to_simplex(a_large)
