"""
Propose a step‑function (non‑negative sequence) minimising
`evaluate_sequence`.  The objective equals

    2·n·max_k (a * a)[k]      (with  ∑ a_i = 1)

so we must find a probability vector whose auto‑convolution is as flat as
possible.  The search consists of three stages:

1. **Broad family scan** – exponential / power decays (monotone) and three
   symmetric families (U‑shaped exponentials, U‑shaped power‑laws and
   symmetric Beta‑shapes).  The best shape is kept.

2. **Adam optimisation of a smooth max** – the max‑convolution is
   approximated by a log‑sum‑exp with a temperature β that is annealed
   exponentially from 0.5 to a large value (≈5 000).  The variables are
   the logits `u` of a soft‑max parametrisation `a = softmax(u)`.  An
   Adam optimiser (first‑ and second‑moment adaptation) is used; the
   gradient of the smoothed objective w.r.t. `u` is computed analytically
   via two FFT‑based convolutions per iteration.

3. **Stochastic hill‑climb polishing** – a greedy/random walk that swaps
   entries or transfers a small mass between two positions, always keeping
   the vector on the probability simplex.  The move size is adapted
   dynamically.

All stages respect the 1000 s wall‑clock limit.  Intermediate scores are
printed for monitoring.  The returned value is a plain Python list of
floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities – no closures, pure functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(convolution(a, a))
    This matches `evaluate_sequence` up to the clipping performed there.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Families of candidate shapes
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def sym_exponential_shape(n: int, lam: float) -> np.ndarray:
    """U‑shaped exponential: a_i ∝ exp(-lam * min(i, n‑1‑i))."""
    i = np.arange(n, dtype=float)
    d = np.minimum(i, n - 1 - i)
    w = np.exp(-lam * d) if lam > 0.0 else np.ones_like(d)
    return w / w.sum()


def sym_power_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law: a_i ∝ (min(i, n‑1‑i)+1)^(-p)."""
    i = np.arange(n, dtype=float)
    d = np.minimum(i, n - 1 - i) + 1.0
    w = d ** (-p) if p > 0.0 else np.ones_like(d)
    return w / w.sum()


def beta_symmetric_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta‑shape: a_i ∝ (i+1)^(α‑1) * (n‑i)^(α‑1),
    with α>0.  For α<1 the distribution is U‑shaped (mass at the ends).
    """
    i = np.arange(n, dtype=float) + 1.0
    w = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (alpha - 1.0))
    return w / w.sum()

# Scanning of parameterised families
# ----------------------------------------------------------------------


def scan_families(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """Search the five families, return the best vector (or None)."""
    best_a = None
    best_sc = float("inf")

    # 1. Exponential (monotone)
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -1, 120)))
    for lam in lam_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # 2. Power‑law (monotone)
    p_grid = np.linspace(0.0, 6.0, 120)
    for p in p_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # 3. Symmetric exponential
    lam_grid2 = np.logspace(-5, -1, 120)
    for lam in lam_grid2:
        if time.time() >= deadline:
            return best_a, best_sc
        a = sym_exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # 4. Symmetric power‑law
    p_grid2 = np.linspace(0.0, 6.0, 120)
    for p in p_grid2:
        if time.time() >= deadline:
            return best_a, best_sc
        a = sym_power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # 5. Symmetric Beta‑shape
    alpha_grid = np.linspace(0.1, 3.0, 120)
    for alpha in alpha_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        a = beta_symmetric_shape(n, alpha)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Adam optimisation of the smooth max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 0.5,
    beta_end: float = 5000.0,
    lr: float = 0.12,
    b1: float = 0.9,
    b2: float = 0.999,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Optimise the smoothed max objective using Adam on logits `u`
    (softmax parametrisation).  Returns the best exact‑objective vector.
    """
    n = start_a.shape[0]
    # initialise logits u so that softmax(u) == start_a (they already sum to 1)
    u = np.log(start_a + eps)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = start_a.copy()
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute a = softmax(u) -----
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        a = exp_u / exp_u.sum()  # ∑ a_i = 1, all >0

        # ----- smooth max via log‑sum‑exp -----
        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        # temperature schedule (exponential increase)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        # weights for the smooth max
        w = np.exp(beta * (conv - maxc))                # stable: maxc subtracted
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)            # length n + (2n‑1) – 1 = 3n‑2
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]       # size n

        # ----- chain rule to logits u (softmax) -----
        grad_dot = np.dot(a, grad_a)                     # scalar ⟨a,∇_a⟩
        grad_u = a * (grad_a - grad_dot)                # ∇_u f

        # ----- Adam update -----
        m = b1 * m + (1.0 - b1) * grad_u
        v = b2 * v + (1.0 - b2) * (grad_u ** 2)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional exact evaluation -----
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    final_a = np.exp(u - np.max(u))
    final_a /= final_a.sum()
    sc = compute_score(final_a)
    if sc < best_sc:
        best_sc = sc
        best_a = final_a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 3_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk: swaps or transfers a small packet of mass.
    The packet size adapts based on recent success.
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # start with a modest transfer fraction
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        # project back onto the probability simplex
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            # enlarge step_scale modestly after success
            step_scale = min(step_scale * 1.09, 0.5)
        else:
            # gradually shrink step_scale after failures
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc
