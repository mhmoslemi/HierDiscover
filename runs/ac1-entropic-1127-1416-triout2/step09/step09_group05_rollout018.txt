def projected_gradient_descent(a, deadline, max_iters=100_000, eta0=0.1):
    a = a.copy()
    n = a.shape[0]
    best_sc = compute_score(a)
    best_a = a.copy()
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # find max conv and its index
        conv = convolution_fft(a, a)
        max_c = conv.max()
        k = np.argmax(conv)
        # gradient
        a_rev = a[::-1]
        # conv_rev: a_rev convolved with indicator vector at position k?
        # gradient = 2 * shift of a_rev
        grad = np.zeros_like(a)
        # for each j compute a_{k-j} if in range
        # this can be done via slicing:
        # For j from max(0, k-(n-1)) to min(k, n-1)
        j_start = max(0, k-(n-1))
        j_end = min(k, n-1)
        # relate: grad[j] = 2 * a[k-j]
        grad_slice_indices = slice(j_start, j_end+1)
        a_slice = a[k - np.arange(j_start, j_end+1)]
        grad[grad_slice_indices] = 2 * a_slice
        # update step size

            a /= s
        # evaluate occasionally
        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

from scipy.optimize import linprog
def solve_lp_for_b(a, M, deadline):
    n = len(a)
    # variables: b_0..b_{n-1}.
    # objective minimize -sum(b) (i.e., maximize sum(b)).
    c = np.full(n, -1.0)  # negative because default is minimize.
    # constraints:
    # For each k: sum_i a_i * b_{k-i} <= M
    # Build A_ub of shape (2n-1, n)
    rows = []
    data = []
    col_idx = []
    # For k = 0..2n-2
    for k in range(2*n-1):
        start_i = max(0, k-(n-1))
        end_i = min(k, n-1)
        # for each i in [start_i, end_i], j = k-i
        # coefficient for b_j is a_i
        # we can fill j = k - i
        for i in range(start_i, end_i+1):
            j = k - i
            rows.append(k)
            col_idx.append(j)
            data.append(a[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, col_idx)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None, None
    b = res.x
    sum_b = b.sum()
    return b, sum_b
```

def build_convolution_constraint_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)

        for i in range(i_start, i_end+1):

"""
Search for a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

as evaluated by `evaluate_sequence`.  The algorithm combines several ideas:

1.  Fast scan of simple 1‑parameter families (exponential, power‑law,
    edge‑exponential) to obtain a decent starting vector.
2.  A few iterations of a linear‑programming “alternating” step:
    given the current vector f, solve
        max   Σ b_i
        s.t.  f * b ≤ M   (M = max(f*f)),   b ≥ 0
    (the convolution constraint is linear in b).  The solution b is
    rescaled to sum = 1 and mixed into f (mirroring the literature
    algorithm).  This dramatically flattens the autocorrelation.
3.  A long exponentiated‑gradient optimisation of a smooth
    log‑sum‑exp approximation of the max‑convolution.
4.  A stochastic hill‑climb (simulated‑annealing style) that moves
    small packets of mass between positions while keeping only
    improving moves.
5.  Multiple random restarts (Dirichlet initialisations) are tried
    within the remaining time budget; the best sequence found is
    returned.

All phases respect the overall wall‑clock budget (default 1000 s).  The
resulting sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np
import scipy.sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (both inputs 1‑D)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector normalised to sum = 1.
    (Same as the value returned by `evaluate_sequence` for such a vector.)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def edge_exponential_shape(n: int, lam: float) -> np.ndarray:
    """
    Symmetric “V” shape: a_i ∝ exp(-lam * min(i, n-1-i)).
    Concentrates mass near the two ends.
    """
    i = np.arange(n, dtype=float)
    d = np.minimum(i, n - 1 - i)
    w = np.exp(-lam * d) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()

# Scan simple families (including the edge‑exponential)
# ----------------------------------------------------------------------


def scan_families(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """
    Exhaustively scan exponential, power‑law and edge‑exponential families.
    Returns the best vector found and its exact score.
    """
    best_a = None
    best_sc = float("inf")

    # exponential λ grid (including λ=0 → uniform)
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -2, 200)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # power‑law p grid (p=0 → uniform)
    p_grid = np.linspace(0.0, 6.0, 200)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # edge‑exponential λ grid
    lam_grid2 = np.logspace(-5, -1, 120)
    for lam in lam_grid2:
        if time.time() >= deadline:
            break
        a = edge_exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # a few random Dirichlet starts (sparseish)
    for _ in range(5):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n) * 0.5)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation of the smoothed max
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 0.5,
    beta_end: float = 2000.0,
    eta0: float = 0.6,
    eps: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed log‑sum‑exp approximation of the max‑convolution
    using a KL‑mirror (exponentiated‑gradient) step.  β is annealed
    exponentially; step size η₀/√t.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weights (log‑sum‑exp)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stabilised exponentials
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smoothed objective
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[:n]

        # step size
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update and renormalisation
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

# Linear‑programming alternating step
# ----------------------------------------------------------------------


def build_conv_constraint_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Construct a sparse matrix A such that A @ b = (a * b)  (full convolution).
    Each row k corresponds to the coefficient vector for (a*b)_k.
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []

    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start to i_end (inclusive)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])

    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_b(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Maximise Σ b_i subject to a * b ≤ M (element‑wise) and b ≥ 0.
    Returns the optimal b (may be all zeros if infeasible) or None.
    """
    A = build_conv_constraint_matrix(a)
    b_ub = np.full(A.shape[0], M, dtype=float)
    c = -np.ones(a.shape[0], dtype=float)          # minimise –Σb  → maximise Σb

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs")
    if not res.success:
        return None
    return res.x


def alternating_optimize(start_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a few alternating LP updates (as described in the problem statement).
    Returns the best vector seen and its exact score.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    step = 0.5  # initial mixing factor
    for it in range(8):                     # hard‑coded small number of outer iterations
        if time.time() >= deadline - 2.0:
            break

        conv = convolution_fft(a, a)
        M = conv.max()
        b = solve_lp_for_b(a, M)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b == 0.0:
            break
        g = b / sum_b                         # renormalise to sum = 1
        # convex combination and re‑normalise (should already be normalised)
        a = (1.0 - step) * a + step * g
        a = np.maximum(a, 0.0)
        a /= a.sum()

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        step *= 0.9                           # slowly reduce the mixing amount

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simple simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray, deadline: float,
               max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Greedy random walk:
        - with prob 0.5 swap two entries,
        - otherwise transfer a fraction of mass between two indices.
    Only moves that improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_sc

        if time.time() < deadline - 5.0:
            a_fin, sc_fin = hill_climb(np.array(best_seq, dtype=float), deadline, max_steps=1_000_000)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.tolist()
                print(f"[{time.time():.1f}] hill‑climb after restart improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a short uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
