def solve_lp_max_sum_scipy(A, M, warm_start=None):
    n = A.shape[1]
    c = -np.ones(n)   # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    # A is csr; convert to appropriate format for hiGHs
    result = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M), bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5})
    if result.success:
        return result.x
    else:
        return None

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
* **Stage 0 – fallback** : if a previously discovered sequence (`height_sequence_1`)
  exists we start from it.
* **Stage 1 – small LP refinement** : on a modest size (n≈512) we repeatedly solve
  the linear programme  

        maximise   Σ b_j
        s.t.      (a ⊛ b)_k ≤ M      for all k
                   b_j ≥ 0

  where M = max(a⊛a).  The LP is solved with SciPy’s HiGHS driver
  (very fast for the small dense Toeplitz matrix).  A convex combination  
  `a←(1‑t)·a + t·b` (followed by projection onto the probability simplex) never
  raises the worst convolution entry, so each successful LP step strictly improves
  the objective.
* **Stage 2 – up‑scale** : the refined short vector is linearly interpolated to a
  larger length (chosen according to the remaining time).  After interpolation we
  project onto the simplex again.
* **Stage 3 – smooth Adam optimisation** : minimise a log‑sum‑exp (soft‑max)
  surrogate of `max(conv(a,a))`.  The smoothing parameter β is annealed from a
  modest value (≈5) to a large one (≈200) so the optimisation first sees a
  smooth landscape and then focuses on the true maximum.  After every step the
  iterate is projected onto the simplex.
* **Stage 4 – exact sub‑gradient polishing** : using the genuine max‑convolution
  we perform projected sub‑gradient descent (`a←a‑η·g`) with a diminishing step
  size.
* **Stage 5 – stochastic hill‑climb** : cheap random moves
  (swap two entries or shift a fraction of mass) are tried; moves that improve
  the objective are accepted and the step‑size is adapted.

All stages respect the global wall‑clock budget; the best sequence seen
throughout the run is returned (as a plain Python list of floats, already
normalised to sum‑to‑one).  The code contains no nested closures or lambdas,
uses only top‑level functions, and relies on NumPy, SciPy (for `linprog`) and
standard library modules.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for vectors already on the probability simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution matrix (Toeplitz) for the LP
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix A such that A @ b = conv(f, b)
    (linear convolution, zero‑padded).
    """
    n = f.shape[0]
    # ensure all entries are strictly positive – otherwise a column would be all zeros
    eps = 1e-12
    f_pos = np.maximum(f, eps)

    # each column j contains the vector f shifted by j rows
    diags = [f_pos] * n
    offsets = np.arange(n)           # 0,1,…,n‑1
    A = sp.diags(diagonals=diags, offsets=offsets,
                 shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum(A: sp.csr_matrix, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    c = -np.ones(n)                     # maximise sum(b)  ⇒  minimise -sum(b)
    ub = np.full(A.shape[0], M)         # right‑hand side of each inequality
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=ub,
                      bounds=bounds,
                      method='highs',
                      options={"presolve": True, "time_limit": 5})
    except Exception:      # pragma: no cover
        return None

    if not res.success:
        return None

    b = res.x
    # Clip spurious tiny negatives caused by numerics
    b = np.maximum(b, 0.0)
    return b

# Optimisation phases
# ----------------------------------------------------------------------


def lp_refinement(a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 15,
                  mix_factor: float = 0.22) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑based improvement of a short vector.
    ``mix_factor`` is the initial weight t in a←(1‑t)a + t b.
    """
    best = a.copy()
    best_score = compute_score(best)
    n = a.shape[0]

    for it in range(max_iter):
        if time.time() >= time_deadline:
            break

        # current worst convolution entry
        conv = convolution_fft(best, best)
        M = float(conv.max())

        # build Toeplitz matrix from the current vector
        A = build_convolution_matrix(best)

        b_opt = solve_lp_max_sum(A, M)
        if b_opt is None:
            print("[LP] solver failed on iteration", it + 1)
            break

        # normalise partner vector (still on simplex)
        b_opt = project_to_simplex(b_opt)

        # convex combination, then project back onto simplex
        cand = (1.0 - mix_factor) * best + mix_factor * b_opt
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        if cand_score < best_score * 0.9999:      # require a noticeable improvement
            best = cand
            best_score = cand_score
            print(f"[LP {it+1}] improved score = {best_score:.6f}")
        else:
            # shrink the mixing factor if we are stagnating
            mix_factor = max(mix_factor * 0.8, 0.02)

    return best, best_score


def smooth_adam_optimize(start_a: np.ndarray,
                         time_deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 5.0,
                         beta_end: float = 200.0,
                         step0: float = 0.9,
                         adam_beta1: float = 0.9,
                         adam_beta2: float = 0.999,
                         adam_eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    The smoothing parameter β is annealed geometrically.
    """
    a = project_to_simplex(start_a.astype(float).copy())
    n = a.shape[0]
    m = np.zeros_like(a)           # first moment
    v = np.zeros_like(a)           # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β (geometric progression)
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights (log‑sum‑exp trick)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient = 2 * (rev(a) * w) restricted to n entries
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)          # length = 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional check (every few hundred iterations)
        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Adam] iteration {it} – new best = {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


def subgradient_optimize(init_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 300_000,
                         step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()
            print(f"[Subgrad] iteration {t} – improved to {best_score:.6f}")

        a = a_new

        # occasional jitter to escape flat regions
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


def hill_climb(seq: np.ndarray,
               time_deadline: float,
               max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic moves (swap or fractional transfer) with simplex projection.
    """
    a = seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10   # initial fraction of mass moved

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_score


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation to a longer length, followed by simplex projection.
    """
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


def central_binomial_coeffs(n: int) -> np.ndarray:
    """
    Truncated central binomial coefficients a_i = C(2i,i)/4^i,
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a = project_to_simplex(a)
    return a

        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        # normalise to sum‑to‑one for internal evaluation
        fb = np.array(fallback, dtype=float)
        if fb.sum() == 0:
            fb = np.full_like(fb, 1.0 / fb.shape[0])
        else:
            fb = fb / fb.sum()
        fb_score = compute_score(fb)
        print(f"[Fallback] internal score = {fb_score:.6f}")
        best_seq = fb.tolist()
        best_score = fb_score
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a final length according to the remaining budget
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 700:
        final_len = 4096
    elif time_left > 400:
        final_len = 2048
    else:
        final_len = 1024

    # ------------------------------------------------------------------
    # 2.  Small‑scale LP refinement (n≈512)
    # ------------------------------------------------------------------
    n_small = 512
    a_small = central_binomial_coeffs(n_small)

    # allocate a modest slice of the budget for the LP stage
    lp_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.10)
    a_small_refined, score_small = lp_refinement(
        a_small,
        time_deadline=lp_deadline,
        max_iter=12,
        mix_factor=0.24,
    )
    print(f"[LP] finished – small score = {score_small:.6f}")

    # ------------------------------------------------------------------
    # 3.  Upscale to the final length
    # ------------------------------------------------------------------
    a_big = upscale_vector(a_small_refined, final_len)

    # ------------------------------------------------------------------
    # 4.  Smooth Adam optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    smooth_deadline = time.time() + (deadline - time.time()) * 0.60
    if time.time() < smooth_deadline:
        a_smooth, smooth_score = smooth_adam_optimize(
            a_big,
            time_deadline=smooth_deadline,
            max_iters=500_000,
            beta_start=6.0,
            beta_end=200.0,
            step0=0.85,
        )
        print(f"[Adam] finished – score = {smooth_score:.6f}")
        if smooth_score < best_score:
            best_score = smooth_score
            best_seq = a_smooth.tolist()

    # ------------------------------------------------------------------
    # 5.  Projected sub‑gradient polishing
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.20
    if time.time() < subgrad_deadline and best_seq is not None:
        refined_a, refined_score = subgradient_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=subgrad_deadline,
            max_iter=300_000,
            step0=0.06,
        )
        print(f"[Subgrad] finished – score = {refined_score:.6f}")
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_a.tolist()

    # ------------------------------------------------------------------
    # 6.  Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline and best_seq is not None:
        polished_seq, polished_score = hill_climb(
            np.array(best_seq, dtype=float),
            time_deadline=hill_deadline,
            max_steps=800_000,
        )
        print(f"[HillClimb] finished – score = {polished_score:.6f}")
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.tolist()

    # ------------------------------------------------------------------
    # 7.  Fallback to a simple uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = compute_score(np.array(best_seq, dtype=float))
        print("[Emergency] using uniform fallback")

    # ------------------------------------------------------------------
    # 8.  Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch completed – final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
