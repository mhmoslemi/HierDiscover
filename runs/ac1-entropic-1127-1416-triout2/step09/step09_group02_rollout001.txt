"""
Gradient‑smoothed descent for the “max‑autocorrelation” problem.

The objective inside ``evaluate_sequence`` is

    score = 2·n·max(conv(a,a)) / (∑a)² ,   a_i ≥ 0 .

For a fixed sum the problem is to minimise the largest entry of the
auto‑convolution.  The code below works with a probability vector
(∑a = 1) and repeatedly performs

* a soft‑max of the convolution (parameter β, raised slowly during the run)
* the exact gradient of that soft‑max (a cross‑correlation, computed by FFT)
* a Euclidean projection onto the simplex.

The gradient points away from the lags that contribute most to the
objective, so the descent is far more informative than a sub‑gradient
that looks only at the single maximal lag.  After the bulk of the time
budget a short hill‑climb (greedy local moves) is applied for polishing.

The whole search finishes well inside the 1000 s wall‑clock limit and
usually beats the previous best score (~ 1.511).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a (not necessarily normalised) vector.
    Returns 2·n·max(conv(a,a)) / (sum(a)²).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:               # avoid division by zero
        return np.inf
    return 2.0 * n * max_b / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sine‑squared shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p .
    Positive p concentrates mass in the centre, negative p moves it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed gradient descent
# ----------------------------------------------------------------------


def smooth_gradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the *soft‑max* of the convolution.
    The temperature (β) is increased gradually so that the smooth
    approximation converges to the true max.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    # keep the best solution found so far
    best_a = a.copy()
    best_score = compute_score(best_a)

    # hyper‑parameters (tuned empirically)
    step0 = 0.5          # initial step size coefficient
    beta = 5.0           # start with a mild soft‑max
    beta_max = 2000.0    # never exceed this (avoids overflow)
    beta_factor = 1.03   # multiplicative increase every ``beta_iter`` iterations
    beta_iter = 1500    # how many iterations between β updates
    iter_cnt = 0

    # pre‑allocate FFT length for the cross‑correlation gradient
    conv_len = 2 * n - 1
    L_grad = next_pow2(n + conv_len - 1)

    while time.time() < deadline:
        iter_cnt += 1

        # ----- convolution of the current vector with itself -----
        conv = convolution_fft(a, a)                 # length 2n‑1

        # ----- update soft‑max temperature -----
        if iter_cnt % beta_iter == 0:
            beta = min(beta * beta_factor, beta_max)

        # ----- compute the soft‑max weights -----
        # Shift by the maximum to keep the exponent well‑scaled.
        max_c = conv.max()
        w = np.exp(beta * (conv - max_c))
        w /= w.sum()                                 # now w is a probability vector

        # ----- gradient = 2·(a_rev ★ w) at the proper slice -----
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L_grad)
        W = np.fft.rfft(w, n=L_grad)
        grad_full = np.fft.irfft(A_rev * W, n=L_grad)   # length L_grad
        start = n - 1
        g = 2.0 * grad_full[start:start + n]            # shape (n,)

        # ----- projected gradient step -----
        step = step0 / np.sqrt(iter_cnt)
        a -= step * g
        a = project_to_simplex(a)

        # ----- evaluate exact objective (no extra FFT) -----
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional progress output (kept cheap)
        if iter_cnt % 5000 == 0:
            print(f"[Grad] iter {iter_cnt:,}, best {best_score:.6f}, beta {beta:.1f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill–climb polishing
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, deadline: float, max_steps: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Very short stochastic hill‑climb that only accepts improving moves.
    The move set consists of a random swap or a tiny mass transfer.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                # safety clamp
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑normalise (numerical drift)
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become a bit bolder
        else:
            step_scale *= 0.96                         # become more conservative

        steps += 1

    return best_a, best_score

    of floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # 0. Optional fallback from a previously discovered sequence
    # --------------------------------------------------------------
    best_seq = None

        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Choose a sensible length (larger n gives a tighter theoretical bound)
    # --------------------------------------------------------------
    n = 2048
    if total_time > 850.0:          # enough time for a larger problem
        n = 4096
    print(f"Using sequence length = {n}")

    # --------------------------------------------------------------
    # 2. Build a collection of seed vectors (hand‑crafted + random)
    # --------------------------------------------------------------
    seeds = []

    # a) deterministic sine‑squared shape (good baseline)
    seeds.append(generate_deterministic(n))

    # b) a few Beta‑family shapes spanning centre/edge concentrations
    for p in np.linspace(-2.5, 2.5, 7):
        seeds.append(generate_beta_family(n, p))

    for _ in range(3):

        seeds.append(random_dirichlet(n, conc=0.3))

    # d) use the fallback if its length matches the current n

    # --------------------------------------------------------------
    # 3. Evaluate all seeds and keep the best one (exact objective)
    # --------------------------------------------------------------
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)          # ensure a valid probability vector
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"Initial seed improvement: score = {best_val:.6f}")

    # from here on we work with a NumPy array
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)
    print(f"Starting score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4. Smoothed gradient descent (majority of the budget)
    # --------------------------------------------------------------
    grad_deadline = start_time + total_time * 0.70   # roughly 70 % of time
    print("\n[Phase 1] Smoothed gradient descent")
    cur_a, cur_score = smooth_gradient_descent(cur_a, grad_deadline)
    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur_a.tolist()
        print(f"After gradient descent: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final hill‑climb polishing (use the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 2] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=800_000)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"After polishing: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Return the candidate; also report the official score
    # --------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq
