def smooth_descent(a, start_score, deadline):
    best_a = a.copy()
    best_score = start_score
    t = 1
    alpha = 5.0
    alpha_max = 60.0
    step0 = 0.1
    n = len(a)
    L = 2*n-1
    while time.time() < deadline:
        # compute conv
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        # softmax weights
        # shift for numeric stability
        shifted = conv - max_conv  # <= 0
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[L-1 : L-n-1 : -1]

        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        # projection onto simplex
        a_new = project_to_simplex(a_new)

        # compute new score
        cur_score = compute_score(a_new)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        # Acceptance not needed (deterministic descent) but may step overshoot
        a = a_new
        # update alpha gradually
        alpha = min(alpha_max, alpha * 1.001)  # increase slowly
        t += 1

        if t % 5000 == 0:
            # jitter to escape
            jitter = np.random.randn(n) * step0 * 0.01 / np.sqrt(t)
            a = project_to_simplex(a + jitter)

    return best_a, best_score
```

"""
Non‑convex minimisation of  2·n·max(conv(a,a))  on the probability simplex.

The search proceeds in three stages:

1. **Diversified seeding** – a large collection of hand‑crafted
   windows (Hann, Hamming, Blackman, Bartlett, Kaiser) and a dense sweep
   of the β‑family a_i∝((i+1)(n‑i))^p together with many Dirichlet draws.
   The best seed (according to the exact objective) initializes the
   optimisation.

2. **Smooth‑max gradient descent** – the nonsmooth `max` is replaced by a
   soft‑max with a temperature `α`.  For a given `α` the gradient of the
   smoothed objective can be written in closed form as a convolution of
   the current vector with the reversed soft‑max weights, enabling an
   O(n log n) update via FFT.  `α` is slowly increased so that the
   optimisation follows the true objective while retaining a useful
   gradient signal.

3. **Greedy polishing** – a short stochastic hill‑climb (swap / local
   mass transfer) that only accepts improvements, finishing the search
   before the wall‑clock deadline.

All operations respect the 1000 s budget; the returned list of floats is
ready for `evaluate_sequence`.  The routine automatically picks a
problem size (2048 → 4096 → 8192) according to the available time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*b using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_sine_squared(n: int) -> np.ndarray:
    """a_i = sin^2(pi*(i+1)/(n+1)), normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))**p  (symmetric polynomial)."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration `conc` (conc=1 → uniform)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def generate_window(n: int, name: str, param: float = None) -> np.ndarray:
    """Standard signal‑processing windows, normalised to unit sum."""
    if name == "bartlett":
        w = np.bartlett(n)
    elif name == "blackman":
        w = np.blackman(n)
    elif name == "hamming":
        w = np.hamming(n)
    elif name in ("hann", "hanning"):
        w = np.hanning(n)
    elif name == "kaiser":
        beta = param if param is not None else 0.0
        w = np.kaiser(n, beta)
    else:
        raise ValueError(f"unknown window {name}")

    w = np.maximum(w, 0.0)          # guard against tiny negativities
    if w.sum() == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    alpha_start: float = 5.0,
    alpha_max: float = 50.0,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Deterministic descent on the soft‑max approximation of the max‑convolution.
    Returns the best vector seen and its (exact) score.
    """
    n = a.shape[0]
    L = 2 * n - 1          # length of the auto‑convolution
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    t = 1
    alpha = alpha_start
    # geometric increase of the temperature
    alpha_factor = np.exp(np.log(alpha_max / alpha_start) / 200_000)

    while time.time() < deadline:
        # auto‑convolution
        conv = convolution_fft(cur_a, cur_a)          # length L
        max_conv = conv.max()

        # soft‑max (stable version)
        shifted = conv - max_conv                     # ≤ 0
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / L)
        else:
            w /= w_sum

        # gradient = 2 * (a * w_rev)[L‑1 : L‑n‑1 : -1]
        w_rev = w[::-1]
        conv_rev = convolution_fft(cur_a, w_rev)     # length = n+L‑1
        grad = 2.0 * conv_rev[L - 1 : L - n - 1 : -1]

        # step size with sqrt‑decay
        step = step0 / np.sqrt(t)
        cur_a = cur_a - step * grad
        cur_a = project_to_simplex(cur_a)

        # occasional jitter to escape shallow plateaux
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            cur_a = project_to_simplex(cur_a + jitter)

        # exact objective (cheap – we already have conv.max())
        cur_score = 2.0 * n * float(max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # temperature schedule
        alpha = min(alpha_max, alpha * alpha_factor)

        t += 1

        # safety abort when iterations explode
        if t > 2_000_000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic improvement that only accepts moves that lower the exact score.
    Moves are either a swap of two entries or a small local mass transfer.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.07

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # numerical re‑normalisation (the move conserves mass up to rounding)
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)   # become bolder
        else:
            step_scale *= 0.96                         # become more cautious

        steps += 1

    return best_a, best_score
