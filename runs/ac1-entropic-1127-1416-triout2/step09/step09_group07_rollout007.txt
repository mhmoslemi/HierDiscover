def adam_exp_gradient_optimize(start_a, deadline, max_iters=120_000, beta_start=0.5, beta_end=300.0, lr_start=0.6):
    a = np.maximum(start_a, 1e-14)
    a /= a.sum()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # compute conv
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute softmax weights
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # compute gradient
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n-1:n-1+n]  # slice to length n

        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr_start / np.sqrt(it)

        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    # final eval
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

def annealing_hill_climb(start_a, deadline, inc_factor=1.02, decay_factor=0.98, recompute_interval=50000):
    a = start_a.copy()
    n = a.shape[0]
    # enforce sum=1
    a = a / a.sum()
    conv = convolution_fft(a, a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_max = cur_max
    best_score = cur_score

    step_scale = 0.05  # initial delta fraction
    max_step = 0.5
    min_step = 1e-4
    start = time.time()
    it = 0

    # temperature schedule
    total_time = deadline - start
    while time.time() < deadline:
        elapsed = time.time() - start
        t_frac = elapsed / total_time
        # T scaling relative to typical deltas: maybe T = cur_max * (0.1 * (1 - t_frac) + 0.001)
        T = cur_max * (0.2 * (1 - t_frac) + 0.001)

        # pick src, dst random
        src = np.random.randint(n)
        if a[src] == 0:
            continue
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1
        # delta random fraction up to step_scale
        delta = step_scale * a[src] * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental convolution update
        conv_cand = conv.copy()
        # linear shifts
        conv_cand[src:src+n] -= 2.0 * delta * a
        conv_cand[dst:dst+n] += 2.0 * delta * a
        # quadratic corrections
        conv_cand[2*src] += delta * delta
        conv_cand[2*dst] += delta * delta
        conv_cand[src+dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max
        delta_score = cand_score - cur_score

        accept = False
        if delta_score < 0:
            accept = True
        else:
            # accept with probability exp(-Δ/T)
            prob = np.exp(-delta_score / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            # update state
            a[src] -= delta
            a[dst] += delta
            # enforce non-neg (numerical)
            a[a < 0] = 0.0
            # renormalize to sum=1 due to rounding errors
            total = a.sum()
            if total != 0:
                a /= total
            conv = conv_cand
            cur_max = cand_max
            cur_score = cand_score
            # adjust step_scale
            step_scale = min(step_scale * inc_factor, max_step)
            # record best
            if cand_score < best_score:
                best_score = cand_score
                best_max = cand_max
                best_a = a.copy()
        else:
            # rejected
            step_scale = max(step_scale * decay_factor, min_step)

        it += 1
        if it % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

    return best_a, best_score

Propose a low‑score step‑function sequence.

The search works in several stages:

1. **Analytic α‑grid** – a one‑parameter family
   a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ is scanned and the best vector (on the
   simplex) is kept.

2. **Smooth Adam‑optimisation** – minimise a log‑sum‑exp surrogate of the
   max‑convolution using an Adam optimiser (KL‑mirror).  The smoothing
   parameter β is exponentially annealed; after each step the vector is
   projected onto the probability simplex.

3. **Strict hill‑climb** – a fast random‑walk that only accepts strictly
   improving mass‑transfer moves, updating the convolution incrementally.

4. **Upscaling & annealed hill‑climb** – the best vector is linearly
   interpolated to a larger length (up to 8192).  A short Adam phase
   refines the shape, then a simulated‑annealing walk (accepting
   occasional worsening moves) performs the final polish.

All phases respect the 1000 s wall‑clock budget, and the best sequence

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector whose entries sum to 1.
    This matches evaluate_sequence when the sum == 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def alpha_start(n: int, alpha: float) -> np.ndarray:

    i = np.arange(n, dtype=float)

    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a

    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Strict hill‑climb (improving moves only)
# ----------------------------------------------------------------------


def fast_hill_climb(start_a: np.ndarray,
                    deadline: float,
                    inc_factor: float = 1.07,
                    decay_factor: float = 0.97,
                    recompute_interval: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Random‑walk polishing that only accepts strictly improving moves.
    Convolution is updated incrementally to avoid an FFT at each step.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)               # exact convolution
    cur_max = conv.max()
    cur_score = 2.0 * n * float(cur_max)

    best_a = a.copy()
    best_score = cur_score
    step_scale = 0.12

    it = 0
    while time.time() < deadline:
        # choose src (donor) and dst (receiver)
        src = np.random.randint(n)
        if a[src] == 0.0:
            continue
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        delta = step_scale * a[src] * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * float(cand_max)

        if cand_score < best_score:
            # Accept move (strict improvement)
            best_score = cand_score
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            best_a = a.copy()
            step_scale = min(step_scale * inc_factor, 0.5)
        else:
            # reject – shrink step
            step_scale *= decay_factor

        it += 1
        if it % recompute_interval == 0:
            conv = convolution_fft(a, a)   # full recompute to avoid drift

    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Annealed hill‑climb (allows occasional uphill moves)
# ----------------------------------------------------------------------


def annealing_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         inc_factor: float = 1.03,
                         decay_factor: float = 0.99,
                         recompute_interval: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing random walk on the simplex.
    Uses the same O(n) incremental convolution update as the strict
    hill‑climb, but accepts worsening moves with probability
    exp(−Δscore/T).  Temperature T decays linearly to a very small
    floor.
    """
    a = start_a.copy()
    n = a.size
    a /= a.sum()                     # ensure exact sum==1
    conv = convolution_fft(a, a)
    cur_max = conv.max()
    cur_score = 2.0 * n * float(cur_max)

    best_a = a.copy()
    best_score = cur_score

    step_scale = 0.08
    max_step = 0.5
    min_step = 1e-5

    start = time.time()
    total_time = deadline - start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        t_frac = elapsed / total_time

        # temperature roughly 20 % of current max at start, decaying to 0.1 %
        T = cur_max * (0.2 * (1.0 - t_frac) + 0.001)

        src = np.random.randint(n)
        if a[src] == 0.0:
            continue
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        delta = step_scale * a[src] * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental convolution update
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * float(cand_max)
        delta_score = cand_score - cur_score

        # acceptance test
        accept = False
        if delta_score < 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit move
            a[src] -= delta
            a[dst] += delta
            # re‑normalise to guard against numerical drift
            total = a.sum()
            if total > 0.0:
                a /= total
            conv = conv_cand
            cur_max = cand_max
            cur_score = cand_score
            step_scale = min(step_scale * inc_factor, max_step)

            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
        else:
            step_scale = max(step_scale * decay_factor, min_step)

        it += 1
        if it % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * float(cur_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# 5. Upscaling helper
# ----------------------------------------------------------------------


def upscale_shape(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(a) to new_n.
    Result is projected back onto the simplex (sum == 1).
    """
    old_n = a.shape[0]
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows a multi‑stage plan and respects the 1000 s wall‑clock limit.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. Fallback (if a global `height_sequence_1` is present)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic start on a modest length
    # ------------------------------------------------------------------
    n0 = 2048
    print(f"[{time.time():.1f}] starting analytic α‑grid, n={n0}")
    a_alpha, sc_alpha = find_best_alpha_start(n0, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam‑based smooth optimisation (still on n0)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] Adam smooth optimisation, n={n0}")
        a0 = np.array(best_seq, dtype=float)
        a0 = project_to_simplex(a0)
        a_adam, sc_adam = adam_exp_gradient_optimize(
            a0,
            deadline - 150.0,
            max_iters=60_000,
            beta_start=0.5,
            beta_end=300.0,
            lr_start=0.6,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Strict hill‑climb on the small vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        print(f"[{time.time():.1f}] strict hill‑climb, n={n0}")
        a_hc, sc_hc = fast_hill_climb(
            np.array(best_seq, dtype=float),
            deadline - 80.0,
        )
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.tolist()
            print(f"[{time.time():.1f}] after strict hill‑climb = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Upscale & annealed hill‑climb (majority of the remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        # pick a larger length based on how much time is left
        remaining = deadline - time.time()
        if remaining > 400:
            n_big = 8192
        elif remaining > 200:
            n_big = 4096
        else:
            n_big = 3072

        print(f"[{time.time():.1f}] upscaling to n={n_big}")
        a_big = upscale_shape(np.array(best_seq, dtype=float), n_big)

        # brief Adam refinement on the large vector
        if time.time() < deadline - 30.0:
            print(f"[{time.time():.1f}] Adam refinement on upscaled vector")
            a_big, sc_big_adam = adam_exp_gradient_optimize(
                a_big,
                deadline - 20.0,
                max_iters=5_000,          # cheap refinement
                beta_start=1.0,
                beta_end=500.0,
                lr_start=0.4,
            )
            # keep it if better (unlikely to be worse)
            if sc_big_adam < best_score:
                best_score = sc_big_adam
                best_seq = a_big.tolist()
                print(f"[{time.time():.1f}] after Adam (large) = {best_score:.6f}")

        # final annealed hill‑climb
        if time.time() < deadline - 5.0:
            print(f"[{time.time():.1f}] annealed hill‑climb on n={n_big}")
            a_final, sc_final = annealing_hill_climb(
                a_big,
                deadline - 2.0,
            )
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.tolist()
                print(f"[{time.time():.1f}] annealed hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback to uniform short vector if everything else failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # return a plain Python list of floats (as required by the harness)
    return [float(x) for x in best_seq]
