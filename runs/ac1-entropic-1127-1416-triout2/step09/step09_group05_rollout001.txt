"""
Optimisation of a non‑negative step function to minimise
      evaluate_sequence(sequence)

The algorithm consists of four phases:

1. **Parameter sweep** – a large collection of one‑ and two‑parameter
   families (exponential, power‑law, beta‑distribution, triangular)
   is scanned on the chosen length `n`.  The best candidate (lowest exact
   score) is kept.

2. **Smoothed sub‑gradient optimisation** – we minimise a log‑sum‑exp
   approximation of the max‑convolution using an Adam optimiser.
   The smoothing parameter β is exponentially annealed from a modest
   starting value to a very large final value (≈10⁵), so the gradient
   approaches a true sub‑gradient of the original max‑convolution.

3. **Stochastic hill‑climbing with simulated annealing** – a greedy
   random walk that occasionally accepts worsening moves according to
   a cooling schedule.  This phase explores the neighbourhood of the
   best point found by the Adam optimiser.

4. **Return** the best normalised sequence (list of Python floats).

All phases respect the overall wall‑clock budget (default 1000 s) and
periodically check the deadline.  The implementation works with any
vector length up to a few‑ten‑thousands (the default is 8192 when time
permits) and uses only NumPy and the already‑imported `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using real FFT.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector normalised to sum = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise to exactly 1
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / v.shape[0]
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Parameterised families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta‑distribution shape.
    a_i ∝ ( (i+0.5)/n )**a  *  (1-(i+0.5)/n )**b
    """
    # centre the points to avoid exact 0 or 1
    x = (np.arange(n, dtype=float) + 0.5) / n
    # safe handling of negative exponents (x>0, 1-x>0)
    w = np.power(x, a) * np.power(1.0 - x, b)
    total = w.sum()
    if total == 0.0 or not np.isfinite(total):
        # fallback to uniform if overflow/underflow occurs
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular bump, peak at the centre, normalised."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    w = 1.0 - np.abs(i - centre) / (centre + 1.0)
    w = np.maximum(w, 0.0)
    return w / w.sum()


# ----------------------------------------------------------------------
# Scan the simple families (including 2‑parameter beta family)
# ----------------------------------------------------------------------


def scan_families(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """
    Exhaustively scan a collection of families and return the best vector.
    Returns (vector, score) or (None, inf) if time expires.
    """
    best_a = None
    best_sc = float("inf")

    # ----- exponential λ grid (including λ=0 → uniform) -----
    lam_grid = np.concatenate(
        (np.array([0.0]), np.logspace(-5, -2, 200))
    )
    for lam in lam_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # ----- power‑law p grid -----
    p_grid = np.linspace(0.0, 6.0, 200)
    for p in p_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # ----- triangular (single shaped) -----
    if time.time() < deadline:
        a = triangular_shape(n)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # ----- beta‑distribution 2‑parameter grid -----
    # parameters are chosen in a modest range that keeps the shape well‑behaved.
    a_vals = np.linspace(-0.9, 4.0, 30)
    b_vals = np.linspace(-0.9, 4.0, 30)
    for a_param in a_vals:
        if time.time() >= deadline:
            break
        for b_param in b_vals:
            if time.time() >= deadline:
                break
            a_vec = beta_shape(n, a_param, b_param)
            sc = compute_score(a_vec)
            if sc < best_sc:
                best_sc = sc
                best_a = a_vec.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Smoothed sub‑gradient optimisation using Adam
# ----------------------------------------------------------------------


def adam_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 400_000,
    beta_start: float = 5.0,
    beta_end: float = 1e5,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Minimises a log‑sum‑exp approximation of max(a*a) with Adam.
    The smoothing parameter β is exponentially annealed from `beta_start`
    to `beta_end`.  The iterates are projected onto the probability simplex.
    Returns the best vector seen (and its exact score).
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smoothed weights (log‑sum‑exp) ----
        conv = convolution_fft(a, a)            # length 2n‑1
        maxc = conv.max()
        # stabilise exponentials
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()                  # same length 2n‑1

        # ---- gradient of the smoothed max ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # slice of length n

        # ---- Adam update (minimisation) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---- project back onto simplex ----
        a = project_to_simplex(a)

        # ---- occasional exact evaluation ----
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final exact evaluation
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 3_000_000,
    start_T: float = 0.12,
    end_T: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk that occasionally accepts worsening moves,
    controlled by a temperature schedule T that decays exponentially.
    The move set consists of swapping two entries or transferring a
    fraction of mass from one position to another.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    steps = 0
    while steps < max_steps and time.time() < deadline:
        # temperature decreasing exponentially
        t = steps / max_steps
        T = start_T * (end_T / start_T) ** t

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small packet of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        delta_sc = sc - best_sc
        if delta_sc < 0 or np.log(np.random.rand() + 1e-20) < -delta_sc / (T + 1e-20):
            # accept move
            best_a = cand
            best_sc = sc
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The search proceeds through:
        1. quick scan of several parametric families,
        2. Adam optimisation of a smooth max‑convolution,
        3. stochastic hill‑climb polishing.
    The routine respects the wall‑clock budget (default 1000 s).
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to be sure we return before the hard timeout
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a workable length based on remaining budget
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Fallback to any existing global sequence
    # -------------------------------------------------
    fb = None
    try:
        fb_seq = globals()["height_sequence_1"]  # type: ignore
        fb_seq = [float(x) for x in fb_seq]
        fb = evaluate_sequence(fb_seq)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_seq
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Scan exponential, power, triangular and beta families
    # -------------------------------------------------
    scan_deadline = deadline - 30.0
    a_scan, sc_scan = scan_families(n, scan_deadline)
    if a_scan is not None and sc_scan < best_score:
        best_score = sc_scan
        best_seq = a_scan.tolist()
        print(f"[{time.time():.1f}] best after parameter scan = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Adam optimisation of the smoothed max
    # -------------------------------------------------
    # allocate a reasonable chunk of the remaining budget
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] starting Adam optimisation")
        adam_deadline = deadline - 150.0
        max_iters = 400_000  # will respect deadline internally
        a_opt, sc_opt = adam_optimize(
            np.array(best_seq, dtype=float),
            adam_deadline,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=1e5,
            lr=0.05,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Final stochastic hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        hill_deadline = deadline
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=float),
            hill_deadline,
            max_steps=3_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a short uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
