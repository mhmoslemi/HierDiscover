def lp_step(a, M):
    n = len(a)
    # Build A_ub where each row k (0..2n-2) has coefficients f[i] for g[j] when i+j=k.
    # Equivalent to A_ub[k, j] = a[k-j] if 0 <= k-j < n else 0.
    # Use sparse matrix.
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # i from i_min .. i_max, then j = k - i.
        # add for each i: coefficient a[i] for g_{k-i}.
        # Could iterate over i.
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Upper bound
    b = np.full(2*n-1, M)
    # Objective
    c = -np.ones(n)  # maximize sum(g)
    # Use highslp
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

#!/usr/bin/env python
"""
Search for a non‑negative step‑function (a list of heights) that minimizes

    2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.  The algorithm mixes several ideas:

1.  A broad sweep of simple analytic families (symmetric product,
    left‑heavy power‑law, exponential “U”, mixtures) on several problem
    sizes.  The best few candidates are kept.

2.  A multi‑restart smoothed‑max optimisation (Adam on a soft‑max
    parametrisation) that gradually sharpens the smoothing parameter β
    and drives the maximum of the self‑convolution down.

3.  An LP‑based improvement step (inspired by the reference alternating
    linear‑programming method).  For the current vector a we solve

        maximise   1ᵀg      (g ≥ 0)
        subject to  a * g ≤ M   (M = max(a*a))

    The solution g is renormalised and mixed with a.  This step reduces
    the peak in a systematic way.

4.  A cheap projected sub‑gradient descent that works directly on the
    true max‑objective.

5.  Symmetrisation (a ← (a+a[::-1])/2) and a final stochastic hill‑climb
    (simulated‑annealing style) to polish the result.

The whole routine respects a global wall‑clock budget (default 1000 s)
and returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a=1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, 1ᵀx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} (monotonically decreasing, normalised)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Convex mixture of the uniform vector and the exponential U‑shape.
    lam∈[0,1] weight of the uniform component.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a


def mixed_two_families(
    n: int, alpha: float, beta: float, lam: float
) -> np.ndarray:
    """
    Mixture of the symmetric‑product family (α) and the left‑heavy power
    family (β).  lam∈[0,1] is the mixture weight for the symmetric part.
    """
    a1 = symmetric_product_family(n, alpha)
    a2 = left_heavy_power_family(n, beta)
    a = lam * a1 + (1.0 - lam) * a2
    return a / a.sum()


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Sample a random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 800.0,
    lr: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).

    The smoothing β is increased exponentially from ``beta_start`` to
    ``beta_end``; the final large β approximates the true max.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    # Adam moments
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- schedule for β (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- current a and its convolution ----
        a = softmax(z)
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ---- gradient wrt a (see Chen et al. “Smoothed Max” derivation) ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 2n‑1
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ---- back‑prop through soft‑max ----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional exact evaluation ----
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheapest iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 1_200_000,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for 2·n·max(a*a)
    with a diminishing stepsize ∝ 1/√t and simplex projection.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑entry (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex.
    Swap or transfer mass between two coordinates, with a tiny temperature
    schedule that permits occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature proportional to current best score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            # accept
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step (optional, requires SciPy)
# ----------------------------------------------------------------------
def lp_improve(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 12,
    mix: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps.
    For the current vector a compute the maximal convolution value M.
    Solve   maximise   ∑ g_j     s.t.   a * g ≤ M,   g ≥ 0.
    The optimum g is renormalised to a probability vector and mixed with a.
    The routine stops when the time budget is exhausted or no further
    improvement is observed.
    """
    try:
        from scipy import sparse, optimize
    except Exception:  # SciPy not available – abort silently
        return a, compute_score(a)

    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = convolution_fft(best_a, best_a)
        M = conv.max()

        # Build the (2n‑1) × n constraint matrix for a * g ≤ M.
        # Row k contains a_i where i = k‑j and 0≤j<n.
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # loop over i where a_i contributes to position k
            for i in range(i_min, i_max + 1):
                rows.append(k)
                cols.append(k - i)          # j = k - i
                data.append(best_a[i])
        A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M)

        # Linear programme: minimise -1ᵀg  (i.e. maximise 1ᵀg)
        c = -np.ones(n)
        res = optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Renormalise to the simplex
        g_norm = g / sum_g

        # Mix and project back to the simplex
        a_new = (1.0 - mix) * best_a + mix * g_norm
        a_new = project_to_simplex(a_new)

        new_score = compute_score(a_new)
        if new_score < best_score - 1e-7:      # clear improvement
            best_a = a_new
            best_score = new_score
            print(f"[{time.time():.1f}] LP improve #{it+1} score = {best_score:.6f}")
        else:
            # No improvement → stop early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The routine respects the global time budget and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from a previous run
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. systematic exploration of families on several problem sizes
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]          # a spread of sizes
    for n in lengths:
        if time.time() >= deadline - 200.0:
            break
        print(f"[{time.time():.1f}] >>> exploring length n = {n}")

        # ---- a tiny helper to evaluate and possibly store a candidate ----
        def try_candidate(a_vec: np.ndarray, tag: str):
            nonlocal best_seq, best_score
            sc = compute_score(a_vec)
            if sc < best_score:
                best_score = sc
                best_seq = a_vec.tolist()
                print(f"[{time.time():.1f}] new best ({tag}) = {best_score:.6f}")

        # ----- symmetric product family (α) -----
        for alpha in np.linspace(0.0, 20.0, 101):
            if time.time() >= deadline - 30.0:
                break
            try_candidate(symmetric_product_family(n, alpha),
                           f"sym‑α={alpha:.2f}")

        # ----- left‑heavy power law (β) -----
        for beta in np.linspace(0.0, 20.0, 101):
            if time.time() >= deadline - 30.0:
                break
            try_candidate(left_heavy_power_family(n, beta),
                           f"left‑β={beta:.2f}")

        # ----- exponential U‑shape (γ) -----
        for gamma in np.linspace(0.0, 8.0, 41):
            if time.time() >= deadline - 30.0:
                break
            try_candidate(exp_u_family(n, gamma),
                           f"exp‑γ={gamma:.2f}")

        # ----- mixtures of symmetric product & left‑heavy (α,β,λ) -----
        for _ in range(120):          # random sampling in the 3‑D space
            if time.time() >= deadline - 30.0:
                break
            alpha = np.random.uniform(0.0, 20.0)
            beta = np.random.uniform(0.0, 20.0)
            lam = np.random.choice([0.0, 0.25, 0.5, 0.75, 1.0])
            mixed = mixed_two_families(n, alpha, beta, lam)
            try_candidate(mixed, f"mix‑αβλ={alpha:.1f},{beta:.1f},{lam}")

        # ----- convex mixtures of uniform & exponential U -----
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for gamma in np.linspace(0.0, 8.0, 17):
                if time.time() >= deadline - 30.0:
                    break
                try_candidate(mixed_uniform_u_family(n, lam, gamma),
                               f"mix‑U‑λ={lam:.2f},γ={gamma:.2f}")

        # ----- random initialisers (monotone & U) -----
        for _ in range(100):
            if time.time() >= deadline - 30.0:
                break
            vec = random_monotone_start(n) if np.random.rand() < 0.5 else random_u_start(n)
            try_candidate(vec, "rand‑starter")

    # ------------------------------------------------------------------
    # 2. LP‑based improvement on the current best vector (if possible)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        n_best = len(best_seq)
        print(f"[{time.time():.1f}] entering LP improvement (n={n_best})")
        a_lp, sc_lp = lp_improve(
            np.array(best_seq, dtype=np.float64),
            deadline - 120.0,
            max_iters=12,
            mix=0.5,
        )
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.tolist()
            print(f"[{time.time():.1f}] after LP best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑max refinement (Adam) with several restarts
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        n_best = len(best_seq)
        a0 = np.array(best_seq, dtype=np.float64)
        restarts = 4
        for r in range(restarts):
            if time.time() >= deadline - 80.0:
                break
            # small multiplicative perturbation of the current best
            perturb = np.random.normal(scale=0.03, size=n_best)
            a_start = a0 * np.exp(perturb)
            a_start = project_to_simplex(a_start)

            sub_deadline = min(deadline - 30.0, time.time() + (deadline - time.time()) * 0.3)
            a_opt, sc_opt = adam_smooth_optimize(
                a_start,
                sub_deadline,
                max_iters=250_000,
                beta_start=5.0,
                beta_end=1200.0,
                lr=0.012,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                a0 = a_opt.copy()          # feed the improvement forward
                print(f"[{time.time():.1f}] Adam restart {r+1} improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient fine‑tuning
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 50.0:
        print(f"[{time.time():.1f}] sub‑gradient fine‑tuning")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            max_iters=2_000_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation test
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Fallback if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 8. Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
