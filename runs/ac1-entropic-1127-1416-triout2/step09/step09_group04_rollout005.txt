Let's implement Exponentiated Gradient (EG) for exact max objective:

Initialize a positive vector (e.g., uniform). At each iteration:

    def exp_grad_optimize(start_a, deadline, max_iter=200_000, eta0=0.5):
        a = start_a.copy()
        best_a = a.copy()
        best_score = compute_score(a)
        n = len(a)
        for t in range(1, max_iter+1):
            cur = time.time()
            if cur >= deadline:
                break
            conv = convolution_fft(a, a)
            k = int(np.argmax(conv))
            # compute gradient: g_i = 2 * a_{k-i} if 0 <= k-i < n else 0.
            # Build index map efficiently:
            g = np.zeros_like(a)
            # valid i: i such that j = k - i in [0, n-1]: i in [k-(n-1), k]
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # set g[i] for i in range i_start..i_end inclusive:
            # g[i] = 2 * a[k-i]
            idx = np.arange(i_start, i_end+1)
            j = k - idx
            g[idx] = 2.0 * a[j]
            # update via EG:
            eta = eta0 / np.sqrt(t)
            a = a * np.exp(-eta * g)
            # renormalize to simplex:
            sum_a = a.sum()
            if sum_a == 0:
                # fallback uniform
                a = np.full_like(a, 1.0/n)
            else:
                a = a / sum_a
            # evaluate occasionally
            if t % 500 == 0:
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
        # final evaluation

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` objective.

Algorithm (≈ 1000 s budget)
--------------------------------
0.  Try a known construction (`height_sequence_1`) as a quick fallback.
1.  **Random‑restart exponential‑gradient (EG) search** on a moderate
    size (n≈1024).  Each restart runs a fast EG optimiser that directly
    minimises the exact max‑convolution using the true sub‑gradient.
    The EG update `a_i ← a_i·exp(−η·g_i)` automatically respects non‑negativity;
    the simplex constraint is enforced by normalising the vector.
    Many cheap restarts quickly explore the space; the best vector is kept.
2.  **Upsampling** – the current best vector is repeatedly split into two
    halves (each entry → two neighbours of half the mass).  This retains the
    total sum while providing a finer search space (2048 entries).
3.  **Fine‑tuned EG optimisation** on the full length (2048).  A slower,
    more accurate run with a decreasing learning‑rate finishes the
    continuous optimisation.
4.  **Stochastic hill‑climb with simulated‑annealing** – random local
    moves (swap, mass‑transfer, multi‑target redistribution) are applied.
    Improving moves are always kept; worsening moves are accepted
    with probability `exp(−Δ/T)` where the temperature `T` decays linearly
    with the remaining wall‑clock time.
5.  Return the best vector found (as a plain Python list).

All heavy work is done with NumPy/FFT – each convolution is O(n log n) and
fits easily inside the 1000 s wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score for a vector with Σa_i = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators (different shapes)
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """A smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start_geometric(n: int) -> np.ndarray:
    """Monotonically decreasing (or increasing) geometric shape."""
    q = np.random.uniform(0.6, 0.98)          # decay factor
    a = np.power(q, np.arange(n))
    a = a / a.sum()
    # sometimes mirror it to obtain an increasing shape
    if np.random.rand() < 0.5:
        a = a[::-1]
    return a


def random_start_piecewise(n: int) -> np.ndarray:
    """Few flat blocks of equal height."""
    blocks = max(2, int(np.sqrt(n) // 3))
    lengths = np.random.multinomial(n, np.full(blocks, 1.0 / blocks))
    a = np.concatenate([np.full(l, 1.0 / n) for l in lengths])
    a = a[:n]                     # safety cut‑off
    return a / a.sum()


def random_start(n: int) -> np.ndarray:
    """Pick one of the generators at random."""
    r = np.random.rand()
    if r < 0.25:
        return random_start_dense(n)
    elif r < 0.45:
        return random_start_sparse(n)
    elif r < 0.65:
        return random_start_gaussian(n)
    elif r < 0.80:
        return random_start_geometric(n)
    else:
        return random_start_piecewise(n)


# ----------------------------------------------------------------------
# 1. Exponential‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient descent on the exact objective
    f(a) = 2·n·max(conv(a,a)).
    The sub‑gradient is obtained from the maximising lag.
    Returns the best vector (on the simplex) seen during the run.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # current convolution and maximising index
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build sub‑gradient g_i = 2*a_{k-i} where defined
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        j = k - idx
        g[idx] = 2.0 * a[j]

        # exponential‑gradient update
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)
        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # tiny jitter to avoid stagnation
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Stochastic hill‑climb with simulated‑annealing polishing
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Random local modifications with a temperature schedule.
    Moves that improve the exact score are always accepted; worsening
    moves are accepted with probability exp(−Δ/T).  The temperature
    decays linearly with elapsed wall‑clock time.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0

    # Initialise temperature roughly 2 % of the current score
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        move_type = np.random.rand()

        if move_type < 0.35:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.70:
            # transfer a (random) fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                frac = np.random.rand() * 0.25   # up to 25 % of source
                delta = frac * cand[i]
                cand[i] -= delta
                cand[j] += delta
        else:
            # split mass from one index into three random targets
            src = np.random.randint(n)
            if cand[src] > 0.0:
                total_delta = np.random.rand() * 0.15 * cand[src]
                cand[src] -= total_delta
                targets = np.random.choice(n, size=3, replace=False)
                split = total_delta / 3.0
                cand[targets] += split

        # safeguard simplex constraints
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        # acceptance test
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: np.ndarray) -> np.ndarray:
    """Double the length by replacing each entry a_i with two halves a_i/2."""
    a = np.array(seq, dtype=float)
    up = np.empty(2 * a.shape[0], dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    up = project_to_simplex(up)   # numerical safety
    return up

    """
    np.random.seed(seed)

    start_time = time.time()
    # keep a small safety margin for the harness after we return
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------

    return best_seq
