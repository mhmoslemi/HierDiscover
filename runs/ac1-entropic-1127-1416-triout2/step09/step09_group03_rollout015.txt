A value ≤ 1.503 is required.  The optimiser works in three stages:

1. **Small‑scale linear‑programming refinement** (n≈512) using SciPy’s
   high‑performance “highs” LP solver.  The LP finds a partner vector
   b that satisfies  a∗b ≤ max(a∗a)  and has maximal ℓ₁‑mass.  Mixing
   a←(1‑t)a+t b (with a tiny t) provably never raises the worst
   convolution value.  Repeating the LP ~15‑20 times drives the
   maximum down substantially.

2. **Smooth Adam optimisation** on the log‑sum‑exp surrogate of the
   max‑convolution.  The temperature β is annealed from 5 to 800,
   the learning‑rate follows 1/√it and the iterate is always
   projected onto the probability simplex.

3. **Projected sub‑gradient polishing** plus a short stochastic hill‑climb.
   The sub‑gradient step follows the active lag (where the convolution
   attains its maximum).  A few hundred thousand iterations are performed
   while respecting the remaining wall‑clock budget.

All stages respect a global deadline (budget‑4 s safety margin) and
keep the best vector seen.  The final vector is returned as a plain
list of floats (already on the simplex, so the harness does not need to
renormalise it).

The code deliberately avoids heavyweight matrix builds for the large
final length (≈4096) – only the small‑scale LP needs the explicit Toeplitz
matrix.  The rest relies on fast FFT‑based convolutions.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * float(n) * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution matrix for the LP (small scale only)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix A such that (A @ b)[k] = conv(f,b)[k].
    Column i contains f shifted down by i rows.
    """
    n = f.shape[0]
    diags = [f] * n
    offsets = np.arange(n)          # 0,1,…,n‑1
    A = sp.diags(
        diagonals=diags,
        offsets=offsets,
        shape=(2 * n - 1, n),
        format="csr",
    )
    return A


def solve_lp_max_sum_scipy(A: sp.csr_matrix, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   subject to   A @ b ≤ M ,   b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]

    # linprog solves  min c·x ; we minimise -sum(b) to maximise sum(b)
    c = -np.ones(n, dtype=float)

    # Upper‑bound constraints
    b_ub = np.full(A.shape[0], M, dtype=float)

    # Bounds for each variable (non‑negative)
    bounds = [(0, None)] * n

    # Use HiGHS – the fastest open‑source LP solver in SciPy
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 30.0},
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    b = np.maximum(res.x, 0.0)      # clip tiny negatives caused by numeric noise
    return b


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 800.0,
    step0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution (log‑sum‑exp) surrogate.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal the temperature β (log‑sum‑exp smoothness)
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)            # full convolution, length 2n‑1
        maxc = conv.max()
        # soft‑max weights (numerically stable)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective w.r.t. a
        a_rev = a[::-1]
        grad_rev = convolution_fft(a_rev, w)    # length 2n‑1
        grad = 2.0 * grad_rev[n - 1 : n - 1 + n]  # slice of length n

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = project_to_simplex(a_new)

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (max‑convolution)
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    2·n·max(conv(a,a)).
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv[k] w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)    # classic 1/√t schedule
        a_new = a - step * g
        a = project_to_simplex(a_new)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Tiny random moves with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    step_scale = 0.05  # fraction of mass moved

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Upscaling utility – small → big
# ----------------------------------------------------------------------


def upscale(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a larger length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Candidate generator – orchestrates the whole pipeline
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for harness bookkeeping
    deadline = start + budget_s - 4.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"[fallback] score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Create a pool of inexpensive initial vectors (small n)
    # ------------------------------------------------------------------
    n_small = 512
    init_vectors = []

    # (a) uniform
    init_vectors.append(np.full(n_small, 1.0 / n_small))

    # (b) decreasing linear (a_i ∝ n-i)
    lin = np.arange(n_small, 0, -1, dtype=float)
    init_vectors.append(project_to_simplex(lin))

    # (c) exponential decay (r≈0.95)
    r = 0.95
    expv = r ** np.arange(n_small)
    init_vectors.append(project_to_simplex(expv))

    # (d) central‑binomial (Catalan) start
    cb = np.empty(n_small, dtype=float)
    cb[0] = 1.0
    for i in range(1, n_small):
        cb[i] = cb[i - 1] * (2 * i - 1) / (2.0 * i)
    init_vectors.append(project_to_simplex(cb))

    # (e) random Dirichlet draws (concentrations 0.3 … 2.0)
    for _ in range(3):
        alpha = np.random.uniform(0.3, 2.0)
        rand_vec = np.random.gamma(alpha, 1.0, size=n_small)
        init_vectors.append(project_to_simplex(rand_vec))

    # Keep the best of the pool as starting point
    for idx, vec in enumerate(init_vectors):
        sc = compute_score(vec)
        if sc < best_val:
            best_val = sc
            best_seq = vec.tolist()
            print(f"[init {idx}] new best score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Small‑scale LP refinement (n≈512)
    # ------------------------------------------------------------------
    a_small = np.array(best_seq, dtype=float) if best_seq is not None else init_vectors[0].copy()
    a_small = project_to_simplex(a_small)

    t_mix = 0.20           # mixing coefficient for convex combination
    lp_iters = 0
    lp_deadline = deadline - 300.0   # reserve time for later stages

    while time.time() < lp_deadline and lp_iters < 25:
        lp_iters += 1
        # compute current worst convolution value
        conv = convolution_fft(a_small, a_small)
        M = float(conv.max())

        # build Toeplitz matrix for the LP
        A = build_convolution_matrix(a_small)

        # solve LP
        b_opt = solve_lp_max_sum_scipy(A, M)
        if b_opt is None:
            print("[LP] solver failed – exiting LP phase.")
            break

        # renormalise partner to the simplex
        b_opt = project_to_simplex(b_opt)

        # convex combination (guaranteed not to increase the max‑conv)
        cand = (1.0 - t_mix) * a_small + t_mix * b_opt
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < best_val * 0.9999:          # accept only if a clear improvement
            best_val = cand_score
            best_seq = cand.tolist()
            a_small = cand
            print(f"[LP {lp_iters}] improved score = {cand_score:.6f}")
            # keep a relatively large mixing factor while we see progress
            t_mix = min(t_mix * 1.05, 0.35)
        else:
            # shrink mixing factor when progress stalls
            t_mix = max(t_mix * 0.85, 0.04)

    # ------------------------------------------------------------------
    # 3. Upscale to a larger length for fine‑grained optimisation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    n_big = 4096 if remaining > 250 else 2048
    a_big = upscale(a_small, n_big)

    # ------------------------------------------------------------------
    # 4. Smooth Adam optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 130.0
    if time.time() < smooth_deadline:
        a_smooth, smooth_score = smooth_adam_optimize(
            a_big,
            deadline=smooth_deadline,
            max_iters=400_000,
            beta_start=5.0,
            beta_end=800.0,
            step0=0.5,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"[smooth] improved to {smooth_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient polishing
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 50.0
    if time.time() < subgrad_deadline and best_seq is not None:
        refined, refined_score = subgradient_optimize(
            np.array(best_seq, dtype=float),
            deadline=subgrad_deadline,
            max_iter=600_000,
            step0=0.055,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined.tolist()
            print(f"[subgrad] improved to {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline and best_seq is not None:
        final_seq, final_score = hill_climb(
            np.array(best_seq, dtype=float),
            deadline=hill_deadline,
            max_steps=400_000,
        )
        if final_score < best_val:
            best_val = final_score
            best_seq = final_seq
            print(f"[hill‑climb] improved to {final_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – fall back to a tiny uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # final check (the harness will do its own clipping)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[final] reported score = {final_score:.6f}")
    return best_seq
