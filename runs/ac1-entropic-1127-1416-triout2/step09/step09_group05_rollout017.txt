def incremental_move(a, conv, i, j, delta):
    """Move mass delta from i to j, update a and conv in place; assumes delta <= a[i].
       Returns old_ai, old_aj, and old_conv copy for possible revert."""
    n = a.shape[0]
    old_ai = a[i]
    old_aj = a[j]
    if delta <= 0 or delta > old_ai:
        return None  # indicates invalid
    # store copy of conv for revert
    old_conv = conv.copy()
    # linear updates using old a
    # because we haven't changed a yet, a still has old values
    conv[i:i+n] += -2.0 * delta * a
    conv[j:j+n] += 2.0 * delta * a
    # quadratic updates
    conv[2*i] += delta * delta
    conv[2*j] += delta * delta
    conv[i + j] += -2.0 * delta * delta
    # update a
    a[i] = old_ai - delta
    a[j] = old_aj + delta
    return old_ai, old_aj, old_conv

def revert_move(a, conv, i, j, old_ai, old_aj, old_conv):

def incremental_move(a, conv, i, j, delta):
    # delta positive move from i to j
    n = a.shape[0]
    old_ai = a[i]
    old_aj = a[j]
    if delta <= 0 or delta > old_ai:
        return None
    old_conv = conv.copy()
    delta_i = -delta
    delta_j = delta
    # linear updates -> use original a vector (pre-modification)
    conv[i:i+n] += 2.0 * delta_i * a
    conv[j:j+n] += 2.0 * delta_j * a
    # quadratic
    conv[2*i] += delta_i * delta_i
    conv[2*j] += delta_j * delta_j
    conv[i + j] += 2.0 * delta_i * delta_j
    # update a
    a[i] = old_ai + delta_i  # = old_ai - delta
    a[j] = old_aj + delta_j  # = old_aj + delta
    return old_ai, old_aj, old_conv

def revert_move(a, conv, i, j, old_ai, old_aj, old_conv):
    a[i] = old_ai
    a[j] = old_aj
    conv[:] = old_conv

"""
Search for a non‑negative step‑function (vector) that minimises
`evaluate_sequence`.  The method:
1. Scan simple one‑parameter families (exponential, power‑law) to obtain
   a good initial shape.
2. Starting from the best shape run a long, fine‑grained hill‑climb that
   transfers a small amount of mass between two coordinates and updates
   the full convolution incrementally.  The move is accepted only if it
   reduces the exact `max(conv)` (hence the objective).  The algorithm
   uses the current maximal convolution index to bias the choice of the
   source coordinate and chooses a destination where its contribution to
   that maximal index is minimal.  This allows many inexpensive moves
   (≈O(n) each) and thus explores far more of the search space within the
   1000 s wall‑clock limit.
3. The best vector found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector normalised to sum = 1.
    (Same as the value returned by `evaluate_sequence` for such a vector.)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def scan_exponential_and_power(
    n: int, deadline: float
) -> tuple[np.ndarray | None, float]:
    """
    Scan exponential and power‑law families.  Returns the best vector
    (as a NumPy array) and its exact score, or (None, inf) if time ran out.
    """
    best_a = None
    best_sc = float("inf")

    # exponential λ grid (including λ=0 → uniform)
    lam_grid = np.concatenate(
        (np.array([0.0]), np.logspace(-5, -2, 200))
    )
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # power‑law p grid (p=0 → uniform)
    p_grid = np.linspace(0.0, 6.0, 200)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Incremental convolution updates used by the hill‑climb
# ----------------------------------------------------------------------


def incremental_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Transfer a positive amount `delta` of mass from index i to index j.
    The vectors `a` and `conv` (full convolution a*a) are updated in‑place.
    Returns a tuple (old_ai, old_aj, old_conv) that can be used to revert
    the operation, or None if the move is illegal.
    """
    n = a.shape[0]
    if delta <= 0.0 or delta > a[i]:
        return None
    old_ai = a[i]
    old_aj = a[j]
    old_conv = conv.copy()

    delta_i = -delta
    delta_j = delta

    # linear part (uses the *old* a, which is still unchanged)
    conv[i : i + n] += 2.0 * delta_i * a
    conv[j : j + n] += 2.0 * delta_j * a

    # quadratic part
    conv[2 * i] += delta_i * delta_i
    conv[2 * j] += delta_j * delta_j
    conv[i + j] += 2.0 * delta_i * delta_j

    # update the probability vector
    a[i] = old_ai + delta_i
    a[j] = old_aj + delta_j

    return old_ai, old_aj, old_conv


def revert_move(
    a: np.ndarray,
    conv: np.ndarray,
    i: int,
    j: int,
    old_ai: float,
    old_aj: float,
    old_conv: np.ndarray,
):
    """
    Restore a and conv to the state before the corresponding `incremental_move`.
    """
    a[i] = old_ai
    a[j] = old_aj
    conv[:] = old_conv


# ----------------------------------------------------------------------
# Hill climb using the incremental update
# ----------------------------------------------------------------------


def hill_climb_incremental(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic hill‑climb that repeatedly moves a small packet of
    mass from a source coordinate (chosen among those contributing most to
    the current maximal convolution entry) to a destination whose
    contribution to that entry is minimal.  The convolution is maintained
    incrementally, so each trial costs O(n) time.  Only improving moves are kept.
    """
    a = start_seq.astype(float).copy()
    a /= a.sum()                     # ensure exact normalisation
    n = a.size
    conv = convolution_fft(a, a)     # full convolution, length 2n‑1
    best_a = a.copy()
    best_conv_max = conv.max()
    best_score = 2.0 * n * best_conv_max

    # adaptive step size (fraction of the source mass)
    step_scale = 0.12
    it = 0
    while it < max_iters and time.time() < deadline:
        # locate the shift where the convolution currently attains its maximum
        k_max = int(conv.argmax())

        # find the index i that contributes most to that maximal entry
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx_range = np.arange(low, high + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        i_local = int(np.argmax(contrib))
        i = idx_range[i_local]

        # safety: if the chosen source has negligible mass, pick a random one
        if a[i] < 1e-12:
            i = np.random.randint(n)

        # destination j – we prefer indices whose partner in the pair (k_max‑j)
        # carries little mass, i.e. where a[k_max‑j] is small (or undefined)
        partner = k_max - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        partner_mass = np.where(mask, a[partner], 0.0)
        # probability inversely proportional to partner_mass
        inv_weights = np.where(mask, 1.0 / (partner_mass + 1e-12), 1.0)
        inv_weights[i] = 0.0                     # avoid picking the source
        prob = inv_weights / inv_weights.sum()
        j = np.random.choice(n, p=prob)

        # ensure j ≠ i
        if j == i:
            j = (i + 1) % n

        # amount to move – a fraction of the source mass
        delta = step_scale * a[i]
        max_allowed = a[i] - 1e-15
        if delta > max_allowed:
            delta = max_allowed

        # try the move
        saved = incremental_move(a, conv, i, j, delta)
        if saved is None:
            it += 1
            continue
        old_ai, old_aj, old_conv = saved
        new_max = conv.max()

        if new_max < best_conv_max - 1e-15:
            # improvement – keep the move and increase the step size a little
            best_conv_max = new_max
            best_a = a.copy()
            best_score = 2.0 * n * best_conv_max
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # revert to the previous state and shrink the step size
            revert_move(a, conv, i, j, old_ai, old_aj, old_conv)
            step_scale = max(step_scale * 0.95, 1e-5)

        # occasionally re‑normalise to avoid drift caused by floating‑point errors
        if it % 10_000 == 0:
            a /= a.sum()
            conv = convolution_fft(a, a)      # clean re‑computation
            # keep the best if the full recomputation revealed improvement
            cur_max = conv.max()
            if cur_max < best_conv_max - 1e-15:
                best_conv_max = cur_max
                best_a = a.copy()
                best_score = 2.0 * n * best_conv_max

        it += 1

    return best_a, best_score

    if sc_uni < best_score:
        best_score, best_seq = sc_uni, uniform.tolist()
        print(f"[{time.time():.1f}] uniform baseline score = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Aggressive incremental hill‑climb
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] starting incremental hill‑climb")
        a_final, sc_final = hill_climb_incremental(
            np.array(best_seq, dtype=float),
            deadline,
            max_iters=5_000_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: fall back to a tiny uniform vector if nothing succeeded
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] tiny uniform fallback score = {best_score:.6f}")

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
