def hill_climb_anneal(start_seq, deadline, max_steps=2000000):
    a = start_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)
    # initialize temperature
    T0 = 1e-2 # maybe
    Tmin = 1e-6
    # compute steps allowed based on time
    step = 0
    while time.time() < deadline and step < max_steps:
        # propose move
        cand = best_a.copy()
        # decide move type
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.rand(n), np.random.rand(n)
            i, j = np.random.randint(n, size=2)
            if i != j:
                # random fraction of a_i
                delta = np.random.rand() * best_a[i] * 0.2  # up to 20% shift
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        delta_sc = sc - best_sc
        # temperature schedule [exponential]
        T = T0 * (0.5 ** (step / max_steps))
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept move
            if sc < best_sc:
                best_sc = sc
                best_a = cand
            a = cand
        step += 1
    return best_a, best_sc

"""
Minimal‐max‐self‑convolution search.

The routine builds a probability vector (non‑negative, sum = 1) and
minimises 2·n·max(a∗a) – the exact value returned by `evaluate_sequence`.
It proceeds in three phases respecting a 1000 s wall‑clock limit:

1. **Broad parameter scan** of several analytic families (exponential,
   power law, symmetric edge‑power, product‑power, double‑exponential and a
   random hybrid).  The best shape is kept.
2. **Smooth max optimisation** using a mirrored (exponentiated‑gradient)
   update on a log‑sum‑exp approximation of the max‑convolution.  The
   schedule for the smoothing parameter β is exponential and the step size
   follows η∝1/√t.
3. **Simulated‑annealing hill‑climb** that proposes swaps or mass transfers,
   accepts improving moves and, with a linearly decreasing temperature,
   also occasionally accepts worsening moves.  This helps escape local
   minima and usually yields a further small improvement.

All phases stop well before the hard timeout; the best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a vector with sum = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families of candidate shapes
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)   (lam ≥ 0)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)   (p ≥ 0)."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def edge_power_shape(n: int, p: float) -> np.ndarray:
    """symmetric edge‑heavy shape:
       a_i ∝ (i+1)^‑p + (n‑i)^‑p   (p ≥ 0)."""
    i = np.arange(n, dtype=float)
    w = (i + 1) ** (-p) + (n - i) ** (-p)
    return w / w.sum()


def product_power_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped shape:
       a_i ∝ [(i+1)*(n‑i)]^‑p   (p ≥ 0)."""
    i = np.arange(n, dtype=float)
    w = ((i + 1) * (n - i)) ** (-p)
    return w / w.sum()


def double_exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) + exp(-lam·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return w / w.sum()


def hybrid_shape(n: int, lam: float, p: float) -> np.ndarray:
    """mix of exponential decay and power‑law at both ends."""
    i = np.arange(n, dtype=float)
    w = (np.exp(-lam * i) * (i + 1) ** (-p) +
         np.exp(-lam * (n - 1 - i)) * (n - i) ** (-p))
    return w / w.sum()

# ----------------------------------------------------------------------
# Smoothed max optimisation (exponentiated‑gradient)
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 500_000,
                        beta_start: float = 0.5,
                        beta_end: float = 5000.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14):
    """
    Minimise a log‑sum‑exp approximation of max(a*a) using a KL‑mirror
    (exponentiated‑gradient) update.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β annealing (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weights (log‑sum‑exp)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))          # stabilised
        w = expw / expw.sum()

        # gradient of the smoothed max (w.r.t. a)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # step size (η ∝ 1/√t)
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update & renormalise
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()

    return best_a, best_sc

# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(start_seq: np.ndarray,
                      deadline: float,
                      max_steps: int = 1_000_000):
    """
    Greedy random walk with a linearly decreasing temperature.
    Swaps or small mass transfers are proposed; worsening moves may be
    accepted with probability exp(-Δ/T).  The best vector seen is returned.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # current state (may be different from the best)
    cur_a = a.copy()
    cur_sc = best_sc

    # temperature schedule – start at ~2 % of the current score
    T0 = max(best_sc * 0.02, 1e-4)
    Tmin = 1e-7

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cur_a[i] > 0:
                delta = np.random.rand() * 0.2 * cur_a[i]   # ≤ 20 % of donor mass
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        delta_sc = sc - cur_sc

        # linearly decaying temperature
        T = max(T0 * (1.0 - step / max_steps), Tmin)

        # Metropolis acceptance
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / T):
            cur_a, cur_sc = cand, sc
            if sc < best_sc:
                best_sc, best_a = sc, cand.copy()

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The algorithm consists of a quick family scan, a long exponentiated‑gradient
    refinement and a final simulated‑annealing hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin before the hard timeout
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a workable length based on the time budget
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Scan a rich collection of analytic families
    # -------------------------------------------------
    a_scan, sc_scan = scan_all_families(n, deadline - 30.0)
    if a_scan is not None and sc_scan < best_score:
        best_seq = a_scan.tolist()
        best_score = sc_scan
        print(f"[{time.time():.1f}] best after family scan = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Smoothed max optimisation (exponentiated‑gradient)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting smooth exponentiated‑gradient")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=500_000,
            beta_start=0.5,
            beta_end=5000.0,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_seq = a_opt.tolist()
            best_score = sc_opt
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Final simulated‑annealing hill‑climb
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final simulated annealing")
        # estimate remaining time and convert to a sensible number of steps
        steps_budget = int((deadline - time.time()) * 400)  # ~400 evals per second
        a_final, sc_final = hill_climb_anneal(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=steps_budget,
        )
        if sc_final < best_score:
            best_seq = a_final.tolist()
            best_score = sc_final
            print(f"[{time.time():.1f}] anneal improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if everything failed fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain list of floats as required by the harness
    return best_seq
