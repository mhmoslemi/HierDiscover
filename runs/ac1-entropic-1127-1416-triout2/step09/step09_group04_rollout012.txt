        iteration += 1

        conv_f_f = np.convolve(f, f)
        M = conv_f_f.max()
        # Build Toeplitz matrix A where each row corresponds to conv f with g
        c = np.concatenate([f, np.zeros(n - 1)])

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` objective.

Algorithm
---------
1. **Random non‑uniform start** – draw a dense Dirichlet vector and scale
   it so that the sum equals √(2 n) (the natural normalisation of the
   objective).  Every entry is forced to stay strictly positive.

2. **Linear‑program refinement** – given the current vector `f`,
   compute `M = max(conv(f,f))`.  For the same `M` solve the linear program

        maximise   Σ g_i
        subject to  conv(f, g) ≤ M   (component‑wise)
                    g_i ≥ 0

   (the convolution with a *fixed* `f` is linear in `g`).  The LP is
   solved with `scipy.optimize.linprog` (highs interior‑point).  The
   solution `g` is then scaled to the required sum and mixed with `f`
   (`f ← (1‑α) f + α g`).  This step is proved to never increase `M`,
   thus driving the objective down.  A few (≤30) LP steps are performed
   per restart.

3. **Multiple restarts** – the LP routine is applied to many random
   starts until the time budget is exhausted.  The best vector seen so
   far is kept.

4. **Stochastic polishing** – a cheap hill‑climbing / simulated‑annealing
   phase (local swaps and mass transfers) further reduces the exact
   `evaluate_sequence` score.

The whole search respects the 1000 s wall‑clock limit and returns the
best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def evaluate_exact(a: np.ndarray) -> float:
    """Exact `evaluate_sequence` score for a vector that already obeys the
    internal clipping.  This is the same as 2·n·max(conv(a,a))/sum(a)^2."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start (all entries > 0)."""
    a = np.random.rand(n) + 1e-12          # avoid exact zeros
    a /= a.sum()
    return a


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its sum equals `target`."""
    if v.sum() == 0:
        # distribute uniformly if everything vanished
        return np.full_like(v, target / v.shape[0])
    return v * (target / v.sum())


# ----------------------------------------------------------------------
# Linear‑program refinement (the heart of the method)
# ----------------------------------------------------------------------


def lp_refine(
    f: np.ndarray,
    target_sum: float,
    time_deadline: float,
    alpha: float = 0.25,
    max_iter: int = 30,
):
    """Iteratively improve `f` by the LP described in the comment
    (maximise Σ g subject to conv(f,g) ≤ M)."""
    import scipy.linalg
    import scipy.optimize

    n = f.shape[0]
    best_f = f.copy()
    best_score = evaluate_exact(best_f)

    # tiny epsilon to keep every entry strictly positive for the next LP
    eps = 1e-12

    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1

        # 1) current maximal convolution coefficient
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # 2) build the convolution matrix A (Toeplitz) such that A @ g = conv(f,g)
        c = np.concatenate([f, np.zeros(n - 1, dtype=float)])
        r = np.concatenate([f[:1], np.zeros(n - 1, dtype=float)])
        A = scipy.linalg.toeplitz(c, r)      # shape (2n‑1, n)

        # 3) linear program: maximise Σ g_i  ⇔ minimise –Σ g_i
        c_obj = -np.ones(n, dtype=float)
        b_ub = np.full(A.shape[0], M, dtype=float)
        bounds = [(0.0, None) for _ in range(n)]

        res = scipy.optimize.linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # If the LP fails (should be rare) abort the refinement.
            print(f"LP failed at iteration {it}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # 4) scale g back to the required total mass
        g_scaled = project_to_sum(g, target_sum)

        # 5) convex combination with the old vector
        f_new = (1.0 - alpha) * f + alpha * g_scaled

        # keep strict positivity (important for the next LP)
        f_new = np.maximum(f_new, eps)

        # re‑normalise exactly to the target sum (numerical drift)
        f_new = project_to_sum(f_new, target_sum)

        # evaluate exact score
        cur_score = evaluate_exact(f_new)

        if cur_score < best_score:
            best_score = cur_score
            best_f = f_new.copy()
            print(f"  LP iteration {it}: improved score = {best_score:.6f}")

        f = f_new

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb with simulated annealing (polishing step)
# ----------------------------------------------------------------------


def hill_climb_anneal(seq, time_deadline, max_steps: int = None):
    """Local random modifications with a temperature schedule.  Always
    accept improvements; sometimes accept worsenings."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = evaluate_exact(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    # initial temperature proportional to the current score
    T0 = best_score * 0.02
    Tmin = best_score * 1e-7

    steps = 0
    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        elapsed = time.time() - start
        T = max(Tmin, T0 * (1.0 - elapsed / (time_deadline - start)))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.04 * cand[i]          # 4 % of the source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑normalise exactly to the target sum (which does not change)
        cand = project_to_sum(cand, cur_a.sum())

        cand_score = evaluate_exact(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects a ~1000 s wall‑clock budget
    and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # problem size – 2048 gives a good trade‑off between quality and speed
    n = 2048
    target_sum = np.sqrt(2.0 * n)           # natural normalisation

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback sequence (if the sandbox provides one)
    # ------------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            fb_val = evaluate_sequence(fb)
            print(f"Fallback score = {fb_val:.6f}")
            if fb_val < best_val:
                best_val = fb_val
                best_seq = fb
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Random‑restart + LP refinement loop
    # ------------------------------------------------------------------
    # reserve a few seconds for the final polishing stage
    reserve_time = 30.0
    restart = 0
    while time.time() < deadline - reserve_time:
        restart += 1
        # allocate a per‑restart slice of time (≈20 s or until budget runs low)
        remaining = deadline - time.time() - reserve_time
        slice_sec = min(25.0, max(8.0, remaining * 0.3))
        slice_deadline = time.time() + slice_sec

        # random dense start (strictly positive)
        f0 = random_start_dense(n)
        f0 = project_to_sum(f0, target_sum)

        cur_score = evaluate_sequence(f0.tolist())
        if cur_score < best_val:
            best_val = cur_score
            best_seq = f0.tolist()
            print(f"Restart {restart}: initial random score = {cur_score:.6f}")

        # LP driven refinement
        f_ref, ref_score = lp_refine(
            f0,
            target_sum=target_sum,
            time_deadline=slice_deadline,
            alpha=0.25,
            max_iter=25,
        )
        if ref_score < best_val:
            best_val = ref_score
            best_seq = f_ref.tolist()
            print(f"Restart {restart}: post‑LP best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Stochastic hill‑climb polishing (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nPolishing with hill‑climb annealing …")
        polished_seq, polished_score = hill_climb_anneal(
            best_seq,
            time_deadline=deadline,
            max_steps=None,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Polished best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Defensive fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [target_sum / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 4. Final reporting
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return plain Python list – the harness expects this.
    return best_seq
