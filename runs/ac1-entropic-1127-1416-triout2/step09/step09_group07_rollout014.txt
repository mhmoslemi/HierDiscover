def adam_optimize(start_a, deadline, max_iters=100_000, beta_start=0.5, beta_end=300.0, lr=0.05):
    # start_y = np.log(start_a + 1e-12)
    y = np.log(start_a + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = None
    best_sc = float('inf')
    n = start_a.shape[0]
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute a via softmax
        y_max = y.max()
        exp_y = np.exp(y - y_max)
        a = exp_y / exp_y.sum()
        # schedule beta
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        # conv
        conv = convolution_fft(a, a)
        # compute w
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()
        # gradient wrt a
        grad_a = 2.0 * convolution_fft(a, w)[n-1:n-1+n]  # transform a * w conv
        # maybe add small epsilon for stability
        # gradient wrt y
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    # final check
    a_final = np.exp(y - np.max(y))
    a_final = a_final / a_final.sum()
    final_sc = compute_score(a_final)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a_final.copy()
    return best_a, best_sc

"""
A multi‑stage optimizer for the step‑function attack.

The algorithm works on a sequence a of length n (chosen according to the
remaining time) that is kept on the probability simplex (sum a = 1 and
a_i ≥ 0).  The score is

    score(a) = 2·n·max(a⊛a)

which equals the value returned by ``evaluate_sequence`` after the
normalisation step performed inside that routine.

Pipeline
--------
0.  Optional fallback: use the globally supplied ``height_sequence_1``.
1.  Analytic start – a one‑parameter family a_i(α) ∝ [(i+½)(n‑½‑i)]^{‑α}
    scanned on a modest grid; the best vector is kept.
2.  Sub‑gradient descent – projected Euclidean sub‑gradient steps on the
    exact (non‑smoothed) objective.
3.  Adam optimisation of a *smooth* approximation of the max‑convolution.
    The smoothed objective is the log‑sum‑exp of the self‑convolution,
    controlled by a temperature β that is exponentially annealed.
    The parameters are re‑parameterised with a soft‑max (y → a) so that
    the simplex constraint is satisfied implicitly.
4.  Fast stochastic hill‑climb – a greedy random walk that moves a small
    packet of mass between two positions (or swaps two entries) while
    updating the convolution incrementally.  Only strictly improving
    moves are accepted; the step size adapts automatically.

All stages stop automatically when the 1000 s wall‑clock budget (minus a
small safety margin) is exhausted.  The best sequence encountered is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This matches the value returned by `evaluate_sequence` when the input
    already satisfies the normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """Search a modest α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([
        np.linspace(-1.5, 0.0, 16),   # centre‑heavy
        np.linspace(0.0, 3.0, 31)     # edge‑heavy
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for 2·n·max(a*a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Adam optimisation of the smooth (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  max_iters: int = 150_000,
                  beta_start: float = 0.5,
                  beta_end: float = 500.0,
                  lr: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Optimise a soft‑max re‑parameterisation ``y → a = softmax(y)`` using
    Adam on a smooth approximation of the max‑convolution.
    The smooth objective is
        f_β(a) = (1/β) log Σ_i exp(β·(a⊛a)_i)
    with β exponentially annealed from ``beta_start`` to ``beta_end``.
    """
    eps = 1e-12
    y = np.log(start_a + eps)               # initialise y so that softmax(y)=start_a
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = None
    best_sc = float("inf")
    n = start_a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- soft‑max mapping ---
        y_max = y.max()
        exp_y = np.exp(y - y_max)
        a = exp_y / exp_y.sum()

        # --- β schedule (exponential) ---
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # --- smoothed max and its gradient ---
        conv = convolution_fft(a, a)               # full self‑convolution
        maxc = conv.max()
        # stable soft‑max over convolution entries
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                    # w_i ≥ 0, Σ w = 1

        # gradient of f_β w.r.t. a:
        #   ∂f/∂a = 2 * (a ⊛ w)  (central part of the full convolution)
        grad_a = 2.0 * convolution_fft(a, w)[n - 1:n - 1 + n]

        # --- chain rule through the soft‑max (mirror) map ---
        dot_ag = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot_ag)

        # --- Adam update ---
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    a_final = np.exp(y - np.max(y))
    a_final = a_final / a_final.sum()
    sc_final = compute_score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Fast stochastic hill‑climb (incremental convolution updates)
# ----------------------------------------------------------------------


def fast_hill_climb(start_a: np.ndarray,
                    deadline: float,
                    inc_factor: float = 1.07,
                    decay_factor: float = 0.97,
                    recompute_interval: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Greedy random‑walk polishing with O(n) incremental convolution updates.
    Only moves that strictly improve the best score are accepted; the
    step size adapts automatically.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)               # exact convolution (length 2n‑1)
    best_a = a.copy()
    best_sc = 2.0 * n * float(conv.max())
    step_scale = 0.12
    steps = 0

    while time.time() < deadline:
        # choose move type (swap or small transfer)
        if np.random.rand() < 0.5:
            # swap the values of two positions
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = a[src] - a[dst]            # positive amount to move
        else:
            # transfer a fraction of the source mass to a random destination
            src = np.random.randint(n)
            if a[src] == 0.0:
                continue
            delta = step_scale * a[src]         # packet size
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1                         # ensure dst ≠ src

        # incremental update of the convolution
        conv_cand = conv.copy()
        # linear shift terms
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic corrections (exact)
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        if cand_score < best_sc:
            # accept the move
            best_sc = cand_score
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            best_a = a.copy()
            step_scale = min(step_scale * inc_factor, 0.5)
        else:
            # reject – shrink the packet size slowly
            step_scale *= decay_factor

        steps += 1
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)   # re‑synchronise to avoid drift

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the 1000 s wall‑clock limit and returns the best
    sequence discovered as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    # -------------------------------------------------
    # Choose a workable length (more length → richer search space)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 16384
    elif remaining > 500:
        n = 8192
    elif remaining > 300:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic start (α‑grid)
    # -------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Projected sub‑gradient refinement (if time permits)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 180.0,
            max_iters=250_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Adam optimisation of the smooth surrogate
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] Adam optimisation (smooth max)")
        a_adam, sc_adam = adam_optimize(
            np.array(best_seq, dtype=float),
            deadline - 90.0,
            max_iters=200_000,
            beta_start=0.5,
            beta_end=800.0,
            lr=0.06,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Fast hill‑climb polishing (majority of remaining budget)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] fast hill‑climb")
        a_final, sc_final = fast_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            inc_factor=1.08,
            decay_factor=0.96,
            recompute_interval=120_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:

        best_seq = [1.0 / n_small] * n_small

    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_seq
