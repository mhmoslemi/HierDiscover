"""
Evolutionary + local‐search optimizer for the step‑function objective.

The algorithm works in three stages under the given time budget:

1. **Seeding** – a diverse set of hand‑crafted and random Dirichlet vectors,
   projected on the simplex, is evaluated and the best few are kept.

2. **Evolutionary loop** – a small population is evolved with
   *Gaussian noise mutation*, *random swap mutation* and *simple crossover*.
   New offspring replace the worst individuals only when they improve the
   exact `2·n·max(conv(a,a))` score.  The mutation strength decays with the
   elapsed time, which gives a coarse global search early on and fine‑grained
   exploitation later.

3. **Local polishing** – a short hill‑climb that applies tiny mass‑transfer
   moves or a smoothing filter, always accepting improvements.  This
   frequently pushes the best candidate a little further.

All heavy calculations use an FFT‑based convolution (`compute_score`) which
is O(n log n) and therefore cheap enough for thousands of evaluations.
The final solution is returned as a plain list of floats, ready for the
provided `evaluate_sequence` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, score, simple generators)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of the result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·(i+1)/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape w_i ∝ ((i+1)(n‑i))**p, normalised."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Sample a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Mutation / crossover helpers
# ----------------------------------------------------------------------


def mutate_noise(a: np.ndarray, sigma: float) -> np.ndarray:
    """Add Gaussian noise (sigma) and re‑normalise onto the simplex."""
    cand = a + np.random.randn(a.shape[0]) * sigma
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s == 0.0:
        # fallback to a fresh random Dirichlet
        return random_dirichlet(a.shape[0])
    return cand / s


def mutate_shuffle(a: np.ndarray, num_swaps: int) -> np.ndarray:
    """Perform a few random pairwise swaps."""
    cand = a.copy()
    n = a.shape[0]
    for _ in range(num_swaps):
        i, j = np.random.randint(n, size=2)
        cand[i], cand[j] = cand[j], cand[i]
    return cand  # already sums to 1


def crossover(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convex combination of two parents followed by simplex projection."""
    w = np.random.rand()
    cand = w * a + (1.0 - w) * b
    s = cand.sum()
    if s == 0.0:
        return random_dirichlet(a.shape[0])
    return cand / s


# ----------------------------------------------------------------------
# Local polishing (hill‑climb with smoothing)
# ----------------------------------------------------------------------


def local_polish(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb that only keeps improving moves.
    Occasionally applies a tiny smoothing filter.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    smooth_kernel = np.ones(5) / 5.0
    step_scale = 0.04
    it = 0
    while time.time() < deadline - 0.5:
        it += 1
        cand = best.copy()
        if np.random.rand() < 0.7:
            # local mass transfer
            i = np.random.randint(n)
            if cand[i] <= 0.0:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = step_scale * np.random.rand() * cand[i]
            cand[i] -= delta
            cand[j] += delta
        else:
            # smoothing move
            cand = np.convolve(best, smooth_kernel, mode='same')
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best = cand
            best_score = cur_score
            step_scale = min(step_scale * 1.08, 0.2)   # become bolder
        else:
            step_scale *= 0.96                         # become more conservative

        # occasional progress report
        if it % 5000 == 0:
            elapsed = time.time() - (deadline - (deadline - time.time()))  # dummy just to print
            print(f"[Polish] iter {it}, best {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Evolutionary search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the wall‑clock budget,
    returns a plain Python list of floats, and prints occasional progress.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # Choose a reasonable length (powers of two are fast for FFT)
    # --------------------------------------------------------------
    if total_time > 800.0:
        n = 4096
    elif total_time > 400.0:
        n = 3072
    else:
        n = 2048
    print(f"[Info] Sequence length set to {n}")

    # --------------------------------------------------------------
    # 0. Assemble an initial diverse population (size ~30)
    # --------------------------------------------------------------
    init_candidates = []

    # a) deterministic sine‑squared shape
    init_candidates.append(generate_deterministic(n))

    # b) beta‑family shapes (both concave and convex)
    for p in np.linspace(-2.5, 2.5, 7):
        init_candidates.append(generate_beta_family(n, p))

    # c) random Dirichlet draws (different concentrations)
    for _ in range(8):
        conc = np.random.choice([0.3, 0.7, 1.0, 2.0])
        init_candidates.append(random_dirichlet(n, conc))

    # d) a randomly permuted deterministic shape (break symmetry)
    det = generate_deterministic(n).copy()
    np.random.shuffle(det)
    init_candidates.append(det)

    # e) fallback from previous runs if available
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.shape[0] == n:
            fallback = np.maximum(fallback, 0.0)
            if fallback.sum() == 0.0:
                fallback = random_dirichlet(n)
            else:
                fallback = fallback / fallback.sum()
            init_candidates.append(fallback)
    except Exception:
        pass

    # Normalise all candidates and evaluate them
    population = []
    scores = []
    for a in init_candidates:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            continue
        a = a / s
        sc = compute_score(a)
        population.append(a)
        scores.append(sc)

    # Keep the top `pop_size` individuals (pop_size ≈ 20)
    pop_size = 20
    if len(population) > pop_size:
        # sort by score (ascending)
        order = np.argsort(scores)
        population = [population[i] for i in order[:pop_size]]
        scores = [scores[i] for i in order[:pop_size]]

    # Remember the current global best
    best_idx = np.argmin(scores)
    best_vec = population[best_idx].copy()
    best_score = scores[best_idx]
    print(f"[Init] Best initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Evolutionary search (major part of the budget)
    # --------------------------------------------------------------
    evol_deadline = deadline - 10.0          # leave ~10 s for polishing
    it = 0
    sigma0 = 0.12                           # initial noise level
    sigma_min = 0.001                       # final noise level
    while time.time() < evol_deadline:
        it += 1
        # ---------- parent selection ----------
        # pick one of the elite (top half) uniformly
        elite_cut = max(1, len(population) // 2)
        parent = population[np.random.randint(elite_cut)]

        # ---------- mutation type ----------
        r = np.random.rand()
        progress = (time.time() - start_time) / total_time
        sigma = sigma_min + (sigma0 - sigma_min) * (1.0 - progress)   # linear decay

        if r < 0.45:
            child = mutate_noise(parent, sigma)
        elif r < 0.85:
            # swap‑mutation – number of swaps grows slowly with n
            num_swaps = max(1, int(0.005 * n * (1.0 - progress)))
            child = mutate_shuffle(parent, num_swaps)
        else:
            # crossover with another random elite individual
            partner = population[np.random.randint(elite_cut)]
            child = crossover(parent, partner)

        # ---------- evaluation ----------
        child_score = compute_score(child)

        # ---------- replacement ----------
        # replace the worst individual if the child is better
        if child_score < scores[-1]:
            population[-1] = child
            scores[-1] = child_score
            # bubble the new entry up to keep the list sorted
            i = len(population) - 1
            while i > 0 and scores[i] < scores[i - 1]:
                population[i], population[i - 1] = population[i - 1], population[i]
                scores[i], scores[i - 1] = scores[i - 1], scores[i]
                i -= 1

            # update global best
            if child_score < best_score:
                best_score = child_score
                best_vec = child.copy()
                elapsed = time.time() - start_time
                print(f"[Evol] t={elapsed:.1f}s iter={it} new best = {best_score:.6f}")

        # occasional progress report
        if it % 50000 == 0:
            elapsed = time.time() - start_time
            print(f"[Evol] elapsed {elapsed:.1f}s, best so far {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Final local polishing (hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Polish] Starting final hill‑climb")
        best_vec, best_score = local_polish(best_vec, deadline)

    # --------------------------------------------------------------
    # 3. Return the result (list of floats) and final official score
    # --------------------------------------------------------------
    final_seq = best_vec.tolist()
    final_val = evaluate_sequence(final_seq)  # the official, possibly slower, check
    print(f"\n[Result] Final reported score = {final_val:.6f}")
    return final_seq
