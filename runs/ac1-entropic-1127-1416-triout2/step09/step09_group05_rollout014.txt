def hill_climb_simulated(start, deadline, max_steps):
    a = start.copy()
    n = a.size
    best = a.copy()
    best_sc = compute_score(best)
    temp0 = 1.0
    temp_end = 1e-4
    # linear or exponential schedule.
    step = 0
    while step < max_steps and time.time() < deadline:
        # propose move: pick i, j distinct.
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        if np.random.rand() < 0.5:
            # swap
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer fraction
            fraction = 0.01 * (np.random.rand())  # up to 1% of mass from i
            delta = fraction * best[i]
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand
        else:
            # accept with probability exp(-(sc - best_sc) / T)
            # compute temperature scaled linearly decreasing
            t = step / max_steps
            T = temp0 * (temp_end / temp0) ** t
            if np.random.rand() < np.exp(-(sc - best_sc) / (T + 1e-12)):
                best = cand
                # note we keep best_sc unchanged, but move to cand for further exploration
        step += 1
    return best, best_sc

"""
Propose a step‑function (non‑negative vector) with a low value of
evaluate_sequence.  The algorithm consists of four phases:

1.   Scan a variety of analytic families (exponential, power‑law,
     triangular, Gaussian) and also a few random permutations of those
     families.  The best vector from this cheap search is kept.
2.   Start a number of short “smooth‑max” optimisations from random
     Dirichlet draws (different concentration parameters).  This
     explores many basins of attraction.
3.   Run a long exponentiated‑gradient (mirror‑descent) optimiser on the
     current best vector.  The smooth objective is a log‑sum‑exp
     approximation of the max‑convolution; the temperature β is
     annealed exponentially up to a very large value so that the
     approximation becomes tight.
4.   Finish with a greedy‑plus‑simulated‑annealing hill‑climb that
     repeatedly swaps or transfers a small packet of mass between two
     coordinates, accepting improving moves and occasionally
     accepting worse ones according to a cooling schedule.

All phases respect the 1000 s wall‑clock budget.  The returned object
is a plain Python list of floats; its score is printed for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) for a vector that is already on the
    probability simplex (sum = 1).  The value coincides with
    `evaluate_sequence` for such a vector.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families (all vectors are normalised to sum = 1)
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular bump."""
    i = np.arange(n, dtype=float)
    w = np.minimum(i + 1, n - i)
    return w / w.sum()


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑mid)²/(2·σ²))."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    return w / w.sum()

# ----------------------------------------------------------------------
# Mirror‑descent optimisation of a smooth approximation of the max‑conv.
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 0.5,
    beta_end: float = 2000.0,
    eta0: float = 0.6,
    eps: float = 1e-14,
):
    """
    Minimise the log‑sum‑exp smoothing of max(conv) using
    exponentiated‑gradient (KL‑mirror) steps.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.size

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β annealing (exponential schedule)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed soft‑max weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stabilise the exponentials
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smoothed objective
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # step size (η_t = η0 / √t)
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update, keep positivity
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc

# Final hill‑climb (greedy + simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_simulated(start_seq: np.ndarray, deadline: float, max_steps: int = 8_000_000):
    """
    Greedy random walk with occasional uphill moves.
    Moves:
      – swap two entries,
      – transfer a small fraction of mass from i to j.
    The temperature decays exponentially, turning the process into a
    pure greedy climb near the end.
    """
    cur = start_seq.copy()
    n = cur.size
    best = cur.copy()
    best_sc = compute_score(best)

    # annealing parameters
    T0 = 1.0
    T_end = 1e-4

    step = 0
    while step < max_steps and time.time() < deadline:
        # propose a move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        if np.random.rand() < 0.5:                     # swap
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # transfer
            fraction = 0.02 * np.random.rand()          # up to 2 % of the source mass
            delta = fraction * cur[i]
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
            cur = cand.copy()
        else:
            # accept with Metropolis probability
            t = step / max_steps
            T = T0 * (T_end / T0) ** t
            if np.random.rand() < np.exp(-(sc - best_sc) / (T + 1e-12)):
                cur = cand.copy()

        step += 1

    return best, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine performs:
        * a quick scan of several analytic families (including random permutations),
        * many short smooth‑max optimisations from random Dirichlet starts,
        * a long exponentiated‑gradient refinement of the current best,
        * a final hill‑climb (greedy + simulated annealing).
    All phases respect the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a workable length according to remaining time
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 16384
    elif remaining > 400:
        n = 8192
    elif remaining > 200:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Scan analytic families (including permutations)
    # -------------------------------------------------
    a_scan, sc_scan = scan_families(n, deadline - 300.0)
    if a_scan is not None and sc_scan < best_score:
        best_score = sc_scan
        best_seq = a_scan.tolist()
        print(f"[{time.time():.1f}] after family scan best = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random Dirichlet starts with short smooth optimisation
    # -------------------------------------------------
    dirichlet_alphas = [0.3, 0.6, 1.0, 2.0]
    for alpha in dirichlet_alphas:
        for _ in range(6):
            if time.time() >= deadline - 200.0:
                break
            a0 = np.random.dirichlet([alpha] * n)
            sc0 = compute_score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.tolist()
                print(f"[{time.time():.1f}] rand Dirichlet(α={alpha}) improved = {best_score:.6f}")

            # short smooth optimisation (≈60 k iterations)
            a_opt, sc_opt = exp_smooth_optimize(
                a0,
                deadline - 150.0,
                max_iters=60_000,
                beta_start=0.5,
                beta_end=800.0,
                eta0=0.7,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}] after short smooth opt = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long exponentiated‑gradient optimisation on the current best
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        print(f"[{time.time():.1f}] starting long smooth optimisation")
        a_long, sc_long = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 40.0,
            max_iters=400_000,
            beta_start=0.5,
            beta_end=3000.0,
            eta0=0.6,
        )
        if sc_long < best_score:
            best_score = sc_long
            best_seq = a_long.tolist()
            print(f"[{time.time():.1f}] after long smooth opt best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Final hill‑climb (greedy + simulated annealing)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb_simulated(
            np.array(best_seq, dtype=float), deadline, max_steps=8_000_000
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb new best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a tiny uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n_small}")

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
