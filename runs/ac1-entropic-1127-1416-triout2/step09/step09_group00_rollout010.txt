    n = len(start_a)
    eps = 1e-12
    a = np.array(start_a, dtype=np.float64)
    a = np.maximum(a, eps)
    a = a / a.sum()

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    best_a = a.copy()
    best_score = compute_score(best_a)
    start_time = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # schedule beta
        elapsed = time.time() - start_time
        total = deadline - start_time
        t_frac = elapsed / total if total>0 else 1.0
        beta = beta_start * (beta_end / beta_start) ** t_frac
        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute softmax weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()
        # gradient wrt a
        a_rev = a[::-1]

        grad_a = 2.0 * conv_rev[n-1:n-1+n]

"""Optimise a non‑negative step‑function to minimise
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in several stages:

1.  Build a very small starter (n=256) from a few random analytic families
    and keep the best shape (sum normalised to 1).

2.  Multiscale up‑sampling: repeatedly double the length up to 4096.
    After each up‑sample the current shape is refined with a
    smoothed‑max Adam optimiser.  The smoothing parameter β is increased
    exponentially during the allotted time, giving a good approximation of
    the true max‑objective while keeping gradients well‑behaved.

3.  A cheap projected sub‑gradient descent on the exact max‑objective
    (step size ∝ 1/√t) cleans up any remaining spikes.

4.  Optional symmetrisation – many good solutions are symmetric.

5.  A final stochastic‑annealing hill‑climb (swap / mass‑transfer moves)
    explores the neighbourhood for a last improvement.

All stages respect the global 1000 s budget and the function finally
returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution using real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (Σa = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for quick initial seeds)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, then normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing simplex vector by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Sample a random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         beta_start: float = 2.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on a = softmax(z).  The smoothing β grows
    exponentially from beta_start to beta_end over the allotted time.
    """
    a = np.maximum(start_a, 1e-12)
    a = a / a.sum()
    z = np.log(a)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12
    best_a = a.copy()
    best_score = compute_score(best_a)

    start_time = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # schedule β according to elapsed fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        beta = beta_start * (beta_end / beta_start) ** frac

        a = softmax(z)                     # stay on the simplex
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights (log‑sum‑exp trick)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a: 2 * (a_rev * w) centred on the original length
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        n = a.shape[0]
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     step0: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for 2·n·max(a*a) with diminishing step size.
    The iterate is projected onto the probability simplex each step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    a = a / a.sum()                     # ensure we start on the simplex

    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))       # index of the active maximal entry
        # sub‑gradient of the active entry
        idx = np.arange(n, dtype=int)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex: random swaps or mass‑transfer moves,
    occasional uphill acceptance governed by a slowly cooling temperature.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # base temperature proportional to current best score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            # accept
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – the harness adds its own checks
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Build a modest starter at n=256 from a handful of families
    # ------------------------------------------------------------------
    n0 = 256
    cur_best = None
    cur_best_score = float("inf")
    # random seeds
    for _ in range(30):
        if time.time() >= deadline:
            break
        typ = np.random.choice(["mono", "u", "pow"])
        if typ == "mono":
            vec = random_monotone_start(n0)
        elif typ == "u":
            vec = random_u_start(n0)
        else:
            beta = np.random.uniform(0.0, 12.0)
            vec = left_heavy_power_family(n0, beta)
        sc = compute_score(vec)
        if sc < cur_best_score:
            cur_best_score = sc
            cur_best = vec.copy()
    # a few points from the symmetric product family
    for alpha in np.linspace(0.0, 12.0, 13):
        if time.time() >= deadline:
            break
        vec = symmetric_product_family(n0, alpha)
        sc = compute_score(vec)
        if sc < cur_best_score:
            cur_best_score = sc
            cur_best = vec.copy()

    if cur_best is None:
        cur_best = np.full(n0, 1.0 / n0, dtype=np.float64)
        cur_best_score = compute_score(cur_best)

    best_seq = cur_best.tolist()
    best_score = cur_best_score
    print(f"[{time.time():.1f}] initial seed n={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Multiscale up‑sampling + Adam refinements
    # ------------------------------------------------------------------
    sizes = [256, 512, 1024, 2048, 4096]   # doubling chain
    for n in sizes[1:]:
        if time.time() >= deadline - 200.0:

        old_n = cur_best.shape[0]
        x_old = np.arange(old_n, dtype=np.float64)
        x_new = np.linspace(0, old_n - 1, n, dtype=np.float64)
        up_vec = np.interp(x_new, x_old, cur_best)
        up_vec = up_vec / up_vec.sum()

        # allocate a slice of the remaining budget for this size
        remaining = deadline - time.time()

            a_opt, sc_opt = adam_smooth_optimize(

        sym = (arr + arr[::-1]) * 0.5
        sym = project_to_simplex(sym)
        sc_sym = compute_score(sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improvement → {sc_sym:.6f}")

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climb (SA)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,

        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improvement → {sc_final:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard: fall back to a modest uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = [1.0 / n_small] * n_small

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
