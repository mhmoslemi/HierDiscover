def maximize_sum_under_convolution_constraint(f, M):
    n = len(f)
    # Build convolution matrix as sparse
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    data = []
    # for each i (col), iterate k from i to i+n-1
    for i in range(n):
        # row indices where f contributes: k = i + j, where j in [0, n-1]
        # So rows = i + np.arange(n)
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
        data.extend(f.tolist())
    # Build sparse matrix of shape (2n-1, n)
    conv_mat = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    b = cp.Variable(n, nonneg=True)
    constraints = [conv_mat @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    if b.value is None:
        # fallback
        return np.zeros_like(f)
    return np.array(b.value).flatten()
```

"""
search for a low‑score non‑negative step‑function.

The algorithm proceeds in several stages, each respecting the global time
budget (≈ 1000 s):

1. **Analytic families** – exhaustive grids for several parametrised families
   (symmetric product, left‑heavy power‑law, exponential decay, and a
   two‑parameter Beta‑family).  The best vector found becomes the current
   incumbent.

2. **Stochastic restarts + Adam on a log‑sum‑exp surrogate** – the true
   max‑convolution is approximated by a smooth log‑sum‑exp with a
   temperature β that is annealed from 5 to a large value (≈ 5000).  The
   variables are re‑parameterised with a soft‑max to enforce non‑negativity
   and the simplex constraint.  Several restarts are performed until the
   remaining time drops below a safety margin.

3. **Mirror (entropy‑mirror) sub‑gradient descent** on the exact
   non‑smooth objective (the maximal entry of the self‑convolution).  The
   step‑size follows ηₜ = η₀ / √t and the iterate is kept on the simplex via
   an explicit normalisation.

4. **Symmetrisation** – many optimal solutions are symmetric, so we
   average the vector with its reverse and project onto the simplex.

5. **Final stochastic hill‑climb with a tiny simulated‑annealing schedule**,
   allowing occasional uphill moves to escape shallow local minima.

The routine returns a plain Python list of floats.  All intermediate scores
are printed for monitoring.
"""

import time
import numpy as np
from scipy import sparse

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective from the statement for a generic vector (no scaling
    is performed here – the function is linear in the sum of the entries).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max()) / (float(a.sum()) ** 2)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Discrete Beta‑like family:
        a_i ∝ (i+½)^p · (n‑½‑i)^q .
    Allows independent left/right shaping.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    left = np.power(i, p, where=(i > 0), out=np.ones_like(i))
    right = np.power(n + 0.5 - i, q, where=(n + 0.5 - i > 0), out=np.ones_like(i))
    w = left * right
    w = np.clip(w, 1e-300, None)        # avoid exact zeros that kill the gradient
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (by sorting a power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) via Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially.
    """
    eps = 1e-12
    z = np.log(start_a + eps)            # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # annealing schedule for β (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # ----- smoothed weights w_k ∝ exp(β·conv_k) -----
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # softmax over positions

        # ----- gradient of f_β w.r.t. a (see derivation in the original code) -----
        a_rev = a[::-1]
        # convolution of reversed a with the weight vector w
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ----- back‑prop through the soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy‑mirror) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    check_interval: int = 1000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for the exact
    objective 2·n·max(a*a).  The step size decays as ηₜ = step0 / √t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))               # active index of the max entry

        # sub‑gradient for the active entry only
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)               # multiplicative update (non‑negative)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated annealing
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_000_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a tiny temperature schedule.
    Moves are either swapping two coordinates or transferring a fraction
    of mass.  Occasional uphill moves are accepted according to a
    Metropolis rule.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor          # initial temperature
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise to the simplex
        cand = cand / cand.sum()

        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a length that balances flexibility and per‑iteration cost.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 4096
    elif budget_s >= 400:
        n = 2048
    else:
        n = 1024
    print(f"[{time.time():.1f}] using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback to a previously known construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families – dense grid search (stops early if time runs out)
    # ------------------------------------------------------------------
    # symmetric product family (alpha ∈ [0,15])
    alphas = np.concatenate([np.linspace(0.0, 5.0, 101), np.linspace(5.0, 15.0, 101)])
    for a in alphas:
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={a:.2f}) = {best_score:.6f}")

    # left‑heavy power‑law family (beta ∈ [0,10])
    betas = np.concatenate([np.linspace(0.0, 5.0, 101), np.linspace(5.0, 10.0, 101)])
    for b in betas:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, b)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {best_score:.6f}")

    # exponential decay family (gamma ∈ [0,10])
    gammas = np.linspace(0.0, 10.0, 121)
    for g in gammas:
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, g)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={g:.2f}) = {best_score:.6f}")

    # two‑parameter Beta‑family (p,q ∈ [-5,5] on a coarse grid)
    ps = np.linspace(-5.0, 5.0, 41)
    qs = np.linspace(-5.0, 5.0, 41)
    for p in ps:
        for q in qs:
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (β‑family p={p:.2f},q={q:.2f}) = {best_score:.6f}")

    # random monotone draws (hundred attempts)
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed optimisation (Adam) – multiple restarts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        start_vec = random_sparse_start(n) if best_seq is None else np.array(best_seq)
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 200.0,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=5000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam‑smooth = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Mirror (entropy‑mirror) sub‑gradient descent on exact max‑objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            step0=0.8,
            check_interval=500,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Symmetrisation – many optimal solutions are symmetric
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb (tiny simulated‑annealing) for last few seconds
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing succeeded fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the external harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq
