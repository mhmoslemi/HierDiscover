def simulated_annealing(a_start, deadline):
    a = a_start.copy()
    n = a.shape[0]
    conv = convolution_fft(a, a)
    best_a = a.copy()
    best_score = 2*n*conv.max()
    current_score = best_score

    step_scale = 0.12
    # temperature initial maybe (best_score) * 0.001?
    T0 = best_score * 0.001  # small relative
    T = T0
    decay = 0.99995
    iter = 0
    # loops
    while time.time() < deadline:
        # propose move
        # decide transfer or swap
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # small transfer
            src = i if a[i] > a[j] else j
            dst = j if src == i else i
            delta = step_scale * a[src]
        else:
            # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # move full difference
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = a[src] - a[dst]

        # compute conv_cand incremental (copy conv)
        conv_cand = conv.copy()
        conv_cand[src:src+n] -= 2.0 * delta * a
        conv_cand[dst:dst+n] += 2.0 * delta * a
        conv_cand[2*src] += delta * delta
        conv_cand[2*dst] += delta * delta
        conv_cand[src+dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        # acceptance
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probabilistic
            # if T is extremely low, acceptance negligible
            if T > 1e-12 and np.random.rand() < np.exp(-(cand_score - current_score) / T):
                accept = True

        if accept:
            # update a
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            current_score = cand_score
            # update best
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                # possibly increase step_scale
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # maybe adjust step_scale down slightly
            step_scale *= 0.97

        iter += 1
        # temperature schedule
        T *= decay

        # occasional recompute full conv to correct drift
        if iter % 200000 == 0:
            conv = convolution_fft(a, a)
            # recompute current_score
            current_score = 2.0 * n * conv.max()
```

"""
Improved search for a low‑score step‑function.

The algorithm follows several stages, each respecting the 1000 s wall‑clock
budget:

1. **Fallback** – if a global variable ``height_sequence_1`` is present it
   is used as a starting point.

2. **Analytic seeds** – a symmetric “Beta‑type’’ family
   a_i ∝ (i+½)^{‑α}(n‑½‑i)^{‑α} and a Gaussian‑type family
   a_i ∝ exp(‑γ·((i‑c)/n)²) are scanned on a modest grid.  The best
   seed (lowest exact score) is kept.

3. **Exponentiated‑gradient smoothing** – minimise a log‑sum‑exp
   surrogate of the max‑convolution using KL‑mirror updates with an
   annealed temperature β.  This quickly approaches a good local optimum.

4. **Projected sub‑gradient descent** – classic Euclidean‑projected
   sub‑gradient method on the true objective 2·n·max(a⊛a).

5. **Simulated‑annealing hill‑climb** – a stochastic walk that
   transfers a packet of mass between two coordinates (or swaps them).
   Moves that worsen the objective are occasionally accepted with a
   temperature that decays exponentially, allowing the search to jump
   out of local minima.

6. **Strict hill‑climb polish** – a final deterministic walk that only
   accepts strictly improving moves, using the same O(n) incremental
   convolution update as the original solver.

All stages keep the vector normalised (sum = 1) and non‑negative.
The best sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This matches what evaluate_sequence returns for a normalised input.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def alpha_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric beta‑type family:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Clip infinities that may arise for extreme α
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """
    Symmetric Gaussian‑type family centred at (n‑1)/2:
        a_i ∝ exp(‑γ·((i‑c)/n)²)
    """
    c = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    dist2 = ((i - c) / n) ** 2
    w = np.exp(-gamma * dist2)
    a = w / w.sum()
    return a


def find_best_analytic(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a modest grid over both families and keep the best vector."""
    best_a = None
    best_sc = float("inf")

    # α grid (symmetric Beta‑type)
    alphas = np.concatenate([
        np.linspace(-1.0, 0.0, 11),   # centre‑heavy
        np.linspace(0.0, 2.5, 26)    # edge‑heavy
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_family(n, alpha)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Gaussian grid (symmetric bell‑shape)
    gammas = np.concatenate([
        np.linspace(0.0, 2.0, 21),   # shallow to narrow
        np.linspace(2.0, 12.0, 21)   # very peaked
    ])
    for gamma in gammas:
        if time.time() >= deadline:
            break
        a = gaussian_family(n, gamma)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exponentiated‑gradient optimisation (smooth max surrogate)
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000,
                        beta_start: float = 0.5,
                        beta_end: float = 500.0,
                        eta0: float = 0.7,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the smoothed objective
        f_β(a) = (1/β) log Σ_x exp( β·(a⊛a)(x) )
    using a KL‑mirror (exponentiated‑gradient) update.
    β is annealed exponentially from ``beta_start`` to ``beta_end``.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smooth max (log‑sum‑exp trick)
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))    # stabilised
        w = expw / expw.sum()                  # softmax weights

        # gradient w.r.t. a (cross‑correlation with the weights)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # step size (decreasing)
        eta = eta0 / np.sqrt(it)

        # KL‑mirror (exponentiated‑gradient) update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 250_000,
                     step0: float = 0.08) -> tuple[np.ndarray, float]:
    """Classic Euclidean projected sub‑gradient method."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: derivative of max component
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Simulated‑annealing stochastic hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(start_a: np.ndarray,
                        deadline: float,
                        init_temp_factor: float = 0.001,
                        decay: float = 0.99995,
                        recompute_interval: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Stochastic walk that transfers or swaps mass between two coordinates.
    Non‑improving moves are accepted with probability exp(-Δ/T),
    where T decays exponentially.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)            # exact convolution
    best_a = a.copy()
    best_sc = 2.0 * n * float(conv.max())
    cur_sc = best_sc

    # temperature initialised as a fraction of the current score
    T = max(best_sc * init_temp_factor, 1e-8)

    step_scale = 0.12
    it = 0
    while time.time() < deadline:
        # ---- propose a move -------------------------------------------------
        if np.random.rand() < 0.5:
            # transfer a small packet from src → dst
            src = np.random.randint(n)
            if a[src] == 0.0:
                continue
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = step_scale * a[src]               # 0 < delta ≤ a[src]
        else:
            # swap the values of two positions (full exchange)
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = a[src] - a[dst]                    # positive

        # ---- incremental convolution update --------------------------------
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_sc = 2.0 * n * cand_max

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            if T > 1e-12 and np.random.rand() < np.exp(-(cand_sc - cur_sc) / T):
                accept = True

        if accept:
            # apply the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a.copy()
                step_scale = min(step_scale * 1.07, 0.5)   # modestly increase packet size
        else:
            # rejected – slowly shrink the packet size
            step_scale *= 0.97

        # temperature decay
        T *= decay
        it += 1

        # occasional full recomputation to curb drift
        if it % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_sc = 2.0 * n * float(conv.max())

    return best_a, best_sc

# 5. Fast deterministic hill‑climb (strict improvement)
# ----------------------------------------------------------------------


def fast_hill_climb(start_a: np.ndarray,
                    deadline: float,
                    inc_factor: float = 1.07,
                    decay_factor: float = 0.97,
                    recompute_interval: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pure improvement walk – only moves that strictly lower the objective are kept.
    Uses the same O(n) incremental convolution update as the SA routine.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)               # exact convolution
    best_a = a.copy()
    best_sc = 2.0 * n * float(conv.max())
    step_scale = 0.12

    steps = 0
    while time.time() < deadline:
        # pick move type
        if np.random.rand() < 0.5:
            # swap (full exchange)
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = a[src] - a[dst]            # positive, results in a swap
        else:
            # small transfer
            src = np.random.randint(n)
            if a[src] == 0.0:
                continue
            delta = step_scale * a[src]         # always ≤ a[src]
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

        # incremental candidate convolution
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_sc = 2.0 * n * cand_max

        if cand_sc < best_sc:
            # accept
            best_sc = cand_sc
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            best_a = a.copy()
            step_scale = min(step_scale * inc_factor, 0.5)
        else:
            # reject – decay step size
            step_scale *= decay_factor

        steps += 1
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)

    return best_a, best_sc
