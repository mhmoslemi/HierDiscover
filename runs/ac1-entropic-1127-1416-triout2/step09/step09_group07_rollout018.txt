"""
Hybrid optimizer for the step‑function attack.

The algorithm proceeds in four (time‑aware) phases:

1️⃣  Analytic seeding – a dense scan of two parametric families
    (power‑law and exponential decay) plus a few random Dirichlet vectors.
    The best‑scoring vector (already normalised to sum = 1) is kept.

2️⃣  Smooth exponential‑gradient descent – minimises a log‑sum‑exp
    surrogate of the max‑convolution using KL‑mirror updates.  The
    temperature β is annealed to focus on the true max.

3️⃣  Euclidean projected sub‑gradient descent on the exact objective.

4️⃣  Fast stochastic hill‑climb with simulated‑annealing acceptance.
    Moves are performed by transferring a packet of mass from a source
    index to a destination index; the convolution is updated in O(n)
    by an incremental formula.  A fraction of the moves target the
    index currently attaining the maximal convolution value, which
    strongly pushes the max down.  The temperature decays linearly with
    the remaining wall‑clock time.  After the main run, short
    perturb‑and‑re‑climb cycles are performed while time permits.

All phases respect the 1000 s budget, and the best sequence found is
returned as a plain list of non‑negative floats (sum≈1, the scale is
irrelevant for the evaluation function).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector whose entries sum to 1.
    This matches the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Analytic families
# ----------------------------------------------------------------------


def power_alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^(-α)   (symmetric, edge‑heavy for α>0).
    Returned vector is normalised to sum‑1.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # avoid degenerate all‑zero vectors for extreme α
    if np.all(w == 0):
        w = np.ones_like(w)
    a = w / w.sum()
    return a


def exponential_decay_start(n: int, lam: float) -> np.ndarray:
    """
    a_i ∝ exp(-λ·min(i, n‑1‑i))   (large λ → mass at the two ends).
    Returned vector is normalised to sum‑1.
    """
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    w = np.exp(-lam * d)
    a = w / w.sum()
    return a


def find_best_analytic_start(n: int, deadline: float):
    """
    Scan several parametric families (α‑grid, λ‑grid) and a few random
    Dirichlet vectors.  Returns the best vector found and its score.
    """
    best_a = None
    best_score = float("inf")

    # 1. power‑law family
    alphas = np.linspace(0.0, 6.0, 61)          # step 0.1
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = power_alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # 2. exponential decay family
    lambdas = np.linspace(0.0, 6.0, 61)        # step 0.1
    for lam in lambdas:
        if time.time() >= deadline:
            break
        a = exponential_decay_start(n, lam)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # 3. a few random Dirichlet starts (useful when time permits)
    for _ in range(12):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score

# 2. Smooth exponentiated‑gradient optimisation
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000,
                        beta_start: float = 0.5,
                        beta_end: float = 2000.0,
                        eta0: float = 0.7,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate f_β using a KL‑mirror (exponentiated‑gradient)
    update.  β grows exponentially from beta_start to beta_end.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smooth max via log‑sum‑exp
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))        # shifted for stability
        w = expw / expw.sum()                     # softmax weights

        # gradient of the smooth objective (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)      # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        # KL‑mirror (exponentiated‑gradient) update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        if it % 400 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 250_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient method for 2·n·max(a⊛a).
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑convolution
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 800 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

def hill_climb_annealing(start_a: np.ndarray,
                         deadline: float,
                         target_prob: float = 0.4,
                         inc_factor: float = 1.07,
                         decay_factor: float = 0.97,
                         recompute_interval: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Random‑walk polishing.
    * Moves consist of a small packet of mass transferred from src → dst.
    * The convolution is updated incrementally in O(n).
    * A linear temperature schedule (proportional to remaining time)
      implements simulated annealing, allowing occasional uphill moves.
    * With probability `target_prob` the source index is chosen as the one
      that contributes most to the current maximal convolution entry.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)               # exact convolution
    current_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_score = current_score

    start_time = time.time()
    total_time = deadline - start_time
    step_scale = 0.12

    steps = 0
    while time.time() < deadline:
        # linear temperature proportional to remaining time
        remaining = max(0.0, deadline - time.time())
        T = (remaining / total_time) * max(0.02 * current_score, 1e-12)

        # ---- pick source index -------------------------------------------------
        if np.random.rand() < target_prob:
            # target the index that currently attains max(conv)
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_vals = np.arange(i_start, i_end + 1)
            j_vals = k_max - i_vals
            prod = a[i_vals] * a[j_vals]
            src = i_vals[np.argmax(prod)]
            if a[src] == 0.0:
                src = np.random.randint(n)
        else:
            src = np.random.randint(n)
            if a[src] == 0.0:
                continue

        # ---- pick destination --------------------------------------------------
        if np.random.rand() < 0.5:
            # random destination
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
        else:
            # destination with minimal current mass
            dst = int(np.argmin(a))
            if dst == src:
                # second‑smallest if clash
                sorted_idx = np.argsort(a)
                for cand in sorted_idx:
                    if cand != src:
                        dst = cand
                        break

        # ---- packet size -------------------------------------------------------
        delta = step_scale * a[src]
        if delta <= 0.0:
            continue

        # ---- incremental convolution update ------------------------------------
        conv_cand = conv.copy()
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        # ---- acceptance (simulated annealing) ----------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        elif T > 0.0 and np.random.rand() < np.exp(-(cand_score - current_score) / T):
            accept = True

        if accept:
            # apply the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            current_score = cand_score

            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                step_scale = min(step_scale * inc_factor, 0.5)
            else:
                step_scale = max(step_scale * 0.999, 1e-5)
        else:
            # reject – shrink the step size a bit
            step_scale = max(step_scale * decay_factor, 1e-5)

        steps += 1
        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)
            current_score = 2.0 * n * float(conv.max())
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

    return best_a, best_score

        pass

    # -------------------------------------------------
    # 1. Analytic seeding (dense grid + few random starts)
    # -------------------------------------------------
    analytic_deadline = deadline - 250.0
    a_analytic, sc_analytic = find_best_analytic_start(n, analytic_deadline)
    if a_analytic is not None and sc_analytic < best_score:
        best_score = sc_analytic
        best_seq = a_analytic.tolist()
        print(f"[{time.time():.1f}] best after analytic = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Smooth exponentiated‑gradient optimisation (single run)
    # -------------------------------------------------
    now = time.time()
    if best_seq is not None and now < deadline - 200.0:
        exp_deadline = now + min(180.0, deadline - now - 150.0)
        print(f"[{time.time():.1f}] exp‑gradient optimisation")
        a_exp, sc_exp = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            exp_deadline,
            max_iters=180_000,
            beta_start=0.5,
            beta_end=2000.0,
            eta0=0.7,
        )
        if sc_exp < best_score:
            best_score = sc_exp
            best_seq = a_exp.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Sub‑gradient refinement
    # -------------------------------------------------
    now = time.time()
    if best_seq is not None and now < deadline - 120.0:
        sub_deadline = now + min(150.0, deadline - now - 110.0)
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            sub_deadline,
            max_iters=180_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Fast hill‑climb with simulated annealing (majority of budget)
    # -------------------------------------------------
    now = time.time()
    if best_seq is not None and now < deadline - 30.0:
        print(f"[{time.time():.1f}] hill‑climb (simulated annealing) – start")
        a_hc, sc_hc = hill_climb_annealing(
            np.array(best_seq, dtype=float),
            deadline,
        )
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

        # ---- Additional short perturb‑and‑re‑climb cycles if time permits ----
        while True:
            now = time.time()
            time_left = deadline - now
            if time_left < 30.0:
                break
            # small random perturbation of the current best
            a_start = np.array(best_seq, dtype=float)
            a_start += 0.02 * np.random.randn(n)
            a_start[a_start < 0.0] = 0.0
            if a_start.sum() == 0.0:
                a_start = np.full(n, 1.0 / n)
            else:
                a_start /= a_start.sum()
            sub_deadline = now + min(25.0, time_left - 5.0)
            a_hc2, sc_hc2 = hill_climb_annealing(a_start, sub_deadline)
            if sc_hc2 < best_score:
                best_score = sc_hc2
                best_seq = a_hc2.tolist()
                print(f"[{time.time():.1f}] perturbed hill‑climb new best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    # Normalise just in case numerical drift accumulated
    best_arr = np.array(best_seq, dtype=float)
    if best_arr.sum() == 0.0:
        best_arr = np.full_like(best_arr, 1.0 / best_arr.size)
    else:
        best_arr /= best_arr.sum()
    final_score = evaluate_sequence(best_arr.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_arr.tolist()
