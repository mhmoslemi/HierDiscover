import cvxpy as cp
from scipy.sparse import csr_matrix

def solve_lp_for_b(a, M):
    n=len(a)
    # Build sparse matrix A (2n-1, n)
    rows = []
    cols = []
    data = []
    for j in range(n):
        # indices where a_i contributes: row = i + j for i in 0..n-1
        row_inds = np.arange(j, j+n)
        rows.append(row_inds)
        cols.append(np.full(n, j))
        data.append(a)   # a is length n
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_var = cp.Variable(n)
    constraints = [A @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if b_var.value is None:
        return None
    return np.array(b_var.value)
```

"""
Propose a step‑function sequence that minimises

    2·n·max(convolve(a,a)) / (sum(a)**2)

under the non‑negative simplex constraint.

Algorithm
---------
0.  Load a fallback sequence if available, otherwise start from the uniform
    vector (sum = 1).

1.  **Linear‑programming refinement** – for the current vector `a`
    (sum = 1) compute its peak `M = max(conv(a,a))`.  Solve the LP

        maximise   sum(b)
        subject to conv(a,b) ≤ M  (element‑wise)
                   b ≥ 0

    (the convolution is linear in `b`).  The optimal `b` puts mass where
    `a` is small while respecting the same peak `M`.  Mix `a` and `b`
    (simple line‑search over a few mixing ratios) and keep the mixture
    that yields the smallest new peak.  Repeat this step a few times
    until no further improvement or time runs out.

2.  **Smooth‑max Adam optimisation** – on the refined vector run many
    iterations of Adam on the differentiable surrogate

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    with an exponentially increasing smoothness parameter `β`.  The
    optimisation stays on the simplex via the soft‑max parametrisation.
    This tightens the peak further.

3.  **Symmetrisation** – many optimal solutions are symmetric.
    Replace `a` by `(a + a[::-1]) / 2` and keep the better of the two.

4.  Return the final list of floats.  All intermediate work respects the
    global wall‑clock budget (default 1000 s).  The evaluation uses
    `evaluate_sequence`, which is invariant under scaling, so we keep the
    sum of the vector equal to 1 throughout the search.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic (full) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def max_conv(a: np.ndarray) -> float:
    """Maximum entry of the a‑self‑convolution."""
    return convolution_fft(a, a).max()


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    return 2.0 * n * max_conv(a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x ≥ 0, sum(x)=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Soft‑max (stable) – used by Adam optimizer
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max (log‑sum‑exp) objective
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 300.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """Minimise the smoothed max of a*a via Adam on the soft‑max parametrisation."""
    n = start_a.shape[0]
    eps = 1e-12
    # initialise logits for soft‑max
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # shape (2n‑1,)

        # gradient wrt a (see derivation in the original script)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 2n‑1
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_sc:
                best_sc = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_sc:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_sc:
        best_sc = sc_fin
        best_a = a_fin.copy()
    return best_a, best_sc

# Linear‑programming refinement utilities

def build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that A @ b == conv(a,b) (full convolution).
    Each column j is the vector a shifted by j positions.
    """
    n = a.shape[0]
    m = 2 * n - 1
    nn = n * n
    rows = np.empty(nn, dtype=np.int32)
    cols = np.empty(nn, dtype=np.int32)
    data = np.empty(nn, dtype=np.float64)

    idx = 0
    for j in range(n):
        rows[idx:idx + n] = np.arange(j, j + n, dtype=np.int32)
        cols[idx:idx + n] = j
        data[idx:idx + n] = a
        idx += n

    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))
    return A


def solve_lp_for_b(a: np.ndarray, M: float, time_limit: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(b)
        subject to conv(a,b) ≤ M   (full convolution)
                   b ≥ 0
    Returns the optimal b (not yet normalised).  Returns None on failure
    or if we run out of time.
    """
    n = a.shape[0]
    A = build_conv_matrix(a)                # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)       # maximise sum(b) == minimise -sum(b)
    bounds = [(0, None)] * n

    # linprog automatically respects a time budget via `options['time_limit']`
    # (available in recent SciPy).  We fall back to a simple check ourselves.
    remaining = max(0.0, time_limit - time.time())
    # SciPy's 'highs' method accepts a `time_limit` in seconds (since 1.7.0)
    # but to stay compatible we simply let it run; the outer driver will abort
    # if the deadline is exceeded.
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
    except Exception as e:
        print(f"linprog exception: {e}")
        return None

    if not res.success:
        print(f"LP not solved successfully: {res.message}")
        return None
    return np.array(res.x, dtype=np.float64)


def mix_optimal(a: np.ndarray, b: np.ndarray, M_current: float) -> tuple[np.ndarray, float]:
    """
    Perform a tiny line‑search over mixing ratios t∈[0,1] and return the
    mixture a_t = (1‑t)·a + t·b with the smallest new peak.
    """
    n = a.shape[0]
    best_a = a
    best_M = M_current
    # candidate mixing ratios – dense enough to catch a small improvement
    cand_ts = np.linspace(0.0, 1.0, 11)  # 0,0.1,…,1.0
    for t in cand_ts:
        a_t = (1.0 - t) * a + t * b
        # a_t stays on the simplex because both a and b sum to 1
        M_t = max_conv(a_t)
        if M_t < best_M - 1e-12:
            best_M = M_t
            best_a = a_t
    return best_a, best_M

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine respects the global time budget
    (default 1000 seconds) and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback (if a previously discovered sequence is available)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Initialise a reasonable starting point if needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 2048
        best_seq = [1.0 / n0] * n0
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] using uniform start, score = {best_score:.6f}")

    # Normalise to sum = 1 for all internal work (the objective is homogeneous)
    a = np.array(best_seq, dtype=np.float64)
    a = a / a.sum()
    n = a.shape[0]

    # ------------------------------------------------------------------
    # 2.  LP‑based refinement (few iterations)
    # ------------------------------------------------------------------
    max_lp_iters = 8
    lp_iter = 0
    while lp_iter < max_lp_iters and time.time() < deadline - 10.0:
        M = max_conv(a)
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] new best after LP start = {best_score:.6f}")

        # Solve LP to get a partner vector b
        b_raw = solve_lp_for_b(a, M, deadline)
        if b_raw is None:
            print("[LP] solver failed – exiting LP loop")
            break

        sum_b = b_raw.sum()
        if sum_b <= 1e-12:
            print("[LP] degenerate solution – exiting LP loop")
            break
        b = b_raw / sum_b                     # re‑normalise to sum = 1

        # Mix a and b, keep best mixture
        a_new, M_new = mix_optimal(a, b, M_current=M)

        # If no improvement, stop the LP phase
        if M_new >= M - 1e-12:
            print("[LP] no improvement from mixing – terminating LP phase")
            break

        a = a_new
        lp_iter += 1
        print(f"[{time.time():.1f}] LP iteration {lp_iter} improved peak to {M_new:.6e}")

    # ------------------------------------------------------------------
    # 3.  Adam optimisation on the smooth‑max surrogate (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # Roughly 2500 Adam steps per second (empirically fast on small n)
        max_iters = max(10_000, int(remaining * 2500))
        print(f"[{time.time():.1f}] starting Adam smoothing (≈{max_iters} iters)")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            deadline,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=800.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            a = a_opt
            print(f"[{time.time():.1f}] after Adam smoothing best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Symmetrisation test (many optima are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)    # enforce sum = 1
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_seq
