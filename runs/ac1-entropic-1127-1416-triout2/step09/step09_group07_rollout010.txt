        i_src = idx[max_idx]
        j_src = j[max_idx]
        src = i_src if a[i_src] >= a[j_src] else j_src
        if a[src] <= 0:
            # fallback random src
            src = np.random.randint(n)
            if a[src] <= 0:
                continue

        # pick dst among few random candidates with smallest a
        # ensure distinct from src
        cand = np.random.choice(n, size=5, replace=False)
        cand = cand[cand != src]
        if cand.size == 0:
            continue
        dst = cand[np.argmin(a[cand])]

        # delta as fraction of src mass
        delta = step_scale * a[src] * np.random.rand()
        if delta <= 0 or delta > a[src]:
            continue

        # incremental conv update
        conv_cand = conv.copy()
        conv_cand[src:src+n] -= 2.0 * delta * a
        conv_cand[dst:dst+n] += 2.0 * delta * a
        conv_cand[2*src] += delta * delta
        conv_cand[2*dst] += delta * delta
        conv_cand[src+dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_sc = 2.0 * n * cand_max

        if cand_sc < cur_sc - 1e-14:
            # accept improvement
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a.copy()
            step_scale = min(step_scale * inc_factor, max_step)
            no_improve_counter = 0
        else:
            step_scale = max(step_scale * decay_factor, min_step)
            no_improve_counter += 1

        iteration += 1
        if iteration % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_sc = 2.0 * n * conv.max()

        # optional early exit if no improvement for long
        if no_improve_counter > 50000 and time.time() + 30 < deadline:
            # break to allow outer restarting
            break

    return best_a, best_sc

"""
Propose a step‑function that minimises the evaluation metric.

Algorithm
---------
1. Choose a suitable length `n` (8192 when time permits).
2. Analytic seeding: scan a one‑parameter family
   a_i(α) ∝ [(i+½)(n‑½‑i)]^(-α) over a modest α‑grid
   and keep the best vector (sum = 1).
3. Generate a few random Dirichlet starts and keep the best.
4. Guided stochastic hill‑climb:
   * Always move mass from the index that contributes most
     to the current maximal convolution entry.
   * Destination is chosen among a few random low‑mass indices.
   * The move size (δ) is a fraction of the source mass
     controlled by an adaptive step scale.
   * Incremental convolution updates avoid a full FFT each step.
   * The step scale is increased on success and decayed otherwise.
   * The climb runs until the remaining wall‑clock budget is exhausted,
     occasionally restarting from a slight perturbation of the current
     best solution.
5. Return the best sequence (as a plain list of floats).

The procedure respects the 1000 s budget, prints progress,
and works solely with NumPy – no file or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector with sum = 1.
    This matches `evaluate_sequence` when the sum constraint holds.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic seed (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    # log‐product avoids overflow for extreme i
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Clip possible infinities
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a dense α‑grid and keep the vector with the smallest exact score.
    """
    best_a = None
    best_score = float("inf")
    # a slightly richer grid than before
    alphas = np.concatenate([
        np.linspace(-2.0, -0.5, 13),   # centre‑heavy (negative α)
        np.linspace(-0.5, 0.0, 11),    # transition
        np.linspace(0.0, 3.0, 31)      # edge‑heavy
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Guided random walk with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_directed(start_a: np.ndarray,
                        deadline: float,
                        step_start: float = 0.25,
                        inc_factor: float = 1.02,
                        decay_factor: float = 0.97,
                        max_step: float = 0.5,
                        min_step: float = 1e-4,
                        recompute_interval: int = 100_000,
                        no_improve_limit: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that always pulls mass from the index
    contributing most to the current maximal convolution entry.
    Only strictly improving moves are accepted; the step size is
    adapted on the fly.  The function returns the best vector seen
    and its exact score.
    """
    a = start_a.copy()
    n = a.shape[0]

    # initial convolution and scores
    conv = convolution_fft(a, a)
    cur_sc = 2.0 * n * float(conv.max())
    best_a = a.copy()
    best_sc = cur_sc

    step_scale = step_start
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # index of the maximal convolution entry
        k_max = int(np.argmax(conv))

        # indices that contribute to this entry
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        j = k_max - idx
        contrib = a[idx] * a[j]

        # pick the strongest contributor
        src_candidate = idx[np.argmax(contrib)]
        # choose the side of the pair that actually has the larger weight
        src = src_candidate if a[src_candidate] >= a[k_max - src_candidate] else (k_max - src_candidate)

        if a[src] <= 0.0:
            # extremely unlikely, but fall back to a random source
            src = np.random.randint(n)
            if a[src] <= 0.0:
                continue

        # destination: a few random low‑mass candidates
        cand = np.random.choice(n, size=5, replace=False)
        cand = cand[cand != src]
        if cand.size == 0:
            continue
        dst = cand[np.argmin(a[cand])]

        # propose a transfer (δ ≤ step_scale·a[src])
        delta = step_scale * a[src] * np.random.rand()
        if delta <= 0.0 or delta > a[src]:
            continue

        # ------ incremental convolution update (exact) ------
        conv_cand = conv.copy()
        # linear part: –2·δ·shift_src + 2·δ·shift_dst
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic correction
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_sc = 2.0 * n * float(cand_max)

        # ----- accept only improvements -----
        if cand_sc < cur_sc - 1e-15:
            # accept move
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a.copy()
            # adapt step size upward
            step_scale = min(step_scale * inc_factor, max_step)
            no_improve = 0
        else:
            # reject – shrink step size
            step_scale = max(step_scale * decay_factor, min_step)
            no_improve += 1

        it += 1

        # periodic full recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = convolution_fft(a, a)
            cur_sc = 2.0 * n * float(conv.max())

        # early restart if we have been stagnant for a while
        if no_improve >= no_improve_limit:
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects a 1000 s wall‑clock limit and returns the best
    vector (plain list of floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a workable length.
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global `height_sequence_1`).
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic start (α‑grid).
    # -------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # -------------------------------------------------
    # 2. A few random Dirichlet seeds (sparse distribution).
    # -------------------------------------------------
    for _ in range(4):
        if time.time() >= deadline - 20.0:
            break
        a_rand = np.random.dirichlet(np.ones(n) * 0.5)
        sc_rand = compute_score(a_rand)
        if sc_rand < best_score:
            best_score = sc_rand
            best_seq = a_rand.tolist()
            print(f"[{time.time():.1f}] random start improved best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Guided hill‑climb (majority of remaining budget).
    #    Run several 30 s chunks, restarting with a tiny jitter
    #    when a chunk stagnates.
    # -------------------------------------------------
    while time.time() < deadline - 5.0:
        sub_deadline = min(deadline, time.time() + 30.0)
        # start from the current best, add an infinitesimal perturbation
        a_start = np.array(best_seq, dtype=float)
        jitter = (np.random.rand(n) - 0.5) * 1e-5
        a_start += jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_hc, sc_hc = hill_climb_directed(a_start, sub_deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.tolist()
            print(f"[{time.time():.1f}] hill‑climb chunk improved best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Final official evaluation.
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
