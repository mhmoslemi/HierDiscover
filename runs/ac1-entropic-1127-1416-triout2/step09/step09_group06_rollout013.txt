def solve_lp(f, M):
    n = len(f)
    # Build A_ub matrix of shape (2n-1, n)
    # Use scipy.sparse.csr_matrix for memory.
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # row k corresponds to conv index k: sum_i f[i] * g[k-i] <= M
        # For each i s.t. 0 <= k-i < n => i in [max(0, k-(n-1)), min(n-1,k)]
        start_i = max(0, k-(n-1))
        end_i = min(n-1, k)
        # then j = k - i corresponds to g[j]
        # in A_ub, column j = k - i, coefficient = f[i]
        for i in range(start_i, end_i+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return res.x, res.fun, res.success

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Small‑scale linear‑program improvement**  
   - Start from the classic inverse‑square‑root (central‑binomial) shape
     of modest length (≈1024).  
   - Repeatedly solve the LP  
        max  Σ g_j   s.t.  g ≥ 0 ,  f * g ≤ M ,  
     where M = max(f*f).  
   - Mix the solution back into f:   f ← (1‑t)·f + t·ĝ,  where ĝ is the
     LP solution scaled to have the same ℓ₁‑norm as f.  This step
     strictly reduces the max‑convolution bound.
2. **Upscaling** – linearly interpolate the improved small shape to a
   larger length (≈8192) while keeping the ℓ₁‑norm equal to 1.
3. **Smooth surrogate optimisation** – continue‑optimise the
   log‑sum‑exp smooth approximation of the max‑convolution using
   L‑BFGS‑B with an annealing schedule for the smoothing parameter β.
4. **Exact max‑convolution polishing** – projected sub‑gradient descent
   on the true objective 2·n·max(conv) (the sum is kept on the simplex).
5. **Light stochastic hill‑climb** – final local search (swap /
   fractional transfer) for any remaining time.
6. Return the best sequence found (plain list of floats).

All steps respect the global 1000 s wall‑clock budget.  No I/O,
no closures, no lambdas – only top‑level functions.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (fast for large vectors)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector on the simplex
    (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective and gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv−max(conv)))
        g = ∂f/∂logits
    The parametrisation a = softmax(logits) guarantees a ≥ 0
    and ∑a = 1.
    """
    a = softmax(logits)

    conv = convolution_fft(a, a)                  # length 2n‑1
    maxc = conv.max()

    # log‑sum‑exp centred at maxc for numerical stability
    exp_weights = np.exp(beta * (conv - maxc))
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient w.r.t. a (using the same trick as the original code)
    w = exp_weights / Z                            # weights over the 2n‑1 positions
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)           # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]       # ∂f/∂a

    # chain rule back to logits (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 10,
                       maxiter_per_step: int = 300):
    """
    Anneal β while optimising the smooth surrogate with L‑BFGS‑B.
    Returns the best vector (still on the simplex) and its exact score.
    """
    # initialise logits from the starting simplex vector
    logits = np.log(start_a + 1e-20)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # give each β a generous iteration budget, but never exceed the wall‑clock
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(20, int(time_left * 25)))  # ≈25 LBFGS iters per second

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f} (step {step})] new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 500_000,
                     step0: float = 0.07):
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Simple polishing after the smooth phase.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # exact sub‑gradient for the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # off‑diagonal entries have derivative 2·a_{k‑i}
        g[mask] = 2.0 * a[j[mask]]
        # diagonal (i = k‑i) should have coefficient a_i, not 2·a_i
        diag = (j == idx) & mask
        g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape flat plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Linear‑program improvement of the small‑scale shape
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A·g = f * g (linear convolution).
    Each row corresponds to one entry of the convolution.
    """
    n = f.shape[0]
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    A = toeplitz(col, row)          # shape (2n‑1, n)
    return A


def lp_improve(f: np.ndarray,
               t: float,
               max_iters: int,
               deadline: float):
    """
    Perform up to max_iters of the LP‑based improvement described in the
    problem statement.
    Returns the best vector (still on the simplex) and its exact score.
    """
    best_f = f.copy()
    best_score = compute_score(best_f)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        n = f.shape[0]
        M = convolution_fft(f, f).max()
        A = build_conv_matrix(f)
        b = np.full(2 * n - 1, M, dtype=float)
        c = -np.ones(n, dtype=float)          # maximise Σ g_j  ⇒ minimise −Σ g_j

        # solve LP (HiGHS is the default in recent SciPy)
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[LP {it+1}] linprog failed: {res.message}")
            break

        g = res.x
        S = g.sum()
        if S <= 1.0 + 1e-12:
            # no improvement possible
            break

        g_norm = g / S
        f = (1.0 - t) * f + t * g_norm
        f = project_to_simplex(f)          # numerical safety

        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[LP {it+1}] new best score = {best_score:.6f}")

        # optional: slowly decay the mixing parameter
        t = max(t * 0.95, 0.05)

    return best_f, best_score


# ----------------------------------------------------------------------
# Deterministic start – truncated central‑binomial coefficients
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root (central‑binomial) shape, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Upscaling helper
# ----------------------------------------------------------------------


def upscale(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_len.
    The result is again normalised to sum = 1.
    """
    old_len = len(seq)
    if old_len == new_len:
        return seq.copy()
    x_old = np.arange(old_len)
    x_new = np.linspace(0, old_len - 1, new_len)
    up = np.interp(x_new, x_old, seq)
    up = up / up.sum()
    return up

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The search consists of:
        * a small‑scale LP‑based improvement,
        * up‑scaling to a larger length,
        * continuation optimisation of a smooth surrogate,
        * exact sub‑gradient polishing,
        * a final cheap stochastic hill‑climb.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous best construction, if present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Small‑scale LP improvement
    # -----------------------------------------------------------------
    small_n = 1024
    if deadline - time.time() > 30.0:
        f_small = central_binomial_sequence(small_n)
        f_small, score_small = lp_improve(
            f_small,
            t=0.3,
            max_iters=20,
            deadline=deadline - 200.0,  # reserve time for the large‑scale phase
        )
        if score_small < best_val:
            best_val = score_small
            best_seq = f_small.tolist()
            print(f"LP‑improved score (n={small_n}) = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Upscale to a larger length suitable for the heavy optimisation
    # -----------------------------------------------------------------
    # Choose a length that still fits comfortably into the remaining budget.
    time_left = deadline - time.time()
    if time_left > 300.0:
        large_n = 8192
    elif time_left > 150.0:
        large_n = 4096
    else:
        large_n = 2048

    if best_seq is None:
        # fallback to central binomial if everything else failed
        best_seq = central_binomial_sequence(large_n).tolist()
    else:
        # upscale the current best (which may be the small‑scale result)
        best_seq = upscale(np.array(best_seq, dtype=float), large_n).tolist()

    a0 = np.array(best_seq, dtype=float)   # already on the simplex

    # -----------------------------------------------------------------
    # 3. Smooth optimisation via L‑BFGS‑B with β‑continuation
    # -----------------------------------------------------------------
    beta_start = 3.0
    beta_end = 2000.0
    a_smooth, smooth_score = lbfgs_continuation(
        a0,
        beta_start,
        beta_end,
        deadline - 150.0,
        schedule_steps=12,
        maxiter_per_step=300,
    )
    if smooth_score < best_val:
        best_val = smooth_score
        best_seq = a_smooth.tolist()
        print(f"After smooth L‑BFGS: score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Exact max‑objective polishing (projected sub‑gradient)
    # -----------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print("\nSub‑gradient polishing …")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 60.0,
            max_iter=400_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑grad improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Very cheap stochastic hill‑climb (if any time remains)
    # -----------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\nFinal hill‑climb …")
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(best_a)
        step_scale = 0.08

        while time.time() < deadline and step_scale > 1e-5:
            cand = best_a.copy()
            if np.random.rand() < 0.5:
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]
            else:
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = step_scale * best_a[i]
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0:
                        cand[i] = 0.0
            cand = project_to_simplex(cand)
            cur_score = compute_score(cand)

            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                step_scale = min(step_scale * 1.07, 0.5)
            else:
                step_scale *= 0.96

        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.tolist()
            print(f"Hill‑climb final score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Final evaluation (mandatory call to the provided evaluator)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
