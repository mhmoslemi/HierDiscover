"""
Propose a step‑function (non‑negative simplex) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

using a multistage search:
    1.  Very fast analytic families (including new mixtures) on several
        problem sizes up to 8192.  A tiny local‑parameter refinement
        (grid‑search around the best α/β/γ/λ) is performed.
    2.  Adam optimisation of a smooth approximation of the max–convolution
        (f_β = (1/β)·log Σ exp(β·conv)) with a much larger β‑schedule
        than the reference implementation.
    3.  L‑BFGS refinement on the same smooth objective (beta fixed at a
        high value) – this often pushes the peak a little lower than Adam.
    4.  A cheap projected sub‑gradient descent on the true max‑objective
        (step ∝ 1/√t) for a final “polish”.
    5.  Symmetrisation (a ← (a+a[::-1])/2) and a stochastic hill‑climb
        (simulated‑annealing style) for the last few seconds.
All stages respect a global wall‑clock budget (default 1000 s).  The
function returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# -------------------- basic utilities -------------------- #

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Return uniform if everything is non‑positive.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# -------------------- analytic families -------------------- #

def uniform_family(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow/underflow.
    w[np.isinf(w)] = np.finfo(np.float64).max
    w[w == 0] = np.finfo(np.float64).tiny
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mix_symprod_uniform(n: int, lam: float, alpha: float) -> np.ndarray:
    """Convex mixture of the uniform vector and the symmetric‑product family."""
    uniform = uniform_family(n)
    sym = symmetric_product_family(n, alpha)
    return lam * uniform + (1.0 - lam) * sym


def mix_leftheavy_uniform(n: int, lam: float, beta: float) -> np.ndarray:
    """Mixture of uniform and a left‑heavy power law."""
    uniform = uniform_family(n)
    lh = left_heavy_power_family(n, beta)
    return lam * uniform + (1.0 - lam) * lh


def mix_expU_uniform(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and exponential U‑shape."""
    uniform = uniform_family(n)
    ushape = exp_u_family(n, gamma)
    return lam * uniform + (1.0 - lam) * ushape

def random_sparse_start(n: int) -> np.ndarray:

def random_monotone_start(n: int) -> np.ndarray:

    r.sort()
    a = r[::-1]                    # decreasing

def random_u_start(n: int) -> np.ndarray:

# -------------------- Adam optimisation (smooth max) -------------------- #

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 2.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """Adam on f_β with an exponential β‑schedule."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)            # initialise soft‑max parameters

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- β schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a and its convolution -----
        a = softmax(z)
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        # smoothed max weights
        exp_shift = np.exp(beta * (conv - maxc))
        w = exp_shift / exp_shift.sum()

        # ----- gradient wrt a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 1000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def lbfgs_smooth_optimize(start_a: np.ndarray,
                          deadline: float,
                          beta: float = 800.0,
                          max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """L‑BFGS on the smoothed max–objective f_β (β fixed, high)."""
    try:
        from scipy.optimize import minimize
    except Exception as e:
        print("SciPy not available – skipping L‑BFGS stage.")
        return start_a, compute_score(start_a)

    n = start_a.shape[0]
    z0 = np.log(start_a + 1e-12)  # soft‑max parameters

    def f_and_grad(z):
        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smoothed max value
        exp_shift = np.exp(beta * (conv - maxc))
        sum_exp = exp_shift.sum()
        f = maxc + (1.0 / beta) * np.log(sum_exp)

        w = exp_shift / sum_exp            # weights
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a
        return float(f), grad_z.astype(np.float64)

    opts = {"maxiter": max_iter, "disp": False, "gtol": 1e-12}
    res = minimize(fun=f_and_grad,
                   x0=z0,
                   method="L-BFGS-B",
                   jac=True,
                   options=opts)

    a_opt = softmax(res.x)
    return a_opt, compute_score(a_opt)

    """Projected sub‑gradient descent on the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of active max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# -------------------- Hill‑climb (simulated annealing) -------------------- #

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a tiny temperature schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature proportional to current best (tiny)
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# -------------------- Main driver -------------------- #

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Small safety margin for the harness.
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback to previously discovered sequence if present.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families on several lengths (including new mixtures).
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]   # explore larger n first

    # Parameter grids (coarse → fine later)
    alphas = np.linspace(0.0, 2.0, 81)          # 0 … 2 step 0.025
    betas = np.linspace(0.0, 5.0, 81)           # 0 … 5 step 0.0625
    gammas = np.linspace(0.0, 6.0, 61)          # 0 … 6 step 0.1
    lams = np.linspace(0.0, 1.0, 11)            # 0 … 1 step 0.1

    for n in lengths:
        if time.time() >= deadline - 250.0:
            break
        print(f"[{time.time():.1f}] Exploring length n = {n}")

        # ---- symmetric product family (α) ----
        for alpha in alphas:
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym α={alpha:.3f}) = {best_score:.6f}")

        # ---- left‑heavy power law (β) ----
        for beta in betas:
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left β={beta:.3f}) = {best_score:.6f}")

        # ---- exponential U‑shape (γ) ----
        for gamma in gammas:
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp γ={gamma:.3f}) = {best_score:.6f}")

        # ---- mixtures of uniform with the previous families ----
        for lam in lams:
            for alpha in [0.2, 0.5, 0.8, 1.2, 1.5]:
                if time.time() >= deadline - 30.0:
                    break
                vec = mix_symprod_uniform(n, lam, alpha)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mixU‑sym lam={lam:.2f},α={alpha}) = {best_score:.6f}")

            for beta in [0.5, 1.0, 2.0, 3.0, 4.0]:
                if time.time() >= deadline - 30.0:
                    break
                vec = mix_leftheavy_uniform(n, lam, beta)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mixU‑lh lam={lam:.2f},β={beta}) = {best_score:.6f}")

            for gamma in [0.5, 1.5, 3.0, 4.5, 5.5]:
                if time.time() >= deadline - 30.0:
                    break
                vec = mix_expU_uniform(n, lam, gamma)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mixU‑exp lam={lam:.2f},γ={gamma}) = {best_score:.6f}")

        # ---- random draws (monotone / sparse / U‑shapes) ----
        for _ in range(70):
            if time.time() >= deadline - 30.0:
                break
            vec = random_monotone_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

        for _ in range(70):
            if time.time() >= deadline - 30.0:
                break
            vec = random_sparse_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑sparse) = {best_score:.6f}")

        for _ in range(70):
            if time.time() >= deadline - 30.0:
                break
            vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑U) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Local refinement of α / β / γ around the current best (if symmetric‑type)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        a_arr = np.array(best_seq, dtype=np.float64)
        n_best = a_arr.shape[0]

        # Detect whether the shape resembles a symmetric product (rough test)
        # If the vector is roughly symmetric we try a fine α‑search.
        if np.max(np.abs(a_arr - a_arr[::-1])) < 0.02:
            # fine‑grained α search around the best α for this n
            # We don't know the α that generated it, so we just sweep a small interval.
            fine_alphas = np.linspace(0.0, 3.0, 121)   # step 0.025
            for alpha in fine_alphas:
                if time.time() >= deadline - 30.0:
                    break
                vec = symmetric_product_family(n_best, alpha)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] refined best (sym α={alpha:.3f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam smooth‑max refinement (more aggressive β‑schedule)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] Starting Adam smooth optimisation")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 60.0,
            max_iters=300_000,
            beta_start=2.0,
            beta_end=2000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. L‑BFGS refinement on the same smooth objective (high β)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] Starting L‑BFGS refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = lbfgs_smooth_optimize(
            a_start,
            deadline - 30.0,
            beta=800.0,
            max_iter=2000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent on the true max‑objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 45.0:
        print(f"[{time.time():.1f}] Starting sub‑gradient polish")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            max_iters=3_000_000,
            step0=0.08,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (many optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final hill‑climb (simulated annealing) for the remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Fallback to a modest uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list (the harness expects this).
    return best_seq
