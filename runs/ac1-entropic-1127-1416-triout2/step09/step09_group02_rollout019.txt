def evaluate_sequence(sequence):
    # protect negatives: max(0, x)
    # clamp to [0, 1000]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return inf
    return 2 * n * max_b / (sum_a**2)

"""
Smooth‑max gradient descent for the step‑function minimisation problem.

The objective to minimise is

    score(a) = 2·n·max_k (a * a)[k]          (a_i ≥ 0, Σ a_i = 1)

Because the max is non‑smooth we approximate it with a
soft‑max (log‑sum‑exp) of the convolution values

    φ_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])  → max_k (a*a)[k]  as β→∞ .

The gradient of φ_β is a weighted sum of the gradients of the
individual convolution entries; each entry k contributes the shifted
vector a_{k‑i}.  Using the soft‑max weights w_k = e^{β·c_k} / Σ e^{β·c_j}
the gradient is simply

    ∇φ_β(a) = 2·(w ⋆ a)                (⋆ = cross‑correlation)

Both the self‑convolution and the cross‑correlation are evaluated with
FFT ( O(n log n) ), making an optimisation with dozens of thousands of
iterations feasible for n≈2000.

Algorithm
---------

1. Generate a handful of sensible seeds (uniform, sine‑squared,
   beta‑family with negative exponents, arcsine‑type, a few sparse
   Dirichlet draws).  Keep the best seed according to the exact score.

2. Starting from that seed run a series of gradient‑descent phases.
   In each phase the smoothing parameter β is increased (1 → 5 → 10 …
   up to 200) and a fixed number of iterations is performed.
   After each iteration the iterate is projected onto the probability
   simplex (Euclidean projection) to respect the non‑negativity and
   normalisation constraints.

3. Occasionally (every 200 iterations) the exact score is evaluated and
   the best vector so far is stored.

4. If time permits a short stochastic hill‑climb (local mass transfer)
   is applied as a final polish.

The whole search respects the supplied time budget; the function returns
the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a) for non‑negative lags 0 … len(a)-1.
    Result length = len(a).  Implemented via FFT:
        (w ⋆ a)[i] = Σ_j w[i+j]·a[j] .
    """
    n = a.shape[0]
    m = w.shape[0]
    L = next_pow2(m + n - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(W * np.conj(A), n=L)
    start = n - 1
    return corr_full[start:start + n]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (Σ a_i = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Seed generators

def generate_uniform(n: int) -> np.ndarray:
    """Uniform distribution (all entries equal)."""
    a = np.full(n, 1.0 / n)
    return a


def generate_sine_squared(n: int) -> np.ndarray:
    """a_i = sin^2(π·(i+1)/(n+1)), normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p .
    p can be negative (mass at edges) or positive (mass at centre).
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_arcsine(n: int) -> np.ndarray:
    """
    Arcsine‑type density  a_i ∝ 1 / sqrt( (i+0.5)*(n-i-0.5) ).
    This puts a lot of mass near the two ends while remaining integrable.
    """
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Sample a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(
    a0: np.ndarray,
    deadline: float,
    beta_schedule: list[float],
    iters_per_beta: int = 800,
    step0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Minimises the exact objective by descending the smoothed
    max–convolution (log‑sum‑exp) using the gradient
        ∇φ_β(a) = 2·(w ⋆ a) ,   w = softmax(β·(a*a)) .
    After each update the iterate is projected onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    step = step0
    for beta in beta_schedule:
        for it in range(iters_per_beta):
            if time.time() >= deadline:
                return best_a, best_score

            # self‑convolution
            conv = convolution_fft(a, a)

            # soft‑max weights (numerically stable)
            mx = conv.max()
            exp_w = np.exp(beta * (conv - mx))
            w = exp_w / exp_w.sum()

            # gradient = 2·(w ⋆ a)
            g = 2.0 * correlation_fft(w, a)

            # gradient step and projection
            a -= step * g
            a = project_to_simplex(a)

            # slowly decay the step size
            step *= 0.9999

            # occasional exact evaluation
            if it % 200 == 0:
                cur = compute_score(a)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()

        # after each β increase we give the step size a tiny boost
        step = max(step, step0 * 0.5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb polishing (only accept improvements)
# ----------------------------------------------------------------------


def hill_climb(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local‑search that only keeps moves which improve the
    exact score.  Uses tiny mass transfers between random positions.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    step_scale = 0.08
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = cur_a.copy()
        i, j = np.random.randint(n, size=2)
        if cand[i] > 0.0 and i != j:
            delta = step_scale * cand[i]
            cand[i] -= delta
            cand[j] += delta

        # re‑normalise (numerical drift)
        cand /= cand.sum()
        new_score = compute_score(cand)

        if new_score < best_score:
            best_score = new_score
            best_a = cand
            cur_a = cand
            cur_score = new_score
            step_scale = min(step_scale * 1.07, 0.5)   # go bolder
        else:
            step_scale *= 0.96                       # become more conservative

        steps += 1

    return best_a, best_score

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the given time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # 0. Choose a manageable length (2048 works well within the budget)
    # ------------------------------------------------------------------
    n = 2048
    total_time = deadline - start_time
    if total_time > 800.0:
        n = 4096
    print(f"[Info] Sequence length = {n}")

    # ------------------------------------------------------------------
    # 1. Generate a set of deterministic / random seeds
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(generate_uniform(n))
    seeds.append(generate_sine_squared(n))
    # a few beta‑family members with strong edge bias
    for p in [-3.0, -2.0, -1.5, -1.2, -1.0, -0.7, -0.5]:
        seeds.append(generate_beta_family(n, p))

    seeds.append(generate_arcsine(n))

    # sparse Dirichlet draws (concentration < 1) – good for exploration
    for _ in range(4):
        seeds.append(random_dirichlet(n, conc=0.2))
        seeds.append(random_dirichlet(n, conc=0.5))

    # ------------------------------------------------------------------
    # 2. Keep the best seed according to the exact objective
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for s in seeds:
        s = s.astype(float)
        s = project_to_simplex(s)
        sc = compute_score(s)
        if sc < best_score:
            best_score = sc
            best_a = s.copy()
            print(f"[Seed] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑max gradient descent (main optimisation phase)
    # ------------------------------------------------------------------
    print("\n[Phase 1] Smoothed‑max gradient descent")
    beta_sched = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 200.0]
    iters_per_beta = max(200, int(2000 * (total_time / 1000.0)))
    cur_a, cur_score = smooth_max_descent(
        best_a, deadline, beta_sched, iters_per_beta=iters_per_beta, step0=0.5
    )
    best_score = cur_score
    print(f"After gradient descent: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional hill‑climb polishing (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 2] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=300_000)
        if cur_score < best_score:
            best_score = cur_score
            print(f"After polishing: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final conversion to list and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq
