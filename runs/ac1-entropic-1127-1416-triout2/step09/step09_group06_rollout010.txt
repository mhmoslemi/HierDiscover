# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns a length‑(len(a)+len(b)-1) array.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector already on the simplex
    (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))
    and g = ∇_logits f.
    The parametrisation a = softmax(logits) guarantees a≥0 and Σa=1.
    """
    a = softmax_logits(logits)

    conv = convolution_fft(a, a)          # length 2n‑1
    maxc = conv.max()

    # centred log‑sum‑exp
    exp_weights = np.exp(beta * (conv - maxc))
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # gradient w.r.t. a
    w = exp_weights / Z                     # same length as conv
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)    # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # chain rule back to logits (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 12,
                       maxiter_per_step: int = 200):
    """
    Anneal the smoothness parameter β while optimising the surrogate with L‑BFGS‑B.
    Returns the best sequence (still on the simplex) and its exact score.
    """
    logits = np.log(start_a + 1e-20)          # safe initial logits
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # allocate a sensible number of L‑BFGS iterations
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(20, int(time_left * 25)))   # ≈25 iters / sec

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.1f} step {step}/{schedule_steps}] new best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 500_000,
                     step0: float = 0.07):
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Simple but effective polishing after the smooth phase.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to avoid plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Deterministic starting families
# ----------------------------------------------------------------------


def central_binomial_sequence(n: int) -> np.ndarray:
    """Inverse‑square‑root (central‑binomial) shape, normalised to the simplex."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    a /= a.sum()
    return a


def power_law_sequence(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-α}, normalised to sum‑one."""
    idx = np.arange(n, dtype=float) + 1.0
    a = idx ** (-alpha)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The algorithm consists of:
        • a quick power‑law α‑grid search,
        • β‑annealed L‑BFGS refinement,
        • projected sub‑gradient polishing,
        • a light stochastic hill‑climb.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the surrounding harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous best construction, if available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose sequence length based on remaining time
    # -----------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 800:
        n = 32768
    elif time_left > 500:
        n = 16384
    elif time_left > 250:
        n = 8192
    else:
        n = 4096
    print(f"Selected length n = {n}")

    # -----------------------------------------------------------------
    # 2. Light α‑grid search on a modest n (4096) to pick a good family
    # -----------------------------------------------------------------
    n_grid = min(4096, n)
    alphas = np.linspace(0.45, 0.75, 13)   # 0.45 … 0.75 in steps of 0.025
    best_alpha = None
    best_alpha_score = float("inf")
    for alpha in alphas:
        seq = power_law_sequence(n_grid, alpha)
        sc = compute_score(seq)
        if sc < best_alpha_score:
            best_alpha_score = sc
            best_alpha = alpha
    print(f"Power‑law grid: best α = {best_alpha:.4f}, approx score = {best_alpha_score:.6f}")

    # also compare the central‑binomial shape on the same grid size
    seq_cb = central_binomial_sequence(n_grid)
    sc_cb = compute_score(seq_cb)
    if sc_cb < best_alpha_score:
        print(f"Central‑binomial beats grid shape (score {sc_cb:.6f})")
        best_alpha = None               # signal to use central‑binomial
        best_alpha_score = sc_cb

    # -----------------------------------------------------------------
    # 3. Initialise full‑size starting sequence
    # -----------------------------------------------------------------
    if best_alpha is None:
        a0 = central_binomial_sequence(n)
    else:
        a0 = power_law_sequence(n, best_alpha)
    a0 /= a0.sum()          # ensure exact simplex (tiny numerical drift)
    init_score = compute_score(a0)
    print(f"Initial sequence score = {init_score:.6f}")

    if init_score < best_val:
        best_val = init_score
        best_seq = a0.tolist()
        print("Initial sequence becomes the current best.")

    # -----------------------------------------------------------------
    # 4. Smooth optimisation via β‑annealed L‑BFGS‑B
    # -----------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 300:
        # reserve ~150 s for subsequent stages
        smooth_deadline = time.time() + time_left - 150.0
        beta_start = 5.0
        beta_end = 2000.0
        a_smooth, smooth_score = lbfgs_continuation(
            a0,
            beta_start,
            beta_end,
            smooth_deadline,
            schedule_steps=12,
            maxiter_per_step=200,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"After L‑BFGS refinement: score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Exact max‑objective polishing (projected sub‑gradient)
    # -----------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 120:
        subgrad_deadline = time.time() + time_left - 30.0
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            subgrad_deadline,
            max_iter=400_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑gradient polishing improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Very light stochastic hill‑climb (if any time remains)
    # -----------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(best_a)
        step_scale = 0.08

        while time.time() < deadline and step_scale > 1e-5:
            cand = best_a.copy()
            if np.random.rand() < 0.5:          # swap two entries
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]
            else:                                # fractional transfer
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = step_scale * best_a[i]
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0.0:
                        cand[i] = 0.0
            cand = project_to_simplex(cand)
            cur_score = compute_score(cand)
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                step_scale = min(step_scale * 1.07, 0.5)
            else:
                step_scale *= 0.96

        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.tolist()
            print(f"Hill‑climb final score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 7. Fallback in the extremely unlikely case that nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
