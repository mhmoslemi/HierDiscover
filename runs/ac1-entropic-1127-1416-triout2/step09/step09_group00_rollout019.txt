"""
Propose a step‑function sequence that minimises the objective
     2·n·max(convolve(a,a)) / (sum(a)**2)

The algorithm consists of three phases:
1. Coarse analytic search – several families (symmetric product,
   left‑heavy power law, exponential “U”, mixtures with the uniform
   vector and a symmetric Beta‑shape) are swept over a set of lengths.
   The best few candidates are kept.

2. Deep smooth optimisation – the kept candidates are refined with a
   Adam optimiser on the smooth log‑sum‑exp approximation of the
   max‑convolution.  The smoothing parameter β is increased from 4 to
   2000 and a large number of iterations is allowed.

3. Sub‑gradient Adam refinement – the smoothed optimum is further
   polished with Adam on the exact sub‑gradient of the max‑convolution.
   Finally a short simulated‑annealing hill‑climb is performed.

All phases respect the global time budget (default 1000 s).  The
function returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a=1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return vectors that sum to 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    # use log to avoid overflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    logw = -alpha * logprod
    logw -= np.max(logw)      # shift for stability
    w = np.exp(logw)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logw = -beta * np.log(i + 0.5)
    logw -= np.max(logw)
    w = np.exp(logw)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logw = -gamma * dist
    logw -= np.max(logw)
    w = np.exp(logw)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Convex mixture of the uniform vector and the U‑shape.
    lam∈[0,1] is the weight of the uniform component.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    a /= a.sum()
    return a


def beta_shape_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta‑shape: a_i ∝ x^{α‑1} (1‑x)^{α‑1},
    where x = (i+½)/n ∈ (0,1).  For α<1 the distribution is U‑shaped.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    logx = np.log(np.clip(x, eps, 1 - eps))
    log1mx = np.log(np.clip(1 - x, eps, 1 - eps))
    logw = (alpha - 1.0) * (logx + log1mx)
    logw -= np.max(logw)
    w = np.exp(logw)
    a = w / w.sum()
    return a


def mixed_uniform_beta_family(n: int, lam: float, alpha: float) -> np.ndarray:
    """
    Convex mixture of uniform and a symmetric Beta‑shape.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    beta = beta_shape_family(n, alpha)
    a = lam * uniform + (1.0 - lam) * beta
    a /= a.sum()
    return a

# Adam on smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 5.0,
                         beta_end: float = 400.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing β is increased exponentially from beta_start to beta_end.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam smooth improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def adam_subgrad_optimize(start_a: np.ndarray,

    At each step the sub‑gradient of the active shift(s) is used.
    """
    n = start_a.shape[0]
    eps = 1e-12
    u = np.log(start_a + eps)        # parameters before soft‑max

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(u)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = softmax(u)
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # active set (within relative tolerance)
        tol = maxc * 1e-12
        if tol < 1e-15:
            tol = 1e-15
        active = conv >= (maxc - tol)
        # build weight vector – uniform over active indices
        w = np.zeros_like(conv)
        cnt = np.count_nonzero(active)
        if cnt > 0:
            w[active] = 1.0 / cnt
        else:
            w[np.argmax(conv)] = 1.0

        # sub‑gradient w.r.t. a
        grad_a = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_u = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(u)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam sub‑grad improved to {best_score:.6f}")

    a_fin = softmax(u)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Long random walk on the simplex.
    – 0.5 probability: swap two entries.
    – 0.5 probability: transfer a fraction of mass from one index to another.
    A tiny temperature schedule permits occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper – up‑sample a vector to a larger length via linear interpolation
# ----------------------------------------------------------------------


def upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Interpolate a probability vector to a new length.
    The result is re‑normalised to sum to 1.
    """
    old_n = seq.size
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new, 0.0)
    seq_new /= seq_new.sum()
    return seq_new

"""
Propose a step‑function sequence that minimises the objective
     2·n·max(convolve(a,a)) / (sum(a)**2)

The algorithm consists of three phases:
1. Coarse analytic search – several families (symmetric product,
   left‑heavy power law, exponential “U”, mixtures with the uniform
   vector and a symmetric Beta‑shape) are swept over a set of lengths.
   The best few candidates are kept.

2. Deep smooth optimisation – the kept candidates are refined with a
   Adam optimiser on the smooth log‑sum‑exp approximation of the
   max‑convolution.  The smoothing parameter β is increased from 4 to
   2000 and a large number of iterations is allowed.

3. Sub‑gradient Adam refinement – the smoothed optimum is further
   polished with Adam on the exact sub‑gradient of the max‑convolution.
   Finally a short simulated‑annealing hill‑climb is performed.

All phases respect the global time budget (default 1000 s).  The
function returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a=1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return vectors that sum to 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    logw = -alpha * logprod
    logw -= np.max(logw)      # stabilise
    w = np.exp(logw)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logw = -beta * np.log(i + 0.5)
    logw -= np.max(logw)
    w = np.exp(logw)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logw = -gamma * dist
    logw -= np.max(logw)
    w = np.exp(logw)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Convex mixture of the uniform vector and the U‑shape.
    lam∈[0,1] is the weight of the uniform component.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    a /= a.sum()
    return a


def beta_shape_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta‑shape: a_i ∝ x^{α‑1} (1‑x)^{α‑1},
    where x = (i+½)/n.
    For α<1 the distribution is U‑shaped.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    logx = np.log(np.clip(x, eps, 1 - eps))
    log1mx = np.log(np.clip(1 - x, eps, 1 - eps))
    logw = (alpha - 1.0) * (logx + log1mx)
    logw -= np.max(logw)
    w = np.exp(logw)
    a = w / w.sum()
    return a


def mixed_uniform_beta_family(n: int, lam: float, alpha: float) -> np.ndarray:
    """
    Convex mixture of uniform and a symmetric Beta‑shape.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    beta = beta_shape_family(n, alpha)
    a = lam * uniform + (1.0 - lam) * beta
    a /= a.sum()
    return a

def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Sample a random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


def random_beta_start(n: int) -> np.ndarray:
    """Sample a random symmetric Beta shape with α∈(0.05,0.95)."""
    alpha = np.random.uniform(0.05, 0.95)
    return beta_shape_family(n, alpha)


# ----------------------------------------------------------------------
# Adam on smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 5.0,
                         beta_end: float = 400.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing β is increased exponentially from beta_start to beta_end.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam smooth improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam on exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def adam_subgrad_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Adam optimiser directly on the true max‑convolution.
    At each step the sub‑gradient of the active shift(s) is used.
    """
    n = start_a.shape[0]
    eps = 1e-12
    u = np.log(start_a + eps)        # parameters before soft‑max

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(u)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = softmax(u)
        conv = convolution_fft(a, a)
        maxc = conv.max()

        tol = maxc * 1e-12
        if tol < 1e-15:
            tol = 1e-15
        active = conv >= (maxc - tol)
        w = np.zeros_like(conv)
        cnt = np.count_nonzero(active)
        if cnt > 0:
            w[active] = 1.0 / cnt
        else:
            w[np.argmax(conv)] = 1.0

        grad_a = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_u = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(u)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam sub‑grad improved to {best_score:.6f}")

    a_fin = softmax(u)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex.
    – 0.5 probability: swap two entries.
    – 0.5 probability: transfer a fraction of mass from one index to another.
    A tiny temperature schedule permits occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper – up‑sample a vector to a larger length via linear interpolation
# ----------------------------------------------------------------------


def upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Interpolate a probability vector to a new length.
    Result is re‑normalised to sum to 1.
    """
    old_n = seq.size
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new, 0.0)
    seq_new /= seq_new.sum()
    return seq_new


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the global time budget and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. fallback (if a previous candidate is available)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families + random draws (coarse)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline - 200.0:
            break
        print(f"[{time.time():.1f}] exploring length n = {n}")

        # ----- symmetric product (α) -----
        for alpha in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        # ----- left‑heavy power law (β) -----
        for beta in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

        # ----- exponential U‑shape (γ) -----
        for gamma in np.linspace(0.0, 6.0, 31):
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

        # ----- mixtures (λ,γ) -----
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for gamma in np.linspace(0.0, 6.0, 13):
                if time.time() >= deadline - 30.0:
                    break
                vec = mixed_uniform_u_family(n, lam, gamma)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

        # ----- Beta shape family (α) -----
        for alpha in np.linspace(0.05, 0.95, 19):
            if time.time() >= deadline - 30.0:
                break
            vec = beta_shape_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (beta‑α={alpha:.2f}) = {best_score:.6f}")

        # ----- mixture uniform + Beta -----
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for alpha in np.linspace(0.05, 0.95, 19):
                if time.time() >= deadline - 30.0:
                    break
                vec = mixed_uniform_beta_family(n, lam, alpha)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mix‑β λ={lam:.2f},α={alpha:.2f}) = {best_score:.6f}")

        # ----- random monotone draws -----
        for _ in range(80):
            if time.time() >= deadline - 30.0:
                break
            vec = random_monotone_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

        # ----- random U‑shape draws -----
        for _ in range(80):
            if time.time() >= deadline - 30.0:
                break
            vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑U) = {best_score:.6f}")

        # ----- random Beta draws -----
        for _ in range(80):
            if time.time() >= deadline - 30.0:
                break
            vec = random_beta_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑Beta) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Deep smooth optimisation (Adam on log‑sum‑exp)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting deep Adam‑smooth refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 80.0,
            max_iters=400_000,
            beta_start=4.0,
            beta_end=2000.0,
            lr=0.015,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after deep smooth best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam on exact sub‑gradient
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting Adam‑sub‑grad refinement")
        a_opt2, sc_opt2 = adam_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=300_000,
            lr=0.015,
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after Adam sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional up‑sampling to a larger length (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        cur_n = len(best_seq)
        new_n = next_pow2(cur_n * 2)
        if new_n <= 16384:
            print(f"[{time.time():.1f}] up‑sampling from {cur_n} to {new_n}")
            a_up = upscale(np.array(best_seq, dtype=np.float64), new_n)
            a_up_opt, sc_up_opt = adam_smooth_optimize(
                a_up,
                deadline - 10.0,
                max_iters=150_000,
                beta_start=5.0,
                beta_end=800.0,
                lr=0.012,
            )
            if sc_up_opt < best_score:
                best_score = sc_up_opt
                best_seq = a_up_opt.tolist()
                print(f"[{time.time():.1f}] after up‑sampling best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation test (often beneficial)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
