"""
search algorithm
----------------
1.  Create a diverse pool of candidate step‑functions (deterministic
    shapes, Dirichlet samples, a Gaussian bump, the historic fallback).

2.  Pick the best of those candidates according to the exact objective
    `2·n·max(conv(a,a))` (i.e. the score returned by
    ``evaluate_sequence`` for a normalised sequence).

3.  Run a *Frank‑Wolfe style* refinement:
    – for the current vector ``f`` compute the maximal auto‑correlation
      ``M = max(f*f)``.
    – solve a linear programme          maximise   Σ b_j
                                         subject to   f * b ≤ M ,   b ≥ 0
      (the convolution constraints are linear in the unknown ``b``).
      The solution ``b`` is then normalised to a probability vector ``g``.
    – replace ``f`` by a convex combination
             ``f ← (1‑t)·f + t·g``   (t starts large and is slowly reduced).
    – repeat until the bound stops improving or the time budget is
      exhausted.  This step drives the maximal convolution down
      aggressively and is the main improvement over the previous
      implementation.

4.  A short stochastic hill‑climb (swap / local mass transfer) that
    only accepts strictly improving moves finishes the search.

All steps respect the supplied wall‑clock budget (default 1000 s) and
return a plain list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def generate_sine_squared(n: int) -> np.ndarray:
    """a_i = sin²(π·(i+1)/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)·(n-i))^p , symmetric, normalised."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    """
    Discrete centred Gaussian, σ = sigma_factor·n,
    then normalised to sum 1.
    """
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a

def simulated_annealing(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    phase_time: float,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk on the simplex.
    Returns the best vector encountered and its exact score.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = max(1e-9, cur_score * 0.04)
    step_initial = 0.35
    min_step = 5e-6
    smooth_kernel = np.ones(5) / 5.0

    it = 0
    while time.time() < deadline - 2.0:
        it += 1
        elapsed = time.time() - (deadline - phase_time)
        frac = elapsed / phase_time
        T = max(1e-12, T0 * (1.0 - frac))
        step_scale = max(min_step, step_initial * (1.0 - frac))

        r = np.random.rand()
        if r < 0.65:
            # local transfer
            i = np.random.randint(n)
            if cur_a[i] <= 0.0:
                continue
            j = (i + 1) % n if np.random.rand() < 0.5 else (i - 1) % n
            if i == j:
                continue
            delta = step_scale * np.random.rand() * cur_a[i]
        elif r < 0.85:
            # global transfer
            i = np.random.randint(n)
            if cur_a[i] <= 0.0:
                continue
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = step_scale * np.random.rand() * cur_a[i]
        else:
            # smoothing move
            a_smooth = np.convolve(cur_a, smooth_kernel, mode="same")
            s = a_smooth.sum()
            if s == 0.0:
                continue
            a_smooth /= s
            new_score = compute_score(a_smooth)
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T):
                cur_a[:] = a_smooth
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = cur_a.copy()
                continue
            else:
                continue

        # apply the transfer
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = compute_score(cur_a)

        # acceptance test
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            cur_a[i] += delta
            cur_a[j] -= delta

        if it % 5000 == 0:
            print(f"[SA] elapsed {elapsed:.1f}s, best {best_score:.6f}")

    return best_a, best_score

# Linear‑programming refinement (core improvement)
# ----------------------------------------------------------------------


def lp_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 30,
    t_initial: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe‑style improvement:
    maximise Σb subject to (a * b) ≤ M,
    then mix the obtained normalised vector into ``a``.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    t = t_initial
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # 1) compute the current bound M
        conv = convolution_fft(cur_a, cur_a)
        M = conv.max()

        # 2) linear program
        b_var = cp.Variable(n, nonneg=True)
        a_param = cp.Parameter(n, nonneg=True, value=cur_a)
        conv_expr = cp.conv(a_param, b_var)               # length 2n‑1
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [conv_expr <= M])
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)

        if b_var.value is None:
            # solver failed – abort early
            print("[LP] solver failed, terminating refinement.")
            break

        b = np.asarray(b_var.value).flatten()
        sum_b = b.sum()
        if sum_b <= 1e-12:
            print("[LP] LP returned zero mass, aborting.")
            break

        g = b / sum_b                     # normalise to sum 1
        # 3) convex combination
        cur_a = (1.0 - t) * cur_a + t * g
        cur_a = project_to_simplex(cur_a)  # numerical safety
        cur_score = compute_score(cur_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
            print(f"[LP] iteration {it:2d}, improved score = {best_score:.6f}")

        # slowly shrink the mixing step to aid convergence
        t *= 0.92
        if t < 0.02:
            t = 0.02

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb polishing (greedy improvement only)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Small stochastic moves (swap or local mass transfer) that are kept
    only when they improve the exact score.  Acts as a final polish.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the given time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # a) optional fallback (historical best)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = np.array(fallback, dtype=float)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # b) choose a manageable length
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 800.0:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # c) generate a portfolio of seed vectors
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(generate_sine_squared(n))

    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    for sigma in (0.12, 0.20, 0.30):
        seeds.append(generate_gaussian(n, sigma))

    for conc in (0.3, 0.7, 1.0, 2.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc))

    if best_seq is not None and len(best_seq) == n:
        seeds.append(best_seq / best_seq.sum())

    # ------------------------------------------------------------------
    # d) pick the best initial seed (exact score)
    # ------------------------------------------------------------------
    for s in seeds:
        s = project_to_simplex(s.astype(float))
        sc = compute_score(s)
        if sc < best_val:
            best_val = sc
            best_seq = s.copy()
            print(f"Initial seed improvement: score = {best_val:.6f}")

    cur_a = best_seq.copy()
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------------------------------------------------------
    # Phase 1 – Simulated annealing (global exploration)
    # ------------------------------------------------------------------
    print("\n[Phase 1] Simulated annealing")
    phase1_time = total_time * 0.30
    phase1_deadline = start_time + phase1_time
    cur_a, cur_score = simulated_annealing(
        cur_a, cur_score, phase1_deadline, phase1_time
    )
    if cur_score < best_val:
        best_val = cur_score
    print(f"After SA: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑programming refinement (core improvement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n[Phase 2] Linear‑programming refinement")
        phase2_deadline = deadline - 30.0
        cur_a, cur_score = lp_refine(
            cur_a, cur_score, phase2_deadline, max_iter=30, t_initial=0.7
        )
        if cur_score < best_val:
            best_val = cur_score
        print(f"After LP refinement: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 3] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=1_200_000)
        if cur_score < best_val:
            best_val = cur_score
        print(f"After polishing: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Return the candidate in the form required by the harness
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq
