"""
Propose a step‑function sequence that minimizes the evaluation metric.

The algorithm works in several stages within the 1000 s budget:

1. **Random low‑dimensional shape search** –  
   The sequence is modelled as a simple parametric shape  
   a_i ∝ (i+α)^p·exp(-λ·i)  (α>0, p<0, λ≥0).  
   A few thousand random draws of (p,α,λ) are tested on a modest
   length (≈1024).  The best parameters give a good starting shape.

2. **Upscaling** –  
   Using the obtained parameters a longer sequence (≈8192‑16384) is built
   by the same formula and normalised to sum = 1.

3. **Smooth surrogate optimisation** –  
   An L‑BFGS‑B continuation on a log‑sum‑exp smooth approximation of the
   max‑convolution (soft‑max re‑parameterisation) refines the shape.

4. **Exact max‑objective polishing** –  
   Projected sub‑gradient descent on 2·n·max(conv) further reduces the
   score.

5. **Light stochastic hill‑climb** –  
   Simple swaps / fractional mass transfers with projection to the simplex
   finish the search.

All intermediate results are kept; the best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT – efficient for large n."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector already on the
    probability simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient (combined)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))
    and g is the gradient of f w.r.t. the soft‑max logits.
    """
    a = softmax_logits(logits)                     # a lies on the simplex

    conv = convolution_fft(a, a)
    maxc = conv.max()

    exp_weights = np.exp(beta * (conv - maxc))     # centred for stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    w = exp_weights / Z                            # soft‑max over conv entries
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)           # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 9,
                       maxiter_per_step: int = 150):
    """
    Anneal the smoothing parameter β while optimising the smooth surrogate
    with L‑BFGS‑B.  Returns the best sequence (still on the simplex) and
    its exact score.
    """
    logits = np.log(start_a + 1e-20)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(20, int(time_left * 30)))  # ≈30 iters / sec

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f} (step {step})] new best score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective polishing – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 400_000,
                     step0: float = 0.07):
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).  Simple
    and robust polishing after the smooth phase.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Simple parametric shape (power‑law + exponential tail)
# ----------------------------------------------------------------------


def shape_params_to_seq(p: float, alpha: float, lam: float, n: int) -> np.ndarray:
    """
    Build a normalised non‑negative sequence of length n:
        a_i ∝ (i + α)^p * exp(-λ·i)
    Parameters:
        p     – power‑law exponent (negative)
        alpha – offset > 0
        lam   – exponential decay (≥0)
    The returned array sums to 1.
    """
    i = np.arange(n, dtype=np.float64)
    # work in log‑space for stability
    log_raw = p * np.log(i + alpha) - lam * i
    log_raw -= np.max(log_raw)                # avoid overflow
    raw = np.exp(log_raw)
    a = raw / raw.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The algorithm combines a low‑dimensional random shape search,
    up‑scaling, L‑BFGS continuation on a smooth surrogate,
    exact sub‑gradient polishing and a final stochastic hill‑climb.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for harness

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback to a previously found construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Random low‑dimensional shape search (tiny n for speed)
    # -----------------------------------------------------------------
    time_now = time.time()
    if deadline - time_now > 200:
        n_search = 1024
        search_deadline = time_now + min(120.0, (deadline - time_now) * 0.5)
        best_params = None
        best_score = float("inf")
        iter_cnt = 0
        while time.time() < search_deadline:
            p = np.random.uniform(-1.5, -0.05)        # power exponent
            alpha = np.random.uniform(0.1, 3.0)       # offset
            lam = np.random.uniform(0.0, 0.02)       # exponential decay
            a = shape_params_to_seq(p, alpha, lam, n_search)
            score = compute_score(a)
            iter_cnt += 1
            if score < best_score:
                best_score = score
                best_params = (p, alpha, lam)
                print(f"[shape search] iter={iter_cnt}, new best score={best_score:.6f}")
        if best_params is not None:
            print(f"Best shape parameters after search: p={best_params[0]:.4f}, "
                  f"α={best_params[1]:.4f}, λ={best_params[2]:.5f}")
        else:
            # fall back to the classic inverse‑square‑root shape
            best_params = (-0.5, 0.5, 0.0)
    else:
        # not enough time – use a reasonable default
        best_params = (-0.5, 0.5, 0.0)

    # -----------------------------------------------------------------
    # 2. Upscale to a bigger problem size
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n_large = 16384
    elif remaining > 300:
        n_large = 8192
    elif remaining > 150:
        n_large = 4096
    else:
        n_large = 2048

    p_opt, alpha_opt, lam_opt = best_params
    a0 = shape_params_to_seq(p_opt, alpha_opt, lam_opt, n_large)
    cur_score = compute_score(a0)
    best_seq = a0.tolist()
    best_val = cur_score
    print(f"Initial up‑scaled shape (n={n_large}) score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Smooth optimisation via L‑BFGS‑B with β‑continuation
    # -----------------------------------------------------------------
    time_now = time.time()
    if deadline - time_now > 200:
        cont_deadline = time_now + min(150.0, (deadline - time_now) * 0.6)
        a_smooth, smooth_score = lbfgs_continuation(
            a0,
            beta_start=5.0,
            beta_end=300.0,
            deadline=cont_deadline,
            schedule_steps=9,
            maxiter_per_step=150,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.tolist()
            print(f"After smooth L‑BFGS: score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Exact max‑objective polishing (projected sub‑gradient)
    # -----------------------------------------------------------------
    time_now = time.time()
    if deadline - time_now > 150:
        polish_deadline = time_now + min(120.0, (deadline - time_now) * 0.8)
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline=polish_deadline,
            max_iter=500_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑grad improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Very cheap stochastic hill‑climb (optional)
    # -----------------------------------------------------------------
    time_now = time.time()
    if deadline - time_now > 10:
        hill_deadline = deadline - 1.0
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(best_a)
        step_scale = 0.08

        while time.time() < hill_deadline and step_scale > 1e-5:
            cand = best_a.copy()
            if np.random.rand() < 0.6:            # simple swap move
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]
            else:                                 # fractional mass transfer
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = step_scale * best_a[i]
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0:
                        cand[i] = 0.0
            cand = project_to_simplex(cand)
            cur_score = compute_score(cand)

            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                step_scale = min(step_scale * 1.07, 0.5)
            else:
                step_scale *= 0.96

        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.tolist()
            print(f"Hill‑climb final score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Final reporting
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
