def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s - 2.0
    n_target = 4096

    # Get base vector
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        # scale to sum=1
        base = base / np.sum(base)
        base = upscale_vector(base, n_target)
    else:
        base = arcsine_initial(n_target)  # sum=1
    # jitter
    rng = np.random.default_rng(seed)
    jitter = rng.normal(scale=1e-4, size=n_target)
    base = np.clip(base + jitter, 0, None)
    base = base / np.sum(base)

    # Adam optimization
    # Determine allowed remaining time for each stage.
    # we allocate roughly half of remaining for Adam, half for subgradient+EG.
    # We'll compute iteration budget based on time.
    # We'll use compute_score for surrogate.

    # stage 1: Adam with smooth max
    remaining = deadline - time.time()
    # set max iters based on remaining perhaps 30 * remaining seconds.
    max_iters_adam = int(min(200_000, remaining * 30))
    # ensure at least 10k loops
    max_iters_adam = max(10_000, max_iters_adam)
    a_adam = adam_optimize_smooth_max(base, deadline, max_iters=max_iters_adam,
                                      tau_start=0.5, tau_end=150.0, base_lr=0.30)

    # stage 2: subgradient descent
    remaining = deadline - time.time()
    max_iters_sg = int(min(250_000, remaining * 30))
    max_iters_sg = max(10_000, max_iters_sg)
    a_sg = subgradient_descent_exact(a_adam, deadline, max_iter=max_iters_sg, step0=0.12)

    # stage 3: exponentiated gradient refinement
    remaining = deadline - time.time()
    max_iters_eg = int(min(200_000, remaining * 30))
    max_iters_eg = max(5_000, max_iters_eg)
    a_final = eg_refine(a_sg, deadline, max_iter=max_iters_eg, eta0=0.2)

    # evaluate final
    try:
        val = evaluate_sequence(a_final.tolist())
        print(f"Final official score = {val}")
    except Exception as e:
        print(f"Evaluation error: {e}")
        val = None

    return a_final.tolist()

"""
Propose a low‑max‑auto‑correlation step function.

The method is a streamlined version of the previous pipeline:

1.  Initialise a candidate from the known fallback (if any) or from the
    classic arcsine (Beta(½,½)) distribution.  The vector is up‑sampled to a
    relatively large length (default 8192) so that the discrete problem
    approximates the continuous one more closely.

2.  Add a tiny random jitter and re‑normalise – this breaks symmetry and
    helps the optimiser escape the exact arcsine fixed‑point.

3.  Optimise the smooth‑max surrogate of the supremum convolution using a
    long Adam run.  The surrogate is
        Φτ(a)= (1/τ)·log Σk exp(τ·(a*a)_k)
    with an exponentially increasing temperature τ (0.5 → ≈150).  The
    soft‑max ensures a≥0 and Σa=1 throughout the run.

4.  Refine the result with a projected sub‑gradient descent on the true
    max‑convolution objective.  The sub‑gradient for the active shift is
    computed by a single FFT‑based convolution.

5.  Polish with an exponentiated‑gradient (mirror‑descent with KL geometry)
    which preserves non‑negativity and the simplex constraint.

6.  If time permits, perform a few short random restarts (jitter + short
    sub‑gradient + EG) to possibly improve the current best.

All stages respect the global timeout (default 1000 s).  The final vector
is returned as a plain list of Python floats; ``evaluate_sequence`` will
handle the normalisation and clipping internally.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a∗b via real FFT (output length = len(a)+len(b)-1).
    Both inputs are assumed 1‑D real vectors.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators and up‑sampling utilities
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discretised arcsine density: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25          # tiny regulariser to avoid division by zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation from len(v) to target_n,
    followed by re‑normalisation to Σ=1.
    """
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 80_000,
    tau_start: float = 0.5,
    tau_end: float = 150.0,
    base_lr: float = 0.30,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φτ(a) = (1/τ)·log Σ_k exp(τ·(a*a)_k)
    The variable is the softmax of logits w (ensuring a≥0, Σa=1).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # a = softmax(w)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights: softmax over τ·conv
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()          # shape (L,)

        # gradient wrt a : 2·(a *↞ weights)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wrt logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (learning‑rate decays like 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging (fast surrogate)
        if it % 20_000 == 0:
            surrogate = 2.0 * n * float(np.max(conv))   # same as compute_score
            print(f"    [Adam] it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution
    objective f(a)=max_k (a*a)_k.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # tolerance to capture all active shifts
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·a *↞ mask_rev
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 30_000 == 0:
            print(f"    [SubG] it={it:,d} max={max_val:.6f}")

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.20,
) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror descent with KL geometry) polish.
    Keeps the vector on the simplex throughout.
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        if it % 30_000 == 0:
            print(f"    [EG]    it={it:,d} max={max_val:.6f}")

    return a

    # ------------------------------------------------------------------
