"""
Optimiser for the “step‑function” challenge (improved version).

The idea is a multi‑scale stochastic + gradient search:

0️⃣  Load a fallback solution if present.
1️⃣  Generate a large and diverse pool of seed vectors (β‑shapes,
    power‑shapes, mixtures, sine‑squared etc.) at a modest size
    n≈1024 and keep the best few according to the *fast* surrogate
    score  2·n·max(conv(a,a))  (the official score for Σa=1).
2️⃣  Refine the selected seeds with a long Adam optimisation of the
    smooth‑max (log‑sum‑exp) surrogate.  The temperature τ is raised
    exponentially up to 400 so that the surrogate becomes almost the
    true max.
3️⃣  Upsample the champion to larger lengths (2048 → 4096 → 8192)
    and repeat Adam – the extra degrees of freedom allow a lower peak.
4️⃣  Perform a deterministic projected sub‑gradient descent on the
    *exact* max‑convolution objective, followed by an exponentiated‑
    gradient polish (mirror‑descent with KL geometry).  Both respect
    the simplex (non‑negative, Σ=1) constraint.
5️⃣  Finally, while time permits, run a simple simulated‑annealing
    loop: add a tiny Gaussian jitter, run a few Adam steps and keep
    the best improvement.  This helps escape possible local minima.
6️⃣  Return the best vector found (converted to a plain list of floats).

All stages respect the global 1000 s limit and keep the globally best
sequence according to the official ``evaluate_sequence`` function.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (no padding artefacts)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate for the official score (assumes Σa = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    # Duchi et al. 2008 implementation
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, v.size + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, w: float) -> np.ndarray:
    """Mixture w·arcsine + (1‑w)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = w * arc + (1.0 - w) * uni
    return mix / mix.sum()


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_sine_squared_shape(n: int) -> np.ndarray:
    """Shape a_i ∝ sin²(π·(i+½)/n).  Naturally non‑negative."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n) ** 2
    return vals / vals.sum()


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(v) → target_n, then renormalise."""
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 60_000,
    tau_start: float = 0.5,
    tau_end: float = 400.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k)
    using the softmax (log‑sum‑exp) gradient.
    The variables are logits w with a = softmax(w) ⇒ a ≥ 0, Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)               # logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax -> a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # softmax weights over τ·conv
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()      # shape (L,)

        # gradient wrt a : 2·conv(a, rev(weights))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # back‑prop through softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (step size decays like 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging
        if it % 20_000 == 0:
            surrogate = compute_score(np.exp(w - np.max(w)) / np.exp(w - np.max(w)).sum())
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.12,
) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # active mask (indices attaining the max, with tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(a, mask_rev) restricted to length n
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """Exponentiated‑gradient polish (mirror descent with KL geometry)."""
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

    # ------------------------------------------------------------------
    best_large_vec = best_small
    best_large_score = compute_score(best_small)

    for target_n in (2048, 4096, 8192):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Upscaling to n={target_n} and Adam refinement ===")
        up = upscale_vector(best_large_vec, target_n)

        # jitter to escape possible plateaus
        jitter = rng.normal(scale=1e-4, size=target_n)
        up = np.clip(up + jitter, 0.0, None)
        up /= up.sum()

        remaining = deadline - time.time()
        max_iters = int(min(80_000, remaining * 25))
        a_opt = adam_optimize_smooth_max(
            up,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=450.0,
            base_lr=0.22,
        )
        surrogate = compute_score(a_opt)
        print(f"    Upscaled surrogate = {surrogate:.6f}")

        if surrogate < best_large_score:
            best_large_score, best_large_vec = surrogate, a_opt

    # ------------------------------------------------------------------
    # 4️⃣  Exact sub‑gradient descent + EG polish on the best large candidate
    # ------------------------------------------------------------------
    if best_large_vec is not None:
        n_best = best_large_vec.shape[0]
        print(f"\n--- Exact refinement (n={n_best}) ---")
        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, remaining * 24))
        refined = subgradient_descent_exact(
            best_large_vec,
            deadline,
            max_iter=max_iter_sub,
            step0=0.09,
        )

        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(180_000, remaining * 24))
            refined = eg_refine(
                refined,
                deadline,
                max_iter=max_iter_eg,
                eta0=0.22,
            )

        try:
            val = evaluate_sequence(refined.tolist())
            print(f"Official score after exact + EG polish = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print("*** New global best! ***")
        except Exception as e:
            print(f"Evaluation failed after refinement: {e}")

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing / local‑search polish
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n--- Final stochastic local search ---")
        a = np.array(best_seq, dtype=float)
        n = len(a)

        while time.time() < deadline - 2.0:
            # small random move: transfer δ from i→j
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(0.0, min(a[i], 0.005))
            if delta == 0.0:
                continue
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # re‑project onto simplex (cheap, but keep sum exactly 1)
            a_candidate = project_to_simplex(a_candidate)

            # a few quick Adam steps to polish
            remaining = deadline - time.time()
            max_iters = int(min(4_000, remaining * 10))
            a_candidate = adam_optimize_smooth_max(
                a_candidate,
                deadline,
                max_iters=max_iters,
                tau_start=0.5,
                tau_end=300.0,
                base_lr=0.18,
            )
            cand_score = compute_score(a_candidate)

            if cand_score < compute_score(a):
                a = a_candidate
                # update best_seq if official improves
                try:
                    official = evaluate_sequence(a.tolist())
                    if official < best_val:
                        best_val, best_seq = official, a.tolist()
                        print(f"    Improved during local search → official {official:.6f}")
                except Exception:
                    pass

    # ------------------------------------------------------------------
    # 6️⃣  Guard – ensure a non‑empty sequence is returned
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
