import cvxpy as cp
from scipy.sparse import csr_matrix

def solve_lp_max_sum(f, M, b_init=None):
    n = len(f)
    # Build Toeplitz matrix A
    # Use csr_matrix for efficiency
    rows = []
    cols = []
    data = []
    for i, fi in enumerate(f):
        # rows are from i to i+n-1 inclusive
        rows_i = np.arange(i, i+n)
        cols_i = np.arange(0, n)
        rows.append(rows_i)
        cols.append(cols_i)
        data.append(np.full(n, fi))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    b = cp.Variable(n)
    constraints = [A @ b <= M, b >= 0]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    if b_init is not None:
        b.value = b_init
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)  # with appropriate options
    return b.value, prob.status
```

"""
Non‑negative step‑function minimiser for `evaluate_sequence`.

Strategy
--------
1. **Analytic start** – the truncated central‑binomial coefficients
   (a_i = C(2i,i)/4^i) give a near‑optimal shape with a theoretical bound
   ≈π/2.  We normalise them to the simplex.

2. **Linear‑programming mixing** – given the current vector f,
   we solve the LP

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M  (k = 0…2n‑2)
                    b ≥ 0

   where M = max(f*f).  The LP finds the “most massive” partner b that does
   not worsen the current worst convolution entry.  Mixing f←(1‑t)f+t·b
   (t≈0.25) provably decreases M.  We repeat this step a few dozen times on a
   modest size (n≈1024) – the LP is tiny enough to be solved many times
   within the 1000 s budget.

3. **Upscaling** – the improved small vector is interpolated to a larger
   length (4096 or 8192) and re‑projected onto the probability simplex.

4. **Smooth max optimisation** – we minimise a log‑sum‑exp surrogate of
   max(conv(a,a)) using Adam momentum.  The schedule for the smoothing
   parameter β is annealed from 5 to 200; the learning step decays with the
   square‑root of the iteration counter.

5. **Exact sub‑gradient polishing** – projected sub‑gradient descent on the
   true max‑objective cleans up any remaining overshoots.

6. **Stochastic hill‑climb** – inexpensive mass‑transfer moves (swap or
   shift a fraction of mass between two positions) further escape tiny
   local minima.

All stages respect the remaining wall‑clock budget and keep the best
sequence found.  The returned value is a plain Python list of non‑negative
floats (already normalised by the optimiser; `evaluate_sequence` will
apply the final clipping/summing anyway).
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the probability simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# LP utilities (stage 2)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the Toeplitz matrix A (size (2n‑1)×n) such that A @ b = conv(f,b).
    Each column i contains the vector f shifted down by i rows.
    """
    n = f.shape[0]
    diags = [np.full(n, fi) for fi in f]          # one diagonal per coefficient
    offsets = list(range(n))                       # offset i places the diagonal on row‑col=i
    A = sp.diags(diagonals=diags, offsets=offsets,
                 shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum(A: sp.csr_matrix, M: float,
                     warm_start: np.ndarray = None) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    b = cp.Variable(n)
    constraints = [A @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    if warm_start is not None:
        b.value = warm_start
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-7, reltol=1e-6, feastol=1e-7)
    except Exception:                     # solver may raise if problem is pathological
        return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    bv = b.value
    if bv is None:
        return None
    # Clip possible tiny negatives caused by numerical noise
    return np.maximum(bv, 0.0)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)           # first‑moment estimate
    v = np.zeros_like(a)           # second‑moment estimate
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing parameter β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smoothed objective (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]     # slice of length n

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()
        a = a_new

        # occasional jitter to shake out plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin for the harness

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. optional fallback (if a previous construction is available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. LP‑based refinement on a modest size (n≈1024)
    # -----------------------------------------------------------------
    n_small = 1024
    # initialise with truncated central‑binomial coefficients
    a_small = np.empty(n_small, dtype=float)
    a_small[0] = 1.0
    for i in range(n_small - 1):
        a_small[i + 1] = a_small[i] * (2 * i + 1) / (2.0 * (i + 1))
    a_small = project_to_simplex(a_small)

    best_small = a_small.copy()
    best_small_score = compute_score(best_small)

    # keep warm‑start for the LP variable
    warm_b = None

    # time allocation for the LP phase
    lp_deadline = deadline - 300.0   # leave ~5 min for the remaining stages
    iteration = 0
    while time.time() < lp_deadline:
        iteration += 1
        conv_f = convolution_fft(best_small, best_small)
        M = float(conv_f.max())
        A = build_convolution_matrix(best_small)
        b_opt = solve_lp_max_sum(A, M, warm_start=warm_b)
        if b_opt is None:
            print("LP solve failed – exiting LP loop.")
            break
        warm_b = b_opt               # reuse as warm start

        # normalise the partner vector
        b_opt = project_to_simplex(b_opt)

        # mix – a modest step size keeps the monotone descent guarantee
        t_mix = 0.25
        candidate = (1.0 - t_mix) * best_small + t_mix * b_opt
        candidate = project_to_simplex(candidate)

        cand_score = compute_score(candidate)
        if cand_score < best_small_score:
            best_small_score = cand_score
            best_small = candidate
            print(f"[LP iter {iteration}] improved score = {best_small_score:.6f}")

    print(f"LP phase finished after {iteration} iterations, best score = {best_small_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Upscale to a larger length for fine‑grained optimisation
    # -----------------------------------------------------------------
    # Choose the final length based on remaining budget
    remaining = deadline - time.time()
    if remaining > 600:
        n_big = 8192
    else:
        n_big = 4096

    a_big = upscale_vector(best_small, n_big)

    # -----------------------------------------------------------------
    # 3. Smooth log‑sum‑exp optimisation (Adam)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    a_opt, scr = smooth_gradient_optimize(
        a_big,
        time_deadline=smooth_deadline,
        max_iters=600_000,
        beta_start=6.0,
        beta_end=200.0,
        step0=1.0,
    )
    if scr < best_val:
        best_val = scr
        best_seq = a_opt.tolist()
        print(f"After smooth optimisation: score={best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing
    # -----------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("\nSub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 100.0,
            max_iter=500_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑grad improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_seq, polished_score = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=800_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb further improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a tiny uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
