"""
Propose a step‑function that minimises the provided `evaluate_sequence`.
The algorithm has three stages (all respecting the 1000 s wall‑clock limit)

1. **Diversified restarts + Adam‑accelerated log‑sum‑exp optimisation**  
   A large number of random starts (dense, sparse and arcsine‑shaped) are
   smoothed with a soft‑max approximation of the max‑convolution.  The
   temperature β is annealed from a modest value (≈5) up to a very sharp
   value (≈400) so that the optimiser finally concentrates on the true
   worst convolution entry.  Adam with a decaying stepsize is used and
   each iterate is projected back onto the probability simplex.

2. **Projected sub‑gradient polishing** – exact max‑convolution gradient
   descent with a decreasing stepsize.  This eliminates the remaining
   slack in the active constraint(s).

3. **Stochastic hill‑climb** – small mass‑transfer moves (including swaps)
   are sampled; improvements are kept and the step‑size is adapted
   automatically.  After the hill‑climb a short final sub‑gradient pass
   is performed.

The search automatically adapts the sequence length (8192 or 16384)
according to the remaining budget.  The best vector found (as a plain
list of floats) is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective (the value returned by `evaluate_sequence` after
    normalising the sum to 1).  The vector is assumed to be on the simplex.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smoothed max‑objective optimisation (log‑sum‑exp) with Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 400.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam‑accelerated minimisation of a soft‑max approximation to
    `max(conv(a,a))`.  The vector is kept on the simplex.
    Returns the best exact score encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal β (geometric schedule)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)                      # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()                                      # probability vector

        # gradient of the smooth objective:
        #   ∂/∂a  Σ w_k (a * a)[k] = 2 * (a_rev * w)[centered]
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]          # slice back to length n

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)
        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)

        # projection onto simplex
        a = project_to_simplex(a_new)

        # keep exact best
        if it % 200 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    # final check
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        cur = compute_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # occasional tiny jitter to escape plateaus
        if t % 1000 == 0:
            jitter = (step0 * 0.004 / np.sqrt(t)) * np.random.randn(n)
            a = project_to_simplex(a + jitter)

    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    deadline: float,
    max_moves: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Small random mass‑transfer moves (swap or shift) with an adaptive
    step‑size.  Operates on a vector that is already on the simplex."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best = a.copy()
    best_score = compute_score(best)

    step_frac = 0.08  # initial fraction of mass to move

    while max_moves > 0 and time.time() < deadline:
        cand = best.copy()

        if np.random.rand() < 0.5:
            # pure swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_frac * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best, best_score = cand, cur
            step_frac = min(step_frac * 1.06, 0.5)   # become bolder
        else:
            step_frac *= 0.97                         # become more cautious

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Helper start‑vector factories (no closures)
# ----------------------------------------------------------------------


def start_dense(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    return a / a.sum()


def start_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    # pick a random sparsity between 1 % and 15 %
    sparsity = rng.uniform(0.01, 0.15)
    k = max(2, int(sparsity * n))
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def start_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    # avoid division by zero at the boundaries
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective.  The routine respects the 1000 s
    wall‑clock budget and always returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val, best_seq = fb_val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose sequence length (larger length gives a tighter theoretical bound)
    # ------------------------------------------------------------------
    # If we have a lot of time left we try the larger length.
    if (deadline - time.time()) > 300 and budget_s >= 900:
        n = 16384
    else:
        n = 8192
    print(f"Chosen sequence length = {n}")

    # ------------------------------------------------------------------
    # 1. Random restarts + smooth Adam optimisation
    # ------------------------------------------------------------------
    # we will keep performing restarts until the time budget for this phase
    # is exhausted (leaving ~250 s for later phases).
    phase_deadline = deadline - 250.0

    start_factories = [
        lambda: start_dense(n, rng),
        lambda: start_sparse(n, rng),
        lambda: start_arcsine(n),
    ]

    restart_counter = 0
    while time.time() < phase_deadline:
        remaining = phase_deadline - time.time()
        if remaining < 30.0:   # not enough time for a useful run
            break

        # pick a start method uniformly at random
        start_a = rng.choice(start_factories)()
        # allocate a slice of the remaining budget for this run
        run_deadline = time.time() + min(remaining * 0.6, 120.0)

        # smooth optimisation
        a_opt, sc = smooth_gradient_optimize(
            start_a,
            deadline=run_deadline,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=400.0,
            step0=0.9,
        )
        restart_counter += 1
        if sc < best_val:
            best_val, best_seq = sc, a_opt.tolist()
            print(f"[restart {restart_counter}] new best (smooth) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Sub‑gradient refinement on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 130.0:
        print("\nSub‑gradient refinement")
        a_ref, sc_ref = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline=deadline - 80.0,
            max_iter=400_000,
            step0=0.07,
        )
        if sc_ref < best_val:
            best_val, best_seq = sc_ref, a_ref.tolist()
            print(f"Sub‑grad improvement → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Hill‑climb polishing (mass transfer moves)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        print("\nHill‑climb polishing")
        a_hc, sc_hc = hill_climb(
            best_seq,
            deadline=deadline - 5.0,
            max_moves=900_000,
        )
        if sc_hc < best_val:
            best_val, best_seq = sc_hc, a_hc.tolist()
            print(f"Hill‑climb improvement → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Very short final sub‑gradient pass (helps escape last plateaus)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        a_fin, sc_fin = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline=deadline,
            max_iter=80_000,
            step0=0.04,
        )
        if sc_fin < best_val:
            best_val, best_seq = sc_fin, a_fin.tolist()
            print(f"Final sub‑grad improvement → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing was found, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness calls the same function)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
