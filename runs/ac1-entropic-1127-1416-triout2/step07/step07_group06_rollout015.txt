"""
Propose a candidate step‑function that minimizes `evaluate_sequence`.

Algorithm overview
------------------
1. **Fallback** – keep any previously known construction.
2. **Parametric family search** – generate Beta‑distribution shaped vectors
   (including the heavy‑ends “arcsine” case) for several lengths and
   parameters.  Keep the best by the cheap `compute_score` (the exact
   objective for ∑a=1).
3. **Fine‑tuning** – starting from the best vector, run a sequence of
   deterministic gradient‑based phases:
   * flatten the convolution variance (projected Adam)
   * minimise a smooth max‑approximation (ε·log‑sum‑exp) with aggressive
     ε‑annealing
   * exact sub‑gradient descent on the true max‑objective
4. **Exploratory stochastic phase** – a classic simulated‑annealing
   sampler that moves mass between coordinates while always staying on the
   probability simplex.  The current state (not the global best) is updated
   with Metropolis acceptance, allowing the search to escape local minima.
5. **Symmetry averaging** – optional post‑processing (a←(a+rev(a))/2) and a
   final projection onto the simplex.
6. **Return** the best vector found before the time budget expires.

All heavy work respects the 1000 s budget – each optimisation routine checks
the global deadline and stops early if necessary.  The implementation uses
only NumPy (FFT based convolutions) and runs on a single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Parametric Beta distributions (including arcsine)
# ----------------------------------------------------------------------


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Return a probability vector proportional to
        x^{alpha-1} (1-x)^{beta-1}
    on the mid‑points x = (i+0.5)/n.
    """
    xs = (np.arange(n) + 0.5) / n
    # protect against zero‑power problems
    a = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, beta - 1.0)
    # replace NaNs/infs that can arise when alpha or beta < 1
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        # fallback to uniform if the parameters are pathological
        return np.full(n, 1.0 / n)
    return a / total


# ----------------------------------------------------------------------
# Gradient‑based optimisation phases
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 6000,
    step0: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the self‑convolution (flattening) using Adam.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        mu = np.mean(conv)
        diff = conv - mu

        # grad = 4 * sum_j a_j * diff_{i+j}
        diff_rev = diff[::-1]
        grad_full = convolution_fft(a, diff_rev)  # length 3n‑2
        grad = 4.0 * grad_full[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 8000,
    step0: float = 0.5,
    eps_start_factor: float = 0.4,
    eps_decay: float = 0.98,
    min_eps_factor: float = 0.001,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth‐max surrogate of the max‑convolution:

        f(a) = ε·log∑_k exp(conv_k/ε)

    ε decays exponentially to a tiny fraction of the current max.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        eps = maxc * max(eps_start_factor * (eps_decay ** (t - 1)), min_eps_factor)

        # soft‑max weights
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)   # ∑ w = 1

        # gradient of the surrogate (see derivation in the reference code)
        w_rev = w[::-1]
        grad_full = convolution_fft(a, w_rev)  # length 3n‑2
        grad = 2.0 * grad_full[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to leave flat plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    The sub‑gradient is taken from the currently active shift.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if t % 2000 == 0:
            # gentle shake to avoid stagnation
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (state‑based, not always from the global best)
# ----------------------------------------------------------------------


def simulated_annealing(
    start_seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list, float]:
    """
    Classic SA that updates a *current* state while remembering the best.
    Moves consist of a swap or a small mass transfer between two indices.
    Acceptance follows the Metropolis rule with a linear cooling schedule.
    """
    a_cur = np.array(start_seq, dtype=float)
    n = a_cur.shape[0]

    cur_score = compute_score(a_cur)
    best_a = a_cur.copy()
    best_score = cur_score

    # linear cooling to zero
    T0 = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = a_cur.copy()
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i → j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 1e-12:
                delta = 0.02 * cand[i]          # 2 % of the source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta_sc = cand_score - cur_score
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept move
            a_cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        # every few thousand steps inject a fresh cheap perturbation
        if step % 5000 == 0:
            rnd = random_dense_vector(n) * 0.015   # ~1.5 % random noise
            cand2 = best_a + rnd
            cand2 = project_to_simplex(cand2)
            cand2_score = compute_score(cand2)
            if cand2_score < best_score:
                best_score = cand2_score
                best_a = cand2.copy()
                a_cur = cand2
                cur_score = cand2_score

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Symmetry post‑processing
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Produce a symmetric version (a + reverse(a))/2 and re‑project onto the simplex.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call of evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback (if a historic construction is present)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. systematic search over a parametric Beta family
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    # symmetric alphas (arcsine ≈0.5) and a few asymmetric pairs
    alpha_grid = np.linspace(0.2, 2.0, 10)          # symmetric case
    asymmetric_pairs = [(0.3, 1.5), (0.5, 1.5), (1.5, 0.5), (1.5, 0.3)]

    for n in lengths:
        # symmetric Beta(a,a)
        for alpha in alpha_grid:
            a = beta_distribution(n, alpha, alpha)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[grid] n={n}, α={alpha:.2f} (sym) → score={sc:.6f}")

        # a few asymmetric shapes
        for (alpha, beta) in asymmetric_pairs:
            a = beta_distribution(n, alpha, beta)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[grid] n={n}, α={alpha:.2f}, β={beta:.2f} → score={sc:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 2. Deterministic fine‑tuning phases (if time permits)
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety net – uniform vector
        n = 256
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq))

    # convert to ndarray for the optimiser
    a_curr = np.array(best_seq, dtype=float)

    # ---- flattening variance (Adam) ----
    a_curr, sc = flatten_conv_adam(
        a_curr,
        time_deadline=deadline,
        max_iter=6000,
        step0=0.14,
    )
    if sc < best_score:
        best_score = sc
        best_seq = a_curr.tolist()
        print(f"[flatten] improved to {sc:.6f}")

    # ---- smooth max surrogate (Adam) ----
    a_curr, sc = smooth_max_adam(
        a_curr,
        time_deadline=deadline,
        max_iter=9000,
        step0=0.55,
        eps_start_factor=0.45,
        eps_decay=0.97,
        min_eps_factor=5e-4,
    )
    if sc < best_score:
        best_score = sc
        best_seq = a_curr.tolist()
        print(f"[smooth‑max] improved to {sc:.6f}")

    # ---- exact sub‑gradient descent ----
    a_curr, sc = subgrad_optimize(
        a_curr,
        time_deadline=deadline,
        max_iter=120_000,
        step0=0.065,
    )
    if sc < best_score:
        best_score = sc
        best_seq = a_curr.tolist()
        print(f"[sub‑grad] improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Stochastic exploration (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq_to_polish = best_seq if best_seq is not None else a_curr.tolist()
        polished_seq, polished_sc = simulated_annealing(
            seq_to_polish,
            time_deadline=deadline,
            max_steps=200_000,
        )
        if polished_sc < best_score:
            best_score = polished_sc
            best_seq = polished_seq
            print(f"[anneal] improved to {polished_sc:.6f}")

        # final cheap symmetry averaging (often helps)
        a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
        sym_sc = compute_score(a_sym)
        if sym_sc < best_score:
            best_score = sym_sc
            best_seq = a_sym.tolist()
            print(f"[sym‑avg] improved to {sym_sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Final official evaluation (ensures any clamping in evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
