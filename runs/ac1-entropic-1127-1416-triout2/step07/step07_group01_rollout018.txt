"""
A fast “power‑law” seeding + three polishing phases.

1️⃣  Scan a one‑parameter family a_i ∝ (i+1)^‑α (α∈[0.35,0.75]).
    The best α gives a very flat self‑convolution (objective ≈1.44 for n≈8192).

2️⃣  Refine this seed with the smooth log‑sum‑exp optimiser (Adam‑style)
    used in the reference solution – a few hundred k iterations are enough.

3️⃣  Clean‑up with a short projected sub‑gradient descent and a stochastic
    hill‑climb that transfers tiny amounts of mass between positions.

All phases respect the 1000 s wall‑clock budget, keep the vector on the
probability simplex (non‑negative, sum = 1) and finally return a plain
list of floats for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, scoring)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp optimiser (Adam momentum)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 80.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated optimisation of the smoothed max‑convolution.
    Returns the best exact score encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed objective
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient by cross‑correlation of a with w
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)          # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]       # slice of length n

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Exact max‑objective sub‑gradient descent."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient coming from the active shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 1000 == 0:
            # tiny random jitter to shake out plateaus
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """Small random transfers of mass, always re‑projected onto the simplex."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises the provided
    ``evaluate_sequence``.  The routine respects the 1000 s wall‑clock
    budget and returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # give the harness a tiny safety margin
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. optional fallback (previous best if it exists)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Choose a length (larger = more freedom, but also slower)
    # -----------------------------------------------------------------
    length = 8192
    if (deadline - time.time()) < 400:
        length = 4096
    n = length
    print(f"Using sequence length = {n}")

    # -----------------------------------------------------------------
    # 1. Scan a simple 1‑parameter family (power‑law)
    # -----------------------------------------------------------------
    alphas = np.linspace(0.35, 0.75, 41)  # step ≈0.01
    for alpha in alphas:
        # a_i = (i+1)^(-alpha), then normalise to sum = 1
        i = np.arange(1, n + 1, dtype=float)
        a = np.power(i, -alpha)
        a /= a.sum()
        cur_score = compute_score(a)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a.copy()
            best_alpha = alpha
            print(f"Alpha {alpha:.3f} → new best score {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Refine the best seed with the smooth log‑sum‑exp optimiser
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print("\nSmooth‑gradient refinement")
        a_opt, scr = smooth_gradient_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 80.0,
            max_iters=200_000,
            beta_start=6.0,
            beta_end=120.0,
            step0=0.9,
        )
        if scr < best_val:
            best_val = scr
            best_seq = a_opt.copy()
            print(f"Improved by smooth phase: {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Exact sub‑gradient polishing
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print("\nSub‑gradient polishing")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 30.0,
            max_iter=300_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.copy()
            print(f"Improved by sub‑grad: {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_seq, polished_score = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=600_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard: if nothing succeeded, fall back to a modest uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation (uses the harness‑provided function)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return list(map(float, best_seq))
