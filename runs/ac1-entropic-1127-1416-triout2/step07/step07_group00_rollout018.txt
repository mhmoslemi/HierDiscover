    n = a0.shape[0]
    a = a0.copy()
    conv = convolution_fft(a, a) # full conv
    current_max = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * float(current_max)
    # SA parameters
    step_scale = 0.2
    # temperature schedule: start with T0, decay to Tf
    T0 = 1e-4

    start_time = time.time()
    it = 0

    while time.time() < deadline:

        total = deadline - start_time
        if total <= 0:
            break
        t_frac = elapsed / total
        T = T0 * ((Tf / T0) ** t_frac)

        i = np.random.randint(n)
        # ensure a_i > - else try again loop small

def mass_transfer_sa(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    n = a0.shape[0]
    a = a0.copy()
    conv = convolution_fft(a)  # compute full conv
    current_max = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * float(current_max)

    step_scale = 0.2
    # acceptance ratios not needed as we enforce improvement only
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute active index and gradient
        k_max = int(np.argmax(conv))
        # compute gradient g
        g = np.zeros(n, dtype=float)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            g[i_range] = 2.0 * a[j_range]
        # pick source: among indices where a>0 and g>0, choose highest g*a
        active = (a > 1e-12) & (g > 0)
        if not np.any(active):
            # fallback random source
            src_candidates = np.where(a > 1e-12)[0]
            if src_candidates.size == 0:
                break
            i = np.random.choice(src_candidates)
        else:
            # choose argmax of g* a maybe
            score = g * a
            i = int(np.argmax(score))
        # pick target: index where g is minimal (prefer zero)
        # target must be different from source
        # gather zeros
        zeros = np.where(g == 0)[0]
        # exclude i
        zeros = zeros[zeros != i]
        if zeros.size > 0:
            j = int(np.random.choice(zeros))
        else:
            # pick index with smallest g not equal i
            candidate = np.arange(n)
            candidate = candidate[candidate != i]
            j = int(candidate[np.argmin(g[candidate])])
        # compute delta
        delta = step_scale * a[i]
        if delta <= 0:
            continue
        # store for revert
        a_i_old = a[i]
        a_j_old = a[j]
        # update conv
        # conv update using original a (pre-modification)
        conv[i:i+n] -= 2.0 * delta * a
        conv[j:j+n] += 2.0 * delta * a

            conv[idx_ij] -= 2.0 * delta * delta

            conv[idx_2i] += delta * delta

            conv[idx_2j] += delta * delta

            a[j] = a_j_old + delta

"""
Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence`.  The strategy is:

1.  Analytic one‑parameter family  a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α)
    – scan a modest α‑grid and keep the best (fast, gives a good start).

2.  A short smooth‐annealing gradient descent (log‑sum‑exp smoothing)
    to reduce the peak of the self‑convolution.

3.  A large “mass‑transfer” simulated‑annealing phase.
    Each move transfers a fraction of the mass from an index that
    contributes heavily to the current maximal convolution entry to an
    index with (almost) zero contribution.  The convolution is updated
    exactly via a simple O(n) formula, so millions of moves fit easily
    inside the 1000 s budget.

4.  A final short smooth‑gradient polishing.

All helper functions are top‑level; no closures are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective when sum(a)=1:
    2·n·max(a⊛a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytic α‑family start
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α), projected onto the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a modest α‑grid and keep the vector with the smallest exact score.
    Returns (vector, score).
    """
    best_a = None
    best_score = float("inf")
    # a slightly finer grid than before
    alphas = np.linspace(0.0, 2.5, 31)  # step ≈0.08
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Smooth log‑sum‑exp gradient descent (annealed β)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(start_a: np.ndarray,
                             deadline: float,
                             max_iters: int = 150_000,
                             beta_start: float = 5.0,
                             beta_end: float = 80.0,
                             step0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max objective
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using an exponentially increasing β and a decreasing step size.
    Returns the best exact score seen.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weights
        conv = convolution_fft(a, a)           # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()    # probability vector (size 2n‑1)

        # gradient = 2·a ⋆ w   (cross‑correlation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # step & projection
        step = step0 / np.sqrt(it)
        a = project_to_simplex(a - step * grad)

        # exact evaluation every few hundred iterations
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Large‑scale mass‑transfer simulated‑annealing
# ----------------------------------------------------------------------


def mass_transfer_sa(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    High‑volume random mass‑transfer moves, each move updates the
    convolution exactly (no FFT).  Moves are kept only if the maximal
    convolution value does not increase.  The step size adapts slowly.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # full convolution, length 2n‑1
    cur_max = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * float(cur_max)

    # adaptive move size (fraction of the source mass)
    step_scale = 0.25
    # simple acceptance counters for adaptive scaling
    accept_cnt = 0
    trial_cnt = 0

    while time.time() < deadline:
        trial_cnt += 1

        # --------------------------------------------------------------
        # identify the active index and its sub‑gradient
        # --------------------------------------------------------------
        k_max = int(np.argmax(conv))

        # gradient g_i = 2 * a_{k_max - i} for indices where this makes sense
        g = np.zeros(n, dtype=float)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        if i_start <= i_end:
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k_max - idx_i
            g[idx_i] = 2.0 * a[idx_j]

        # choose a source index (large contribution) and a target (almost zero)
        src_mask = (a > 1e-14) & (g > 0)
        if np.any(src_mask):
            # pick index maximising g_i * a_i (heuristic)
            src_i = int(np.argmax(g * a))
            if not src_mask[src_i]:
                # fallback to any positive‑mass index
                src_i = int(np.random.choice(np.where(a > 1e-14)[0]))
        else:
            # completely random source if gradient is zero everywhere
            src_i = int(np.random.choice(np.where(a > 1e-14)[0]))

        zero_g = np.where(g == 0)[0]
        zero_g = zero_g[zero_g != src_i]          # avoid self‑transfer
        if zero_g.size > 0:
            tgt_j = int(np.random.choice(zero_g))
        else:
            # pick the index with the smallest gradient (excluding source)
            cand = np.arange(n)
            cand = cand[cand != src_i]
            tgt_j = int(cand[np.argmin(g[cand])])

        # --------------------------------------------------------------
        # propose a mass transfer
        # --------------------------------------------------------------
        delta = step_scale * a[src_i]
        if delta <= 0.0:
            continue

        # remember old values for a possible revert
        old_src = a[src_i]
        old_tgt = a[tgt_j]

        # ----- update convolution exactly (using the original a) -----
        # note: a has not been changed yet, so we can safely use it
        conv[src_i:src_i + n] -= 2.0 * delta * a
        conv[tgt_j:tgt_j + n] += 2.0 * delta * a

        idx_ij = src_i + tgt_j
        if idx_ij < conv.size:
            conv[idx_ij] -= 2.0 * delta * delta
        idx_2i = 2 * src_i
        if idx_2i < conv.size:
            conv[idx_2i] += delta * delta
        idx_2j = 2 * tgt_j
        if idx_2j < conv.size:
            conv[idx_2j] += delta * delta

        new_max = conv.max()

        # ----- accept only if the peak does not increase -----
        if new_max <= cur_max + 1e-15:          # tiny tolerance for round‑off
            # accept
            a[src_i] = old_src - delta
            a[tgt_j] = old_tgt + delta
            cur_max = new_max
            accept_cnt += 1

            # update best record
            cur_score = 2.0 * n * float(cur_max)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # enlarge step_scale a little when we make progress
                step_scale = min(step_scale * 1.03, 0.55)
        else:
            # reject – revert the convolution update
            conv[src_i:src_i + n] += 2.0 * delta * a
            conv[tgt_j:tgt_j + n] -= 2.0 * delta * a
            if idx_ij < conv.size:
                conv[idx_ij] += 2.0 * delta * delta
            if idx_2i < conv.size:
                conv[idx_2i] -= delta * delta
            if idx_2j < conv.size:
                conv[idx_2j] -= delta * delta
            # shrink step_scale slowly to improve acceptance later
            step_scale = max(step_scale * 0.98, 1e-4)

        # occasionally adjust step_scale based on acceptance ratio
        if trial_cnt % 5000 == 0:
            acc_ratio = accept_cnt / trial_cnt
            if acc_ratio < 0.05:
                step_scale = max(step_scale * 0.9, 1e-4)
            elif acc_ratio > 0.30:
                step_scale = min(step_scale * 1.1, 0.55)
            # reset counters for the next block
            accept_cnt = 0
            trial_cnt = 0

    return best_a, best_score
