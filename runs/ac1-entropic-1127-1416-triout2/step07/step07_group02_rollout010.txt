"""
LP‑based refinement for the max‑convolution bound.

The algorithm works as follows

1.  Initialise a probability vector (non‑negative, sum = 1).  The
    arcsine distribution is a good start; other seeds can be added
    easily.

2.  Repeatedly solve a linear programme

        max   Σ g_i
        s.t.  (f * g)_k ≤ M   for all k∈{0,…,2n‑2}
              g_i ≥ 0

    where * is the linear convolution, f is the current sequence and
    M = max(f * f).  The constraints are linear in g, so the LP can be
    solved with SciPy’s high‑performance ‘highs’ solver.

    The optimal g has a larger total mass than f while respecting the
    same max‑convolution bound.  Mixing the two sequences (with a
    decreasing mixing factor) and renormalising yields a new f with a
    strictly smaller 2·n·max(conv)/sum².  This step is guaranteed to
    never increase the objective.

3.  After a few LP iterations (limited by the global time budget) a
    final polishing phase runs a projected sub‑gradient descent on the
    true objective to squeeze out the last bits of improvement.

The whole search respects the 1000 s budget and always returns a
valid non‑negative sequence.  If no improvement is found a uniform
fallback is used.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Fast convolution (real FFT) – used by the surrogate score and sub‑grad
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """Cheap, scale‑invariant surrogate: 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic seed families (only the arcsine is used by default)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑like distribution, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β) family on {0,…,n‑1}, regularised with eps."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


# ----------------------------------------------------------------------
# Linear‑programming helper – builds the convolution matrix A so that
#      A @ g = f * g   (linear in g)
# ----------------------------------------------------------------------


def conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Returns a sparse matrix A of shape (2·n‑1, n) with
    A @ g == np.convolve(f, g)   (linear convolution, zero‑padded).
    Each diagonal offset j (0 ≤ j < n) contains the constant f[j].
    """
    n = f.shape[0]
    L = 2 * n - 1
    # every diagonal has length n (rows j … j+n‑1)
    data = [np.full(n, f_j, dtype=np.float64) for f_j in f]
    offsets = np.arange(n)  # 0 … n‑1
    A = sp.diags(data, offsets, shape=(L, n), format="csc")
    return A


# ----------------------------------------------------------------------
# Linear‑programming refinement (alternating scheme)
# ----------------------------------------------------------------------


def lp_refine(initial_a: np.ndarray,
              deadline: float,
              max_iters: int = 20,
              t_init: float = 0.5,
              t_decay: float = 0.9) -> np.ndarray:
    """
    Alternating LP refinement.

    Parameters
    ----------
    initial_a : np.ndarray
        Starting probability vector (non‑negative, sum = 1).
    deadline : float
        Absolute time (seconds since epoch) after which the routine stops.
    max_iters : int
        Hard limit on the number of LP solves.
    t_init, t_decay :
        Mixing factor (1−t)*a + t*g; t is multiplied by t_decay after each
        iteration, gradually shrinking the step size.

    Returns
    -------
    np.ndarray
        Refined probability vector (still normalised).
    """
    a = np.maximum(initial_a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()

    n = a.size
    best_a = a.copy()
    best_val = fast_score(a)
    t = t_init

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current bound M = max(f * f)
        conv_aa = convolution_fft(a, a)
        M = float(np.max(conv_aa))

        # build linear constraints A @ g <= M
        A = conv_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # solve LP: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method="highs", options={"presolve": True})
        if not res.success:
            # abort if the solver fails – the current a is still usable
            print(f"    LP iteration {it}: solver failed ({res.message})")
            break

        g = np.maximum(res.x, 0.0)

        # convex combination and renormalisation
        a_new = (1.0 - t) * a + t * g
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new / s
        else:
            a_new = np.full_like(a_new, 1.0 / n)

        # compute surrogate score
        val = fast_score(a_new)
        print(f"    LP it={it:2d}  t={t:.3f}  approx score={val:.6f}")

        # keep the best iterate seen so far
        if val < best_val:
            best_val, best_a = val, a_new.copy()

        # prepare next loop
        a = a_new
        t *= t_decay

        # tiny early‑stop: no improvement in surrogate after a couple of steps
        if it >= 5 and abs(val - best_val) < 1e-9:
            break

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max) – final polishing step
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_descent_exact(init_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 150_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv(a,a)) / (sum(a)²).
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance to identify the active indices of the max
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution (see derivation in comments)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5000 == 0:
            approx = fast_score(a)
            print(f"    Sub‑grad it={it:,d}  approx score≈{approx:.6f}")

    return a


# ----------------------------------------------------------------------
# Simple up‑sampling (linear interpolation) between lengths
# ----------------------------------------------------------------------


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) → new_n, keeping non‑negativity."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq = new_seq / s
    else:
        new_seq = np.full(new_n, 1.0 / new_n)
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine combines
    • an LP‑based alternating improvement,
    • a final projected sub‑gradient polish,
    and respects the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Try to reuse a previously found construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Length schedule – start small, up‑sample, then refine larger sizes
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]   # larger lengths give tighter asymptotic bounds

    current = None   # holds the best vector from the previous stage

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1. Initialise a seed for this length
        # --------------------------------------------------------------
        if current is not None:
            a0 = upsample(current, n)
        else:
            # primary analytic seed
            a0 = arcsine_initial(n)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

        # --------------------------------------------------------------
        # 2. LP alternating refinement (bounded by remaining time)
        # --------------------------------------------------------------
        lp_deadline = deadline - 2.0   # keep a little slack for polishing
        a_lp = lp_refine(a0, lp_deadline, max_iters=25,
                         t_init=0.5, t_decay=0.9)

        # --------------------------------------------------------------
        # 3. Official evaluation of the LP result
        # --------------------------------------------------------------
        try:
            val = evaluate_sequence(a_lp.tolist())
            print(f"   LP‑refined official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_lp.tolist()
                print(f"   *** New global best after LP: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Evaluation failed after LP: {e}")

        # --------------------------------------------------------------
        # 4. Final polishing with exact sub‑gradient descent
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            print("   Polishing with sub‑gradient descent …")
            a_polish = subgradient_descent_exact(
                np.array(best_seq if best_seq else a_lp, dtype=float),
                deadline,
                max_iter=150_000,
                step0=0.07,
            )
            try:
                val2 = evaluate_sequence(a_polish.tolist())
                print(f"   Polished official score = {val2:.6f}")
                if val2 < best_val:
                    best_val, best_seq = val2, a_polish.tolist()
                    print(f"   *** Updated global best after polish: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Polishing evaluation failed: {e}")

        # keep the best vector for up‑sampling to the next (larger) length
        if best_seq is not None:
            current = np.array(best_seq, dtype=float)
        else:
            current = a_lp.copy()

        # early exit if we already beat the target
        if best_val <= 1.503:
            print(f"Target reached (score={best_val:.6f}); stopping early.")
            break

    # ------------------------------------------------------------------
    # 5. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
