"""
Step‑function optimisation – enhanced search.

The algorithm proceeds in stages:

1.  Choose a sequence length (preferably 8192 or 4096) and
    optimise a one‑parameter “arcsine‑like’’ family
    a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ using a univariate scalar optimiser.
    This yields a very good deterministic starting point.

2.  Refine the solution by:
    * mixing with the uniform distribution,
    * applying a power‑transform,
    * a short Adam‑accelerated soft‑max (log‑sum‑exp) descent,
    * exact sub‑gradient descent on the true max‑convolution,
    * a stochastic hill‑climber with mass‑transfer moves,
    * optional symmetry averaging.

All vectors are kept on the simplex (non‑negative, sum = 1), so the
fast surrogate score `2·n·max(conv)` equals the real evaluation metric.
The routine respects the supplied time budget (≈ 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy for scalar optimisation
# ----------------------------------------------------------------------
try:
    from scipy.optimize import minimize_scalar
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Utility functions – FFT convolution and simplex projection
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def best_arcsine_vector(n: int):
    """
    Find the exponent p in the family
        w_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ
    that minimises the surrogate score.
    Returns the normalised vector and its score.
    """
    eps = 1e-12
    i = np.arange(n, dtype=float)

    if _SCIPY_AVAILABLE:
        def f(p):
            w = ((i + eps) * (n - 1 - i + eps)) ** (-p)
            a = w / w.sum()
            return compute_score(a)

        res = minimize_scalar(
            f,
            bounds=(0.2, 1.0),
            method="bounded",
            options={"xatol": 1e-4, "maxiter": 30},
        )
        p_opt = res.x
    else:
        # fallback grid search (very cheap)
        p_grid = np.linspace(0.2, 1.0, 17)
        best_sc = float("inf")
        p_opt = 0.5
        for p in p_grid:
            w = ((i + eps) * (n - 1 - i + eps)) ** (-p)
            a = w / w.sum()
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                p_opt = p

    w = ((i + eps) * (n - 1 - i + eps)) ** (-p_opt)
    a = w / w.sum()
    return a, compute_score(a)


def uniform_vector(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# 1. Smooth soft‑max descent (Adam optimiser)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f(a) = eps·log Σ_k exp(conv_k/eps)
    where conv = a⋆a.

    Returns  grad = 2 * (w_rev ⊛ a),
    with w = softmax(conv/eps).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)                 # Σ w = 1

    # cross‑correlation w_rev ⊛ a gives Σ_j a_j w_{i+j}
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.2,
    eps_factor: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth soft‑max surrogate.
    ε is decreased exponentially; the algorithm stops when the time budget
    expires.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Adam parameters
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature – never smaller than 1 % of the current max
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        # Adam update (biased estimates)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)                       # basic decay
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 500 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent (max‑conv only)
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true objective 2·n·max(a⋆a).
    The sub‑gradient is taken at the shift where the convolution attains its
    maximum.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i} (zero outside the support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # tiny random shake every 1500 iterations
        if t % 1500 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climbing (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber that only accepts strictly improving moves.
    Moves:
        * swap two entries,
        * transfer a fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # transfer size adapts during the run
    step_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # become bolder
        else:
            step_frac *= 0.96                       # become more cautious

        max_steps -= 1

    return best_a, best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Enforce (approx.) symmetry: (a + reverse(a))/2 and re‑project."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` score.  The routine respects the given time budget
    (≈ 1000 s) and returns a plain Python list ready for the evaluation
    harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 0. Deterministic family search (arcsine‑like)
    # ------------------------------------------------------------------
    lengths = [8192, 4096]                     # larger sizes give finer granularity
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # ---- optimise exponent p for arcsine‑type shape -----------------
        a_best, sc_best = best_arcsine_vector(n)
        if sc_best < best_score:
            best_score, best_seq = sc_best, a_best.tolist()
            print(f"[init] n={n} best p‑shape score={sc_best:.6f}")

        # ---- mix with uniform ------------------------------------------------
        uniform = uniform_vector(n)
        for lam in np.linspace(0.5, 1.0, 11):
            if time.time() >= deadline:
                break
            cand = lam * a_best + (1.0 - lam) * uniform   # sum stays 1
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.tolist()
                print(f"[mix] n={n} λ={lam:.2f} score={sc:.6f}")

        # ---- power‑transform of the best shape -----------------------------
        for tau in np.linspace(0.8, 1.2, 9):
            if time.time() >= deadline:
                break
            cand = a_best ** tau
            cand = cand / cand.sum()
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.tolist()
                print(f"[pow] n={n} τ={tau:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # If nothing succeeded (unlikely) fall back to a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq))
        print(f"[fallback uniform] n={n} score={best_score:.6f}")

    a0 = np.array(best_seq, dtype=float)

    # ------------------------------------------------------------------
    # 1. Adam‑accelerated soft‑max gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_gd, sc_gd = adam_softmax_descent(
            a0,
            time_deadline=deadline - 2.0,
            max_iter=8000,
            step0=0.35,
            eps_factor=0.25,
        )
        if sc_gd < best_score:
            best_score, best_seq = sc_gd, a_gd.tolist()
            print(f"[adam] improved score={sc_gd:.6f}")

    # ------------------------------------------------------------------
    # 2. Exact sub‑gradient descent for fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        a_sub, sc_sub = subgradient_descent(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 2.0,
            max_iter=60000,
            step0=0.06,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.tolist()
            print(f"[subgrad] improved score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 3. Hill‑climbing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_hc, sc_hc = hill_climb(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 2.0,
            max_steps=180_000,
        )
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.tolist()
            print(f"[hill‑climb] improved score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # 4. Symmetry averaging (cheap, often beneficial)
    # ------------------------------------------------------------------
    a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score, best_seq = sc_sym, a_sym.tolist()
        print(f"[symmetry] improved score={sc_sym:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
