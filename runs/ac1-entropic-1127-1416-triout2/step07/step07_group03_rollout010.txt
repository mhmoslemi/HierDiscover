"""
step_function_optimizer.py
~~~~~~~~~~~~~~~~~~~~~~~~~~
A time‑bounded search for a non‑negative step function that minimises
``evaluate_sequence``.  The routine combines

* a large pool of analytically‑generated seeds (arcsine, beta, power,
  Dirichlet, mixtures, and the previous best if available);
* a smooth‑max Adam phase that minimises a log‑sum‑exp surrogate of the
  max‑auto‑convolution;
* a KL‑mirror (exponentiated‑gradient) polish that works on the exact
  non‑smooth objective;
* a final exact sub‑gradient descent;
* a tiny stochastic hill‑climbing refinement.

The code respects a wall‑clock budget (default 1000 s) and always
returns a valid non‑empty sequence.  All heavy work is performed with
FFT‑based convolutions for speed.
"""

import time
from typing import List
import numpy as np


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * b using real FFT (no circular wrap).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official objective for a vector with sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projection utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators (all return a vector that sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Discrete beta‑distribution‑like shape."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law envelope."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def random_dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point from Dirichlet(alpha) and normalise to 1."""
    vec = rng.gamma(alpha, 1.0, size=n)
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vec / total

# Optimisation primitives

def smooth_adam_optimize(z0: np.ndarray,
                         deadline: float,
                         n: int,
                         beta0: float = 5.0,
                         beta_max: float = 400.0,
                         ramp_iters: int = 50000,
                         lr: float = 0.09) -> np.ndarray:
    """
    Adam optimisation on the smooth log‑sum‑exp surrogate of max(conv).
    The iterate lives on the simplex via a soft‑max mapping.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = np.exp(z - np.max(z))                # soft‑max without explicit normalisation
        a /= a.sum()
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # raise temperature linearly
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                    # ≤ 0
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of surrogate w.r.t. a
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1: n - 1 + n]   # length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final projection onto simplex
    a = np.exp(z - np.max(z))
    a /= a.sum()
    return a


def exponentiated_gradient_polish(a0: np.ndarray,
                                 deadline: float,
                                 max_iter: int = 70_000,
                                 step0: float = 0.03) -> np.ndarray:
    """
    KL‑mirror (exponentiated‑gradient) descent on the exact max‑convolution.
    """
    a = np.clip(a0, 1e-15, None)
    a /= a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) w.r.t. a
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # step size decays like 1/sqrt(it)
        step = step0 / np.sqrt(it)
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


def subgradient_descent_exact(init_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 120_000,
                              step0: float = 0.09) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Operates on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    ExactSG it={it:,d} score={compute_score(a):.6f}")

    return a


def stochastic_hill_climb(a0: np.ndarray,
                          deadline: float,
                          init_sigma: float = 0.02,
                          decay: float = 0.999) -> np.ndarray:
    """
    Very cheap random walk around a0.  At each step a Gaussian
    perturbation is added, the result is projected onto the simplex,
    and the move is kept if it improves the official score.
    """
    rng = np.random.default_rng()
    best = a0.copy()
    best_val = compute_score(best)
    sigma = init_sigma

    while time.time() < deadline:
        cand = best + sigma * rng.standard_normal(best.shape)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        cand_val = compute_score(cand)
        if cand_val < best_val:
            best, best_val = cand, cand_val
            # occasional log
            if rng.random() < 0.01:
                print(f"    HillClimb improvement: {best_val:.6f}")

        sigma *= decay  # shrink perturbation slowly

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The routine respects the wall‑clock budget and always returns a
    non‑empty list of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback from a previous run if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Optimise several lengths, biggest first
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    # allocate a fraction of the total budget to each length (more for larger)
    length_fractions = [0.55, 0.30, 0.15]

    for n, frac in zip(lengths, length_fractions):
        if time.time() >= deadline - 10.0:
            break
        length_budget = (deadline - time.time()) * frac
        length_deadline = time.time() + length_budget
        print(f"\n=== Optimising length {n} (budget ≈ {length_budget:.1f}s) ===")

        # --------------------------------------------------------------
        # 1a. Build a diverse pool of seeds
        # --------------------------------------------------------------
        seeds = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
        ]

        # random power‑law seeds
        for _ in range(4):
            p = rng.uniform(0.05, 3.5)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # random beta‑shape seeds
        for _ in range(4):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # Dirichlet samples with various concentration parameters
        for alpha in [0.3, 1.0, 3.0]:
            seeds.append(random_dirichlet_seed(n, alpha, rng))

        # mixtures of arcsine & uniform
        for _ in range(2):
            w = rng.uniform(0.0, 1.0)
            seeds.append(generate_mixture_shape(n, w))

        # include previous best (rescaled / repeated) if we have one
        if best_seq:
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()
            seeds.append(cand)

        # --------------------------------------------------------------
        # 1b. Keep only the most promising seeds according to the fast surrogate
        # --------------------------------------------------------------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]   # keep at most five per length

        # --------------------------------------------------------------
        # 1c. Run the optimisation pipeline for each seed
        # --------------------------------------------------------------
        per_seed_budget = max(2.0,
                              (length_deadline - time.time()) /
                              (len(top_seeds) * 1.5 + 1e-9))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 8.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_budget:.1f}s")
            seed_deadline = min(time.time() + per_seed_budget, deadline - 4.0)

            # initialise Adam via log‑softmax (avoid log(0))
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # -----------------------------------------------------------------
            # 2. Smooth‑max Adam optimisation (more aggressive schedule)
            # -----------------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=400.0,
                ramp_iters=50000,
                lr=0.09,
            )

            # -----------------------------------------------------------------
            # 3. Official evaluation of the Adam result
            # -----------------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   After Adam official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best (Adam) = {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation of Adam result failed: {e}")

            # -----------------------------------------------------------------
            # 4. Mirror‑descent polishing (exact objective)
            # -----------------------------------------------------------------
            if time.time() < deadline - 5.0:
                polish_deadline = deadline - 2.0
                a_polished = exponentiated_gradient_polish(
                    np.array(best_seq, dtype=float),
                    polish_deadline,
                    max_iter=60_000,
                    step0=0.03,
                )
                try:
                    val2 = evaluate_sequence(a_polished.tolist())
                    print(f"   After EG polish official score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_polished.tolist()
                        print(f"   *** New global best (EG) = {best_val:.6f} ***")
                except Exception as e:
                    print(f"   EG polishing failed: {e}")

            # -----------------------------------------------------------------
            # 5. Quick stochastic hill‑climbing on the current best
            # -----------------------------------------------------------------
            if time.time() < deadline - 2.0:
                hill_deadline = deadline - 1.0
                a_hill = stochastic_hill_climb(
                    np.array(best_seq, dtype=float),
                    hill_deadline,
                    init_sigma=0.03,
                    decay=0.998,
                )
                try:
                    val3 = evaluate_sequence(a_hill.tolist())
                    print(f"   After hill‑climb score = {val3:.6f}")
                    if val3 < best_val:
                        best_val, best_seq = val3, a_hill.tolist()
                        print(f"   *** New global best (Hill‑climb) = {best_val:.6f} ***")
                except Exception as e:
                    print(f"   Hill‑climb evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 6. Final global polishing using exact sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global exact sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=150_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final sub‑gradient polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 7. Tiny convex combination sweep with uniform (helps flatten peaks)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nScanning convex combinations with uniform …")
        base = np.array(best_seq, dtype=float)
        uniform = np.full(base.shape, 1.0 / base.shape[0])
        for lam in np.linspace(0.0, 0.12, 13):
            cand = (1.0 - lam) * base + lam * uniform
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            try:
                val = evaluate_sequence(cand.tolist())
                if val < best_val:
                    best_val, best_seq = val, cand.tolist()
                    print(f"   *** New best via mixing λ={lam:.3f}: {best_val:.6f} ***")
            except Exception:
                continue

    # ------------------------------------------------------------------
    # 8. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
