"""
Propose a non‑negative step function (a probability vector) that
minimises the evaluation function used in the challenge.

Algorithm
---------
1. **Analytic seeds** – scan a small grid of the one‑parameter family
   a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α) and also a simple decreasing linear shape.
   Keep the best seed (lowest exact score).

2. **Adam‑based smoothed optimisation** – minimise the
   log‑sum‑exp smooth approximation of the max‑convolution.
   The smoothing parameter β is annealed from a modest value to a large one,
   while Adam (adaptive moments) provides fast convergence.
   After each update the vector is projected back onto the probability simplex.

3. **Long stochastic hill‑climb** – from the best vector obtained above,
   perform many random mass‑transfer moves; a move is kept if it improves the
   true (non‑smoothed) score.  This polishing stage can escape tiny local
   minima that the gradient step may miss.

All heavy work uses only NumPy and real FFTs, stays inside the 1000 s wall‑clock
budget and returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of `evaluate_sequence` when sum(a) == 1.
    Returns 2·n·max(conv) where conv = a ⊛ a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    if np.all(v <= 0):
        # uniform point is the projection of the all‑non‑positive vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for seeding
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return the normalised vector a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=float)
    # work in log‑space to avoid overflow/underflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(
    n: int, deadline: float, alphas: np.ndarray | None = None
) -> tuple[np.ndarray, float]:
    """Search a modest grid of α‑values, keep the one with smallest exact score."""
    if alphas is None:
        alphas = np.linspace(0.0, 2.5, 26)  # step ≈0.1
    best_a = None
    best_score = float("inf")
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def linear_decreasing_start(n: int) -> np.ndarray:
    """Simple decreasing shape a_i ∝ (n‑i)."""
    i = np.arange(n, dtype=float)
    w = n - i
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(
    start_a: np.ndarray,
    deadline: float,
    lr: float = 0.04,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    beta_start: float = 5.0,
    beta_end: float = 250.0,
    beta_ramp_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth objective

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    by Adam with an exponential schedule for β.
    Returns the best exact (non‑smoothed) score seen and the associated vector.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1

        # ---- β schedule (linear ramp up to beta_end) ----
        frac = min(t / beta_ramp_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        # ---- smooth weights ----
        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))       # stable shift
        w = exp_weights / exp_weights.sum()

        # ---- gradient of the smooth max (cross‑correlation) ----
        grad = 2.0 * convolution_fft(a[::-1], w)[n - 1 : n - 1 + n]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---- projection onto the simplex (keeps sum = 1, non‑negative) ----
        a = project_to_simplex(a)

        # ---- occasional exact evaluation ----
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] adam iter {t:>6d}  best_score {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Long stochastic hill‑climb (purely on the true objective)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
      – with prob≈0.5 swap two entries,
      – otherwise transfer a small fraction of mass from one index to another.
    Keep a move only if the exact score improves.
    """
    a = start_seq.copy()
    n = a.size

    best_a = a.copy()
    best_score = compute_score(best_a)

    # start with a modest step size; it will adapt automatically
    step_scale = 0.10

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.4)   # enlarge a bit when we succeed
        else:
            step_scale *= 0.97                         # shrink otherwise

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows the pipeline:
       * analytic seeding,
       * Adam‑based smoothed optimisation,
       * final hill‑climb polishing.
    It respects the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin – the harness adds its own checks
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose vector length.
    # 8192 gives a good trade‑off between expressivity and speed.
    # --------------------------------------------------------------
    n = 8192
    if (deadline - start_time) > 700:
        # if we have a lot of time, go a bit bigger – still cheap.
        n = 12288
    print(f"[{time.time():.1f}] vector length = {n}")

    # --------------------------------------------------------------
    # 0. optional fallback from a previously known good sequence
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fb = evaluate_sequence(height_sequence_1)  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Analytic seeds (α‑grid + linear decreasing)
    # --------------------------------------------------------------
    sec_deadline = deadline - 30.0
    a_alpha, sc_alpha = find_best_alpha_start(n, sec_deadline)
    a_linear = linear_decreasing_start(n)
    sc_linear = compute_score(a_linear)

    # pick the better of the two
    if sc_linear < sc_alpha:
        start_a = a_linear
        start_sc = sc_linear
        print(f"[{time.time():.1f}] best analytic start = linear (score {start_sc:.6f})")
    else:
        start_a = a_alpha
        start_sc = sc_alpha
        print(f"[{time.time():.1f}] best analytic start = α‑grid (score {start_sc:.6f})")

    if start_sc < best_score:
        best_score = start_sc
        best_seq = start_a.tolist()

    # --------------------------------------------------------------
    # 2. Adam‑based smoothed optimisation (most of the remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting Adam optimisation")
        a_adam, sc_adam = adam_optimize(
            np.array(best_seq, dtype=float) if best_seq is not None else start_a,
            deadline - 110.0,
            lr=0.04,
            beta_start=5.0,
            beta_end=250.0,
            beta_ramp_iters=250_000,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[{time.time():.1f}] after Adam best_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Final stochastic hill‑climb polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Safeguard – in the very unlikely event that nothing succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] fallback to uniform (score {best_score:.6f})")

    # --------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
