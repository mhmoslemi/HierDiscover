"""
Optimisation of a non‑negative step function (probability vector).

The search consists of three stages:

0. (optional) load a previously discovered good vector.
1. Start from a one‑parameter analytic family a_i(α) ∝ [(i+½)(n‑½‑i)]^{‑α}
   and refine α by a cheap local grid search.
2. Refine the best vector with a soft‑max (exponential) parametrisation.
   Adam optimisation on the smooth log‑sum‑exp approximation of the max‑convolution
   is performed while β (the smoothness) is annealed.
3. A short projected sub‑gradient phase followed by a stochastic hill‑climb
   provides the final polishing.

All operations respect the simplex constraint (∑a_i = 1, a_i ≥ 0) so the
objective value returned by `evaluate_sequence` is exactly
`2·n·max(a*a)` (the scaling factor of the original problem cancels).
The routine adapts the length of the vector to the remaining wall‑clock time
(8192 entries when enough time is available, otherwise 4096).  It prints
intermediate scores and finally returns the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(p: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` when the vector is already a
    probability distribution (∑p = 1).  Returns 2·n·max(p⊛p).
    """
    n = p.shape[0]
    conv = convolution_fft(p, p)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic one‑parameter family and refinement
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    Return the normalised vector a_i(α) ∝ [(i+½)(n‑½‑i)]^{‑α}.
    The result lies on the probability simplex.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def coarse_alpha_grid(n: int, deadline: float) -> tuple[np.ndarray, float, float]:
    """Scan a modest α‑grid and return (vector, score, α)."""
    best_vec = None
    best_score = float("inf")
    best_alpha = 0.0
    alphas = np.linspace(0.0, 2.5, 26)          # step ≈ 0.1
    for a in alphas:
        if time.time() >= deadline:
            break
        vec = alpha_start(n, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
            best_alpha = a
    return best_vec, best_score, best_alpha


def refine_alpha_around(
    n: int,
    start_alpha: float,
    start_vec: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float, float]:
    """
    Local refinement of α: repeatedly sample a finer grid in a shrinking
    neighbourhood around the current best α.
    """
    best_alpha = start_alpha
    best_vec = start_vec
    best_score = compute_score(start_vec)

    radius = 0.2
    for _ in range(5):                     # five refinement rounds
        if time.time() >= deadline:
            break
        alphas = np.linspace(best_alpha - radius,
                             best_alpha + radius, 21)
        for a in alphas:
            if time.time() >= deadline:
                break
            if a < 0.0:
                continue
            vec = alpha_start(n, a)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_vec = vec.copy()
                best_alpha = a
        radius *= 0.5
    return best_vec, best_score, best_alpha


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def adam_softmax_optimize(
    start_p: np.ndarray,
    deadline: float,
    beta_start: float = 1.0,
    beta_end: float = 200.0,
    lr: float = 0.05,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on logits (soft‑max) minimising the smoothed objective

        f_β(p) = (1/β)·log Σ_k exp(β·(p⊛p)_k)

    The smoothing parameter β is increased exponentially from `beta_start`
    to `beta_end`.  The function returns the best exact score observed.
    """
    n = start_p.shape[0]
    # initialise logits (add small epsilon to avoid log(0))
    u = np.log(start_p + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta_ratio = beta_end / beta_start

    best_p = start_p.copy()
    best_score = compute_score(best_p)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # ---------- annealing ----------
        t = it / max_iters
        beta = beta_start * (beta_ratio ** t)

        # ---------- forward pass ----------
        # probability vector
        max_u = np.max(u)                     # stability
        exp_u = np.exp(u - max_u)
        p = exp_u / exp_u.sum()               # soft‑max, automatically on simplex

        conv = convolution_fft(p, p)          # length 2n‑1
        maxc = conv.max()
        # log‑sum‑exp weights
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()                           # shape (2n‑1,)

        # ---------- gradient w.r.t. p ----------
        p_rev = p[::-1]
        conv_rev = convolution_fft(p_rev, w)  # length 3n‑2
        grad_p = 2.0 * conv_rev[n - 1:n - 1 + n]   # ∂/∂p of smoothed max

        # ---------- gradient w.r.t. logits ----------
        mean = np.dot(p, grad_p)
        grad_u = p * (grad_p - mean)

        # ---------- Adam update ----------
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u ** 2)
        m_hat = m / (1 - 0.9 ** it)
        v_hat = v / (1 - 0.999 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        u -= step

        # ---------- occasional exact evaluation ----------
        if it % 500 == 0:
            # recompute exact score for the current soft‑max vector
            sc = compute_score(p)
            if sc < best_score:
                best_score = sc
                best_p = p.copy()
                print(f"[{time.time():.1f}] Adam iter {it:6d}, β={beta:.2f}, best={best_score:.6f}")

    # final check
    sc = compute_score(best_p)
    if sc < best_score:
        best_score = sc
        best_p = best_p.copy()
    return best_p, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (true max‑objective)
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_vec: np.ndarray,
    deadline: float,
    step0: float = 0.07,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Classical projected sub‑gradient descent on 2·n·max(p⊛p)."""
    n = start_vec.shape[0]
    a = start_vec.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # active index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] subgrad iter {t:6d}, best={best_score:.6f}")

    sc = compute_score(best_a)
    if sc < best_score:
        best_score = sc
        best_a = best_a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Long stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
      - with prob 0.5 swap two entries,
      - otherwise move a fraction of mass from i to j.
    A move is kept only if the exact score improves.
    """
    n = start_seq.size
    best_a = start_seq.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12  # typical transferred mass fraction

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1
        if steps % 100_000 == 0:
            print(f"[{time.time():.1f}] hill‑climb step {steps:,}, best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to create a sparse random start (used for optional restarts)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Generate a random probability vector with a few non‑zero entries."""
    sparsity = np.random.uniform(0.02, 0.15)          # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The algorithm follows the four‑stage plan described above and respects
    the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a length that fits the remaining budget
    # ------------------------------------------------------------------
    n = 4096
    if deadline - start_time > 600:      # ample time → longer vectors
        n = 8192
    print(f"[{time.time():.1f}] chosen length = {n}")

    # ------------------------------------------------------------------
    # 0. Fallback – try a previously discovered construction if present
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic α‑grid + local refinement
    # ------------------------------------------------------------------
    vec_alpha, sc_alpha, alpha0 = coarse_alpha_grid(n, deadline - 30.0)
    if vec_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = vec_alpha.tolist()
        print(f"[{time.time():.1f}] after coarse α‑grid best = {best_score:.6f}")

    vec_refined, sc_refined, alpha_opt = refine_alpha_around(
        n, alpha0, vec_alpha, deadline - 20.0
    )
    if vec_refined is not None and sc_refined < best_score:
        best_score = sc_refined
        best_seq = vec_refined.tolist()
        print(f"[{time.time():.1f}] after α‑refinement (α={alpha_opt:.4f}) best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the soft‑max parametrisation
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] starting Adam soft‑max optimisation")
        start_vec = np.array(best_seq, dtype=float)
        # reserve time for the two final polishing stages
        adam_deadline = deadline - 150.0
        a_adam, sc_adam = adam_softmax_optimize(
            start_vec,
            adam_deadline,
            beta_start=1.0,
            beta_end=250.0,
            lr=0.07,
            max_iters=250_000,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient refinement (shorter than before)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 60.0,
            step0=0.06,
            max_iters=300_000,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: fall back to a modest uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will call evaluate_sequence again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
