def optimize_lbfgs(a0, beta, maxiter, deadline):
    eps = 1e-12
    y0 = np.log(np.maximum(a0, eps))
    # define fun
    def fun(y):
        # compute smooth objective and gradient
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        conv = convolution_fft(a, a)
        maxc = conv.max()
        diff = conv - maxc
        # Clip diff: keep <=0
        diff = np.where(diff>0, 0, diff)
        exp_weights = np.exp(beta * diff)
        sum_exp = exp_weights.sum()
        # f = maxc + np.log(sum_exp) / beta
        f = maxc + np.log(sum_exp) / beta
        w = exp_weights / sum_exp
        # gradient w.r.t a
        a_rev = a[::-1]
        grad_conv = convolution_fft(a_rev, w)
        grad_a = 2.0 * grad_conv[len(a)-1:len(a)-1+len(a)]
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)
        return f, grad_y.astype(np.float64)
    # callback to early stop if time
    iteration = [0]
    def callback(y):
        iteration[0] += 1
        if time.time() >= deadline:
            raise KeyboardInterrupt
    try:
        res = minimize(fun, y0, method='L-BFGS-B', jac=True, callback=callback,
                       options={'maxiter': maxiter, 'disp': False})
    except KeyboardInterrupt:
        # abort early
        a_opt = np.exp(y0 - np.max(y0))
        a_opt /= a_opt.sum()
        return a_opt
    y_opt = res.x
    a_opt = np.exp(y_opt - np.max(y_opt))

"""
Proposes a non‑negative step‑function (sequence) that minimises
`evaluate_sequence`.  The search consists of the following stages:

1.  **Analytical edge‑heavy start** – a fine grained scan of the
    family a_i ∝ [(i+½)(n‑½‑i)]^(‑α) on the probability simplex.
2.  **Quasi‑Newton refinement** – minimise the log‑sum‑exp
    approximation of max(a*a) using L‑BFGS with a continuation
    on the inverse‑temperature β (β grows up to ≈10⁵).  The
    optimisation works in the soft‑max (exponential) parametrisation,
    which guarantees non‑negativity and exact sum‑to‑one.
3.  **Adam‑style projected sub‑gradient** on the true max‑objective.
    This yields fast local improvements once the peak is already
    isolated.
4.  **Long stochastic hill‑climb** – random mass‑transfer moves
    (accepting only improving steps) that polish the final solution.

All heavy work uses NumPy (FFT‑based convolutions).  The routine respects
the 1000 s wall‑clock limit and returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` when sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytic edge‑heavy start (one‑parameter α family)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return normalised a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a

        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. L‑BFGS optimisation of a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


# Try to import scipy; if unavailable we fall back to the previous method.
try:
    from scipy.optimize import minimize
    SCIPY_OK = True
except Exception:                     # pragma: no cover
    SCIPY_OK = False


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a point on the simplex)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def l2b_smooth_obj_grad(y: np.ndarray, beta: float):
    """
    Returns (f, grad) where
        f  = max(a*a) approximated by log‑sum‑exp with temperature β,
        grad = ∂f/∂y   (y are the soft‑max parameters).
    """
    a = softmax(y)

    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    diff = conv - maxc
    diff[diff > 0] = 0.0                               # safeguard against round‑off
    # weights of the smooth max
    exp_w = np.exp(beta * diff)
    sum_exp = exp_w.sum()
    # smoothed max value
    f = maxc + np.log(sum_exp) / beta

    w = exp_w / sum_exp                                 # probability vector, length 2n‑1

    # gradient w.r.t. a (cross‑correlation of a_rev and w)
    a_rev = a[::-1]
    tmp = convolution_fft(a_rev, w)
    n = a.shape[0]
    grad_a = 2.0 * tmp[n - 1:n - 1 + n]                # length n

    # chain rule: ∂a/∂y = diag(a) - a aᵀ
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    return float(f), grad_y.astype(np.float64)


def lbfgs_optimize(start_a: np.ndarray,
                   beta: float,
                   max_iter: int,
                   deadline: float) -> np.ndarray:
    """
    Runs L‑BFGS on the log‑sum‑exp objective with inverse‑temperature β.
    Returns the (normalised) vector a that minimises the smooth objective.
    """
    eps = 1e-12
    y0 = np.log(np.maximum(start_a, eps))

    def fun(y):
        f, g = l2b_smooth_obj_grad(y, beta)
        return f, g

    # callback – abort if we run out of time
    def callback(_):
        if time.time() >= deadline:
            raise KeyboardInterrupt

    try:
        res = minimize(fun,
                       y0,
                       method='L-BFGS-B',
                       jac=True,
                       callback=callback,
                       options={'maxiter': max_iter, 'disp': False})
    except KeyboardInterrupt:                          # pragma: no cover
        # early termination – return the current iterate
        y_opt = y0
    else:
        y_opt = res.x

    a_opt = softmax(y_opt)
    return a_opt


def cascade_lbfgs(start_a: np.ndarray,
                  betas,
                  max_iter_stage: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Runs a sequence of L‑BFGS optimisations with increasing β.
    Returns the best vector (and its exact score) encountered.
    """
    a = start_a.copy()
    best_a, best_sc = a, compute_score(a)
    for beta in betas:
        if time.time() >= deadline - 4.0:
            break
        # allocate a modest number of L‑BFGS iterations per β
        a = lbfgs_optimize(a, beta, max_iter_stage, deadline - 2.0)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] β={beta:.0e} → score {sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Adam‑style projected sub‑gradient on the true max‑objective
# ----------------------------------------------------------------------


def adam_max_optimize(start_a: np.ndarray,
                      deadline: float,
                      max_iters: int = 300_000,
                      lr: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the exact max‑convolution objective.
    Returns the best vector seen and its exact score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                # active index

        # sub‑gradient for the active index (one‑hot weight)
        w = np.zeros_like(conv)
        w[k] = 1.0
        a_rev = a[::-1]
        tmp = convolution_fft(a_rev, w)
        grad = 2.0 * tmp[n - 1:n - 1 + n]

        # Adam update on the simplex (project after the step)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = project_to_simplex(a)

        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Long stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(start_a: np.ndarray,
               deadline: float,
               max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Simple random walk on the simplex that accepts only improving moves.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc, best_a = sc, cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    The routine follows the 4‑stage plan described above and obeys the
    1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # tiny safety margin

    # --------------------------------------------------------------
    # Choose a convenient length (power of two, FFT‑friendly)
    # --------------------------------------------------------------
    n = 8192
    if (deadline - start_time) > 600:           # more time → larger vector
        n = 16384
    print(f"[{time.time():.1f}] chosen length = {n}")

    # --------------------------------------------------------------
    # 0. fallback – keep a previously known good construction if present
    # --------------------------------------------------------------
    best_seq = None
    best_sc = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_sc:
            best_sc, best_seq = fb, fallback
    except Exception:   # pragma: no cover
        pass

    # --------------------------------------------------------------
    # 1. analytical α‑grid start (more refined than previous version)
    # --------------------------------------------------------------
    a_alpha, sc_alpha = best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_sc:
        best_sc, best_seq = sc_alpha, a_alpha.tolist()
        print(f"[{time.time():.1f}] after α‑grid best = {best_sc:.6f}")

    # --------------------------------------------------------------
    # 2. L‑BFGS cascade on log‑sum‑exp (high β → true max)
    # --------------------------------------------------------------
    if SCIPY_OK and best_seq is not None and time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] starting L‑BFGS cascade")
        betas = [5e0, 2e1, 8e1, 5e2, 2e3, 1e4, 5e4, 2e5]   # exponential schedule
        a_lbfgs, sc_lbfgs = cascade_lbfgs(
            np.array(best_seq, dtype=float),
            betas,
            max_iter_stage=150,               # cheap L‑BFGS at each β
            deadline=deadline - 180.0
        )
        if sc_lbfgs < best_sc:
            best_sc, best_seq = sc_lbfgs, a_lbfgs.tolist()
            print(f"[{time.time():.1f}] post‑L‑BFGS best = {best_sc:.6f}")

    # --------------------------------------------------------------
    # 3. Adam‑style projected sub‑gradient refinement (max‑objective)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] Adam sub‑gradient refinement")
        a_adam, sc_adam = adam_max_optimize(
            np.array(best_seq, dtype=float),
            deadline - 110.0,
            max_iters=350_000,
            lr=0.03,
        )
        if sc_adam < best_sc:
            best_sc, best_seq = sc_adam, a_adam.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_sc:.6f}")

    # --------------------------------------------------------------
    # 4. Final long hill‑climb polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_sc:
            best_sc, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_sc:.6f}")

    # --------------------------------------------------------------
    # Safeguard – if nothing succeeded fall back to a modest uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_sc = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
