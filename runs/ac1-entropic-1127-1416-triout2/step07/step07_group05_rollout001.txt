"""

Algorithm (≈ 1000 s total):

     ``compute_score`` (identical to ``evaluate_sequence`` for sum‑=‑1 vectors).
  2️⃣ Upsample the best candidate to ``n_big = 4096`` and run a short Adam

  3️⃣ Refine the Adam result with an L‑BFGS‑B optimisation of the same smooth
     surrogate using a high temperature (τ≈300) – this gives a very accurate

  4️⃣ Projected sub‑gradient descent on the *true* max‑convolution objective,

  5️⃣ A fast simulated‑annealing (SA) phase performs tiny mass‑shifts between

  6️⃣ The best sequence discovered during the whole run is finally evaluated

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (length = len(a)+len(b)-1).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a sequence with sum(a)=1:
        2·n·max(conv(a,a)).
    This equals ``evaluate_sequence`` after normalising the sum to 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto {x ≥ 0, Σx = 1}.
    Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_asymmetric_power_shape(
    n: int, p: float, eps_left: float, eps_right: float
) -> np.ndarray:
    """
    Asymmetric generalised arcsine:
       a_i ∝ 1 / ((i+eps_left)*(n‑½‑i+eps_right))**p .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + eps_left) * (n - 0.5 - i + eps_right) + 1e-12) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation from len(v) → target_n,
    then re‑normalise to sum = 1.
    """
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 30_000,
    tau_start: float = 0.5,
    tau_end: float = 120.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate.
    Variables are logits w with a = softmax(w) → a ≥ 0, Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)  # initialise logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        conv = convolution_fft(a, a)

        # smooth‑max weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()

        # gradient wrt a
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wrt logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def smooth_max_obj_and_grad(w: np.ndarray, tau: float) -> Tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate
        Φτ(softmax(w)) = (1/τ)·log∑exp(τ·conv(a,a)).
    Returns (value, grad_w) where grad_w respects the softmax parameterisation.
    """
    # softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = convolution_fft(a, a)
    scaled = conv * tau
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    value = (max_scaled + np.log(sum_exp)) / tau

    # softmax weights
    weights = exp_scaled / sum_exp

    # gradient wrt a
    n = a.shape[0]
    L = conv.shape[0]
    rev_weights = weights[::-1]
    conv_rev = convolution_fft(a, rev_weights)
    grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # gradient wrt logits
    dot = np.dot(a, grad_a)
    grad_w = a * (grad_a - dot)

    return value, grad_w


def lbfgs_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    tau: float = 300.0,
    max_iter: int = 800,
) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth‑max surrogate.
    The optimisation variable is the logits w (a = softmax(w)).
    """
    n = a_init.shape[0]
    w0 = np.log(a_init + 1e-12)

    # SciPy minimise with supplied gradient
    def fun(w):
        val, grad = smooth_max_obj_and_grad(w, tau)
        return val, grad

    options = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    res = minimize(fun, w0, method="L-BFGS-B", jac=True, options=options)

    w_opt = res.x
    # final softmax
    w_shift = w_opt - np.max(w_opt)
    a_opt = np.exp(w_shift)
    a_opt /= a_opt.sum()
    return a_opt


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
    noise_scale: float = 1e-4,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution
    objective, with a tiny Gaussian perturbation to aid escape from shallow
    plateaus.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) on original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # add small random noise (helps exploration)
        subgrad += noise_scale * np.random.randn(n)

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a


def simulated_annealing_refine(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
    start_temp: float = 0.05,
    cooling: float = 0.9995,
    eps_factor: float = 5e-4,
) -> np.ndarray:
    """
    Very cheap SA that moves a small amount of mass between two coordinates.
    The objective is the exact ``compute_score`` (identical to the official
    score after normalising the sum to 1).  The current best vector is always
    kept.
    """
    a = a_init.copy()
    cur_score = compute_score(a)
    n = len(a)
    temp = start_temp
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # choose donor and recipient
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount to move
        eps = eps_factor * rng.random() * a[i]
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # re‑normalise (sum unchanged ⇒ no need)
        new_score = compute_score(a_new)

        # Metropolis criterion
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temp):
            a = a_new
            cur_score = new_score

        temp *= cooling
        if temp < 1e-12:
            temp = 1e-12

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with minimal ``evaluate_sequence``.
    The routine follows the multi‑phase scheme described in the module‑level
    docstring and respects the global 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    best_seq: List[float] = []
    best_score: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_score:
                best_score, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Random parametric search on a modest size (n=2048)
    # ------------------------------------------------------------------
    n_search = 2048
    pool: List[Tuple[float, np.ndarray]] = []

    # deterministic starters
    pool.append((compute_score(np.full(n_search, 1.0 / n_search)), np.full(n_search, 1.0 / n_search)))
    pool.append((compute_score(arcsine_initial(n_search)), arcsine_initial(n_search)))

    # random families (bounded by remaining time)
    while time.time() < deadline - 5.0 and len(pool) < 2000:
        typ = rng.integers(0, 6)
        if typ == 0:  # beta‑like
            a = rng.uniform(0.05, 3.0)
            b = rng.uniform(0.05, 3.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n_search))
            cand = generate_beta_shape(n_search, a, b, eps)
        elif typ == 1:  # power‑shape
            p = rng.uniform(0.2, 2.5)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n_search, p, eps)
        elif typ == 2:  # mixture arcsine/uniform
            w = rng.random()
            cand = generate_mixture_shape(n_search, w)
        elif typ == 3:  # triangular bump
            peak = rng.integers(0, n_search)
            cand = generate_triangular_shape(n_search, int(peak))
        elif typ == 4:  # asymmetric power shape
            p = rng.uniform(0.2, 2.5)
            eps_l = rng.uniform(0.0, 1.0)
            eps_r = rng.uniform(0.0, 1.0)
            cand = generate_asymmetric_power_shape(n_search, p, eps_l, eps_r)
        else:  # uniform with small random perturbations
            base = np.full(n_search, 1.0 / n_search)
            jitter = rng.normal(scale=1e-3, size=n_search)
            cand = base + jitter
            cand = np.clip(cand, 0.0, None)
            cand = cand / cand.sum()

        pool.append((compute_score(cand), cand))

    # keep a few best seeds
    pool.sort(key=lambda x: x[0])
    top_seeds_small = [vec for _, vec in pool[:5]]
    print(f"\nRandom search (n={n_search}) complete – best surrogate = {pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Upsample to a larger size and Adam optimisation
    # ------------------------------------------------------------------
    n_big = 4096
    best_big = None
    best_big_score = float("inf")

    for idx, seed_vec in enumerate(top_seeds_small, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Adam optimisation on upscaled seed {idx}/{len(top_seeds_small)} ---")
        up = upscale_vector(seed_vec, n_big)

        # slight random jitter before optimisation
        jitter = rng.normal(scale=1e-4, size=n_big)
        up = up + jitter
        up = np.clip(up, 0.0, None)
        up = up / up.sum()

        remaining = deadline - time.time()
        max_iters = int(min(30_000, remaining * 25))  # a few thousand iterations
        a_opt = adam_optimize_smooth_max(
            up,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=120.0,
            base_lr=0.23,
        )
        surrogate = compute_score(a_opt)
        print(f"    Post‑Adam surrogate = {surrogate:.6f}")

        if surrogate < best_big_score:
            best_big_score, best_big = surrogate, a_opt

    if best_big is None:
        # fallback to upscaled arcsine if everything failed
        best_big = upscale_vector(arcsine_initial(n_search), n_big)
        best_big_score = compute_score(best_big)

    # ------------------------------------------------------------------
    # 3️⃣ L‑BFGS‑B refinement on the smooth‑max surrogate (τ large)
    # ------------------------------------------------------------------
    print("\n--- L‑BFGS‑B refinement (smooth‑max, τ=300) ---")
    tau_lbfgs = 300.0
    remaining = deadline - time.time()
    max_lbfgs_iter = int(min(800, remaining * 1.5))  # very cheap, L‑BFGS‑B converges fast
    a_lbfgs = lbfgs_optimize_smooth_max(best_big, deadline, tau=tau_lbfgs, max_iter=max_lbfgs_iter)
    lbfgs_score = compute_score(a_lbfgs)
    print(f"    L‑BFGS‑B surrogate = {lbfgs_score:.6f}")
    if lbfgs_score < best_big_score:
        best_big_score, best_big = lbfgs_score, a_lbfgs

    # ------------------------------------------------------------------
    # 4️⃣ Exact sub‑gradient descent + EG polish (with noise)
    # ------------------------------------------------------------------
    print("\n--- Exact sub‑gradient descent (with noise) ---")
    remaining = deadline - time.time()
    max_sub_iters = int(min(200_000, remaining * 30))
    a_sub = subgradient_descent_exact(best_big, deadline, max_iter=max_sub_iters,
                                      step0=0.09, noise_scale=5e-5)

    print("\n--- Exponentiated‑gradient polish ---")
    remaining = deadline - time.time()
    max_eg_iters = int(min(150_000, remaining * 30))
    a_eg = eg_refine(a_sub, deadline, max_iter=max_eg_iters, eta0=0.22)

    exact_score = compute_score(a_eg)
    print(f"    Exact + EG surrogate = {exact_score:.6f}")

    if exact_score < best_big_score:
        best_big_score, best_big = exact_score, a_eg

    # ------------------------------------------------------------------
    # 5️⃣ Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    print("\n--- Simulated‑annealing fine‑tuning ---")
    remaining = deadline - time.time()
    max_sa_iters = int(min(40_000, remaining * 30))
    a_sa = simulated_annealing_refine(best_big, deadline, max_iter=max_sa_iters,
                                      start_temp=0.06, cooling=0.9996, eps_factor=4e-4)

    sa_score = compute_score(a_sa)
    print(f"    Post‑SA surrogate = {sa_score:.6f}")
    if sa_score < best_big_score:
        best_big_score, best_big = sa_score, a_sa

    # ------------------------------------------------------------------
    # 6️⃣ Final official evaluation
    # ------------------------------------------------------------------
    final_candidate = best_big.tolist()
    try:
        final_val = evaluate_sequence(final_candidate)
        print(f"\nFinal official score = {final_val:.6f}")
        if final_val < best_score:
            best_score, best_seq = final_val, final_candidate
    except Exception as e:
        print(f"Evaluation failed on final candidate: {e}")

    # ------------------------------------------------------------------
    # 7️⃣ Guard – always return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_score = evaluate_sequence(best_seq)
        print(f"Returned fallback uniform sequence with score = {best_score:.6f}")

    print(f"\nSearch finished. Best official score = {best_score:.6f}")
    return best_seq
