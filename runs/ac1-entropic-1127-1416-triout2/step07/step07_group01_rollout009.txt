"""
Propose a sequence of non‑negative heights that minimises the
`evaluate_sequence` objective.

The search works in three phases and deliberately uses a modest
dimension (n≈1024) so that many optimisation runs fit into the
1000 s wall‑clock budget.

1. **Smooth max optimisation** – Adam‑accelerated gradient descent on a
   log‑sum‑exp surrogate of the ‑max‑ (β is annealed up to a large
   value, yielding an almost exact max).  Hundreds of independent
   random restarts are performed; the best exact score is kept.

2. **Projected sub‑gradient refinement** – a classic sub‑gradient
   method applied to the true max‑objective, again with simplex
   projection.

3. **Stochastic hill‑climb polishing** – small mass‑transfer moves
   accepted only when they improve the exact score; the step size
   adapts during the run.

All intermediate vectors are kept on the probability simplex
(non‑negative, sum = 1) so that the evaluation function’s scaling‑invariance
is respected.  The routine finally returns a plain Python list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using the real FFT (fast for large arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth max optimisation (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a soft‑max surrogate of  max(conv(a,a)) with Adam momentum.
    The routine returns the best *exact* score encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ------- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ------- smoothed objective ---------------------------------------
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        # log‑sum‑exp weights (numerically stable)
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:   # extremely unlikely, but protect against division by zero
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # ------- gradient of the soft‑max (cross‑correlation) -------------
        a_rev = a[::-1]
        # conv_rev has length 3n‑2 ; we need the central n entries
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ------- Adam update ----------------------------------------------
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasionally evaluate the *exact* score
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal entry (see analysis)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # small jitter every thousand iterations to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 2_000_000,
) -> tuple[list[float], float]:
    """
    Random local moves (swap or small mass transfer) with acceptance only
    when the exact score improves.  Step size adapts automatically.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # initial fraction of mass to move between two entries
    step_scale = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                       # become more conservative

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  All work respects the 1000 s time‑budget.
    Returns a plain Python list of floats (the sequence).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – the harness adds a tiny overhead after we return
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a manageable length – smaller n lets us do many restarts
    # ------------------------------------------------------------------
    n = 1024
    # if we have a lot of time left we can afford a slightly larger n
    if deadline - time.time() > 800:
        n = 2048
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # Helper – generate a random start on the simplex (dense or sparse)
    # ------------------------------------------------------------------
    def random_start():
        """Return a random probability vector of length `n`."""
        if np.random.rand() < 0.5:
            # dense – Dirichlet(α=1) i.e. uniform random on simplex
            a = np.random.rand(n)
            a = a / a.sum()
        else:
            # sparse – place mass on √n positions
            k = max(2, int(np.sqrt(n)))
            idx = np.random.choice(n, size=k, replace=False)
            a = np.zeros(n, dtype=float)
            a[idx] = 1.0 / k
            a = project_to_simplex(a)
        return a

    # ------------------------------------------------------------------
    # 2. Random‑restart smooth optimisation
    # ------------------------------------------------------------------
    # Reserve a little time for the refinement phases
    reserve_for_refine = 120.0
    while time.time() < deadline - reserve_for_refine:
        remaining = deadline - time.time() - reserve_for_refine
        # give each restart a modest time slice (but never less than 5 s)
        time_limit = time.time() + max(5.0, min(30.0, remaining * 0.5))

        a0 = random_start()
        a_opt, scr = smooth_gradient_optimize(
            a0,
            time_deadline=time_limit,
            max_iters=500_000,
            beta_start=5.0,
            beta_end=500.0,
            step0=1.0,
        )
        exact = compute_score(a_opt)
        if exact < best_val:
            best_val = exact
            best_seq = a_opt.tolist()
            print(f"New best after smooth phase: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print("\nSub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 10.0,
            max_iter=600_000,
            step0=0.05,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Improved by sub‑grad: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nHill‑climb polishing")
        polished_seq, polished_score = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=2_000_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback to a simple uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will call `evaluate_sequence`
    # again, but we report the score here for debugging).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
