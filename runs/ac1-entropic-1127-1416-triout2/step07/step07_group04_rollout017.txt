def evaluate_sequence(sequence):
    if not isinstance(sequence, list) -> inf

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s - 0.5
    n_target = 16384
    # generate initial candidate list of shapes
    best_seq = None
    best_score = float('inf')
    # baseline arcsine
    for eps in [0.0, 1e-8, 1e-6, 1e-4, 1e-2, 0.05, 0.25]:
        a = arcsine_eps(n_target, eps)
        score = compute_score(a)
        if score < best_score:
            best_score = score; best_seq = a
    # power shape with p in [0.3,0.4,0.5,0.6,0.7]
    for p in [0.3,0.4,0.5,0.6,0.7]:
        a = generate_power_shape(n_target, p, eps=1e-8)
        score = compute_score(a)
        if score < best_score:
            best_score = score; best_seq = a

    # mixture arcsine+uniform
    arc = arcsine_eps(n_target, 0) # no eps
    uni = np.full(n_target, 1.0/n_target)
    for w in np.linspace(0,1,11):
        a = w*arc + (1-w)*uni
        a = a / a.sum()
        score = compute_score(a)
        if score < best_score:
            best_score = score; best_seq = a

    # Random perturbations (hill climbing)
    rng = np.random.default_rng(seed)
    max_trials = 2000
    while time.time() < deadline - 5 and max_trials>0:
        # propose by adding gaussian noise
        noise = rng.normal(scale=0.01, size=n_target)
        a_candidate = best_seq + noise
        a_candidate = np.maximum(a_candidate, 0)
        a_candidate = a_candidate / a_candidate.sum()
        score = compute_score(a_candidate)
        if score < best_score:
            best_score = score; best_seq = a_candidate
        max_trials -= 1

    # Gradient refinement using Adam with smooth max
    remaining = deadline - time.time()
    if remaining > 10:
        max_iters = int(min(150_000, remaining * 30))
        best_seq = adam_optimize_smooth_max(best_seq, deadline,
                                            max_iters=max_iters,
                                            tau_start=0.5,
                                            tau_end=300.0,
                                            base_lr=0.5)

        best_score = compute_score(best_seq)

    # Exact polishing subgradient + eg if time left
    remaining = deadline - time.time()
    if remaining > 5:
        max_iter_sub = int(min(200_000, remaining * 30))
        best_seq = subgradient_descent_exact(best_seq, deadline,
                                             max_iter=max_iter_sub,
                                             step0=0.05)
        best_score = compute_score(best_seq)

    remaining = deadline - time.time()
    if remaining > 2:
        max_iter_eg = int(min(150_000, remaining * 30))
        best_seq = eg_refine(best_seq, deadline,
                             max_iter=max_iter_eg,
                             eta0=0.2)

        best_score = compute_score(best_seq)

    # final official evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final official score = {final_score:.6f}")
    return best_seq.tolist()

def arcsine_eps(n, eps):
    i = np.arange(n, dtype=float)
    denom = np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = 1.0 / denom

"""
Hybrid shape‑search optimizer for the step‑function bound.

The algorithm proceeds in several cheap, time‑aware stages:

1.  Choose a large discretisation (default 16384 points) and generate a
    few analytically‑motivated families (arcsine, power‑law, beta, and
    mixtures with the uniform distribution).  The exact score
    `2·n·max(conv(a,a))` is evaluated with an FFT‑based convolution; the
    best candidate is retained.

2.  Perform a short stochastic hill‑climb (Gaussian noise + simplex
    projection) to explore the neighbourhood of the current best.

3.  Refine the candidate with a smooth‑max Adam optimiser.  The temperature
    τ is linearly increased from 0.5 to a large final value (≈ 300) so that
    the smoothed objective approaches the true max.

4.  Polish the result with a projected Euclidean sub‑gradient descent
    followed by an exponentiated‑gradient (mirror‑descent) phase.
    Both phases respect a decaying step‑size ∝ 1/√t.

All stages stop automatically when the global 1000 s wall‑clock budget is
exhausted, guaranteeing that a non‑empty sequence is returned.  The final
sequence is normalised to sum = 1, which makes the evaluation function
scale‑invariant.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (fast for the lengths used here).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact surrogate of the official score for a probability vector
    (Σa = 1).  The surrogate equals the value returned by
    ``evaluate_sequence`` for the same vector.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))

# Shape generators

def arcsine_eps(n: int, eps: float = 0.0) -> np.ndarray:
    """Discrete arcsine density on a grid of size n, with optional epsilon."""
    i = np.arange(n, dtype=float)
    denom = np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = 1.0 / denom
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-9) -> np.ndarray:
    """Generalised arcsine shape: ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-9) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float, eps: float = 0.0) -> np.ndarray:
    """
    Mixture weight_arc·arcsine + (1‑weight_arc)·uniform.
    The result is normalised to sum = 1.
    """
    arc = arcsine_eps(n, eps)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

# Optimisers

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 300.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth‐max approximation of max(conv(a,a)).
    The optimisation variable is the log‑its ``w`` (softmax → ``a``),
    guaranteeing non‑negativity and unit sum.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)    # logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a (stable)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)                # length L

        # smooth‑max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp                # shape (L,)

        # gradient w.r.t. a : 2·conv(a, weights_rev)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)         # length n+L-1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        # renormalise (project onto simplex under KL geometry)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine runs a fast parameter sweep, a light stochastic
    hill‑climb and three optimisation phases (smooth‑max Adam,
    Euclidean sub‑gradient, exponentiated‑gradient).  All stages respect a
    global 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0       # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback: a previously discovered construction (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"[fallback] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 0 – choose discretisation size (large enough for sub‑percent precision)
    # ------------------------------------------------------------------
    n_target = 16384
    # If the deadline is already tight we fall back to a smaller grid.
    remaining = deadline - time.time()
    if remaining < 200.0:
        n_target = 8192
    print(f"\n=== Stage 0: grid size n = {n_target} ===")

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric sweep
    # ------------------------------------------------------------------
    print("\n--- Stage 1: parametric sweep ---")
    candidates = []

    # arcsine with several epsilons
    for eps in (0.0, 1e-8, 1e-6, 1e-4, 1e-2, 0.05, 0.25):
        a = arcsine_eps(n_target, eps)
        candidates.append(a)

    # power‑law family (p controls the tail)
    for p in (0.3, 0.4, 0.5, 0.6, 0.7):
        a = generate_power_shape(n_target, p, eps=1e-9)
        candidates.append(a)

    # symmetric beta family (α = β)
    for alpha in (0.4, 0.5, 0.6, 0.8, 1.0):
        a = generate_beta_shape(n_target, alpha, alpha, eps=1e-9)
        candidates.append(a)

    # mixture of arcsine and uniform
    arc = arcsine_eps(n_target, eps=0.0)
    uni = np.full(n_target, 1.0 / n_target)
    for w in np.linspace(0.0, 1.0, 11):
        a = w * arc + (1.0 - w) * uni
        a = a / a.sum()
        candidates.append(a)

    # evaluate all candidates
    for a in candidates:
        sc = compute_score(a)
        if sc < best_val:
            best_val, best_seq = sc, a.tolist()
            print(f"  new best from sweep: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – stochastic hill‑climb (few thousand random perturbations)
    # ------------------------------------------------------------------
    print("\n--- Stage 2: stochastic hill‑climb ---")
    a_cur = np.array(best_seq, dtype=float)
    max_trials = 3000
    while time.time() < deadline - 5 and max_trials > 0:
        # relative noise magnitude adapts to current vector scale
        sigma = 0.02 * np.mean(a_cur[a_cur > 0]) if np.any(a_cur > 0) else 0.02
        noise = rng.normal(scale=sigma, size=n_target)
        a_new = a_cur + noise
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total == 0.0:
            a_new = np.full_like(a_new, 1.0 / n_target)
        else:
            a_new /= total
        sc = compute_score(a_new)
        if sc < best_val:
            best_val, best_seq, a_cur = sc, a_new.tolist(), a_new
            print(f"  improvement via hill‑climb: score={best_val:.6f}")
        max_trials -= 1

    # ------------------------------------------------------------------
    # Stage 3 – smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    print("\n--- Stage 3: smooth‑max Adam ---")
    remaining = deadline - time.time()
    if remaining > 10.0:
        max_iters = int(min(150_000, remaining * 30))
        a_cur = np.array(best_seq, dtype=float)
        a_cur = adam_optimize_smooth_max(
            a_cur,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=300.0,
            base_lr=0.5,
        )
        best_val = compute_score(a_cur)
        best_seq = a_cur.tolist()
        print(f"  Adam finished: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – exact polishing (sub‑gradient then EG)
    # ------------------------------------------------------------------
    print("\n--- Stage 4: exact polishing ---")
    remaining = deadline - time.time()
    if remaining > 10.0:
        max_iter_sub = int(min(200_000, remaining * 20))
        a_cur = subgradient_descent_exact(np.array(best_seq, dtype=float),
                                          deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.05)
        best_val = compute_score(a_cur)
        best_seq = a_cur.tolist()
        print(f"  after sub‑gradient: score={best_val:.6f}")

    remaining = deadline - time.time()
    if remaining > 5.0:
        max_iter_eg = int(min(150_000, remaining * 20))
        a_cur = eg_refine(np.array(best_seq, dtype=float),
                          deadline,
                          max_iter=max_iter_eg,
                          eta0=0.20)
        best_val = compute_score(a_cur)
        best_seq = a_cur.tolist()
        print(f"  after EG polish: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"\nFinal official score = {final_score:.6f}")
        # keep the official score as the benchmark for return
        if final_score < best_val:
            best_val = final_score
    except Exception as exc:
        print(f"Official evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print(f"Returned trivial uniform fallback: score={best_val:.6f}")

    print(f"\nSearch completed. Best score = {best_val:.6f}")
    return best_seq
