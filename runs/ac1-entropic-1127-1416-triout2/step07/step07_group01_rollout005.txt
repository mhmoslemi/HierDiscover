    n_local = n  # defined from outer.
    # define closure
    def fun_and_grad(y):
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        conv = convolution_fft(a, a)
        maxc = conv.max()
        diff = conv - maxc
        exp_beta = np.exp(beta * diff)
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n_local-1 : n_local-1 + n_local]

def hill_climb(seq, deadline, max_steps=500000):
   a = np.array(seq, dtype=float)
   n = a.shape[0]
   best_a = a.copy()
   best_score = compute_score(best_a)  # uses sum=1
   step_scale = 0.1

   while max_steps>0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        # project onto simplex
        cand = project_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97
        max_steps -= 1
   return best_a.tolist(), best_score

"""
Propose a non‑negative step‑function (probability vector) that minimises
`evaluate_sequence`.  The algorithm works on the simplex via a soft‑max
parameterisation and uses L‑BFGS optimisation on a smooth
log‑sum‑exp approximation of the max‑convolution.  A low‑β stage finds a
coarse shape, a high‑β stage refines it, and a final stochastic hill‑climb
polishes the result.  The search adapts the sequence length to the
available wall‑clock budget (up to 16384 entries) and respects the 1000 s
limit.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_y(y: np.ndarray) -> np.ndarray:
    """Map an unconstrained vector y to the simplex via soft‑max."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(y: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smoothed objective
        L(y) = max(conv) + (1/β)·log Σ exp(β·(conv - max(conv)))
    where conv = a*a and a = softmax(y).  Gradient is expressed w.r.t. y.
    """
    a = softmax_from_y(y)                     # a ≥ 0, sum = 1
    conv = convolution_fft(a, a)               # length 2n‑1
    maxc = conv.max()

    # log‑sum‑exp part (weights for the soft max)
    diff = conv - maxc                         # ≤ 0, max = 0
    exp_beta = np.exp(beta * diff)
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp                     # probability vector, length 2n‑1

    # gradient w.r.t. a (2·(a_rev * w) evaluated on the central n positions)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]   # size n

    # push the gradient through the soft‑max (Jacobian of softmax)
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    # smoothed value
    L = maxc + (1.0 / beta) * np.log(sum_exp)

    return L, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the simplex (via the soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


def optimize_lbfgs(y0: np.ndarray, n: int, beta: float,
                   max_iter: int, deadline: float) -> np.ndarray:
    """
    Runs L‑BFGS on the smoothed objective for a fixed β.
    Returns the probability vector a = softmax(y_opt).
    """
    # SciPy will call `smooth_obj_and_grad(y, n, beta)` and expects (f, g)
    options = {"maxiter": max_iter, "disp": False}
    result = scipy.optimize.minimize(
        smooth_obj_and_grad,
        y0,
        args=(n, beta),
        method="L-BFGS-B",
        jac=True,
        options=options,
    )
    y_opt = result.x
    a_opt = softmax_from_y(y_opt)          # already normalised
    return a_opt


# ----------------------------------------------------------------------
# Small stochastic hill‑climb (mass transfers) for final polishing
# ----------------------------------------------------------------------


def hill_climb_polish(seq, deadline, max_steps=300_000):
    """
    Tiny stochastic moves (swap or transfer a fraction of mass) projected
    back onto the simplex.  Returns a (list, score) pair.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
