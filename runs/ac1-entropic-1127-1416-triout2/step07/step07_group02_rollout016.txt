"""
Propose a step‑function (non‑negative simplex vector) that minimizes
``evaluate_sequence``.  The algorithm consists of three stages:

1.  **Fast analytic sweep** –  a wide family of Beta‑type shapes
    (including the arcsine distribution) is sampled on several lengths.
    The cheap surrogate ``fast_score`` (exact objective for normalized
    vectors) is used to pick the most promising candidate.

2.  **Continuation Adam on the smooth‑max** –  the selected candidate is
    re‑parameterised by a soft‑max (so the simplex constraint is built‑in)
    and a smooth approximation of the max‑convolution
    ``L_β(a)=log‑sum‑exp(β·conv(a,a))`` is minimised.
    The temperature β is increased from 5 to a large value (≈1500) while
    Adam adapts its step.

3.  **Exact sub‑gradient polishing** –  a projected sub‑gradient (mirror)
    descent on the true objective ``2·n·max(conv)/sum²`` followed by a
    final Euclidean projection step (``subgradient_descent_exact``) yields
    the final answer.

All phases respect the global time budget (default 1000 s) and a
fallback uniform sequence is returned if nothing else succeeds.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast exact surrogate (scale‑invariant) used for quick screening
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised simplex vector:
        2·n·max(conv(a,a)) / (sum(a)²)
    (no clipping, no safety checks – cheap and accurate for our uses).
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families of candidate shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Discretised arcsine density  f(x) ∝ 1/√(x(1‑x)).
    The tiny epsilon prevents division by zero.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-8) -> np.ndarray:
    """
    Discrete Beta‑type density  f(i) ∝ (i+eps)^{α‑1}·(n‑1‑i+eps)^{β‑1}.
    Parameters α,β>0 control the mass near the left/right boundaries.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    Power‑law variant  f(i) ∝ ((i+½)(n‑½‑i)+eps)^{‑p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx=1} (the probability simplex)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (the same routine used in the previous SOTA)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.06,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on
        f(a) = 2·n·max(conv(a,a)) / (sum(a)²) .
    Uses the exact max‑convolution sub‑gradient.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # indices where the maximum is attained (numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d}  score≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max optimisation with Adam (continuation on β)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 1500.0,
    ramp_iters: int = 80_000,
    lr: float = 0.03,
) -> np.ndarray:
    """
    Minimise the smooth surrogate L_β(a)=log‑sum‑exp(β·conv(a,a))
    while keeping a on the simplex by the soft‑max map.
    β is increased linearly from ``beta0`` to ``beta_max`` (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights (log‑sum‑exp trick)
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of the smooth‑max w.r.t. a: 2·(a ★ rev(w_norm))
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    Adam it={t:,d}  β={beta:.1f}  score≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) using the exact max‑subgradient
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.2,
    decay: float = 0.9995,
) -> np.ndarray:
    """
    KL‑mirror descent (exponential‑gradient) on the exact objective.
    The iterate always stays on the simplex.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        # occasional progress report
        if int(step * 1e6) % 100_000 == 0:
            print(f"    Mirror step={step:.6f}  score≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine performs a cheap analytic sweep, a long Adam continuation,
    a mirror‑descent polish and finally exact sub‑gradient descent.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # a tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0 – fallback to a previously stored good sequence (if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1 – quick analytic sweep on a few lengths
    # --------------------------------------------------------------
    sweep_lengths = [4096, 2048, 1024]
    sweep_candidates: List[Tuple[float, np.ndarray]] = []

    sweep_deadline = time.time() + 30.0          # spend at most ~30 s here
    print("\n=== Analytic sweep ===")
    for n in sweep_lengths:
        if time.time() >= sweep_deadline:
            break
        print(f"\nScanning length {n}")

        candidates: List[np.ndarray] = []

        # core families
        candidates.append(arcsine_initial(n))
        candidates.append(np.full(n, 1.0 / n))

        # beta‑family grid (α,β) – moderate resolution
        alphas = np.linspace(0.2, 4.0, 6)
        betas = np.linspace(0.2, 4.0, 6)
        for a in alphas:
            for b in betas:
                candidates.append(generate_beta_shape(n, a, b, eps=1e-8))

        # power‑law family
        for p in np.linspace(0.2, 2.2, 7):
            candidates.append(generate_power_shape(n, p, eps=1e-8))

        # random Dirichlet draws for extra diversity
        for _ in range(5):
            rand = rng.random(n)
            rand = rand / rand.sum()
            candidates.append(rand)

        # evaluate cheaply and retain the best few
        scored = [(fast_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        for sc, vec in scored[:3]:          # keep three per length
            sweep_candidates.append((sc, vec))
            print(f"  length {n} candidate score ≈{sc:.6f}")

    # --------------------------------------------------------------
    # 2 – pick the overall best candidate from the sweep
    # --------------------------------------------------------------
    if not sweep_candidates:
        # Very unlikely, but keep a uniform fallback
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print("\nNo candidates found in sweep – using uniform fallback.")
        return best_seq

    sweep_candidates.sort(key=lambda x: x[0])
    seed_score, seed_vec = sweep_candidates[0]
    print(f"\nBest analytic seed: length {seed_vec.size}, score≈{seed_score:.6f}")

    # --------------------------------------------------------------
    # 3 – intensive Adam optimisation on the selected seed
    # --------------------------------------------------------------
    n = seed_vec.size
    # allocate the bulk of the remaining time to Adam
    remaining = deadline - time.time()
    adam_deadline = time.time() + max(10.0, 0.6 * remaining)

    print("\n=== Adam continuation (smooth‑max) ===")
    z0 = np.log(seed_vec + 1e-12)   # log‑softmax initialisation
    a_adam = smooth_adam_optimize(
        z0,
        adam_deadline,
        n=n,
        beta0=5.0,
        beta_max=1500.0,
        ramp_iters=80_000,
        lr=0.03,
    )
    try:
        val_adam = evaluate_sequence(a_adam.tolist())
        print(f"   Adam result official score = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.tolist()
    except Exception as e:
        print(f"   Adam evaluation failed: {e}")

    # --------------------------------------------------------------
    # 4 – Mirror‑descent refinement (exact max sub‑gradient)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mir_deadline = deadline - 1.0
        print("\n=== Mirror‑descent refinement ===")
        a_mirror = mirror_descent_optimize(
            np.array(best_seq, dtype=float),
            mir_deadline,
            step0=0.2,
            decay=0.9995,
        )
        try:
            val_mirror = evaluate_sequence(a_mirror.tolist())
            print(f"   Mirror‑refined official score = {val_mirror:.6f}")
            if val_mirror < best_val:
                best_val, best_seq = val_mirror, a_mirror.tolist()
                print("*** Updated best after mirror step ***")
        except Exception as e:
            print(f"   Mirror evaluation failed: {e}")

    # --------------------------------------------------------------
    # 5 – Final Euclidean sub‑gradient polish (exact objective)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\n=== Final Euclidean sub‑gradient polish ===")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=250_000,
            step0=0.06,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final evaluation failed: {e}")

    # --------------------------------------------------------------
    # 6 – safety net – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
