assistantfinal**Solution Overview**

The goal is to minimise  

\[
E(a)=\frac{2\,n\;\max(a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the discrete linear convolution of the non‑negative step heights `a`.  
Because the objective is *scale invariant* we can fix  

\[
\sum_i a_i = 1,
\]

so the problem reduces to  

\[
\min_{a\ge 0,\;\sum a=1}\; 2\,n\;\max(a*a)\;.
\]

The function is non‑convex because the convolution is quadratic, therefore we use a **multi‑start smooth‑max Adam optimiser** followed by **exact nonsmooth polishing** (projected sub‑gradient + exponentiated‑gradient) and a tiny stochastic local‑search at the end.  
All phases respect the global 1000 s wall‑clock budget.

Key ideas
---------

* **Warm‑starts** – a handful of analytically‑motivated shapes (arcsine, several Beta‑distributions, uniform) are used as seeds.
* **Smooth‑max Adam** – we minimise a differentiable approximation of the max‑convolution  

  \[
  \operatorname{SM}_\tau(x)=\frac1{\tau}\log\Bigl(\sum_i e^{\tau\,x_i}\Bigr),
  \]

  with a linear temperature schedule `τ∈[0.5, τ_end]`.  
  The optimisation variable is a *logits* vector `w`; `a = softmax(w)` automatically satisfies the simplex constraints.
* **Very long Adam runs** – the temperature is pushed up to `τ_end=800` and up to 300 k iterations are performed (still well inside the 1000 s limit).  
  This concentrates the gradient on the true maximal entries of the convolution.
* **Exact polishing** – after the smooth phase we switch to true sub‑gradients of the max‑convolution and perform
  * Euclidean projected sub‑gradient descent,
  * Mirror‑descent with exponentiated‑gradient updates.
* **Local stochastic refinement** – a quick random‑perturbation + projection loop explores the neighbourhood of the best point for any remaining time.
* **Time‑aware budgeting** – before every major loop we recompute the remaining time and shrink iteration limits accordingly, guaranteeing termination before the deadline.

The implementation below follows exactly this plan.  All helper functions are top‑level, there are no closures or file I/O, and the only external call is the supplied `evaluate_sequence`.

```python
"""
Multi‑start smooth‑max Adam optimiser + exact polishing.

* warm‑starts from arcsine, several Beta‑shapes and the uniform density,
* long Adam optimisation with a very high temperature (τ up to 800),
* projected Euclidean sub‑gradient descent,
* exponentiated‑gradient (mirror descent) polishing,
* a final stochastic local search.

The routine respects a global 1000 s budget and returns the best
non‑negative sequence it discovered (sum = 1).
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discretised arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-6) -> np.ndarray:
    """
    Beta‑type shape:
        (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1},
    normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (logits → softmax)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 800.0,
                             base_lr: float = 0.8,
                             noise_scale: float = 0.0) -> np.ndarray:
    """
    Adam optimiser on the smooth approximation of max(conv).

    Parameters
    ----------
    a_init : np.ndarray
        Starting probability vector (sum = 1, non‑negative).
    deadline : float
        Unix time at which the optimisation must stop.
    max_iters : int
        Upper bound on the number of iterations.
    tau_start, tau_end : float
        Linear temperature schedule for the smooth‑max.
    base_lr : float
        Base learning‑rate (scaled by 1/√t).
    noise_scale : float
        If >0, isotropic Gaussian noise is added to the probabilities
        every 10 000 iterations (helps escape shallow basins).

    Returns
    -------
    np.ndarray
        Optimised probability vector (still on the simplex).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    rng = np.random.default_rng(0)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # logits → probabilities (softmax)
        w_shift = w - np.max(w)               # for numeric stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # occasional isotropic noise to improve exploration
        if noise_scale > 0.0 and (it % 10_000) == 0:
            sigma = noise_scale * (1.0 - it / max_iters)
            a = a + rng.normal(scale=sigma, size=n)
            a = np.clip(a, 0.0, None)
            a = project_to_simplex(a)
            w = np.log(a + 1e-12)

        # ---------- gradient of smooth‑max ----------
        conv = convolution_fft(a, a)               # length L
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp             # softmax‐weights over the L entries

        # gradient w.r.t. a : 2·conv(a, rev(weights))
        rev_weights = weights[::-1]                # length L
        conv_rev = convolution_fft(a, rev_weights)  # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (learning‑rate decays like 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # (optional) progress report
        if (it % 50_000) == 0:
            surrogate = compute_score(np.exp(w - np.max(w)) / np.exp(w - np.max(w)).sum())
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (nonsmooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.06) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        conv_rev = convolution_fft(a, rev_mask)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polishing (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        conv_rev = convolution_fft(a, rev_mask)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

# Stochastic local refinement (optional final phase)
# ----------------------------------------------------------------------


def stochastic_local_refine(a_best: np.ndarray,
                            deadline: float,
                            max_trials: int = 2000) -> np.ndarray:
    """
    Simple random perturbation + projection loop.
    Keeps the best point seen under the smooth surrogate (compute_score).
    """
    n = a_best.size
    best_a = a_best.copy()
    best_score = compute_score(best_a)
    rng = np.random.default_rng(1)

    trial = 0
    while time.time() < deadline and trial < max_trials:
        # scale of perturbation slowly shrinks
        remaining = deadline - time.time()
        sigma = 0.02 * (remaining / (deadline - start_time + 1e-9)) * (1.0 / np.sqrt(trial + 1))
        cand = best_a + rng.normal(scale=sigma, size=n)
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)
        score = compute_score(cand)
        if score < best_score:
            best_score = score
            best_a = cand
        trial += 1
    return best_a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.

    Strategy
    --------
    1.  Build several analytic seeds (arcsine, uniform, Beta‑shapes).
    2.  Run a short Adam optimisation on each seed; keep the best.
    3.  Continue a *long* Adam run on that best seed with a very high
        temperature (τ up to 800) – this tightly approximates the true
        max‑convolution.
    4.  Exact polishing:
        * projected Euclidean sub‑gradient descent,
        * exponentiated‑gradient (mirror‑descent) refinement.
    5.  If time permits, perform a tiny stochastic local search.
    6.  Return the best sequence found (as a list of floats).
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Seed generation (all seeds are normalised to sum = 1)
    # ------------------------------------------------------------------
    n = 2048                       # final resolution (fits in the time budget)
    seeds: List[np.ndarray] = []

    # deterministic seeds
    seeds.append(arcsine_initial(n))
    seeds.append(np.full(n, 1.0 / n))                # uniform

    # several Beta‑shapes – they cover a wide variety of “U‑shaped” vs
    # “bell‑shaped” behaviours.
    for a, b in [(0.3, 0.3), (0.4, 0.4), (0.6, 0.6),
                 (0.7, 0.7), (0.4, 0.6), (0.6, 0.4),
                 (0.5, 0.9), (0.9, 0.5)]:
        seeds.append(generate_beta_shape(n, a, b, eps=1e-6))

    # arcsine‑uniform mixtures
    for w in [0.2, 0.5, 0.8]:
        seeds.append(generate_mixture_shape(n, weight_arc=w))

    # ------------------------------------------------------------------
    # 2. Warm‑start Adam on every seed (short run, cheap)
    # ------------------------------------------------------------------
    print("\n--- Phase 0: short warm‑start Adam on multiple seeds ---")
    warm_iters = 60_000
    for idx, a0 in enumerate(seeds, 1):
        if time.time() >= deadline - 5.0:
            break
        print(f"   Seed {idx}/{len(seeds)}")
        a_opt = adam_optimize_smooth_max(
            a0,
            deadline,
            max_iters=warm_iters,
            tau_start=0.5,
            tau_end=120.0,
            base_lr=0.6,
            noise_scale=0.0,
        )
        s = compute_score(a_opt)
        print(f"       surrogate after warm‑start = {s:.6f}")
        if s < best_val:
            best_val = s
            best_seq = a_opt.tolist()
            print("       *** new best seed ***")

    # ------------------------------------------------------------------
    # 3. Prolonged Adam optimisation on the current best seed
    # ------------------------------------------------------------------
    print("\n--- Phase 1: long Adam optimisation (high temperature) ---")
    if not best_seq:
        # fallback to a deterministic seed if everything failed
        best_seq = arcsine_initial(n).tolist()
        best_val = compute_score(np.array(best_seq))

    a_current = np.array(best_seq, dtype=float)

    remaining = deadline - time.time()
    max_iters_long = int(min(300_000, remaining * 30))
    a_current = adam_optimize_smooth_max(
        a_current,
        deadline,
        max_iters=max_iters_long,
        tau_start=0.5,
        tau_end=800.0,          # very high temperature → accurate max
        base_lr=0.9,
        noise_scale=5e-4,       # tiny stochastic kicks help escape plateaus
    )
    current_score = compute_score(a_current)
    print(f"   Surrogate after long Adam = {current_score:.6f}")
    if current_score < best_val:
        best_val = current_score
        best_seq = a_current.tolist()
        print("   *** new global best after long Adam ***")

    # ------------------------------------------------------------------
    # 4. Exact polishing (sub‑gradient + exponentiated‑gradient)
    # ------------------------------------------------------------------
    print("\n--- Phase 2: exact polishing (sub‑gradient) ---")
    remaining = deadline - time.time()
    max_sub_iter = int(min(400_000, remaining * 30))
    a_current = subgradient_descent_exact(
        a_current,
        deadline,
        max_iter=max_sub_iter,
        step0=0.05,
    )
    score_sub = compute_score(a_current)
    print(f"   After sub‑gradient: surrogate = {score_sub:.6f}")
    if score_sub < best_val:
        best_val = score_sub
        best_seq = a_current.tolist()
        print("   *** new best after sub‑gradient ***")

    # Exponentiated‑gradient refinement (if time permits)
    if time.time() < deadline - 2.0:
        print("\n--- Phase 2b: exponentiated‑gradient refinement ---")
        remaining = deadline - time.time()
        max_eg_iter = int(min(250_000, remaining * 30))
        a_current = eg_refine(
            a_current,
            deadline,
            max_iter=max_eg_iter,
            eta0=0.22,
        )
        score_eg = compute_score(a_current)
        print(f"   After EG: surrogate = {score_eg:.6f}")
        if score_eg < best_val:
            best_val = score_eg
            best_seq = a_current.tolist()
            print("   *** new best after EG ***")

    # ------------------------------------------------------------------
    # 5. Tiny stochastic local search (optional, uses any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n--- Phase 3: stochastic local refinement ---")
        a_current = stochastic_local_refine(
            np.array(best_seq, dtype=float),
            deadline,
            max_trials=1200,
        )
        score_local = compute_score(a_current)
        print(f"   After local search: surrogate = {score_local:.6f}")
        if score_local < best_val:
            best_val = score_local
            best_seq = a_current.tolist()
            print("   *** new best after local search ***")

    # ------------------------------------------------------------------
    # 6. Final official evaluation
    # ------------------------------------------------------------------
    try:
        official = evaluate_sequence(best_seq)
        print(f"\nFinal official score = {official:.6f}")
        # In case the official score is *slightly* different from the surrogate
        # we keep the sequence that gave the smallest official value.
        if official < best_val:
            best_val = official
    except Exception as exc:
        print(f"Official evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        print("Fallback to uniform distribution.")
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
```
