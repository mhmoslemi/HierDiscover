def active_subgrad_adam(a0, time_deadline, max_iter=200_000, step0=0.1, tol_rel=1e-6, jitter_freq=1000, jitter_scale=0.003):

   a = a0.copy()

   m = np.zeros_like(a); v = np.zeros_like(a)

   best_a = a.copy(); best_sc = compute_score(a); n = len(a); idx_start = 2*n - 2

   for t in range(1,max_iter+1):

        if time.time()>time_deadline: break

        conv = convolution_fft(a, a)   # shape 2n-1

        M = np.max(conv)

        # active mask
        eps = max(1e-12, M * tol_rel)
        active_idx = np.where(conv >= M - eps)[0]
        mask = np.zeros_like(conv)
        if active_idx.size > 0:
            mask[active_idx] = 1.0 / active_idx.size
        else:
            mask[np.argmax(conv)] = 1.0

        # gradient
        conv_rev = convolution_fft(a, mask[::-1])
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam
        beta1=0.9; beta2=0.999; eps_adam=1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        step = step0 / np.sqrt(t)   # maybe / (np.sqrt(t)+1)?
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

        if t % jitter_freq == 0:
            # small random noise
            jitter = np.random.randn(n) * (jitter_scale * step0 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

   return best_a, best_sc

"""
Multiphase optimizer for the self‑convolution peak minimisation problem.

Key ideas
---------
* Random or shape‑biased initial vectors on the probability simplex.
* Phase 1 – flatten the variance of the convolution (quadratic objective) via
  Adam.  This yields a very flat autocorrelation.
* Phase 2 – minimise a smooth soft‑max surrogate (ε·log∑exp) with Adam,
  progressively shrinking ε.
* Phase 3 – “active‑set” sub‑gradient descent: instead of using only the
  single maximal shift we average the gradients of all shifts whose value is
  within a relative tolerance of the current maximum.  Adam is used for an
  adaptive step‑size.
* Phase 4 – a tiny linear‑programming “LP‑boost” (only for modest lengths
  ≤ 1024).  For a fixed f we solve
        maximise   ∑b_j
        subject to   f⋆b ≤ M   and   b ≥ 0
  where M = max(f⋆f).  The solution b is mixed with the current f
  (convex combination) and projected back onto the simplex.  This step
  often produces a noticeable drop in the peak.
* Phase 5 – simulated‑annealing hill‑climbing with mass‑transfer moves.
* Phase 6 – symmetry averaging (a←(a+rev(a))/2) which empirically reduces
  the peak further.

All phases respect the 1000 s time budget and keep the best sequence seen.
The routine returns a plain list of floats ready for the harness.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisation
# ----------------------------------------------------------------------


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


def biased_U_shape(n: int) -> np.ndarray:
    """U‑shaped (low in the centre, high at the edges) distribution."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    w = (i - centre) ** 2 + 1e-3
    return w / np.sum(w)


def biased_V_shape(n: int) -> np.ndarray:
    """V‑shaped (high in the centre, low at the edges) distribution."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    w = np.abs(i - centre) + 1e-3
    return w / np.sum(w)


# ----------------------------------------------------------------------
# Phase 1 – variance flattening (L2) using Adam
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 3000,
    step0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise   ∑ (conv_k - mean(conv))²   (i.e. variance of the convolution)
    with Adam.  The gradient of this quartic loss is
        g = 4 * Σ_j a_j * (conv_{i+j} - μ)
    where μ = mean(conv).  After each step we project onto the simplex.
    Returns the best vector seen (exact score).
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)                 # length 2n‑1
        mu = np.mean(conv)                           # = sum_a**2 / (2n‑1)
        diff = conv - mu

        # gradient = 4 * Σ_j a_j * diff_{i+j}
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)      # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 2 – smooth max surrogate (ε·log‑sum‑exp) with Adam
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 8000,
    step0: float = 0.5,
    eps_factor: float = 0.25,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution:
        f(a) = ε·log∑_k exp(conv_k/ε)
    Gradient is 2·Σ_j a_j·w_{i+j} where w is the soft‑max over conv/ε.
    Adam is used for adaptive steps.
    Returns the best vector (exact score) observed.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε shrinks but never below 1 % of maxc
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)           # soft‑max, ∑ w = 1

        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to help escape flat regions
        if t % 400 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc

def active_subgrad_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.15,
    tol_rel: float = 1e-4,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    jitter_every: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective 2·n·max(a⊛a).
    The sub‑gradient is the average over all shifts whose value is within
    `tol_rel` of the current maximum (active set).  Adam is used for
    adaptivity.
    Returns the best vector found.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        M = np.max(conv)

        # active set mask
        eps_act = max(1e-12, M * tol_rel)
        active_idx = np.where(conv >= M - eps_act)[0]
        mask = np.zeros_like(conv)
        if active_idx.size == 0:
            mask[np.argmax(conv)] = 1.0
        else:
            mask[active_idx] = 1.0 / active_idx.size

        # gradient = 2 * (a ⋆ mask_rev) restricted to first n entries
        conv_rev = convolution_fft(a, mask[::-1])
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % jitter_every == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – tiny LP‑boost (only for modest n)
# ----------------------------------------------------------------------


def lp_boost(
    a: np.ndarray,
    time_deadline: float,
    mix_t: float = 0.2,
    max_steps: int = 5,
) -> tuple[np.ndarray, float]:
    """
    For the current vector a (length n) solve a tiny LP:
        maximise   ∑ b_j
        subject to  (a⋆b)_k ≤ M  ∀k,   b ≥ 0,
    where M is the current max(a⋆a).  The solution b is mixed back into a
    with weight `mix_t` and projected onto the simplex.  Only invoked
    for n ≤ 1024 (larger n would make the LP too heavy).
    Returns the updated vector and its (exact) score.
    """
    n = len(a)
    if n > 1024:
        return a, compute_score(a)

    M = np.max(convolution_fft(a, a))

    # build the Toeplitz convolution matrix A (shape (2n‑1, n))
    col = np.concatenate([a, np.zeros(n - 1)])
    row = np.zeros(n)
    row[0] = a[0]
    A = toeplitz(col, row)[: 2 * n - 1, :]

    # linear programme: minimise -∑b  (i.e. maximise ∑b)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    a_curr = a.copy()
    for _ in range(max_steps):
        if time.time() >= time_deadline:
            break
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break
        b0 = res.x
        # convex combination and project back
        a_new = (1.0 - mix_t) * a_curr + mix_t * b0
        a_new = project_to_simplex(a_new)
        a_curr = a_new
        # recompute M for the next LP (optional, cheap)
        M = np.max(convolution_fft(a_curr, a_curr))
        b_ub = np.full(2 * n - 1, M)

    return a_curr, compute_score(a_curr)

# Phase 5 – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 250_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with simulated annealing.
    Moves:
        * swap two positions
        * transfer a fraction of mass from i → j
    Accepted if it improves the exact score, otherwise with probability
    exp(-Δ / T) where T cools linearly.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        if step % 5000 == 0:
            # occasional larger perturbation
            rnd = random_dense_vector(n) * 0.02
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Produce a symmetric version (a + reverse(a))/2 and re‑project.
    Symmetric vectors often have flatter autocorrelations.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a
    plain Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback: previously known construction (if present)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    outer_iter = 0

    while time.time() < deadline - 5.0:
        outer_iter += 1
        n = int(np.random.choice(lengths))

        # ---- 1. initialise -------------------------------------------------
        init_kind = np.random.choice(["dirichlet", "Ushape", "Vshape", "mix"])
        if init_kind == "dirichlet":
            a0 = random_dense_vector(n)
        elif init_kind == "Ushape":
            a0 = biased_U_shape(n)
        elif init_kind == "Vshape":
            a0 = biased_V_shape(n)
        else:  # mix of random and U‑shape
            a0 = 0.5 * random_dense_vector(n) + 0.5 * biased_U_shape(n)
            a0 = a0 / a0.sum()

        # quick filter – keep if it already out‑performs the incumbent
        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[{outer_iter}] init ({init_kind}) n={n}, score={best_score:.6f}")

        # ---- 2. flattening phase -----------------------------------------
        if time.time() < deadline - 6.0:
            a_flat, sc_flat = flatten_conv_adam(
                a0,
                time_deadline=deadline - 3.0,
                max_iter=4000,
                step0=0.15,
            )
            if sc_flat < best_score:
                best_score = sc_flat
                best_seq = a_flat.tolist()
                print(f"[{outer_iter}] flatten n={n}, score={best_score:.6f}")

        # ---- 3. smooth max surrogate --------------------------------------
        if time.time() < deadline - 8.0:
            a_max, sc_max = smooth_max_adam(
                a_flat if 'a_flat' in locals() else a0,
                time_deadline=deadline - 5.0,
                max_iter=8000,
                step0=0.55,
                eps_factor=0.22,
            )
            if sc_max < best_score:
                best_score = sc_max
                best_seq = a_max.tolist()
                print(f"[{outer_iter}] smooth‑max n={n}, score={best_score:.6f}")

        # ---- 4. active‑set sub‑gradient descent (Adam) --------------------
        if time.time() < deadline - 12.0:
            a_sub, sc_sub = active_subgrad_adam(
                a_max,
                time_deadline=deadline - 7.0,
                max_iter=150_000,
                step0=0.18,
                tol_rel=1e-4,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{outer_iter}] active‑subgrad n={n}, score={best_score:.6f}")

        # ---- 5. tiny LP boost (only for small n) --------------------------
        if n <= 1024 and time.time() < deadline - 7.0:
            a_lp, sc_lp = lp_boost(
                a_sub if 'a_sub' in locals() else a_max,
                time_deadline=deadline - 5.0,
                mix_t=0.25,
                max_steps=5,
            )
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.tolist()
                print(f"[{outer_iter}] LP‑boost n={n}, score={best_score:.6f}")

        # ---- 6. simulated‑annealing hill‑climb ----------------------------
        if time.time() < deadline - 5.0:
            seq_to_refine = best_seq if best_seq is not None else a_sub.tolist()
            polished_seq, polished_sc = hill_climb_anneal(
                seq_to_refine,
                time_deadline=deadline - 2.0,
                max_steps=250_000,
            )
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished_seq
                print(f"[{outer_iter}] anneal improved: score={best_score:.6f}")

        # ---- 7. symmetry averaging -----------------------------------------
        if time.time() < deadline - 2.0:
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer_iter}] symmetry averaged: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
