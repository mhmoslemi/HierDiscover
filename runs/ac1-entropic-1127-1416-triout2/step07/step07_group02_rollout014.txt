def subgradient_adam_optimize(a0, deadline, n, lr=0.02, max_iter=200000):
    # a0 sum=1 (maybe not exactly due to rounding)
    # map a0 to logits via log(a0+eps)
    z = np.log(a0 + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        a = _softmax(z)
        conv = convolution_fft(a, a)
        mx = conv.max()
        # mask for active indices:
        tol = mx*1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[2*n-2 - np.arange(n)]  # L=2n-1
        # gradient w.r.t logits:
        dot = np.dot(subgrad, a)
        grad_z = a * (subgrad - dot)

        m = beta1*m + (1-beta1)*grad_z

"""
Propose a step‑function that minimises ``evaluate_sequence``.
The method combines:

* Analytic families (arcsine, Beta, power, mixtures) to generate many
  diverse seeds for a chosen length (up to 4096).
* A continuation Adam optimiser on a smooth‑max surrogate (log‑sum‑exp).
* A direct sub‑gradient Adam optimiser that works on the true max‑
  objective (non‑smooth) while staying on the simplex via a soft‑max.
* A simple hill‑climbing (random mass‑transfer) local search.
* A final polishing phase with the exact projected sub‑gradient method
  from the previous state‑of‑the‑art.

All phases respect the global 1000 s time budget.  The code never
returns an empty sequence – a uniform fallback is used if needed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(a,a))/sum(a)^2 (no clipping)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discretised arcsine density on {0,…,n‑1}.  Normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(alpha,beta) shape on a uniform grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power law shape 1/[(i+½)(n‑½‑i)+eps]^p, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Convex mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

# Optimisers
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 300.0,
    ramp_iters: int = 60_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    minimise the smooth approximation L_β(a)=log‑sum‑exp(β·conv(a,a)).
    The parameter β is increased linearly from beta0 to beta_max.
    The iterate a lives on the simplex via a soft‑max map.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    Adam it={t:,d}  β={beta:.1f}  score≈{fast_score(a):.6f}")

    return _softmax(z)


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    n: int,
    lr: float = 0.02,
    max_iter: int = 80_000,
) -> np.ndarray:
    """
    Adam optimiser on the true objective 2·n·max(conv(a,a)).
    The iterate stays on the simplex via logits → soft‑max.
    The gradient is a sub‑gradient obtained from the active max‑indices.
    """
    # initialise logits from a0 (avoid log(0))
    z = np.log(a0 + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    L = 2 * n - 1

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        a = _softmax(z)                     # on simplex
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance to capture all near‑max entries
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑convolution
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # gradient wrt logits (soft‑max chain rule)
        dot = np.dot(subgrad, a)
        grad_z = a * (subgrad - dot)

        # Adam step
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    SubAdam it={t:,d}  score≈{fast_score(a):.6f}")

    return _softmax(z)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑objective."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    Sub‑grad it={it:,d}  score≈{fast_score(a):.6f}")

    return a


def annealing_local_search(
    a0: np.ndarray,
    deadline: float,
    n: int,
    rng: np.random.Generator,
    max_moves: int = 5_000,
) -> np.ndarray:
    """
    Simple hill‑climbing: repeatedly move a small amount of mass from one
    coordinate to another and keep the move if the fast surrogate improves.
    The search stops when time runs out or after ``max_moves`` proposals.
    """
    a = a0.copy()
    cur_score = fast_score(a)
    total_time = deadline - time.time()
    moves = 0

    while time.time() < deadline and moves < max_moves:
        moves += 1
        # pick a source index proportionally to its current mass
        src = rng.choice(n, p=a)
        if a[src] <= 1e-14:
            continue
        # pick a different destination uniformly
        dst = rng.integers(0, n - 1)
        if dst >= src:
            dst += 1
        # propose moving a fraction of the source mass
        # the step size shrinks as the deadline approaches
        remaining = deadline - time.time()
        scale = 0.3 * (remaining / total_time)  # from 0.3 down to 0
        eps = a[src] * scale * rng.random()
        if eps <= 0 or eps > a[src]:
            continue
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # re‑normalise to avoid drift due to rounding
        a_new /= a_new.sum()
        new_score = fast_score(a_new)
        if new_score < cur_score:
            a, cur_score = a_new, new_score
    return a

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The method follows a multi‑phase workflow:

1. **Analytic seeding** – a wide pool of candidate shapes (arcsine,
   Beta‑family, power‑law, mixtures, random Dirichlet) is generated for
   each chosen length (up to 4096).  The cheap surrogate
   ``fast_score`` (2·n·max(conv)/sum²) selects a few promising seeds.

2. **Continuation Adam on a smooth‑max surrogate** – the objective
   ``log‑sum‑exp(β·conv(a,a))`` is minimised while β is increased from a
   modest value (≈5) to a large one (≈400).  The iterate lives on the
   probability simplex via a soft‑max parametrisation.

3. **Direct sub‑gradient Adam** – a second Adam phase works on the
   exact non‑smooth objective 2·n·max(conv(a,a)); the sub‑gradient of the
   max‑convolution is obtained from the active indices.  This phase
   refines the solution obtained in step 2.

4. **Local hill‑climbing (annealing)** – small random transfers of mass
   between coordinates are accepted only when they lower the fast
   surrogate.  The schedule shrinks the step size as the deadline
   approaches.

5. **Final polishing** – the best candidate found is fed to the exact
   projected sub‑gradient descent from the previous state‑of‑the‑art.

All phases respect the global 1000 s budget; a uniform fallback is used
if no candidate is produced.  The returned sequence is a list of
non‑negative floats that summation‑normalises automatically via the
evaluation routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """
    Cheap scale‑invariant surrogate:
    2·n·max(conv(a,a))/sum(a)²   (no clipping or safety checks).
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discretised arcsine density on {0,…,n‑1}, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β) shape on a uniform grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape 1/[(i+½)(n‑½‑i)+eps]ᵖ, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform distributions."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

# Optimisers
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to one."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 300.0,
    ramp_iters: int = 60_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimise the smooth surrogate L_β(a)=log‑sum‑exp(β·conv(a,a)).
    The temperature β is increased linearly from ``beta0`` to ``beta_max``.
    The iterate is constrained to the simplex by the soft‑max map.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly schedule β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a: 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)          # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]     # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    Adam it={t:,d}  β={beta:.1f}  score≈{fast_score(a):.6f}")

    return _softmax(z)


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    n: int,
    lr: float = 0.02,
    max_iter: int = 80_000,
) -> np.ndarray:
    """
    Adam optimiser on the exact objective 2·n·max(conv(a,a)).
    The iterate stays on the simplex via logits → soft‑max.
    The sub‑gradient is obtained from the active max‑indices.
    """
    z = np.log(a0 + 1e-12)          # initialise logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    L = 2 * n - 1

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        a = _softmax(z)
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # back‑prop through soft‑max
        dot = np.dot(subgrad, a)
        grad_z = a * (subgrad - dot)

        # Adam step
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    SubAdam it={t:,d}  score≈{fast_score(a):.6f}")

    return _softmax(z)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))/sum(a)².
    Used as a final polishing step.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    Exact sub‑grad it={it:,d}  score≈{fast_score(a):.6f}")

    return a


def annealing_local_search(
    a0: np.ndarray,
    deadline: float,
    n: int,
    rng: np.random.Generator,
    max_moves: int = 5_000,
) -> np.ndarray:
    """
    Simple hill‑climbing: repeatedly move a small amount of mass from one
    index to another and keep the move if the surrogate improves.
    The step size shrinks as the deadline approaches.
    """
    a = a0.copy()
    cur_score = fast_score(a)
    total_time = deadline - time.time()
    moves = 0

    while time.time() < deadline and moves < max_moves:
        moves += 1
        src = rng.choice(n, p=a)               # heavier coordinates are more likely sources
        if a[src] <= 1e-14:
            continue
        dst = rng.integers(0, n - 1)
        if dst >= src:
            dst += 1

        remaining = deadline - time.time()
        scale = 0.3 * (remaining / total_time)   # decays to zero
        eps = a[src] * scale * rng.random()
        if eps <= 0 or eps > a[src]:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        a_new /= a_new.sum()                    # re‑normalise (tiny drift)
        new_score = fast_score(a_new)
        if new_score < cur_score:
            a, cur_score = a_new, new_score

    return a
