def adam_optimize(start_a, deadline, max_iters=200_000, beta_start=5.0, beta_end=200.0, lr=0.1, eps=1e-8):
    a = start_a.copy()
    n = a.shape[0]
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # schedule of beta
        t = it / max_iters
        beta = beta_start * (beta_end/beta_start)**t
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # weights
        exp_weights = np.exp(beta*(conv - maxc))
        w = exp_weights / np.sum(exp_weights)
        # gradient:
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n-1:n-1+n]   # shape n
        # Adam updates
        m = 0.9*m + 0.1*grad
        v = 0.999*v + 0.001*grad*grad
        step = lr / (np.sqrt(v) + eps)
        a = project_to_simplex(a - step * m)
        # evaluate
        if it % 200 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score
```

def simulated_annealing(start_a, deadline, max_iter=1000000):
    a = start_a.copy()
    n = a.size
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    iter = 0
    while time.time() < deadline and iter < max_iter:
        T = max(0.001, (deadline - time.time()) / (deadline - start_time))  # decaying temperature
        # propose move:
        cand = a.copy()
        # choose move type:
        if np.random.rand()<0.5:
            i,j = np.random.randint(n, size=2)
            # swap mass:
            cand[i],cand[j] = cand[j],cand[i]
        else:
            i,j = np.random.randint(n, size=2)
            # transfer a small fraction from i to j
            frac = 0.05 * np.random.rand()  # between 0 and 5% of source
            delta = frac * cand[i]
            cand[i] -= delta
            cand[j] += delta
        # project to simplex (ensuring stay nonneg, sum 1)
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < np.exp((cur_score - cand_score)/T):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        iter+=1
    return best_a, best_score

def upsample_vector(v, new_n):
    old_n = len(v)
    if new_n == old_n:
        return v.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    v_new = np.interp(x_new, x_old, v)
    v_new = project_to_simplex(v_new)
    return v_new

def smooth_adam_optimize(start_a, deadline, max_iters=200_000, beta_start=5.0, beta_end=200.0, lr=0.05):
    a=start_a.copy()
    n=a.shape[0]
    m=np.zeros_like(a)
    v=np.zeros_like(a)
    best_a=a.copy()
    best_score=compute_score(a)
    it=1
    while it <= max_iters and time.time() < deadline:
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start)**t
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute softmax weights
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()
        # gradient via cross-correlation
        grad = 2.0 * convolution_fft(a[::-1], w)[n-1:n-1+n]
        # Adam updates
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        step = lr / (np.sqrt(v) + 1e-8)
        a = project_to_simplex(a - step * m)
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        it += 1
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

1. **Coarse optimisation on a small system** (n≈2048).  
   – Scan a one‑parameter family a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ.  
   – Refine the best candidate with a smoothed‐gradient Adam optimiser
     (β is annealed from 5 to ≈200).

2. **Upscaling** the vector to the final size (n≈8192).  
   The small‑scale optimum is interpolated and projected onto the simplex.

3. **Fine optimisation on the full size**.  
   – Run the same Adam optimiser (more iterations, tighter β).  
   – Optional projected sub‑gradient descent.  
   – A long stochastic hill‑climb that accepts improving moves and
     occasional worsening moves (simulated‑annealing style).

All operations respect the simplex (non‑negative, unit sum) so the
objective `evaluate_sequence` equals `2·n·max(conv)` and can be
computed cheaply with `compute_score`.  The routine respects the
1000 s wall‑clock limit and prints progress timestamps.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of evaluate_sequence for a vector that already satisfies
    sum(a)=1 and a≥0.  Returns 2·n·max(a⊛a) (the true objective value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}.
    """
    if np.all(v <= 0):
        # all non‑positive → uniform point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families – the “edge‑heavy” start
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    Produce a symmetric vector a_i ∝ [(i+½)(n‑½‑i)]⁻ᵅ and project onto the simplex.
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against overflow/underflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a coarse grid of α, optionally a refined grid around the best
    value, and return the best vector together with its exact score.
    """
    best_a = None
    best_score = float("inf")
    # coarse grid
    alphas = np.linspace(0.0, 2.5, 26)          # step ≈0.1
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_alpha = alpha

    # refined grid around the best α (if time permits)
    if best_a is not None and time.time() < deadline:
        lo = max(0.0, best_alpha - 0.2)
        hi = best_alpha + 0.2
        alphas_ref = np.linspace(lo, hi, 21)  # step ≈0.02
        for alpha in alphas_ref:
            if time.time() >= deadline:
                break
            a = alpha_start(n, alpha)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed objective – Adam optimiser
# ----------------------------------------------------------------------


def smooth_adam_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 5.0,
                         beta_end: float = 200.0,
                         lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Minimises the smoothed objective
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using an Adam style update and an exponential β‑schedule.
    Returns the best exact score (computed with `compute_score`) seen.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential annealing)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # probability vector, sum to 1

        # gradient of the smoothed objective
        grad = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a - step * m_hat)

        # occasional exact evaluation
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (true max‑objective)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the exact objective
    2·n·max(a⊛a).  Uses a 1/√t stepsize schedule.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                      # active index

        # sub‑gradient for the active entry k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing flavour)
# ----------------------------------------------------------------------


def stochastic_hill_climb(start_a: np.ndarray,
                          deadline: float,
                          max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Long random walk on the simplex:
    – with prob 0.5 swap two entries,
    – otherwise transfer a fraction of mass.
    Worsening moves are occasionally accepted with a temperature
    decreasing linearly with the remaining time.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(a)

    # temperature parameters
    start_T = 0.02
    end_T = 0.0005

    steps_done = 0
    while steps_done < max_steps and time.time() < deadline:
        remaining = deadline - time.time()
        total = deadline - (deadline - max_steps)  # not used, just for scaling

        # linearly decreasing temperature
        T = end_T + (start_T - end_T) * (remaining / (deadline - (deadline - max_steps)))

        cand = a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.08 * cand[i]          # move up to ~8 % of source mass
                cand[i] -= delta
                cand[j] += delta
                # safeguard against tiny negatives
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        # acceptance rule
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand                       # also move the current point
        else:
            # accept with Metropolis probability
            if np.random.rand() < np.exp(-(cand_score - best_score) / max(T, 1e-12)):
                a = cand

        steps_done += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling / interpolation
# ----------------------------------------------------------------------


def upsample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a simplex vector `v` from its current size
    to `new_n`, preserving the sum (re‑projected onto the simplex).
    """
    old_n = v.shape[0]
    if new_n == old_n:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    v_new = np.interp(x_new, x_old, v)
    v_new = project_to_simplex(v_new)
    return v_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Finds a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows the three‑phase plan described above and
    respects the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Time bookkeeping – keep a small safety margin for the harness.
    # ------------------------------------------------------------------
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Optional fallback – use a previously discovered sequence if it exists.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if 'height_sequence_1' in globals():
        try:
            fallback = [float(x) for x in globals()['height_sequence_1']]
            fb = evaluate_sequence(fallback)
            print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
            if fb < best_score:
                best_score = fb
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – small‑scale optimisation (fast, many iterations)
    # ------------------------------------------------------------------
    # decide a modest size for the exploratory phase
    small_n = 2048
    phase1_deadline = min(deadline, start_time + 0.35 * budget_s)  # ≈35 % of total budget
    print(f"[{time.time():.1f}] phase‑1 (small n={small_n}) start")
    a_small, sc_small = find_best_alpha_start(small_n, phase1_deadline - 30.0)
    if a_small is not None and sc_small < best_score:
        best_score = sc_small
        best_seq = a_small.tolist()
        print(f"[{time.time():.1f}] best after α‑grid (small) = {best_score:.6f}")

    # Refine the small‑scale candidate with Adam + smoothing
    if time.time() < phase1_deadline - 20.0:
        a_small, sc_small = smooth_adam_optimize(
            a_small,
            phase1_deadline - 10.0,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=200.0,
            lr=0.06,
        )
        if sc_small < best_score:
            best_score = sc_small
            best_seq = a_small.tolist()
            print(f"[{time.time():.1f}] after small Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to the final size
    # ------------------------------------------------------------------
    # Choose final length depending on remaining time
    remaining = deadline - time.time()
    if remaining > 400.0:
        final_n = 16384
    elif remaining > 200.0:
        final_n = 8192
    else:
        final_n = 4096

    print(f"[{time.time():.1f}] upscaling to n={final_n}")
    a_large = upsample_vector(np.array(best_seq, dtype=float), final_n)

    # ------------------------------------------------------------------
    # Phase 3 – fine optimisation on the full size
    # ------------------------------------------------------------------
    # 3a) Smoothed Adam optimisation (the heavy work)
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] large‑scale Adam start")
        a_large, sc_large = smooth_adam_optimize(
            a_large,
            deadline - 100.0,
            max_iters=350_000,
            beta_start=5.0,
            beta_end=300.0,
            lr=0.05,
        )
        if sc_large < best_score:
            best_score = sc_large
            best_seq = a_large.tolist()
            print(f"[{time.time():.1f}] after large Adam best = {best_score:.6f}")

    # 3b) Projected sub‑gradient refinement (optional, if time permits)
    if time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement start")
        a_large, sc_sub = subgrad_optimize(
            a_large,
            deadline - 40.0,
            max_iters=300_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_large.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # 3c) Long stochastic hill‑climb / simulated‑annealing polishing
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] stochastic hill‑climb start")
        a_large, sc_hc = stochastic_hill_climb(
            a_large,
            deadline,
            max_steps=900_000,
        )
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_large.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if everything failed fall back to a uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        final_n = max(256, final_n)
        best_seq = [1.0 / final_n] * final_n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will call this as well)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] search finished – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats as required.
    return best_seq
