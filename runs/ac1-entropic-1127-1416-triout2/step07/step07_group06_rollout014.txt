"""
Propose a low‑peak step‑function.

The algorithm consists of three stages:

1. **Coarse parametric search** –  For several lengths `n` we generate a
   *U‑shaped* family of vectors
        a_i(p) ∝ ((i+1)*(n-i))^{-p}  (p≥0)
   and linearly mix it with the uniform vector.  The mixture weight λ
   and the exponent p are swept on a grid, each candidate is projected
   onto the probability simplex (the mixture already sums to one) and
   evaluated with a fast FFT–based routine `compute_score`.  The best
   candidate seen so far is kept.

2. **Local refinement** – The best vector from the grid search is fed
   into two deterministic optimisers that work directly on the exact
   objective:
        * a smooth–max (log‑sum‑exp) surrogate minimised with Adam,
        * a classic sub‑gradient descent on the true max‑convulation.
   Both steps respect the simplex constraint after every update.

3. **Optional random shuffling / symmetry averaging** – a few random
   permutations of the current best vector are tested (they keep the
   ℓ₁‑norm).  Finally a cheap symmetric averaging `a←(a+rev(a))/2` is
   attempted because symmetric vectors often have flatter autocorrelations.

All work is limited by an explicit deadline (`budget_s` ≈ 1000 s); the
best sequence found before the deadline is returned as a plain Python
list of floats.  The routine is completely self‑contained, uses only
NumPy, and respects the required non‑negative, ℓ₁‑normalised format.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum(a)==1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, ∑x=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric U‑shaped families
# ----------------------------------------------------------------------


def u_shape(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U” shape:
        a_i ∝ ((i+1)*(n-i))^{-p}
    Normalised to sum 1.
    """
    idx = np.arange(n, dtype=np.float64)
    # use log‑exp to avoid overflow/underflow
    raw = np.exp(-p * np.log((idx + 1) * (n - idx)))
    raw_sum = raw.sum()
    return raw / raw_sum


def mixed_candidate(n: int, p: float, lam: float) -> np.ndarray:
    """
    Mixture of the uniform vector and the U‑shape.
    lam=1  → uniform, lam=0 → pure U‑shape.
    """
    if lam == 1.0:
        return np.full(n, 1.0 / n)
    if lam == 0.0:
        return u_shape(n, p)
    uniform = np.full(n, 1.0 / n)
    shape = u_shape(n, p)
    return lam * uniform + (1.0 - lam) * shape


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) optimiser – Adam
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.5,
    eps_factor: float = 0.22,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Minimise the smooth max‐surrogate of the convolution."""
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # decreasing ε, never below 1 % of maxc
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        # soft‑max weights (∑w = 1)
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)

        # gradient g_i = 2·∑_j a_j·w_{i+j}
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter
        if t % 400 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def subgrad_optimize(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Classical sub‑gradient descent on 2·n·max(a⊛a)."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active lag
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # tiny random shake every thousand steps
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (often beneficial)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a+rev(a))/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    The routine respects a ~1000 s time budget and returns a plain list
    of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Fallback if a previously known construction is available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse parametric search over lengths / p / λ
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]
    lam_grid = [0.0, 0.25, 0.5, 0.75, 1.0]               # mixture weight
    p_grid = np.linspace(0.01, 2.0, 21)                 # exponent candidates

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Grid search for n={n} ===")
        uniform = np.full(n, 1.0 / n)

        for p in p_grid:
            if time.time() >= deadline - 5.0:
                break
            shape = u_shape(n, p)

            for lam in lam_grid:
                if time.time() >= deadline - 5.0:
                    break

                if lam == 0.0:
                    cand = shape
                elif lam == 1.0:
                    cand = uniform
                else:
                    cand = lam * uniform + (1.0 - lam) * shape

                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.tolist()
                    print(
                        f"New best: score={best_score:.6f} (n={n}, p={p:.3f}, λ={lam})"
                    )

        # ------------------------------------------------------------------
        # 1.5️⃣  Small random permutation sweep for the current best n
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            a_best = np.array(best_seq, copy=False)
            for _ in range(30):
                if time.time() >= deadline - 5.0:
                    break
                perm = np.random.permutation(n)
                a_perm = a_best[perm]
                sc = compute_score(a_perm)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_perm.tolist()
                    a_best = a_perm
                    print(f"Permute improved: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Deterministic local refinement (smooth‑max → sub‑gradient)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=float)

        # ---- smooth‑max surrogate (Adam) ----
        if time.time() < deadline - 2.0:
            a_s, sc_s = smooth_max_adam(
                a,
                time_deadline=deadline - 2.0,
                max_iter=8000,
                step0=0.5,
                eps_factor=0.22,
            )
            if sc_s < best_score:
                best_score = sc_s
                best_seq = a_s.tolist()
                a = a_s
                print(f"Smooth‑max improved: score={best_score:.6f}")

        # ---- exact sub‑gradient descent ----
        if time.time() < deadline - 2.0:
            a_sub, sc_sub = subgrad_optimize(
                a,
                time_deadline=deadline - 2.0,
                max_iter=50000,
                step0=0.07,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                a = a_sub
                print(f"Sub‑gradient improved: score={best_score:.6f}")

        # ---- optional symmetry averaging (cheap) ----
        if time.time() < deadline - 2.0:
            a_sym = final_symmetry_average(a)
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = a_sym.tolist()
                print(f"Symmetry averaging improved: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Safety net – if nothing was found, use a tiny uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq))
        print("Safety fallback uniform vector used.")

    # ------------------------------------------------------------------
    # 4️⃣  Final official evaluation (uses the provided function)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
