def smooth_gradient_optimize(start_a: np.ndarray,
                             deadline: float,
                             max_iters: int = 500_000,
                             beta_start: float = 5.0,
                             beta_end: float = 200.0,
                             step0: float = 0.8,
                             momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Accelerated gradient descent on the log‑sum‑exp smooth approximation
    of the max‑convolution.  Uses Nesterov‑style momentum and an
    exponentially increasing β to tighten the approximation.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule — exponential increase from beta_start to beta_end
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution a*a
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable log‑sum‑exp weights
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient = 2 * (a_rev * w) slice
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        step = step0 / math.sqrt(it)
        # momentum update

        a = project_to_simplex(a + v)

                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 500_000,
               temp_start: float = 0.02,
               temp_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk on the simplex.
    Moves: swap two entries or transfer a fraction of mass.
    A move is kept if it improves the exact score, otherwise it may
    be accepted with a probability that decays with the temperature.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12
    step_scale_min = 1e-4

    steps = 0
    while steps < max_steps and time.time() < deadline:
        # linear temperature schedule
        progress = steps / max_steps
        temperature = temp_start * (temp_end / temp_start) ** progress

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        improved = sc < best_score - 1e-12
        if improved:
            best_score = sc
            best_a = cand
            # slowly increase step_scale to explore larger moves
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # Metropolis acceptance
            if temperature > 0 and np.random.rand() < math.exp(-(sc - best_score) / temperature):
                best_score = sc
                best_a = cand
                step_scale = min(step_scale * 1.02, 0.5)
            else:
                step_scale = max(step_scale * 0.97, step_scale_min)
        steps += 1

    return best_a, best_score

"""
Refined multi‑stage optimizer for the max‑convolution objective used by the
challenge.  The method improves on the previous four‑stage pipeline by

* using a finer α‑grid for the analytical start,
* applying an accelerated smoothed‑max gradient descent with momentum and a
  larger final β (closer approximation to the true max),
* allowing several random sparse restarts,
* employing a longer temperature‑controlled hill‑climb.

The search works on a longer sequence (up to 16384 entries) when the time
budget permits, which gives the optimiser more degrees of freedom and a
better chance to lower the evaluation score.
All helper functions are top‑level and avoid closures, satisfying the
harness requirements.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution of real vectors using an FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact score used by ``evaluate_sequence`` when sum(a)=1.
    Returns 2·n·max(a⊛a) (the objective is scale‑invariant).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    a_i(α) ∝ [(i+½)(n‑½‑i)]^{‑α}, normalised to the simplex.
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # work in log‑space to avoid overflow/underflow
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a modest grid of α‑values, keep the one yielding the smallest exact score.
    Returns (best_vector, best_score).
    """
    best_a = None
    best_score = float("inf")
    # a slightly finer grid than before
    alphas = np.linspace(0.0, 2.5, 31)  # step ≈0.08
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Accelerated smoothed‑max gradient descent (log‑sum‑exp)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(start_a: np.ndarray,
                             deadline: float,
                             max_iters: int = 500_000,
                             beta_start: float = 5.0,
                             beta_end: float = 200.0,
                             step0: float = 0.8,
                             momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using an exponentially increasing β, a 1/√t stepsize and Nesterov‑style
    momentum.  The routine returns the best exact score seen.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    v = np.zeros_like(a)                       # momentum buffer
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- β schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- smoothed max weights -----
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()        # probability vector of length 2n‑1

        # ----- gradient (cross‑correlation) -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]     # slice of length n

        # ----- step with momentum and simplex projection -----
        step = step0 / math.sqrt(it)
        v = momentum * v - step * grad
        a = project_to_simplex(a + v)

        # ----- keep the best exact score -----
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Long stochastic hill‑climb with temperature‑controlled acceptance
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 800_000,
               temp_start: float = 2e-2,
               temp_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
        – with 50 % probability swap two entries,
        – otherwise transfer a fraction of mass from one index to another.
    A move that does not improve the exact score may still be accepted
    with a Metropolis probability that decays with a temperature schedule.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12          # typical fraction of source mass to move
    step_scale_min = 1e-4

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # linearised temperature schedule (exponential decay)
        progress = step / max_steps
        temperature = temp_start * (temp_end / temp_start) ** progress

        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # swap move
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass‑transfer move
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        improved = sc < best_score - 1e-12
        if improved:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # grow step size slowly
        else:
            # Metropolis acceptance
            if temperature > 0.0 and np.random.rand() < math.exp(-(sc - best_score) / temperature):
                best_score = sc
                best_a = cand
                step_scale = min(step_scale * 1.02, 0.5)
            else:
                step_scale = max(step_scale * 0.97, step_scale_min)

    return best_a, best_score

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock limit and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin – the harness adds its own checks
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length selection – longer vectors give more flexibility.
    # ------------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 600:
        n = 16384
    elif remaining > 300:
        n = 12288
    else:
        n = 8192
    print(f"[{time.time():.1f}] selected length n={n}")

    # ------------------------------------------------------------------
    # Store best result found so far
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback to a previously found construction, if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Edge‑heavy analytical start (grid over α)
    # ------------------------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random sparse restarts → accelerated smooth optimiser
    # ------------------------------------------------------------------
    def random_sparse_start():
        """Generate a random probability vector with 5‑12 % non‑zeros."""
        sparsity = np.random.uniform(0.04, 0.12)
        k = max(2, int(round(sparsity * n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
        return project_to_simplex(a)

    # keep trying restarts while a comfortable amount of time remains
    while time.time() < deadline - 300.0:
        a0 = random_sparse_start()
        a_opt, sc_opt = smooth_gradient_optimize(
            a0,
            deadline - 200.0,
            max_iters=300_000,
            beta_start=6.0,
            beta_end=120.0,
            step0=0.7,
            momentum=0.85,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random restart = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Deep accelerated smooth optimisation from the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] deep smooth optimisation")
        a_smooth, sc_smooth = smooth_gradient_optimize(
            np.array(best_seq, dtype=float),
            deadline - 130.0,
            max_iters=600_000,
            beta_start=5.0,
            beta_end=200.0,
            step0=0.8,
            momentum=0.9,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{time.time():.1f}] after deep smooth best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
