"""
Optimiser based on a two‑stage scheme

1️⃣  Coarse LP refinement (n≈1024).  
    • Start from the arcsine density (known good shape).  
    • Compute M = max(conv(f,f)).  
    • Solve a linear programme  max Σ g_j   s.t.   f∗g ≤ M ,  g ≥ 0.  
      The LP is tiny (≤ 2 · n variables) and can be solved with SciPy/HiGHS.  
    • Normalise the LP solution to a probability vector and mix it with f.
      The mixing factor is chosen by a tiny 1‑D search that minimises the
      resulting max‑convolution.

2️⃣  Upsampling + smooth‑max Adam optimisation on the fine grid (n=4096‑8192).  
    The variable is parametrised by logits → softmax, guaranteeing non‑negativity
    and unit sum.  A log‑sum‑exp (temperature τ) approximates the max; τ is
    increased from 0.5 to 300.  Adam with a decaying learning‑rate refines the
    shape.

3️⃣  Exact non‑smooth polishing.  
    • Projected Euclidean sub‑gradient descent on the true max‑convolution.  
    • Exponentiated‑gradient (mirror‑descent) polish.

The whole pipeline obeys the 1000 s wall‑clock budget, keeps the best
candidate found and guarantees a non‑empty return.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a∗b using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate of the official score: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points,
    then re‑normalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Adam optimiser with smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 300_000,
                             tau_start: float = 0.5,
                             tau_end: float = 300.0,
                             base_lr: float = 0.6) -> np.ndarray:
    """
    Adam optimisation on the log‑sum‑exp smooth approximation of max(conv).
    a = softmax(w) guarantees a≥0 and Σa=1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a∗a
        conv = convolution_fft(a, a)

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp

        # gradient w.r.t. a : 2·conv_rev(weights)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging
        if it % 40000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (non‑smooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 500_000,
                              step0: float = 0.08) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 100000 == 0:
            print(f"    Sub‑grad it={it:,d} max={max_val:.6e}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 300_000,
              eta0: float = 0.24) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 100000 == 0:
            print(f"    EG it={it:,d} max={max_val:.6e}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming helpers (coarse phase)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray):
    """
    Construct the (2n‑1)×n matrix C such that C @ g = conv(f, g).
    Sparse CSR format – suitable for SciPy's HiGHS LP solver.
    """
    from scipy.sparse import csr_matrix

    n = f.size
    rows = np.arange(n)[:, None] + np.arange(n)          # shape (n,n)
    rows = rows.ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)
    L = 2 * n - 1
    return csr_matrix((data, (rows, cols)), shape=(L, n))


def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   max Σ g_j    s.t.   conv(f, g) ≤ M ,   g ≥ 0.
    Returns the optimal g (may not be normalised).
    """
    from scipy.optimize import linprog

    n = f.shape[0]
    L = 2 * n - 1
    A = build_convolution_matrix(f)          # shape (L, n), sparse
    b = np.full(L, M, dtype=np.float64)
    c = -np.ones(n)                          # maximise Σg ⇒ minimise −Σg
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    # fallback – return the current f (no improvement possible)
    return f.copy()


def lp_refine(n: int, deadline: float, max_iters: int = 12) -> np.ndarray:
    """
    Coarse LP‑based refinement on a grid of size n.
    Returns a probability vector (sum = 1).
    """
    f = arcsine_initial(n)      # deterministic start
    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_f_f = convolution_fft(f, f)
        M = conv_f_f.max()

        g_opt = solve_lp_max_sum(f, M)
        sum_g_opt = g_opt.sum()
        if sum_g_opt <= 0:
            break
        g = g_opt / sum_g_opt      # normalise to a probability vector

        # pre‑compute the three convolutions needed for mixing
        conv_f_g = convolution_fft(f, g)
        conv_g_g = convolution_fft(g, g)

        # tiny 1‑D search for the mixing factor that most reduces max‑conv
        t_candidates = np.array([0.05, 0.10, 0.15, 0.20, 0.30])
        best_t = 0.0
        best_M = M
        for t in t_candidates:
            combo = ((1 - t) ** 2) * conv_f_f + (t ** 2) * conv_g_g + 2 * t * (1 - t) * conv_f_g
            cur_M = combo.max()
            if cur_M < best_M - 1e-12:
                best_M = cur_M
                best_t = t

        if best_t == 0.0:   # no improving mixing factor found
            print(f"    LP iter={it}: stagnation, stop")
            break

        # update f and re‑normalise (floating drift)
        f = (1 - best_t) * f + best_t * g
        f = f / f.sum()

        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        print(f"    LP iter={it}  t={best_t:.3f}  score={cur_score:.6f}  max={best_M:.6e}")

    return best_f

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence`` score.
    The routine combines a coarse LP refinement (n≈1024) with a high‑resolution
    Adam + exact polishing pipeline.  The total wall‑clock time never exceeds
    ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  (optional) fall back to a previously discovered sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse LP refinement (n=1024)
    # ------------------------------------------------------------------
    n_coarse = 1024
    time_for_lp = min(250.0, deadline - time.time() - 600.0)   # leave ample time for fine phase
    lp_deadline = time.time() + max(0.0, time_for_lp)
    print("\n--- Phase 1: coarse LP refinement (n={}) ---".format(n_coarse))
    f_coarse = lp_refine(n_coarse, lp_deadline, max_iters=15)

    # ------------------------------------------------------------------
    # 2️⃣  Upsample to a finer grid and run smooth‑max Adam
    # ------------------------------------------------------------------
    # Choose high resolution depending on remaining budget
    remaining = deadline - time.time()
    if remaining > 350.0:
        n_high = 8192
    elif remaining > 200.0:
        n_high = 4096
    else:
        n_high = 2048
    print(f"\n--- Phase 2: upsample to n={n_high} and Adam fine‑tune ---")
    a_high = upsample(f_coarse, n_high)

    # Adam parameters – allocate most of the remaining time to this phase
    remaining = deadline - time.time()
    max_iter_adam = int(min(500_000, remaining * 30))
    a_high = adam_optimize_smooth_max(
        a_high,
        deadline,
        max_iters=max_iter_adam,
        tau_start=0.5,
        tau_end=300.0,
        base_lr=0.6,
    )
    surrogate = compute_score(a_high)
    print(f"High‑res Adam surrogate score = {surrogate:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exact polishing (sub‑gradient + exponentiated‑gradient)
    # ------------------------------------------------------------------
    print("\n--- Phase 3: exact polishing ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(600_000, remaining * 30))
    a_refined = subgradient_descent_exact(a_high, deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.08)

    remaining = deadline - time.time()
    if remaining > 30.0:
        max_iter_eg = int(min(400_000, remaining * 30))
        a_refined = eg_refine(a_refined, deadline,
                              max_iter=max_iter_eg,
                              eta0=0.24)

    # ------------------------------------------------------------------
    # 4️⃣  Evaluate the candidate and keep the best
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(a_refined.tolist())
        print(f"Official score after polishing = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_refined.tolist()
            print("*** New global best found! ***")
    except Exception as e:
        print(f"Evaluation failed after polishing: {e}")

    # ------------------------------------------------------------------
    # 5️⃣  Optional final global polish if time permits
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n--- Final global polish (extra time) ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub2 = int(min(400_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sub2,
                                      step0=0.05)

        remaining = deadline - time.time()
        if remaining > 20.0:
            max_iter_eg2 = int(min(250_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg2,
                          eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
