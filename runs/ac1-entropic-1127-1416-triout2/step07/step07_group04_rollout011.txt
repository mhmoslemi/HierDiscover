def propose_candidate(...):
    start = time.time()
    deadline = start + budget_s - 5

    # define n (target)
    target_n = 4096

    # baseline seed
    seeds = []

    # arcsine

            fallback_arr = upsample(fallback_arr, target_n)

        fallback_arr = fallback_arr / fallback_arr.sum()
        seeds.append(fallback_arr)

    # random beta shapes

    for _ in range(3):
        alpha = rng.uniform(0.1, 4.0)
        beta = rng.uniform(0.1, 4.0)

        remaining = deadline - time.time()
        # decide iterations based on remaining and number of seeds left.

def local_random_search(a, deadline, max_steps=200000):
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    rng = np.random.default_rng()
    step = 0
    while time.time() < deadline and step < max_steps:
        i = rng.integers(0,n)
        j = rng.integers(0,n-1)
        if j >= i:
            j += 1
        # choose delta relative to a[i], up to a[i]
        max_delta = best_a[i]
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * 0.05  # move up to 5% of a[i]
        if delta <= 0:
            continue
        new_a = best_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # ensure positivity
        if new_a[i] < 0:
            continue
        new_score = compute_score(new_a)
        if new_score < best_score:
            best_a = new_a
            best_score = new_score
        step += 1
    return best_a

"""
Propose a low‑discrepancy step function.

The pipeline mixes several ideas that proved effective in earlier
experiments and adds a small linear‑programming (LP) refinement step
which was missing before.

1.  **Coarse optimisation (n=1024)** – start from a classic arcsine
    density (optionally mixed with a fallback sequence).  Adam is used
    on a smooth approximation of the max‑convolution with a temperature
    schedule that ends at τ≈200.

2.  **LP tightening** – for the current coarse shape `f` we compute
    `M = max(conv(f,f))` and solve the linear program

        maximise   Σ g_j
        subject to  g ≥ 0,
                    conv(f, g) ≤ M.

    The optimal `g` is scaled to a probability vector and mixed with
    `f` (step size `t≈0.05`).  A few repetitions of this step
    dramatically lower the active constraints of the max‑convolution.

3.  **Upsampling** – the improved coarse shape is linearly interpolated
    to a finer grid (`n = 4096` or `2048` if the time budget is low).

4.  **Fine Adam** – the same smooth‑max Adam optimiser is run again,
    now with a much larger final temperature (`τ≈2000`) to obtain a
    very sharp approximation of the true max.

5.  **Exact polishing** – a projected sub‑gradient descent on the
    genuine max‑convolution followed by an exponentiated‑gradient (mirror
    descent) step.

6.  **Local random search** – a cheap stochastic hill‑climb that moves
    a tiny amount of mass between two bins if it reduces the surrogate
    `2·n·max(conv)`.  This finishes the remaining seconds.

All stages respect a global 1000 s wall‑clock budget; the function always
returns a non‑empty sequence (otherwise a uniform distribution is used).
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (fast for the sizes we use).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1} (Duchi et al., 2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    For a probability vector (Σ a = 1) the surrogate equals the official value:
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‐type density on a uniform grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform, normalised."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points, then renormalise.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (used before the exact phase)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimiser of the smooth approximation of max(conv(a,a)).
    The optimisation variable are logits `w` such that
        a = softmax(w)  (automatically Σa = 1, a ≥ 0).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)                # initialise logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)               # for numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)           # length L

        # smooth‑max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp        # shape (L,)

        # gradient w.r.t. a : 2·conv(a, rev(weights))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)   # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)           # decaying step‑size
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional diagnostic
        if it % 30000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (nonsmooth) polishing – sub‑gradient + exponentiated gradient
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Projected Euclidean sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror descent with KL geometry) polish.
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Local stochastic refinement (cheap hill‑climb)
# ----------------------------------------------------------------------


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_steps: int = 200_000) -> np.ndarray:
    """
    Simple stochastic hill‑climb: move a tiny amount of mass from a random
    index to another and keep the change if it reduces the surrogate.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    rng = np.random.default_rng()

    step = 0
    while time.time() < deadline and step < max_steps:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if best_a[i] <= 0.0:
            continue
        # move at most 5 % of the source weight
        delta = rng.random() * 0.05 * best_a[i]
        if delta <= 0.0:
            continue

        new_a = best_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # keep feasibility (non‑negative, sum = 1)
        if new_a[i] < 0.0:
            continue

        new_score = compute_score(new_a)
        if new_score < best_score:
            best_a = new_a
            best_score = new_score

        step += 1
        if step % 5000 == 0:
            # occasional progress report
            print(f"    Local search step {step:,d} best surrogate={best_score:.6f}")

    return best_a


# ----------------------------------------------------------------------
# LP tightening (optional, used only when cvxpy is available)
# ----------------------------------------------------------------------


def lp_tighten(f: np.ndarray, M: float, mix: float = 0.05) -> np.ndarray:
    """
    One LP refinement step:
        maximise Σ g_j  subject to g ≥ 0,  conv(f, g) ≤ M.
    The optimiser returns a new probability vector that is mixed
    with the current `f` using weight `mix`.
    """
    try:
        import cvxpy as cp
    except Exception:                     # cvxpy not present – skip
        return f

    n = f.size
    L = 2 * n - 1

    # Build the convolution matrix B where (B @ g)[k] = (f * g)_k
    B = np.zeros((L, n), dtype=float)
    for col in range(n):
        B[col:col + n, col] = f

    g = cp.Variable(n)
    constraints = [g >= 0,
                   B @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    # a modest iteration limit to stay within the time budget
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)

    if g.value is None:
        return f

    g_opt = np.maximum(g.value, 0.0)
    sum_g = g_opt.sum()
    if sum_g <= 0.0:
        return f
    g_scaled = g_opt / sum_g

    # mix and renormalise
    f_new = (1.0 - mix) * f + mix * g_scaled
    f_new /= f_new.sum()
    return f_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    The routine respects a 1000 s wall‑clock limit and returns the best
    sequence found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 1) initialise a coarse shape (n=1024)
    # ------------------------------------------------------------------
    n_coarse = 1024
    a0 = arcsine_initial(n_coarse)

    # mix with a possible fallback construction if it exists
    if "height_sequence_1" in globals():
        fallback = np.asarray(globals()["height_sequence_1"], dtype=float)
        if fallback.size != n_coarse:
            fallback = upsample(fallback, n_coarse)
        fallback = fallback / fallback.sum()
        a0 = 0.5 * a0 + 0.5 * fallback
        a0 /= a0.sum()

    print("\n--- Phase 1: coarse Adam optimisation (n={}) ---".format(n_coarse))
    remaining = deadline - time.time()
    max_iters_coarse = int(min(150_000, remaining * 30))
    a_coarse = adam_optimize_smooth_max(
        a0,
        deadline,
        max_iters=max_iters_coarse,
        tau_start=0.5,
        tau_end=200.0,
        base_lr=0.5,
    )
    print(f"Coarse surrogate after Adam = {compute_score(a_coarse):.6f}")

    # ------------------------------------------------------------------
    # 2) LP tightening (optional, executed if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                      # need a few seconds for the LPs
        print("\n--- Phase 2: LP tightening (up to 3 LP steps) ---")
        for lp_round in range(3):
            if time.time() >= deadline:
                break
            conv = convolution_fft(a_coarse, a_coarse)
            M = float(np.max(conv))
            a_coarse = lp_tighten(a_coarse, M, mix=0.05)
            print(f"  LP round {lp_round+1}, surrogate={compute_score(a_coarse):.6f}")

    # ------------------------------------------------------------------
    # 3) Upsample to fine resolution (n=4096, fallback to 2048)
    # ------------------------------------------------------------------
    target_n = 4096
    if deadline - time.time() < 150.0:        # not enough time for 4096‑point polish
        target_n = 2048
    print(f"\n--- Phase 3: upsample to n={target_n} ---")
    a_fine = upsample(a_coarse, target_n)

    # ------------------------------------------------------------------
    # 4) Fine Adam with a very high temperature schedule
    # ------------------------------------------------------------------
    print("\n--- Phase 4: fine Adam optimisation ---")
    remaining = deadline - time.time()
    max_iters_fine = int(min(250_000, remaining * 30))
    a_fine = adam_optimize_smooth_max(
        a_fine,
        deadline,
        max_iters=max_iters_fine,
        tau_start=0.5,
        tau_end=2000.0,
        base_lr=0.5,
    )
    print(f"Fine surrogate after Adam = {compute_score(a_fine):.6f}")

    # ------------------------------------------------------------------
    # 5) Exact polishing: sub‑gradient + exponentiated gradient
    # ------------------------------------------------------------------
    print("\n--- Phase 5: exact polishing (sub‑gradient) ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(400_000, remaining * 30))
    a_fine = subgradient_descent_exact(
        a_fine,
        deadline,
        max_iter=max_iter_sub,
        step0=0.25,
    )
    print(f"Surrogate after sub‑gradient = {compute_score(a_fine):.6f}")

    print("\n--- Phase 5b: exact polishing (exponentiated‑gradient) ---")
    remaining = deadline - time.time()
    if remaining > 2.0:
        max_iter_eg = int(min(300_000, remaining * 30))
        a_fine = eg_refine(
            a_fine,
            deadline,
            max_iter=max_iter_eg,
            eta0=0.25,
        )
        print(f"Surrogate after EG = {compute_score(a_fine):.6f}")

    # ------------------------------------------------------------------
    # 6) Cheap stochastic local refinement (runs only if time remains)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Phase 6: final stochastic refinement ---")
        steps = int(min(200_000, remaining * 3000))   # a few thousand steps per second
        a_fine = local_random_search(a_fine, deadline, max_steps=steps)
        print(f"Surrogate after local search = {compute_score(a_fine):.6f}")

    # ------------------------------------------------------------------
    # 7) Finalisation and official evaluation
    # ------------------------------------------------------------------
    best_seq = a_fine.tolist()
    # Guard against pathological zero‑sum (should not happen)
    total = sum(best_seq)
    if total == 0.0:
        best_seq = [1.0 / len(best_seq)] * len(best_seq)
    else:
        best_seq = [float(x / total) for x in best_seq]

    try:
        official = evaluate_sequence(best_seq)
        print(f"\nOfficial evaluation of the final candidate = {official:.6f}")
    except Exception as exc:
        print(f"\nOfficial evaluation failed: {exc}")
        official = float("inf")

    # If something went terribly wrong return a safe uniform distribution
    if not np.isfinite(official):
        print("Returning a uniform fallback sequence.")
        best_seq = [1.0 / target_n] * target_n

    print("\nSearch completed.")
    return best_seq
