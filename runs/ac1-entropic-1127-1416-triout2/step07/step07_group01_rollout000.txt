The search is performed in three phases:

1. **Coarse multi‑resolution search** – a short optimisation on a small
   vector (default `n≈2048`).  Many random starts are generated, each of them
   is refined by a smooth log‑sum‑exp objective (Adam optimiser).  The best
   coarse vector is kept.

2. **Upsampling** – the coarse solution is interpolated to a longer vector
   (`n≈8192`).  Because the objective is scale‑invariant we only have to
   re‑normalise to a unit sum.

3. **Fine optimisation** – the up‑sampled vector is polished with
   *smooth* optimisation, a *projected sub‑gradient* method on the exact
   max‑convolution, and finally a stochastic hill‑climb that moves tiny
   amounts of mass between indices.

All phases respect the global wall‑clock limit (≈ 1000 s).  The routine
always returns the best sequence found (as a plain Python list).  The
implementation contains only top‑level functions, uses only `numpy`,
and makes no filesystem or network accesses.

```python
"""
Multi‑resolution optimisation of a non‑negative step function.

* Phase 1 – many cheap restarts on a small problem (n≈2048) using a
  smoothed log‑sum‑exp objective with Adam momentum.

* Phase 2 – interpolate the best coarse solution to a larger size
  (n≈8192) and renormalise.

* Phase 3 – fine polishing:
      – Adam on the smooth max,
      – projected sub‑gradient descent on the exact max,
      – stochastic hill‑climb (mass transfers).

The routine respects the 1000 s budget and returns the best vector
found as a Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp optimisation (Adam momentum)
# ----------------------------------------------------------------------


def smooth_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 140.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max (log‑sum‑exp) of a*a.
    Returns the best vector (unit‑sum) seen and its exact score.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal beta
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smooth max
        conv = convolution_fft(a, a)                 # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()

        # gradient of the smooth max (cross‑correlation)
        a_rev = a[::-1]
        corr = convolution_fft(a_rev, w)            # length 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]       # slice length n

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a = project_to_simplex(a_new)

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective sub‑gradient optimisation
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(conv(a,a)).
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry (2 * shifted copy of a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a = project_to_simplex(a_new)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # small jitter every 1000 steps to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Small random mass‑transfer moves with projection onto the simplex.
    Accepts a move only if it improves the exact score.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a, best_score

# Helper – generate different kinds of starts
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Dense Dirichlet‑like start."""
    a = np.random.rand(n)
    return project_to_simplex(a)


def random_start_sparse(n: int) -> np.ndarray:
    """Sparse start – a few equal non‑zero entries."""
    k = max(2, int(round(np.random.uniform(0.02, 0.10) * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑type shape: a_i ∝ 1/√(i·(n‑i+1))."""
    i = np.arange(1, n + 1)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    return project_to_simplex(a)


def generate_random_start(n: int) -> np.ndarray:
    """Pick a start type at random."""
    r = np.random.rand()
    if r < 0.5:
        return random_start_dense(n)
    if r < 0.8:
        return random_start_sparse(n)
    return arcsine_start(n)


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def upsample_to(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_len = v.shape[0]
    if old_len == new_len:
        return v.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(new_x, old_x, v)
    return project_to_simplex(v_new)
```
