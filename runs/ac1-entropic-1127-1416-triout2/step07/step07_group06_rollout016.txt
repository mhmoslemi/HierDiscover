    old_n = a_old.shape[0]
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)

    return a_new

def lp_step(a, t=0.2, time_deadline...):
    n = len(a)
    M = np.max(convolution_fft(a, a))
    # Build B matrix
    B = np.zeros((2*n-1, n), dtype=float)
    for j in range(n):
        B[j:j+n, j] = a
    # Solve LP
    c = -np.ones(n)
    res = linprog(c, A_ub=B, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return a  # fallback
    g0 = res.x
    sum_g = g0.sum()
    if sum_g <= 0:
        return a
    scale = 1.0 / sum_g
    g = g0 * scale
    # mixture
    a_new = (1 - t) * a + t * g
    a_new = project_to_simplex(a_new)
    return a_new

"""
Improved step‑function optimizer.

The optimizer follows a multi‑scale scheme:

* **Scale ladder** – start with a modest length (512) and progressively
  upscale the best distribution to larger lengths (1024, 2048, …).
* **Rich initial families** – for each size a collection of deterministic
  shapes (uniform, exponential, power‑law, arcsine) and a few random
  Dirichlet draws are tried.  The best one seeds the optimisation.
* **Flattening** – Adam minimises the variance of the self‑convolution,
  driving it towards a flat (≈uniform) profile.
* **Smooth‑max** – Adam minimises a log‑sum‑exp surrogate of the max‑convolution.
* **Linear‑programming boost** – one (or a few) LP steps maximise the total
  mass of a vector `g` under the constraint `f * g ≤ M` (where `M` is the
  current peak).  After rescaling `g` the mixture ``(1‑t)·f + t·g`` is
  projected back onto the probability simplex.  The LP solves a simple
  convex problem (SciPy HiGHS) and typically yields a noticeable drop in the
  peak.
* **Exact sub‑gradient** – fine‑tunes the true objective with a classic
  sub‑gradient descent.
* **Simulated‑annealing hill‑climb** – stochastic mass transfers finish
  the search.
* **Symmetry averaging** – optionally symmetrises the final vector.

All stages respect the global time budget (≈ 1000 s).  The code
requires only `numpy` and, if available, `scipy.optimize.linprog`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)

# Optimisation phases

def flatten_conv_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 3000,
    step0: float = 0.1,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam minimisation of the variance of the self‑convolution.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        mu = np.mean(conv)
        diff = conv - mu

        # gradient = 4 * Σ_j a_j * diff_{i+j}
        # Compute via convolution with reversed diff
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.5,
    eps_factor: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam minimisation of the smooth (log‑sum‑exp) surrogate for max‑convolution.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # epsilon decays but never below 1 % of maxc
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        # log‑sum‑exp weights
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)           # soft‑max, ∑ w = 1

        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter helps escape plateaus
        if t % 400 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact objective
        2·n·max(a⊛a) .
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i}
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random shake every thousand iterations
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list, float]:
    """
    Simulated‑annealing hill climbing.
    Moves are swaps or mass transfers between two indices.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12  # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]     # move 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0.0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional diversification
        if step % 5000 == 0:
            rnd = random_dense_vector(n) * 0.02
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Symmetrise the vector: (a + rev(a))/2 and re‑project.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

# Linear‑programming boost (optional, requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
except Exception:          # pragma: no cover
    linprog = None


def lp_improve(a: np.ndarray, t_mix: float = 0.2, time_deadline: float = None) -> np.ndarray:
    """
    One LP step: maximise Σ g_j subject to (a * g)_i ≤ M (all i),
    where M = max convolution of a with itself.
    The resulting vector `g` is rescaled to sum‑1 and mixed
    with the current `a` using mixing factor `t_mix`.
    """
    if linprog is None:
        # SciPy not available – skip LP step
        return a

    n = a.shape[0]
    M = np.max(convolution_fft(a, a))

    # Build the (2n‑1) × n constraint matrix:
    #   row i, column j  ->  a_{i‑j}  (zero if out of range)
    rows = []
    cols = []
    data = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(a.tolist())
    A_ub = np.zeros((2 * n - 1, n), dtype=float)
    # vectorised fill using slicing (faster than looping over rows)
    for j in range(n):
        A_ub[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n)           # minimise –Σ g_j  → maximise Σ g_j
    bounds = [(0, None)] * n

    # Solve the LP – use HiGHS if possible (default in recent SciPy)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0.0:
        return a

    # Rescale to unit sum (the LP maximises the sum under the peak bound)
    g = g0 / sum_g0

    # Blend with the current vector
    a_new = (1.0 - t_mix) * a + t_mix * g
    a_new = project_to_simplex(a_new)
    return a_new

def upscale_vector(a_old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from length `len(a_old)` to `new_n`,
    followed by simplex projection.
    """
    old_n = a_old.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = a_new / a_new.sum()
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns
    a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final call
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Scale ladder – start small, grow larger
    # ------------------------------------------------------------------
    scale_ladder = [512, 1024, 2048, 4096, 8192]

    # Most recent "best" vector at the current scale (None for the first level)
    prev_best = None

    for n in scale_ladder:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n--- Optimising length {n} ---")

        # ------------------------------------------------------------------
        # 1. Build a set of diverse initial vectors for this length
        # ------------------------------------------------------------------
        init_vectors = []

        # a) deterministic families
        uniform = np.full(n, 1.0 / n)
        init_vectors.append(uniform)

        i = np.arange(n)

        # arcsine‑like shape
        arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        arcs = arcs / arcs.sum()
        init_vectors.append(arcs)

        # exponential decays (left‑hand side heavy)
        for lam in [0.001, 0.005, 0.01, 0.05]:
            expv = np.exp(-lam * i)
            expv = expv / expv.sum()
            init_vectors.append(expv)

        # symmetric power‑law (product of distances to the ends)
        for p in [-1.5, -1.0, -0.5, 0.5, 1.0]:
            pw = ((i + 1) * (n - i)) ** p
            pw = pw / pw.sum()
            init_vectors.append(pw)

        # random Dirichlet draws
        for _ in range(3):
            init_vectors.append(random_dense_vector(n))

        # b) upscale the best vector from the previous scale, if any
        if prev_best is not None:
            up = upscale_vector(np.array(prev_best, dtype=float), n)
            init_vectors.append(up)

        # ------------------------------------------------------------------
        # 2. Run a quick filter on each candidate, keep the best one
        # ------------------------------------------------------------------
        for a0 in init_vectors:
            if time.time() >= deadline:
                break
            sc = compute_score(a0)
            if sc < best_score:
                best_score = sc
                best_seq = a0.tolist()
                print(f"  init candidate improved score={best_score:.6f}")

        # Use the current best (projected) as starting point for the heavy phases
        a_cur = np.array(best_seq, dtype=float)

        # ------------------------------------------------------------------
        # 3. Flattening phase (variance of convolution)
        # ------------------------------------------------------------------
        time_rem = deadline - time.time()
        if time_rem > 4.0:
            a_flat, sc_flat = flatten_conv_adam(
                a_cur,
                time_deadline=deadline - 1.0,
                max_iter=2500 if n <= 2048 else 1500,
                step0=0.14 if n <= 1024 else 0.08,
            )
            if sc_flat < best_score:
                best_score = sc_flat
                best_seq = a_flat.tolist()
                a_cur = a_flat.copy()
                print(f"  flattening improved score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 4. Smooth‑max surrogate optimisation
        # ------------------------------------------------------------------
        time_rem = deadline - time.time()
        if time_rem > 6.0:
            a_smooth, sc_smooth = smooth_max_adam(
                a_cur,
                time_deadline=deadline - 1.0,
                max_iter=5000 if n <= 2048 else 3000,
                step0=0.5,
                eps_factor=0.22,
            )
            if sc_smooth < best_score:
                best_score = sc_smooth
                best_seq = a_smooth.tolist()
                a_cur = a_smooth.copy()
                print(f"  smooth‑max improved score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 5. Optional LP boost – run a few quick LP updates
        # ------------------------------------------------------------------
        if linprog is not None and time.time() < deadline - 4.0:
            # perform up to three LP mixing steps with decreasing mixing weight
            for mix in (0.25, 0.15, 0.08):
                a_lp = lp_improve(a_cur, t_mix=mix, time_deadline=deadline - 1.0)
                sc_lp = compute_score(a_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.tolist()
                    a_cur = a_lp.copy()
                    print(f"  LP boost (mix={mix}) improved score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 6. Exact sub‑gradient descent (fine‑tune the true max)
        # ------------------------------------------------------------------
        time_rem = deadline - time.time()
        if time_rem > 8.0:
            a_sub, sc_sub = subgrad_optimize(
                a_cur,
                time_deadline=deadline - 2.0,
                max_iter=40000,
                step0=0.07,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                a_cur = a_sub.copy()
                print(f"  sub‑gradient improved score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 7. Simulated‑annealing hill‑climb
        # ------------------------------------------------------------------
        time_rem = deadline - time.time()
        if time_rem > 10.0:
            polished_seq, polished_sc = hill_climb_anneal(
                a_cur.tolist(),
                time_deadline=deadline - 2.0,
                max_steps=150_000,
            )
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished_seq
                a_cur = np.array(polished_seq, dtype=float)
                print(f"  annealing improved score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 8. Symmetry averaging (often helps the tail)
        # ------------------------------------------------------------------
        a_sym = final_symmetry_average(a_cur)
        sym_sc = compute_score(a_sym)
        if sym_sc < best_score:
            best_score = sym_sc
            best_seq = a_sym.tolist()
            a_cur = a_sym.copy()
            print(f"  symmetry averaging improved score={best_score:.6f}")

        # store the best vector at this scale for potential up‑scaling
        prev_best = best_seq

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = float("inf")

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
