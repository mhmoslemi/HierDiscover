"""
Propose a step‑function sequence with a low value of
``evaluate_sequence`` (i.e. minimise 2·n·max(conv(a,a)) with sum(a)=1).

Algorithm outline
-----------------
1. **Seeding** – generate a large, diverse pool of candidate
   probability vectors of length ``n=2048`` using several parametric
   families (uniform, arcsine, beta‑like, power‑law, exponential,
   Gaussian, Dirichlet and mixtures thereof).  Their fast surrogate
   score ``2·n·max(conv)`` is evaluated and the best few are retained.

2. **First Adam pass** – each retained seed is refined with a modest
   Adam optimisation of a smooth‑max surrogate (log‑sum‑exp of the
   convolution).  This yields a set of improved vectors.

3. **Focused Adam & random restarts** – the current best vector is
   jittered several times and Adam is run again with a longer schedule
   to escape possible local minima.

4. **Exact refinement** – the best candidate is fed to a projected
   sub‑gradient descent on the true max‑convolution objective, and then
   polished with an exponentiated‑gradient (mirror‑descent) step.

5. **Evaluation & return** – the official ``evaluate_sequence`` is called
   on the polished vector; the best one found within the global 1000 s
   budget is returned (a uniform fallback is guaranteed).

All steps respect the time limit and print progress information.
"""

import time
import numpy as np
from typing import List, Tuple

# Low‑level utilities

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution by real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector a is assumed to be normalised (sum = 1) – then the
    denominator of ``evaluate_sequence`` disappears and the score is
    exactly 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-lam·i) + exp(-lam·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian a_i ∝ exp(-(i-µ)²/(2σ²))."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    vals = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point from Dirichlet(alpha,...,alpha)."""
    vec = rng.dirichlet(np.full(n, alpha))
    # guard against pathological zeros (unlikely)
    if vec.sum() <= 0.0:
        return np.full(n, 1.0 / n)
    return vec / vec.sum()


def generate_mixture_three_shape(
    n: int,
    w_arc: float,
    w_uni: float,
    sigma: float,
) -> np.ndarray:
    """
    Mixture w_arc·arcsine + w_uni·uniform + (1‑w_arc‑w_uni)·Gaussian.
    """
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    gauss = generate_gaussian_shape(n, sigma)
    rem = 1.0 - w_arc - w_uni
    if rem < 0.0:
        rem = 0.0
    mix = w_arc * arc + w_uni * uni + rem * gauss
    total = mix.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return mix / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(v) to target_n, then renormalise."""
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 30_000,
    tau_start: float = 0.5,
    tau_end: float = 120.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimiser of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k)
    using the temperature schedule τ∈[tau_start, tau_end].
    The variables are logits w with a = softmax(w) → a ≥0, Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits (add tiny offset for log)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (exponential in iteration index)
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # a = softmax(w)
        w_shift = w - np.max(w)          # numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # soft‑max weights on τ·conv
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()

        # gradient w.r.t a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # chain rule through softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (learning‑rate decays ~1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution
    objective 2·n·max(conv(a,a)).
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask for entries attaining the maximum
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) restricted to original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """
    Exponentiated‑gradient polishing (mirror‑descent with KL geometry).
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

            cand = generate_gaussian_shape(n, sigma)
        elif typ == 4:          # Dirichlet draw (α in (0.05, 1))
            alpha = rng.uniform(0.05, 1.0)
            cand = generate_dirichlet_shape(n, alpha, rng)
        elif typ == 5:          # mixture arcsine / uniform / Gaussian
            w_arc = rng.random()
            w_uni = rng.random() * (1.0 - w_arc)
            sigma = rng.uniform(0.05 * n, 0.3 * n)
            cand = generate_mixture_three_shape(n, w_arc, w_uni, sigma)
        else:                   # random perturbation of arcsine
            jitter = rng.normal(scale=1e-3, size=n)
            cand = arc + jitter
            cand = np.clip(cand, 0.0, None)
            cand = cand / cand.sum()

        pool.append((compute_score(cand), cand))

    # keep the best few for the expensive Adam stage
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:12]]
    print(f"[seeding] generated {len(pool)} candidates, best surrogate = {pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. Light Adam refinement on the top seeds
    # ------------------------------------------------------------------
    refined: List[Tuple[float, np.ndarray]] = []
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"[Adam‑stage‑1] seed {idx}/{len(top_seeds)}")
        remaining = deadline - time.time()
        iters = int(min(30_000, remaining * 30))
        a_opt = adam_optimize_smooth_max(
            seed_vec,
            deadline,
            max_iters=iters,
            tau_start=0.5,
            tau_end=120.0,
            base_lr=0.25,
        )
        surr = compute_score(a_opt)
        print(f"    surrogate after Adam = {surr:.6f}")
        refined.append((surr, a_opt))

    refined.sort(key=lambda x: x[0])
    best_vec = refined[0][1] if refined else arc

    # ------------------------------------------------------------------
    # 3. Focused Adam with several random restarts from the current best
    # ------------------------------------------------------------------
    restarts = 5
    for r in range(restarts):
        if time.time() >= deadline - 5.0:
            break
        jitter = rng.normal(scale=2e-3, size=n)
        pert = best_vec + jitter
        pert = np.clip(pert, 0.0, None)
        pert = pert / pert.sum()
        remaining = deadline - time.time()
        iters = int(min(70_000, remaining * 30))
        a_opt = adam_optimize_smooth_max(
            pert,
            deadline,
            max_iters=iters,
            tau_start=0.5,
            tau_end=150.0,
            base_lr=0.22,
        )
        surr = compute_score(a_opt)
        print(f"[restart {r+1}] surrogate = {surr:.6f}")
        if surr < refined[0][0]:
            refined[0] = (surr, a_opt)
            best_vec = a_opt

    best_vec = refined[0][1]          # champion after Adam passes

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient descent + exponentiated‑gradient polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        print("[exact refinement] starting sub‑gradient descent")
        remaining = deadline - time.time()
        sub_iters = int(min(250_000, remaining * 30))
        a_sub = subgradient_descent_exact(best_vec, deadline, max_iter=sub_iters, step0=0.09)

        if time.time() < deadline - 3.0:
            print("[EG polish] starting exponentiated‑gradient")
            remaining = deadline - time.time()
            eg_iters = int(min(180_000, remaining * 30))
            a_sub = eg_refine(a_sub, deadline, max_iter=eg_iters, eta0=0.22)

        # official evaluation
        try:
            val = evaluate_sequence(a_sub.tolist())
            print(f"[final] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_sub.tolist()
                print("*** New global best discovered ***")
        except Exception as e:
            print("[error] evaluation failed:", e)

    # ------------------------------------------------------------------
    # 5. If still time, a quick global polish on the best vector found
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("[global polish] additional sub‑gradient + EG")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        sub_iters = int(min(300_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.07)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            eg_iters = int(min(200_000, remaining * 30))
            a = eg_refine(a, deadline, max_iter=eg_iters, eta0=0.18)

        final_val = evaluate_sequence(a.tolist())
        print(f"[global polish] final official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after global polish ***")

    # ------------------------------------------------------------------
    # 6. Guard – always return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        fallback = [1.0 / n] * n
        best_val = evaluate_sequence(fallback)
        best_seq = fallback
        print("[fallback] used uniform sequence, score =", best_val)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
