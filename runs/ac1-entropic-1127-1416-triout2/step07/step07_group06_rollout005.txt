"""
propose_candidate.py

A multi‑restart, symmetry‑aware optimiser for the step‑function problem.
The pipeline for each restart:

1. **Random (or best‑known) start** – Dirichlet on the probability simplex.
2. **Flattening** – minimise the variance of the self‑convolution with Adam.
   The iterate is symmetrised after every update and projected onto the simplex.
3. **Smooth‑max (log‑sum‑exp) phase** – minimise a smooth approximation of the
   ``max`` of the convolution.  The temperature ε is annealed exponentially
   to a very small value, giving a close approximation of the true objective.
4. **Exact sub‑gradient descent with back‑tracking line‑search** – uses the
   genuine max‑convolution sub‑gradient but adapts the step size so that each
   iteration strictly improves the score (when possible).
5. **Optional stochastic hill‑climb** – targeted mass‑transfer moves that are
   biased by the current worst (max) shift, plus a classic simulated‑annealing
   component.
6. **Final symmetry averaging** – forces a perfectly symmetric sequence and
   re‑projects onto the simplex (symmetry almost always lowers the peak).

All phases respect the global time budget (≈1000 s).  The best feasible
sequence encountered is returned as a plain ``list[float]`` ready for the
evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a ⊛ b using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for vectors with ``sum(a)=1``.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex (non‑negative, sum = 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize_and_project(v: np.ndarray) -> np.ndarray:
    """
    Enforce a[i] = a[n‑1‑i] and project back onto the simplex.
    """
    sym = (v + v[::-1]) * 0.5
    return project_to_simplex(sym)


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet with α=1)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Phase‑2 – flattening (variance of the convolution) using Adam
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 4000,
    step0: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimises the variance of ``a⊛a``.
    Each update is symmetrised and projected onto the simplex.
    Returns the best vector (exact score) observed.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)                 # length 2n‑1
        mu = np.mean(conv)                           # = 1/(2n‑1) for sum=1
        diff = conv - mu

        # gradient = 4 * Σ_j a_j * diff_{i+j}
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)      # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = symmetrize_and_project(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase‑3 – smooth max (log‑sum‑exp) minimisation
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 8000,
    step0: float = 0.55,
    eps_start_factor: float = 0.5,
    eps_decay: float = 0.998,
    eps_min_factor: float = 1e-4,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of ``max(a⊛a)``:
        f(a) = ε·log Σ_k exp(conv_k/ε)
    The temperature ε is annealed exponentially to a tiny fraction of ``max(conv)``.
    Symmetrisation and simplex projection are applied after every update.
    Returns the best vector (exact score) seen.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # exponential decay of ε, never smaller than a tiny fraction of maxc
        eps = maxc * max(eps_start_factor * (eps_decay ** (t - 1)), eps_min_factor)

        # stable log‑sum‑exp
        shifted = conv - maxc               # max of shifted is 0
        exp_vals = np.exp(shifted / eps)   # all ≤ 1
        w = exp_vals / np.sum(exp_vals)    # soft‑max weights, Σ w = 1

        # gradient of the smooth max: 2 * (a * w) convolution
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = symmetrize_and_project(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to leave flat regions
        if t % 600 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = symmetrize_and_project(a + jitter)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase‑4 – exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgrad_optimize_line_search(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true objective 2·n·max(a⊛a).
    At each iteration a back‑tracking line‑search (geometric halving) is used
    to guarantee a non‑increasing exact score whenever possible.
    Symmetrisation + simplex projection are applied after every trial step.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑convolution at shift k:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # descent direction = -g
        direction = -g

        # back‑tracking line‑search
        alpha = step0 / np.sqrt(t)
        improved = False
        while alpha > 1e-12:
            a_trial = a + alpha * direction
            a_trial = symmetrize_and_project(a_trial)
            sc = compute_score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # if no improvement, still project (to keep feasibility) but keep a unchanged
            a = symmetrize_and_project(a)

        # occasional small random shake to escape plateaus
        if t % 1500 == 0:
            shake = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = symmetrize_and_project(a + shake)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase‑5 – stochastic hill‑climb (targeted + classic moves)
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 180_000,
) -> tuple[list, float]:
    """
    Simulated‑annealing hill‑climber.
    Three move types:
        1) swap two entries,
        2) transfer a fraction of mass i → j,
        3) *targeted* transfer from an index that contributes most to the
           current worst shift to an index that contributes least.
    Acceptance follows a linear cooling schedule.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12                     # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        # linear temperature decay
        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        move_type = np.random.rand()
        if move_type < 0.4:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.8:
            # generic transfer
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 1e-12:
                delta = 0.05 * cand[i]                      # 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        else:
            # targeted transfer using the current worst shift
            conv = convolution_fft(cand, cand)
            k = int(np.argmax(conv))
            # contributions to the worst shift
            contrib = np.zeros_like(cand)
            for idx in range(n):
                j = k - idx
                if 0 <= j < n:
                    contrib[idx] = cand[idx] * cand[j]
            # source: index with largest contribution
            i_src = int(np.argmax(contrib))
            # destination: index with smallest value (but not the same)
            non_zero = np.where(cand > 1e-12)[0]
            if len(non_zero) > 1:
                # pick a destination far from the source contribution
                cand_vals = cand.copy()
                cand_vals[i_src] = np.max(cand_vals) + 1.0   # ignore source
                i_dst = int(np.argmin(cand_vals))
                delta = 0.05 * cand[i_src]
                cand[i_src] -= delta
                cand[i_dst] += delta
                if cand[i_src] < 0.0:
                    cand[i_src] = 0.0

        cand = symmetrize_and_project(cand)

        cur_sc = compute_score(cand)
        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept (always keeps feasibility)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional larger perturbation to diversify the search
        if step % 5000 == 0:
            rnd = random_dense_vector(n) * 0.02   # 2 % random noise
            cand = best_a + rnd
            cand = symmetrize_and_project(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Enforce perfect symmetry and re‑project onto the simplex."""
    return symmetrize_and_project(a)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine respects the given time budget (≈ 1000 s) and returns a
    plain Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final official evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameter ranges and helper to update the incumbent
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    def maybe_update(candidate, score, source):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"[{source}] improved: n={len(candidate)}, score={best_score:.6f}")

    outer_iter = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (multiple restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        outer_iter += 1
        n = int(np.random.choice(lengths))

        # --------------------------------------------------------------
        # 1. Initial vector (occasionally reuse a truncation of the incumbent)
        # --------------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.2 and len(best_seq) >= n:
            a0 = np.array(best_seq[:n], dtype=float)
            a0 = project_to_simplex(a0)
        else:
            a0 = random_dense_vector(n)

        # keep a quick check – if the random start already beats incumbent
        cur_sc = compute_score(a0)
        maybe_update(a0.tolist(), cur_sc, "init")

        # --------------------------------------------------------------
        # 2. Flattening phase
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 6.0:
            break
        a_flat, sc_flat = flatten_conv_adam(
            a0,
            time_deadline=deadline - 2.0,
            max_iter=4000,
            step0=0.15,
        )
        maybe_update(a_flat.tolist(), sc_flat, "flatten")

        # --------------------------------------------------------------
        # 3. Smooth‑max (log‑sum‑exp) phase
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        a_smax, sc_smax = smooth_max_adam(
            a_flat,
            time_deadline=deadline - 3.0,
            max_iter=8000,
            step0=0.55,
            eps_start_factor=0.5,
            eps_decay=0.998,
            eps_min_factor=1e-4,
        )
        maybe_update(a_smax.tolist(), sc_smax, "smooth‑max")

        # --------------------------------------------------------------
        # 4. Exact sub‑gradient descent with line‑search
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 12.0:
            a_sub, sc_sub = subgrad_optimize_line_search(
                a_smax,
                time_deadline=deadline - 5.0,
                max_iter=100_000,
                step0=0.07,
            )
            maybe_update(a_sub.tolist(), sc_sub, "sub‑grad")

        # --------------------------------------------------------------
        # 5. Stochastic hill‑climb (shortened if time is low)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            seq_to_polish = best_seq if best_seq is not None else a_sub.tolist()
            polished_seq, polished_sc = hill_climb_anneal(
                seq_to_polish,
                time_deadline=deadline - 3.0,
                max_steps=150_000,
            )
            maybe_update(polished_seq, polished_sc, "anneal")

        # --------------------------------------------------------------
        # 6. Final cheap symmetry averaging (often helpful)
        # --------------------------------------------------------------
        if best_seq is not None:
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            maybe_update(a_sym.tolist(), sym_sc, "symmetry")

    # ------------------------------------------------------------------
    # Safety net – if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
