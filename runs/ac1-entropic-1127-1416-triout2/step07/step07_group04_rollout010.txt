"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. **Meta‑search (cheap surrogate)**  
   Generate many symmetric candidate shapes on a fixed grid (default 4096
   points): pure arcsine, uniform, mixtures of arcsine+uniform,
   parametrised power‑law shapes and symmetric Beta shapes.  For each
   candidate compute a fast surrogate `2·n·max(conv(a,a))` using FFTs.
   Keep the best few (default 4) according to the surrogate.

2. **Local refinement**  
   Starting from each top surrogate candidate run a three‑phase
   optimisation that respects the exact objective:
   * **Smooth‑max Adam** – minimise a log‑sum‑exp approximation of the
     max‑convolution, with a warm‑up temperature schedule τ∈[0.5,τ_max]
     (τ_max≈500) and a decaying Adam learning‑rate.
   * **Projected sub‑gradient descent** on the true max‑convolution.
   * **Exponentiated‑gradient (mirror descent)** for a final KL‑prox
     polishing.

   All stages stop early when the global time budget (≈1000 s) is
   exhausted.

3. **Selection**  
   Evaluate the refined sequences with the official `evaluate_sequence`
   function and keep the best one.  If time permits a final polishing
   of the global best is performed.

The code respects all competition rules: no file I/O, only NumPy and
standard library, and it returns a non‑empty list of non‑negative
floats.
"""

import time

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        # Uniform distribution if the vector is non‑positive
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    Input a is assumed to be normalised (Σa = 1).  Returns
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-9) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{-p}, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-9) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) → new_n points, then renormalise.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of a smooth log‑sum‑exp approximation of max(conv).
    The optimisation variable is the logits w (softmax → a) which
    automatically satisfies a ≥ 0, Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a (length L)
        conv = convolution_fft(a, a)

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)   # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (optional, can be removed)
        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

        deadline,
        max_iters=max_iters_adam,
        tau_start=0.5,
        tau_end=500.0,
        base_lr=0.5,
    )

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 2.0:
        return a

    max_iters_sub = int(min(300_000, remaining * 30))
    a = subgradient_descent_exact(a, deadline, max_iter=max_iters_sub, step0=0.07)

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient polishing
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 1.0:
        return a

    max_iters_eg = int(min(250_000, remaining * 30))
    a = eg_refine(a, deadline, max_iter=max_iters_eg, eta0=0.25)

    return a


# ----------------------------------------------------------------------
# Candidate generation (cheap surrogate)
# ----------------------------------------------------------------------


def build_candidate_pool(n: int, rng: np.random.Generator) -> List[Tuple[str, np.ndarray]]:
    """
    Produce a list of candidate shapes together with a textual description.
    The pool contains arcsine, uniform, mixtures, power‑law and Beta families.
    """
    candidates = []

    # Base shapes
    arcs = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)

    candidates.append(("arcsine", arcs))
    candidates.append(("uniform", uni))

    # Mixtures arcsine + uniform
    for w in np.linspace(0.0, 1.0, 21):
        mix = w * arcs + (1.0 - w) * uni
        # Already normalised (both components sum to 1)
        candidates.append((f"mix_{w:.2f}", mix))

    # Power‑law shapes (generalised arcsine)
    for p in np.linspace(0.30, 0.80, 25):
        shape = generate_power_shape(n, p, eps=1e-9)
        candidates.append((f"power_{p:.3f}", shape))

    # Symmetric Beta shapes
    for a in np.linspace(0.30, 1.50, 25):
        shape = generate_beta_shape(n, a, a, eps=1e-9)
        candidates.append((f"beta_{a:.3f}", shape))

    # Random perturbations of the arcsine shape (helps escape local minima)
    for _ in range(8):
        pert = arcs + 0.005 * rng.standard_normal(n)
        pert = np.maximum(pert, 0.0)
        pert = project_to_simplex(pert)
        candidates.append(("arcsine_perturb", pert))

    return candidates


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine follows the meta‑search → local‑refinement strategy
    described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a small safety margin
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Fallback to any previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose a working resolution
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # Use the finer grid if we still have enough time
    target_n = 4096 if remaining > 250.0 else 2048
    print(f"\nWorking resolution: n = {target_n}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap surrogate meta‑search
    # ------------------------------------------------------------------
    pool = build_candidate_pool(target_n, rng)
    surrogates = []
    for desc, arr in pool:
        a = arr.astype(float)
        a = a / a.sum()
        surrogate = compute_score(a)
        surrogates.append((surrogate, desc, a))

    surrogates.sort(key=lambda x: x[0])
    top_k = 4
    top_candidates = surrogates[:top_k]
    print(f"Top {top_k} surrogate candidates (lower is better):")
    for i, (surr, desc, _) in enumerate(top_candidates, 1):
        print(f"  {i}. {desc:<15} surrogate = {surr:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – local refinement of each top candidate
    # ------------------------------------------------------------------
    for idx, (surr, desc, a0) in enumerate(top_candidates, 1):
        if time.time() >= deadline - 3.0:
            break  # not enough time for another full run

        print(f"\nRefining candidate {idx}/{top_k}: {desc} (surrogate {surr:.6f})")
        a_refined = refine_sequence(a0, deadline)

        # Official evaluation (might be slow, but we have few candidates)
        try:
            val = evaluate_sequence(a_refined.tolist())
            print(f"  Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_refined.tolist()
                print("*** New global best found! ***")
            # Early exit if we already beat the target
            if best_val < 1.503:
                print("Target achieved – stopping early.")
                break
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Phase 2 – final global polish (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal global polishing of the best sequence.")
        a = np.array(best_seq, dtype=float)

        # One more pass of sub‑gradient + EG with tighter schedules
        remaining = deadline - time.time()
        if remaining > 4.0:
            max_iters_sub = int(min(200_000, remaining * 30))
            a = subgradient_descent_exact(a, deadline,
                                          max_iter=max_iters_sub,
                                          step0=0.05)

        remaining = deadline - time.time()
        if remaining > 2.0:
            max_iters_eg = int(min(150_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iters_eg,
                          eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        print("No valid sequence found – falling back to uniform distribution.")
        fallback_n = target_n
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
