def lp_iterative_improvement(a, time_deadline):
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    while time.time() < time_deadline:
        # compute conv and M
        conv = convolution_fft(a, a)
        M = conv.max()
        # solve LP for g
        try:
            g = solve_lp_for_g(a, M)
        except Exception:
            break
        if g is None or not np.any(g):
            break
        # normalize g
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g / sum_g
        # mixing factor
        t = 0.2  # could adapt
        a_new = (1 - t) * a + t * g
        # project onto simplex
        a_new = project_to_simplex(a_new)
        a = a_new
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

            # if no improvement after some iterations, break
            # maybe reduce t
            t = max(t * 0.5, 0.01)
            if t < 0.01:
                break
    return best_a, best_score

"""
Search for a non‑negative step‑function (probability vector) that minimises
`evaluate_sequence`.  The algorithm works in several stages:

1.  Build a modest initial pool of deterministic shapes (uniform, Gaussian,
    triangular, Hann, Blackman) and, if present, fall back to the previously
    discovered `height_sequence_1`.  The best of these is kept.
2.  (Optional) Linear‑programming improvement:  for the current vector `a`
    solve  max ∑g_j   s.t.   a∗g ≤ M  (where M = max(a∗a)).  Mix the solution
    back into `a` and repeat while time permits.  This pushes mass away from
    positions that create the current peak.
3.  Many short Adam‑accelerated smooth‑max optimisations from random starts.
    The smoothed objective is a log‑sum‑exp approximation of the true
    max‑convolution; the routine projects onto the probability simplex after
    each Adam step.
4.  A projected sub‑gradient descent on the exact max‑objective for fine
    polishing.
5.  A stochastic hill‑climb with simulated‑annealing style acceptance that
    swaps or transfers tiny amounts of mass between two positions.

All stages respect the overall wall‑clock budget (≈1000 s).  The best vector
found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used only for the LP stage)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog  # type: ignore
except Exception:  # pragma: no cover
    linprog = None  # LP stage will be skipped if SciPy is unavailable

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise just in case of round‑off
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


def symmetrize_and_normalise(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry and re‑normalise to sum 1."""
    v = (v + v[::-1]) / 2.0
    s = v.sum()
    if s > 0:
        v /= s
    else:
        v = np.full_like(v, 1.0 / v.shape[0])
    return v


# ----------------------------------------------------------------------
# Smoothed max‑objective optimisation (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 80.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
    enforce_symmetry: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a,a)) using Adam momentum.
    Returns the best vector encountered together with its exact score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smoothed objective ---------------------------------------
        conv = convolution_fft(a, a)           # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()    # a probability distribution

        # ---- gradient (cross‑correlation) -----------------------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # slice length n

        # ---- Adam update -----------------------------------------------
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)
        if enforce_symmetry:
            a_new = symmetrize_and_normalise(a_new)

        a = a_new

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation of the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
    enforce_symmetry: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)
        if enforce_symmetry:
            a_new = symmetrize_and_normalise(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)
            if enforce_symmetry:
                a = symmetrize_and_normalise(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb polishing (small mass‑transfer moves + annealing)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb with simulated‑annealing acceptance.
    Swaps or transfers a small fraction of mass, always projecting onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved
    temperature = 0.02  # initial temperature for annealing

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        # accept if better, otherwise accept with probability exp(-Δ/temperature)
        if cur_score < best_score or np.random.rand() < np.exp(-(cur_score - best_score) / max(temperature, 1e-12)):
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                step_scale = min(step_scale * 1.05, 0.5)
                temperature = max(temperature * 0.95, 1e-5)
            else:
                # accepted uphill – keep the candidate but don’t update best
                best_a = cand
                # slightly increase temperature on uphill moves
                temperature = min(temperature * 1.05, 0.5)

        else:
            # rejected – shrink step size, cool down
            step_scale *= 0.97
            temperature = max(temperature * 0.99, 1e-5)

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Deterministic window generators (used for initial candidates)
# ----------------------------------------------------------------------


def window_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def window_gaussian(n: int, sigma_ratio: float = 0.15) -> np.ndarray:
    x = np.arange(n)
    mu = (n - 1) / 2.0
    sigma = sigma_ratio * n
    w = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return project_to_simplex(w)


def window_triangle(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    w = np.concatenate([inc, dec])
    return project_to_simplex(w)


def window_hann(n: int) -> np.ndarray:
    x = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    return project_to_simplex(w)


def window_blackman(n: int) -> np.ndarray:
    x = np.arange(n)
    w = (
        0.42
        - 0.5 * np.cos(2 * np.pi * x / (n - 1))
        + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    )
    return project_to_simplex(w)


# ----------------------------------------------------------------------
# Linear‑programming helper (LP stage)
# ----------------------------------------------------------------------


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max∑g_j  s.t.  f∗g ≤ M,  g ≥ 0.
    Returns the optimal g (may be dense).  Raises if LP fails.
    """
    if linprog is None:
        raise RuntimeError("SciPy linprog not available")
    n = f.shape[0]
    m = 2 * n - 1

    rows = np.arange(m)[:, None]            # shape (m,1)
    cols = np.arange(n)                     # shape (n,)
    offset = rows - cols                    # shape (m,n)
    mask = (offset >= 0) & (offset < n)

    A = np.zeros((m, n), dtype=float)
    A[mask] = f[offset[mask]]

    c = -np.ones(n, dtype=float)            # maximise ∑g → minimise -∑g
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    return res.x


def lp_iterative_improvement(a: np.ndarray, time_deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above and mix the solution back.
    Returns an improved vector and its exact score.
    """
    if linprog is None:
        return a, compute_score(a)  # LP not possible – return unchanged

    best_a = a.copy()
    best_score = compute_score(best_a)

    # mixing factor (how far we move towards the LP solution each iteration)
    mix = 0.20

    while time.time() < time_deadline:
        conv = convolution_fft(a, a)
        M = conv.max()
        try:
            g = solve_lp_for_g(a, M)
        except Exception:
            break
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g  # renormalise to sum 1
        a = (1.0 - mix) * a + mix * g
        a = project_to_simplex(a)  # keep on the simplex
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # if no improvement for several iterations, reduce the mixing step
            mix = max(mix * 0.7, 0.05)
            if mix < 0.05:
                break
    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises the provided
    `evaluate_sequence`.  The routine respects the 1000 s wall‑clock budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------------------
    # Choose a reasonable length (larger gives more freedom)
    # -------------------------------------------------------------
    # Use a longer vector only if enough time remains; otherwise be safe with 1024.
    if (deadline - time.time()) > 200:
        n = 2048
    else:
        n = 1024
    print(f"Chosen sequence length = {n}")

    # -------------------------------------------------------------
    # Build a pool of deterministic initial candidates
    # -------------------------------------------------------------
    init_candidates = [
        window_uniform(n),
        window_gaussian(n),
        window_triangle(n),
        window_hann(n),
        window_blackman(n),
    ]

    # fall back to the previously discovered sequence if present
    try:
        fb = np.array(height_sequence_1, dtype=float)  # type: ignore
        if fb.size > n:
            fb = fb[:n]
        elif fb.size < n:
            fb = np.pad(fb, (0, n - fb.size), mode="constant")
        fb = project_to_simplex(fb)
        init_candidates.append(fb)
    except Exception:
        pass

    # evaluate the deterministic pool
    best_a = None
    best_score = float("inf")
    for cand in init_candidates:
        scr = compute_score(cand)
        if scr < best_score:
            best_score = scr
            best_a = cand.copy()
    print(f"Best deterministic start score = {best_score:.6f}")

    # -------------------------------------------------------------
    # 0. Optional linear‑programming improvement
    # -------------------------------------------------------------
    lp_deadline = min(deadline - 150.0, time.time() + 30.0)
    if time.time() < lp_deadline:
        best_a, best_score = lp_iterative_improvement(best_a, lp_deadline)
        print(f"After LP stage score = {best_score:.6f}")

    # -------------------------------------------------------------
    # 1. Random restarts + Adam‑accelerated smooth optimisation
    # -------------------------------------------------------------
    # allocate about 55 % of the remaining budget to these short runs
    remaining = deadline - time.time()
    smooth_deadline = time.time() + remaining * 0.55

    def random_start():
        """Generate a random probability vector (dense or sparse)."""
        if np.random.rand() < 0.7:  # dense most of the time
            a = np.random.rand(n)
        else:  # sparse start
            k = max(2, int(round(np.random.uniform(0.02, 0.3) * n)))
            idx = np.random.choice(n, size=k, replace=False)
            a = np.zeros(n, dtype=float)
            a[idx] = 1.0 / k
        return project_to_simplex(a)

    while time.time() < smooth_deadline:
        a0 = random_start()
        per_run_deadline = min(smooth_deadline, time.time() + 5.0)  # ~5 s per run
        a_opt, scr = smooth_gradient_optimize(
            a0,

            max_iters=200_000,
            beta_start=6.0,
            beta_end=150.0,
            step0=0.9,
            enforce_symmetry=False,
        )
        if scr < best_score:
            best_score = scr
            best_a = a_opt.copy()
            print(f"New best after smooth restart: score = {best_score:.6f}")

    # -------------------------------------------------------------
    # 2. Sub‑gradient refinement on the current best vector
    # -------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 60.0
        refined_a, refined_score = subgrad_optimize(
            best_a,
            time_deadline=sub_deadline,
            max_iter=300_000,
            step0=0.07,
            enforce_symmetry=False,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_a = refined_a.copy()
            print(f"Improved by sub‑grad: score = {best_score:.6f}")

    # -------------------------------------------------------------

            time_deadline=deadline,
            max_steps=800_000,

    print(f"\nFinished search. Final reported score = {final_score:.6f}")
