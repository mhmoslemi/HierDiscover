    pop = [random_dirichlet(n) for _ in range(population_size)]
    # optionally include arcsine vector
    pop[0] = arcsine_initialisation(n)
    scores = [compute_score(ind) for ind in pop]
    best_idx = np.argmin(scores)
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    gen = 0
    while time.time() < deadline and gen < max_generations:
        for i in range(population_size):
            # choose three distinct random indices not i
            idxs = list(range(population_size))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # optionally ensure nonnegative: clip
            mutant = np.maximum(mutant, 0)
            # crossover
            cross = np.random.rand(n) < CR
            # ensure at least one dimension from mutant
            if not np.any(cross):
                cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])
            # project to simplex
            trial = project_to_simplex(trial)
            trial_score = compute_score(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score

                    best_score = trial_score
                    best = trial.copy()
        gen += 1
        if gen % 5 == 0:
            print(f"[DE] n={n}, gen={gen}, best={best_score:.6f}, time={time.time()-start_time:.1f}s")

"""
Propose a candidate step‐function that minimizes
    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The search combines:

* random restarts with a Differential‑Evolution (DE) population on the simplex
* a simple targeted hill‑climber that moves mass away from the shift producing the
  current maximum of the convolution,
* optional symmetry averaging,
* a final call to the provided ``evaluate_sequence`` for the official score.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
so that the fast surrogate ``compute_score`` equals the true
``evaluate_sequence`` value.  The algorithm respects the 1000 s budget
and reports progress via ``print`` statements.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to ``x``."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two 1‑D arrays via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` when ``sum(a) == 1``.
    Returns 2·n·max(conv) where conv = a⋆a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        # zero vector -> uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - 1.0                # cumulative sum – 1
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial vectors on the simplex
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine‑type density:
        w_i ∝ 1 / √( i·(n‑1‑i) )
    (with a tiny epsilon to avoid division by zero) and project to simplex.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_simplex_vector(n: int) -> np.ndarray:
    """Draw a random point on the simplex – simply normalise uniform noise."""
    v = np.random.rand(n)
    return project_to_simplex(v)


# ----------------------------------------------------------------------
# Differential‑Evolution optimiser on the simplex
# ----------------------------------------------------------------------


def de_optimize(
    n: int,
    pop_size: int,
    max_generations: int,
    deadline: float,
    F: float = 0.7,
    CR: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Classic DE (``DE/rand/1/bin``) constrained to the probability simplex.
    Returns the best vector discovered and its fast score.
    """
    # initialise population (one member is the arcsine start, the rest are random)
    population = [random_simplex_vector(n) for _ in range(pop_size)]
    population[0] = arcsine_initialisation(n)

    scores = [compute_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_vec = population[best_idx].copy()
    best_score = scores[best_idx]

    generation = 0
    while generation < max_generations and time.time() < deadline:
        for i in range(pop_size):
            # Choose three distinct indices different from i
            choices = list(range(pop_size))
            choices.remove(i)
            r1, r2, r3 = np.random.choice(choices, 3, replace=False)

            mutant = population[r1] + F * (population[r2] - population[r3])
            # enforce non‑negativity before projection
            mutant = np.maximum(mutant, 0.0)

            # binomial crossover
            cross = np.random.rand(n) < CR
            if not np.any(cross):
                cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = project_to_simplex(trial)

                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:

                    best_vec = trial.copy()

        generation += 1
        # occasional progress report
        if generation % 20 == 0:

        # safety break if we are about to run out of time
        if time.time() + 0.1 >= deadline:
            break

# ----------------------------------------------------------------------
# Simple hill‑climber with mass‑transfer moves (targeting the max‑shift)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climber that only accepts strictly improving moves.
    Two move types are used:
        • swap two entries,
        • transfer a fraction of mass from a pair that participates
          in the current maximal convolution shift to a random third index.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)
    step_frac = 0.07  # start with a modest transfer fraction

    steps = 0
    while steps < max_steps and time.time() < deadline:
        # compute the current shift giving the maximal convolution value
        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))

        # identify the pair (i, j = k_max - i) that contributes the most
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        pair_idxs = idx[mask]
        if pair_idxs.size > 0:
            products = a[pair_idxs] * a[j[mask]]
            i = pair_idxs[np.argmax(products)]
            j_i = k_max - i
        else:
            # degenerate case – fall back to a random swap
            i, j_i = np.random.randint(n, size=2)

        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap
            cand[i], cand[j_i] = cand[j_i], cand[i]
        else:
            # transfer
            if i != j_i:
                delta = step_frac * min(cand[i], cand[j_i])
                if delta > 0.0:
                    # destination index (different from i and j_i)
                    l = np.random.randint(n)
                    while l == i or l == j_i:
                        l = np.random.randint(n)
                    cand[i] -= delta
                    cand[j_i] -= delta
                    cand[l] += 2.0 * delta

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.08, 0.5)  # become bolder
        else:
            step_frac *= 0.96  # become more cautious

        a = cand
        steps += 1

    return best_a, best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace ``a`` by (a + rev(a))/2 and re‑project to simplex."""
    sym = (a + a[::-1]) * 0.5
    return project_to_simplex(sym)

"""
Propose a candidate step‐function that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The search combines

* random restarts with a Differential‑Evolution (DE) population on the simplex,
* a cheap hill‑climber that moves mass away from the shift producing the current
  maximal convolution value,
* optional symmetry averaging,
* a final call to the supplied ``evaluate_sequence`` for the official score.

All vectors are kept on the probability simplex (non‑negative, Σa = 1) so that
the fast surrogate ``compute_score`` equals the true ``evaluate_sequence``
value.  The routine respects the 1000 s time budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` when sum(a)=1 → 2·n·max(conv)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initial vectors on the simplex
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine–like density:
        w_i ∝ 1 / √(i·(n‑1‑i)).
    The vector is projected to the simplex.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_simplex_vector(n: int) -> np.ndarray:
    """Uniformly sampled point on the simplex (simple normalisation of uniform noise)."""
    v = np.random.rand(n)
    return project_to_simplex(v)
