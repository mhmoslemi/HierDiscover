"""
Propose a step function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Fast utilities** – FFT based linear convolution and a cheap proxy
   ``compute_score(a) = 2·n·max(conv(a,a))`` (the official score when the
   sequence is normalised to sum = 1).

2. **Diverse parametric families** – generate candidates from several
   one‑parameter families (power‑law, beta‑distribution, geometric,
   monotone power, mixtures of the arcsine envelope and uniform/spike
   components).  The parameters are sampled uniformly at random in
   broad intervals, producing both symmetric and strongly asymmetric
   shapes.

3. **Random search** – for the allotted wall‑clock time a large
   number of candidates is sampled (different lengths 1024/2048/4096
   are tried).  Each candidate is normalised to sum = 1 and its fast
   score is evaluated.  The best few candidates are kept for polishing.

4. **Projected sub‑gradient polishing** – the current best candidate is
   refined by a long run of projected sub‑gradient descent on the true
   non‑smooth objective (the same routine used in the reference
   solution).  This respects the simplex constraints and typically
   reduces the score further.

5. **Final check** – if no candidate was found (should never happen) a
   uniform sequence of length 512 is returned as a safe fall‑back.

The whole routine respects the 1000 s wall‑clock budget and always
returns a non‑empty list of non‑negative numbers.
"""

from typing import List
import numpy as np

# FFT‑based convolution utilities (shared by all phases)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for the official objective when ``sum(a) = 1``.
    Returns   2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # mask of (near)‑maximal entries; tolerance avoids numerical noise
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # convolution with reversed mask gives the sub‑gradient
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Parametric families of candidate shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_geometric_shape(n: int, r: float) -> np.ndarray:
    """Monotonically decreasing geometric law."""
    if not (0.0 < r < 1.0):
        return np.full(n, 1.0 / n)
    vals = r ** np.arange(n, dtype=float)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mono_power_shape(n: int, p: float) -> np.ndarray:
    """Pure power law a_i ∝ (i+1)^{-p} (asymmetric, decreasing)."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / ((i + 1.0) ** p)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_spike_mixture(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and two end‑spikes."""
    arc = arcsine_initial(n)
    spikes = np.zeros(n, dtype=float)
    spikes[0] = spikes[-1] = 0.5
    mix = weight_arc * arc + (1.0 - weight_arc) * spikes
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The routine respects the 1000 s wall‑clock budget and always returns
    a non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.3  # tiny safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 0.  Fallback to a previously stored solution if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray([float(x) for x in globals()["height_sequence_1"]], dtype=float)
            if cand.size > 0:
                cand = cand / cand.sum()
                val = compute_score(cand)
                print(f"Fallback (height_sequence_1) proxy score = {val:.6f}")
                if val < best_score:
                    best_score, best_seq = val, cand
                    best_n = cand.size
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Randomised search over a few parametric families
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]  # modest sizes that still give a good bound
    families = ["power", "beta", "geom", "mono_pow", "mix_arc_uni", "mix_arc_spike", "arc"]

    while time.time() < deadline - 8.0:          # reserve a few seconds for polishing
        # pick a random length (bias towards larger for finer resolution)
        n = int(rng.choice(lengths, p=np.array([1, 2, 3]) / 6.0))

        # pick a random family
        fam = rng.choice(families)

        if fam == "power":
            p = rng.uniform(0.05, 3.0)
            eps = rng.uniform(0.0, 2.0)
            a = generate_power_shape(n, p, eps)

        elif fam == "beta":
            a_par = rng.uniform(0.1, 5.0)
            b_par = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            a = generate_beta_shape(n, a_par, b_par, eps)

        elif fam == "geom":
            r = rng.uniform(0.5, 0.9999)          # r close to 1 gives near‑uniform
            a = generate_geometric_shape(n, r)

        elif fam == "mono_pow":
            p = rng.uniform(0.05, 3.0)
            a = generate_mono_power_shape(n, p)

        elif fam == "mix_arc_uni":
            w = rng.uniform(0.0, 1.0)
            a = generate_mixture_shape(n, w)

        elif fam == "mix_arc_spike":
            w = rng.uniform(0.0, 1.0)
            a = generate_spike_mixture(n, w)

        else:      # arcsine baseline
            a = arcsine_initial(n)

        # sanity check – ensure a is a valid probability vector
        if a.ndim != 1 or a.size != n:
            continue
        if not np.isfinite(a).all() or (a < 0).any():
            continue
        total = a.sum()
        if total <= 0.0:
            continue
        a = a / total

        # fast proxy score
        scr = compute_score(a)

        # keep the best overall
        if scr < best_score:
            best_score, best_seq, best_n = scr, a.copy(), n
            print(f"New global best (proxy) = {best_score:.6f}  n={best_n}")

        # occasional aggressive polishing of a promising candidate
        # (run a short sub‑gradient pass only if it looks promising)
        if scr < best_score * 0.98:
            sub_deadline = time.time() + min(3.0, deadline - time.time() - 5.0)
            if sub_deadline > time.time():
                a_polished = subgradient_descent_exact(
                    a,
                    sub_deadline,
                    max_iter=30_000,
                    step0=0.07,
                )
                scr_pol = compute_score(a_polished)
                if scr_pol < best_score:
                    best_score, best_seq, best_n = scr_pol, a_polished.copy(), n
                    print(f"Polished candidate improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Global polishing of the best candidate found so far
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n=== Final global polishing ===")
        final_seq = subgradient_descent_exact(
            best_seq,
            deadline,
            max_iter=200_000,
            step0=0.07,
        )
        final_score = compute_score(final_seq)
        if final_score < best_score:
            best_seq, best_score = final_seq, final_score
            print(f"*** Final polishing improved best proxy score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        n0 = 512
        best_seq = np.full(n0, 1.0 / n0)
        best_score = compute_score(best_seq)
        print("Fallback to uniform sequence.")

    # Convert to plain Python list for the external checker
    result = best_seq.tolist()
    # Optionally print the official score for debugging purposes
    try:
        official = evaluate_sequence(result)
        print(f"\nSearch finished. Official score = {official:.6f}")
    except Exception:
        # evaluate_sequence may not yet be defined in the isolated test harness
        print("\nSearch finished (official evaluation unavailable).")
    return result
