def arcsine_integral_shape(n):
    i = np.arange(n)
    # boundaries for each bin: i/n to (i+1)/n
    left = i / n
    right = (i+1) / n
    vals = (2/np.pi)*(np.arcsin(np.sqrt(right)) - np.arcsin(np.sqrt(left)))
    # arcsin(sqrt(0)) is 0; arcsin(sqrt(1)) = pi/2
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0/n)
    return vals / total
```

"""
Step‑function optimizer for the “evaluate_sequence” challenge.

The optimizer works in several timed stages:

0️⃣  Load a fallback sequence (if present) and keep it if it scores well.
1️⃣  Generate a large pool of diverse candidate shapes at a modest size
    (n ≈ 1024).  Shapes include the exact arcsine integral, Beta‑like
    densities, power‑law families, mixtures and simple triangular bumps.
2️⃣  Keep the few best seed vectors and refine them with Adam on the smooth‑max
    (log‑sum‑exp) surrogate of the max‑convolution.  The temperature τ is
    annealed from 0.5 to a large value (≈250) so that the surrogate converges
    to the true max.
3️⃣  Upsample the champion to larger lengths (2048 → 4096) using linear
    interpolation, add a tiny random jitter and run a second Adam phase with
    a longer schedule and a higher τ‑target.
4️⃣  Perform exact Euclidean‑projected sub‑gradient descent on the true
    max‑convolution objective, followed by an exponentiated‑gradient polish.
5️⃣  If time permits, a final short polish (sub‑gradient + EG) is applied.
6️⃣  Throughout the run we keep the best ever vector (according to the
    official ``evaluate_sequence`` score) and return it at the end.

All vectors are kept on the probability simplex (non‑negative and summed to 1),
so the official score equals the surrogate ``2·n·max(conv)``.
The algorithm respects the global 1000 s budget, prints progress and finally
returns the best sequence as a plain list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors via FFT.
    The result has length len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate = official score when Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine discrete approximation a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def arcsine_integral_shape(n: int) -> np.ndarray:
    """
    Exact discretisation of the continuous arcsine law:
    a_i = ∫_{i/n}^{(i+1)/n} (1/π)/√(x(1‑x)) dx.
    """
    left = i = np.arange(n, dtype=float) / n
    right = (np.arange(1, n + 1, dtype=float)) / n
    vals = (2.0 / np.pi) * (np.arcsin(np.sqrt(right)) - np.arcsin(np.sqrt(left)))
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if tot <= 0 or not np.isfinite(tot):
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if tot <= 0 or not np.isfinite(tot):
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_asymmetric_power_shape(
    n: int, p: float, eps_left: float, eps_right: float
) -> np.ndarray:
    """
    Asymmetric generalised arcsine:
    a_i ∝ 1 / ((i+eps_left)*(n‑½‑i+eps_right))**p .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + eps_left) * (n - 0.5 - i + eps_right) + 1e-12) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if tot <= 0 or not np.isfinite(tot):
        return np.full(n, 1.0 / n)
    return vals / tot


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(v) to target_n,
    followed by renormalisation to Σ=1.
    """
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 30_000,
    tau_start: float = 0.5,
    tau_end: float = 250.0,
    base_lr: float = 0.3,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·(a*a)_k)
    The temperature τ is increased exponentially from ``tau_start`` to
    ``tau_end``.
    The variables are stored as logits w with a = softmax(w) guaranteeing
    non‑negativity and Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting point (add epsilon for log)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # a = softmax(w)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # soft‑max weights over τ·conv  (log‑sum‑exp gradient)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()  # shape (L,)

        # gradient w.r.t. a : 2·(a *⟂* weights)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)          # length 3n‑2
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wr.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (step size decays like 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging (using the fast surrogate)
        if it % 20_000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.10,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true
    max‑convolution objective.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) on the original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.22,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror‑descent with KL geometry).
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
