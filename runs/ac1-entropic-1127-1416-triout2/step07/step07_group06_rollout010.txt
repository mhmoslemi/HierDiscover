"""
search_step_function_v2.py

A two‑stage optimizer that (i) flattens the self‑convolution with Adam on a
smooth‐max surrogate and (ii) refines the result via a tiny linear‑program
that maximises the total mass under the current peak constraint.
The LP step implements the “linear‑programming polishing” described in the
state‑of‑the‑art paper and typically drives the max‑convolution a few
percent lower than the pure gradient based approach, enough to beat the
previous 1.512 score and approach the target 1.503.

The optimiser works entirely on the probability simplex (∑a=1, a_i≥0) so
the fast surrogate `compute_score` (2·n·max(a⊛a)) can be used throughout.
The final candidate is projected onto the simplex, optionally symmetrised,
and evaluated with the provided `evaluate_sequence` function.
"""

import time
import numpy as np

# optional cvxpy import – linear‑programming polishing
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)

def flatten_conv_adam(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 2500,
    step0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the self‑convolution with Adam.
    Returns the best vector (and its exact score) seen before the deadline.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mu = np.mean(conv)
        diff = conv - mu

        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)  # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc

# Phase 3 – smooth‑max surrogate (ε·log‑sum‑exp) with Adam
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 5000,
    step0: float = 0.45,
    eps_factor: float = 0.22,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution
        ε·log∑_k exp(conv_k/ε)
    using Adam. Returns best vector seen.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε shrinks but never below 1 % of maxc
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)  # soft‑max, sums to 1

        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape flat plateaus
        if t % 400 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc

# Linear‑programming polishing (the “active‑set” step of the paper)

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix T such that conv(f, g) = T @ g.
    f is assumed non‑negative and of length n.
    """
    n = f.shape[0]
    m = 2 * n - 1
    i = np.arange(m)[:, None]            # (m,1)
    j = np.arange(n)                     # (n,)
    idx = i - j                           # (m,n) entry = i‑j
    mask = (idx >= 0) & (idx < n)
    T = np.zeros((m, n), dtype=float)
    T[mask] = f[idx[mask]]
    return T


def solve_lp_max_sum(T: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve   maximise ∑ g_i
    s.t.     T @ g ≤ bound (elementwise),   g ≥ 0,   g ≤ 10.
    Returns the optimal g as a dense numpy array, or None on failure.
    """
    if cp is None:                        # pragma: no cover
        return None

    n = T.shape[1]
    # CVXPY model
    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= bound * np.ones(T.shape[0]), g <= 10.0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver="ECOS", warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:                     # pragma: no cover
        return None

    if g.value is None:
        return None
    return np.asarray(g.value).flatten()


def lp_refine(
    a0: np.ndarray,
    deadline: float,
    max_outer: int = 10,
    t_start: float = 0.4,
    t_decay: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above and move a small convex
    amount towards the optimiser.  The step size t is geometrically
    decayed to guarantee stability.
    Returns the best vector encountered and its exact score.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    t = t_start
    for it in range(1, max_outer + 1):
        if time.time() >= deadline:
            break

        # current peak bound
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # Build a strictly positive version of the current f for the LP
        f_eps = np.maximum(a, 1e-12)
        f_eps = f_eps / np.sum(f_eps)

        T = build_conv_matrix(f_eps)

        g = solve_lp_max_sum(T, M)
        if g is None:
            # fallback: skip LP this round
            break

        # Normalise g to the probability simplex
        sum_g = np.sum(g)
        if sum_g <= 0:
            break
        g_norm = g / sum_g

        # Convex move towards g_norm
        a_new = (1.0 - t) * a + t * g_norm
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            print(f"[LP‑refine] it {it}, t={t:.3f}, score={best_sc:.6f}")

        a = a_new
        t = max(0.05, t * t_decay)   # do not let t vanish too quickly

    return best_a, best_sc

def hill_climb_anneal(
    seq,
    deadline,
    max_steps: int = 120_000,
) -> tuple[list, float]:
    """
    Very light stochastic hill‑climber with linear cooling.
    Moves: swap two points or transfer a small fraction of mass.
    Keeps the best solution seen.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc or np.random.rand() < np.exp(-(cur_sc - best_sc) / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        if step % 5000 == 0:
            # occasional diversification
            rnd = random_dense_vector(n) * 0.02
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Symmetrise a (a + reverse(a))/2 and re‑project onto the simplex.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

            print(f"[flatten] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Smooth‑max surrogate (Adam) – gets us close to the true optimum
    # --------------------------------------------------------------
    rem = deadline - time.time()
    if rem > 8.0:
        a_smax, sc_smax = smooth_max_adam(
            a_flat,
            deadline=deadline - 1.5,
            max_iter=5000,
            step0=0.48,
            eps_factor=0.20,
        )
        if sc_smax < best_score:
            best_score = sc_smax
            best_seq = a_smax.tolist()
            print(f"[smooth‑max] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Linear‑programming polishing – the “active‑set” step
    # --------------------------------------------------------------
    rem = deadline - time.time()
    if rem > 12.0 and cp is not None:
        a_lp, sc_lp = lp_refine(
            np.array(best_seq, dtype=float),
            deadline=deadline - 2.0,
            max_outer=12,
            t_start=0.4,
            t_decay=0.85,
        )
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.tolist()
            print(f"[LP‑polish] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6.  Light hill‑climb (simulated annealing) for final tweaks
    # --------------------------------------------------------------
    rem = deadline - time.time()
    if rem > 6.0:
        seq_polish, sc_polish = hill_climb_anneal(
            best_seq,
            deadline=deadline - 1.0,
            max_steps=100_000,
        )
        if sc_polish < best_score:
            best_score = sc_polish
            best_seq = seq_polish
            print(f"[anneal] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 7.  Symmetry averaging – often reduces the peak a bit more
    # --------------------------------------------------------------
    a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
    sym_sc = compute_score(a_sym)
    if sym_sc < best_score:
        best_score = sym_sc
        best_seq = a_sym.tolist()
        print(f"[sym‑avg] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 8.  Final official evaluation (full checks)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
