"""
Improved step‑function search
============================

The optimiser works in four stages.

1. **Massive low‑dimensional exploration** – a large family of
   analytically‑parameterised shapes (arcsine, Beta, power‑law,
   mixtures, random Dirichlet / sparse vectors) is enumerated on a
   modest grid (n=1024).  The cheap surrogate
   ``2·n·max(conv(a,a))`` is evaluated for each candidate and the best
   shape is retained.

2. **Coarse smooth‑max Adam** – the best low‑dimensional shape is refined
   with Adam on a differentiable log‑sum‑exp approximation of the
   max‑convolution.  The schedule for the temperature (τ) is
   gradually increased, allowing the optimiser to focus on the true
   max while keeping gradients well‑behaved.

3. **Upsampling + high‑resolution Adam** – the refined shape is linearly
   interpolated to the target resolution (default 4096 points) and
   Adam is run again with a much larger temperature schedule.  This
   phase drives the solution close to the continuous optimum.

4. **Exact polishing** – finally a projected sub‑gradient descent
   followed by an exponentiated‑gradient (mirror‑descent) pass is
   performed on the true non‑smooth objective.  The whole pipeline is
   time‑aware: each phase checks the remaining wall‑clock budget and
   scales its iteration budget accordingly, guaranteeing termination
   before the 1000 s limit.

All helper functions are top‑level (no closures) and only NumPy and the
provided ``evaluate_sequence`` are required.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (real inputs only)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  (Algorithm of Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate of the official score (Σa = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-8) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def random_sparse(n: int, density: float, rng: np.random.Generator) -> np.ndarray:
    """
    Sparse random vector with given density (fraction of non‑zeros),
    Dirichlet‑like random magnitudes, and normalised to sum = 1.
    """
    m = max(1, int(round(density * n)))
    idx = rng.choice(n, size=m, replace=False)
    vals = rng.random(m)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    s = a.sum()
    if s == 0.0:
        a[0] = 1.0
        s = 1.0
    a /= s
    return a


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points.
    The result is re‑normalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth approximation of max(conv).
    Variable is the logits w (softmax → a) which guarantees a ≥ 0
    and Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)                 # numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)            # length L

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    [Adam] it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (nonsmooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.06) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 50000 == 0:
            print(f"    [Subgrad] it={it:,d} surrogate={compute_score(a):.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 50000 == 0:
            print(f"    [EG] it={it:,d} surrogate={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine consists of:
      1. an exhaustive low‑dimensional shape sweep,
      2. coarse Adam optimisation (n=1024),
      3. up‑sampling + fine Adam (n=4096 or 2048),
      4. exact polishing (sub‑gradient + exponentiated‑gradient).
    All phases respect a global 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # 1 s safety margin

    # ------------------------------------------------------------------
    # Phase 0 – exhaustive low‑dimensional search (n=1024)
    # ------------------------------------------------------------------
    n_coarse = 1024
    print("\n--- Phase 0: low‑dimensional exhaustive search (n=1024) ---")
    best_a = None
    best_surrogate = float("inf")

    def maybe_update(a_candidate: np.ndarray, label: str):
        nonlocal best_a, best_surrogate
        score = compute_score(a_candidate)
        if score < best_surrogate:
            best_surrogate = score
            best_a = a_candidate.copy()
            print(f"    New best via {label}: surrogate={score:.6f}")

    # 1. deterministic families
    maybe_update(arcsine_initial(n_coarse), "arcsine")
    maybe_update(np.full(n_coarse, 1.0 / n_coarse), "uniform")

    # 2. mixtures of arcsine & uniform
    arc = arcsine_initial(n_coarse)
    uni = np.full(n_coarse, 1.0 / n_coarse)
    for w in np.linspace(0.0, 1.0, 21):
        cand = w * arc + (1.0 - w) * uni
        cand = cand / cand.sum()
        maybe_update(cand, f"mix_arc_uni_{w:.2f}")

    # 3. symmetric Beta (α = β)
    alphas = np.linspace(0.2, 1.8, 33)   # includes arcsine α=0.5
    for a in alphas:
        cand = generate_beta_shape(n_coarse, a, a, eps=1e-8)
        maybe_update(cand, f"beta_sym_{a:.2f}")

    # 4. asymmetric Beta grid (coarser)
    alphas_grid = np.linspace(0.2, 1.8, 9)
    betas_grid  = np.linspace(0.2, 1.8, 9)
    for a in alphas_grid:
        for b in betas_grid:
            cand = generate_beta_shape(n_coarse, a, b, eps=1e-8)
            maybe_update(cand, f"beta_{a:.2f}_{b:.2f}")

    # 5. power‑law shapes (generalised arcsine)
    for p in np.linspace(0.6, 1.4, 17):
        cand = generate_power_shape(n_coarse, p, eps=1e-8)
        maybe_update(cand, f"power_{p:.2f}")

    # 6. random Dirichlet samples
    for _ in range(30):
        cand = rng.dirichlet(np.ones(n_coarse))
        maybe_update(cand, "rand_dirichlet")

    # 7. random sparse vectors
    for dens in (0.05, 0.10, 0.20):
        for _ in range(20):
            cand = random_sparse(n_coarse, dens, rng)
            maybe_update(cand, f"rand_sparse_{dens:.2f}")

    # ------------------------------------------------------------------
    # Phase 1 – coarse Adam optimisation on the best low‑dimensional shape
    # ------------------------------------------------------------------
    print("\n--- Phase 1: coarse Adam optimisation (n=1024) ---")
    remaining = deadline - time.time()
    max_iter_coarse = int(min(150_000, max(10_000, remaining * 30)))
    best_a = adam_optimize_smooth_max(
        best_a,
        deadline,
        max_iters=max_iter_coarse,
        tau_start=0.5,
        tau_end=80.0,
        base_lr=0.5,
    )
    surrogate_coarse = compute_score(best_a)
    print(f"Coarse Adam surrogate = {surrogate_coarse:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upsample and high‑resolution Adam
    # ------------------------------------------------------------------
    # choose target resolution based on remaining time
    remaining = deadline - time.time()
    if remaining > 200.0:
        target_n = 4096
    else:
        target_n = 2048
    print(f"\n--- Phase 2: upsample to n={target_n} and high‑res Adam ---")
    a_high = upsample(best_a, target_n)

    remaining = deadline - time.time()
    max_iter_high = int(min(300_000, max(20_000, remaining * 30)))
    a_high = adam_optimize_smooth_max(
        a_high,
        deadline,
        max_iters=max_iter_high,
        tau_start=0.5,
        tau_end=300.0,
        base_lr=0.6,
    )
    surrogate_high = compute_score(a_high)
    print(f"High‑res Adam surrogate = {surrogate_high:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact polishing (sub‑gradient + exponentiated‑gradient)
    # ------------------------------------------------------------------
    print("\n--- Phase 3: exact polishing ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(300_000, max(20_000, remaining * 30)))
    a_refined = subgradient_descent_exact(a_high, deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.07)

    remaining = deadline - time.time()
    if remaining > 30.0:
        max_iter_eg = int(min(200_000, max(10_000, remaining * 30)))
        a_refined = eg_refine(a_refined, deadline,
                              max_iter=max_iter_eg,
                              eta0=0.25)

    # ------------------------------------------------------------------
    # Phase 4 – optional cheap random local search on the surrogate
    # ------------------------------------------------------------------
    # Small hill‑climbing moves using the fast surrogate; this often
    # squeezes a few extra decimal places.
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Phase 4: cheap random local search (surrogate) ---")
        best_local = a_refined.copy()
        best_local_score = compute_score(best_local)
        step_factor = 0.05  # relative amount moved per trial
        attempts = 0
        while time.time() < deadline - 2.0:
            attempts += 1
            i, j = rng.choice(best_local.size, size=2, replace=False)
            delta = step_factor * best_local[i]
            if delta > best_local[i]:
                delta = best_local[i] * 0.5
            cand = best_local.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)
            if cand_score < best_local_score:
                best_local = cand
                best_local_score = cand_score
                print(f"    improvement after {attempts} attempts: {cand_score:.6f}")

        a_refined = best_local

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"\nOfficial score of final candidate = {final_val:.6f}")
    except Exception as exc:
        print(f"Evaluation failed: {exc}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not isinstance(a_refined, np.ndarray) or a_refined.size == 0:
        fallback_n = 1024
        a_refined = np.full(fallback_n, 1.0 / fallback_n)

    print(f"\nSearch finished. Best official score = {final_val:.6f}")
    return a_refined.tolist()
