    a = w / w.sum()

"""
Propose a step‐function (non‑negative sequence) that minimises the evaluation
metric

    score = 2·n·max(conv) / (sum(a)**2)

The core idea is simple yet powerful:

* The continuous arcsine distribution on [0,1] has the smallest possible
  supremum norm of the self‑convolution among all probability densities.
  Discretising this density on a large number of points makes the discrete
  convolution almost flat, i.e. the constant 2·n·max(conv) converges to
  1.5 from above.
* By using a very large length (up to 2^17 ≈ 131 072) we get arbitrarily
  close to the theoretical limit 1.5.  A few tiny mixes with the uniform
  distribution smooth the inevitable discretisation artefacts.
* The whole search stays well inside the 1000 s budget – a handful of FFTs
  on 2^18 points are cheap.  If, after the scan, the score is still above
  the target, a short sub‑gradient optimisation is launched as a fallback.

The function returns a plain ``list[float]`` ready for the external
``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution and fast score (sum = 1)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for vectors that already satisfy
    ``sum(a) == 1``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple projections – keep vectors on the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def arcsine_distribution(n: int, eps: float = 1e-12) -> np.ndarray:
    """
    Discrete arcsine density: a_i ∝ 1 / sqrt(i·(n‑1‑i)).
    The distribution is symmetric and already close to optimal.
    """
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = w / np.sum(w)
    return a


def beta_shape_distribution(n: int, alpha: float, eps: float = 1e-12) -> np.ndarray:
    """
    Generalised Beta(α,α) shape on {0,…,n‑1}.
    Density ∝ (i·(n‑1‑i))^{α‑1}.
    For α = 0.5 we recover the arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    exponent = alpha - 1.0
    # use log‑space to avoid overflow for tiny α
    log_w = exponent * (np.log(i + eps) + np.log(n - 1 - i + eps))
    log_w -= np.max(log_w)                     # stabilise
    w = np.exp(log_w)
    a = w / np.sum(w)
    return a


def sin_power_distribution(n: int, power: float = 1.0) -> np.ndarray:
    """
    Weight proportional to sin(π·(i+0.5)/n)^{power}.
    Another smooth, symmetric shape.
    """
    i = np.arange(n, dtype=np.float64)
    angles = np.pi * (i + 0.5) / n
    w = np.sin(angles) ** power
    a = w / np.sum(w)
    return a


def mix_with_uniform(a: np.ndarray, mix: float) -> np.ndarray:
    """Convex combination a ↦ (1‑mix)·a + mix·uniform."""
    n = a.shape[0]
    if mix <= 0.0:
        return a
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    b = (1.0 - mix) * a + mix * uniform
    # a tiny projection to fight numerical drift
    return project_to_simplex(b)


# ----------------------------------------------------------------------
# A light sub‑gradient optimiser (fallback if needed)
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Classic sub‑gradient descent on the exact objective
    ``2·n·max(a⋆a)``.  The routine stops when the time budget expires.
    """
    a = a0.copy()
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        # occasional tiny random shake to escape possible stalls
        if t % 1500 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return a

# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the final call to ``evaluate_sequence``

    # lengths – powers of two are cheap for FFT
    lengths = [131072, 65536, 32768, 16384]

        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 4. Final official evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    # Return the *list* required by the harness
    return best_seq.tolist()
