"""
Improved step‑function optimizer
================================

The optimizer searches for a non‑negative sequence a (∑a=1) that minimizes  

        2·n·max(conv(a,a))

where conv is the linear (self)‑convolution.  This is exactly the value
returned by ``evaluate_sequence`` after the sum‑normalisation.

Key ideas
---------

* **Coarse exploration of a low‑dimensional family.**  A few parametric
  shapes (arcsine, power, beta, mixtures) are generated on a modest grid
  (n≈1024).  The best of them is chosen as a warm‑start.

* **Two‑stage smooth‑max Adam.**  The non‑smooth max is approximated by the
  log‑sum‑exp (temperature τ).  Stage 1 works on the coarse grid,
  stage 2 refines a higher‑resolution up‑sampled version (n≈4096 or 8192).

* **Exact nonsmooth polishing.**  Projected sub‑gradient descent followed
  by an exponentiated‑gradient (mirror‑descent) pass reduces the true
  max‑convolution.

* **Randomised local refinement.**  Multiplicative Gaussian noise is
  injected into the current best vector, the result is re‑normalised to the
  simplex and accepted if it improves the objective.  The noise amplitude
  decays with the remaining time (simulated‑annealing style).

All phases respect a global 1000 s wall‑clock budget.  The routine is fully
deterministic apart from the initial ``seed`` and prints progress
information.  The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (efficient for large vectors).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # edge case: all non‑positive -> uniform
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector a with sum = 1:
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-4) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float = 1e-4) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform, normalised."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n points, re‑normalised."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth approx. of max(conv).
    Variables are logits w -> a = softmax(w) (automatically on the simplex).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax -> a
        w_shift = w - np.max(w)  # stability
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)  # length L

        # smooth max via log‑sum‑exp (weights)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp  # shape (L,)

        # gradient w.r.t. a : 2·conv(a, reversed(weights))
        rev_w = weights[::-1]
        # convolution a * rev_w produces vector length n+L-1; we need the slice
        conv_rev = convolution_fft(a, rev_w)  # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (lightweight)
        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"[Adam] it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a

# Exact (nonsmooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.06) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, reversed(mask))
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

# Random local search (multiplicative noise)
# ----------------------------------------------------------------------


def random_local_search(start_a: np.ndarray,
                        deadline: float,
                        sigma_start: float = 0.3,
                        sigma_end: float = 0.001,
                        max_no_improve: int = 5000) -> np.ndarray:
    """
    Simple simulated‑annealing style search: multiplicative Gaussian noise,
    re‑normalisation to the simplex, accept only improvements.
    Stops when time runs out or a long streak without improvement occurs.
    """
    a = start_a.copy()
    best = a.copy()
    best_score = compute_score(best)
    n = a.size

    rng = np.random.default_rng()
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # linearly decay sigma with remaining time
        frac = (time.time() - (deadline - (deadline - start_time))) / (deadline - start_time)
        # protect against slight overshoot
        frac = min(max(frac, 0.0), 1.0)
        sigma = sigma_start * (1 - frac) + sigma_end * frac

        # propose a multiplicative perturbation (always positive)
        proposal = a * np.exp(sigma * rng.standard_normal(n))
        proposal /= proposal.sum()  # project onto simplex (positivity guaranteed)

        score = compute_score(proposal)
        it += 1

        if score < best_score:
            best_score = score
            best = proposal.copy()
            a = proposal  # move the centre of the search
            no_improve = 0
            print(f"[Random] it={it:,d} sigma={sigma:.5f} new best={best_score:.6f}")
        else:
            no_improve += 1
            # occasional reset to best if we wander too far
            if no_improve >= max_no_improve:
                a = best.copy()
                no_improve = 0

        # safety check to avoid endless tight loops
        if it % 50000 == 0:
            print(f"[Random] it={it:,d} current best={best_score:.6f}")

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low

    budget, combines a parametric warm‑start, two‑stage Adam optimisation,
    exact nonsmooth polishing, and a final stochastic local refinement.

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # 2 s safety margin

    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a known construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"[Fallback] known construction score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – coarse exploration on n=1024
    # ------------------------------------------------------------------
    n_coarse = 1024
    print("\n--- Phase 1: coarse shape exploration (n={}) ---".format(n_coarse))

    candidate_shapes = []

    # deterministic seeds
    candidate_shapes.append(arcsine_initial(n_coarse))
    candidate_shapes.append(np.full(n_coarse, 1.0 / n_coarse))  # uniform

    # power family
    for p in [0.6, 0.8, 1.0, 1.2, 1.5]:
        candidate_shapes.append(generate_power_shape(n_coarse, p))

    # beta family grid
    alphas = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    betas = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    for a in alphas:

    # mixtures with uniform
    for w in [0.25, 0.5, 0.75]:

    print("\nSearch finished. Best official score = {:.6f}".format(best_val))
    return best_seq
