```
def minimize_max_convolution(f, deadline):
    n = len(f)
    m = 2*n - 1
    # Build A (m x n)
    A = np.zeros((m, n))
    for i in range(n):
        A[i:i+n, i] = f
    # variable order: first n entries b, last entry t
    # constraints: A b - t <= 0 => A b - 1*t <= 0
    # Represent as A_ub * x <= b_ub, where x = [b, t]
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    # equality sum(b) = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1,1))])
    b_eq = np.array([1.0])
    # bounds: b_i >=0, no upper bound; t >= 0
    bounds = [(0, None)]*n + [(0, None)]
    # objective minimize t => c = [0,...,0,1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        b_opt = res.x[:n]
        t_opt = res.x[-1]
        return b_opt, t_opt, res
    else:
        raise ValueError("LP failed")
```

The program searches for a non‑negative step function that minimises  

\[
\text{score}= \frac{2\,n\;\max (a\star a)}{\bigl(\sum a_i\bigr)^2},
\qquad a\star a = a\;\text{convolution}\;a .
\]

Because the score is invariant under scaling we can restrict ourselves to
probability vectors (`∑aᵢ = 1`).  
The algorithm consists of three phases:

1. **Analytic start** – a short scan of the 1‑parameter family  
   \(a_i(\alpha)\propto[(i+\tfrac12)(n-\tfrac12-i)]^{-\alpha}\) and a
   fallback to the last known good vector (`height_sequence_1`).  
   The best vector from this stage is used as the initial point.

2. **LP‑based descent** – given the current distribution `f`,
   solve the linear program  

   \[
   \begin{aligned}
   \min_{b,t}\;& t \\[2pt]
   \text{s.t. }& (f*b)_k \le t \quad\forall k\\
               & b_j\ge 0\;(j=0\ldots n-1)\\
               &\sum_j b_j = 1 .
   \end{aligned}
   \]

   The optimal `b` has the smallest possible value of  
   \(\max (f*b)\).  Replacing `f←b` strictly reduces the objective (or
   leaves it unchanged).  The LP is solved with SciPy’s HiGHS
   interior‑point solver; the constraint matrix is a Toeplitz matrix
   built from the current `f`.  The LP is repeated until no further
   improvement is observed or the time budget is exhausted.

3. **Fine‑grained hill climbing** – a long stochastic walk on the
   probability simplex (swap or small mass‑transfer moves).  A move is
   kept only if it strictly improves the exact score.  This phase
   eliminates the remaining tiny local‑search noise.

All heavy work is performed with NumPy (FFT for fast convolutions) and
SciPy (linear programming).  The total wall‑clock time never exceeds the
provided budget (default 1000 s).  The returned object is a plain Python
list of floats, ready for `evaluate_sequence`.

```python
"""
search for a step function with minimal evaluate_sequence score.

The algorithm:
   1. quick analytical start – scan a one‑parameter family a_i(α)
   2. LP‑based descent – iteratively solve
          min_t  t  subject to  (f * b)_k ≤ t ,  b≥0 , Σb = 1
      and replace f by the optimal b.
   3. final stochastic hill‑climb on the simplex.

All helper functions are top‑level; no closures or file I/O are used.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact score for a probability vector (Σ a = 1).

    Returns 2·n·max(a⊛a) where n = len(a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic start – 1‑parameter family a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return the normalised vector a_i(α)."""
    i = np.arange(n, dtype=float)
    # compute log of the product to stay stable
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Grid‑search α ∈ [0,2.5] and keep the best exact score."""
    best_a = None
    best_score = float("inf")
    alphas = np.linspace(0.0, 2.5, 26)          # step ≈0.1
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based descent: minimise the maximal convolution with the current f
# ----------------------------------------------------------------------


def lp_minimise_max_convolution(f: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve
        min_t t
        s.t. (f * b)_k ≤ t   (k = 0..2n‑2)
             b ≥ 0, Σ b = 1 .
    Returns (b_opt, t_opt).
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build dense matrix A where A[k, j] = f_{k-j} if 0≤k-j<n, else 0.
    A = np.zeros((m, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f

    # Assemble the LP in the form  min cᵀx  s.t.  A_ub x ≤ b_ub , A_eq x = b_eq .
    # Variables are (b_0,…,b_{n‑1}, t)  →  dimension n+1.
    A_ub = np.hstack([A, -np.ones((m, 1), dtype=float)])   # A·b - t ≤ 0
    b_ub = np.zeros(m, dtype=float)

    A_eq = np.hstack([np.ones((1, n), dtype=float), np.zeros((1, 1), dtype=float)])
    b_eq = np.array([1.0], dtype=float)

    bounds = [(0.0, None)] * n + [(0.0, None)]   # b_j ≥0 , t ≥0
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0                                 # minimise t

    # Solve with HiGHS (fast interior‑point implementation)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True})

    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    b_opt = res.x[:n]
    t_opt = res.x[-1]

    # Numerical cleanup: enforce non‑negativity and renormalise.
    b_opt = np.maximum(b_opt, 0.0)
    s = b_opt.sum()
    if s == 0.0:
        raise RuntimeError("LP returned zero‑mass solution")
    b_opt /= s
    return b_opt, t_opt


# ----------------------------------------------------------------------
# Long stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
        - with probability 0.5 swap two entries,
        - otherwise move a fraction of mass from i to j.
    Keep a move only if the exact score improves.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.12

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97
        max_steps -= 1
    return best_a, best_score

    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose length – 2048 gives a good trade‑off between flexibility
    # and the quadratic cost of the LP.
    # --------------------------------------------------------------
    n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    # --------------------------------------------------------------
    # 0. fallback to the previously known good sequence, if any
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. analytical α‑grid start (quick and cheap)
    # --------------------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_seq = a_alpha.tolist()
        best_score = sc_alpha
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # If still nothing, fall back to a uniform vector (should never happen)
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Normalise the current candidate to a probability vector.
    # --------------------------------------------------------------
    f = np.array(best_seq, dtype=float)
    f = np.maximum(f, 0.0)
    f /= f.sum()

    # --------------------------------------------------------------
    # 2. LP‑based descent (iterative improvement)
    # --------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline - 30.0:
        lp_iter += 1
        # current exact score
        cur_score = compute_score(f)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = f.tolist()
            print(f"[{time.time():.1f}] LP iter {lp_iter}: new best score = {best_score:.6f}")

        # Solve the LP that minimises the maximal convolution with f.
        try:
            b_opt, t_opt = lp_minimise_max_convolution(f)
        except Exception as e:
            print(f"[{time.time():.1f}] LP failed on iter {lp_iter}: {e}")
            break

        new_score = 2.0 * n * t_opt
        print(f"[{time.time():.1f}] LP iter {lp_iter}: t_opt={t_opt:.6e}, score={new_score:.6f}")

        # Stop if we cannot improve any further
        if new_score >= best_score - 1e-12:
            # no improvement → stop the LP phase
            break

        # Accept the new distribution
        f = b_opt
        best_score = new_score
        best_seq = f.tolist()

    # --------------------------------------------------------------
    # 3. Final long hill‑climb polishing (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(np.array(best_seq, dtype=float),
                                       deadline,
                                       max_steps=500_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
```
