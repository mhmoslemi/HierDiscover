"""
Hybrid search for a low‑value step‑function.

Algorithm
---------
1.  For a list of lengths (default 4096, 2048, 1024) generate a large
    pool of *analytic* and *random* seeds:
        • arcsine, uniform and mixtures,
        • power‑family a[i] ∝ ((i+½)(n‑½‑i)+ε)^‑p  with a dense grid of p,
        • beta‑family a[i] ∝ (i+ε)^{α‑1}(n‑1‑i+ε)^{β‑1} with a grid of (α,β),
        • Dirichlet draws (different concentration parameters).

    All seeds are normalised to ℓ₁‑norm 1.  The cheap surrogate
    S(a)=2·n·max(conv(a,a)) (exactly the score of `evaluate_sequence`
    when the sum is 1) is used to keep only the most promising few
    candidates (≤ 5 per length).

2.  Each kept candidate is refined by a *smooth‑max* optimisation.
    The objective
        L_β(a)=log‑sum‑exp(β·conv(a,a))
    smoothly approximates the true max.  We re‑parameterise a via the
    soft‑max of an unconstrained log‑vector `z`, thus staying on the
    probability simplex automatically.  A continuation on the smoothness
    parameter `β` (from 2 up to 400) is performed while the Adam optimiser
    runs until the per‑seed time‑budget expires.

3.  The Adam result is evaluated with the real `evaluate_sequence`.
    If it improves the global best, the best sequence is updated.

4.  A short *exact sub‑gradient* descent (projected onto the simplex)
    is run on the current global best to polish the solution.

5.  When the global time budget is almost exhausted a cheap *random
    local search* (Gaussian perturbations + simplex projection) attempts
    to squeeze out a final improvement.

All steps respect the overall 1000 s budget and guarantee that a
non‑empty, non‑negative sequence is returned.  The code uses only
NumPy and the provided `evaluate_sequence` function.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """Cheap surrogate 2·n·max(conv(a,a)) (exact for sum(a)=1)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Symmetric 1/√(i·(n‑i)) shape, normalised to sum 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-10) -> np.ndarray:
    """a[i] ∝ ((i+½)(n‑½‑i)+eps)^{-p}, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float = 1e-10
) -> np.ndarray:
    """a[i] ∝ (i+eps)^{α‑1} (n‑1‑i+eps)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_dirichlet_seed(rng: np.random.Generator, n: int, conc: float) -> np.ndarray:
    """Dirichlet draw with concentration parameter `conc`."""
    return rng.dirichlet(np.full(n, conc))


# ----------------------------------------------------------------------
# Smooth‑max + Adam optimiser (continuation on β)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 2.0,
    beta_max: float = 400.0,
    ramp_iters: int = 50000,
    lr: float = 0.03,
) -> np.ndarray:
    """
    Minimise the smooth approximation L_β(a)=log‑sum‑exp(β·conv(a,a))
    while keeping a on the simplex via the soft‑max map.
    β grows linearly from `beta0` to `beta_max`.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # weights for the smooth‑max (log‑sum‑exp)
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(
                f"    Adam it={t:,d}  β={beta:.1f}  score≈{fast_score(a):.6f}"
            )

    return _softmax(z)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected onto the simplex)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on
        f(a)=2·n·max(conv(a,a))/sum(a)^2 .
    The iterate stays on the probability simplex.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for “max‑active” entries
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # diminishing step‑size
        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d}  score≈{fast_score(a):.6f}")

    return a

def propose_candidate(
    seed: int = 42, budget_s: int = 1000, **kwargs
) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  Returns a list of heights (ℓ₁‑norm = 1).
    """
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use the previously saved sequence if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Scan lengths (large first)
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a rich pool of seeds
        # --------------------------------------------------------------
        seed_pool: List[np.ndarray] = []

        # deterministic analytic families
        seed_pool.append(arcsine_initial(n))
        seed_pool.append(np.full(n, 1.0 / n))

        # power‑family (dense grid on the exponent)
        for p in np.linspace(0.05, 2.0, 25):
            seed_pool.append(generate_power_shape(n, p, eps=1e-12))

        # beta‑family (grid on (α,β) around the arcsine point)
        alphas = np.linspace(0.3, 1.0, 8)
        betas = np.linspace(0.3, 1.0, 8)
        for a in alphas:
            for b in betas:
                seed_pool.append(generate_beta_shape(n, a, b, eps=1e-12))

        # random Dirichlet draws with several concentration levels
        for _ in range(12):
            conc = rng.uniform(0.05, 2.0)
            seed_pool.append(generate_dirichlet_seed(rng, n, conc))

        # --------------------------------------------------------------
        # 1b. Quick pre‑selection with the cheap surrogate
        # --------------------------------------------------------------
        scored = [(fast_score(vec), vec) for vec in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]   # keep at most five

        # --------------------------------------------------------------
        # 1c. Allocate time per seed and run the heavy optimiser
        # --------------------------------------------------------------
        remaining_seeds = len(top_seeds)
        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 2.0:
                break

            # a modest share of the remaining budget
            rem = max(0.0, deadline - time.time())
            per_seed_budget = min(30.0, rem / (remaining_seeds * 1.5))
            remaining_seeds -= 1
            seed_deadline = time.time() + per_seed_budget

            print(
                f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_budget:.1f}s"
            )

            # ------------------------------------------------------------------
            # 2.  Smooth‑max Adam optimisation (continuation on β)
            # ------------------------------------------------------------------
            z0 = np.log(seed_vec + 1e-12)        # initialise logits
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=2.0,
                beta_max=400.0,
                ramp_iters=50000,
                lr=0.03,
            )

            # ------------------------------------------------------------------
            # 3.  Official evaluation
            # ------------------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ------------------------------------------------------------------
            # 4.  Quick exact sub‑gradient polishing (short time slice)
            # ------------------------------------------------------------------
            if time.time() < deadline - 1.0:
                sub_deadline = deadline - 0.5
                a_arr = np.array(best_seq, dtype=float)
                a_arr = subgradient_descent_exact(
                    a_arr,
                    sub_deadline,
                    max_iter=80_000,
                    step0=0.12,
                )
                try:
                    val2 = evaluate_sequence(a_arr.tolist())
                    print(f"   Sub‑grad polished score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_arr.tolist()
                        print(f"   *** Updated best after sub‑grad: {best_val:.6f} ***")
                except Exception:
                    pass

    # ----------------------------------------------------------------------
    # 5.  Global polishing (if time allows)
    # ----------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nGlobal polishing with sub‑gradient descent …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=250_000,
            step0=0.12,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final polished score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # 6.  Random local search (tiny perturbations + projection)
    # ----------------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\nLocal random search …")
        a_curr = np.array(best_seq, dtype=float)
        best_local_val = best_val

        while time.time() < deadline:
            # Gaussian perturbation; scale shrinks slowly with time
            remaining = deadline - time.time()
            scale = 0.03 * (remaining / (budget_s * 0.5) + 0.1)
            cand = a_curr + rng.normal(scale=scale, size=a_curr.shape)
            cand = np.clip(cand, 0, None)
            cand = _project_to_simplex(cand)

            try:
                val = evaluate_sequence(cand.tolist())
            except Exception:
                continue

            if val < best_local_val:
                best_local_val, a_curr = val, cand
                print(f"   Local improvement: {best_local_val:.6f}")

        best_seq = a_curr.tolist()
        best_val = best_local_val

    # ----------------------------------------------------------------------
    # 7.  Guard – guarantee a non‑empty return
    # ----------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
