def lp_refine_step(f, t, deadline):
    if time.time() >= deadline: return f
    M = np.max(convolution_fft(f, f))
    n = len(f)
    # Build A matrix.
    i = np.arange(n)
    k = np.arange(2*n-1)[:, None]
    j = k - i[None,:]  # shape (2n-1, n)
    mask = (j >= 0) & (j < n)
    A = np.where(mask, f[j], 0.0)
    # Solve LP
    from scipy.optimize import linprog
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # Use hiGHs, may also set `options={'presolve': True}`
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f
    g0 = res.x
    sum_g0 = np.sum(g0)
    if sum_g0 <= 0:
        return f
    g = g0 / sum_g0
    # Move towards g
    f_new = (1 - t) * f + t * g
    # Project to simplex
    f_new = project_to_simplex(f_new)
    return f_new

Improved step‑function optimisation for the `evaluate_sequence` metric.

Main ideas
----------
* Start from a discrete arcsine density (the classic near‑optimal shape).
* Refine the density with several stages:

    SCIPY_OK = False

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    # Numerical noise can produce tiny negative values – clip
    return np.maximum(conv, 0.0)


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a) == 1.
    Returns 2·n·max(conv) where conv = a⋆a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
        {x ≥ 0, Σ x = 1}
    """
    if np.all(v <= 0):
        # If everything is non‑positive, the projection is the uniform vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine‑like density:
        a_i ∝ 1 / sqrt(i·(n‑1‑i)).
    The distribution is symmetric, non‑negative and already yields a fairly
    flat self‑convolution.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_sparse_vector(n: int) -> np.ndarray:
    """A random probability vector with a modest number of non‑zero entries."""
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 1. Soft‑max (smooth max) gradient and Adam optimiser
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f_eps(a) = eps·log Σ_k exp(conv_k / eps)
    where conv = a⋆a.

    The analytical form is
        ∇f = 2·(w_rev ⊛ a)
    with w = softmax(conv/eps) (a probability vector, Σ w = 1).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)

    # cross‑correlation w_rev ⊛ a
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 4000,
    step0: float = 0.2,
    eps_factor: float = 0.25,
    min_eps_factor: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth soft‑max surrogate.
    The temperature ε is exponentially decayed but never drops below
    `min_eps_factor * max(conv)`.  Returns the best vector seen.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature schedule – never smaller than a tiny fraction of maxc
        eps = maxc * max(eps_factor * (0.94 ** (t - 1)), min_eps_factor)

        grad = softmax_gradient(a, eps)

        # Adam biased moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)          # simple decay
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to avoid stagnation
        if t % 600 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.001 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent (max‑conv only)
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true objective
        F(a) = 2·n·max(a⋆a).
    The sub‑gradient is taken at a shift where the convolution attains its
    maximum.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i} for indices that exist
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional random shake
        if t % 1500 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Linear‑programming refinement (LP‑step)
# ----------------------------------------------------------------------


def _build_lp_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A ∈ R^{(2n‑1)×n} such that
        (A @ g)[k] = (f ⋆ g)[k]   for all k.
    """
    n = f.shape[0]
    i = np.arange(n, dtype=np.int64)
    k = np.arange(2 * n - 1, dtype=np.int64)[:, None]        # shape (2n‑1,1)
    j = k - i[None, :]                                        # shape (2n‑1, n)
    mask = (j >= 0) & (j < n)
    # Use fancy indexing – this creates a dense array of shape (2n‑1, n)
    A = np.where(mask, f[j], 0.0)
    return A


def lp_refine_step(
    f: np.ndarray,
    mix: float,
    time_deadline: float,
) -> np.ndarray:
    """
    One LP refinement step:

    1. Compute M = max(f⋆f).
    2. Solve  max Σ g_j   s.t.  f⋆g ≤ M ,  g ≥ 0 .
    3. Scale the solution to have sum = 1.
    4. Mix the current vector with the scaled solution:
           f ← (1‑mix)·f + mix·g .
    The result is projected back onto the simplex.
    """
    if not SCIPY_OK:
        # SciPy not available – simply return the original vector.
        return f

    n = f.shape[0]
    M = float(np.max(convolution_fft(f, f)))

    # Build constraint matrix A and right‑hand side.
    A = _build_lp_matrix(f)                     # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Linear programme: minimise -1·g (i.e. maximise Σ g)
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None) for _ in range(n)]

    # Solve with the high‑performance “highs” interior‑point method.
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "disp": False},
        )
    except Exception:      # pragma: no cover
        return f

    if not res.success or res.x is None:
        return f

    g0 = res.x
    sum_g0 = np.sum(g0)
    if sum_g0 <= 0.0:
        return f

    # Scale so that the total mass equals 1.
    g = g0 / sum_g0

    # Mix and re‑project.
    f_new = (1.0 - mix) * f + mix * g
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# 4. Stochastic hill‑climber (strictly improving moves)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber that only accepts *strictly* improving moves.
    Moves consist of either swapping two entries or transferring a
    fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_frac = 0.08  # initial transfer size (relative to the source entry)

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # become bolder
        else:
            step_frac *= 0.96                       # become more cautious

        max_steps -= 1

    return best_a, best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Enforce (approx.) symmetry by averaging the vector with its reverse.
    The result is projected back onto the simplex.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` score.  The routine respects the time budget and
    returns a plain Python list ready for the evaluation harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Give a small safety margin for the final call to evaluate_sequence.
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously known construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Search configuration
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]   # work from larger to smaller
    max_outer = 20                      # outer restarts (time‑limited)

    for outer in range(max_outer):
        if time.time() >= deadline - 6.0:
            break

        n = int(np.random.choice(lengths))
        print(f"\n[outer {outer}] chosen length = {n}")

        # --------------------------------------------------------------
        # 0. Initial vector – mostly arcsine, occasionally a sparse random.
        # --------------------------------------------------------------
        if np.random.rand() < 0.7:
            a0 = arcsine_initialisation(n)
        else:
            a0 = random_sparse_vector(n)

        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[init] score = {cur_sc:.6f}")

        # --------------------------------------------------------------
        # 1. Adam‑accelerated soft‑max gradient descent
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left < 10.0:
            break
        # Rough estimate: ~30‑35 gradient steps per second on a single core.
        iters = max(2_000, int(time_left * 30))
        a_gd, sc_gd = adam_softmax_descent(
            a0,
            time_deadline=deadline - 1.5,
            max_iter=iters,
            step0=0.28,
            eps_factor=0.22,
            min_eps_factor=0.001,
        )
        sc_exact = compute_score(a_gd)
        if sc_exact < best_score:
            best_score = sc_exact
            best_seq = a_gd.tolist()
            print(f"[adam]  improved score = {sc_exact:.6f}")

        # --------------------------------------------------------------
        # 2. Exact sub‑gradient descent for fine tuning
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left > 12.0:
            a_sub, sc_sub = subgradient_descent(
                a_gd,
                time_deadline=deadline - 2.0,
                max_iter=80_000,
                step0=0.07,
            )
            sc_exact = compute_score(a_sub)
            if sc_exact < best_score:
                best_score = sc_exact
                best_seq = a_sub.tolist()
                print(f"[sub‑grad] improved score = {sc_exact:.6f}")
        else:
            a_sub = a_gd

        # --------------------------------------------------------------
        # 3. LP‑refinement (a few alternating updates)
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        if SCIPY_OK and time_left > 15.0:
            a_lp = a_sub
            mix_factor = 0.22
            for lp_iter in range(3):
                if time.time() >= deadline - 4.0:
                    break
                a_lp = lp_refine_step(a_lp, mix=mix_factor, time_deadline=deadline - 2.0)
                sc_lp = compute_score(a_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.tolist()
                    print(f"[LP {lp_iter+1}] improved score = {sc_lp:.6f}")
                # gradually decrease the mixing weight to stabilise
                mix_factor = max(0.05, mix_factor * 0.7)
        else:
            a_lp = a_sub

        # --------------------------------------------------------------
        # 4. Hill‑climbing polishing
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left > 10.0:
            polished, pol_sc = hill_climb(
                np.array(best_seq, dtype=float),
                time_deadline=deadline - 2.0,
                max_steps=200_000,
            )
            if pol_sc < best_score:
                best_score = pol_sc
                best_seq = polished.tolist()
                print(f"[hill‑climb] improved score = {pol_sc:.6f}")

        # --------------------------------------------------------------
        # 5. Optional symmetry averaging (cheap and often beneficial)
        # --------------------------------------------------------------
        a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[symmetry] improved score = {sc_sym:.6f}")

    # ------------------------------------------------------------------
    # Fallback – a uniform vector (guaranteed finite result)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (calls the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
