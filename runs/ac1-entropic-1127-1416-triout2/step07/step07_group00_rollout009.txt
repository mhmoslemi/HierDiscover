"""
Search for a non‑negative step function that minimises the evaluate_sequence metric.

The method combines:
  - a fast analytic family (edge‑heavy power law) to obtain a good start;
  - several random restarts optimised by an Adam‑style projected gradient descent
    on a log‑sum‑exp smooth approximation of the max‑convolution;
  - a classic projected sub‑gradient phase for fine tuning;
  - a long stochastic hill‑climb that moves mass between coordinates.

All heavy work uses FFT‑based convolutions, runs well inside the allowed
wall‑clock budget and never calls external solvers.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using real FFT (a and b both real)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    # implementation from Duchi et al.
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def compute_score(a: np.ndarray) -> float:
    """Exact objective when sum(a)=1 (avoids the heavy evaluate_sequence)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def alpha_start(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family a_i ∝ [(i+½)(n‑½‑i)]^(‑α) – normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against overflow/underflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a

def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Sweep a modest grid of α and keep the best (fast because it uses compute_score)."""
    best_a = None
    best_score = float("inf")
    alphas = np.linspace(0.0, 2.5, 30)  # step ≈0.086
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 120.0,
                         lr0: float = 0.12,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth approximation
        f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
    The smoothing parameter β is increased exponentially from `beta_start`
    to `beta_end`.  The routine returns the best exact score observed.
    """
    a = start_a.astype(np.float64).copy()
    n = a.size

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    # pre‑allocate for speed
    a_rev = a[::-1]

    for it in range(1, max_iters + 1):
        # time check only every few hundred steps
        if it % 500 == 0 and time.time() >= deadline:
            break

        # ----- β schedule (geometric) -----
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # ----- smooth max -------------------------------------------------
        conv = convolution_fft(a, a)          # length 2n‑1
        c_max = conv.max()
        # stable log‑sum‑exp
        exp_shifted = np.exp(beta * (conv - c_max))
        w = exp_shifted / exp_shifted.sum()    # softmax weights, length 2n‑1

        # ----- gradient (cross‑correlation) -------------------------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]  # slice of length n

        # ----- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / np.sqrt(it)
        a_new = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # ----- projection onto simplex ------------------------------------
        a = project_to_simplex(a_new)

        # ----- occasional exact evaluation --------------------------------
        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                # tighten learning‑rate a little after a good improvement
                # (no hard rule – just a hint for the user via print)
                # print(f"[{time.time():.1f}] Adam it {it} improved to {sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 400_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method on the true max‑convolution."""
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if t % 500 == 0 and time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of (a*a)_k  = 2 * (a shifted by k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb:
        – swap two entries with probability 0.5,
        – otherwise transfer a fraction of mass from one index to another.
    Moves that improve the exact score are kept; the step size is
    automatically increased/decreased.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(best_a)

    # start with moving about 10 % of the source mass
    step_scale = 0.10

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_score

def random_dirichlet_start(n: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a random point on the simplex from a Dirichlet distribution."""
    # lower concentration → sparser vectors
    a = np.random.gamma(concentration, 1.0, size=n)
    a /= a.sum()
    return a

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    fallback_seq = globals().get("height_sequence_1")

            print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
            if fb < best_score:
                best_score = fb
                best_seq = [float(x) for x in fallback_seq]
        except Exception as e:
            print(f"[{time.time():.1f}] fallback failed: {e}")

    # ------------------------------------------------------------------
    # 1.  analytical edge‑heavy start (grid over α)
    # ------------------------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha

    # ------------------------------------------------------------------
    # 2.  Random Dirichlet restarts with Adam‑style smooth optimisation

    # allocate a modest number of restarts; each gets its own budget

            beta_start=6.0,

        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam restart = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Longer Adam optimisation from the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 130.0:
        print(f"[{time.time():.1f}] fine‑tuning with Adam")
        a_fine, sc_fine = adam_smooth_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 100.0,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=150.0,
            lr0=0.10,
        )
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.tolist()
            print(f"[{time.time():.1f}] after fine‑tune Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Projected sub‑gradient refinement (more iterations than before)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 40.0,
            max_iters=400_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=600_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # safety fallback – uniform distribution (very unlikely to be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small

    # ------------------------------------------------------------------
    # final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq

"""
Search for a non‑negative step function that minimises the `evaluate_sequence`
objective.

The algorithm proceeds through five stages:

1. **Analytic edge‑heavy start** – a quick grid over the one‑parameter family
   a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α) gives a decent initial point.
2. **Random Dirichlet restarts** – each start is refined by a projected Adam
   optimiser that minimises a log‑sum‑exp smooth approximation of the
   max‑convolution.  The smoothing parameter β is annealed from a modest value
   up to a large one.
3. **Fine‑tuning Adam** – a longer Adam run on the current best vector.
4. **Projected sub‑gradient refinement** – classic sub‑gradient descent on the
   true max‑convolution objective.
5. **Long stochastic hill‑climb** – random swaps / mass‑transfer moves on the
   simplex, keeping any improvement.

All convolutions use an FFT‑based routine, the simplex projection follows
Duchi‑et‑al., and the code respects the 1000 s wall‑clock limit.
The returned value is a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        # all non‑positive → uniform point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a)=1.
    Returns the exact objective 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# 1. Edge‑heavy analytical start
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    Return the normalised vector a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α).
    The vector lies on the probability simplex.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a modest grid of α‑values, keep the one with smallest exact score.
    Returns (best_vector, best_score).
    """
    best_a = None
    best_score = float("inf")
    alphas = np.linspace(0.0, 2.5, 30)  # ≈0.086 step
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected Adam on the smooth max‑convolution
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 120.0,
                         lr0: float = 0.12,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smoothed objective

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k).

    β is increased geometrically from `beta_start` to `beta_end`.
    Returns the best exact score encountered and the associated vector.
    """
    a = start_a.astype(np.float64).copy()
    n = a.size

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        # time check only periodically → reduces overhead
        if it % 500 == 0 and time.time() >= deadline:
            break

        # ----- β schedule (geometric) -----
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # ----- smooth max (log‑sum‑exp) -----
        conv = convolution_fft(a, a)          # length 2n‑1
        c_max = conv.max()
        exp_shift = np.exp(beta * (conv - c_max))
        w = exp_shift / exp_shift.sum()        # soft‑max weights

        # ----- gradient (cross‑correlation) -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]  # slice of length n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / np.sqrt(it)
        a_new = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # ----- projection onto simplex -----
        a = project_to_simplex(a_new)

        # ----- occasional exact evaluation -----
        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final evaluation
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 400_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for the exact objective
    2·n·max(a*a).  Uses a 1/√t stepsize.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if t % 500 == 0 and time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of (a*a)_k  = 2·a shifted by k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Long stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
      – with probability 0.5 swap two entries,
      – otherwise transfer a fraction of mass from one index to another.
    Moves that improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size

    best_a = a.copy()
    best_score = compute_score(best_a)

    # start moving about 10 % of the source mass
    step_scale = 0.10

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Miscellaneous helpers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, concentration: float = 0.5) -> np.ndarray:
    """Sample a random point on the simplex from a Dirichlet distribution."""
    a = np.random.gamma(concentration, 1.0, size=n)
    a /= a.sum()
    return a

    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient refinement (longer than before)

    return best_seq
