"""
Propose a low‑dimensional‐to‑high‑dimensional optimisation pipeline for the
step‑function challenge.

Strategy
--------
0️⃣  Load a fallback solution if it exists.
1️⃣  Perform a cheap parameter sweep on a modest size (n≈256) over several
    analytically‑motivated families (arcsine, power‑law, beta‑law, random Dirichlet)
    – keep the best shape.
2️⃣  Refine that shape with a long Adam optimisation on the smooth‑max
    surrogate (log‑sum‑exp of the convolution).  The temperature τ is raised
    exponentially up to a large value so that the surrogate approaches the real
    max.
3️⃣  Upsample the champion to larger lengths (2048 → 4096 → 8192) and repeat Adam.
    Multiple jittered restarts are tried; the best result at each scale is kept.
4️⃣  Run a few iterations of exact Euclidean‑projected sub‑gradient descent
    followed by an exponentiated‑gradient polish (the same tools used before).
5️⃣  Finish with a tiny random‑swap local search that conserves the ℓ₁‑norm.
    Every improvement is evaluated with the official ``evaluate_sequence``.
All stages respect a global 1000 s budget and maintain the best sequence found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (length = len(a)+len(b)-1).
    Both inputs are real 1‑D arrays.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The input is assumed to be normalised so that Σa = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = u - cssv / np.arange(1, v.size + 1)
    rho_idx = np.where(rho > 0)[0]
    if rho_idx.size == 0:
        theta = 0.0
    else:
        rho = rho_idx[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (families used for the coarse grid)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine law a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-12)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    vals = left * right
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, w: float) -> np.ndarray:
    """Mixture (w)·arcsine + (1‑w)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = w * arc + (1.0 - w) * uni
    return mix / mix.sum()


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(v) to target_n,
    followed by re‑normalisation to Σ=1.
    """
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up


# ----------------------------------------------------------------------
# Optimisers on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 100_000,
    tau_start: float = 0.5,
    tau_end: float = 300.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k).
    The temperature τ grows exponentially from ``tau_start`` to ``tau_end``.
    Variables are logits w such that a = softmax(w) guaranteeing non‑negativity
    and Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from a_init (add tiny offset to avoid log(0))
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponentially increasing temperature
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()          # shape (L,)

        # gradient of Φ_τ w.r.t. a: 2·conv_rev(weights)
        rev_weights = weights[::-1]
        conv_rev = convolution_fft(a, rev_weights)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # back‑prop through softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (step size decays like 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent & exponentiated‑gradient polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = project_to_simplex(a_init.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """Exponentiated‑gradient polish (mirror descent with KL geometry)."""
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with minimal official score.
    The routine follows the multi‑stage pipeline described at the top of the file.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Optional fallback
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep on a small dimension (n≈256)
    # ------------------------------------------------------------------
    n_grid = 256
    candidates: List[np.ndarray] = []

    # deterministic baselines
    candidates.append(arcsine_initial(n_grid))
    candidates.append(np.full(n_grid, 1.0 / n_grid))

    # power‑law family (p, eps)
    for p in np.linspace(0.35, 0.80, 10):
        for eps in np.linspace(0.0, 2.0, 5):
            candidates.append(generate_power_shape(n_grid, p, eps))

    # beta‑law family (α,β,eps)
    for a in np.linspace(0.3, 2.0, 7):
        for b in np.linspace(0.3, 2.0, 7):
            eps = rng.uniform(0.0, 2.0)
            candidates.append(generate_beta_shape(n_grid, a, b, eps))

    # mixture of arcsine and uniform
    for w in np.linspace(0.0, 1.0, 6):
        candidates.append(generate_mixture_shape(n_grid, w))

    # a handful of random Dirichlet samples
    for _ in range(80):
        candidates.append(rng.dirichlet(np.ones(n_grid)))

    # keep the best few according to the fast surrogate
    best_grid_vec = None
    best_grid_score = float("inf")
    for vec in candidates:
        sc = compute_score(vec)
        if sc < best_grid_score:
            best_grid_score, best_grid_vec = sc, vec

    print(f"[Stage 1] Best surrogate on n={n_grid} → {best_grid_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam refinement on the small‑scale champion
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    max_iters_small = int(min(180_000, remaining * 20))
    a_small = adam_optimize_smooth_max(
        best_grid_vec,
        deadline,
        max_iters=max_iters_small,
        tau_start=0.5,
        tau_end=300.0,
        base_lr=0.30,
    )
    try:
        val = evaluate_sequence(a_small.tolist())
        print(f"[Stage 2] Official after small‑scale Adam = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_small.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Upscaling cascade (2048 → 4096 → 8192) with Adam restarts
    # ------------------------------------------------------------------
    upscale_targets = (2048, 4096, 8192)
    current_vec = a_small

    for target_n in upscale_targets:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n[Stage 3] Upscaling to n={target_n}")

        # linear interpolation to the new size
        up = upscale_vector(current_vec, target_n)

        # try a few jittered restarts and keep the best
        best_local = up
        best_local_score = compute_score(up)
        trials = 3
        for tr in range(trials):
            if time.time() >= deadline - 5.0:
                break
            v = up + rng.normal(scale=5e-4, size=target_n)
            v = np.clip(v, 0.0, None)
            v /= v.sum()

            remaining = deadline - time.time()
            max_iters = int(min(120_000, remaining * 20))
            v_opt = adam_optimize_smooth_max(
                v,
                deadline,
                max_iters=max_iters,
                tau_start=0.5,
                tau_end=500.0,
                base_lr=0.22,
            )
            sc = compute_score(v_opt)
            if sc < best_local_score:
                best_local, best_local_score = v_opt, sc

        current_vec = best_local
        print(f"    Upscaled surrogate = {best_local_score:.6f}")

        # occasional official evaluation
        try:
            val = evaluate_sequence(current_vec.tolist())
            print(f"    Official after up‑scale {target_n} = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, current_vec.tolist()
                print("*** New global best! ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4️⃣  Exact sub‑gradient descent + exponentiated‑gradient polish
    # ------------------------------------------------------------------
    if current_vec is not None:
        n_best = len(current_vec)
        print(f"\n[Stage 4] Exact refinement on n={n_best}")
        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, remaining * 20))
        refined = subgradient_descent_exact(
            current_vec,
            deadline,
            max_iter=max_iter_sub,
            step0=0.10,
        )
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(180_000, remaining * 20))
            refined = eg_refine(
                refined,
                deadline,
                max_iter=max_iter_eg,
                eta0=0.22,
            )

        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official after exact polish = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print("*** New global best after refinement ***")
        except Exception as e:
            print(f"    Evaluation failed after refinement: {e}")

    # ------------------------------------------------------------------
    # 5️⃣  Tiny random‑swap local search (mass‑preserving)
    # ------------------------------------------------------------------
    if best_seq:
        a = np.array(best_seq, dtype=float)
        n = len(a)
        print("\n[Stage 5] Local random‑swap search")
        while time.time() < deadline - 1.0:
            i, j = rng.choice(n, size=2, replace=False)
            # propose moving a small fraction of a[i] to a[j]
            delta = rng.uniform(0.0, 1.0) * a[i] * 0.005   # ≤0.5 % of a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # no need to renormalise – total mass unchanged
            if a_new[i] < 0:
                continue
            try:
                val_new = evaluate_sequence(a_new.tolist())
            except Exception:
                continue
            if val_new < best_val:
                best_val, best_seq = val_new, a_new.tolist()
                a = a_new
                print(f"    Improvement via swap: {val_new:.6f}")
        # final safeguard
        a = np.array(best_seq, dtype=float)

    # ------------------------------------------------------------------
    # 6️⃣  Guard – always return something
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
