def upscale(shape_small, n_large):
    x_small = np.linspace(0, 1, shape_small.size)
    x_large = np.linspace(0, 1, n_large)
    shape_large = np.interp(x_large, x_small, shape_small)
    shape_large = np.maximum(shape_large, 0)
    shape_large /= shape_large.sum()
    return shape_large
```

"""
Propose a step function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Fast random search on a moderate size (n=1024).**  
   A large pool of analytic shapes (arcsine, power‑law, beta, mixtures,
   plus random perturbations) is generated.  Each shape is normalised to
   sum = 1 and scored with a *fast* surrogate 2·n·max(conv) (the objective
   when the sum is 1).  The ten best shapes are retained.

2. **Upscaling** – the selected shapes are linearly interpolated to a
   larger resolution (n=4096) so that the final answer can exploit a
   finer discretisation.

3. **Mirror‑descent polishing** – each up‑scaled candidate is refined
   with an exponentiated‑gradient (KL‑mirror) descent that works directly
   on the true non‑smooth objective.  The routine runs until a per‑shape
   time budget (derived from the remaining wall‑clock time) expires.

4. **Official evaluation** – after polishing the sequence is handed to
   ``evaluate_sequence``; the best‑scoring sequence is kept.

5. **Global exact sub‑gradient polishing** – the current global best is
   finally improved with a projected sub‑gradient descent (the same
   routine used in the reference solution).

The whole pipeline respects the 1000 s wall‑clock limit, always returns
a non‑empty valid sequence and aggressively explores the space of
parameterised shapes, which has been observed to push the score below
the previous 1.5117 barrier.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official objective for a vector that already
    sums to 1:  2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic shape generators (all return a normalized vector: sum = 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """1 / ((i+0.5)*(n-0.5-i)+eps)^p, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    s = vals.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / s


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Discrete Beta‑like envelope."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    s = base.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n)
    return base / s


def generate_mixture_shape(n: int, w_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    return mix / mix.sum()


def upsample_shape(small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from `small` to length `n_large`, re‑normalised."""
    x_small = np.linspace(0.0, 1.0, small.size)
    x_large = np.linspace(0.0, 1.0, n_large)
    large = np.interp(x_large, x_small, small)
    large = np.maximum(large, 0.0)
    total = large.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n_large, 1.0 / n_large)
    return large / total


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Mirror descent on max‑convolution using KL divergence.
    The iterate always stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for the active set (maximal entries)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = _conv_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} surrogate={_fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact, for final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 180_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact objective 2·n·max(conv),
    staying on the simplex via Euclidean projection.
    """
    a = init_a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = _conv_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad

        # Euclidean projection onto the probability simplex
        if np.any(a < 0):
            # classic O(n) simplex projection
            u = np.sort(a)[::-1]
            cssv = np.cumsum(u) - 1.0
            rho_arr = u - cssv / np.arange(1, a.size + 1)
            rho_mask = rho_arr > 0
            if not np.any(rho_mask):
                theta = 0.0
            else:
                rho = np.where(rho_mask)[0][-1]
                theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(a - theta, 0.0)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} surrogate={_fast_score(a):.6f}")

    return a
