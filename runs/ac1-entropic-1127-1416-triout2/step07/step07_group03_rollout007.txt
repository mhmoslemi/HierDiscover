def exponentiated_gradient_polish(a0, deadline, max_iter=50000, step=0.015):
    a = np.clip(a0, 1e-15, None)
    a /= a.sum()
    n = a.size
    L = 2*n - 1
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(conv.max())
        tol = mx*1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

Propose a step function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Fallback** – if a previous best sequence ``height_sequence_1`` exists,
   evaluate it and keep it as the current best.

2. **Parametric seeding (length 4096)** – generate a diverse pool of candidates
   using a few analytic families (arcsine, power‑law, beta, Gaussian,
   Hann/Blackman windows) with random parameters.  All candidates are
   normalised to sum = 1.

3. **Pre‑selection** – rank the pool by a fast surrogate
   ``compute_score`` (exact objective for unit‑sum vectors) and keep the
   best few (≤10) seeds.

4. **Smooth‑max Adam refinement** – each selected seed is refined on the
   smooth log‑sum‑exp surrogate of the max‑convolution.  The temperature
   β is linearly increased from a low value (≈5) to a high value (≈250);
   the Adam optimiser runs until the global deadline is reached.

5. **KL‑mirror polishing** – the current global best is further improved
   by exponentiated‑gradient (mirror) descent on the true non‑smooth
   objective.

6. **Exact projected sub‑gradient polishing** – a final projected
   sub‑gradient descent (the same routine used in the reference solution)
   is run for a short horizon.

7. **Local random search** – a cheap neighbourhood search (pairwise mass
   transfers) is performed while time permits to squeeze out tiny
   improvements.

8. **Final evaluation** – the resulting vector is returned as a list of
   floats; the official ``evaluate_sequence`` is called once for reporting.

The routine respects the 1000 s wall‑clock budget, always returns a
non‑empty valid sequence and aims to beat the previous bound
(~1.51168) by systematic parameter exploration and multi‑stage
optimisation.

import time

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate for the official objective (unit‑sum vectors)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Returns the value of the official objective for a non‑negative
    vector ``a`` (any scale).  It uses the exact convolution, but
    assumes the denominator ``sum(a)**2`` is already taken into account.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_conv = float(conv.max())
    sum_a = float(a.sum())
    if sum_a <= 0.0:
        return float("inf")
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def power_shape(n: int, p: float, delta: float, eps: float = 1e-12) -> np.ndarray:
    """Generalised power‑law shape: ((i+δ)*(n‑1‑i+δ)+ε)^(‑p)."""
    i = np.arange(n, dtype=float)
    base = ((i + delta) * (n - 1 - i + delta) + eps) ** (-p)
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-12) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discretised Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def hann_window(n: int) -> np.ndarray:
    """Standard Hann window, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def blackman_window(n: int) -> np.ndarray:
    """Standard Blackman window, normalised."""
    i = np.arange(n, dtype=float)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 250.0,
    ramp_iters: int = 40000,
    lr: float = 0.07,
    max_iter: int = 200_000,
) -> np.ndarray:
    """
    Minimise the smooth log‑sum‑exp surrogate of ``max(conv(a,a))``.
    The iterates stay on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        a = softmax(z)  # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(conv.max())

        # linearly raise β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‐max via log‑sum‑exp weighting
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2 * (a ⋆ w_rev) (restricted to original length)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate score={compute_score(a):.6f}")

    # final vector on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Mirror descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)  # avoid exact zeros
    a /= a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(conv.max())
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)) / (sum a)^2.
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(conv.max())
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Tiny local random neighbourhood search
# ----------------------------------------------------------------------


def local_random_search(a: np.ndarray, deadline: float, trials: int = 5000, sigma: float = 1e-3) -> np.ndarray:
    """
    Simple neighbourhood search: repeatedly transfer a small random mass
    between two random coordinates while staying on the simplex.
    """
    best = a.copy()
    best_score = compute_score(best)
    n = a.size
    rng = np.random.default_rng()

    for _ in range(trials):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.normal(scale=sigma)
        if best[i] + delta < 0 or best[j] - delta < 0:
            continue
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = project_to_simplex(cand)
        score = compute_score(cand)
        if score < best_score:
            best = cand
            best_score = score
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The function respects the 1000 s wall‑clock budget and always returns
    a non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback from a previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Optimise the main length (prefer the largest we can afford)
    # ------------------------------------------------------------------
    n = 4096  # target length – large enough to improve the bound

    # ------- generate a pool of parametric seeds ----------------------
    seeds: List[np.ndarray] = []

    # 1a. analytic deterministic seeds
    seeds.append(arcsine_initial(n))
    seeds.append(np.full(n, 1.0 / n))
    seeds.append(hann_window(n))
    seeds.append(blackman_window(n))

    # 1b. random power‑law shapes
    for _ in range(12):
        p = rng.uniform(0.35, 0.80)          # around the arcsine exponent 0.5
        delta = rng.uniform(0.0, 6.0)        # smoothing at the boundaries
        eps = rng.uniform(0.0, 1e-3)
        shape = power_shape(n, p, delta, eps)
        seeds.append(shape)

    # 1c. random beta‑shape seeds
    for _ in range(12):
        a = rng.uniform(0.2, 4.0)
        b = rng.uniform(0.2, 4.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        shape = beta_shape(n, a, b, eps)
        seeds.append(shape)

    # 1d. Gaussian windows (different spreads)
    for _ in range(8):
        sigma = rng.uniform(0.05 * n, 0.30 * n)
        shape = gaussian_shape(n, sigma)
        seeds.append(shape)

    # 1e. mixture of arcsine and uniform (single weight)
    for _ in range(6):
        w = rng.uniform(0.0, 1.0)
        mix = w * arcsine_initial(n) + (1.0 - w) * np.full(n, 1.0 / n)
        mix /= mix.sum()
        seeds.append(mix)

    # 1f. include previously best (if any) by up‑sampling / truncating
    if best_seq:
        old = np.array(best_seq, dtype=float)
        if old.size >= n:
            cand = old[:n]
        else:
            reps = int(np.ceil(n / old.size))
            cand = np.tile(old, reps)[:n]
        cand = np.clip(cand, 1e-15, None)
        cand /= cand.sum()
        seeds.append(cand)

    # ------- fast pre‑selection ---------------------------------------
    scored = [(compute_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:10]]  # keep at most ten promising seeds
    print(f"Pre‑selection: best surrogate score = {scored[0][0]:.6f}")

    # ------- refinement of each top seed -------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 30.0:
            print("Time low – stopping further refinements.")
            break

        print(f"\nRefining seed {idx}/{len(top_seeds)} (length {n})")
        # initialise Adam via log‑softmax (avoid log(0))
        eps = 1e-12
        z0 = np.log(seed_vec + eps)

        # Run Adam with a per‑seed deadline
        seed_deadline = min(deadline - 5.0, time.time() + 25.0)  # at most ~25 s per seed
        a_refined = smooth_adam_optimize(
            z0,
            seed_deadline,
            n=n,
            beta0=5.0,
            beta_max=250.0,
            ramp_iters=40000,
            lr=0.07,
        )

        # Evaluate the refined candidate
        val = compute_score(a_refined)
        print(f"   Surrogate score after Adam = {val:.6f}")
        # Convert to official score for reporting (identical when sum≈1)
        try:
            official = evaluate_sequence(a_refined.tolist())
            print(f"   Official score = {official:.6f}")
        except Exception as e:
            official = float("inf")
            print(f"   Official evaluation failed: {e}")

        if official < best_val:
            best_val, best_seq = official, a_refined.tolist()
            print(f"   *** New global best: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2. Global polishing on the current best
    # ------------------------------------------------------------------
    if best_seq:
        a_arr = np.array(best_seq, dtype=float)

        # KL‑mirror polishing (exponentiated gradient)
        if time.time() < deadline - 20.0:
            print("\nStarting KL‑mirror (exponentiated‑gradient) polishing …")
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline,
                max_iter=80_000,
                step=0.015,
            )
            try:
                val = evaluate_sequence(a_arr.tolist())
                print(f"   After EG polish official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"   *** Improved after EG polish: {best_val:.6f} ***")
            except Exception as e:
                print(f"   EG polishing failed: {e}")

        # Exact sub‑gradient polishing (short horizon)
        if time.time() < deadline - 10.0:
            print("\nFinal exact sub‑gradient polishing …")
            a_arr = subgradient_descent_exact(
                a_arr,
                deadline,
                max_iter=100_000,
                step0=0.07,
            )
            try:
                val = evaluate_sequence(a_arr.tolist())
                print(f"   After final sub‑gradient official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print("*** Global best improved by final sub‑gradient ***")
            except Exception as e:
                print(f"   Final sub‑gradient polishing failed: {e}")

        # Tiny random local search for any remaining edge‑case improvement
        if time.time() < deadline - 5.0:
            print("\nLocal random neighbourhood search …")
            a_arr = local_random_search(
                np.array(best_seq, dtype=float),
                deadline,
                trials=10_000,
                sigma=5e-4,
            )
            try:
                val = evaluate_sequence(a_arr.tolist())
                print(f"   After local search official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print("*** Improved after local random search ***")
            except Exception as e:
                print(f"   Local search evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a simple uniform distribution of moderate size
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
