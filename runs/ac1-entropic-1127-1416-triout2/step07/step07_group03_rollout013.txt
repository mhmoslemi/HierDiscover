"""
Propose a step‑function (non‑negative sequence) that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Seed generation** – create a diverse pool of candidates for a single
   length (default 4096).  Seeds are taken from analytic families
   (arcsine, beta, power‑law, arcsine/uniform mixture) and, if available,
   the previously best known sequence.

2. **Fast screening** – compute a cheap surrogate ``2·n·max(conv)`` (the
   exact value when the vector is normalised to unit L¹‑norm) and keep only
   the most promising seeds.

3. **Mirror‑descent (exponentiated‑gradient) optimisation** – starting from each
   top seed run a projected multiplicative update on the true non‑smooth
   objective.  The step size follows a 1/√t schedule, which makes the method
   aggressive early and stable later.  The iterate stays on the simplex
   automatically.

4. **Optional exact sub‑gradient polishing** – after the mirror‑descent
   phase a few iterations of Euclidean projected sub‑gradient descent are
   performed if time permits.

5. **Return** – the best sequence (unit L¹‑norm) found within the 1000 s budget.
   Because the unit‑norm objective equals ``evaluate_sequence`` this avoids an
   extra costly call.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (no circular wrap).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate for the official objective (unit L¹‑norm)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` when ``a`` sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for seed generation
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to unit L¹‑norm."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-12)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape, normalised to unit L¹‑norm."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law envelope, normalised to unit L¹‑norm."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (used by the fallback)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (optional final polish)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))
    staying on the probability simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional progress report
        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (KL‑mirror) descent on the true non‑smooth objective
# ----------------------------------------------------------------------


def exponentiated_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.35,
    max_iter: int = 10**9,
) -> np.ndarray:
    """
    Mirror descent with KL divergence (multiplicative updates) directly on
    the original max‑convolution objective.  The step size follows
    step = step0 / sqrt(t).  The iterate always stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        step = step0 / np.sqrt(it)
        a = a * np.exp(-step * subgrad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a

            cand = np.array(cand, dtype=float)
            cand = cand / cand.sum()               # ensure unit L¹‑norm
            val = compute_score(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand.tolist()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a single length (large enough to capture the shape)
    # ------------------------------------------------------------------
    # 4096 is a good compromise between quality and speed.
    n = 4096
    print(f"\n=== Optimising length {n} ===")

    # --------------------------------------------------------------
    # 1a. Build a rich pool of seeds
    # --------------------------------------------------------------
    seeds = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),          # uniform
    ]

    # random beta shapes
    for _ in range(12):
        a = rng.uniform(0.05, 2.5)
        b = rng.uniform(0.05, 2.5)
        eps = rng.uniform(0.0, 1.0)
        seeds.append(generate_beta_shape(n, a, b, eps))

    # random power‑law shapes
    for _ in range(12):
        p = rng.uniform(0.05, 2.5)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_power_shape(n, p, eps))

    # mixture of arcsine & uniform
    for _ in range(6):
        w = rng.uniform(0.0, 1.0)
        seeds.append(generate_mixture_shape(n, w))

    # include the current best (if any)
    if best_seq:
        old = np.array(best_seq, dtype=float)
        if old.size >= n:
            cand = old[:n]
        else:
            reps = int(np.ceil(n / old.size))
            cand = np.tile(old, reps)[:n]
        cand = np.clip(cand, 1e-15, None)
        cand = cand / cand.sum()
        seeds.append(cand)

    # --------------------------------------------------------------
    # 1b. Fast surrogate screening – keep only the most promising seeds
    # --------------------------------------------------------------
    scored = [(compute_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top_k = 5
    top_seeds = [vec for _, vec in scored[:top_k]]
    print(f"Top {top_k} seeds (surrogate scores):")
    for i, (val, _) in enumerate(scored[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # --------------------------------------------------------------
    # 1c. Mirror‑descent optimisation on each top seed
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    per_seed_budget = max(2.0, remaining / (len(top_seeds) + 1e-9))

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 0.5:
            break

        print(f"\nSeed {idx}/{len(top_seeds)} – allocated ≈{per_seed_budget:.1f}s")
        seed_deadline = min(time.time() + per_seed_budget, deadline - 0.2)

        # Run exponentiated‑gradient descent with decaying step size
        a_opt = exponentiated_gradient_descent(
            seed_vec,
            seed_deadline,
            step0=0.35,
        )

        val = compute_score(a_opt)
        print(f"   Final surrogate score = {val:.6f}")

        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()
            print(f"   *** New global best: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 5. Optional final polishing with Euclidean sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal exact sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=150_000,
            step0=0.08,
        )
        val2 = compute_score(a_arr)
        if val2 < best_val:
            best_val, best_seq = val2, a_arr.tolist()
            print(f"   *** Improved after final polish: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 256
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq, dtype=float))

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
