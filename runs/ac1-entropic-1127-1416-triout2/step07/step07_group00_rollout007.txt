"""
Hybrid analytical start + massive incremental hill‑climb.

* Initialise a non‑negative step‑function on the probability simplex by
  scanning a one‑parameter family a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α).

* From this start run a fast, fully‑vectorised random‑mass‑transfer
  optimisation.  Each move transfers a small fraction of the mass from a
  source index i to a destination j, updates the full a∗a convolution
  analytically (no FFT) and accepts the move if it improves the
  infinity‑norm of the convolution, or with a temperature‑controlled
  Metropolis probability to escape local minima.

* The step‑size (fraction of transferred mass) adapts to the recent
  acceptance ratio, and the convolution is periodically recomputed from
  scratch to avoid numerical drift.

The routine respects the 1000 s wall‑clock limit, prints periodic progress,
and finally returns the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective when sum(a)=1:
        score = 2·n·max(a∗a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytical family a_i(α) and simple search over α
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α)  – already on the simplex.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(
    n: int, deadline: float, max_grid: int = 30
) -> tuple[np.ndarray, float]:
    """Grid‑search α∈[0,2.5] and keep the best exact score."""
    best_a = None
    best_sc = np.inf
    alphas = np.linspace(0.0, 2.5, max_grid)
    for alp in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alp)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

# Main search routine
# ----------------------------------------------------------------------
