"""
propose_candidate
-----------------

A time‑bounded multi‑scale optimizer for the step‑function evaluation
problem.  The objective

    2·n·max(a⊛a)   (with ∑a = 1, a_i ≥ 0)

is the sup‑norm of the self‑convolution.  The algorithm proceeds in
stages:

1. **Coarse search** – start on a modest length (1024) and drive the
   convolution toward a flat (almost uniform) profile using an Adam
   optimiser on the variance of the convolution.

2. **Smooth max descent** – replace the hard max by a
   ε·log‑sum‑exp surrogate and continue with Adam while slowly shrinking
   ε, which gives a reliable path toward the true peak.

3. **Exact sub‑gradient** – once the surrogate is tiny we switch to a
   classic sub‑gradient descent on the exact max objective.

4. **Scale‑up** – the best vector is interpolated to a larger length
   (2048 → 4096 → 8192 → 16384).  The same three optimisation phases are
   repeated at each scale, exploiting the extra degrees of freedom to
   flatten the convolution further (the theoretical lower bound tends to
   1 as n → ∞).

5. **Stochastic hill‑climb** – a simulated‑annealing loop performs
   random mass transfers and swaps, occasionally accepting worsening
   moves to escape local minima.

All updates are projected onto the probability simplex (non‑negative,
∑a = 1) so the candidate is always valid for `evaluate_sequence`.  The
routine respects the supplied time budget (≈ 1000 s) and returns the
best sequence found.

The implementation is deliberately self‑contained, uses only NumPy and
the standard library, and prints occasional progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a) (the exact objective used by the judge).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and Σv = 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex via a Dirichlet."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


def upsample_vector(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a longer length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, a)
    # small numerical drift may move the sum away from 1 – re‑project
    return project_to_simplex(up)


# ----------------------------------------------------------------------
# Phase 2 – flattening (variance minimisation) via Adam
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 5000,
    step0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the convolution (equivalently L2 distance to a
    uniform profile) using Adam.  The gradient of the variance L = Σ (c‑μ)²
    is 4 * (a * (c‑μ))_|i|, where `*` denotes convolution.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mu = np.mean(conv)                     # = 1/(2n‑1) at optimum
        diff = conv - mu

        # Gradient = 4 * a * diff   (restricted to the original n entries)
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)      # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – smooth max (log‑sum‑exp) descent with Adam
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 8000,
    step0: float = 0.45,
    eps_factor: float = 0.25,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Optimise the smooth approximation
        f(a) = ε·log Σ exp(conv_k / ε)
    using Adam.  ε starts as a fraction of the current max and decays
    geometrically (never below 1 % of that max).  The gradient is
        g_i = 2·Σ_j a_j·w_{i+j}
    where w is the soft‑max distribution over the convolution entries.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # epsilon schedule
        eps = maxc * max(eps_factor * (0.997 ** (t - 1)), 0.01)

        # soft‑max weights (stable formulation)
        shifted = conv - maxc          # avoid overflow
        exp_vals = np.exp(shifted / eps)
        w = exp_vals / np.sum(exp_vals)   # Σ w = 1

        # gradient = 2 * a * w   (convolution with reversed w)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to avoid stagnation
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – exact sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on   f(a) = 2·n·max(a⊛a).
    The sub‑gradient at the peak shift k is
        g_i = 2·a_{k-i}   (indices out of range give 0).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build sub‑gradient
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional random shake to boost exploration
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 5 – simulated‑annealing hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    deadline,
    max_steps: int = 250_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with a linear cooling schedule.
    Allowed moves:
      * swap two positions,
      * transfer a fraction of the mass from i → j.
    The move is always accepted if it improves the exact score,
    otherwise accepted with probability exp(-Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.14  # initial temperature (tuned empirically)

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction (up to 6 %) of source mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.06 * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # larger perturbation every few thousand steps
        if step % 6000 == 0:
            rnd = random_dense_vector(n) * 0.025
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Make the vector symmetric (a+reverse(a))/2 and re‑project."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the time budget (≈ 1000 s) and returns a plain
    Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the global best
    # ------------------------------------------------------------------
    def maybe_update(candidate):
        nonlocal best_seq, best_score
        sc = compute_score(candidate)
        if sc < best_score:
            best_score = sc
            best_seq = candidate.tolist()
            print(f"   *** new global best: score={best_score:.6f} (n={len(candidate)})")
        return sc

    # ------------------------------------------------------------------
    # Multi‑scale schedule – start small, gradually grow.
    # ------------------------------------------------------------------
    scales = [1024, 2048, 4096, 8192, 16384]   # all powers of two for FFT speed

    for scale_idx, n in enumerate(scales, 1):
        if time.time() >= deadline - 8.0:
            break

        print(f"\n=== Scale {scale_idx}/{len(scales)}: n = {n} ===")

        # --- initialise -------------------------------------------------
        if best_seq is not None:
            # use the current best, up‑sample if necessary
            a0 = upsample_vector(np.array(best_seq, dtype=float), n)
        else:
            a0 = random_dense_vector(n)

        # quick sanity check
        maybe_update(a0)

        # ------------------ Phase 2 : flattening -----------------------
        remaining = deadline - time.time()
        if remaining < 4.0:
            break
        a_flat, sc_flat = flatten_conv_adam(
            a0,
            deadline=deadline - 2.0,
            max_iter=5000,
            step0=0.14,
        )
        maybe_update(a_flat)

        # ------------------ Phase 3 : smooth max -----------------------
        remaining = deadline - time.time()
        if remaining < 6.0:
            break
        a_smax, sc_smax = smooth_max_adam(
            a_flat,
            deadline=deadline - 3.0,
            max_iter=8000,
            step0=0.48,
            eps_factor=0.23,
        )
        maybe_update(a_smax)

        # ------------------ Phase 4 : exact sub‑gradient ---------------
        remaining = deadline - time.time()
        if remaining > 12.0:
            a_sub, sc_sub = subgrad_optimize(
                a_smax,
                deadline=deadline - 5.0,
                max_iter=80000,
                step0=0.06,
            )
            maybe_update(a_sub)

        # ------------------ Phase 5 : hill‑climb anneal ----------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            seq_to_polish = best_seq if best_seq is not None else a_sub.tolist()
            polished_seq, polished_sc = hill_climb_anneal(
                seq_to_polish,
                deadline=deadline - 2.0,
                max_steps=200_000,
            )
            maybe_update(np.array(polished_seq, dtype=float))

            # symmetry averaging (cheap but often beneficial)
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            maybe_update(a_sym)

        # Early stop if we have already passed the target 1.503
        if best_score <= 1.503:
            print("Target achieved – early exit.")
            break

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should never happen) use uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq, dtype=float))

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
