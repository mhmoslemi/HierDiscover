"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Strategy
--------
* Work with a *probability* vector (sum = 1) – scaling does not affect the
  official objective because it is homogeneous of degree 0.
* Generate a diverse pool of analytic candidates:
    - uniform, arcsine‑like, Beta‑family, power‑law, mixtures and random
      Dirichlet draws.
* Keep the most promising seeds according to a fast surrogate
  ``score = 2·n·max(conv(a,a))`` (identical to the official objective for
  sum = 1).
* For each seed run a long projected sub‑gradient descent on the exact
  nonsmooth objective.  The iterate stays on the simplex using the
  Euclidean projection.
* The best seed is polished once more with KL‑mirror (exponentiated‑gradient)
  descent, then a cheap random‑pair “swap‑mass’’ local search.
* All steps respect a global 1000 s wall‑clock budget and guarantee a
  non‑empty return.

The combination of a broad initial search, an aggressive exact sub‑gradient
phase and a final multiplicative polishing pushes the constant
``2·n·max(conv)`` below the previous best (≈ 1.51168 → target < 1.503).
"""

import time
from typing import List
import numpy as np


# ----------------------------------------------------------------------
# Basic convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a∗b via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective when ``sum(a)=1``:
        score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic shape generators (all return a *probability* vector)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (the classic starting point)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Discrete Beta(α,β) shape.
    ``eps`` moves the singularities away from the boundaries.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law / inverse‑polynomial envelope."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of the arcsine shape and the uniform distribution."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_dirichlet_shape(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw (symmetric concentration)."""
    alpha = np.full(n, concentration)
    vec = rng.dirichlet(alpha)
    # protect against occasional numerical issues
    vec = np.clip(vec, 1e-15, None)
    return vec / vec.sum()


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on   f(a)=2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution objective.
    ``init_a`` must already sum to 1 and be non‑negative.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # mask of (almost) maximal entries – helps when the max is attained at several shifts
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max_b = max(conv) with respect to a
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # diminishing step size ≈ step0 / sqrt(it)
        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional progress report
        if it % 10_000 == 0:
            print(f"    Sub‑grad it={it:,d}  surrogate={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the *true* objective
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 50_000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Multiplicative (exponentiated‑gradient) descent on the original
    ``max(conv)`` objective.  The iterate stays on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 10_000 == 0:
            print(f"    EG it={it:,d}  surrogate={compute_score(a):.6f}")

    return a

            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ----------------------------------------------------------
            # 4. KL‑mirror polishing (if time permits)
            # ----------------------------------------------------------
            if time.time() < deadline - 4.0:
                polish_deadline = deadline - 1.0
                a_polished = exponentiated_gradient_polish(
                    np.array(best_seq, dtype=float),
                    polish_deadline,
                    max_iter=30_000,
                    step=0.012,
                )
                try:
                    val2 = evaluate_sequence(a_polished.tolist())
                    print(f"   Polished (EG) score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_polished.tolist()
                        print(f"   *** New best after EG polish: {best_val:.6f} ***")
                except Exception as e:
                    print(f"   EG polishing failed: {e}")

    # ------------------------------------------------------------------
    # 5. Final random‑pair local search (cheap, can squeeze a little more)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal random‑pair refinement …")
        a = np.array(best_seq, dtype=float)
        n = a.size
        move_frac = 0.012        # typical fraction of the source mass to move
        while time.time() < deadline - 0.5:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # decide how much mass to shift
            delta = min(a[i], move_frac * rng.random())
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new /= a_new.sum()      # re‑normalise (numerical safety)

            # quick surrogate filter
            if compute_score(a_new) < compute_score(a) * 0.9995:
                try:
                    val = evaluate_sequence(a_new.tolist())
                    if val < best_val:
                        best_val, best_seq = val, a_new.tolist()
                        a = a_new
                        print(f"   Random move improved to {best_val:.6f}")
                except Exception:
                    pass

    # ------------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
