"""
Hybrid parametric + gradient search for a low‑value step sequence.

The algorithm proceeds in three phases:

1. **Broad parametric seeding** –  a large pool of analytic shapes
   (arcsine, power‑law, beta‑like, mixtures and simple triangles) is
   generated for a fixed length (default 4096).  Every candidate is
   normalised to sum = 1 and scored with a very cheap surrogate
   `2·n·max(conv)` obtained via an FFT‑based convolution.  The best
   few (≤15) seeds are kept.

2. **Smooth‑max Adam refinement** – each retained seed is refined by
   minimising a smooth log‑sum‑exp approximation of the max‑convolution
   (`β` is ramped from 5 to a large value).  The variables are kept on
   the probability simplex via a soft‑max parametrisation and Adam
   updates.

3. **Exact polishing** – the result of Adam is further improved with
   KL‑mirror (exponentiated‑gradient) descent on the true
   non‑smooth objective, followed by a long projected sub‑gradient
   descent.  The best sequence found according to the official
   ``evaluate_sequence`` function is returned.  The whole pipeline
   respects a 1000 s wall‑clock budget and always yields a non‑empty
   valid sequence.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective when the vector already
    sums to 1:   2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Families of analytic initial shapes (all return a vector summing to 1)
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.25) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_shape(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def triangular_shape(n: int) -> np.ndarray:
    """Simple symmetric triangle: ramps up then down."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=float)
    dec = np.arange(n - half, 0, -1, dtype=float)
    tri = np.concatenate([inc, dec])[:n]
    tri /= tri.sum()
    return tri


# ----------------------------------------------------------------------
# Projection utilities for the simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 150.0,
    ramp_iters: int = 30000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Minimise the smooth log‑sum‑exp surrogate of max(conv) using Adam.
    The iterates stay on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly raise the temperature β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp weighting
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2 * (a ⋆ w_rev) (restricted to original length)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate score={compute_score(a):.6f}")

    return softmax(z)  # final vector on the simplex


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 50000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Mirror descent on the original max‑convolution objective using the
    KL divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for the set of active (maximal) indices
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} surrogate score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # mask of (near)‑maximal entries
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} surrogate score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Candidate‑generation utilities
# ----------------------------------------------------------------------


def generate_candidate_pool(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Return a diverse list of analytic seeds of length ``n``."""
    seeds: List[np.ndarray] = []

    # arcsine family with several epsilons
    for eps in [0.0, 0.1, 0.25, 0.5, 1.0, 2.0]:
        seeds.append(arcsine_shape(n, eps))

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # random power‑law shapes
    for _ in range(30):
        p = rng.uniform(0.1, 4.0)
        eps = rng.uniform(0.0, 4.0)
        seeds.append(generate_power_shape(n, p, eps))

    # random beta‑like shapes
    for _ in range(30):
        a = rng.uniform(0.1, 6.0)
        b = rng.uniform(0.1, 6.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(generate_beta_shape(n, a, b, eps))

    # mixture of arcsine & uniform
    for _ in range(10):
        w = rng.uniform(0.0, 1.0)
        seeds.append(generate_mixture_shape(n, w))

    # simple symmetric triangle
    seeds.append(triangular_shape(n))

    return seeds


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The routine respects the 1000 s wall‑clock budget and always returns
    a non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to a previously saved sequence if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Primary optimisation length (larger lengths often give tighter bounds)
    # ------------------------------------------------------------------
    n = 4096
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 1a.  Build a large pool of analytic seeds
    # ------------------------------------------------------------------
    seeds = generate_candidate_pool(n, rng)

    # Include previous best (rescaled / tiled) if we have one
    if best_seq:
        old = np.array(best_seq, dtype=float)
        if old.size >= n:
            cand = old[:n]
        else:
            reps = int(np.ceil(n / old.size))
            cand = np.tile(old, reps)[:n]
        cand = np.clip(cand, 1e-15, None)
        cand = cand / cand.sum()
        seeds.append(cand)

    # ------------------------------------------------------------------
    # 1b.  Fast surrogate screening – keep only the most promising seeds
    # ------------------------------------------------------------------
    scored = [(compute_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top_k = min(15, len(scored))
    top_seeds = [vec for _, vec in scored[:top_k]]
    print(f"Selected top {top_k} seeds after surrogate screening.")

    # ------------------------------------------------------------------
    # 2.  Refine each top seed with Adam → EG → evaluation
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break

        print(f"\n--- Seed {idx}/{len(top_seeds)} (n={n}) ---")
        # allocate a time slice for this seed (roughly 30 % of the remaining budget)
        remaining = deadline - time.time()
        if remaining < 30.0:
            print("  Not enough time left for another seed.")
            break
        seed_time = min(40.0, remaining * 0.35)
        seed_deadline = time.time() + seed_time

        # ---- Adam on smooth surrogate ----
        z0 = np.log(seed_vec + 1e-12)
        a_adam = smooth_adam_optimize(
            z0,
            seed_deadline,
            n=n,
            beta0=5.0,
            beta_max=150.0,
            ramp_iters=30000,
            lr=0.07,
        )

        # ---- KL‑mirror polishing (exponentiated gradient) ----
        # Give it as much remaining time as possible, but leave a tiny margin.
        eg_deadline = min(deadline - 2.0, time.time() + 20.0)
        a_eg = exponentiated_gradient_polish(
            a_adam,
            eg_deadline,
            max_iter=50000,
            step=0.015,
        )

        # ---- Official evaluation ----
        try:
            val = evaluate_sequence(a_eg.tolist())
            print(f"   Official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_eg.tolist()
                print(f"   *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3.  Global exact sub‑gradient polishing on the current best
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global exact sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=250_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
