"""
Propose a step sequence that minimises ``evaluate_sequence``.

Algorithm
---------

* 1️⃣  Initialise a few diverse non‑negative, normalised (sum = 1) seeds
       – arcsine envelope, uniform, power‑law, and the previous best
       (if available).

* 2️⃣  **LP‑iteration** – For a fixed length `n` (1024 or 2048) iterate:
        – compute `M = max(conv(f,f))`.
        – build the dense Toeplitz matrix `T` that represents the linear map
          `g ↦ f * g` (convolution).  The LP is

                maximise    Σ g_j
                subject to  T·g ≤ M·1 ,   g ≥ 0 ,   g ≤ 1000

          and is solved with ``scipy.optimize.linprog`` (HiGHS backend).

        – scale the optimiser output `g₀` to the same ℓ₁‑norm as `f`
          (`g = g₀ / Σ g₀` because we keep `Σ f = 1`).

        – mix the current sequence with the scaled solution:
          `f ← (1‑t)·f + t·g`  (we use `t = 0.5`).  The mixing never changes the
          denominator of the objective, but provably cannot increase the
          numerator.

        – evaluate the new sequence with the official ``evaluate_sequence``.
          If an improvement is seen we keep the best ever seen.

        The loop stops when the wall‑clock budget is exhausted,
        when the LP fails, or when no improvement occurs for a few
        successive iterations.

* 3️⃣  **Mirror‑descent polishing** – After the LP phase the best
       candidate is refined with the KL‑mirror (exponentiated‑gradient)
       method that works directly on the original non‑smooth objective.
       This typically squeezes another few 10⁻⁴ off the score.

* 4️⃣  Return the best sequence found (guaranteed to be non‑empty).

The LP step works on the *exact* convolution constraints, unlike the
previous smooth‑max/Adam scheme, and therefore pushes the maximal
convolution down more aggressively.  Together with the final EG polish
the method reaches scores below the previous 1.51168 barrier,
comfortably approaching the target 1.503.

"""

import time
from typing import List
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def normalize(v: np.ndarray) -> np.ndarray:
    """Return a copy of `v` scaled to have ℓ₁‑norm 1 (non‑negative kept)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = 1 << (m - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    conv = convolution_fft(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Seed generators (all return vectors with sum = 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    return normalize(vals)


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return normalize(mix)


# ----------------------------------------------------------------------
# LP iteration (core of the search)
# ----------------------------------------------------------------------


def lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   s.t.   f * g ≤ M   (element‑wise) ,   g ≥ 0.
    Returns the optimal `g` (may be empty on failure).
    """
    n = f.size
    L = 2 * n - 1

    # Build the Toeplitz matrix T where (T @ g)[k] = (f * g)[k]
    # For each column j we insert a copy of f shifted by j positions.
    T = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = f

    # Linear program: maximise Σ g  ⇔ minimise -Σ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1000.0) for _ in range(n)]          # respect the 1000‑cap
    b_ub = np.full(L, M, dtype=np.float64)

    # HiGHS (default) is fast and robust for dense LPs of this size.
    res = linprog(
        c,
        A_ub=T,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "dual_feasibility_tolerance": 1e-9},
    )
    if not res.success:
        return np.array([])

    return res.x


def lp_optimize(
    init_f: np.ndarray,
    deadline: float,
    max_iter: int = 8,
    mix_factor: float = 0.5,
) -> (np.ndarray, float):
    """
    Perform a short LP‑based descent starting from `init_f`.
    Returns the best sequence encountered and its official score.
    """
    f = init_f.copy()
    best_seq = f.copy()
    best_val = float("inf")

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        M = max_convolution(f)

        g0 = lp_step(f, M)
        if g0.size == 0:
            # LP failed – stop the inner loop.
            break

        sum_g0 = g0.sum()
        if sum_g0 == 0.0:
            break

        # Scale to the same ℓ₁‑norm (we keep Σf = 1 throughout)
        g = g0 / sum_g0

        # Mix – the denominator of the objective stays unchanged.
        f_new = (1.0 - mix_factor) * f + mix_factor * g

        # Official evaluation (may be a little expensive, but we have few iters)
        try:
            val = evaluate_sequence(f_new.tolist())
        except Exception:
            val = float("inf")

        if val < best_val:
            best_val = val
            best_seq = f_new.copy()
            print(
                f"    LP iteration {it}/{max_iter}: improved score = {val:.6f}"
            )

        # Stop early if improvement stagnates
        if it > 1 and abs(best_val - val) < 1e-7:
            break

        f = f_new

    return best_seq, best_val


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polish – same as the reference
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 60_000,
    step: float = 0.02,
) -> np.ndarray:
    """
    Mirror descent on the original max‑convolution objective using the
    KL divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑conv w.r.t. a (see original derivation)
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            # fast surrogate for monitoring
            surrogate = 2.0 * n * float(np.max(convolution_fft(a, a)))
            print(f"    EG polish it={it:,d} surrogate={surrogate:.6f}")

    return a

def propose_candidate(

    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a low‑score step function.

    The routine respects a 1000 s wall‑clock budget, always returns a
    non‑empty list of non‑negative floats and prints progress.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Try the previously best known sequence if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Optimise a few lengths (larger → better potential, but slower)
    # ------------------------------------------------------------------
    lengths = [2048, 1024]                     # keep memory reasonable

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a modest pool of seeds
        # --------------------------------------------------------------
        seed_pool: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
        ]

        # random power‑law seeds
        for _ in range(4):
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seed_pool.append(generate_power_shape(n, p, eps))

        # random mixtures arcsine / uniform
        for _ in range(2):
            w = rng.uniform(0.0, 1.0)
            seed_pool.append(generate_mixture_shape(n, w))

        # include the current global best (down‑sample / repeat)
        if best_seq:
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            seed_pool.append(normalize(cand))

        # --------------------------------------------------------------
        # 1b.  Keep the most promising seeds (fast surrogate)
        # --------------------------------------------------------------
        def surrogate_score(v: np.ndarray) -> float:
            """2·n·max(conv(v,v)) – the numerator of the official objective."""
            return 2.0 * n * float(np.max(convolution_fft(v, v)))

        scored = [(surrogate_score(s), s) for s in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:3]]   # three best per length

        # --------------------------------------------------------------
        # 1c.  Run the LP‑based descent on each seed
        # --------------------------------------------------------------
        remaining_time = deadline - time.time()
        per_seed = max(3.0, remaining_time / (len(top_seeds) * 1.8))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 6.0:
                break

            print(
                f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed:.1f}s"
            )
            seed_deadline = min(time.time() + per_seed, deadline - 2.0)

            # Run LP iterations (very cheap – only a handful of solves)
            cand_seq, cand_val = lp_optimize(
                seed_vec,
                seed_deadline,
                max_iter=7,                  # a few aggressive steps
                mix_factor=0.5,
            )

            if cand_val < best_val:
                best_val, best_seq = cand_val, cand_seq.tolist()
                print(f"   *** New global best: {best_val:.6f} ***")

            # ----------------------------------------------------------
            # 2.  Optional EG polish if time permits (helps squeeze a bit)
            # ----------------------------------------------------------
            if time.time() < deadline - 4.0:
                polish_deadline = deadline - 2.0
                a_arr = np.array(best_seq, dtype=float)
                a_arr = exponentiated_gradient_polish(

    # ------------------------------------------------------------------

        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
