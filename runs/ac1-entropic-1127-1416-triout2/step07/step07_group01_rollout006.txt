"""
Improved optimiser for the `evaluate_sequence` challenge.

The optimiser follows three phases, each respecting the 1000 s wall‑clock
budget:

1. **Multiple Adam‑accelerated smooth‐max runs** – starting from random
   dense or sparse probability vectors we minimise a smoothed
   approximation of the true maximum convolution using an Adam style
   momentum scheme.  The schedule for the smoothing parameter β goes
   from a modest value (≈5) up to a very large value (≈150) so that the
   optimiser eventually focuses on the true max.

2. **Projected sub‑gradient descent** on the exact max‑objective.
   This stage cleans up any remaining overshoot from the smooth phase.

3. **Two complementary polishing stages**  
   – a generic hill‑climb that swaps or transfers a fraction of mass,
   – a *targeted* hill‑climb that explicitly attacks the index attaining
   the maximal convolution value by moving mass away from the most
   harmful coordinates to locations that do not contribute to that same
   index.  Both stages use a simple adaptive step‑size.

All helper routines are defined at the module level (no closures).  The
function `propose_candidate` is the entry point required by the
harness; it returns a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generator
# ----------------------------------------------------------------------


def random_start(n: int) -> np.ndarray:
    """Return a random probability vector (dense or sparse) of length n."""
    if np.random.rand() < 0.6:               # dense start (60 %)
        a = np.random.rand(n)
        a = a / a.sum()
    else:                                    # sparse start (40 %)
        p = np.random.uniform(0.03, 0.5)
        k = max(2, int(round(p * n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
        a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Smoothed max‑objective optimisation (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 150.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a,a)) using Adam‑style momentum.
    The routine keeps the best exact (non‑smooth) score encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smooth objective & weights --------------------------------
        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()              # probability vector

        # ---- gradient (cross‑correlation) -------------------------------
        a_rev = a[::-1]
        # convolve reversed a with the weight distribution; the result
        # has length 3n‑2, we need the central n elements.
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]        # slice of length n

        # ---- Adam update -----------------------------------------------
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final exact evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation of the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that swaps entries or transfers a small fraction
    of mass between two positions, always projecting back onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb that attacks the maximal convolution index
# ----------------------------------------------------------------------


def targeted_hill_climb(
    seq,
    time_deadline,
    max_iter: int = 300_000,
) -> tuple[list[float], float]:
    """
    Greedy mass‑transfer moves that directly reduce the entry of the
    convolution attaining the current maximum.  In each iteration the
    algorithm identifies the index `i` contributing most to the maximal
    convolution value and moves a fraction of its mass to a location that
    does not affect that same convolution entry.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_frac = 0.12  # initial transfer fraction

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()
        idx = int(np.argmax(conv))

        # contributions of each coordinate to conv[idx]
        i_arr = np.arange(n)
        j_arr = idx - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        i_max = int(np.argmax(contrib))
        # choose a destination that does *not* contribute to this max entry
        out_of_range = ~mask
        if np.any(out_of_range):
            dest_candidates = np.where(out_of_range)[0]
        else:
            # fallback: pick among indices with smallest contribution
            k_small = max(1, n // 200)  # ~0.5 % of positions
            dest_candidates = np.argpartition(contrib, k_small)[:k_small]

        p = int(np.random.choice(dest_candidates))

        # compute how much mass to move
        delta = step_frac * a[i_max]
        if delta > a[i_max]:
            delta = a[i_max]
        if delta <= 0:
            # nothing to move – shrink step and continue
            step_frac = max(step_frac * 0.8, 1e-4)
            continue

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[p] += delta

        # safeguard against numerical drift
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            a = a_new
            step_frac = min(step_frac * 1.06, 0.6)
        else:
            step_frac *= 0.94  # become more conservative after a miss

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    Search for a non‑negative sequence that minimises the provided
    `evaluate_sequence`.  The routine respects the 1000 s wall‑clock
    budget and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # 0. optional fallback from a previously discovered sequence
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # Choose a reasonable length (larger often gives more freedom)
    # -------------------------------------------------------------
    length = 4096
    if (deadline - time.time()) > 500 and budget_s >= 800:
        length = 8192
    n = length
    print(f"Chosen sequence length = {n}")

    # -------------------------------------------------------------
    # 1. Multiple Adam‑accelerated smooth optimisation restarts
    # -------------------------------------------------------------
    # Allocate roughly 60 % of remaining budget for this phase.
    smooth_phase_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.60)

    while time.time() < smooth_phase_deadline - 120.0:
        # give each restart a modest time slice (≈30 s) so that we get many starts
        restart_deadline = time.time() + 30.0
        a0 = random_start(n)
        a_opt, scr = smooth_gradient_optimize(
            a0,
            time_deadline=restart_deadline,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=150.0,
            step0=0.9,
        )
        if scr < best_val:
            best_val = scr
            best_seq = a_opt.tolist()
            print(f"New best after smooth restart: score={best_val:.6f}")

    # -------------------------------------------------------------
    # 2. Sub‑gradient refinement (exact max)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print("\nSub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 90.0,
            max_iter=400_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Improved by sub‑grad: score={best_val:.6f}")

    # -------------------------------------------------------------
    # 3. First polishing: generic hill‑climb
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print("\nGeneric hill‑climb polishing")
        polished_seq, polished_score = hill_climb_permutation(
            best_seq,
            time_deadline=deadline - 60.0,
            max_steps=500_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Generic hill‑climb improved best to {best_val:.6f}")

    # -------------------------------------------------------------
    # 4. Targeted hill‑climb focusing on the maximal convolution entry
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print("\nTargeted hill‑climb polishing")
        targeted_seq, targeted_score = targeted_hill_climb(
            best_seq,
            time_deadline=deadline,
            max_iter=300_000,
        )
        if targeted_score < best_val:
            best_val = targeted_score
            best_seq = targeted_seq
            print(f"Targeted hill‑climb improved best to {best_val:.6f}")

    # -------------------------------------------------------------
    # Safeguard – fall back to a modest uniform vector if nothing found
    # -------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (uses the harness‑provided function)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
