def random_start(n):
    r = np.random.rand()
    if r < 0.4:
        a = np.random.rand(n)
        a = a / a.sum()
    elif r < 0.7:
        # sparse

    else:

        alpha = np.random.uniform(0.3, 1.2)

"""
Non‑negative step‑function optimiser.

The algorithm consists of four stages:

1.  **Fallback** – try a previously‑found sequence if it exists.
2.  **LP‑based minimax** – starting from an arcsine (Beta(0.5,0.5)) shape we
    solve a linear programme `max Σ g_j`     s.t. conv(f,g) ≤ M,
    where `M = max(conv(f,f))`.  The solution is mixed with the current
    `f` (convex combination) and the process is repeated a few times.
    This dramatically flattens the auto‑convolution.
3.  **Smooth max‑objective optimisation** – many random restarts are
    refined with a log‑sum‑exp smooth surrogate of the max‑operator,
    using Adam‑style momentum and a schedule for the smoothing parameter.
4.  **Final polishing** – a simulated‑annealing hill‑climb on the exact
    objective finishes the search.

All steps respect the 1000 s wall‑clock budget and the routine always
returns a plain list of floats (the best sequence found).
"""

import time
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Parameterised beta‑shape start vectors
# ------------------------------------------------------------

def beta_shape(n: int, alpha: float) -> np.ndarray:
    """Discrete Beta(alpha,alpha) density on n points, normalised to sum = 1."""
    i = np.arange(n) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    dens = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
    dens = dens / dens.sum()
    return dens


def random_start_vector(n: int) -> np.ndarray:
    """Generate a diverse random starting probability vector."""
    r = np.random.rand()
    if r < 0.4:                                   # dense random
        a = np.random.rand(n)
        a = a / a.sum()
    elif r < 0.7:                                 # sparse random
        k = max(2, int(round(np.random.uniform(0.02, 0.15) * n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
    else:                                          # beta‑shaped
        alpha = np.random.uniform(0.3, 1.2)
        a = beta_shape(n, alpha)
    return a

# Linear‑programming minimax optimisation
# ------------------------------------------------------------

try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that
    conv(f, g) = A @ g  for any g.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def lp_minimax_optimize(start_f: np.ndarray,
                        deadline: float,
                        max_outer_iters: int = 6,
                        mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Iteratively improve `f` by solving a LP that maximises the total mass
    of a companion vector `g` under the constraint   conv(f,g) ≤ M,
    where `M` is the current maximal auto‑convolution value.
    The new `f` is a convex combination of the old one and the normalised
    solution `g`.  Returns the best vector seen and its exact score.
    """
    if not SCIPY_AVAILABLE:
        return start_f, compute_score(start_f)

    f = start_f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    n = f.shape[0]

    for it in range(max_outer_iters):
        if time.time() >= deadline:
            break

        # maximal auto‑convolution of the current f
        M = float(convolution_fft(f, f).max())

        # construct the linear constraints A g ≤ M
        A = build_conv_matrix(f)

        # linear programme: maximise Σ g_j  (i.e. minimise −Σ g_j)
        c = -np.ones(n, dtype=float)
        b = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # LP failed – abort the loop
            print(f"LP iteration {it}: solver failed ({res.message})")
            break

        g = res.x
        sum_g = g.sum()

        # If the LP cannot increase the total mass we have reached a fixed point
        if sum_g <= 1.0 + 1e-12:
            print(f"LP iteration {it}: sum(g) = {sum_g:.6f} ≤ 1  →  stop")
            break

        # normalise g and mix it with the current f
        g_norm = g / sum_g
        f = (1.0 - mix_factor) * f + mix_factor * g_norm
        f = project_to_simplex(f)          # guard against numerical drift

        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"LP iteration {it}: improved score = {best_score:.6f}")

    return best_f, best_score

# Smoothed max‑objective optimisation (log‑sum‑exp) with Adam
# ------------------------------------------------------------

def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 6.0,
    beta_end: float = 200.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a,a)) using Adam‑style
    momentum.  Returns the best exact score encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max (log‑sum‑exp)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smoothed objective (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

def hill_climb_anneal(seq, time_deadline, max_steps: int = 800_000):
    """
    Stochastic hill‑climb with occasional uphill moves (Metropolis rule).
    Always stays on the probability simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # a rough lower bound (average value of the convolution)
    lower_bound = 2.0 * n / (2 * n - 1.0)
    temperature0 = max(0.001, (best_score - lower_bound) * 0.2)

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        cand = best_a.copy()

        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a small fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = np.random.rand() * 0.05 * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
        else:
            # Metropolis acceptance
            temp = temperature0 * (1.0 - step / max_steps)
            if temp > 0.0 and np.random.rand() < np.exp(-(cur_score - best_score) / temp):
                best_a = cand

        # occasional global jitter to shake the system
        if step % 5000 == 0:
            jitter = np.random.randn(n) * 0.01
            best_a += jitter
            best_a = project_to_simplex(best_a)

    return best_a.tolist(), best_score

        print("\nStage 1: LP‑based minimax optimisation")
        # start from a smooth arcsine shape
        f0 = beta_shape(n, alpha=0.5)
        f0 = project_to_simplex(f0)
        f_opt, score_opt = lp_minimax_optimize(f0, deadline)
        if score_opt < best_val:
            best_val = score_opt
            best_seq = f_opt.tolist()
            print(f"LP stage best score = {best_val:.6f}")
    else:
        print("\nStage 1 skipped (scipy not available or insufficient time)")

    # ---------------------------------------------------------
    # 2. Smooth log‑sum‑exp Adam optimiser with many random restarts
    # ---------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 250.0:
        print("\nStage 2: Smooth Adam optimisation (random restarts)")
        # keep a buffer for the later stages
        stage2_deadline = time.time() + remaining - 160.0
        restart_id = 0
        while time.time() < stage2_deadline:
            a0 = random_start_vector(n)
            a_opt, scr = smooth_gradient_optimize(
                a0,
                time_deadline=stage2_deadline,
                max_iters=400_000,
                beta_start=5.0,
                beta_end=180.0,
                step0=0.9,
            )
            if scr < best_val:
                best_val = scr
                best_seq = a_opt.tolist()
                print(f"Restart {restart_id}: new best = {best_val:.6f}")
            restart_id += 1

    # ---------------------------------------------------------
    # 3. Sub‑gradient refinement on the current best vector
    # ---------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 120.0:
        print("\nStage 3: Sub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 70.0,
            max_iter=600_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑grad improved best to {best_val:.6f}")

    # ---------------------------------------------------------
    # 4. Final hill‑climb polishing (simulated annealing)
    # ---------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 10.0:
        print("\nStage 4: Hill‑climb polishing")
        polished_seq, polished_score = hill_climb_anneal(
            best_seq,
            time_deadline=deadline,
            max_steps=1_200_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ---------------------------------------------------------
    # Safeguard – fall back to a modest uniform vector if nothing found
    # ---------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
