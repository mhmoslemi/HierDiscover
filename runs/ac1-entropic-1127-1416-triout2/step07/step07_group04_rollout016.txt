"""
Optimiser for the “step‑function” attack.

The goal is to minimise

    score = 2·n·max(conv(a,a)) / (sum(a)²),

where ``a`` is a non‑negative sequence.  The score is scale‑invariant,
hence we keep the sequence normalised to the probability simplex
(∑a = 1).  The algorithm works in three phases:

1. **Coarse global search** – many random shape generators (arcsine,
   power‑law, Beta, mixture) are sampled.  Each seed is refined by a
   short Adam run on the smooth‑max approximation of the ‑∞‑norm of
   the convolution.

2. **Heavy polishing** – the best candidate (including the deterministic
   arcsine start) is further optimised with a long Adam stage,
   Euclidean projected sub‑gradient descent on the true max‑convolution,
   and a mirror‑descent (exponentiated‑gradient) finish.  The pipeline
   respects a global 1000 s wall‑clock budget.

3. **Final global polishing** – if time permits, the current best
   sequence is polished once more with generous iteration limits.

All helper routines are pure NumPy; no file or network access is used.
The function ``propose_candidate`` is the entry point that returns the
best sequence found before the deadline.
"""

import time
import numpy as np
from typing import List, Tuple


# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    v = v.astype(float, copy=True)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score for a probability vector (∑a = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a uniform grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 1e-4
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-6) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^‑p, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float,
                       eps: float = 1e-6) -> np.ndarray:
    """Beta‑type shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n points, then renormalise."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth max (log‑sum‑exp) of the convolution.
    The optimisation variable is the logits w, with ``a = softmax(w)``.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # initialise logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule for the smooth‑max
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a (guarantees a ≥ 0, Σa = 1)
        w_shift = w - np.max(w)          # stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)     # length L

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp   # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)   # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 25000 == 0:
            surrogate = compute_score(a)
            print(f"    [Adam] it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.06) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev = mask[::-1]
        conv_rev = convolution_fft(a, rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev = mask[::-1]
        conv_rev = convolution_fft(a, rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single seed
# ----------------------------------------------------------------------


def run_optimisation(init_a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Runs the three‑stage optimisation (Adam → sub‑gradient → EG) on *init_a*
    and returns the polished sequence together with its official score
    (computed via ``evaluate_sequence``).  All stages respect *deadline*.
    """
    # --- Stage 1: smooth‑max Adam ------------------------------------
    rem = deadline - time.time()
    if rem <= 0:
        return init_a, float("inf")
    max_iters_adam = int(min(300_000, rem * 30))
    a = adam_optimize_smooth_max(init_a, deadline,
                                 max_iters=max_iters_adam,
                                 tau_start=0.5,
                                 tau_end=400.0,
                                 base_lr=0.5)

    # --- Stage 2: exact sub‑gradient descent -------------------------
    rem = deadline - time.time()
    if rem > 0:
        max_iters_sub = int(min(400_000, rem * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iters_sub,
                                      step0=0.06)

    # --- Stage 3: exponentiated‑gradient polish ----------------------
    rem = deadline - time.time()
    if rem > 0:
        max_iters_eg = int(min(300_000, rem * 30))
        a = eg_refine(a, deadline,
                      max_iter=max_iters_eg,
                      eta0=0.25)

    # --- Official evaluation -----------------------------------------
    try:
        val = evaluate_sequence(a.tolist())
    except Exception:
        val = float("inf")
    return a, val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine respects a global 1000 s wall‑clock budget,
    trying a deterministic arcsine start, then a handful of random seeds,
    each refined by a quick Adam pass followed by a full polish when
    promising.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose a working resolution (larger -> finer but slower)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 800:
        n_target = 8192
    elif time_left > 400:
        n_target = 4096
    else:
        n_target = 2048
    print(f"\nTarget length for the search: n = {n_target}")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic arcsine start (heavy polishing)
    # ------------------------------------------------------------------
    print("\n--- Phase 1: heavy optimisation of the deterministic arcsine start ---")
    a0 = arcsine_initial(n_target)
    a_opt, val = run_optimisation(a0, deadline)
    if val < best_val:
        best_val = val
        best_seq = a_opt.tolist()
        print(f"New global best after Phase 1: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – quick random restarts
    # ------------------------------------------------------------------
    shape_kinds = ["arcsine", "power", "beta", "mix"]
    restart_counter = 0
    while time.time() < deadline - 5.0:
        # limit the number of restarts to keep the runtime reasonable
        if restart_counter >= 30:
            break
        kind = rng.choice(shape_kinds)

        if kind == "arcsine":
            init = arcsine_initial(n_target)
        elif kind == "power":
            p = rng.uniform(0.6, 1.4)
            init = generate_power_shape(n_target, p, eps=1e-6)
        elif kind == "beta":
            alpha = rng.uniform(0.3, 2.5)
            beta = rng.uniform(0.3, 2.5)
            init = generate_beta_shape(n_target, alpha, beta, eps=1e-6)
        else:   # mix
            w = rng.uniform(0.0, 1.0)
            init = generate_mixture_shape(n_target, weight_arc=w)

        # small random perturbation and renormalisation
        noise = rng.normal(scale=0.02, size=n_target)
        init = np.maximum(init + noise, 0.0)
        init = init / init.sum()

        # ----- quick Adam run (no exact polishing) -----
        quick_rem = deadline - time.time()
        if quick_rem < 30.0:
            break
        quick_iters = int(min(120_000, quick_rem * 15))
        a_quick = adam_optimize_smooth_max(init, deadline,
                                            max_iters=quick_iters,
                                            tau_start=0.5,
                                            tau_end=150.0,
                                            base_lr=0.5)
        quick_score = compute_score(a_quick)

        # If the quick surrogate already beats the current best, run a full polish.
        if quick_score < best_val:
            print(f"\n[Restart {restart_counter}] Promising quick score {quick_score:.6f}"
                  f" (type={kind}) – entering full polishing")
            a_full, val_full = run_optimisation(a_quick, deadline)
            if val_full < best_val:
                best_val = val_full
                best_seq = a_full.tolist()
                print(f"*** New best after restart {restart_counter}: {best_val:.6f}")
        restart_counter += 1

    # ------------------------------------------------------------------
    # Phase 3 – final global polish (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Final global polish (extra time) ---")
        a = np.array(best_seq, dtype=float)
        a, val = run_optimisation(a, deadline)
        if val < best_val:
            best_val = val
            best_seq = a.tolist()
            print(f"*** Updated global best after final polish: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback (uniform) official score = {best_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
