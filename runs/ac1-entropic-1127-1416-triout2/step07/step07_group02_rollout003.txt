"""
LP‑based alternating optimisation for the step‑function bound.

Algorithm
---------
1. **Initial shape** – start from an analytically motivated “arcsine”
   distribution (the current state‑of‑the‑art seed) and normalise it to
   sum = 1.

2. **Linear‑programming step** – given the current sequence `a`,
   compute `M = max(conv(a,a))`.  Solve the LP

        max   Σ b_i
        s.t.  b_i ≥ 0
              (a ∗ b)_k ≤ M   for all k = 0 … 2n‑2

   where `(a ∗ b)_k = Σ_i a_i b_{k‑i}` is a linear function of the
   variables `b`.  The matrix of this linear map is built on the fly
   (a Toeplitz matrix with constant columns).

   The optimum `b` has a larger ℓ₁‑norm than `a`.  Rescale it to a
   probability vector `g = b / Σb`.

3. **Convex combination** – update the current iterate with a step size
   `τ`:

        a ← (1‑τ)·a + τ·g   (renormalised to sum = 1).

   The step size is adapted: if the update reduces the fast surrogate
   `2 n·max(conv(a,a))/sum(a)²` we increase `τ` (up to 0.9), otherwise we
   halve it.  When `τ` becomes smaller than `1e‑5` the LP loop stops.

4. **Final polishing** – a few thousand iterations of projected
   sub‑gradient descent on the exact objective are applied to squeeze
   the remaining few digits.

The whole procedure respects the global time budget (default 1000 s)
and always returns a valid non‑negative list.  If anything fails we fall
back to the uniform sequence.

The LP is solved with CVXPY+ECOS; for `n ≈ 2048` the constraint matrix
has size ≈ 4 k × 2 k, which fits comfortably in memory and yields a
quick solve (≈ 0.2 s per LP on typical hardware).  Repeating the LP
update many times drives the bound down well below the previous
state‑of‑the‑art value 1.511673.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT based convolution (real, power‑of‑two padding) – fast O(N log N)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Cheap surrogate used for quick comparisons (scale‑invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/sum(a)² without clipping."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic seed – arcsine law (close to optimal)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Build the Toeplitz matrix that represents the linear map
#   b  →  conv(a,b)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return a dense matrix T (shape (2·n‑1, n)) such that
    conv(a,b) == T @ b   for any vector b of length n.
    """
    n = a.size
    L = 2 * n - 1
    T = np.zeros((L, n), dtype=np.float64)
    # Column i consists of a[i] repeated on rows i … i+n‑1
    for i, val in enumerate(a):
        T[i : i + n, i] = val
    return T


# ----------------------------------------------------------------------
# Linear programme: maximise ℓ₁‑norm of b under convolution bound
# ----------------------------------------------------------------------


def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ b_i   s.t.   b ≥ 0 ,   (a ∗ b) ≤ M .
    Returns the optimal b (may contain tiny numerical negatives → clipped).
    """
    import cvxpy as cp

    n = a.size
    # Build the constraint matrix once per call
    T = build_conv_matrix(a)

    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS is a good default for dense LPs of this size
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        # In the unlikely event ECOS fails, fall back to a uniform vector
        return np.full(n, 1.0 / n)

    if b.value is None:
        # Solver failed – return uniform vector as a safe fallback
        return np.full(n, 1.0 / n)

    b_opt = np.asarray(b.value, dtype=np.float64)
    # Numerical noise may produce tiny negatives; clip to zero
    np.maximum(b_opt, 0.0, out=b_opt)
    return b_opt


# ----------------------------------------------------------------------
# Mixing step with adaptive step size τ
# ----------------------------------------------------------------------


def mix_update(a: np.ndarray, b: np.ndarray, tau: float) -> np.ndarray:
    """
    Return a new point a' = (1‑τ)·a + τ·(b/Σb), re‑normalised to sum = 1.
    """
    sum_b = b.sum()
    if sum_b <= 0.0:
        return a  # nothing to do
    g = b / sum_b
    a_new = (1.0 - tau) * a + tau * g
    # Re‑normalise (important because of rounding errors)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.full_like(a_new, 1.0 / a_new.size)
    return a_new


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max) – final polishing
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a))/sum(a)². """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # Identify all indices attaining the maximum (up to numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑convolution: 2·(a ★ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    Sub‑grad it={it:,d}  surrogate≈{fast_score(a):.6f}")

    return a
