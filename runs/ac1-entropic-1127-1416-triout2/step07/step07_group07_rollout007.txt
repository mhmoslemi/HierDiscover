"""
Propose a high‑quality non‑negative step‑function minimizing

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines three complementary phases:

1. **Smart initialization** – a discrete arcsine‑like density (known to give
   a low convolution peak) projected to the probability simplex.

2. **Iterative linear‑program improvement** – for moderate lengths (≤4096)
   we solve a small LP
        max   Σ g_j
        s.t.  (a * g)_k ≤ M   (k = 0 … 2n‑2)
              g ≥ 0 ,
   where M = max(conv(a,a)).  The optimal `g` increases the total mass
   without raising the current max‑convolution bound.  After normalising `g`
   we mix it with the current vector and repeat a few times.

3. **Fine‑tuning** – Adam‑accelerated soft‑max gradient descent on a smooth
   surrogate of the max‑operator, followed by a simple hill‑climber and a
   symmetry‑averaging step.

All operations respect the simplex (non‑negative, sum = 1), so the fast
`compute_score` (which assumes ∑a = 1) equals the true `evaluate_sequence`
value.  The routine respects the given time budget (≈ 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT convolution, simplex projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, accurate)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` when sum(a) == 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        # any point on the simplex works; pick uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine–type density:
        a_i ∝ 1 / sqrt(i·(n‑1‑i)).
    This distribution is symmetric and already yields a relatively flat
    self‑convolution.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_sparse_vector(n: int) -> np.ndarray:
    """Random probability vector with a modest number of non‑zero entries."""
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Gradient descent on a smooth approximation of max(conv)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of f(a) = eps·log Σ_k exp(conv_k/eps) where conv = a⋆a.
    Returns  grad = 2·(w_rev ⊛ a)  with  w = softmax(conv/eps).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)
    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)       # Σ w = 1
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 10_000,
    step0: float = 0.5,
    eps_factor: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth soft‑max surrogate.
    The temperature (ε) is reduced exponentially.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature never smaller than 1 % of the current max
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        # Adam update (biased estimates)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)               # simple decay
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # Occasional tiny jitter to shake‑up stagnation
        if t % 500 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple hill‑climber (strictly improving moves)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber that only accepts strictly improving
    swaps or mass‑transfer moves.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # adaptive transfer fraction
    step_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # safeguard against negative rounding errors
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # become bolder after success
        else:
            step_frac *= 0.96                       # become more cautious

        max_steps -= 1

    return best_a, best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Enforce (approx.) symmetry: (a + reverse(a))/2 and re‑project."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Linear‑program helper (moderate lengths only)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix C satisfying  (f * g)[k] = Σ_j C[k,j]·g[j].
    Shape: (2·n‑1, n) .
    """
    n = f.shape[0]
    m = 2 * n - 1
    C = np.zeros((m, n), dtype=np.float64)

    # Fill each row by the appropriate slice of f
    for k in range(m):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # indices j_start … j_end correspond to f[k‑j]
        C[k, j_start:j_end + 1] = f[k - np.arange(j_start, j_end + 1)]
    return C


def lp_expand_mass(f: np.ndarray, M: float, time_deadline: float) -> np.ndarray:
    """
    Solve
        max Σ g_j
        s.t.  (f * g)[k] ≤ M  (all k)
              g ≥ 0
    Returns the optimal g (may be dense).  If the LP fails, returns the
    original vector unchanged.
    """
    n = f.shape[0]
    C = build_convolution_matrix(f)                # (2n‑1) × n
    b = np.full(C.shape[0], M, dtype=np.float64)

    # linprog solves a * x ≤ b  with min c·x ; we maximise Σ g ⇔ minimise −Σ g
    c = -np.ones(n, dtype=np.float64)

    # Pass a very generous time limit to the LP solver – it will abort
    # automatically when the global deadline is reached.
    remaining = max(0.0, time_deadline - time.time())
    options = {"presolve": True, "time_limit": remaining}

    res = linprog(
        c,
        A_ub=C,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options=options,
    )

    if not res.success or res.x is None:
        print("   [LP] solver failed or timed out – skipping this step")
        return f.copy()
    return res.x

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation metric.  The routine respects the supplied time budget
    (≈ 1000 s) and returns a plain Python list ready for the evaluation
    harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try a few different lengths (larger n affords a lower theoretical bound)
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]   # descending order – stop when out of time
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Searching with length n={n} ===")
        # ---- 0. initialise -------------------------------------------------
        if np.random.rand() < 0.6:
            a = arcsine_initialisation(n)
        else:
            a = random_sparse_vector(n)

        cur_sc = compute_score(a)
        print(f"   init score = {cur_sc:.6f}")
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a.tolist()

        # ---- 1. (optional) LP‑mass‑expansion for moderate n ---------------
        # We only attempt the LP when the matrix size is reasonable.
        if n <= 4096 and time.time() < deadline - 5.0:
            # Run a few LP iterations (typically 3‑5) – each iteration expands the
            # total mass while respecting the current max‑convolution bound.
            max_lp_iters = 5
            for it in range(max_lp_iters):
                if time.time() >= deadline - 4.0:
                    break
                M = np.max(convolution_fft(a, a))
                print(f"   [LP] iteration {it + 1}, current M = {M:.6f}")
                g = lp_expand_mass(a, M, deadline)
                sum_g = np.sum(g)
                if sum_g <= 1e-12:
                    break
                # Normalise the new vector and blend it with the old one.
                g_norm = g / sum_g
                alpha = 0.4 if it == 0 else 0.3   # more aggressive at first
                a = project_to_simplex((1 - alpha) * a + alpha * g_norm)

                new_sc = compute_score(a)
                print(f"   [LP] after mixing score = {new_sc:.6f}")
                if new_sc < best_score:
                    best_score = new_sc
                    best_seq = a.tolist()
                # Stop early if improvement stalls
                if np.abs(new_sc - cur_sc) < 1e-8:
                    break
                cur_sc = new_sc

        # ---- 2. Adam‑accelerated soft‑max gradient descent -----------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            # Roughly target ~30 gradient evaluations per second
            max_gd_iters = int(min(30_000, remaining * 30))
            a, sc = adam_softmax_descent(
                a,
                time_deadline=deadline - 2.0,
                max_iter=max_gd_iters,
                step0=0.5,
                eps_factor=0.2,
            )
            exact_sc = compute_score(a)
            print(f"   after Adam GD best score = {exact_sc:.6f}")
            if exact_sc < best_score:
                best_score = exact_sc
                best_seq = a.tolist()

        # ---- 3. Hill‑climbing polishing ------------------------------------
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_hc_steps = int(min(200_000, remaining * 150))   # ~150 steps/s
            a_hc, sc_hc = hill_climb(
                np.array(best_seq, dtype=float),
                time_deadline=deadline - 1.0,
                max_steps=max_hc_steps,
            )
            final_sc = compute_score(a_hc)
            print(f"   after hill‑climb score = {final_sc:.6f}")
            if final_sc < best_score:
                best_score = final_sc
                best_seq = a_hc.tolist()

        # ---- 4. Symmetrisation (cheap but often beneficial) ---------------
        if best_seq is not None:
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"   after symmetry averaging score = {sym_sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (guarantees a finite score)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
