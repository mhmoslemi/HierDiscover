import cvxpy as cp

def lp_step(a, M_current):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    lam = cp.Variable()
    constraints = [cp.conv(a, b) <= lam,
                   lam >= 0,
                   cp.sum(b) == 1]
    obj = cp.Minimize(lam)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    b_val = np.array(b.value).astype(float)
    lam_val = lam.value
    return b_val, lam_val

"""
search‑candidate – a fast multi‑stage optimizer for the L∞ self‑convolution
objective.

The algorithm works in four phases:
1.  Random parametric “arcsine‑like” families (a_i ∝ ((i+α)(n‑1‑i+α))⁻ᵖ)
    are sampled cheaply and projected onto the probability simplex.
    The best vector is kept.
2.  A few iterations of Adam on the smooth soft‑max surrogate
    (ε·log Σ exp(conv/ε)) are applied to the current best vector.
3.  Exact sub‑gradient descent on the true objective
    (2·n·max(a⋆a)) refines the shape.
4.  A simple hill‑climber (swap / mass‑transfer moves) finishes the search,
    followed by a symmetry averaging step.

All steps respect the time budget (≈ 1000 s).  The final sequence is returned
as a plain Python list, ready for the external `evaluate_sequence` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT–based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact replica of `evaluate_sequence` when sum(a)=1.
    Returns 2·n·max(conv) where conv = a⋆a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        # uniform distribution is the projection of a non‑positive vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_sparse_vector(n: int) -> np.ndarray:
    """Random probability vector with a modest number of non‑zero entries."""
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def generate_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Parametric arcsine‑like shape a_i ∝ ((i+α)(n‑1‑i+α))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    # avoid exact zero at the edges when α=0 and p>0
    denom = (i + alpha) * (n - 1 - i + alpha)
    # raise to -p, taking care of possible overflow
    with np.errstate(divide='ignore', over='ignore'):
        w = np.power(denom, -p)
    w[np.isinf(w)] = 0.0
    if np.all(w == 0):
        w = np.ones_like(w)
    return project_to_simplex(w)


# ----------------------------------------------------------------------
# Adam‑accelerated soft‑max gradient descent (smooth surrogate)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of eps·log Σ exp(conv/eps)."""
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)  # soft‑max weights, Σ w = 1

    # cross‑correlation w_rev ⊛ a  (same as a * w_rev)
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)

    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 4000,
    step0: float = 0.2,
    eps_factor: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth surrogate."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature ε (never too small)
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 500 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.002 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on 2·n·max(a⋆a)."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i} inside the support
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny random shake
        if t % 1500 == 0:
            a = project_to_simplex(a + np.random.randn(n) * (step0 * 0.004 / np.sqrt(t)))

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climber (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Hill‑climber that only accepts strict improvements."""
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_frac = 0.08  # initial transfer proportion

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # be bolder
        else:
            step_frac *= 0.96                       # be more cautious

        max_steps -= 1

    return best_a, best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Enforce (approx.) symmetry: (a + reverse(a))/2 and re‑project."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

    if best_seq is not None and time.time() < deadline - 10.0:
        a0 = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        # allocate a modest number of Adam steps (fast but effective)
        max_iters = int(min(8000, remaining * 25))   # ~25 grad evaluations per sec
        a_adam, sc_adam = adam_softmax_descent(
            a0,
            time_deadline=deadline - 5.0,
            max_iter=max_iters,
            step0=0.25,
            eps_factor=0.22,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[Adam] improved score = {sc_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient descent for fine tuning
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        a_start = np.array(best_seq, dtype=float)
        a_sub, sc_sub = subgradient_descent(
            a_start,
            time_deadline=deadline - 3.0,
            max_iter=70000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[sub‑grad] improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Hill‑climb polishing + symmetry averaging
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_cur = np.array(best_seq, dtype=float)
        polished, pol_sc = hill_climb(
            a_cur,
            time_deadline=deadline - 1.0,
            max_steps=150_000,
        )
        if pol_sc < best_score:
            best_score = pol_sc
            best_seq = polished.tolist()
            print(f"[hill‑climb] improved score = {pol_sc:.6f}")

        # tiny symmetry step (often gives a small boost)
        a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
        sym_sc = compute_score(a_sym)
        if sym_sc < best_score:
            best_score = sym_sc
            best_seq = a_sym.tolist()
            print(f"[symmetry] improved score = {sym_sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (guarantees a finite result)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
