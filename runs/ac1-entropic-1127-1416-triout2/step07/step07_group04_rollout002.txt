"""
Propose a low‑dimensional step sequence with a low evaluate_sequence score.

The algorithm is a multi‑scale, stochastic optimisation pipeline:

1.  Start from a small grid (n=1024) with a randomly chosen “nice” shape
    (arcsine, beta‑type, power‑law or a mixture).  Run a smooth‑max Adam
    optimiser (softmax on the logits) to reduce the surrogate objective
    2·n·max(conv(a,a)).  The temperature of the smooth‑max is increased
    from τ≈0.5 to τ≈300 to obtain a tighter approximation of the true max.

2.  Upsample the resulting vector to a larger resolution (first 4096,
    then – if time permits – 8192).  At each resolution the Adam phase is
    repeated, followed by a long Euclidean projected sub‑gradient descent
    on the exact max‑convolution and an exponentiated‑gradient (mirror‑descent)
    polish.  All loops monitor the global deadline (1000 s) and stop early
    when needed.

3.  Keep the best candidate seen (including a possible fallback stored in
    the global variable ``height_sequence_1``).  After the main optimisation
    a cheap local random‑move hill‑climbing phase (using the fast surrogate)
    explores the neighbourhood of the best sequence; the final official
    score is evaluated and, if improved, the best sequence is updated.

The whole routine works with NumPy only, uses only the provided
``evaluate_sequence`` for the final check, and respects the 1000 s wall‑clock
budget.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform, normalised."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points.
    The result is re‑normalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of a smooth approximation of max(conv).
    Variables are the logits w → a = softmax(w) guaranteeing a ≥ 0,
    Σa = 1.  The temperature τ linearly interpolates from τ_start to τ_end.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)                 # stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)            # length L

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient wrt a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wrt logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (nonsmooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.06) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Helper to obtain a random initial shape
# ----------------------------------------------------------------------


def random_initial_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random but well‑behaved non‑negative shape normalised to sum = 1."""
    choice = rng.integers(0, 5)
    if choice == 0:
        return arcsine_initial(n)
    if choice == 1:
        alpha = rng.uniform(0.3, 4.0)
        beta = rng.uniform(0.3, 4.0)
        return generate_beta_shape(n, alpha, beta, eps=1e-6)
    if choice == 2:
        p = rng.uniform(0.3, 2.0)
        return generate_power_shape(n, p, eps=1e-6)
    if choice == 3:
        w = rng.uniform(0.0, 1.0)
        return generate_mixture_shape(n, w)
    # last option: perturb arcsine with small Gaussian noise
    base = arcsine_initial(n)
    noise = rng.normal(scale=0.02, size=n)
    pert = base + noise
    pert = np.clip(pert, 0, None)
    if pert.sum() == 0:
        pert = np.full(n, 1.0 / n)
    else:
        pert /= pert.sum()
    return pert

    surrogate_coarse = compute_score(a_coarse)

    # Phase‑2: upsample to an intermediate resolution (4096) and refine
    # Decide whether we have enough time for a larger grid (8192)
