```python
from scipy.optimize import linprog
import numpy as np
import time

def build_conv_matrix(a):
    n = len(a)
    rows = np.zeros((2*n-1, n), dtype=float)
    for k in range(2*n-1):
        # each row: a[i] where i = k - j
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # j = k - i
        # i range inclusive
        # assign rows[k, k-i] = a[i]
        # So for i in [i_start,i_end], set rows[k, k-i] = a[i]
        rows[k, k - np.arange(i_start, i_end+1)] = a[i_start:i_end+1]
    return rows
```

"""
Search for a non‑negative step‑function minimizing
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in three stages:

1. **LP‑driven refinement** –  repeatedly solve a linear program
   maximizing the total mass of a new candidate `b` under the
   constraints (a * b)[k] ≤ M (where M is the current worst
   autocorrelation).  The optimal `b` is symmetrised,
   normalised to unit sum and mixed with the current `a`.
   This is the core “Frank‑Wolfe‑like’’ step that steadily reduces
   the peak of the convolution.

2. **Exact sub‑gradient descent** –  after the LP stage the best
   vector found is polished with a classic sub‑gradient method on the
   true objective (max‑convolution).  The step size decays as 1/√t
   and the iterate is projected onto the probability simplex after
   each update.

3. **Stochastic hill‑climbing** –  a final simulated‑annealing phase
   performs random mass transfers and swaps; improvements are always
   kept and occasional uphill moves are accepted with a linearly
   cooling schedule.  The result is symmetrised for a last
   push.

All phases respect the overall time budget (~1000 s).  The code
uses only NumPy and SciPy (highs solver) and prints progress
information.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for many calls)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns the exact value of ``evaluate_sequence`` (since scaling cancels).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    v_i ≥ 0 and Σ v_i = 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet via exponentials)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Linear‑programming helper
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that A @ b = a ⊛ b.
    Each column i of A contains ``a`` shifted down by i rows.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve    max Σ b_i   s.t.   a ⊛ b ≤ M  (component‑wise),  b ≥ 0.
    Returns the optimal b (as a NumPy array) or ``None`` if the solver fails.
    """
    n = a.shape[0]
    A = build_conv_matrix(a)                # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises; we minimise -Σb_i to maximise Σb_i.
    res = linprog(
        c=-np.ones(n, dtype=np.float64),
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "verbose": 0},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on   f(a) = 2·n·max(a⊛a) .
    Returns the best vector seen and its score.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2 * a_{k-i} for indices that fall into the vector
        g = np.zeros_like(a)
        if k < n:
            # overlap on the left side
            g[: k + 1] = 2.0 * a[k::-1]
        else:
            # overlap on the right side
            start = k - n + 1
            g[start:] = 2.0 * a[: n - start][::-1]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape flat regions
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber with a simple linear cooling schedule.
    Moves: swap two entries or transfer a small fraction of mass.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]      # transfer 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0.0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional diversification
        if step % 5000 == 0:
            rnd = random_dense_vector(n) * 0.02
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n‑1‑i} and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

                print(f"[outer {outer}] symmetry averaging improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
