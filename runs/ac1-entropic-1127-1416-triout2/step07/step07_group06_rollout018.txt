"""
Multi‑phase optimiser for the step‑function evaluation problem.

Goal
----
Find a non‑negative sequence `a` (length up to a few thousand) that minimises

    evaluate_sequence(a) = 2·n·max(a⊛a) / (sum(a)²)

The optimiser works inside the 1000 s time budget, keeps the best sequence
found and returns it as a plain Python list.

Approach
--------
1. **Parametric seeding** – generate a few families of deterministic shapes
   (Beta‑U‑shaped, polynomial, sinusoidal) on several lengths.  The best
   shape is used as a warm‑start for the subsequent numerical optimisation.

2. **Flattening (L2‑variance) phase** – Adam on the variance of the
   linear convolution `a⊛a`.  This drives the autocorrelation towards a flat
   profile.

3. **Smooth‑max surrogate phase** – minimise an ε·log‑sum‑exp
   approximation of the true max using Adam with an exponentially decaying
   ε.  This gives a smooth gradient while still targeting the peak.

4. **Exact sub‑gradient descent** – classical sub‑gradient steps on the
   true max‑objective for final fine‑tuning.

5. **Simulated‑annealing hill‑climb** – stochastic mass‑transfer moves
   (swap, fraction transfer, local re‑distribution) with a linear cooling
   schedule.  Occasional large perturbations keep the walk exploratory.

6. **Symmetrisation** – a cheap averaging with the reversed vector often
   reduces the peak further.

All updates are projected onto the probability simplex (sum = 1,
non‑negative) which guarantees a valid input for `evaluate_sequence`.

The code respects the time budget, prints progress and finally returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (real input → real output)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a⊛b using real FFT."""
    m = a.size + b.size - 1
    n = _next_pow2(m)
    A = np.fft.rfft(a, n=n)
    B = np.fft.rfft(b, n=n)
    C = A * B
    conv = np.fft.irfft(C, n=n)[:m]
    return conv

# Simple utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Fast replica of evaluate_sequence for vectors with sum(a)=1."""
    # we know sum(a)=1 → denominator disappears
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, ≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (sum = 1)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Parametric initial families (Beta‑U‑shape, polynomial, sinusoid)
# ----------------------------------------------------------------------


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """U‑shaped (α,β<1) or bell‑shaped (α,β>1) Beta density on a grid."""
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n
    # avoid 0⁰ or 1⁰ when α or β < 1
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    shape = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    shape = shape / np.sum(shape)
    return shape


def _poly_shape(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ (i·(n‑1‑i))^p."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    dist = i * (n - 1 - i)
    shape = np.power(dist + 1e-12, p)  # +1e-12 to avoid 0⁰
    shape = shape / np.sum(shape)
    return shape


def _sinusoid_shape(n: int, power: float) -> np.ndarray:
    """Shape a_i ∝ sin(π·(i+0.5)/n)^power (peaked in the middle)."""
    i = np.arange(n, dtype=float) + 0.5
    vals = np.sin(np.pi * i / n) ** power
    shape = vals / np.sum(vals)
    return shape


def _generate_parametric_candidates(n: int, budget: float, deadline: float):
    """Yield a handful of promising deterministic vectors for length n."""
    # a small grid of parameters (chosen empirically)
    candidates = []

    # Beta families – edge‑heavy for α,β<1, centre‑heavy otherwise
    for a in (0.3, 0.5, 0.8, 1.2, 2.0):
        for b in (0.3, 0.5, 0.8, 1.2, 2.0):
            if time.time() >= deadline:
                return candidates
            vec = _beta_shape(n, a, b)
            candidates.append(vec)

    # Polynomial (U‑shape when p<0)
    for p in (-0.8, -0.5, -0.2, 0.0, 0.5, 1.0):
        if time.time() >= deadline:
            return candidates
        vec = _poly_shape(n, p)
        candidates.append(vec)

    # Sinusoid powers (peaked centre for large power)
    for pw in (0.5, 1.0, 2.0, 4.0):
        if time.time() >= deadline:
            return candidates
        vec = _sinusoid_shape(n, pw)
        candidates.append(vec)

    # a few random Dirichlet draws as a safety net
    for _ in range(5):
        if time.time() >= deadline:
            break
        candidates.append(_random_dirichlet(n))

    return candidates


# ----------------------------------------------------------------------
# Optimisation phases (flattening, smooth‑max, sub‑gradient, anneal)
# ----------------------------------------------------------------------


def _flatten_phase(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam on the variance of the convolution."""
    a = a0.copy()
    n = a.size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a, best_sc = a.copy(), _score(a)
    idx_start = 2 * n - 2

    step0 = 0.12
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        mu = np.mean(conv)
        diff = conv - mu
        # gradient = 4 * Σ_j a_j * diff_{i+j}
        grad = 4.0 * _conv_fft(a, diff[::-1])[idx_start - np.arange(n)]

        # Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 / (np.sqrt(t) + 1.0)

        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)

        sc = _score(a_new)
        if sc < best_sc:
            best_sc, best_a = sc, a_new.copy()

        a = a_new

    return best_a, best_sc


def _smooth_max_phase(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam on ε·log∑exp(conv/ε).  ε decays exponentially."""
    a = a0.copy()
    n = a.size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a, best_sc = a.copy(), _score(a)
    idx_start = 2 * n - 2
    step0 = 0.45
    beta1, beta2 = 0.9, 0.999
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        maxc = np.max(conv)
        # start with a relatively large ε and shrink it
        eps = maxc * max(0.1 * (0.995 ** (t - 1)), 1e-3)

        # soft‑max weights (∑w = 1)
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)

        grad = 2.0 * _conv_fft(a, w[::-1])[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = _project_simplex(a_new)

        sc = _score(a_new)
        if sc < best_sc:
            best_sc, best_a = sc, a_new.copy()

        a = a_new

        # occasional jitter to escape flat plateaux
        if t % 400 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = _project_simplex(a + jitter)

    return best_a, best_sc


def _subgrad_phase(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on 2·n·max(a⊛a)."""
    a = a0.copy()
    n = a.size
    best_a, best_sc = a.copy(), _score(a)

    step0 = 0.07
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2·a_{k‑i}   (with appropriate bounds)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = _project_simplex(a_new)

        sc = _score(a_new)
        if sc < best_sc:
            best_sc, best_a = sc, a_new.copy()

        a = a_new

        # tiny random shake every 1000 steps
        if t % 1000 == 0:
            a = _project_simplex(a + np.random.randn(n) *
                                 (step0 * 0.005 / np.sqrt(t)))

    return best_a, best_sc


def _anneal_phase(seq, deadline: float) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climber with a linear cooling schedule."""
    a = np.array(seq, dtype=float)
    n = a.size
    best_a = a.copy()
    best_sc = _score(best_a)

    T0 = 0.12
    step = 1
    max_steps = 200_000

    while step <= max_steps and time.time() < deadline:
        step += 1
        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]              # swap
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]                     # move 5% of source
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = _project_simplex(cand)
        cur_sc = _score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc, best_a = cur_sc, cand.copy()
            else:
                # accept anyway (required for SA) but keep best unchanged
                pass

        # aggressive diversification every 5 k steps
        if step % 5000 == 0:
            perturb = 0.02 * _random_dirichlet(n)  # 2 % fresh noise
            cand = best_a + perturb
            cand = _project_simplex(cand)
            cur_sc = _score(cand)
            if cur_sc < best_sc:
                best_sc, best_a = cur_sc, cand.copy()

    return best_a, best_sc


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Average with its reverse and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return _project_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a tiny safety margin for the final evaluation call
    deadline = start + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback – previously known good construction (if available)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score, best_seq = fb_score, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Length candidates (increase slowly, longer sequences cost more time)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]        # keep FFTs cheap enough

    outer_it = 0
    while time.time() < deadline - 5.0:
        outer_it += 1
        # ------------------------------------------------------------------
        # pick a length (biased towards longer ones as time runs out)
        # ------------------------------------------------------------------
        if time.time() > deadline * 0.7:
            n = lengths[-1]                     # final passes use the longest
        else:
            n = int(np.random.choice(lengths))

        # ------------------------------------------------------------------
        # 1) generate a few deterministic seeds (Beta, polynomial, sinusoid)
        # ------------------------------------------------------------------
        param_cands = _generate_parametric_candidates(
            n, budget=budget_s, deadline=deadline
        )
        for vec in param_cands:
            cur_sc = _score(vec)
            if cur_sc < best_score:
                best_score, best_seq = cur_sc, vec.tolist()
                print(f"[{outer_it}] param‑seed n={n} score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 2) pick a random start (or a truncated version of current best)
        # ------------------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.2:
            a0 = np.array(best_seq[:n], dtype=float)
            a0 = _project_simplex(a0)
        else:
            a0 = _random_dirichlet(n)

        # quick filter – keep if already beats incumbent
        cur_sc = _score(a0)
        if cur_sc < best_score:
            best_score, best_seq = cur_sc, a0.tolist()
            print(f"[{outer_it}] random‑init n={n} score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 3) flattening phase (variance of convolution)
        # ------------------------------------------------------------------
        rem = deadline - time.time()
        if rem < 4.0:
            break
        a_flat, sc_flat = _flatten_phase(a0, deadline - 2.0)
        if sc_flat < best_score:
            best_score, best_seq = sc_flat, a_flat.tolist()
            print(f"[{outer_it}] flatten n={n} score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 4) smooth‑max surrogate phase
        # ------------------------------------------------------------------
        rem = deadline - time.time()
        if rem < 6.0:
            break
        a_smax, sc_smax = _smooth_max_phase(a_flat, deadline - 3.0)
        if sc_smax < best_score:
            best_score, best_seq = sc_smax, a_smax.tolist()
            print(f"[{outer_it}] smooth‑max n={n} score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 5) exact sub‑gradient descent
        # ------------------------------------------------------------------
        if deadline - time.time() > 10.0:
            a_sub, sc_sub = _subgrad_phase(a_smax, deadline - 5.0)
            if sc_sub < best_score:
                best_score, best_seq = sc_sub, a_sub.tolist()
                print(f"[{outer_it}] sub‑grad n={n} score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 6) simulated‑annealing hill‑climb
        # ------------------------------------------------------------------
        if deadline - time.time() > 8.0:
            seq_start = best_seq if best_seq is not None else a_sub.tolist()
            polished, polished_sc = _anneal_phase(seq_start, deadline - 2.0)
            if polished_sc < best_score:
                best_score, best_seq = polished_sc, polished.tolist()
                print(f"[{outer_it}] anneal n={n} score={best_score:.6f}")

            # final cheap symmetrisation
            a_sym = _symmetrize(np.array(best_seq, dtype=float))
            sym_sc = _score(a_sym)
            if sym_sc < best_score:
                best_score, best_seq = sym_sc, a_sym.tolist()
                print(f"[{outer_it}] symmetrised n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing has been found, return a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback‑uniform] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
