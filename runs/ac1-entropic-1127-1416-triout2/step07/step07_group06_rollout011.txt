"""
Propose a step‑function (non‑negative, sum‑to‑one) that minimises
   evaluate_sequence(sequence) = 2·n·max(conv) .

Algorithm (≈100 s per outer run):
1. Random Dirichlet start.
2. Linear‑programming “mass‑pumping” step:
   maximise Σ g_j   subject to   f⋆g ≤ M   (M = max(f⋆f)),  g≥0.
   The solution is normalised and, if it improves the peak,
   replaces the current vector.
   A small‐mixing step is tried if the direct replacement does not improve.
   The LP is solved with SciPy’s high‑performance “highs” method.
3. Smooth max‑surrogate minimisation (ε·log‑sum‑exp) with Adam.
4. Exact sub‑gradient descent on the true ∞‑norm using the
   average sub‑gradient of all active (max) constraints.
5. Simple simulated‑annealing hill‑climber for a final polish.
The best sequence found over several restarts (different lengths) is
returned.  All updates keep the vector on the probability simplex,
so the final call to `evaluate_sequence` is safe.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random dense start
# ----------------------------------------------------------------------


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Phase 2 – smooth max surrogate minimisation (Adam)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.5,
    eps_factor: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of the max‑convolution:
        f(a) = ε·log Σ exp(conv/ε)
    Using Adam and projection onto the simplex.
    Returns the best vector (exact score) seen.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε decays but never smaller than 1 % of maxc
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 1e-12)

        # soft‑max weights
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)  # Σ w = 1

        # gradient of the smooth max (see derivation in the original code)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)          # length 3n‑2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – exact sub‑gradient descent (averaged over all active maxes)
# ----------------------------------------------------------------------


def subgrad_optimize_aggregated(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective 2·n·max(a⊛a).
    Uses the averaged sub‑gradient of *all* positions attaining the maximum.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        # active set (all indices within a tiny tolerance of the max)
        active = np.where(conv >= maxc - 1e-12)[0]

        # build a mask that averages over the active shifts
        mask = np.zeros_like(conv)
        if active.size > 0:
            mask[active] = 1.0 / active.size

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)          # length 3n‑2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # diminishing step size
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        # occasional random jitter to escape plateaus
        if t % 5000 == 0:
            a_new += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a_new = project_to_simplex(a_new)

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – LP based “mass‑pump” improvement
# ----------------------------------------------------------------------


def lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme:
        max Σ g_j
        s.t. f ⊛ g ≤ M   (element‑wise)
             g ≥ 0
    Returns the optimiser g (may be None on failure).
    """
    n = len(f)
    # Build the (2n‑1)×n Toeplitz matrix that implements the convolution f⊛g
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.zeros(n)
    row[0] = f[0]
    A = scipy.linalg.toeplitz(col, row)          # shape (2n‑1, n)

    # Objective is –Σg (linprog minimises)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    res = scipy.optimize.linprog(
        c=c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x


def lp_improve(f0: np.ndarray, time_deadline: float, max_iter: int = 15) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP step to drive down max(f⊛f).
    Stops when no improvement is observed or time runs out.
    Returns the best vector and its exact score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_sc = compute_score(best_f)

    for it in range(max_iter):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(f, f)
        M = np.max(conv)

        g = lp_step(f, M)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g <= 1e-12:
            break
        g_norm = g / sum_g                     # renormalise to Σg = 1

        sc_g = compute_score(g_norm)
        if sc_g < best_sc:
            best_f = g_norm
            best_sc = sc_g
            f = best_f
            print(f"LP iter {it}: direct improvement → {best_sc:.6f}")
            continue

        # try convex combinations if the direct replacement fails
        improved = False
        for t in (0.2, 0.4, 0.6, 0.8):
            mix = (1 - t) * f + t * g_norm
            mix = project_to_simplex(mix)
            sc_mix = compute_score(mix)
            if sc_mix < best_sc:
                best_f = mix
                best_sc = sc_mix
                f = best_f
                improved = True
                print(f"LP iter {it}: mixing t={t:.1f} → {best_sc:.6f}")
                break
        if not improved:
            # no further improvement possible
            break

    return best_f, best_sc


# ----------------------------------------------------------------------
# Phase 5 – simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 150_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with a linear cooling schedule.
    Moves: swap two entries or transfer a small fraction of mass.
    Accepts worsening moves with probability exp(-Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12  # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]  # move 5 % of mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional random perturbation to diversify exploration
        if step % 5000 == 0:
            rnd = random_dense_vector(n) * 0.02
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a
    plain Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # Try a few different lengths – larger n gives more freedom but is slower.
    lengths = [1024, 2048]  # both fit comfortably into the time budget
    for n in lengths:
        # a few independent restarts for each length
        for restart in range(3):
            if time.time() >= deadline - 5.0:
                break

            # ---------- 0. random start ----------
            a0 = random_dense_vector(n)

            # ---------- 1. LP‑based improvement ----------
            a_lp, sc_lp = lp_improve(
                a0,
                time_deadline=deadline - 2.0,
                max_iter=12,
            )
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.tolist()
                print(f"[n={n} r={restart}] LP step best {best_score:.6f}")

            # ---------- 2. smooth‑max surrogate (Adam) ----------
            if time.time() < deadline - 4.0:
                a_smooth, sc_smooth = smooth_max_adam(
                    a_lp,
                    time_deadline=deadline - 2.0,
                    max_iter=25000,
                    step0=0.6,
                    eps_factor=0.2,
                )
                if sc_smooth < best_score:
                    best_score = sc_smooth
                    best_seq = a_smooth.tolist()
                    print(f"[n={n} r={restart}] smooth‑max best {best_score:.6f}")

            # ---------- 3. exact sub‑gradient descent ----------
            if time.time() < deadline - 3.0:
                a_sub, sc_sub = subgrad_optimize_aggregated(
                    a_smooth,
                    time_deadline=deadline - 2.0,
                    max_iter=80000,
                    step0=0.12,
                )
                if sc_sub < best_score:
                    best_score = sc_sub
                    best_seq = a_sub.tolist()
                    print(f"[n={n} r={restart}] sub‑grad best {best_score:.6f}")

            # ---------- 4. simulated‑annealing hill‑climb ----------
            if time.time() < deadline - 2.0:
                polished_seq, polished_sc = hill_climb_anneal(
                    best_seq if best_seq is not None else a_sub.tolist(),
                    time_deadline=deadline - 1.0,
                    max_steps=150000,
                )
                if polished_sc < best_score:
                    best_score = polished_sc
                    best_seq = polished_seq
                    print(f"[n={n} r={restart}] anneal improved {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case we found nothing
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n

    # Final official evaluation (uses the provided evaluate_sequence)
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
