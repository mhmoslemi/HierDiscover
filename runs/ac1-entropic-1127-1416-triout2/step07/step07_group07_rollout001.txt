"""
Propose a step‑function that minimises
    score = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

* Initialise a probability vector (sum = 1) with a discrete arcsine
  density – this already yields a fairly flat self‑convolution.

* Iteratively improve the vector by solving a linear programme.
  For the current vector f we know its worst self‑convolution value
  M = max(f⋆f).  The LP
        maximise   Σ_j b_j
        subject to A_f b ≤ M,   b ≥ 0
  where (A_f)_{k,j}=f_{k‑j} for 0≤k≤2n‑2 (zero padding).  The
  constraints guarantee that the cross‑convolution f⋆b does not exceed
  the current worst value M.  After solving we rescale b to have sum = 1
  and mix it with f:
        f←(1‑t)·f + t·b̂          (t slowly decays)

  Mixing preserves the simplex constraint and strictly reduces the
  maximal self‑convolution when possible.

* After a few LP‑mixing steps we polish the result with a cheap
  hill‑climber that transfers a small amount of mass between two
  coordinates only when the score improves.

* The best vector seen during the whole search is returned (as a plain
  Python list) and finally evaluated with the provided `evaluate_sequence`
  function.

The whole routine respects a time budget (≈ 1000 s) and never returns
an empty or invalid list.  It works for lengths up to a few thousand
(e.g. 1024 – 2048) which keep the LP size manageable while giving enough
degrees of freedom to improve the bound below the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a)=1.
    Returns the part 2·n·max(conv) (the denominator is 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Numerical implementation of the algorithm from
    "Efficient Projections onto the l1‑Ball for Learning in High Dimensions".
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine‑like density: a_i ∝ 1/√(i·(n‑1‑i)).
    The distribution is symmetric and already gives a relatively flat
    self‑convolution.  The result is projected onto the simplex.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_sparse_vector(n: int) -> np.ndarray:
    """Generate a random probability vector with a modest number of non‑zero entries."""
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Linear‑programming matrix construction
# ----------------------------------------------------------------------


def build_toeplitz_matrix(f: np.ndarray) -> coo_matrix:
    """
    Build the (2·n‑1) × n matrix A_f with entries
        (A_f)_{k,j} = f_{k‑j}   if 0 ≤ k‑j < n, else 0.
    This matrix satisfies A_f b = f ⋆ b (full convolution).
    """
    n = f.shape[0]
    nn = n * n
    rows = np.empty(nn, dtype=int)
    cols = np.empty(nn, dtype=int)
    data = np.empty(nn, dtype=float)

    pos = 0
    for i in range(n):
        fi = f[i]
        rows[pos:pos + n] = np.arange(i, i + n)
        cols[pos:pos + n] = np.arange(n)
        data[pos:pos + n] = fi
        pos += n

    return coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


# ----------------------------------------------------------------------
# Simple hill‑climber (used for final polishing)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: perform only improving moves.
    Two move types are used – swapping two entries and transferring a
    fraction of mass from one entry to another.  The step size adapts
    during the run.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two coordinates
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0:
                delta = step_frac * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # become bolder
        else:
            step_frac *= 0.96                       # become more cautious

        max_steps -= 1

    return best_a, best_sc
