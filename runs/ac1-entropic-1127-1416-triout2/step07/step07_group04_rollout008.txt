"""
Improved step‑function search
==============================

This implementation adds several ideas on top of the previous optimizer:

* **Multiple seeds** – a handful of analytically motivated shapes
  (arcsine, beta‑type, power‑law, mixture) and a random perturbation are
  tried.  The best‑performing seed is kept.

* **Dynamic resolution** – if enough time is available we work with
  8192 points (instead of 4096).  A coarser 1024‑point optimisation is
  performed first and then the result is linearly up‑sampled.

* **More aggressive smooth‑max Adam** – the temperature τ is increased
  up to 300, the learning‑rate is a bit larger and the iteration budget
  is scaled with the remaining wall‑clock time.

* **Simplified simplex projection** – in the sub‑gradient phase we
  replace the costly O(n log n) Euclidean projection by a cheap
  “zero‑out‑and‑renormalise’’ step; the algorithm stays on the
  probability simplex and is much faster for long vectors.

* **Perturb‑and‑polish** – after a seed finishes we perturb the current
  best sequence with small Gaussian noise and give it a short polish,
  which often escapes shallow local minima.

All helper functions are defined at top level (no closures).  The main
entry point `propose_candidate` respects the 1000 s budget and always
returns a non‑empty sequence.  The returned list can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008) – kept for compatibility; the sub‑gradient
    routine uses a faster normalisation instead.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-6) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float = 1e-6) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform, normalised."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points.
    The result is re‑normalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    # re‑normalise
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total

# Optimisers

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth approximation of max(conv).
    Variables are logits w (softmax → a) guaranteeing a ≥ 0 and Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # initialise logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)                 # stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if it % 25000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.06) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    For speed we use a cheap “zero‑out‑and‑renormalise’’ projection.
    """
    a = initial_a.copy()
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad

        # cheap projection onto the simplex
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose target resolution depending on available time
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    target_n = 8192 if remaining_time > 300 else 4096
    print(f"\nTarget length: {target_n} (remaining time ≈ {remaining_time:.1f}s)")

    # ------------------------------------------------------------------
    # Seed generators
    # ------------------------------------------------------------------
    seed_functions = [
        lambda n: arcsine_initial(n),
        lambda n: generate_beta_shape(n, 0.8, 0.8, 1e-6),
        lambda n: generate_beta_shape(n, 1.2, 1.2, 1e-6),
        lambda n: generate_power_shape(n, 0.6, 1e-6),
        lambda n: generate_mixture_shape(n, 0.5),
        # random perturbation of the uniform distribution
        lambda n: (rng.random(n) + 0.001)
    ]

    # ------------------------------------------------------------------
    # Loop over seeds
    # ------------------------------------------------------------------
    for idx, gen in enumerate(seed_functions):
        if time.time() >= deadline - 5.0:
            break  # not enough time for another seed
        print(f"\n=== Seed {idx + 1}/{len(seed_functions)} ===")

        # ----- Coarse optimisation (n = 1024) -----
        n_coarse = 1024
        a0 = gen(n_coarse)
        a0 = a0 / a0.sum()                       # normalise to the simplex

        remaining = deadline - time.time()
        max_iter_coarse = int(min(120_000, remaining * 25))
        a_coarse = adam_optimize_smooth_max(
            a0,
            deadline,
            max_iters=max_iter_coarse,
            tau_start=0.5,
            tau_end=50.0,
            base_lr=0.5,
        )
        surrogate_coarse = compute_score(a_coarse)
        print(f"Coarse Adam surrogate = {surrogate_coarse:.6f}")

        # ----- Upsample to high resolution -----
        a_high = upsample(a_coarse, target_n)

        # ----- High‑resolution Adam -----
        remaining = deadline - time.time()
        max_iter_high = int(min(300_000, remaining * 20))
        a_refined = adam_optimize_smooth_max(
            a_high,
            deadline,
            max_iters=max_iter_high,
            tau_start=0.5,
            tau_end=300.0,
            base_lr=0.6,
        )
        surrogate_high = compute_score(a_refined)
        print(f"High‑res Adam surrogate = {surrogate_high:.6f}")

        # ----- Exact sub‑gradient descent -----
        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, remaining * 20))
        a_refined = subgradient_descent_exact(
            a_refined,
            deadline,
            max_iter=max_iter_sub,
            step0=0.07,
        )

        # ----- Exponentiated‑gradient polish -----
        remaining = deadline - time.time()
        max_iter_eg = int(min(150_000, remaining * 20))
        a_refined = eg_refine(
            a_refined,
            deadline,
            max_iter=max_iter_eg,
            eta0=0.25,
        )

        # ----- Official evaluation -----
        try:
            val = evaluate_sequence(a_refined.tolist())
            print(f"Seed {idx + 1} official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_refined.tolist()
                print("*** New global best! ***")
        except Exception as e:
            print(f"Evaluation failed for seed {idx + 1}: {e}")

        # ----- Perturb‑and‑polish around the current best (if time) -----
        if best_seq and time.time() < deadline - 5.0:
            print("  Perturb‑and‑polish current best...")
            base = np.array(best_seq, dtype=float)
            noise = rng.normal(scale=0.004, size=base.shape)
            pert = np.maximum(0.0, base + noise)
            pert /= pert.sum()

            # short sub‑gradient polish
            remaining = deadline - time.time()
            max_iter_sub = int(min(120_000, remaining * 25))
            pert = subgradient_descent_exact(
                pert,
                deadline,
                max_iter=max_iter_sub,
                step0=0.05,
            )
            # short EG polish
            remaining = deadline - time.time()
            max_iter_eg = int(min(80_000, remaining * 25))
            pert = eg_refine(
                pert,
                deadline,
                max_iter=max_iter_eg,
                eta0=0.20,
            )
            try:
                val = evaluate_sequence(pert.tolist())
                print(f"  Perturbed best score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = pert.tolist()
                    print("*** Updated best after perturbation ***")
            except Exception as e:
                print(f"  Evaluation after perturbation failed: {e}")

    # ------------------------------------------------------------------
    # Final global polish (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Final global polish ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 20))
        a = subgradient_descent_exact(
            a,
            deadline,
            max_iter=max_iter_sub,
            step0=0.05,
        )
        remaining = deadline - time.time()
        max_iter_eg = int(min(200_000, remaining * 20))
        a = eg_refine(
            a,
            deadline,
            max_iter=max_iter_eg,
            eta0=0.20,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated best after final polish ***")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
