def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns a length ``len(a) + len(b) - 1`` array.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_surrogate_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    Input a is assumed to be normalised (Σa = 1).  The surrogate equals
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the O(n) algorithm of
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(w: np.ndarray) -> np.ndarray:
    """Stable softmax to map logits to a probability vector."""
    w_shift = w - np.max(w)
    e = np.exp(w_shift)
    return e / e.sum()

# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at *peak*."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight)·arcsine + (1‑weight)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_asymmetric_power_shape(
    n: int, p: float, eps_left: float, eps_right: float
) -> np.ndarray:
    """
    Asymmetric generalised arcsine:
    a_i ∝ 1 / ((i+eps_left)*(n‑½‑i+eps_right))**p .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + eps_left) * (n - 0.5 - i + eps_right) + 1e-12) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation from length ``len(v)`` to ``target_n``, followed by
    renormalisation to Σ=1.
    """
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up /= up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 100_000,
    tau_start: float = 0.5,
    tau_end: float = 150.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φτ(a) = (1/τ)·log Σ_k exp(τ·conv_k(a,a)).
    The temperature τ grows exponentially from ``tau_start`` to ``tau_end``.
    Variables are logits w (a = softmax(w)), guaranteeing non‑negativity and
    Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # Initialise logits from a_init (add tiny epsilon for log stability)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (exponential)
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax mapping
        a = softmax_logits(w)

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()      # shape (L,)

        # gradient wrt a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wrt logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update with 1/√t decay
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (using the fast surrogate)
        if it % 20000 == 0:
            surrogate = compute_surrogate_score(softmax_logits(w))
            print(
                f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}"
            )

    # final softmax
    a_opt = softmax_logits(w)
    return a_opt


def subgradient_descent_exact(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution
    objective (max(conv)).  The step size decays like 1/√t.
    """
    a = project_to_simplex(a_init.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # tolerance to collect all indices attaining the maximum
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) restricted to original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror descent with KL geometry) polishing.
    """
    a = a_init.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a


def stochastic_local_search(
    a_init: np.ndarray,
    deadline: float,
    max_steps: int = 30_000,
    max_move_frac: float = 0.07,
) -> np.ndarray:
    """
    Very cheap hill‑climbing: repeatedly transfer a small amount of mass from
    a random source index to a random target index and keep the move if it
    lowers the smooth‑max surrogate.
    """
    a = a_init.copy()
    n = len(a)
    best_score = compute_surrogate_score(a)

    rng = np.random.default_rng()
    for step in range(max_steps):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # transfer up to max_move_frac of the source mass
        delta = a[i] * (rng.random() * max_move_frac)
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # re‑normalise (should already sum to 1, but rounding may drift)
        a_candidate /= a_candidate.sum()
        cand_score = compute_surrogate_score(a_candidate)
        if cand_score < best_score:
            a = a_candidate
            best_score = cand_score
            # occasional logging
            if step % 5000 == 0:
                print(f"    Local search step {step:,d} improved surrogate to {best_score:.6f}")

    return a

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with minimal official score.
    The method follows the multi‑stage pipeline described in the module
    docstring and returns the best sequence found before the time budget is
    exhausted.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Load a fallback sequence if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            # normalise to sum 1 for internal use
            fallback_arr = np.array(fallback, dtype=float)
            if fallback_arr.sum() == 0:
                fallback_arr = np.full_like(fallback_arr, 1.0 / fallback_arr.size)
            else:
                fallback_arr = fallback_arr / fallback_arr.sum()
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception as e:
            print(f"Failed to load fallback: {e}")

    # ------------------------------------------------------------------
    # Helper to update the global best using the official evaluator
    # ------------------------------------------------------------------
    def try_update(seq: np.ndarray, src: str):
        nonlocal best_seq, best_val
        try:
            score = evaluate_sequence(seq.tolist())
            if score < best_val:
                best_val, best_seq = score, seq.tolist()
                print(f"New global best ({src}) = {score:.6f}")
        except Exception as exc:
            print(f"Evaluation error ({src}): {exc}")

    # ------------------------------------------------------------------
    # 1️⃣  Small‑scale diverse seeding (n≈512)
    # ------------------------------------------------------------------
    n_small = 512
    pool: List[Tuple[float, np.ndarray]] = []

    uniform = np.full(n_small, 1.0 / n_small)
    pool.append((compute_surrogate_score(uniform), uniform.copy()))
    arc = arcsine_initial(n_small)
    pool.append((compute_surrogate_score(arc), arc.copy()))

    # generate many random parametrised shapes (time‑bounded)
    while time.time() < deadline - 5.0 and len(pool) < 800:
        typ = rng.integers(0, 6)
        if typ == 0:      # beta‑like
            a = rng.uniform(0.05, 0.6)
            b = rng.uniform(0.05, 0.6)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n_small))
            cand = generate_beta_shape(n_small, a, b, eps)
        elif typ == 1:    # power‑shape
            p = rng.uniform(0.2, 3.0)
            eps = rng.uniform(0.0, 3.0)
            cand = generate_power_shape(n_small, p, eps)
        elif typ == 2:    # mixture arcsine/uniform
            w = rng.random()
            cand = generate_mixture_shape(n_small, w)
        elif typ == 3:    # triangular bump
            peak = rng.integers(0, n_small)
            cand = generate_triangular_shape(n_small, int(peak))
        elif typ == 4:    # asymmetric power shape
            p = rng.uniform(0.2, 3.0)
            eps_l = rng.uniform(0.0, 2.0)
            eps_r = rng.uniform(0.0, 2.0)
            cand = generate_asymmetric_power_shape(n_small, p, eps_l, eps_r)
        else:              # random Dirichlet (smooth)
            alpha = rng.random(size=n_small) * 2.0 + 0.1
            cand = rng.gamma(alpha)
            cand = cand / cand.sum()
        pool.append((compute_surrogate_score(cand), cand))

    pool.sort(key=lambda x: x[0])
    top_seeds_small = [vec for _, vec in pool[:5]]
    print(f"\nSeeding complete (n={n_small}) – best surrogate = {pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam refinement on the small seeds
    # ------------------------------------------------------------------
    refined_small: List[Tuple[float, np.ndarray]] = []
    for idx, seed_vec in enumerate(top_seeds_small, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Adam on small seed {idx}/{len(top_seeds_small)} (n={n_small}) ---")
        remaining = deadline - time.time()
        max_iters = int(min(120_000, remaining * 30))   # generous but safe
        a_opt = adam_optimize_smooth_max(
            seed_vec,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=120.0,
            base_lr=0.28,
        )
        surrogate = compute_surrogate_score(a_opt)
        print(f"    Post‑Adam surrogate = {surrogate:.6f}")
        refined_small.append((surrogate, a_opt))

    refined_small.sort(key=lambda x: x[0])
    best_small = refined_small[0][1]   # champion at n=512
    try_update(best_small, "best_small (post‑Adam)")

    # ------------------------------------------------------------------
    # 3️⃣  Upscale to medium size (n=2048) and Adam again
    # ------------------------------------------------------------------
    n_medium = 2048
    if time.time() < deadline - 5.0:
        print(f"\n=== Upscaling to n={n_medium} and Adam ===")
        up = upscale_vector(best_small, n_medium)

        # small random jitter to escape local plateaus
        jitter = rng.normal(scale=1e-3, size=n_medium)
        up = up + jitter
        up = np.clip(up, 0.0, None)
        up /= up.sum()

        remaining = deadline - time.time()
        max_iters = int(min(180_000, remaining * 30))
        a_med = adam_optimize_smooth_max(
            up,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=150.0,
            base_lr=0.24,
        )
        surrogate_med = compute_surrogate_score(a_med)
        print(f"    Medium surrogate (n={n_medium}) = {surrogate_med:.6f}")
        try_update(a_med, "medium (post‑Adam)")
    else:
        a_med = None

    # ------------------------------------------------------------------
    # 4️⃣  Upscale to large size (n=8192) and Adam again
    # ------------------------------------------------------------------
    n_large = 8192
    if a_med is not None and time.time() < deadline - 5.0:
        print(f"\n=== Upscaling to n={n_large} and Adam ===")
        up_large = upscale_vector(a_med, n_large)

        # add a bit more jitter (scale relative to length)
        jitter = rng.normal(scale=5e-4, size=n_large)
        up_large = up_large + jitter
        up_large = np.clip(up_large, 0.0, None)
        up_large /= up_large.sum()

        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 30))
        a_large = adam_optimize_smooth_max(
            up_large,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=180.0,
            base_lr=0.22,
        )
        surrogate_large = compute_surrogate_score(a_large)
        print(f"    Large surrogate (n={n_large}) = {surrogate_large:.6f}")
        try_update(a_large, "large (post‑Adam)")
    else:
        a_large = None

    # ------------------------------------------------------------------
    # 5️⃣  Exact sub‑gradient + EG polishing on the large candidate
    # ------------------------------------------------------------------
    if a_large is not None and time.time() < deadline - 10.0:
        n_best = a_large.shape[0]
        print(f"\n--- Exact refinement on n={n_best} ---")
        # Sub‑gradient descent
        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 30))
        refined = subgradient_descent_exact(
            a_large,
            deadline,
            max_iter=max_iter_sub,
            step0=0.10,
        )
        # EG polish
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(200_000, remaining * 30))
            refined = eg_refine(
                refined,
                deadline,
                max_iter=max_iter_eg,
                eta0=0.20,
            )
        try_update(refined, "large (subgrad+EG)")

        # ------------------------------------------------------------------
        # 6️⃣  Stochastic local search for final tweaks
        # ------------------------------------------------------------------
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_steps = int(min(40_000, remaining * 30))
            refined = stochastic_local_search(
                refined,
                deadline,
                max_steps=max_steps,
                max_move_frac=0.05,
            )
            try_update(refined, "large (local search)")

    # ------------------------------------------------------------------
    # 7️⃣  Final safeguard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a modest uniform vector
        fallback_n = 1024
        uniform = [1.0 / fallback_n] * fallback_n
        best_seq = uniform
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform (n={fallback_n}) – score = {best_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
