def hill_climb_refine(a, deadline, initial_step=0.02, decay=0.999, min_step=1e-5):
    while time.time() < deadline:
        step = max(min_step, step * decay)
        # choose i,j
        i = rng.integers(len(a))
        j = rng.integers(len(a))
        if i == j:
            continue
        delta = step * a[i] * rng.random()
        if delta > a[i]:
            delta = a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # guarantee non-neg
        # evaluate
        score_new = compute_score(a_new)
        if score_new < best_score:
            a = a_new; best_score = score_new

"""
Propose a non‑negative step sequence that minimizes the official
``evaluate_sequence`` objective.

Strategy
--------

1. **Random parametric seeding** – generate a large pool of candidate
   shapes (beta‑distribution, power‑law, arcsine, mixtures, a
   configurable “γ‑shape”).  All candidates are normalised to sum to 1.

2. **Fast evaluation** – each candidate is scored with a cheap FFT‑based
   surrogate ``compute_score`` (identical to the official objective when
   the L1‑norm equals 1).  The best candidate seen so far is kept.

3. **Polishing** – the current global best is refined by three
   complementary optimisation phases (all respecting the simplex):
   * KL‑mirror (exponentiated‑gradient) descent on the true max‑convolution.
   * Adam on a smooth log‑sum‑exp surrogate of the max (temperature ↑ to
     sharpen the approximation).
   * Projected sub‑gradient descent for a final squeeze.

The routine monitors the wall‑clock and always returns a valid (non‑empty)
sequence before the 1000 s budget expires.
"""

# ----------------------------------------------------------------------
# Imports and low‑level utilities
# ----------------------------------------------------------------------
import time
from typing import List
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries sum to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all normalise to L1‑norm = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β)‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – the denominator is raised to p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_gamma_shape(n: int, gamma: float, delta: float = 0.5, eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    The classic arcsine corresponds to γ=0.5, δ≈0.5, ε≈0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 180_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # mask of indices attaining the maximum (numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # reverse mask for convolution with a
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimisation (optional sharper polishing)
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 300.0,
    ramp_iters: int = 50_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam on the log‑sum‑exp smooth approximation of max(conv).
    Operates in the log‑domain (soft‑max enforces the simplex).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a : 2·(a ⋆ w_rev)
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    return softmax(z)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low official ``evaluate_sequence`` score.
    The function respects the 1000 s wall‑clock limit and always returns a
    non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Random parametric search over several lengths
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]   # larger lengths explored first

    for n in lengths:
        if time.time() >= deadline - 20.0:
            break
        print(f"\n--- Random search for length {n} ---")

        # size‑aware time budget for this length
        length_deadline = min(deadline - 5.0, time.time() + (deadline - time.time()) * 0.3)

        iter_count = 0
        # rough estimate: ~0.0006 s per candidate (FFT + bookkeeping)
        max_iters = int((length_deadline - time.time()) / 0.0006) + 1
        max_iters = min(max_iters, 70_000)          # hard cap per length

        while time.time() < length_deadline and iter_count < max_iters:
            iter_count += 1

            # --------------------------------------------------------------
            # pick a generator and random parameters
            # --------------------------------------------------------------
            shape_type = rng.choice(["beta", "power", "gamma", "mixture", "arcsine", "uniform"])

            if shape_type == "beta":
                alpha = 10 ** rng.uniform(-1.0, 0.7)   # ≈[0.1,5]
                beta = 10 ** rng.uniform(-1.0, 0.7)
                eps = rng.uniform(0.0, max(0.5, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)

            elif shape_type == "power":
                p = rng.uniform(0.2, 0.9)            # around the arcsine exponent 0.5
                eps = rng.uniform(0.0, 2.0)
                cand = generate_power_shape(n, p, eps)

            elif shape_type == "gamma":
                gamma = rng.uniform(0.25, 0.75)      # explore around 0.5
                delta = rng.uniform(0.1, 1.0)
                eps = rng.uniform(0.0, 0.5)
                cand = generate_gamma_shape(n, gamma, delta, eps)

            elif shape_type == "mixture":
                w = rng.uniform(0.0, 1.0)
                cand = generate_mixture_shape(n, w)

            elif shape_type == "arcsine":
                cand = arcsine_initial(n)

            else:   # uniform
                cand = np.full(n, 1.0 / n)

            # normalise and guard against pathological zeros
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()

            # --------------------------------------------------------------
            # fast evaluation
            # --------------------------------------------------------------
            val = compute_score(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.tolist()
                print(f"New best (len={n}) it={iter_count:,d} score={val:.6f}")

        # --------------------------------------------------------------
        # 2.  Brief polish of the current best for this length (if time)
        # --------------------------------------------------------------
        if not best_seq:
            continue

        remaining = deadline - time.time()
        if remaining > 12.0:
            print(f"  Polishing length {n} with exponentiated‑gradient …")
            a_arr = np.array(best_seq, dtype=float)
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline=deadline,
                max_iter=int(min(80_000, remaining * 2000)),
                step=0.012,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"  Polished (EG) new best score={val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Global polishing – a three‑stage refinement
    # ------------------------------------------------------------------
    if best_seq:
        remaining = deadline - time.time()
        if remaining > 20.0:
            print("\n--- Global polishing phase ---")
            a_arr = np.array(best_seq, dtype=float)

            # 3a. KL‑mirror (exponentiated‑gradient) – longer run
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline=deadline,
                max_iter=int(min(150_000, remaining * 1500)),
                step=0.01,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"After EG polish score={val:.6f}")

            # 3b. Adam on smooth surrogate (sharpen max)
            if time.time() < deadline - 5.0:
                eps = 1e-12
                z0 = np.log(np.clip(a_arr, eps, None))
                a_arr = smooth_adam_optimize(
                    z0,
                    deadline=deadline,
                    n=a_arr.size,
                    beta0=5.0,
                    beta_max=400.0,
                    ramp_iters=80_000,
                    lr=0.07,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"After Adam polish score={val:.6f}")

            # 3c. Final exact sub‑gradient descent
            if time.time() < deadline - 3.0:
                a_arr = subgradient_descent_exact(
                    a_arr,
                    deadline=deadline,
                    max_iter=300_000,
                    step0=0.07,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"After final sub‑gradient score={val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee that we return something valid
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 5.  Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
