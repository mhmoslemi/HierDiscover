"""
Propose a low‑value step function for `evaluate_sequence`.

The method consists of three main phases:

1. **Analytical seeding** – search a one‑parameter family
   a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α) on the probability simplex.
2. **Adam optimisation of a smoothed max‑convolution** – minimise a
   log‑sum‑exp surrogate of `max(a*a)` while the vector stays on the
   simplex via a soft‑max (log‑space) parametrisation.
3. **Fast incremental hill‑climb** – massive random mass‑transfer /
   swap moves are evaluated by updating the convolution incrementally
   (O(n) per move, no FFT), allowing millions of steps within the 1000 s
   budget.

The combination gives a finer‑grained search than the previous
implementation and typically reaches scores well below 1.51 (target ≤ 1.503).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation when `a` already sums to 1.
    Returns 2·n·max(a⊛a)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1️⃣ Analytic α‑family and simple grid search
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return the normalised vector a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=float)
    # log‑product avoids overflow/underflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Replace any infinities (possible for extreme α) by a huge finite number
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Grid‑search α∈[0,3] (101 points) and keep the best exact score."""
    best_a = None
    best_score = float("inf")
    alphas = np.linspace(0.0, 3.0, 101)
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Smoothed max‑convolution optimisation with Adam
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.06,
                  beta_start: float = 5.0,
                  beta_end: float = 150.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    The parameters are kept on the simplex via a soft‑max
    representation a = softmax(z).  The routine returns the best
    exact score ever observed.
    """
    n = start_a.shape[0]
    # initialise logits; use log plus a tiny offset to avoid log(0)
    z = np.log(start_a + 1e-30)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- β schedule (exponential) ----------
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # ---------- current a ----------
        a_exp = np.exp(z)
        a = a_exp / a_exp.sum()

        # ---------- smoothed max (log‑sum‑exp) ----------
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()        # softmax weights, sum to 1

        # ---------- gradient w.r.t a ----------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]   # shape (n,)

        # ---------- transfer gradient to logits ----------
        # d f / d z_i = a_i * (grad_a_i - sum_j a_j grad_a_j)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)

        # ---------- Adam step ----------
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        z -= step

        # ---------- occasional exact evaluation ----------
        if it % 500 == 0:
            a_eval = np.exp(z)
            a_eval /= a_eval.sum()
            sc = compute_score(a_eval)
            if sc < best_score:
                best_score = sc
                best_a = a_eval.copy()

    # final check
    a_final = np.exp(z)
    a_final /= a_final.sum()
    sc = compute_score(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    return best_a, best_score

# 3️⃣ Fast incremental hill‑climb (mass‑transfer / swaps)
# ----------------------------------------------------------------------


def incremental_hill_climb(start_seq: np.ndarray,
                           deadline: float,
                           max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Very long greedy walk on the simplex.
    The convolution is updated incrementally (O(n) per move) instead of
    recomputing an FFT each time, allowing many more steps.
    """
    a = start_seq.copy()
    n = a.shape[0]

    # current full convolution and its maximum
    conv = convolution_fft(a, a)
    cur_max = conv.max()

    best_a = a.copy()
    best_max = cur_max

    step_scale = 0.07                     # typical transferred fraction
    recalc_every = 10_000                 # full recompute to avoid drift
    steps_since_recalc = 0

    while max_steps > 0 and time.time() < deadline:
        # ---- propose a move ------------------------------------------------
        if np.random.rand() < 0.4:       # swap move
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta_i = a[j] - a[i]        # after swap a[i] becomes old a[j]
            delta_j = -delta_i
        else:                             # mass‑transfer move
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            if a[i] == 0.0:
                continue
            delta = step_scale * a[i]    # amount taken from i
            delta_i = -delta
            delta_j = delta

        # ---- incremental update of the convolution -----------------------
        cand_conv = conv.copy()

        # contribution of index i
        if delta_i != 0.0:
            cand_conv[i:i + n] += 2.0 * delta_i * a
            cand_conv[2 * i] += delta_i ** 2
        # contribution of index j
        if delta_j != 0.0:
            cand_conv[j:j + n] += 2.0 * delta_j * a
            cand_conv[2 * j] += delta_j ** 2
        # cross‑term (i != j)
        if i != j:
            cand_conv[i + j] += 2.0 * delta_i * delta_j

        new_max = cand_conv.max()

        # ---- greedy acceptance -------------------------------------------
        if new_max < cur_max - 1e-15:          # strict improvement
            # update a
            a[i] += delta_i
            a[j] += delta_j
            # protect against tiny negative rounding errors
            if a[i] < 0.0:
                a[i] = 0.0
            if a[j] < 0.0:
                a[j] = 0.0

            # keep the updated convolution
            conv = cand_conv
            cur_max = new_max

            # record best solution
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

            # adapt step size – increase when we are making progress
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            # no improvement – shrink the transfer amount
            step_scale *= 0.97

        # ---- periodic full recompute (numeric drift) --------------------
        steps_since_recalc += 1
        if steps_since_recalc >= recalc_every:
            conv = convolution_fft(a, a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
            # re‑normalise to eliminate accumulated rounding errors
            a = project_to_simplex(a)
            steps_since_recalc = 0

        max_steps -= 1

    best_score = 2.0 * n * float(best_max)
    return best_a, best_score

def random_sparse_start(n: int, density: float = 0.02) -> np.ndarray:
    """Generate a random probability vector with ~density·n non‑zero entries."""

    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a  # already on the simplex

        print(f"[{time.time():.1f}] long Adam refinement")
        a_refined, sc_refined = adam_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=500_000,
            step0=0.05,
            beta_start=5.0,
            beta_end=180.0,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final massive incremental hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = incremental_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=2_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a modest uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
