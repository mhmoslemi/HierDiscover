"""
Propose a sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The code implements a multi‑scale, exponentiated‑gradient optimisation.
Key ideas
* Parameterise the vector on the simplex via soft‑max of unconstrained logits.
* Optimise a smooth surrogate of the max‑operator (log‑sum‑exp) with Adam.
* Initialise from several analytically motivated distributions
  (uniform, arcsine, random Dirichlet, sparse) to explore different basins.
* After the smooth phase perform a cheap hill‑climb that moves small amounts
  of mass between coordinates while always staying on the simplex.
All heavy work uses FFT‑based convolutions (`O(n log n)`).  The routine
keeps track of the time‑budget (default 1000 s) and always returns a
plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using a real FFT.
    Returned length = len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective value when `a` already satisfies sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex handling via the soft‑max map
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max that maps ℝⁿ → probability simplex."""
    v_max = np.max(v)
    e = np.exp(v - v_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Initialisation primitives
# ----------------------------------------------------------------------


def uniform_start(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def arcsine_start(n: int) -> np.ndarray:
    """
    Discretised arcsine distribution on [0,1]:
    p_i ∝ 1 / sqrt(x_i·(1−x_i)),  x_i = (i+0.5)/n.
    """
    i = np.arange(n) + 0.5
    x = i / n
    w = 1.0 / np.sqrt(x * (1.0 - x) + 1e-15)
    return w / w.sum()


def random_dirichlet_start(n: int, alpha: float = 1.0) -> np.ndarray:
    """Sample a Dirichlet(α,…,α) vector."""
    g = np.random.gamma(shape=alpha, scale=1.0, size=n)
    return g / g.sum()


def sparse_start(n: int, sparsity: float = 0.02) -> np.ndarray:
    """Very sparse vector with a few randomly placed masses."""
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = np.random.rand(k) + 0.1
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth optimisation on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def smooth_optimize_logits(
    v0: np.ndarray,
    n: int,
    time_deadline: float,
    max_iters: int = 120_000,
    beta_start: float = 6.0,
    beta_end: float = 120.0,
    step0: float = 0.7,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‐convolution objective on the
    log‑soft‑max parametrisation.
    Returns best probability vector (sum=1) and its exact score.
    """
    v = v0.astype(float).copy()
    m = np.zeros_like(v)
    vhat = np.zeros_like(v)

    best_a = softmax(v)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- forward pass ------------------------------------------------
        a = softmax(v)                     # stays on simplex automatically
        conv = convolution_fft(a, a)       # length 2n‑1
        maxc = conv.max()
        # smooth weights (log‑sum‑exp)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()            # probability distribution over lag indices

        # ---- gradient w.r.t. a -----------------------------------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)     # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ---- chain rule for the soft‑max --------------------------------
        dot = np.dot(grad_a, a)                     # scalar ⟨∇L, a⟩
        grad_v = grad_a * a - dot * a               # ∇_v L

        # ---- Adam update ------------------------------------------------
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad_v
        vhat = adam_beta2 * vhat + (1.0 - adam_beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - adam_beta1 ** it)
        vhat_hat = vhat / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        v -= step * m_hat / (np.sqrt(vhat_hat) + adam_eps)

        # occasional clipping to avoid overflow in the soft‑max
        if it % 500 == 0:
            v = np.clip(v, -50, 50)

        # ---- track the best exact score ---------------------------------
        if it % 200 == 0:
            a_cur = softmax(v)
            cur_score = compute_score(a_cur)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_cur.copy()

    # final check
    a_final = softmax(v)
    final_score = compute_score(a_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a_final.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb polishing (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq: list[float],
               time_deadline: float,
               max_steps: int = 400_000) -> tuple[list[float], float]:
    """
    Simple hill‑climb that repeatedly transfers a small fraction of
    probability mass between two coordinates.
    """
    a = np.array(seq, dtype=float)
    n = a.size

    best_a = a.copy()
    best_score = compute_score(best_a)

    # initial step‑size as a fraction of the current mass at the donor
    step_scale = 0.12

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        # choose the move type
        if np.random.rand() < 0.55:               # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        else:                                      # swap (helps tiny reshuffles)
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # project back onto simplex (here only a simple renormalisation)
        # because the move conserves total mass, we only need to clip negatives
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = uniform_start(n)
        else:
            cand /= cand_sum

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                         # become more cautious

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the 1000 s wall‑clock limit and returns a list of
    floats (a probability vector – sum=1 – which is invariant under scaling).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the outer harness
    deadline = start_time + budget_s - 1.5

    # -------------------------------------------------------------
    # Choose a working length – larger gives more freedom but costs
    # -------------------------------------------------------------
    remaining_total = deadline - time.time()
    if remaining_total > 800:
        n = 8192
    elif remaining_total > 400:
        n = 4096
    else:
        n = 2048
    print(f"[INFO] Working sequence length = {n}")

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # Build a pool of starting vectors
    # -------------------------------------------------------------
    starts = [
        uniform_start(n),
        arcsine_start(n),
        random_dirichlet_start(n, alpha=0.8),
        random_dirichlet_start(n, alpha=2.0),
        sparse_start(n, sparsity=0.03),
    ]

    # -------------------------------------------------------------
    # Phase 1 – smooth Adam optimisation from each start
    # -------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        now = time.time()
        if now >= deadline - 120.0:
            break                     # not enough time for later stages

        # allocate a slice of the remaining budget to this restart
        slice_deadline = min(deadline - 30.0, now + (deadline - now) * 0.4)
        v0 = np.log(np.maximum(a0, 1e-15))

        a_opt, scr = smooth_optimize_logits(
            v0,
            n,
            time_deadline=slice_deadline,
            max_iters=120_000,
            beta_start=5.0,
            beta_end=130.0,
            step0=0.8,
        )
        if scr < best_val:
            best_val = scr
            best_seq = a_opt.tolist()
            print(f"[PH1] New best after start {idx}: score={best_val:.6f}")

    # -------------------------------------------------------------
    # Phase 2 – final hill‑climb polishing (if time permits)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n[PH2] Hill‑climb polishing")
        polished_seq, polished_score = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=600_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"[PH2] Improved to {best_val:.6f}")

    # -------------------------------------------------------------
    # Safeguard – fall back to a modest uniform vector if nothing found
    # -------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform_start(256).tolist()
        best_val = compute_score(np.array(best_seq))

    # final verification using the provided security‑checked evaluator
    final_score = evaluate_sequence(best_seq)
    print(f"\n[FIN] Finished search. Final reported score = {final_score:.6f}")

    return best_seq
