"""
Propose a low‑score step‑function using a multi‑stage optimiser.

The pipeline (all operations stay on the probability simplex so the scale
is irrelevant for the original evaluate_sequence):

1. **Initial shapes** – deterministic candidates (arcsine, Hann,
   triangular, Gaussian) + Dirichlet random draws.  The arcsine
   distribution is the continuous solution whose self‑convolution is
   uniform, therefore it provides a very good starting point.

2. **Flattening** – minimise the variance of the linear convolution
   (∑(c‑μ)²) with Adam.  This directly flattens the autocorrelation.

3. **Smooth‑max descent** – minimise a smooth approximation of the
   max‑convolution ε·log∑exp(c/ε).  ε is decreased slowly (continuation)
   so the optimiser first sees a smooth landscape and later focuses on
   the true peak.

4. **Exact sub‑gradient** – fine‑tune the true objective
   2·n·max(a⊛a) using a classic sub‑gradient step (g_i = 2·a_{k‑i}).

5. **Simulated‑annealing hill‑climb** – stochastic mass transfers
   (swap or move a fraction of mass) with a linear temperature schedule.
   Improves the objective beyond the gradient steps.

6. **Symmetry averaging** – (a+reverse(a))/2 often reduces the peak
   further; the result is projected back onto the simplex.

All phases are time‑checked against the global deadline (≈1000 s) and the
best sequence seen is kept.  The code adapts the number of iterations to
the remaining time, prefers larger lengths (8192–16384) and uses a
variety of initialisations to avoid local minima.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution (a ⊛ b) via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


def arcsine_shape(n: int) -> np.ndarray:
    """U‑shaped arcsine distribution (continuous solution of the problem)."""
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def hann_shape(n: int) -> np.ndarray:
    """Hann window (bell shaped)."""
    i = np.arange(n, dtype=float)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / np.sum(a)


def triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular pyramid."""
    i = np.arange(n, dtype=float)
    a = np.minimum(i + 1, n - i)
    return a / np.sum(a)


def gaussian_shape(n: int, sigma_factor: float = 0.3) -> np.ndarray:
    """Centered Gaussian."""
    i = np.arange(n, dtype=float)
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - (n - 1) / 2.0) / sigma) ** 2)
    return a / np.sum(a)


# ----------------------------------------------------------------------
# Phase 2 – flattening the convolution (L2 variance) using Adam
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 10_000,
    step0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the convolution (Σ (c‑μ)²) with Adam,
    projecting onto the simplex after each step.
    Returns the best vector seen and its exact score.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    idx_start = 2 * n - 2

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)                 # length 2n‑1
        mu = np.mean(conv)                           # = sum_a**2 / (2n‑1) = 1/(2n‑1)
        diff = conv - mu

        # gradient = 4 * Σ_j a_j * diff_{i+j}
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)      # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – smooth max (log‑sum‑exp) descent with Adam
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 12_000,
    step0: float = 0.6,
    eps_factor: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Minimise ε·log∑exp(c/ε) where c = a⊛a, using Adam.
    ε follows a slow continuation schedule (never below 1 % of maxc).
    Returns the best exact score observed.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    idx_start = 2 * n - 2

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # continuation: ε decays but stays ≥ 1 % of maxc
        eps_cur = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        # log‑sum‑exp surrogate
        exp_vals = np.exp((conv - maxc) / eps_cur)
        w = exp_vals / np.sum(exp_vals)                 # soft‑max, Σw = 1

        # gradient of the surrogate: 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to shake out of plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true objective 2·n·max(a⊛a).
    The sub‑gradient at the peak index k is g_i = 2·a_{k‑i}.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the max‑index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random shake every 2000 iterations
        if t % 2000 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 5 – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    deadline: float,
    max_steps: int = 300_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with a linear cooling schedule.
    Moves are swaps or small mass transfers; always stay feasible.
    Returns the best sequence (as a Python list) and its score.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12  # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i → j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]          # 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0.0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional diversification
        if step % 5000 == 0:
            rnd = random_dense_vector(n) * 0.02   # 2 % random noise
            cand2 = best_a + rnd
            cand2 = project_to_simplex(cand2)
            sc2 = compute_score(cand2)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = cand2.copy()

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    (a + reverse(a)) / 2  followed by projection.
    Symmetric vectors often have a flatter self‑convolution.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a
    plain Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin (the harness will call evaluate_sequence once more)
    deadline = start_time + budget_s - 1.5

    best_seq = None
    best_score = float("inf")
    TARGET = 1.503  # a convenient early‑stop target

    # ------------------------------------------------------------------
    # Lengths to explore (larger lengths give a lower theoretical bound)
    # ------------------------------------------------------------------
    lengths = [8192, 12_288, 16_384]   # keep FFTs cheap enough

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate quickly (sum=1 → score = 2·n·maxconv)
    # ------------------------------------------------------------------
    def quick_score(vec):
        return compute_score(vec)

    # ------------------------------------------------------------------
    # Outer loop – generate candidates until time runs out
    # ------------------------------------------------------------------
    outer = 0
    while time.time() < deadline - 5.0:
        outer += 1
        n = int(np.random.choice(lengths))

        # ------------------------------------------------------------------
        # 1. Choose an initial shape
        # ------------------------------------------------------------------
        init_funcs = [
            arcsine_shape,
            hann_shape,
            triangular_shape,
            gaussian_shape,
            random_dense_vector,
        ]
        init_fun = np.random.choice(init_funcs)
        if init_fun is random_dense_vector:
            a0 = init_fun(n)
        else:
            a0 = init_fun(n)

        # initial quick check
        cur_sc = quick_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[{outer}] init (n={n}) score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 2. Flattening phase
        # ------------------------------------------------------------------
        rem = deadline - time.time()
        if rem < 6.0:
            break
        a_flat, sc_flat = flatten_conv_adam(
            a0,
            deadline=deadline - 2.0,
            max_iter=8000,
            step0=0.18,
        )
        if sc_flat < best_score:
            best_score = sc_flat
            best_seq = a_flat.tolist()
            print(f"[{outer}] flatten (n={n}) score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 3. Smooth‑max phase (continuation)
        # ------------------------------------------------------------------
        rem = deadline - time.time()
        if rem < 8.0:
            break
        a_smax, sc_smax = smooth_max_adam(
            a_flat,
            deadline=deadline - 3.0,
            max_iter=12000,
            step0=0.7,
            eps_factor=0.28,
        )
        if sc_smax < best_score:
            best_score = sc_smax
            best_seq = a_smax.tolist()
            print(f"[{outer}] smooth‑max (n={n}) score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 4. Exact sub‑gradient descent
        # ------------------------------------------------------------------
        rem = deadline - time.time()
        if rem > 12.0:
            a_sub, sc_sub = subgrad_optimize(
                a_smax,
                deadline=deadline - 5.0,
                max_iter=200_000,
                step0=0.045,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{outer}] sub‑grad (n={n}) score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 5. Simulated‑annealing hill‑climb
        # ------------------------------------------------------------------
        rem = deadline - time.time()
        if rem > 10.0:
            seq_to_opt = best_seq if best_seq is not None else a_sub.tolist()
            polished, polished_sc = hill_climb_anneal(
                seq_to_opt,
                deadline=deadline - 2.0,
                max_steps=250_000,
            )
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished
                print(f"[{outer}] anneal (n={n}) score={best_score:.6f}")

            # ------------------------------------------------------------------
            # 6. Symmetry averaging (cheap, often beneficial)
            # ------------------------------------------------------------------
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = quick_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer}] symmetry (n={n}) score={best_score:.6f}")

        # early stop if we have already beaten the target
        if best_score <= TARGET:
            print(f"Target achieved ({best_score:.6f}); exiting early.")
            break

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing found (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq, dtype=float))

    # ------------------------------------------------------------------
    # Final official evaluation (guards against any remaining scaling)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print("\nSearch finished. Final reported score = {:.6f}".format(final_score))
    return best_seq
