def fast_score(a):  # a is np.ndarray sum to 1
    n = a.shape[0]
    # compute convolution using FFT
    m = a.shape[0] + a.shape[0] - 1
    # compute optimal power-of-two length for FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    max_conv = conv.max()
    return 2.0 * n * max_conv

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0
    # Determine length
    remaining = deadline - time.time()
    n = 2048
    if remaining > 600:
        n = 8192
    elif remaining > 300:
        n = 4096
    else:
        n = 2048

    print(f"Using n={n}")

    # initial candidates list
    alphas = [0.5, 0.45, 0.55, 0.4, 0.6, 0.35, 0.65]
    best_seq = None
    best_score = float('inf')
    # also uniform
    seq_uniform = np.full(n, 1.0/n)
    score = fast_score(seq_uniform)
    best_seq = seq_uniform
    best_score = score
    print(f"Uniform score {best_score:.6f}")

    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = generate_beta_distribution(n, alpha)
        score = fast_score(a)
        print(f"Alpha {alpha:.2f} score {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a

    # local random search from best_seq
    a_cur = best_seq.copy()
    cur_score = best_score

    # set initial step scale based on n
    step_scale = 0.1  # fraction of mass to move
    while time.time() < deadline:
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a_cur[i]
        if delta <= 0:
            continue
        cand = a_cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # maintain positivity automatically
        # ensure normalization (sum unchanged)
        cand_score = fast_score(cand)
        if cand_score < cur_score:
            a_cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_cur.copy()
                print(f"New best after local move: {best_score:.6f}")

"""
Arc‑sine (Beta(½,½)) distribution framework.

The algorithm constructs a probability vector on the simplex whose
self‑convolution is (in the continuous limit) exactly uniform, which
minimises the objective `2·n·max(conv(a,a))`.  The construction is
parametrised by a Beta shape α; we try several α around 0.5, keep the
best, and then perform a cheap stochastic local‑search that transfers
small amounts of mass between entries.  All evaluation during the
search uses an FFT‑based exact score, while the final score is obtained
through the provided `evaluate_sequence` function.
"""

import time
import numpy as np

def fast_score(a):
    """Exact objective for a vector `a` (does not require it to sum to 1)."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s <= 0:
        return float("inf")
    a = a / s
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    return 2.0 * n * float(conv.max())

def beta_distribution(n, alpha):
    """Return a probability vector proportional to (x(1‑x))^{α‑1} on n points."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # protect against exact 0/1 – never occurs because of +0.5
    pdf = (x * (1.0 - x)) ** (alpha - 1.0)
    total = pdf.sum()
    if total == 0:
        pdf = np.full(n, 1.0 / n)
    else:
        pdf = pdf / total
    return pdf

def local_search(a, time_deadline, init_step=0.10):
    """Stochastic mass‑transfer hill‑climb that only accepts improving moves."""
    n = a.shape[0]
    cur = a.copy()
    cur_score = fast_score(cur)
    best = cur.copy()
    best_score = cur_score
    step = init_step
    while time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * cur[i]
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # maintain non‑negativity (delta <= cur[i] guarantees it)
        cand_score = fast_score(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                # increase step size a little – confidence in direction
                step = min(step * 1.05, 0.5)
        else:
            # slowly shrink step when move is not accepted
            step *= 0.97
    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Choose a reasonable length (larger gives a finer discretisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n = 8192
    elif remaining > 400:
        n = 4096
    else:
        n = 2048
    print(f"Chosen sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Scan a few Beta‑shapes around ½ (the continuous optimum)
    # ------------------------------------------------------------------
    alphas = [0.5, 0.45, 0.55, 0.4, 0.6, 0.35, 0.65]
    best_vec = np.full(n, 1.0 / n)   # uniform as a warm start
    best_val = fast_score(best_vec)
    print(f"Uniform initial score (fast) = {best_val:.6f}")

    for alpha in alphas:
        if time.time() >= deadline:
            break
        cand = beta_distribution(n, alpha)
        val = fast_score(cand)
        print(f"alpha={alpha:.2f} → fast score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_vec = cand
            print(f"  ** new best after Beta scan")

    # ------------------------------------------------------------------
    final_seq = best_vec.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

"""
Arc‑sine (Beta(½,½)) construction + cheap stochastic polishing.

The continuous probability law with density
    f(x) ∝ 1/√(x(1−x)),   x∈(0,1)
has the remarkable property that its self‑convolution is exactly uniform.
Discretising this “arcsine” law on a grid of length *n* yields a
probability vector whose auto‑convolution is almost flat, which minimises
the objective

    2·n·max(conv(a,a)) / (sum(a)²)   (the harness evaluates this).

The search therefore proceeds as follows:

1. Choose a length (2048 / 4096 / 8192) according to the remaining budget.
2. Build several Beta‑shaped vectors  a_i ∝ (x_i·(1−x_i))^{α−1}
   for α around ½ (the continuous optimum).  The fast FFT‑based score
   `fast_score` is used to keep the best one.
3. Optionally inspect a few random dense starts – cheap and sometimes
   lucky.
4. A very cheap stochastic “mass‑transfer” hill‑climb refines the best
   vector: repeatedly pick two positions *i*,*j* and move a small
   fraction of mass from *i* to *j*, accepting the move only when the
   exact FFT‑score improves.
5. The final candidate is evaluated with the official `evaluate_sequence`
   function and returned.

All heavy work uses only O(n log n) FFTs; the final convolution performed
by the harness (O(n²)) is executed only once.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast exact evaluation (FFT based) – used throughout the search
# ----------------------------------------------------------------------
def fast_score(a):
    """Exact objective 2·n·max(conv(a,a)) for a probability vector a."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    a = a / s                     # enforce sum‑to‑one (the objective is scale‑invariant)
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two ≥ m for an efficient real FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]   # linear convolution
    return 2.0 * n * float(conv.max())

# ----------------------------------------------------------------------
# Generate a discretised Beta(α,α) distribution on n points
# ----------------------------------------------------------------------
def beta_distribution(n, alpha):
    """
    Return a probability vector proportional to
        (x·(1−x))^{α−1},   x = (i+0.5)/n,  i=0,…,n−1 .
    For α=0.5 this is the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                     # points in (0,1)
    # Avoid exact 0 or 1 – the +0.5 shift guarantees they are never hit.
    pdf = (x * (1.0 - x)) ** (alpha - 1.0)
    total = pdf.sum()
    if total == 0.0:
        # fall back to uniform if something went wrong
        return np.full(n, 1.0 / n)
    return pdf / total

# ----------------------------------------------------------------------
# Very cheap stochastic local optimisation (mass‑transfer hill‑climb)
# ----------------------------------------------------------------------
def local_search(a, time_deadline, init_step=0.12):
    """
    Perform a simple hill‑climb that moves a fraction of mass from one
    entry to another.  Only improving moves are kept; the step size is
    slowly adapted.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = fast_score(cur)
    best = cur.copy()
    best_score = cur_score
    step = init_step

    while time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * cur[i]                     # keep non‑negativity
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = fast_score(cand)
        if cand_score < cur_score:                # accept only improvements
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                step = min(step * 1.05, 0.5)       # grow step when we succeed
        else:
            step *= 0.97                           # shrink step on rejection
    return best, best_score

    return final_seq
