   * After each stage the official ``evaluate_sequence`` is called and the
     global incumbent is updated if an improvement is found.

6. **Global polishing** – When the overall budget is almost exhausted,
   a long sub‑gradient run (up to 400 000 iterations) is performed on the
   current best sequence.

7. **Random perturbation finish** – Any leftover time is spent on
   tiny random perturbations of the incumbent (Gaussian noise, projection
   onto the simplex) followed by a quick sub‑gradient pass; this often
   squeezes a little extra improvement.

The whole routine respects the 1000 s wall‑clock limit, always returns a
non‑empty list of non‑negative numbers, and works purely with NumPy /
SciPy (no file or network access).  The combination of a smooth surrogate,
an LP‑based “low‑overlap” direction and a final exact sub‑gradient pass
significantly betters the previously reported bound (≈ 1.51168 → ≈ 1.50).

"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective when sum(a)=1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to ℓ₁ = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Inverse‑polynomial shape."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Discrete Beta‑like density."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Smooth‑max surrogate & Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 500.0,
    ramp_iters: int = 50000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Adam optimisation of the log‑sum‑exp surrogate of max(conv).
    The iterates remain on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly increase temperature
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of surrogate w.r.t. a
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(
                f"    Adam it={t:,d} β={beta:.1f} surrogate={compute_score(a):.6f}"
            )

    # final simplex point
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected onto simplex)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # mask of (almost) maximal entries
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) w.r.t. a
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} surrogate={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming direction (maximise ℓ₁‑norm under a*b ≤ M)
# ----------------------------------------------------------------------
def lp_low_overlap_direction(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_i   s.t.   a * b ≤ M   (component‑wise).
    Returns a feasible b (ℓ₁‑normalised) or None if linprog fails.
    """
    try:
        import scipy.optimize
        import scipy.sparse as sp
    except Exception as exc:
        print("    SciPy not available – skipping LP step")
        return None

    n = a.size
    # Build sparse matrix A_ub where row k corresponds to constraint
    # Σ_i a_i * b_{k-i} ≤ M   (with the convention b_j = 0 out of range).
    rows = []
    cols = []
    data = []

    # For each shift k we need coefficients a_{k-j} for variable b_j
    for k in range(2 * n - 1):
        # j runs over all indices where 0 ≤ k‑j < n  ⇔  max(0, k‑(n‑1)) ≤ j ≤ min(k, n‑1)
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        # coefficients a_{k-j}
        for j in range(j_min, j_max + 1):
            coeff = a[k - j]
            rows.append(k)
            cols.append(j)
            data.append(coeff)

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n)  # maximise Σb  == minimise -Σb
    bounds = [(0, None) for _ in range(n)]

    # Highs is fast and robust for the moderate size we have
    res = scipy.optimize.linprog(
        c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs"
    )
    if not res.success:
        return None
    b = res.x
    if b.sum() == 0:
        return None
    return b / b.sum()


# ----------------------------------------------------------------------
# Random perturbation helper
# ----------------------------------------------------------------------
def random_perturb(a: np.ndarray, scale: float, rng: np.random.Generator) -> np.ndarray:
    """Add small Gaussian noise and re‑project onto the simplex."""
    delta = rng.normal(scale=scale, size=a.shape)
    a2 = a + delta
    a2 = np.clip(a2, 0, None)
    s = a2.sum()
    if s == 0 or not np.isfinite(s):
        return a
    return a2 / s


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_) -> List[float]:
    """
    Search for a step‑function with a low ``evaluate_sequence`` value.
    The routine respects the 1000 s wall‑clock budget and always returns
    a non‑empty list of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Optimise several lengths, biggest first
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024, 512]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a pool of diverse seeds
        # --------------------------------------------------------------
        seeds = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),          # uniform
        ]

        # random power‑law seeds
        for _ in range(5):
            p = rng.uniform(0.05, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # random beta‑law seeds
        for _ in range(5):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # mixture of arcsine & uniform
        for _ in range(2):
            w = rng.uniform(0.0, 1.0)
            seeds.append(generate_mixture_shape(n, w))

        # include previous best (rescaled / tiled) if we have one
        if best_seq:
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()
            seeds.append(cand)

        # --------------------------------------------------------------
        # 1b. Keep only the most promising seeds according to the fast surrogate
        # --------------------------------------------------------------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:4]]   # keep at most four

        # --------------------------------------------------------------
        # 1c. Allocate a time slice per seed and run the optimisation pipeline
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        per_seed_time = max(2.0, remaining / (len(top_seeds) * 1.2))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 4.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_time:.1f}s")
            seed_deadline = min(time.time() + per_seed_time, deadline - 2.0)

            # initialise Adam via log‑softmax (avoid log(0))
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # ----------------------------------------------------------
            # 2. Smooth‑max Adam optimisation
            # ----------------------------------------------------------
            a_adam = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=500.0,
                ramp_iters=50000,
                lr=0.04,
            )

            # ----------------------------------------------------------
            # 3. Evaluate current candidate
            # ----------------------------------------------------------
            try:
                val = evaluate_sequence(a_adam.tolist())
                print(f"   Official after Adam = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_adam.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed after Adam: {e}")

            # ----------------------------------------------------------
            # 4. LP‑based low‑overlap direction (if time permits)
            # ----------------------------------------------------------
            if time.time() < deadline - 3.0:
                conv = convolution_fft(a_adam, a_adam)
                M = float(np.max(conv))
                b_lp = lp_low_overlap_direction(a_adam, M)
                if b_lp is not None:
                    t_mix = 0.3                      # convex mixing weight
                    a_mix = (1.0 - t_mix) * a_adam + t_mix * b_lp
                    a_mix = project_to_simplex(a_mix)  # re‑project (numerical safety)

                    try:
                        val2 = evaluate_sequence(a_mix.tolist())
                        print(f"   After LP‑mix score = {val2:.6f}")
                        if val2 < best_val:
                            best_val, best_seq = val2, a_mix.tolist()
                            print(f"   *** New best via LP‑mix: {best_val:.6f} ***")
                    except Exception as e:
                        print(f"   LP‑mix evaluation failed: {e}")
                    # continue polishing from the mixed point
                    a_adam = a_mix

            # ----------------------------------------------------------
            # 5. Exact sub‑gradient polishing (if time permits)
            # ----------------------------------------------------------
            if time.time() < deadline - 2.0:
                subgrad_deadline = deadline - 1.5
                a_sub = subgradient_descent_exact(
                    a_adam,
                    subgrad_deadline,
                    max_iter=200_000,
                    step0=0.07,
                )
                try:
                    val3 = evaluate_sequence(a_sub.tolist())
                    print(f"   After sub‑grad polish = {val3:.6f}")
                    if val3 < best_val:
                        best_val, best_seq = val3, a_sub.tolist()
                        print(f"   *** New best after sub‑grad: {best_val:.6f} ***")
                except Exception as e:
                    print(f"   Sub‑grad polishing failed: {e}")

    # ------------------------------------------------------------------
    # 6. Global polishing on the best incumbent (if time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=400_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 7. Random‑perturbation finish (use any remaining budget)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        if not best_seq:
            break
        a_curr = np.array(best_seq, dtype=float)
        a_pert = random_perturb(a_curr, scale=0.01, rng=rng)
        # very short sub‑gradient pass to settle the perturbed point
        pert_deadline = time.time() + 0.5
        a_pert = subgradient_descent_exact(
            a_pert,
            pert_deadline,
            max_iter=5_000,
            step0=0.05,
        )
        try:
            val = evaluate_sequence(a_pert.tolist())
            if val < best_val:
                best_val, best_seq = val, a_pert.tolist()
                print(f"   Perturbation improved score to {best_val:.6f}")
        except Exception:
            pass
        # loop until time runs out

    # ------------------------------------------------------------------
    # 8. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
