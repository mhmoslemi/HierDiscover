def adam_optimize(start_a, deadline, max_iters=200000, step0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    a = start_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute convolved via FFT
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute smoothed gradient using log-sum-exp (as before)
        beta = 5.0 * (80.0/5.0) ** (t/max_iters) # same schedule? maybe simpler.
        # Use large beta for approximating max (maybe constant large)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()
        # gradient = 2 * (a_rev * w) (as earlier)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = project_to_simplex(a)
        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def directed_hill_climb(start_a, deadline, max_steps=400000):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    step_scale = 0.1
    while max_steps>0 and time.time()<deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute products for shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j>=0) & (j<n)
        prod = np.zeros_like(a)
        prod[mask] = a[idx[mask]] * a[j[mask]]
        # identify i_max, j_min among mask
        if not np.any(mask):
            # shift invalid? shouldn't happen
            break
        i_max = np.argmax(prod)  # max product index
        i_min = np.argmin(prod)  # min product (maybe zero)
        # propose move from i_max to i_min

"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

The search consists of four stages:

1. **Fallback & analytical edge‑heavy start** – same α‑grid used previously.
2. **Decreasing power‑law family** – randomised search over a simple
   monotone family a_i ∝ (i+δ)^(-α) possibly mixed with a uniform part.
   This family is far more expressive for the “anti‑symmetry’’ needed to
   reduce the central convolution term.
3. **Smooth log‑sum‑exp annealed gradient descent** – the same routine as
   before but started from the best vector of stage 2 and allowed to run
   as long as time permits.
4. **Directed stochastic hill‑climb** – repeatedly moves a small amount
   of mass from the index that contributes most to the current maximal
   convolution entry to the index that contributes the least, accepting
   only improving moves (with a tiny simulated‑annealing wiggle).

All heavy work uses NumPy/FFT only, runs comfortably inside the 1000 s
wall‑clock budget and never requires external LP solvers.
The function returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, projection, scoring)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a)=1.
    Returns the exact score 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Edge‑heavy analytical start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Symmetric family a_i ∝ [(i+½)(n‑½‑i)]^(‑α) and normalised."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Grid search α∈[0,2.5] (26 points)."""
    best_a = None
    best_score = float("inf")
    alphas = np.linspace(0.0, 2.5, 26)
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Power‑law family utilities
# ----------------------------------------------------------------------


def power_law_vector(n: int, alpha: float, delta: float) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+δ)^(-α), normalised to sum 1."""
    i = np.arange(n, dtype=float)
    w = (i + delta) ** (-alpha)
    a = w / w.sum()
    return a


def mixture_vector(n: int, alpha: float, delta: float, w_uniform: float) -> np.ndarray:
    """
    Mixture of uniform and power‑law:
         a = w·U + (1‑w)·PL
    where both components already sum to 1, so the result also sums to 1.
    """
    uniform = np.full(n, 1.0 / n, dtype=float)
    pl = power_law_vector(n, alpha, delta)
    a = w_uniform * uniform + (1.0 - w_uniform) * pl
    return a


# ----------------------------------------------------------------------
# 3. Smooth log‑sum‑exp annealed gradient descent
# (identical to the previous implementation, but wrapped for reuse)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(start_a: np.ndarray,
                             deadline: float,
                             max_iters: int = 300_000,
                             beta_start: float = 5.0,
                             beta_end: float = 80.0,
                             step0: float = 0.6) -> tuple[np.ndarray, float]:
    """Annealed, projected gradient descent on the smoothed max‑convolution."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient (cross‑correlation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # step & projection
        step = step0 / np.sqrt(it)
        a = project_to_simplex(a - step * grad)

        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Directed stochastic hill‑climb (mass‑transfer guided by the active
#    convolution entry)
# ----------------------------------------------------------------------


def directed_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of mass from the index that contributes most
    to the current maximal convolution entry to the index that contributes
    the least. Only strictly improving moves are kept (with a tiny
    temperature wiggle to escape plateaus).
    """
    a = start_a.copy()
    n = a.size

    best_a = a.copy()
    best_score = compute_score(best_a)

    # tiny simulated‑annealing temperature parameters
    base_temp = 1e-5
    temp_factor = 0.9995

    step_scale = 0.12  # start moving ~12 % of the source mass

    while max_steps > 0 and time.time() < deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # indices that actually overlap for shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        if not np.any(mask):
            break

        # products a_i * a_{k-i}
        valid_i = idx[mask]
        valid_j = j[mask]
        prod = a[valid_i] * a[valid_j]

        i_max = valid_i[np.argmax(prod)]      # worst contributor
        i_min = valid_i[np.argmin(prod)]      # best place to receive mass

        # propose a small transfer
        delta = step_scale * a[i_max]
        if delta <= 0.0:
            break
        cand = a.copy()
        cand[i_max] -= delta
        cand[i_min] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)

        # improvement test – allow a negligible uphill move with temperature
        temperature = base_temp
        if sc < best_score - 1e-12 or np.random.rand() < np.exp(-(sc - best_score) / temperature):
            best_score = sc
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # cool the temperature very slowly
        base_temp *= temp_factor
        max_steps -= 1

    return best_a, best_score
