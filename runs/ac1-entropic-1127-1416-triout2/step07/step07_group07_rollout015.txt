def eg_descent(a0, time_deadline, max_iter=20000, eta0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute subgradient g
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # step size schedule
        eta = eta0 / np.sqrt(t)
        # multiplicative update
        a = a * np.exp(-eta * g)
        # renormalize to sum=1
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a
        # evaluate
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc
```

def anneal_refine(a0, deadline):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    # compute start time
    start = time.time()
    # schedule parameters
    max_iter = 200000
    while time.time() < deadline and max_iter > 0:
        # compute fraction of elapsed time
        elapsed = time.time() - start
        total_budget = deadline - start
        frac = elapsed / total_budget
        # temperature descending linearly from high to low
        T = (1 - frac) * 0.05 + frac * 0.0005
        # propose random transfer
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # fraction of mass to move
        alpha = np.random.rand() * 0.5
        delta = alpha * a[i]
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # project onto simplex? Actually after transfer sum unchanged and non-negativity guaranteed (cand[i]>=0)
        # we may need to clip small negative due to floating.
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand) # ensures exactly sum=1
        sc = compute_score(cand)
        if sc < best_sc or np.exp(-(sc - best_sc) / T) > np.random.rand():
            a = cand
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
        max_iter -= 1
    return best_a, best_sc

    # fallback if nothing

```python
"""
Search for a non‑negative step‑function minimizing the evaluation metric.

The method combines several optimisation stages:
1. Initialise a probability vector on the simplex using arcsine, sparse,
   or Beta‑shaped distributions.
2. Adam optimisation on a smooth soft‑max surrogate of the max–convolution.
3. Exponentiated‑gradient (mirror‑descent) updates on the exact max.
4. Classical sub‑gradient descent for fine‑tuning.
5. Greedy hill‑climbing followed by a simulated‑annealing polishing phase.
6. Optional symmetry averaging.

All stages respect the simplex constraint (non‑negative, sum = 1),
so the cheap `compute_score` equals the exact `evaluate_sequence` value.
The routine stops before the supplied time budget and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – FFT convolution, simplex projection, score
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when the vector has unit sum.
    Returns 2·n·max(conv) where conv = a⋆a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine‑like density: a_i ∝ 1/√(i·(n‑1‑i)).
    The distribution is symmetric and already gives a relatively flat
    self‑convolution.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_sparse_vector(n: int) -> np.ndarray:
    """Random probability vector with a modest number of non‑zero entries."""
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def beta_initialisation(n: int) -> np.ndarray:
    """
    Discrete Beta‑shaped density: a_i ∝ i^{α‑1} (n‑i+1)^{β‑1},
    where α,β are sampled randomly in (0.1, 5).  This yields a variety
    of U‑shaped, bell‑shaped and flat profiles.
    """
    alpha = np.random.uniform(0.1, 5.0)
    beta = np.random.uniform(0.1, 5.0)
    i = np.arange(1, n + 1, dtype=float)  # 1‑based index
    w = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    # Guard against overflow/underflow
    w = np.where(np.isfinite(w), w, 0.0)
    return project_to_simplex(w)


# ----------------------------------------------------------------------
# 1. Adam‑accelerated soft‑max descent (smooth surrogate)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f(a) = eps·log Σ_k exp(conv_k/eps)
    where conv = a⋆a.

    Returns  grad = 2 * (w_rev ⊛ a),
    with w = softmax(conv/eps).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)                 # Σ w = 1

    # cross‑correlation w_rev ⊛ a gives Σ_j a_j w_{i+j}
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 4000,
    step0: float = 0.2,
    eps_factor: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth soft‑max surrogate.
    ε is decreased exponentially; the algorithm stops when the time budget
    expires.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature – never smaller than 1 % of the current max
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        # Adam update (biased estimates)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)                       # basic decay
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 500 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent (max‑conv only)
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true objective 2·n·max(a⋆a).
    The sub‑gradient is taken at the shift where the convolution attains its
    maximum.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i} (zero outside the support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # tiny random shake every 1500 iterations
        if t % 1500 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climbing (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber that only accepts strictly improving moves.
    Moves:
        * swap two entries,
        * transfer a fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # transfer size adapts during the run
    step_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # numerical safeguard
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # become bolder
        else:
            step_frac *= 0.96                       # become more cautious

        max_steps -= 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Exponentiated‑gradient (mirror descent) on the exact max
# ----------------------------------------------------------------------


def eg_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 30_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the KL‑divergence (exponentiated
    gradient).  At each iteration the sub‑gradient of the max‑convolution
    is used; the update is a multiplicative weight step followed by a
    normalisation back onto the simplex.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max at position k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step size decays like O(1/√t)
        eta = eta0 / np.sqrt(t)

        # multiplicative update, then re‑normalise
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# 5. Simulated‑annealing refinement (accepting occasional uphill moves)
# ----------------------------------------------------------------------


def anneal_refine(
    a0: np.ndarray,
    deadline: float,
    total_budget: float,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing wrapper around random mass‑transfer moves.
    Temperature decreases linearly from start_T to end_T over the
    remaining time budget.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    start_T = 0.02
    end_T = 0.0005

    n = a.shape[0]
    while time.time() < deadline:
        # linear temperature schedule based on elapsed time
        elapsed = time.time() - (deadline - total_budget)
        frac = min(max(elapsed / total_budget, 0.0), 1.0)  # in [0,1]
        T = (1.0 - frac) * start_T + frac * end_T

        # propose a random transfer (or swap) move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random fraction of the source mass
        alpha = np.random.rand() * 0.5  # up to 50 % of source
        delta = alpha * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # numerical safety
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_sc = compute_score(cand)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand.copy()
            a = cand
            continue

        # accept with probability exp(−Δ/T)
        delta_score = cur_sc - best_sc
        prob = np.exp(-delta_score / (T + 1e-12))
        if prob > np.random.rand():
            a = cand

    return best_a, best_sc

# 6. Optional symmetry averaging
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Enforce (approx.) symmetry: (a + reverse(a))/2 and re‑project."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` score.  The routine respects the given time budget
    (≈ 1000 s) and returns a plain Python list ready for the evaluation
    harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    total_budget = max(0.0, budget_s - 2.0)   # keep a tiny safety margin
    deadline = start_time + total_budget

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Outer search – multiple random starts with varying lengths
    # ------------------------------------------------------------------
    lengths = [16384, 8192, 4096, 2048, 1024]   # larger sizes give more freedom
    outer = 0
    while time.time() < deadline - 5.0:
        outer += 1
        n = int(np.random.choice(lengths))

        # ------------------------------------------------------------------
        # 0a. Choose an initial distribution on the simplex
        # ------------------------------------------------------------------
        r = np.random.rand()
        if r < 0.5:
            a0 = arcsine_initialisation(n)
        elif r < 0.8:
            a0 = random_sparse_vector(n)
        else:
            a0 = beta_initialisation(n)

        # Quick evaluation of the raw initial point
        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[{outer}] init (n={n})  score={cur_sc:.6f}")

        # ------------------------------------------------------------------
        # 2. Adam‑accelerated soft‑max gradient descent
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 6.0:
            break
        a_adam, _ = adam_softmax_descent(
            a0,
            time_deadline=deadline - 2.0,

"""
Search for a non‑negative step‑function minimising the evaluation metric.

The algorithm proceeds through several optimisation phases:

1. Initialise a probability vector on the simplex using the arcsine law,
   a sparse random pattern or a discrete Beta‑shape (random α,β).

2. Adam optimisation on a smooth soft‑max surrogate of the max‑convolution.

3. Exponentiated‑gradient (mirror‑descent) updates on the true max
   using the exact sub‑gradient.

4. Classical sub‑gradient descent for fine‑grained tuning.

5. Greedy hill‑climbing followed by a simulated‑annealing polishing step
   that accepts occasional uphill moves.

6. Optional symmetry averaging (a←(a+rev(a))/2).

All stages respect the simplex constraint (non‑negative, sum = 1), which
means the cheap `compute_score` equals the exact `evaluate_sequence`
value when the sum is normalised to 1.  The routine respects the supplied
time budget (≈ 1000 s) and returns the best sequence found as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when the vector is on the simplex.
    Returns 2·n·max(a⋆a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
    (Wang & Carreira‑Perpiñán, 2013).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_sparse_vector(n: int) -> np.ndarray:
    """Random probability vector with a modest number of non‑zeros."""
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def beta_initialisation(n: int) -> np.ndarray:
    """
    Discrete Beta‑shaped density:
        a_i ∝ i^{α‑1} (n‑i+1)^{β‑1},
    with α,β sampled uniformly in (0.1, 5).  This yields a wide variety of
    shapes (U‑shaped, bell‑shaped, almost flat, …).
    """
    alpha = np.random.uniform(0.1, 5.0)
    beta = np.random.uniform(0.1, 5.0)
    i = np.arange(1, n + 1, dtype=float)  # 1‑based index
    w = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    w = np.where(np.isfinite(w), w, 0.0)
    return project_to_simplex(w)


# ----------------------------------------------------------------------
# 1. Adam on a smooth soft‑max surrogate
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of   f(a)=eps·log Σ exp(conv/eps)   w.r.t. a."""
    conv = convolution_fft(a, a)
    maxc = np.max(conv)
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)                 # Σ w = 1
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 4000,
    step0: float = 0.2,
    eps_factor: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth surrogate; returns best vector found."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if t % 500 == 0:            # gentle jitter
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution objective."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if t % 1500 == 0:            # small random shake
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Greedy hill‑climbing
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climber that only accepts strictly improving swaps or mass
    transfers.  The step size (`step_frac`) adapts automatically.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                   # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)
        else:
            step_frac *= 0.96

        max_steps -= 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Exponentiated‑gradient (mirror descent) on the exact max
# ----------------------------------------------------------------------


def eg_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 30_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using KL‑divergence.
    At each iteration the sub‑gradient of the current max‑convolution
    entry is used; the update is multiplicative followed by renormalisation.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# 5. Simulated‑annealing polishing
# ----------------------------------------------------------------------


def anneal_refine(
    a0: np.ndarray,
    deadline: float,
    total_budget: float,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves with a linear temperature schedule.
    Occasionally accepts uphill moves according to the Metropolis rule.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    start_T = 0.02
    end_T = 0.0005
    n = a.shape[0]

    while time.time() < deadline:
        # linear schedule based on elapsed time
        elapsed = time.time() - (deadline - total_budget)
        frac = np.clip(elapsed / total_budget, 0.0, 1.0)
        T = (1.0 - frac) * start_T + frac * end_T

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # transfer up to 50 % of the source mass
        alpha = np.random.rand() * 0.5
        delta = alpha * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_sc = compute_score(cand)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand.copy()
            a = cand
            continue

        # Metropolis acceptance
        delta_score = cur_sc - best_sc
        prob = np.exp(-delta_score / (T + 1e-12))
        if prob > np.random.rand():
            a = cand

    return best_a, best_sc

    a_sym = (a + a[::-1]) * 0.5

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
