"""
Improved search for a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm consists of

1. **Analytical families** – a one‑parameter family a_i(α) ∝
   [(i+½)(n‑½‑i)]^(‑α) and a two‑parameter mixture of two such families.
   A tiny golden‑section search refines the best α.

2. **Adam optimiser on the smooth log‑sum‑exp surrogate** – a modern
   first‑order method (Adam) with an annealing schedule for the
   smoothing parameter β.  The iterate is projected onto the probability
   simplex after each update.

3. **Projected sub‑gradient descent** – a classic 1/√t step‑size method
   applied to the true max‑objective; it clean‑ups the result of Adam.

4. **Simulated‑annealing hill‑climb** – a long stochastic walk that
   transfers a small amount of mass between two coordinates, accepting
   worsening moves with a temperature that slowly decays.

All heavy work uses only NumPy, runs inside the 1000 s wall‑clock limit
and never calls external LP/QP solvers.  The function `propose_candidate`
returns a plain Python list of floats ready for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact score as defined by `evaluate_sequence` for a probability vector
    (i.e. sum(a) == 1).  The function returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive → uniform point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def shape_alpha(n: int, alpha: float) -> np.ndarray:
    """
    Return the normalised symmetric shape
        a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α)
    on the probability simplex.
    """
    i = np.arange(n, dtype=float)
    # work in log‑space for stability
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def mix_two_shapes(n: int, a1: float, a2: float, w: float) -> np.ndarray:
    """
    Return a convex mixture of two α‑shapes:
        a = w·shape_alpha(α1) + (1‑w)·shape_alpha(α2)
    and renormalise to sum = 1.
    """
    s1 = shape_alpha(n, a1)
    s2 = shape_alpha(n, a2)
    a = w * s1 + (1.0 - w) * s2
    a /= a.sum()
    return a


def golden_section_alpha(n: int, lo: float, hi: float, deadline: float,
                         max_iter: int = 30) -> float:
    """
    Refine the best α in [lo, hi] by a golden‑section search.
    Stops early if the wall‑clock deadline is reached.
    """
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    resphi = 2.0 - phi  # = 1/phi²
    c = hi - resphi * (hi - lo)
    d = lo + resphi * (hi - lo)
    fc = compute_score(shape_alpha(n, c))
    fd = compute_score(shape_alpha(n, d))

    it = 0
    while (hi - lo) > 1e-4 and time.time() < deadline and it < max_iter:
        it += 1
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - resphi * (hi - lo)
            fc = compute_score(shape_alpha(n, c))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + resphi * (hi - lo)
            fd = compute_score(shape_alpha(n, d))

    return (lo + hi) / 2.0

def adam_optimize(start_a: np.ndarray, deadline: float,
                  max_iters: int = 250_000,
                  beta_start: float = 6.0, beta_end: float = 100.0,
                  lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed log‑sum‑exp surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    The iterate is projected onto the simplex each step.
    Returns the best exact score seen and the corresponding vector.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- annealing schedule for β ----
        t_frac = t / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # ---- smoothed weights ----
        conv = convolution_fft(a, a)  # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # ---- gradient of the smooth objective ----
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto the simplex ----
        a = project_to_simplex(a)

        # check exact score every few hundred iterations
        if t % 300 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray, deadline: float,
                    max_iters: int = 300_000,
                    step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective
    2·n·max(a⊛a).  Uses a 1/√t step‑size.
    Returns the best exact score observed.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 400 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def annealed_hill_climb(start_a: np.ndarray, deadline: float,
                        max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Long stochastic hill‑climb with simulated‑annealing acceptance.
    Move types:
      – swap two coordinates,
      – transfer a small fraction of mass from i to j.
    The temperature decays exponentially.
    Returns the best vector and its exact score.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(best_a)

    # temperature schedule
    start_temp = 1e-3
    end_temp = 1e-7
    temp_decay = (end_temp / start_temp) ** (1.0 / max_steps)
    temp = start_temp

    step_scale = 0.12  # initial fraction of source mass to move

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        cand = best_a.copy()
        if np.random.rand() < 0.5:               # swap move
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # mass‑transfer move
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            temp = start_temp                           # reset temperature a bit
        else:
            # probabilistic acceptance
            prob = np.exp(-(sc - best_score) / temp)
            if np.random.rand() < prob:
                best_a = cand
                best_score = sc
                step_scale = min(step_scale * 1.07, 0.5)
            else:
                step_scale *= 0.97

        temp *= temp_decay

    return best_a, best_score

def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    # choose sparsity between 1 % and 10 % of the coordinates
    proportion = np.random.uniform(0.01, 0.10)
    k = max(2, int(round(proportion * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_mixture_search(n: int, budget: float,
                          max_trials: int = 2000) -> tuple[np.ndarray, float]:
    """
    Sample random two‑α mixtures and keep the best.
    Stops either after `max_trials` or when `budget` seconds have elapsed.
    """
    best_a = None
    best_score = float("inf")
    start = time.time()
    for _ in range(max_trials):
        if time.time() - start >= budget:
            break
        a1 = np.random.uniform(0.0, 2.5)
        a2 = np.random.uniform(0.0, 2.5)
        w = np.random.rand()
        a = mix_two_shapes(n, a1, a2, w)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Finds a non‑negative sequence that (approximately) minimises
    `evaluate_sequence`.  The routine respects the 1000 s wall‑clock
    limit and returns a Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin – the harness adds its own checks
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose a working length.  8192 is a good compromise; we go to
    # 16384 only if a lot of time is left.
    # --------------------------------------------------------------
    n = 8192
    if deadline - start_time > 800:
        n = 16384
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  fallback – use the global sequence if present
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Uniform baseline
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    sc_uniform = compute_score(uniform)
    if sc_uniform < best_score:
        best_score = sc_uniform
        best_seq = uniform.tolist()
        print(f"[{time.time():.1f}] uniform baseline = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. One‑parameter α‑family (coarse grid + golden‑section refinement)
    # --------------------------------------------------------------
    # coarse scan
    alphas = np.linspace(0.0, 2.5, 26)           # step ≈0.1
    best_alpha = None
    for a in alphas:
        if time.time() >= deadline - 100:
            break
        cand = shape_alpha(n, a)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] α={a:.3f} improved → {best_score:.6f}")
        if best_alpha is None or sc < compute_score(shape_alpha(n, best_alpha)):
            best_alpha = a

    # golden‑section refinement around the best coarse α
    if best_alpha is not None and time.time() < deadline - 70:
        lo = max(0.0, best_alpha - 0.2)
        hi = min(2.5, best_alpha + 0.2)
        refined_alpha = golden_section_alpha(n, lo, hi, deadline - 60)
        cand = shape_alpha(n, refined_alpha)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] refined α={refined_alpha:.5f} → {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Random mixture of two α‑shapes
    # --------------------------------------------------------------
    if time.time() < deadline - 50:
        mix_budget = (deadline - time.time()) / 2.0   # give it half of the remaining time
        mix_seq, mix_sc = random_mixture_search(n, mix_budget, max_trials=3000)
        if mix_seq is not None and mix_sc < best_score:
            best_score = mix_sc
            best_seq = mix_seq.tolist()
            print(f"[{time.time():.1f}] random mixture improved → {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Random sparse restarts followed by Adam optimisation
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # create a fresh sparse start
        a0 = random_sparse_start(n)
        # short Adam run to polish the start
        a_opt, sc_opt = adam_optimize(
            a0,
            deadline - 5.0,
            max_iters=80_000,
            beta_start=5.0,
            beta_end=50.0,
            lr=0.025,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] sparse‑Adam gave new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Refine the current best vector with Adam (long run)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 25.0:
        print(f"[{time.time():.1f}] long Adam refinement")
        a_refined, sc_refined = adam_optimize(
            np.array(best_seq, dtype=float),
            deadline - 15.0,
            max_iters=300_000,
            beta_start=6.0,
            beta_end=120.0,
            lr=0.02,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] after long Adam → {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Projected sub‑gradient refinement (shorter but precise)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 8.0,
            max_iters=250_000,
            step0=0.06,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad → {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Final long simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print(f"[{time.time():.1f}] final annealed hill‑climb")
        a_final, sc_final = annealed_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved → {best_score:.6f}")

    # --------------------------------------------------------------
    # Safeguard: fall back to a tiny uniform vector if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final official evaluation (the harness will repeat this)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
