"""
Optimiser for the `evaluate_sequence` objective.

The search is performed in four stages:

1. **Parametric warm‑starts** – a short sweep over simple families
   (geometric and power‑law decays) yields a few decent baselines.
2. **Random restarts + smoothed log‑sum‑exp Adam optimisation** – many
   independent starts are refined with a smooth approximation of the
   max‑convolution.  A higher final β (≈ 300) gives a tighter proxy for
   the true max, while Adam‑style momentum speeds convergence.
3. **Projected sub‑gradient descent** – the exact non‑smooth objective
   2·n·max(conv) is minimised from the best vector found so far.
4. **Simulated‑annealing polishing** – small mass transfers between
   random positions are accepted with a temperature schedule,
   allowing occasional uphill moves to escape shallow local minima.

All stages respect the 1000 s wall‑clock limit and the routine always
returns a plain Python list of non‑negative floats (the vector is kept
on the simplex, i.e. it sums to 1, which leaves the scaling invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using a real FFT.
    The result has length len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp optimisation (Adam momentum)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 300.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a,a)) using Adam‑style updates.
    The routine keeps the best exact (non‑smooth) score encountered.
    """
    a = project_to_simplex(start_a.astype(float).copy())
    n = a.shape[0]

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal β linearly in log‑space
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max via log‑sum‑exp
        conv = convolution_fft(a, a)
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()

        # gradient of the smoothed objective = 2 * (a_rev * w)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active lag k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # small jitter every 1000 iterations to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (mass‑transfer moves)
# ----------------------------------------------------------------------


def anneal_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
    init_temp: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a temperature schedule.
    Small fractions of mass are moved between random positions.
    """
    a = project_to_simplex(start_a.astype(float).copy())
    cur_score = compute_score(a)
    best_a, best_score = a.copy(), cur_score
    n = a.shape[0]

    # initial step size (fraction of source mass)
    step_scale = 0.15
    # linearly decreasing temperature
    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # temperature goes from init_temp to ~0
        temperature = init_temp * (1.0 - step / max_steps)

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        diff = cand_score - cur_score

        if diff < 0 or np.exp(-diff / max(temperature, 1e-12)) > np.random.rand():
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
                # when we improve, slightly increase step size
                step_scale = min(step_scale * 1.04, 0.5)
        else:
            # no improvement – slowly shrink step size
            step_scale *= 0.99

    return best_a, best_score

# Helper factories for simple parametric families

def geometric_vector(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to sum = 1."""
    if r <= 0.0:
        # pure spike at the first index
        a = np.zeros(n)
        a[0] = 1.0
        return a
    powers = np.power(r, np.arange(n, dtype=float))
    total = powers.sum()
    return powers / total


def powerlaw_vector(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    indices = np.arange(1, n + 1, dtype=float)
    vals = np.power(indices, -p)
    return vals / vals.sum()

        if restart_deadline <= time.time():
            break

        a0 = random_start()
        a_opt, score_opt = smooth_gradient_optimize(
            a0,
            time_deadline=restart_deadline,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=300.0,
            step0=0.9,
        )
        if score_opt < best_val:
            best_val, best_seq = score_opt, a_opt.tolist()
            print(f"New best after smooth phase: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Sub‑gradient refinement on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print("\nStarting sub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 70.0,
            max_iter=300_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val, best_seq = refined_score, refined_a.tolist()
            print(f"Improved by sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5.  Simulated‑annealing polishing (mass‑transfer moves)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nAnnealing polishing")
        polished_a, polished_score = anneal_optimize(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=800_000,
            init_temp=1.0,
        )
        if polished_score < best_val:
            best_val, best_seq = polished_score, polished_a.tolist()
            print(f"Annealing improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.  Fallback to a tiny uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (uses the harness‑provided function)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
