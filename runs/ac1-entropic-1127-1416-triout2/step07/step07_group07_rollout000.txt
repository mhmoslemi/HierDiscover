"""
Search for a non‑negative step function minimizing the metric

    score = 2·n·max(conv) / (sum(a)**2)

The approach combines several complementary optimisation phases that
fit comfortably into the 1000 s budget:

  1. Initialise a probability vector on the simplex using either a
     discrete arcsine law (which already flattens the self‑convolution) or
     a random sparse distribution.

  2. Adam‑accelerated gradient descent on a smooth soft‑max surrogate of
     the max‑convolution.  The surrogate is  ε·log∑exp(conv/ε)  and its
     gradient is computed analytically via a single FFT.  The temperature
     ε is decreased exponentially; the objective is evaluated exactly at
     each iterate and the best iterate is retained.

  3. Classic sub‑gradient descent on the true L∞‑objective
     (i.e. on the current maximum convolution entry).  The step size is
     1/√t and the iterate is projected back onto the simplex.

  4. A simulated‑annealing hill‑climber that proposes two elementary
     moves – swapping two entries or transferring a fraction of mass
     from one index to another.  Moves are accepted with the Metropolis
     probability exp(−Δ/T) where the temperature T is cooled
     geometrically.  The best ever state is stored.

  5. Optional symmetry averaging (a ← (a+rev(a))/2) followed by a
     simplex projection – cheap and often improves the flatness.

All vectors are kept on the probability simplex, therefore the fast
`compute_score` (which assumes sum(a)=1) is exactly the same as the
official `evaluate_sequence` value up to the invariant scaling.
The routine respects the given time budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast and numerically stable)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a)=1.
    Returns 2·n·max(conv) which equals the exact score because the denominator
    (sum(a)**2) equals 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initialisation strategies
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine‑type density: a_i ∝ 1/√(i·(n‑1‑i)).
    The distribution is symmetric and yields a relatively flat
    self‑convolution.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_sparse_vector(n: int) -> np.ndarray:
    """Random probability vector with a modest number of non‑zero entries."""
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 1. Adam on the smooth soft‑max surrogate of max(conv)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f(a) = eps·log Σ_k exp(conv_k/eps)
    where conv = a⋆a.

    The gradient is  2·(w_rev ⊛ a)  with
        w = softmax(conv/eps)   (Σ w = 1).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)  # Σ w = 1

    w_rev = w[::-1]
    # correlation of a with w_rev has length 2n‑1
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 20000,
    step0: float = 0.3,
    eps_factor: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    eps is reduced exponentially; the loop stops when the time budget expires.
    Returns the best iterate seen.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature schedule – never let eps drop below 1 % of the current max
        eps = maxc * max(eps_factor * (0.97 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 700 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.003 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent directly on 2·n·max(a⋆a).
    The sub‑gradient is taken at *all* indices whose convolution value is
    within 0.1 % of the current maximum (active set), which yields a richer
    descent direction than using a single index.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # active set: entries whose value is close to the maximum
        tol = maxc * 0.001  # 0.1 %
        active = np.where(conv >= maxc - tol)[0]

        # build the sub‑gradient as a sum over the active shifts
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # average the contribution
        if len(active) > 1:
            g /= len(active)

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # light random shake every few thousand steps
        if t % 2000 == 0:
            a = project_to_simplex(a + np.random.randn(n) * (step0 * 0.004 / np.sqrt(t)))

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------


def hill_climb_sa(
    a_start: np.ndarray,
    time_deadline: float,
    steps: int = 300_000,
    init_temp: float = 1e-3,
    final_temp: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climber that accepts worsening moves with a Metropolis probability.
    Moves are either a swap of two entries or a transfer of a fraction of mass.
    The temperature decays geometrically from `init_temp` to `final_temp`.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_sc = compute_score(a)

    best_a = a.copy()
    best_sc = cur_sc

    # adaptive transfer size
    step_frac = 0.02

    start_time = time.time()
    for step in range(steps):
        if time.time() >= time_deadline:
            break

        # linearised progress for temperature schedule
        progress = step / steps
        T = init_temp * ((final_temp / init_temp) ** progress)

        # propose a move
        cand = a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_frac * cand[i]
                cand[i] -= delta
                cand[j] += delta
                # numerical guard
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑project to simplex (keeps sum=1 and non‑negative)
        cand = project_to_simplex(cand)

        cand_sc = compute_score(cand)
        delta_sc = cand_sc - cur_sc

        # Metropolis acceptance
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            a = cand
            cur_sc = cand_sc
            # adapt step size
            step_frac = min(step_frac * 1.03, 0.5)
        else:
            step_frac *= 0.97

        # keep the best ever seen
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc

    """Enforce (approx.) reflection symmetry and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
