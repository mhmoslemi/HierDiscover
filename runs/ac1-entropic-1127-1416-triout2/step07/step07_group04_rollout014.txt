def solve_lp_max_sum(f, M):

    n = len(f); L=2*n-1.

    A = scipy.linalg.toeplitz(np.r_[f, np.zeros(n-1)], np.r_[f[0], np.zeros(n-1)])

    # optionally A = A[:L,:] but size Lx n.

    # Use linprog.

    c = -np.ones(n)  # we minimize -sum => maximize sum

    bounds = [(0, None)] * n

    # Inequality constraints: A @ b <= M

    # Since linprog expects A_ub x <= b_ub

    A_ub = A

    b_ub = np.full(L, M)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})

    if not res.success: return None

    b = res.x

    return b

def solve_lp_max_sum(f, M):
    from scipy.optimize import linprog
    n = f.shape[0]
    L = 2 * n - 1
    # Build convolution matrix A
    # Use scipy.linalg.toeplitz
    from scipy.linalg import toeplitz
    first_col = np.r_[f, np.zeros(n - 1)]
    first_row = np.r_[f[0], np.zeros(n - 1)]
    A = toeplitz(first_col, first_row)[:L, :]
    # constraints: A @ b <= M
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(L, M),
                  bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None

def solve_lp_max_sum(f, M):
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz
    n = f.shape[0]
    L = 2*n - 1
    A = toeplitz(np.r_[f, np.zeros(n - 1)], np.r_[f[0], np.zeros(n - 1)])[:L, :]
    # Solve max sum(b) -> minimize -sum(b)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(L, M), bounds=[(0, None)]*n,
                  method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
step‑function optimizer with LP‑driven improvements
=================================================

The search proceeds in several independent *restarts* (until the
time‑budget expires).  Each restart consists of

1.  Randomly initialise a shape on a modest grid (n≈1024) – arcsine,
    beta‑type, power‑type or a mixture with a uniform component.

2.  A smooth‑max Adam phase (softmax parametrisation) quickly drives the
    surrogate ``2·n·max(conv)`` down.

3.  A few *LP‑updates*:
    given the current vector `f` (∑f=1) we compute `M = max(f*f)`.
    The linear program

        maximise   Σ b_j
        subject to  f * b ≤ M   (componentwise)
                    b_j ≥ 0

    yields a vector `b` that is as large as possible while keeping the
    convolution bound.  Rescaling `b` to sum = 1 and mixing
    `f ← (1‑t)·f + t·b` (projected onto the simplex) provably never
    increases `M`.  Only a handful of LP solves are needed – we stay on
    the coarse grid where the Toeplitz convolution matrix fits easily in
    memory.

4.  Upsample to the high‑resolution grid (default 4096 points) and run a
    second, longer Adam phase with a much larger temperature schedule
    (τ from 0.5 to 200).

5.  Exact polishing using Euclidean projected sub‑gradient descent
    followed by exponentiated‑gradient (mirror descent).

Every stage respects the global deadline (≈ 1000 s).  The best sequence
found (as a plain Python list) is returned.  If a fallback construction
`height_sequence_1` exists it is evaluated first.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    a is assumed to be normalised to Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-6) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float = 1e-6) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points and renormalisation.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


def random_initial_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random shape from a small family."""
    typ = rng.integers(0, 4)
    if typ == 0:  # arcsine
        return arcsine_initial(n)
    if typ == 1:  # beta
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        return generate_beta_shape(n, alpha, beta, eps=1e-6)
    if typ == 2:  # power
        p = rng.uniform(0.5, 2.0)
        return generate_power_shape(n, p, eps=1e-6)
    # mixture
    w = rng.uniform(0.0, 1.0)
    return generate_mixture_shape(n, w)


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (softmax parametrisation)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 150_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of a smooth approximation of max(conv(a,a)).
    The optimisation variable lives in the probability simplex via a softmax.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)                 # stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)            # length L

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if it % 25000 == 0:
            surrogate = objective_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (nonsmooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Linear‑programming update (LP‑improvement)
# ----------------------------------------------------------------------


def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   max Σ b_j   s.t.   conv(f,b) ≤ M  (componentwise),   b ≥ 0.
    Returns the optimal b (may be None on failure).
    """
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz

    n = f.shape[0]
    L = 2 * n - 1

    # Toeplitz convolution matrix: rows = linear convolution f * b
    first_col = np.r_[f, np.zeros(n - 1)]
    first_row = np.r_[f[0], np.zeros(n - 1)]
    A = toeplitz(first_col, first_row)[:L, :]   # shape (L, n)

    c = -np.ones(n)                               # maximise Σb  → minimise –Σb
    bounds = [(0, None)] * n
    b_ub = np.full(L, M)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if res.success:
        return res.x
    return None


def lp_improve(f: np.ndarray,
               t: float = 0.5,
               max_iter: int = 5,
               deadline: float = None) -> np.ndarray:
    """
    Apply a few LP‑driven mixing steps.  The vector f is kept on the simplex.
    """
    f = project_to_simplex(f)

    for it in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break

        conv = convolution_fft(f, f)
        M = float(np.max(conv))

        b0 = solve_lp_max_sum(f, M)
        if b0 is None:
            break

        s0 = b0.sum()
        if s0 <= 0.0:
            break

        # scale to unit sum
        b = b0 / s0

        new_f = (1.0 - t) * f + t * b
        new_f = project_to_simplex(new_f)

        new_M = float(np.max(convolution_fft(new_f, new_f)))
        if new_M < M - 1e-14:      # genuine improvement
            f = new_f
        else:
            # shrink the mixing factor if no progress
            t *= 0.5
            if t < 1e-3:
                break

    return f

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs several restarts, each consisting of a coarse Adam
    optimisation, a handful of LP‑updates, up‑sampling to a high‑resolution
    grid, a second long Adam phase and exact polishing.  The best sequence
    found before the deadline is returned.
    """
    import sys
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin of a few seconds
    deadline = start_time + budget_s - 5.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # Optional fallback from a previous construction (if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    attempt = 0
    # ------------------------------------------------------------------
    # Main restart loop – stop when the remaining time is too small
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:   # reserve ~30 s for a final polish
        attempt += 1
        print(f"\n=== Restart {attempt} ===")
        n_coarse = 1024
        a = random_initial_shape(n_coarse, rng)
        a = project_to_simplex(a)

        # --------------------------------------------------------------
        # Phase 1 – coarse Adam optimisation (smooth‑max)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        max_iters_coarse = int(min(120_000, remaining * 25))
        print(f"Phase 1: Adam on coarse grid (n={n_coarse}), iters≈{max_iters_coarse}")
        a = adam_optimize_smooth_max(a,
                                     deadline,
                                     max_iters=max_iters_coarse,
                                     tau_start=0.5,
                                     tau_end=80.0,
                                     base_lr=0.5)

        # --------------------------------------------------------------
        # Phase 2 – a few LP‑based improvement steps
        # --------------------------------------------------------------
        print("Phase 2: LP‑driven mixing")
        a = lp_improve(a,
                       t=0.5,
                       max_iter=5,
                       deadline=deadline)

        # --------------------------------------------------------------
        # Phase 3 – upsample to high resolution
        # --------------------------------------------------------------
        target_n = 4096
        # If we are running out of time, fall back to 2048 points
        if deadline - time.time() < 200.0:
            target_n = 2048
        print(f"Phase 3: Upsample to n={target_n}")
        a = upsample(a, target_n)

        # --------------------------------------------------------------
        # Phase 4 – long Adam refinement on the high‑resolution grid
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        max_iters_high = int(min(250_000, remaining * 25))
        print(f"Phase 4: High‑res Adam, iters≈{max_iters_high}")
        a = adam_optimize_smooth_max(a,
                                     deadline,
                                     max_iters=max_iters_high,
                                     tau_start=0.5,
                                     tau_end=200.0,
                                     base_lr=0.5)

        # --------------------------------------------------------------
        # Phase 5 – exact polishing (sub‑gradient + EG)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 25))
        print(f"Phase 5: sub‑gradient descent, iters≈{max_iter_sub}")
        a = subgradient_descent_exact(a,
                                      deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.07)

        remaining = deadline - time.time()
        if remaining > 8.0:
            max_iter_eg = int(min(200_000, remaining * 25))
            print(f"Phase 5: exponentiated‑gradient polish, iters≈{max_iter_eg}")
            a = eg_refine(a,
                          deadline,
                          max_iter=max_iter_eg,
                          eta0=0.25)

        # --------------------------------------------------------------
        # Official evaluation
        # --------------------------------------------------------------
        try:
            val = evaluate_sequence(a.tolist())
            print(f"Attempt {attempt}: official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print("*** New global best! ***")
            if best_val < 1.503:       # early exit if target met
                print("Target reached – exiting early.")
                break
        except Exception as exc:
            print(f"Evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # Final global polish (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n--- Final global polish ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(200_000, remaining * 25))
        a = subgradient_descent_exact(a,
                                      deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.05)

        remaining = deadline - time.time()
        if remaining > 5.0:
            max_iter_eg = int(min(150_000, remaining * 25))
            a = eg_refine(a,
                          deadline,
                          max_iter=max_iter_eg,
                          eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        print("No valid candidate found – returning uniform sequence.")
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
