def propose_candidate(seed=123, budget_s=1000):

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2

    best_seq = None
    best_val = float('inf')

    # fallback
    if 'height_sequence_1' in globals():
        try: fallback = list(map(float, globals()['height_sequence_1']))
            val = evaluate_sequence(fallback)
            if val < best_val: best_val, best_seq = val, fallback
        except: pass

    # define kernel for optimization
    def softmax(w):
        w_shift = w - np.max(w)
        exp_w = np.exp(w_shift)
        return exp_w / exp_w.sum()

    def phi_and_grad(w, tau):
        a = softmax(w)
        conv = convolution_fft(a, a)
        L = conv.size
        # compute weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = np.sum(exp_scaled)
        weights = exp_scaled / sum_exp
        # grad wrt a:
        rev_weights = weights[::-1]
        conv_rev = convolution_fft(a, rev_weights)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(a.shape[0])]
        # grad wrt w (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)
        phi = (max_scaled + np.log(sum_exp)) / tau
        return phi, grad_w

    # control schedule
    tau_schedule = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0]

    # list of target sizes
    target_sizes = [1024, 2048, 4096]  # start moderate, then large
    # ensure we don't exceed time for each; we can break if time is low

    for n in target_sizes:
        if time.time() >= deadline - 30: break
        print(f"\n=== Optimizing for size n={n} ===")
        # generate seeds
        seeds = []
        # arcsine
        seeds.append(arcsine_initial(n))
        # uniform
        seeds.append(np.full(n, 1.0/n))
        # mixture arcsine/uniform with random weight
        for _ in range(3):
            w = rng.random()
            seeds.append(generate_mixture_shape(n, w))
        # beta shapes random
        for _ in range(5):
            a = rng.uniform(0.05, 0.5)
            b = rng.uniform(0.05, 0.5)
            eps = rng.uniform(0.0, max(1.0, 0.02*n))
            seeds.append(generate_beta_shape(n, a, b, eps))
        # power shapes
        for _ in range(5):
            p = rng.uniform(0.2, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))
        # Dirichlet random
        for _ in range(5):
            alpha = rng.uniform(0.1, 5.0)
            seeds.append(rng.dirichlet(np.full(n, alpha)))
        # optionally include fallback if same size
        if best_seq is not None and len(best_seq) == n:
            seeds.append(np.array(best_seq))

        # we may prune duplicates or near duplicates? Not necessary.

        # run optimization on each seed
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline - 5:
                break
            # convert to probability simplex (should already sum to 1)
            seed_vec = seed_vec / seed_vec.sum()
            # jitter small noise for variance

            seed_vec = np.clip(seed_vec + jitter, 1e-12, None)
            seed_vec = seed_vec / seed_vec.sum()
            # convert to logits
            w = np.log(seed_vec + 1e-12)

            # continuation loop
            for tau in tau_schedule:
                if time.time() >= deadline - 5:
                    break
                # optimization using L-BFGS-B

                max_iter = 50  # adjust

                    phi, grad = phi_and_grad(w_vec, tau)
                    return phi, grad
                # call minimize

                               x0=w,

        best_val = evaluate_sequence(best_seq)
    print(f"\nSearch complete. Best official score = {best_val:.6f}")
    return best_seq

"""
Propose a step‑function sequence that minimizes ``evaluate_sequence``.

Algorithm
---------
1.  Gather a few diverse initial shapes (arcsine, uniform,
    mixtures, beta‑like, power‑like, random Dirichlet).  All are
    normalised to sum = 1 → they lie on the probability simplex.

2.  For each initial shape run a *continuation* of smooth‑max
    optimisation:
        •  Introduce a temperature τ and minimise the smooth‑max
           Φ_τ(a) = (1/τ)·log Σ_k exp(τ·(a∗a)_k) .
        •  Variables are logits w with a = softmax(w); this guarantees
           non‑negativity and Σa=1.
        •  For a given τ we solve the unconstrained problem
           min_w Φ_τ(softmax(w)) using SciPy's L‑BFGS‑B, supplying the
           exact gradient (derived from the weighted cross‑correlation).
        •  τ is increased in a schedule [1, 2, 5, 10, 20, 40, 80, 150];
           the solution from one τ is warm‑started for the next.

3.  After the final (large) τ we have a candidate a that is close to a
    true local optimum of max(a∗a).  Its official score is obtained with
    ``evaluate_sequence``.

4.  The best candidate found so far is further polished by
    *exact* sub‑gradient descent on the true max‑convolution objective
    (projected onto the simplex) followed by an exponentiated‑gradient
    (mirror‑descent) stage – the same refinements that proved effective
    in the previous code.

5.  All stages respect a global 1000 s budget; the best sequence
    encountered (according to the official evaluator) is returned.

The core novelty is the use of a high‑quality L‑BFGS optimisation on the
smooth‑max surrogate instead of the Adam optimiser used before.  This
provides faster, more accurate convergence and yields a lower final
bound.
"""

import time
import math
from typing import List, Tuple

import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (output length = len(a)+len(b)-1).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` when Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
    (Duchi et al., 2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (seed factories)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_asymmetric_power_shape(
    n: int, p: float, eps_left: float, eps_right: float
) -> np.ndarray:
    """
    Asymmetric generalised arcsine:
    a_i ∝ 1 / ((i+eps_left)*(n‑½‑i+eps_right))**p .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + eps_left) * (n - 0.5 - i + eps_right) + 1e-12) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Optimisation helpers (smooth‑max + L‑BFGS)
# ----------------------------------------------------------------------


def _softmax(w: np.ndarray) -> np.ndarray:
    w_shift = w - np.max(w)
    exp_w = np.exp(w_shift)
    return exp_w / exp_w.sum()


def phi_and_grad(w: np.ndarray, tau: float) -> Tuple[float, np.ndarray]:
    """
    Smooth‑max Φ_τ and its gradient w.r.t. logits w.
    a = softmax(w) → Σa = 1.
    """
    a = _softmax(w)
    conv = convolution_fft(a, a)                # length L = 2n‑1
    L = conv.size

    # weights = softmax(τ·conv)
    scaled = conv * tau
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    weights = exp_scaled / sum_exp               # shape (L,)

    # gradient w.r.t. a : 2·conv_rev where rev(weights) is the reversed weight vector
    rev_weights = weights[::-1]
    conv_rev = convolution_fft(a, rev_weights)   # length = n+L‑1
    grad_a = 2.0 * conv_rev[L - 1 - np.arange(a.shape[0])]

    # gradient w.r.t. logits (softmax Jacobian)
    dot = np.dot(a, grad_a)
    grad_w = a * (grad_a - dot)

    # smooth‑max value
    phi = (max_scaled + np.log(sum_exp)) / tau
    return float(phi), grad_w.astype(np.float64)


def l_bfgs_continuation(
    w0: np.ndarray,
    tau_schedule: List[float],
    deadline: float,
    maxiter_per_tau: int = 50,
) -> np.ndarray:
    """
    Runs L‑BFGS on Φ_τ for each τ in ``tau_schedule``,
    warm‑starting the next stage with the previous solution.
    Returns the final logits w.
    """
    w = w0.copy()
    for tau in tau_schedule:
        if time.time() >= deadline - 2.0:
            break
        # SciPy minimise; fun returns (value, grad)
        res = minimize(
            fun=lambda x: phi_and_grad(x, tau),
            x0=w,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_tau, "ftol": 1e-12, "disp": False},
        )
        w = res.x
    return w

# ----------------------------------------------------------------------


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)
    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with minimal ``evaluate_sequence``.
    The routine combines diverse seeding, L‑BFGS continuation on the
    smooth‑max surrogate, and a final sub‑gradient + exponentiated‑gradient
    polishing stage.  All work is bounded by the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # leave a safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Load a fallback solution if one exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Define optimisation parameters
    # ------------------------------------------------------------------
    tau_schedule = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0]
    # we will treat three problem sizes; larger sizes give more degrees of freedom
    target_sizes = [1024, 2048, 4096]

    for n in target_sizes:
        if time.time() >= deadline - 30:
            break
        print(f"\n=== Optimising for length n={n} ===")

        # --------------------------------------------------------------
        # 2️⃣  Build a modest pool of seeds (always normalised)
        # --------------------------------------------------------------
        seeds: List[np.ndarray] = []

        # deterministic pleasant shapes
        seeds.append(arcsine_initial(n))
        seeds.append(np.full(n, 1.0 / n))

        # random mixtures of arcsine/uniform
        for _ in range(3):
            w = rng.random()
            seeds.append(generate_mixture_shape(n, w))

        # beta‑like random shapes
        for _ in range(5):
            a = rng.uniform(0.05, 0.5)
            b = rng.uniform(0.05, 0.5)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # power‑type shapes
        for _ in range(5):
            p = rng.uniform(0.2, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # random Dirichlet draws (different concentration)
        for _ in range(5):
            alpha = rng.uniform(0.1, 5.0)
            seeds.append(rng.dirichlet(np.full(n, alpha)))

        # include the current global best if it has the same length
        if best_seq and len(best_seq) == n:
            seeds.append(np.array(best_seq, dtype=float))

        # --------------------------------------------------------------
        # 3️⃣  Optimise each seed via L‑BFGS continuation
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(seeds, start=1):
            if time.time() >= deadline - 5:
                break

            # tiny jitter to avoid exact symmetry sticking
            jitter = rng.normal(scale=1e-4, size=n)
            seed_vec = np.clip(seed_vec + jitter, 1e-12, None)
            seed_vec = seed_vec / seed_vec.sum()

            # initialise logits
            w0 = np.log(seed_vec + 1e-12)

            # run continuation
            w_opt = l_bfgs_continuation(
                w0,
                tau_schedule,
                deadline,
                maxiter_per_tau=60,  # generous per‑temperature budget
            )
            a_opt = _softmax(w_opt)

            # quick surrogate score (identical to official when Σa=1)
            surrogate = compute_score(a_opt)
            print(
                f"   n={n}, seed {idx}/{len(seeds)} → surrogate={surrogate:.6f}"
            )

            # official evaluation
            try:
                val = evaluate_sequence(a_opt.tolist())
            except Exception as e:
                print(f"   Evaluation error: {e}")
                continue

            print(f"   Official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"*** New global best! score={best_val:.6f} ***")

        # --------------------------------------------------------------
        # 4️⃣  Polish the current best of this size (if any)
        # --------------------------------------------------------------
        if best_seq and len(best_seq) == n:
            a_cur = np.array(best_seq, dtype=float)
            remaining = deadline - time.time()
            if remaining > 10:
                max_iter_sub = int(min(200_000, remaining * 25))
                a_cur = subgradient_descent_exact(
                    a_cur, deadline, max_iter=max_iter_sub, step0=0.09
                )
                if time.time() < deadline - 2:
                    max_iter_eg = int(min(150_000, (deadline - time.time()) * 25))
                    a_cur = eg_refine(
                        a_cur, deadline, max_iter=max_iter_eg, eta0=0.22
                    )
                try:
                    val = evaluate_sequence(a_cur.tolist())
                    if val < best_val:
                        best_val, best_seq = val, a_cur.tolist()
                        print(f"Polish improvement → new best = {best_val:.6f}")
                except Exception as e:
                    print(f"Polish evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 5️⃣  Fallback safety net
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence used, score={best_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
