def adam_smooth_optimize(start_a, deadline, max_iters=200000, beta_start=5.0, beta_end=200.0, lr=0.01):
    # start from a (on simplex), compute z = np.log(a+eps)
    # m, v arrays zero
    # For iter in range(max_iters):
         if time.time() >= deadline: break
         beta schedule: t = iter/max_iters
         beta = beta_start * (beta_end/beta_start)**t
         a = softmax(z)
         # conv
         conv = convolution_fft(a, a)
         maxc = conv.max()
         exp_w = np.exp(beta * (conv - maxc))
         w = exp_w / exp_w.sum()
         # compute gradient wrt a:
         a_rev = a[::-1]
         conv_rev = convolution_fft(a_rev, w)
         grad_a = 2.0 * conv_rev[n-1:n-1+n]
         # gradient wrt z
         dot_ag = np.dot(a, grad_a)
         grad_z = a * grad_a - dot_ag * a
         # Adam update
         m = beta1*m + (1-beta1)*grad_z
         v = beta2*v + (1-beta2)*(grad_z**2)
         m_hat = m / (1 - beta1**(t+1))
         v_hat = v / (1 - beta2**(t+1))
         z -= lr * m_hat / (np.sqrt(v_hat) + eps)
         # keep best a via compute_score evaluation periodically

"""
A fast multi‑stage optimizer for the non‑negative step‑function problem.

The algorithm
---------------
1.  Analytic families (symmetric product, left‑heavy power law) – dense
    grid search on the simplex, keep the best.
2.  Random monotone vectors – generate a few sorted random draws and test.
3.  Smooth “log‑sum‑exp” optimisation using Adam on the soft‑max
    parametrisation of the simplex.  The smoothing parameter β is
    annealed exponentially.
4.  Very long projected sub‑gradient descent on the true max‑objective
    (the cheapest iteration – only one FFT per step).  The step size is
    1/√t and the loop runs until the time budget is exhausted.
5.  Symmetrisation test – average the vector with its reverse and keep the
    better of the two.
6.  Final stochastic hill‑climb with a simple simulated‑annealing rule
    (swap or mass‑transfer moves, sometimes accepting uphill steps).
    The walk is kept on the probability simplex.

All stages respect the global wall‑clock limit (default 1000 s).  The
returned object is a plain Python list of floats ready for
`evaluate_sequence`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the probability simplex:
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # safety
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)        # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) using Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 150.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially from beta_start
    to beta_end.
    """
    n = start_a.shape[0]
    # initialise z from the start vector (log‑softmax)
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- β schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a -----
        a = softmax(z)

        # ----- smoothed weights -----
        conv = convolution_fft(a, a)            # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ----- gradient w.r.t. a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheapest iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for the true max‑objective
          2·n·max(a*a)
    using a diminishing stepsize  ∝ 1/√t.  The iterate is projected onto
    the probability simplex after each step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                        # index of active max

        # sub‑gradient of max‑entry (only the active terms)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        # occasional exact check
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with a tiny simulated‑annealing component
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Long random walk on the simplex.
    – With probability 0.5 swap two entries.
    – Otherwise move a fraction of mass from one index to another.
    A simple temperature schedule allows occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # a small temperature scale – we want only tiny occasional uphill steps
    T0 = best_score * 1e-4
    T = T0

    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            # accept move
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # expand exploration step size a little
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # rejected – shrink the transfer magnitude slowly
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)
    return best_a, best_score
