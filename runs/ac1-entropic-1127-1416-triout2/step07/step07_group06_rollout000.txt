def solve_max_sum_g(f, M):
    n = len(f)
    row = 2*n-1
    # build sparse matrix
    diags = [f]  # list of arrays each of shape (n,)
    offsets = np.arange(n)
    C = sp.diags(diags, offsets, shape=(row, n), format='csc')
    # Constraint: C @ g <= M * np.ones(row)
    b = np.full(row, M)
    # bounds: (0, None)
    res = linprog(c=-np.ones(n), A_ub=C, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback to zero vector
        return np.zeros_like(f)
    g0 = res.x
    # normalize to sum=1
    s = g0.sum()
    if s > 1e-12:
        g = g0 / s
    else:
        g = g0
    return g
```

def solve_lp_max_sum(f, M):
    import scipy.sparse as sp
    from scipy.optimize import linprog
    n = len(f)
    row = 2*n - 1
    # build sparse matrix
    diags = [f]  # note: f is array
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(row, n), format='csc')
    b = np.full(row, M, dtype=float)
    c = -np.ones(n)
    # bounds
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # If LP fails, fallback to zero vector
        return np.zeros_like(f)
    g0 = res.x
    s = g0.sum()
    if s > 1e-12:
        g = g0 / s
    else:
        g = g0
    return g

"""
Step‑function optimizer using a Frank–Wolfe style linear‑programming
subroutine.

The algorithm:
1. Start from a good known sequence (`height_sequence_1`) or a random
   Dirichlet vector on the probability simplex (∑a = 1).
2. Repeatedly solve a linear program:
       max  Σ g_j   subject to   f ⊛ g ≤ M   and   g ≥ 0,
   where M = max(f ⊛ f).  The convolution f ⊛ g is linear in g, so the
   constraints are linear.  This is a Frank–Wolfe step that provably
   drives the peak of the self‑convolution down.
3. Mix the current vector f with the LP solution g using a diminishing
   step size (t = 2/(k+2)) and re‑project onto the simplex.
4. After a handful of LP steps (typically <30) perform a short
   simulated‑annealing hill‑climb to polish the result.
5. Finish with a symmetry‑averaging projection (a ← (a+rev(a))/2).

All operations respect the 1000 s time budget; the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution utilities (same as in the previous code)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (Frank–Wolfe step)
# ----------------------------------------------------------------------


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve
        max Σ g_j
    s.t.   f ⊛ g ≤ M   (element‑wise)
           g ≥ 0
    The convolution f ⊛ g is linear in g, so the constraints are a set of
    linear inequalities.  The feasible set is a polytope; we solve with
    SciPy's high‑performance 'highs' interior‑point method.
    The returned vector is renormalised to sum to 1.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the sparse convolution matrix C where (C @ g)[k] = (f⊛g)[k].
    # Each diagonal offset d (0 ≤ d < n) contains the constant value f[d].
    diag_data = [np.full(n, f[d], dtype=float) for d in range(n)]
    C = sp.diags(diag_data, offsets=np.arange(n), shape=(rows, n), format="csc")

    # Upper‑bound constraints: C @ g ≤ M
    b = np.full(rows, M, dtype=float)

    # Objective: maximise Σ g_j  → minimise -Σ g_j
    c = -np.ones(n, dtype=float)

    # Bounds for each variable: g_j ≥ 0
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(
        c,
        A_ub=C,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # If the LP fails (should be rare) fall back to the current vector.
        return f.copy()

    g0 = res.x
    total = np.sum(g0)
    if total > 1e-12:
        g = g0 / total
    else:
        g = g0
    return g


# ----------------------------------------------------------------------
# Frank‑Wolfe optimisation loop
# ----------------------------------------------------------------------


def frank_wolfe_optimize(
    a0: np.ndarray, time_deadline: float, max_iter: int = 30
) -> tuple[np.ndarray, float]:
    """
    Apply a few Frank–Wolfe LP steps to lower the peak of the
    self‑convolution.  The iteration stops early if the time budget is
    exhausted or if improvement stalls.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # current peak
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # solve the LP for the current f = a
        g = solve_lp_max_sum(a, M)

        # Frank–Wolfe step size (standard diminishing schedule)
        step = 2.0 / (it + 2.0)

        a_new = (1.0 - step) * a + step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            print(
                f"[FW] iter {it:2d} – score improved to {best_sc:.6f} (step={step:.3f})"
            )

        # simple convergence test
        if abs(cur_sc - best_sc) < 1e-9:
            break

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (lightweight version)
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 50000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with a linear cooling schedule.
    Moves consist of swapping two indices or transferring a small
    fraction of the mass.  The method keeps feasibility by projection.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12  # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]  # move 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)

        cur_sc = compute_score(cand)
        delta_sc = cur_sc - best_sc

        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional diversification
        if step % 5000 == 0:
            jitter = random_dense_vector(n) * 0.02
            cand = best_a + jitter
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (helps to flatten the autocorrelation)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)
