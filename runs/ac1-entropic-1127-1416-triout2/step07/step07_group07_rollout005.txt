"""
Propose a step‑function (non‑negative vector) that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

The evaluation function `evaluate_sequence` is already provided in the
environment.  The algorithm consists of three phases:

1. **Parametric sweep** – a symmetric Beta‑type family
       a_i ∝ ((i+δ)*(N‑1‑i+δ))^{-β}
   with β∈[0.25,0.75] and a small shift δ≥0.  For several lengths (N∈{8192,
   4096, 2048}) the best vector is kept.  This family contains the
   discrete arcsine distribution (β=0.5,δ=0) but also smoother
   alternatives that empirically lower the convolution peak.

2. **Local refinement** – a few iterations of classic sub‑gradient descent
   on the true objective followed by a simple hill‑climber that only
   accepts strictly improving transfers.  The refinement works on the
   probability simplex, therefore the score `2·N·max(conv)` is identical
   to the value returned by `evaluate_sequence`.

3. **Symmetry averaging** – the final vector is symmetrised (averaged
   with its reverse) and re‑projected onto the simplex; this often
   reduces the maximal convolution entry a bit more.

All heavy work is performed with fast FFT‑based convolutions.  The whole
search respects the supplied time budget (≈1000 s).  The returned list
contains non‑negative numbers whose sum is exactly 1, so the final score
is the same as the fast surrogate used during the search.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, real‑valued)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a)=1.
    Returns 2·n·max(a⋆a) where n = len(a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0,
    Σ x = 1}.  Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        # Return uniform distribution if all entries are non‑positive.
        return np.full_like(v, 1.0 / v.shape[0])

    # Sort v in descending order.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric family – symmetric Beta‑type densities
# ----------------------------------------------------------------------


def beta_distribution(n: int, beta: float, delta: float) -> np.ndarray:
    """
    Returns a symmetric probability vector of length n with entries

        a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-β} .

    The vector is normalised to sum to 1.
    """
    i = np.arange(n, dtype=float)
    term = (i + delta) * (n - 1 - i + delta) + 1e-12
    w = term ** (-beta)
    # Guard against infinities / NaNs (should not happen with the +1e‑12).
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = np.sum(w)
    if s == 0.0:               # extremely unlikely, fall back to uniform
        return np.full(n, 1.0 / n)
    return w / s


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on  f(a) = 2·n·max(a⋆a) .
    The sub‑gradient is taken at any index where the convolution attains its
    maximal value.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # shift with maximal value

        # sub‑gradient: g_i = 2 * a_{k-i} (zero outside support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # tiny random shake to avoid stagnation
        if t % 1500 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple hill‑climbing (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climber that only accepts moves that strictly improve the
    objective.  Two move types are used:
        * swap two entries,
        * transfer a fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # Adaptive step fraction for the transfer move.
    step_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                max_steps -= 1
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                max_steps -= 1
                continue
            cand[i] -= delta
            cand[j] += delta
            # Numerical safeguard
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # become bolder
        else:
            step_frac *= 0.96                       # become more cautious

        max_steps -= 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (cheap post‑processing)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + reverse(a))/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects a ~1000 s time budget and
    returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a couple of seconds for the final evaluation call.
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Parametric sweep over the Beta family
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048]            # largish supports give finer granularity
    betas = np.linspace(0.30, 0.70, 9)      # 0.30 … 0.70 in steps of 0.05
    deltas = [0.0, 0.5, 1.0, 2.0, 5.0]     # tweak the edge regularisation

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        for beta in betas:
            for delta in deltas:
                if time.time() >= deadline - 5.0:
                    break
                a0 = beta_distribution(n, beta, delta)
                sc = compute_score(a0)
                if sc < best_score:
                    best_score = sc
                    best_seq = a0
                    print(
                        f"[init] n={n}, beta={beta:.3f}, delta={delta}, "
                        f"score={sc:.6f}"
                    )

    # ------------------------------------------------------------------
    # 2. Local optimisation (sub‑gradient + hill‑climb) on the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        # Ensure the vector is a proper probability vector (tiny re‑normalisation)
        best_seq = best_seq / np.sum(best_seq)

        # Sub‑gradient descent – limited iterations because each iteration
        # needs an FFT.
        while time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            # Allocate a modest number of iterations according to remaining time.
            max_iter = int(min(50_000, remaining * 30))   # ~30 sub‑grad steps per sec
            a_sub, sc_sub = subgradient_descent(
                best_seq,
                time_deadline=deadline - 2.0,
                max_iter=max_iter,
                step0=0.06,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub
                print(f"[sub‑grad] improved score = {sc_sub:.6f}")

            # Hill‑climb – number of steps adapts to the time left.
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            max_steps = int(min(300_000, remaining * 200))  # ~200 steps per sec
            a_hc, sc_hc = hill_climb(
                best_seq,
                time_deadline=deadline - 2.0,
                max_steps=max_steps,
            )
            if sc_hc < best_score:
                best_score = sc_hc
                best_seq = a_hc
                print(f"[hill‑climb] improved score = {sc_hc:.6f}")

            # Symmetry averaging – cheap, often gives a tiny boost.
            a_sym = final_symmetry_average(best_seq)
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = a_sym
                print(f"[symmetrise] improved score = {sc_sym:.6f}")

            # Break if we are about to exceed the deadline.
            if time.time() >= deadline - 5.0:
                break

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 4. Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    # Safety re‑normalisation (the simplex projection may have introduced
    # a tiny numerical drift).
    best_seq = best_seq / np.sum(best_seq)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_seq.tolist()
