"""
Hybrid grid‑search + multi‑phase optimizer for the step‑function attack.

The routine performs three high‑level stages within the 1000 s budget:

1. **Exploratory grid** – a collection of parametric families (power,
   beta, mixture, uniform) are instantiated on a modest grid (n≈2048).
   Each candidate receives a short Adam + sub‑gradient polish.
   The best few candidates are kept for the second stage.

2. **Upscaling & deep refinement** – the selected candidate(s) are
   linearly up‑sampled to a higher resolution (default 4096) and
   undergo a long Adam run with an aggressive temperature schedule
   (τ from 0.5 to 200).  Afterwards a projected sub‑gradient descent
   and an exponentiated‑gradient polish are applied.

3. **Final polishing** – the current global best is refined once more
   with the exact nonsmooth methods (if time permits).

All variables are kept on the probability simplex (non‑negative,
Σa = 1) via a softmax parametrisation (Adam) or Euclidean projection.
The objective being minimised is exactly the value returned by
``evaluate_sequence``; the implementation guarantees a non‑empty
return even if the search is interrupted early.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns length len(a)+len(b)-1 vector.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-6) -> np.ndarray:
    """
    Generalised arcsine: ((i+½)*(n‑½‑i)+eps)^{-p}, normalised.
    Larger p concentrates mass near the edges.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-6) -> np.ndarray:
    """
    Beta‑type shape on the discrete grid:
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float, eps: float = 1e-6) -> np.ndarray:
    """
    Mixture of an arcsine shape and the uniform distribution.
    weight_arc ∈ [0,1] is the weight of the arcsine component.
    """
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points,
    followed by renormalisation to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (softmax parametrisation)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 150_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the log‑sum‑exp (smooth‑max) surrogate.
    Variable is the logits w; a = softmax(w) lives on the simplex.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-16)          # initialise logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # Linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max via log‑sum‑exp
        scaled = conv * tau
        max_s = np.max(scaled)
        exp_scaled = np.exp(scaled - max_s)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp  # size L

        # gradient wrt a:
        # d/da max_smooth = Σ_k w_k * d/da conv_k, and
        # d/da conv_k = 2 * (a shifted by k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)          # length n+L-1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wrt logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)          # diminishing step size
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional diagnostics
        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    [Adam] it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact nonsmooth polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Projected Euclidean sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries attaining (or nearly attaining) the max
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a -= eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent with KL geometry) polish.
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The procedure mixes a fast parametric grid search with a deep
    multi‑phase refinement (Adam → sub‑gradient → exponentiated‑gradient).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (previously discovered construction)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"[Fallback] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 1 – exploratory grid on a modest resolution (n≈2048)
    # ------------------------------------------------------------------
    n_grid = 2048
    print(f"\n--- Stage 1: grid exploration (n={n_grid}) ---")

    # Define a modest collection of parametric families
    param_grid: List[Tuple[str, dict]] = [
        ("power", {"p": 1.0, "eps": 1e-6}),
        ("power", {"p": 1.2, "eps": 1e-6}),
        ("power", {"p": 1.5, "eps": 1e-6}),
        ("power", {"p": 1.8, "eps": 1e-6}),
        ("power", {"p": 2.0, "eps": 1e-6}),
        ("power", {"p": 2.5, "eps": 1e-8}),
        ("beta",  {"alpha": 0.4, "beta": 0.4, "eps": 1e-6}),
        ("beta",  {"alpha": 0.3, "beta": 0.3, "eps": 1e-6}),
        ("beta",  {"alpha": 0.6, "beta": 0.6, "eps": 1e-6}),
        ("mixture", {"weight_arc": 0.7}),
        ("mixture", {"weight_arc": 0.5}),
        ("uniform", {}),
    ]

    # Keep the best few candidates (by surrogate) for stage 2
    candidate_pool: List[Tuple[float, np.ndarray]] = []

    for typ, pars in param_grid:
        if time.time() >= deadline - 30.0:
            print("[Stage 1] time is running out, stopping grid search")
            break

        if typ == "power":
            a0 = generate_power_shape(n_grid, pars["p"], pars["eps"])
        elif typ == "beta":
            a0 = generate_beta_shape(n_grid, pars["alpha"], pars["beta"], pars["eps"])
        elif typ == "mixture":
            a0 = generate_mixture_shape(n_grid, pars["weight_arc"])
        elif typ == "uniform":
            a0 = np.full(n_grid, 1.0 / n_grid)
        else:
            continue

        # short Adam + sub‑gradient polish
        remaining = deadline - time.time()
        # allocate a small fraction of the remaining budget to each candidate
        max_it_adam = int(min(80_000, remaining * 12))
        a_adam = adam_optimize_smooth_max(
            a0,
            deadline,
            max_iters=max_it_adam,
            tau_start=0.5,
            tau_end=30.0,
            base_lr=0.4,
        )
        max_it_sub = int(min(80_000, (deadline - time.time()) * 12))
        a_refined = subgradient_descent_exact(a_adam, deadline,
                                              max_iter=max_it_sub,
                                              step0=0.08)

        # quick official evaluation
        try:
            val = evaluate_sequence(a_refined.tolist())
        except Exception:
            val = float("inf")

        surrogate = compute_score(a_refined)
        print(f"[Grid] {typ} {pars} → surrogate={surrogate:.6f} official={val:.6f}")

        # retain a copy for later upscaling if it looks promising
        candidate_pool.append((val, a_refined.copy()))

        # update global best immediately
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()
            print(f"*** New global best from grid! score={best_val:.6f} ***")

    # Keep at most three best candidates (excluding the current global best)
    candidate_pool.sort(key=lambda x: x[0])
    top_candidates = [seq for _, seq in candidate_pool[:3]]

    # ------------------------------------------------------------------
    # Stage 2 – high‑resolution refinement of the best candidates
    # ------------------------------------------------------------------
    target_n = 4096
    if deadline - time.time() < 120.0:
        target_n = 2048  # fallback if time is short

    print(f"\n--- Stage 2: upscale to n={target_n} and deep refinement ---")

    for idx, cand in enumerate(top_candidates):
        if time.time() >= deadline - 30.0:
            print("[Stage 2] time is running out, stop further candidates")
            break

        # Upsample to high resolution
        a_high = upsample(cand, target_n)

        # Length‑aware deep Adam
        remaining = deadline - time.time()
        max_it_adam_high = int(min(250_000, remaining * 15))
        a_adam_high = adam_optimize_smooth_max(
            a_high,
            deadline,
            max_iters=max_it_adam_high,
            tau_start=0.5,
            tau_end=200.0,
            base_lr=0.5,
        )

        # Exact polish (sub‑gradient + EG)
        max_it_sub_high = int(min(200_000, (deadline - time.time()) * 15))
        a_sub = subgradient_descent_exact(a_adam_high, deadline,
                                          max_iter=max_it_sub_high,
                                          step0=0.07)

        if time.time() < deadline - 2.0:
            max_it_eg_high = int(min(180_000, (deadline - time.time()) * 15))
            a_polished = eg_refine(a_sub, deadline,
                                   max_iter=max_it_eg_high,
                                   eta0=0.22)
        else:
            a_polished = a_sub

        # Official evaluation
        try:
            val = evaluate_sequence(a_polished.tolist())
        except Exception:
            val = float("inf")

        print(f"[Refine #{idx+1}] official score = {val:.6f}")

        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()
            print(f"*** New global best after high‑res refinement! score={best_val:.6f} ***")

    # ------------------------------------------------------------------
    # Stage 3 – final global polish (if any time remains)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n--- Stage 3: final global polishing ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_it_sub_final = int(min(300_000, remaining * 12))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_it_sub_final,
                                      step0=0.05)

        if time.time() < deadline - 2.0:
            max_it_eg_final = int(min(200_000, (deadline - time.time()) * 12))
            a = eg_refine(a, deadline,
                          max_iter=max_it_eg_final,
                          eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
