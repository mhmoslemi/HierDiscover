    a = np.exp(w_shift)

"""
step‑function optimiser – multi‑scale Adam + max‑Adam + sub‑gradient

The optimiser works in three phases:

0️⃣  Load a fallback vector (if present) and keep it as a seed.
1️⃣  Generate a large pool of diverse shapes for n=1024, keep the best few,
    and refine them with Adam on a *smooth‑max* surrogate (log‑sum‑exp of the
    convolution).  The temperature is increased to ≈ 300 so the surrogate
    almost coincides with the true max.
2️⃣  Upscale the champion to larger lengths (2048 → 4096 → 8192).  After a
    short Adam warm‑up the vector is optimised with *Adam on the exact max*
    (sub‑gradient of the max‑convolution) – this is far more aggressive than
    the smooth surrogate.  Finally a long Euclidean projected sub‑gradient
    descent followed by an exponentiated‑gradient polish are applied.

All stages respect the global 1000 s budget, keep the best vector according
to the official ``evaluate_sequence`` function and finally return it.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate used during the search.
    The vector a is assumed to sum to 1, therefore the official score is
    exactly 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex Σx=1, x≥0.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangle, apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_asymmetric_power_shape(
    n: int, p: float, eps_left: float, eps_right: float
) -> np.ndarray:
    """
    Asymmetric generalised arcsine:
    a_i ∝ 1 / ((i+ε_l)*(n‑½‑i+ε_r))**p .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + eps_left) * (n - 0.5 - i + eps_right) + 1e-12) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(v) to target_n, followed by renormalisation."""
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up /= up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 80_000,
    tau_start: float = 0.5,
    tau_end: float = 300.0,
    base_lr: float = 0.2,
) -> np.ndarray:
    """
    Adam on the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k)
    with an exponential temperature schedule.
    The variables are logits w with a = softmax(w), guaranteeing non‑negativity
    and Σa = 1.
    Returns the best vector seen during the run (according to the surrogate).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_w = w.copy()
    best_surrogate = np.inf

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        conv = convolution_fft(a, a)

        # smooth‑max
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        surrogate = (np.log(exp_scaled.sum()) + max_scaled) / tau

        if surrogate < best_surrogate:
            best_surrogate = surrogate
            best_w = w.copy()

        # softmax weights over τ·conv
        weights = exp_scaled / exp_scaled.sum()          # shape (L,)

        # gradient wrt a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wr.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging
        if it % 20_000 == 0:
            print(f"    smooth‑Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # return the best vector observed
    w_shift = best_w - np.max(best_w)
    a_best = np.exp(w_shift)
    a_best /= a_best.sum()
    return a_best


def adam_optimize_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimiser that works directly on the true max‑convolution objective.
    The gradient used is a sub‑gradient obtained from the active (max) indices.
    Returns the best vector seen during the run.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_w = w.copy()
    best_obj = np.inf

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # softmax to obtain current a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # keep best
        obj = 2.0 * n * max_val
        if obj < best_obj:
            best_obj = obj
            best_w = w.copy()

        # mask of (almost) maximal entries
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        grad_a = 2.0 * convolution_fft(a, rev_mask)[L - 1 - np.arange(n)]

        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # very sparse logging
        if it % 30_000 == 0:
            print(f"    max‑Adam it={it:,d} obj={obj:.6f}")

    # final softmax using the historically best logits
    w_shift = best_w - np.max(best_w)
    a_best = np.exp(w_shift)
    a_best /= a_best.sum()
    return a_best


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.15,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        subgrad = 2.0 * convolution_fft(a, rev_mask)[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.20,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent w.r.t. KL).
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        subgrad = 2.0 * convolution_fft(a, rev_mask)[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget and returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.9  # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  fallback (if a previous run stored a candidate)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback candidate official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1️⃣  generate a rich pool for n=1024
    # --------------------------------------------------------------
    n_small = 1024
    pool: List[Tuple[float, np.ndarray]] = []

    # deterministic starters
    pool.append((compute_score(np.full(n_small, 1.0 / n_small)), np.full(n_small, 1.0 / n_small)))
    pool.append((compute_score(arcsine_initial(n_small)), arcsine_initial(n_small)))

    for _ in range(800):
        if time.time() >= deadline - 30:
            break
        typ = rng.integers(0, 7)
        if typ == 0:
            a = rng.uniform(0.01, 0.9)
            b = rng.uniform(0.01, 0.9)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n_small))
            cand = generate_beta_shape(n_small, a, b, eps)
        elif typ == 1:
            p = rng.uniform(0.05, 3.0)
            eps = rng.uniform(0.0, 5.0)
            cand = generate_power_shape(n_small, p, eps)
        elif typ == 2:
            w = rng.random()
            cand = generate_mixture_shape(n_small, w)
        elif typ == 3:
            peak = rng.integers(0, n_small)
            cand = generate_triangular_shape(n_small, peak)
        elif typ == 4:
            p = rng.uniform(0.05, 3.0)
            eps_l = rng.uniform(0.0, 2.0)
            eps_r = rng.uniform(0.0, 2.0)
            cand = generate_asymmetric_power_shape(n_small, p, eps_l, eps_r)
        elif typ == 5:
            # Dirichlet with concentration α ∈ (0.01, 2)
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n_small, alpha))
        else:
            # random perturbation of arcsine
            cand = arcsine_initial(n_small) * (1.0 + rng.normal(scale=0.2, size=n_small))
            cand = np.clip(cand, 0.0, None)
            cand = cand / cand.sum()
        pool.append((compute_score(cand), cand))

    pool.sort(key=lambda x: x[0])
    top_seeds_small = [vec for _, vec in pool[:6]]
    print(f"\nInitial seeding finished – best surrogate = {pool[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam (smooth‑max) refinement of the top seeds
    # --------------------------------------------------------------
    refined_small: List[Tuple[float, np.ndarray]] = []
    for idx, seed_vec in enumerate(top_seeds_small, start=1):
        if time.time() >= deadline - 20:
            break
        remaining = deadline - time.time()
        max_iters = int(min(120_000, remaining * 30))
        print(f"\n--- Smooth‑Adam on seed {idx}/{len(top_seeds_small)} (max_iters={max_iters}) ---")
        a_opt = adam_optimize_smooth_max(
            seed_vec,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=300.0,
            base_lr=0.22,
        )
        surrogate = compute_score(a_opt)
        print(f"    after smooth‑Adam surrogate = {surrogate:.6f}")
        refined_small.append((surrogate, a_opt))

    refined_small.sort(key=lambda x: x[0])
    best_small = refined_small[0][1]
    print(f"\nBest small‑scale surrogate after smooth‑Adam = {refined_small[0][0]:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Upscale progressively and apply aggressive max‑Adam
    # --------------------------------------------------------------
    best_large_vec = None
    best_large_obj = float("inf")

    for target_n in (2048, 4096, 8192):
        if time.time() >= deadline - 25:
            break
        print(f"\n=== Upscaling to n={target_n} ===")
        up = upscale_vector(best_small, target_n)

        # small random jitter to escape potential symmetries
        up += rng.normal(scale=5e-4, size=target_n)
        up = np.clip(up, 0.0, None)
        up /= up.sum()

        remaining = deadline - time.time()
        max_iters = int(min(180_000, remaining * 25))
        a_opt = adam_optimize_max(
            up,
            deadline,
            max_iters=max_iters,
            base_lr=0.18,
        )
        obj = compute_score(a_opt)  # this is the official objective for sum=1
        print(f"    after max‑Adam surrogate (official) = {obj:.6f}")
        if obj < best_large_obj:
            best_large_obj, best_large_vec = obj, a_opt

    # --------------------------------------------------------------
    # 4️⃣  Exact sub‑gradient + EG polish on the best large candidate
    # --------------------------------------------------------------
    if best_large_vec is not None:
        n_best = best_large_vec.shape[0]
        print(f"\n--- Exact refinement on n={n_best} ---")
        remaining = deadline - time.time()
        max_sub = int(min(600_000, remaining * 30))
        refined = subgradient_descent_exact(best_large_vec, deadline, max_iter=max_sub, step0=0.12)

        if time.time() < deadline - 5:
            remaining = deadline - time.time()
            max_eg = int(min(250_000, remaining * 30))
            refined = eg_refine(refined, deadline, max_iter=max_eg, eta0=0.18)

        try:
            val = evaluate_sequence(refined.tolist())
            print(f"Official score after full polish = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print("*** New global best! ***")
        except Exception as e:
            print(f"Evaluation failed after polishing: {e}")

    # --------------------------------------------------------------
    # 5️⃣  Final safety – ensure a non‑empty solution
    # --------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform – official score = {best_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
