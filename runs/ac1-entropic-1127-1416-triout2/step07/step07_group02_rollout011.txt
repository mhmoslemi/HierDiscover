"""
Optimisation of a non‑negative step function.
-------------------------------------------------
The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear convolution of the coefficient vector with itself.
The value is invariant under scaling, therefore we may restrict the search to
probability vectors (sum = 1).

The algorithm works in three stages:

1. **Exploratory sweep** – a broad collection of analytically‑motivated
   families (arcsine, beta, inverse‑power, centred‑power, sinusoid windows,
   Hann, random Dirichlet) is generated for several lengths
   (8192, 4096, 2048, 1024).  Each candidate is scored with the cheap exact
   surrogate `fast_score`.  The best few (default 3) per length are kept.

2. **Local refinement** – every kept seed is refined by a continuation
   Adam optimiser that minimises a smooth approximation of the max‑convolution
   (log‑sum‑exp with a temperature β that is ramped from 5 to 800).  The
   parameters are kept on the probability simplex via a soft‑max mapping.
   After Adam, a short mirror‑descent step (exponential‑gradient) uses the
   true sub‑gradient of the max‑convolution to push the solution further
   down.

3. **Global polishing** – the current best vector is finally polished with a
   projected sub‑gradient descent (exact max) for the remaining time budget.

The routine respects a total time budget (default 1000 s), always returns a
valid non‑negative sequence, and prints occasional progress information.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate used for pre‑selection (scale‑invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/sum(a)^2 (exact surrogate, no clipping)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families for seed generation
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = vals / vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type density ∝ (i+eps)^{α‑1} (n‑1‑i+eps)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_inv_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Inverse‑power density ∝ 1 / ((i+½)(n‑½‑i)+eps)^{p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_centered_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Centered power density ∝ ((i+½)(n‑½‑i)+eps)^{p}."""
    i = np.arange(n, dtype=float)
    base = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_sine_power_shape(n: int, q: float) -> np.ndarray:
    """Sinusoid window raised to a power: sin(π(i+½)/n)^{q}."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n) ** q
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_hann_shape(n: int) -> np.ndarray:
    """Classical Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=float)
    vals = 0.5 * (1.0 - np.cos(2.0 * np.pi * (i + 0.5) / n))
    return vals / vals.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex via Dirichlet(1,…,1)."""
    vec = rng.dirichlet(np.ones(n))
    return vec


# ----------------------------------------------------------------------
# Projection utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Adam optimiser on smooth max (log‑sum‑exp) with continuation on β
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 800.0,
    ramp_iters: int = 120_000,
    lr: float = 0.02,
) -> np.ndarray:
    """
    Minimise the smooth surrogate  L_β(a) = log‑sum‑exp(β·conv(a,a))
    while enforcing a on the simplex via a soft‑max.
    β is increased linearly from beta0 to beta_max (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                      # a >= 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights (log‑sum‑exp trick)
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of L_β w.r.t. a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)          # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]     # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 10_000 == 0:
            print(f"    Adam it={t:,d}  β={beta:.1f}  surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) with the exact max‑subgradient
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.16,
    decay: float = 0.9985,
) -> np.ndarray:
    """
    Exponential‑gradient descent on the true objective.
    The iterate stays on the simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        # occasional debug output
        if int(step * 1e6) % 200_000 == 0:
            print(f"    Mirror step={step:.6f}  score≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polish)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.13,
) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    Sub‑grad it={it:,d}  score≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine builds many analytically‑motivated seeds,
    selects the most promising by the cheap surrogate, refines each with a
    continuation Adam optimiser (smooth max), a short mirror‑descent pass,
    and finally a global projected sub‑gradient polish.  The total running
    time never exceeds the supplied budget (default 1000 s).  If the search
    fails to improve a safe fallback uniform sequence is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # --------------------------------------------------------------
    # Use a previously known construction as an extra seed, if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Generate a rich pool of seeds for several lengths
    # --------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]   # ordered from largest to smallest
    all_top_seeds: List[Tuple[int, np.ndarray]] = []   # (n, vector)

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Scanning length {n} ===")

        seed_vectors: List[np.ndarray] = []

        # deterministic analytical families
        seed_vectors.append(arcsine_initial(n))
        seed_vectors.append(np.full(n, 1.0 / n))
        seed_vectors.append(generate_hann_shape(n))

        # inverse‑power sweep
        for p in np.linspace(0.15, 1.40, 12):
            seed_vectors.append(generate_inv_power_shape(n, p, eps=1e-8))

        # centred‑power sweep
        for p in np.linspace(0.15, 1.40, 12):
            seed_vectors.append(generate_centered_power_shape(n, p, eps=1e-8))

        # sin‑power sweep (captures raised‑cosine family)
        for q in np.linspace(0.5, 4.0, 10):
            seed_vectors.append(generate_sine_power_shape(n, q))

        # beta‑family grid (symmetric and mildly asymmetric)
        alphas = [0.2, 0.5, 0.8, 1.2, 1.6, 2.5, 4.0]
        betas = alphas
        for a in alphas:
            for b in betas:
                seed_vectors.append(generate_beta_shape(n, a, b, eps=0.0))

        # random Dirichlet draws
        for _ in range(8):
            seed_vectors.append(generate_random_dirichlet(n, rng))

        # ----------------------------------------------------------
        # 1b.  Rank by cheap surrogate and keep a few best
        # ----------------------------------------------------------
        scored = [(fast_score(vec), vec) for vec in seed_vectors]
        scored.sort(key=lambda x: x[0])
        top = [vec for _, vec in scored[:3]]   # keep up to three per length
        for vec in top:
            all_top_seeds.append((n, vec))
        print(f"  kept {len(top)} top seeds (estimated scores: {[round(s[0],6) for s in scored[:3]]})")

    # --------------------------------------------------------------
    # 2.  Refine each kept seed within an allocated time slice
    # --------------------------------------------------------------
    remaining = len(all_top_seeds)
    for idx, (n, seed_vec) in enumerate(all_top_seeds, start=1):
        if time.time() >= deadline - 3.0:
            break
        # allocate a slice respecting the remaining budget
        remain_time = deadline - time.time()
        per_seed = max(5.0, min(30.0, remain_time / (remaining * 1.5)))
        seed_deadline = time.time() + per_seed
        remaining -= 1

        print(f"\nSeed {idx}/{len(all_top_seeds)} (n={n}) – allocated ≈{per_seed:.1f}s")
        # initialise Adam via log‑softmax (avoids division by zero)
        z0 = np.log(seed_vec + 1e-12)

        # ------------------------------------------------------
        # 2a. Continuation Adam optimisation (smooth max)
        # ------------------------------------------------------
        a_opt = smooth_adam_optimize(
            z0,
            seed_deadline,
            n=n,
            beta0=5.0,
            beta_max=800.0,
            ramp_iters=120_000,
            lr=0.02,
        )

        # ------------------------------------------------------
        # 2b. Official evaluation of the Adam result
        # ------------------------------------------------------
        try:
            val = evaluate_sequence(a_opt.tolist())
            print(f"   Official score after Adam = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"   *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Evaluation failed (Adam result): {e}")

        # ------------------------------------------------------
        # 2c. Mirror‑descent refinement (exact max sub‑gradient)
        # ------------------------------------------------------
        if time.time() < deadline - 2.0:
            md_deadline = deadline - 0.5
            a_mirror = mirror_descent_optimize(
                np.array(best_seq, dtype=float),
                md_deadline,
                step0=0.16,
                decay=0.9985,
            )
            try:
                val2 = evaluate_sequence(a_mirror.tolist())
                print(f"   Mirror‑refined score = {val2:.6f}")
                if val2 < best_val:
                    best_val, best_seq = val2, a_mirror.tolist()
                    print(f"   *** Updated best after mirror step: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Mirror refinement evaluation failed: {e}")

    # --------------------------------------------------------------
    # 3.  Final global polishing with projected sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=300_000,
            step0=0.13,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing evaluation failed: {e}")

    # --------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
