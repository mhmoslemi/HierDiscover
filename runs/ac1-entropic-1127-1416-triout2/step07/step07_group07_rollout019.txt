def golden_section_search(func, a, b, tol=1e-4, max_iter=30):
    phi = (np.sqrt(5)-1)/2
    c = b - phi * (b-a)
    d = a + phi * (b-a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b-a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b-a)
            fd = func(d)
        if abs(b-a) < tol:
            break
    return (c if fc<fd else d), min(fc,fd)

"""
Search for a non‑negative step‑function that minimizes the
evaluation metric 2·n·max(conv) / (sum(a)^2).  The search combines:

* a parametric Beta(α,α) family (including the arcsine case α=½),
  explored with a golden‑section line‑search;
* Adam optimisation on a smooth soft‑max surrogate;
* exact sub‑gradient descent on the true max‑convolution;
* an incremental hill‑climber that transfers a tiny amount of mass
  between two positions while updating the convolution in O(n).

All operations stay on the probability simplex (sum=1, a_i ≥0)
so the cheap fast score `2·n·max(conv)` equals the official
`evaluate_sequence` value.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initialisation(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)

def random_sparse_vector(n: int) -> np.ndarray:
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a

def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    conv = convolution_fft(a, a)
    maxc = np.max(conv)
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)       # Σ w = 1
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad

def adam_softmax_descent(a0: np.ndarray, time_deadline: float,
                         max_iter: int = 4000,
                         step0: float = 0.2,
                         eps_factor: float = 0.3):
    a = a0.astype(float).copy()

        if cur_sc < best_sc:

            best_a = a_new.copy()
        a = a_new

def subgradient_descent(a0: np.ndarray, time_deadline: float,
                        max_iter: int = 120_000,
                        step0: float = 0.08):

def hill_climb_incremental(a_start: np.ndarray,
                           conv_start: np.ndarray,

        conv_new = conv.copy()

        conv_new[j:j + n] += 2.0 * delta * a

            conv_new[idx2i] += delta * delta
        idx2j = 2 * j
        if idx2j < conv_new.shape[0]:
            conv_new[idx2j] += delta * delta

        conv_new[idxij] -= 2.0 * delta * delta

        a_candidate[j] += delta

            conv = conv_new

            step_frac *= 0.995

def beta_shape(alpha: float, n: int) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the index set {0,…,n‑1},
    discretised and normalised to unit sum.
    """
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    eps = 1e-12
    x = (np.arange(n, dtype=float) + 0.5) / n          # points in (0,1)
    # w = (x * (1-x))**(alpha-1)  – protect against under‑/overflow
    base = x * (1.0 - x) + eps
    w = np.power(base, alpha - 1.0)
    total = np.sum(w)
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return w / total

def golden_section_search(func, a, b, tol=1e-5, max_iter=30):
    """Return (x_opt, f_opt) that approximately minimises func on [a,b]."""
    phi = (np.sqrt(5) - 1) / 2  # about 0.618
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of   f(a) = eps·log Σ_k exp(conv_k/eps)   with conv = a⋆a.
    Returns  grad = 2·(w_rev ⊛ a)  where w = softmax(conv/eps).
    """

    maxc = np.max(conv)

    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad

    try:

        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1)  Scan the 1‑parameter Beta family (α≈½ is the arcsine case)
    # ------------------------------------------------------------------
    def score_alpha(alpha):
        a = beta_shape(alpha, n)
        return compute_score(a)

    # Golden‑section search in a plausible interval around 0.5
    left, right = 0.30, 0.90
    opt_alpha, opt_sc = golden_section_search(score_alpha, left, right,
                                              tol=1e-4, max_iter=25)
    a0 = beta_shape(opt_alpha, n)
    cur_sc = compute_score(a0)
    if cur_sc < best_score:
        best_score = cur_sc
        best_seq = a0.tolist()
        print(f"[beta] α={opt_alpha:.5f}  score={cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam‑accelerated optimisation on the soft‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        # allocate roughly 30 gradient steps per second of remaining time
        iters = min(8000, int((deadline - time.time()) * 30))
        a_gd, sc_gd = adam_softmax_descent(
            a0,
            time_deadline=deadline - 2.0,
            max_iter=iters,
            step0=0.3,
            eps_factor=0.2,
        )
        if sc_gd < best_score:
            best_score = sc_gd
            best_seq = a_gd.tolist()
            print(f"[adam] improved score = {sc_gd:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact sub‑gradient descent (fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        a_sub, sc_sub = subgradient_descent(
            a_gd if best_seq == a_gd.tolist() else a0,
            time_deadline=deadline - 3.0,
            max_iter=120_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[sub‑grad] improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Massive incremental hill‑climbing – many cheap mass transfers
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_start = np.array(best_seq, dtype=float)
        conv_start = convolution_fft(a_start, a_start)

        # rough estimate: ~400 moves per second on a single core
        moves = int((deadline - time.time()) * 400)
        a_hc, sc_hc = hill_climb_incremental(
            a_start,
            conv_start,
            time_deadline=deadline - 2.0,
            max_steps=moves,
        )
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.tolist()
            print(f"[hill‑climb] improved score = {sc_hc:.6f}")

        # optional cheap symmetry averaging
        a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
        sym_sc = compute_score(a_sym)
        if sym_sc < best_score:
            best_score = sym_sc
            best_seq = a_sym.tolist()
            print(f"[sym avg] improved score = {sym_sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (guaranteed finite result)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 512
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
