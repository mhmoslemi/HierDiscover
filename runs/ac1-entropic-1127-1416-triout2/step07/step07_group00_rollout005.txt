def subgrad_optimize_opt(start_a, deadline, max_iters=500000, step_factor=1.0):

  a = start_a.copy()

  best_a = a.copy(); best_F = compute_score(a)

  for t in range(1, max_iters+1):

    if time.time() >= deadline: break

    conv = convolution_fft(a,a)

    Cmax = conv.max() ; idx = np.argmax(conv)

    F = 2*a.shape[0]*Cmax

    # compute subgrad as earlier
    k = idx

    idx_i = np.arange(a.shape[0])

    j = k - idx_i

    mask = (j>=0) & (j<a.shape[0])

    g = np.zeros_like(a)

    g[mask] = 2.0 * a[j[mask]]

    # compute step size
    denom = np.dot(g,g) + 1e-12

    # If not improve, step might be too small; we still make step
    step = step_factor * (F - best_F) / denom if F > best_F else step_factor * 0.001

    # ensure step positive
    step = max(step, 1e-5)

    a = project_to_simplex(a - step * g)

    # check per 500 iter

    if t % 500 == 0:

        sc = compute_score(a)

        if sc < best_F:

            best_F = sc; best_a = a.copy()

  return best_a, best_F

The program searches for a non‑negative step function (a probability vector) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the aperiodic convolution of `a` with itself.  
Because `F` is homogeneous of degree 0 we may restrict the search to the simplex
\(\sum a_i = 1\); on that set  

\[
F(a)=2n\;\max (a*a).
\]

The optimiser works in five stages:

1. **Analytic families** – a one‑parameter “α‑edge” family and a two‑parameter
   “β‑Beta” family are scanned on a fine grid.  The best vector on the simplex
   is kept as a starting point.

2. **Projected Adam on the smoothed objective** – we minimise the
   log‑sum‑exp approximation of the max,  

   \[
   f_\beta(a)=\frac1\beta\log\!\sum_k\exp\!\bigl(\beta\,(a*a)_k\bigr),
   \]

   with an exponentially increasing smoothing parameter `β`.  Adam’s
   adaptive steps are projected onto the simplex after each update.

3. **Polyak‑step sub‑gradient descent** – a classic sub‑gradient method on the
   true max‑objective, using the optimal step size
   \(\frac{F(a)-F_{\text{best}}}{\|g\|^2}\).

4. **Simulated‑annealing hill‑climb** – random mass‑transfer moves
   (or swaps) are accepted either if they improve the exact score or with a
   decreasing temperature, allowing occasional uphill steps.

5. **Final evaluation** – the best vector is passed to the supplied
   `evaluate_sequence` for the official score and returned as a plain
   Python list.

All heavy work is done with NumPy‑FFT (‑ `O(n log n)`) and the programme
adapts the vector length (`n = 8192` or `16384`) to the remaining time budget.
The total wall‑clock time is kept below the 1000 s limit.

Optimiser for the step‑function attack.

The algorithm proceeds through the following stages:

1️⃣  Analytic start – scan the one‑parameter edge family
    a_i ∝ [(i+½)(n‑½‑i)]^{‑α}

    max‑convolution using Adam with Euclidean projection onto the probability
    simplex.

3️⃣  Polyak‑step sub‑gradient descent on the true max‑objective.

4️⃣  Simulated‑annealing hill‑climb (mass‑transfer / swaps).

5️⃣  Return the best vector as a plain list of floats.

#  Basic linear‑algebra helpers (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using real FFT.
    Returns length len(a)+len(b)-1 array.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of 2·n·max(a*a) when sum(a)=1.
    This is the objective that evaluate_sequence would return (apart from
    clipping, which is irrelevant for our normalised vectors).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, ∑x = 1}.
    """
    if np.all(v <= 0):
        # zero vector → uniform simplex point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
#  Analytic families for the initial guess
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    One‑parameter family a_i ∝ [(i+½)(n‑½‑i)]^{‑α}.
    Returned vector lies on the simplex.
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # safety against overflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Two‑parameter Beta‑like family
    a_i ∝ (i+½)^{‑p}·(n‑½‑i)^{‑q}.
    Normalised to the simplex.
    """
    i = np.arange(n, dtype=float)
    # work in log‑space to avoid overflow/underflow
    log_w = -p * np.log(i + 0.5) - q * np.log(n - 0.5 - i)
    w = np.exp(log_w)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def random_sparse_start(n: int) -> np.ndarray:
    """
    Create a random sparse probability vector (k non‑zero entries).
    The sparsity level is chosen uniformly between 2 % and 50 % of n.
    """
    prop = np.random.uniform(0.02, 0.5)
    k = max(2, int(round(prop * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_dense_start(n: int) -> np.ndarray:
    """
    Dirichlet(α) start with α = 0.5 (moderately sparse).
    """
    a = np.random.gamma(0.5, 1.0, size=n)
    return project_to_simplex(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- β schedule (exponential increase) -----
        beta = beta_start * (beta_end / beta_start) ** (t / max_iters)

        # ----- smoothed objective weights (softmax) -----
        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        # stable softmax
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ----- gradient of smoothed objective -----
        a_rev = a[::-1]
        rev_conv = convolution_fft(a_rev, w)            # length 3n‑2
        grad = 2.0 * rev_conv[n - 1:n - 1 + n]          # shape (n,)

        # ----- Adam moments -----
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)

        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        # ----- update & projection -----
        step = step_size * m_hat / (np.sqrt(v_hat) + 1e-12)
        a = project_to_simplex(a - step)

        # ----- keep the best exact score -----
        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
#  Stage 3 – Polyak‑step sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 400_000,
                     step_factor: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method with Polyak step size.
    Returns (best_vector, best_exact_score).
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        idx = np.arange(n, dtype=int)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # current exact objective
        cur_val = 2.0 * n * conv.max()

        # Polyak step size (only if we have made progress already)
        denom = np.dot(g, g) + 1e-12
        if cur_val > best_score:
            step = step_factor * (cur_val - best_score) / denom
        else:
            # no improvement yet – take a tiny exploratory step
            step = step_factor * 1e-3 / np.sqrt(t)

        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
#  Stage 4 – Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Long stochastic hill‑climb with a simple temperature schedule.
    Mass‑transfer or swap moves are accepted if they improve the score
    or with probability exp(-(Δ)/T).  Returns the best vector seen.
    """
    a_curr = start_a.copy()
    n = a_curr.size
    cur_score = compute_score(a_curr)

    best_a = a_curr.copy()
    best_score = cur_score

    # linear cooling – start relatively hot, finish cold
    temp = 0.12
    decay = np.exp(np.log(1e-6 / temp) / max_steps)  # final temperature ~1e‑6

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        cand = a_curr.copy()

        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.1 * cand[i]      # move ~10 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a_curr = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        # cool down
        temp *= decay

    return best_a, best_score


# ----------------------------------------------------------------------
#  Main entry point required by the scoring harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows the four‑stage pipeline described above and respects
    the 1000 s wall‑clock limit.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0.  Try a fallback sequence if the global variable exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a workable length (bigger n gives a lower theoretical bound)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # If we have more than ~600 s after the fallback, we can afford n=16384
    n = 8192
    if remaining > 600:
        n = 16384
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Analytic families – coarse grid + a few random draws
    # ------------------------------------------------------------------
    # α‑grid
    alphas = np.linspace(0.0, 5.0, 51)   # step 0.1
    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        cand = alpha_start(n, alpha)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] new best after α={alpha:.2f} → {best_score:.6f}")

    # β‑grid (p,q)
    ps = np.linspace(0.0, 5.0, 11)      # step 0.5
    for p in ps:
        for q in ps:
            if time.time() >= deadline - 30.0:
                break
            cand = beta_family(n, p, q)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.tolist()
                print(f"[{time.time():.1f}] new best β(p,q)=({p:.2f},{q:.2f}) → {best_score:.6f}")

    # a few random (p,q) draws for extra coverage
    while time.time() < deadline - 30.0:
        p = np.random.uniform(0.0, 5.0)
        q = np.random.uniform(0.0, 5.0)
        cand = beta_family(n, p, q)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] random β(p,q)=({p:.2f},{q:.2f}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Random restarts with Adam (smooth log‑sum‑exp)
    # ------------------------------------------------------------------
    restart_deadline = deadline - 200.0
    while time.time() < restart_deadline:
        # initialise either from a sparse random vector or a small perturbation
        if np.random.rand() < 0.5:
            a0 = random_sparse_start(n)
        else:
            # perturb current best slightly
            a0 = random_dense_start(n) * 0.2 + np.array(best_seq) * 0.8
            a0 = project_to_simplex(a0)

        a_opt, sc_opt = adam_optimize(
            a0,
            deadline - 180.0,
            max_iters=250_000,
            beta_start=6.0,
            beta_end=400.0,
            step_size=0.25,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] Adam restart improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Sub‑gradient refinement (Polyak step)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 60.0,
            max_iters=300_000,
            step_factor=0.8,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_fin, sc_fin = hill_climb_sa(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=800_000,
        )
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Safeguard: if nothing succeeded, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7.  Official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
