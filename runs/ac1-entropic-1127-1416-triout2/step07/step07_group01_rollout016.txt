def propose_candidate(seed=12345,budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    n = 8192

    best_seq = None
    best_val = float('inf')
    # fallback if exists
    try:
        fallback = [float(x) for x in height_sequence_1]

            print(f'Fallback score {fv:.6f}')

        pass

    # define evaluate wrapper using compute_score to avoid repeated use of evaluate_sequence (faster)
    def score_vec(a):
        return compute_score(a)  # a is np.ndarray normalized sum=1

    # 1. parametric shape grid search
    print('Starting parametric shape search')
    ps = np.concatenate([np.linspace(-3.0, -0.1, 30), np.linspace(-0.1, 3.0, 30)])
    for p in ps:
        if time.time() >= deadline:
            break
        a = shape_vector(p, n)
        cur = score_vec(a)
        if cur < best_val:
            best_val = cur
            best_seq = a.tolist()
            print(f'New best shape p={p:.3f} score={cur:.6f}')

    # 2. random Dirichlet search
    print('Random Dirichlet search')
    while time.time() < deadline - 5.0:
        # batch size based on remaining time
        batch = 200
        for _ in range(batch):
            if time.time() >= deadline - 5.0:
                break
            # sample concentration parameter alpha from log-uniform
            log_alpha = np.random.uniform(-2, 2)  # alpha between 0.01 and 100
            alpha = 10**log_alpha

            # using Gamma distribution
            g = np.random.gamma(alpha, 1.0, size=n)
            a = g / g.sum()
            cur = score_vec(a)
            if cur < best_val:
                best_val = cur
                best_seq = a.tolist()
                print(f'New best Dirichlet alpha={alpha:.3f} score={cur:.6f}')
    # If have a best candidate, refine with smooth gradient
    if best_seq is not None and time.time() < deadline - 200.0:
        print('Smooth gradient refinement')
        a0 = np.array(best_seq)
        # use longer time for refinement
        time_limit = deadline - 100.0
        a_ref, score_ref = smooth_gradient_optimize(

            max_iters=800_000,

            beta_end=200.0,

            adam_beta1=0.9,
            adam_beta2=0.999,
            adam_eps=1e-8,
        )
        if score_ref < best_val:
            best_val = score_ref
            best_seq = a_ref.tolist()

    # subgradient polishing if time left

"""
Hybrid search for a non‑negative step‑function that minimises
`evaluate_sequence`.  The routine combines:

*   a cheap parametric sweep (U‑shaped / convex power‑law families);
*   massive random Dirichlet sampling;
*   a log‑sum‑exp (smooth max) Adam optimiser with a long β‑anneal;
*   projected sub‑gradient polishing on the exact max‑objective;
*   a final stochastic hill‑climb that transfers tiny amounts of mass.

All helpers are defined at top level – no closures – and the routine
adapts to the 1000 s wall‑clock limit.  The best vector found is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution using a real FFT.  Returns an array of length
    len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective when the vector already sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric family – power‑law U‑shapes
# ----------------------------------------------------------------------


def shape_vector(p: float, n: int) -> np.ndarray:
    """
    Symmetric power‑law shape.
    a_i ∝ ((i+1)*(n-i))**p ,  i=0,…,n‑1.
    p<0 gives a U‑shape (mass near the edges), p>0 a hill‑shape.
    Returned vector is already normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    vals = ((i + 1.0) * (n - i)) ** p
    # protect against under‑flow – fall back to uniform if everything vanished
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=float)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smoothed max‑objective optimisation (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 80.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the max‑convolution using Adam‑style
    momentum.  The best exact (non‑smooth) score encountered is returned.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smooth objective & weights --------------------------------
        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()              # probability vector

        # ---- gradient (cross‑correlation) -------------------------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]        # slice of length n

        # ---- Adam update -----------------------------------------------
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation of the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry (indices that contribute)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 1000 == 0:
            # tiny jitter to escape plateaus
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb polishing (small mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that swaps entries or transfers a
    small fraction of mass between two positions, always projecting
    back onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Helper for Dirichlet sampling
# ----------------------------------------------------------------------


def random_dirichlet(alpha: float, n: int) -> np.ndarray:
    """
    Sample a symmetric Dirichlet(alpha) on the simplex of dimension n.
    """
    g = np.random.gamma(alpha, 1.0, size=n)
    return g / g.sum()
