The program searches for a non‑negative step‑function (a probability vector) that minimises  

\[
\text{score}= \frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

where `*` denotes discrete convolution.  
The optimisation proceeds in four stages:

1. **Random restarts** – many independent starts are generated (dense or sparse) and
   refined by a smooth **log‑sum‑exp** approximation of the max‑objective.
   Adam‑style momentum and an annealing schedule for the smoothing parameter `β`
   are used.

2. **Exact sub‑gradient descent** – the true max‑objective is optimised with a
   projected sub‑gradient method (the gradient of the currently maximal
   convolution entry).

3. **Greedy max‑descent** – the current maximal convolution index `k` is
   identified, its exact sub‑gradient `g_i = a_{k-i}` is built and a small
   amount of probability mass is transferred from the index with the largest
   gradient component to the one with the smallest.  The step size is
   continuously adapted; this step directly reduces the offending maximal
   entry.

4. **Hill‑climb polishing** – stochastic pairwise swaps or tiny mass‑transfer
   moves are tried; any improvement is kept.

All stages respect the 1000 s wall‑clock limit.  The best sequence found is
returned as a plain Python list of floats.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""Optimise a non‑negative step‑function to minimise `evaluate_sequence`.

The algorithm uses several complementary phases:
  1. Random restarts + Adam‑accelerated log‑sum‑exp smoothing.
  2. Projected sub‑gradient descent on the exact max‑objective.
  3. Greedy max‑descent that transfers probability mass from the
     most‑responsible indices of the current maximal convolution entry.
  4. Stochastic hill‑climb polishing.

All phases are time‑aware and respect the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generator
# ----------------------------------------------------------------------


def random_start(n: int) -> np.ndarray:
    """Create a random probability vector (dense or sparse)."""
    if np.random.rand() < 0.6:                     # dense start
        a = np.random.rand(n)
        a = a / a.sum()
    else:                                          # sparse start
        # pick between 2% and 40% of the coordinates
        proportion = np.random.uniform(0.02, 0.4)
        k = max(2, int(round(proportion * n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
        a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# 1. Smoothed max‑objective optimisation (log‑sum‑exp) with Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 400.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a,a)) using Adam momentum.
    The routine keeps the best exact score encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed objective
        conv = convolution_fft(a, a)           # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()    # softmax weights

        # gradient of the weighted sum = 2 * (a_rev * w)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Sub‑gradient optimisation of the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to escape flat plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Greedy max‑descent (targeted mass transfer)
# ----------------------------------------------------------------------


def greedy_max_descent(
    init_a: np.ndarray,
    time_deadline: float,
    max_steps: int = 300_000,
    step_start: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of probability mass from the index that
    contributes most to the current maximal convolution entry to the index
    that contributes least.  The step size is adapted and the best vector
    encountered is returned.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    step = step_start
    no_improve = 0

    while max_steps > 0 and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        maxc = conv.max()
        k = int(np.argmax(conv))

        # Build the exact (unnormalised) sub‑gradient of the maximal entry
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        g = np.zeros_like(a)
        g[idx] = a[k - idx]                       # g_i = a_{k-i}

        # Choose source (largest g) and target (smallest g)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or a[src] < 1e-15:
            # nothing to move – break early
            break

        delta = step * a[src]                     # proportional move
        if delta > a[src]:
            delta = a[src]

        # Perform move
        a[src] -= delta
        a[dst] += delta

        # Occasionally re‑project to avoid numerical drift
        if max_steps % 1000 == 0:
            a = project_to_simplex(a)

        cur_score = compute_score(a)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            step = min(step * 1.05, 0.5)          # become bolder after improvement
            no_improve = 0
        else:
            # revert the move and shrink the step
            a[src] += delta
            a[dst] -= delta
            step *= 0.96
            no_improve += 1
            if no_improve >= 5000:                # stagnation guard
                break

        max_steps -= 1

    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Final hill‑climb polishing (small random moves)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline,
    max_steps: int = 500_000,
    step_start: float = 0.05,
) -> tuple[list[float], float]:
    """Stochastic hill‑climb that swaps entries or transfers a tiny fraction."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step = step_start
    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns the best list of floats found before the wall‑clock deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # Choose a reasonable length (larger gives more design freedom)
    # -------------------------------------------------------------
    n = 8192
    # if we have plenty of time we can try a longer vector
    if budget_s >= 900 and (deadline - time.time()) > 400:
        n = 16384
    print(f"Sequence length = {n}")

    # -------------------------------------------------------------
    # 1. Random restarts + Adam‑accelerated smooth optimisation
    # -------------------------------------------------------------
    # allocate roughly 40 % of the overall budget to this phase
    restart_deadline = start_time + 0.40 * budget_s
    while time.time() < restart_deadline:
        remaining = deadline - time.time()
        if remaining < 30:                     # reserve time for later phases
            break
        a0 = random_start(n)
        a_opt, scr = smooth_gradient_optimize(
            a0,
            time_deadline=restart_deadline,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=400.0,
            step0=0.9,
        )
        if scr < best_val:
            best_val = scr
            best_seq = a_opt.tolist()
            print(f"New best after smooth phase: {best_val:.6f}")

    # -------------------------------------------------------------
    # 2. Sub‑gradient refinement
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150:
        print("\nSub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 130,
            max_iter=350_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Improved by sub‑grad: {best_val:.6f}")

    # -------------------------------------------------------------
    # 3. Greedy max‑descent (targeted mass transfers)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70:
        print("\nGreedy max‑descent")
        gd_a, gd_score = greedy_max_descent(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 50,
            max_steps=250_000,
            step_start=0.10,
        )
        if gd_score < best_val:
            best_val = gd_score
            best_seq = gd_a.tolist()
            print(f"Greedy descent improved best to {best_val:.6f}")

    # -------------------------------------------------------------
    # 4. Final hill‑climb polishing
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("\nHill‑climb polishing")
        final_seq, final_score = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=700_000,
            step_start=0.05,
        )
        if final_score < best_val:
            best_val = final_score
            best_seq = final_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # -------------------------------------------------------------
    # Safeguard – fall back to a modest uniform vector if nothing found
    # -------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (uses the harness‑provided function)
    final_report = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_report:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
```
