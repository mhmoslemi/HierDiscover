"""
Propose a step‐function with a low evaluate_sequence score.

Algorithm
---------
1.  Grid search over a one‑parameter family
        a_i ∝ ((i+ε)(n‑1‑i+ε))^(‑p) ,   p∈[0,0.7]
    (p=0 → uniform, p≈0.5 → arcsine).  For each length n the
    best p (and a few post‑smoothings) is kept.

2.  Simulated‑annealing refinement:
    – start from the best vector found above,
    – repeatedly move a small random amount of mass from one index to
      another,
    – accept moves that improve the objective or, with probability
      exp(−Δ/T), worsening moves (temperature T is slowly lowered).

3.  Optional symmetry averaging (a←(a+rev(a))/2) and a final
    re‑normalisation.

All evaluations use the fast surrogate   score = 2·n·max(conv(a,a))
   which is exactly the value returned by evaluate_sequence when
   Σa = 1.  The full evaluator is called only once at the very end.
"""

import time
import numpy as np
from math import log, exp

# ----------------------------------------------------------------------
# FFT based convolution and cheap score (assumes sum(a)=1)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.shape[0] + b.shape[0] - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def _score_from_prob(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Parametric families of candidate vectors
# ----------------------------------------------------------------------


def _powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ ((i+ε)*(n‑1‑i+ε))^(‑p)   ,   p≥0.
    p=0 → uniform, p≈0.5 → discrete arcsine.
    """
    eps = 1e-12
    i = np.arange(n, dtype=float)
    w = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _smooth(a: np.ndarray, window: int = 3) -> np.ndarray:
    """Simple moving‑average smoothing (keeps the vector length)."""
    if window <= 1:
        return a.copy()
    kernel = np.ones(window, dtype=float) / float(window)
    # mode='same' pads with zeros – harmless because we renormalise anyway
    smoothed = np.convolve(a, kernel, mode='same')
    # eliminate tiny negatives caused by rounding
    smoothed = np.maximum(smoothed, 0.0)
    smoothed_sum = smoothed.sum()
    if smoothed_sum == 0:
        return np.full_like(a, 1.0 / a.shape[0])
    return smoothed / smoothed_sum


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _anneal_optimize(
    a_start: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the true objective.
    Moves a random fraction of mass from i → j; accepts worsening moves
    with probability exp(−Δ/T).  Temperature is geometrically cooled.
    Returns the best vector seen.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_score = _score_from_prob(a)
    best_a = a.copy()
    best_score = cur_score

    # annealing schedule
    T = 0.05                # initial temperature
    T_min = 1e-7
    cool = 0.9995           # cooling factor per iteration
    # step size (max fraction of source mass moved)
    step_frac = 0.05

    itr = 0
    while time.time() < deadline and itr < max_iters:
        itr += 1

        # pick distinct source and target indices
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] <= 1e-14:
            continue

        # propose moving a random fraction of the source mass
        delta = step_frac * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # renormalise only to counter floating‑point drift
        # (sum remains 1 analytically)
        cand_sum = cand.sum()
        if cand_sum != 1.0:
            cand /= cand_sum

        cand_score = _score_from_prob(cand)
        dE = cand_score - cur_score

        accept = False
        if dE < 0:
            accept = True
        else:
            # Metropolis criterion
            prob = exp(-dE / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[anneal] iter {itr:,}  new best {best_score:.6f}")

        # cool down
        T = max(T * cool, T_min)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------
