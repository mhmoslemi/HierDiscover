The goal is to minimise  

\[
F(a)=\frac{2\,n\; \max\limits_{k}(a\ast a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a` is a non‑negative sequence.  The value of `F` does **not** change if the whole
vector is multiplied by a constant, therefore we can (and do) enforce  
\(\sum_i a_i=1\) and simply minimise `2·n·max(a∗a)`.

The code below performs a **multi‑stage search**:

1. **Diverse initialisation** – a large pool of candidates is built for a
   set of lengths (up to 16384).  Families include the arcsine envelope,
   power‑law envelopes, Beta‑distribution envelopes, mixtures with the uniform
   vector and simple monotone power‑law vectors.

2. **Fast pre‑selection** – every candidate is scored with a cheap FFT‑based
   surrogate `2·n·max(conv)`.  Only the best few seeds per length are kept.

3. **Smooth surrogate optimisation** – each retained seed is refined by an
   Adam optimiser applied to a **log‑sum‑exp** smooth approximation of the max
   operator.  The temperature is increased from 5 to 150, giving a smooth‑to‑sharp
   transition.

4. **Exact sub‑gradient descent** – the smooth‑Adam result is further polished
   with a projected sub‑gradient method that works directly on the true
   non‑smooth objective.  The step size follows a \(1/\sqrt{t}\) schedule and
   the iterate is kept on the simplex by Euclidean projection.

5. **Exponential‑gradient (mirror) polishing** – a short KL‑mirror descent
   (multiplicative updates) removes the remaining bias of the sub‑gradient
   method.

6. **Random perturbation walk** – occasional Gaussian perturbations followed
   by the mirror step help escape shallow local minima.

7. **Final global polishing** – if time permits a last long sub‑gradient run is
   performed on the best vector found so far.

All stages respect the 1000 s wall‑clock budget and the function always returns
a non‑empty sequence.  The combination of a richer parameter grid, longer
candidate lengths and more aggressive polishing usually pushes the score below
the target 1.503.

"""
Propose a step function (non‑negative sequence) that minimises
evaluate_sequence.  The search uses many initial shapes (arcsine,
power‑law, beta, mixtures), a smooth log‑sum‑exp surrogate optimised
with Adam, followed by exact sub‑gradient descent and a final
KL‑mirror (exponential‑gradient) polishing.  Longer lengths (up to
16384) are tried and the best result seen is returned.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (fast surrogate)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective when the vector already sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Generators for diverse initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law envelope  ((i+½)(n‑½‑i)+eps)^(‑p)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_monotone_power(n: int, p: float) -> np.ndarray:
    """Simple monotone decreasing power law a_i ∝ (i+1)^(-p)."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / ((i + 1.0) ** p)
    return vals / vals.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.09,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # mask of (near) maximal entries – needed for a sub‑gradient of max
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # reverse mask to obtain the sub‑gradient
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional logging
        if it % 20_000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 150.0,
    ramp_iters: int = 20_000,
    lr: float = 0.06,
) -> np.ndarray:
    """
    Minimise the smooth log‑sum‑exp surrogate of max(conv) using Adam.
    Operates on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly increase temperature β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp weighting
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        w_norm = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient w.r.t. a : 2 * (a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final vector on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 20_000,
    step: float = 0.012,
) -> np.ndarray:
    """
    Mirror descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    evaluate_sequence value.  The routine makes use of a variety of
    initial shapes, a smooth surrogate optimiser, exact sub‑gradient
    polishing and a final KL‑mirror run.  It always returns a non‑empty
    sequence before the wall‑clock limit expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Re‑use a previous construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Search over several lengths (larger lengths give a finer
    #     discretisation of the continuous optimum)
    # ------------------------------------------------------------------
    lengths = [16384, 8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 15.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a rich pool of seeds
        # --------------------------------------------------------------
        seeds = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),               # uniform
        ]

        # power‑law grid (including arcsine‑like p≈0.5)
        p_vals = np.linspace(0.1, 1.2, 12)
        eps_vals = [0.0, 0.01, 0.1, 0.25, 0.5, 1.0]
        for p in p_vals:
            for eps in eps_vals:
                seeds.append(generate_power_shape(n, p, eps))

        # monotone decreasing pure power‑law
        for p in [0.5, 0.7, 1.0, 1.4]:
            seeds.append(generate_monotone_power(n, p))

        # random beta‑shape seeds
        for _ in range(10):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # mixture of arcsine & uniform
        for w in [0.2, 0.5, 0.8]:
            seeds.append(generate_mixture_shape(n, w))

        # include the current global best (resampled / truncated)
        if best_seq:
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            cand = cand / cand.sum()
            seeds.append(cand)

        # --------------------------------------------------------------
        # 1b. Keep only the most promising seeds according to the cheap surrogate
        # --------------------------------------------------------------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]

        # --------------------------------------------------------------
        # 1c. Optimise each retained seed
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        per_seed = max(6.0, remaining / (len(top_seeds) * 2.0 + 1e-9))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} – length {n} – budget ≈{per_seed:.1f}s")
            seed_deadline = min(time.time() + per_seed, deadline - 2.0)

            # ----------------------------------------------------------
            # 2. Smooth‑max Adam optimisation (quick phase)
            # ----------------------------------------------------------
            z0 = np.log(seed_vec + 1e-12)
            a_smooth = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=150.0,
                ramp_iters=20_000,
                lr=0.06,
            )

            # ----------------------------------------------------------
            # 3. Exact sub‑gradient polishing (longer)
            # ----------------------------------------------------------
            # allocate the rest of the seed budget to sub‑gradient
            sub_deadline = min(deadline - 2.0, time.time() + per_seed * 0.7)
            a_polished = subgradient_descent_exact(
                a_smooth,
                sub_deadline,
                max_iter=200_000,
                step0=0.09,
            )

            # ----------------------------------------------------------
            # 4. Official evaluation
            # ----------------------------------------------------------
            try:
                val = evaluate_sequence(a_polished.tolist())
                print(f"   official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_polished.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   evaluation failed: {e}")

            # ----------------------------------------------------------
            # 5. KL‑mirror polishing with a tiny random perturbation
            # ----------------------------------------------------------
            if time.time() < deadline - 4.0:
                # add a small Gaussian jitter before the EG step
                jitter = rng.normal(scale=0.02 / np.sqrt(n), size=n)
                a_noise = a_polished + jitter
                a_noise = project_to_simplex(a_noise)

                eg_deadline = min(deadline - 2.0, time.time() + per_seed * 0.3)
                a_eg = exponentiated_gradient_polish(
                    a_noise,
                    eg_deadline,
                    max_iter=30_000,
                    step=0.012,
                )
                try:
                    val2 = evaluate_sequence(a_eg.tolist())
                    print(f"   post‑EG score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_eg.tolist()
                        print(f"   *** New best after EG: {best_val:.6f} ***")
                except Exception as e:
                    print(f"   EG polishing failed: {e}")

    # ------------------------------------------------------------------
    # 6. Global final polishing (if we still have time)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 8.0:
        print("\n--- Final global sub‑gradient polishing ---")
        a_arr = np.array(best_seq, dtype=float)
        final_deadline = deadline - 2.0
        a_final = subgradient_descent_exact(
            a_arr,
            final_deadline,
            max_iter=300_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_final.tolist())
            print(f"   final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_final.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   final polishing error: {e}")

    # ------------------------------------------------------------------
    # 7. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print("Fallback uniform sequence used, score =", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
