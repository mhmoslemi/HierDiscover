def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    best_seq = None
    best_score = float('inf')
    best_official = float('inf')

    length_choices = [8192, 4096, 2048, 1024]

    for n in length_choices:
        if time.time() >= deadline - 5:
            break

        print(f"\n=== Length {n} (time left {deadline-time.time():.1f}s) ===")
        # generate seeds
        seeds = []
        # deterministic seeds
        seeds.append(arcsine_initial(n))
        seeds.append(np.full(n, 1.0 / n))
        # mixture
        seeds.append(generate_mixture_shape(n, weight_arc=0.7))
        # sweep beta family
        for alpha in np.linspace(0.3, 2.5, 7):
            seeds.append(generate_beta_shape(n, alpha, alpha, eps=0.0))
        # power shapes
        for p in np.linspace(0.2, 1.5, 7):
            seeds.append(generate_power_shape(n, p, eps=1e-6))
        # random Dirichlet
        for _ in range(10):
            a = rng.dirichlet(np.ones(n) * np.exp(rng.uniform(-2, 2))) # maybe varying concentration
            seeds.append(a)

        # keep top seeds by fast_score
        scored = [(fast_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top = [vec for _, vec in scored[:5]]
        # allocate time per seed
        per_seed_time = max(2.0, (deadline - time.time()) / (len(top) * 2))
        for idx, seed_vec in enumerate(top):
            if time.time() >= deadline - 2:
                break
            print(f" Seed {idx+1}/{len(top)} - time budget {per_seed_time:.1f}s")
            seed_deadline = time.time() + per_seed_time

            # Prepare initial logit vector for softmax
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # Run smooth_adam
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=2000.0,
                ramp_iters=200_000,
                lr=0.03,
            )
            # cheap score
            sc = fast_score(a_opt)
            print(f"   Fast_score after Adam: {sc:.6f}")

            # evaluate official
            try:
                off = evaluate_sequence(a_opt.tolist())
                print(f"   Official score: {off:.6f}")
                if off < best_official:
                    best_official = off
                    best_seq = a_opt.tolist()
                    best_score = sc
                    print(f"   *** New best! ***")
            except Exception as e:
                print(f"   evaluation error: {e}")

            # Mirror descent refinement if time left
            if time.time() < deadline - 2:
                md_deadline = deadline - 0.5
                a_mirror = mirror_descent_optimize(
                    np.array(best_seq, dtype=float),
                    md_deadline,
                    step0=0.15,
                    decay=0.9999,
                )
                try:
                    off_m = evaluate_sequence(a_mirror.tolist())
                    print(f"   Mirror refined official: {off_m:.6f}")
                    if off_m < best_official:
                        best_official = off_m
                        best_seq = a_mirror.tolist()
                        best_score = fast_score(a_mirror)
                        print("   *** Updated best after mirror ***")
                except Exception:
                    pass

    # Final global polishing or random perturbation if time permits
    if best_seq and time.time() < deadline - 1:
        print("\nFinal polishing with subgradient descent...")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(a_arr, deadline, max_iter=300_000, step0=0.05)
        try:
            off = evaluate_sequence(a_arr.tolist())
            print(f"   Final score after subgradient: {off:.6f}")
            if off < best_official:
                best_official = off
                best_seq = a_arr.tolist()
                best_score = fast_score(a_arr)
        except Exception as e:
            print(f"   final evaluation error: {e}")

    # If still nothing, fallback uniform
    if not best_seq:
        n0 = 512
        best_seq = [1.0/n0]*n0
        best_official = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official {best_official:.6f}")
    return best_seq

"""
Search for a non‑negative step sequence that minimises
``evaluate_sequence``.

The algorithm proceeds in several stages:

1.  Choose a set of sequence lengths (up to 8192).  Larger n gives a
    lower theoretical bound because the average convolution value is
    1/(2n‑1).

2.  For each length generate a diverse pool of “seed” vectors:
    - arcsine‑type shape,
    - uniform,
    - mixtures of arcsine and uniform,
    - Beta(α,β) families,
    - power‑law families,
    - a handful of random Dirichlet draws.

3.  Keep the most promising seeds according to the cheap surrogate
    ``fast_score`` (the exact objective without clipping).

4.  Refine every retained seed with a continuation Adam optimiser that
    minimises the smooth‐max ``log‑sum‑exp(β·conv)`` while staying on the
    probability simplex via a soft‑max parametrisation.  The temperature
    β is annealed from 5 to 2000.

5.  After the Adam phase perform a *mirror‑descent* (exponential‑gradient)
    step that uses the exact sub‑gradient of the true max‑convolution.

6.  The best candidate found so far is polished with a long projected
    sub‑gradient descent (the exact objective).

7.  Finally, a cheap random local‑search (pairwise mass transfers) explores
    the neighbourhood of the polished solution.

All phases respect the overall time budget (default 1000 s).  The
function returns a list of floats that is always a valid, non‑negative
step function.  If nothing is found a uniform fallback is used.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate used for pre‑selection (scale‑invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv(a,a))/sum(a)^2 without clipping.
    ``a`` must be non‑negative.  The function is cheap (FFT) and used as
    a proxy for the true ``evaluate_sequence``.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic generators for diverse starting shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Shape proportional to 1/((i+½)*(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with random concentration)."""
    # Random concentration parameter between 0.1 and 5
    conc = 10 ** rng.uniform(-2.0, 0.7)
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx_i = 1, x_i ≥ 0."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent for f(a)=2·n·max(conv(a,a))/sum(a)^2."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance to capture all max entries (possibly multiple)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution (see notes)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    Sub‑grad it={it:,d}  surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) with Adam optimiser (continuation on β)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 2000.0,
    ramp_iters: int = 150_000,
    lr: float = 0.03,
) -> np.ndarray:
    """
    Minimises the smooth approximation L_β(a)=log‑sum‑exp(β·conv(a,a)),
    enforcing a∈simplex via the soft‑max transform.  β is increased
    linearly from ``beta0`` to ``beta_max`` (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of log‑sum‑exp w.r.t. a:
        # d/da Σ w_i * (a★e_i) = 2·(a ★ w_rev)   (★ = convolution)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 8_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) with exact max‑subgradient
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.15,
    decay: float = 0.9999,
) -> np.ndarray:
    """
    Exponential‑gradient (mirror) descent on the exact max‑convolution
    objective.  The iterate stays on the simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative KL‑mirror step
        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        # occasional progress report
        if int(step * 1e6) % 200_000 == 0:
            print(f"    Mirror step={step:.6f} surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Simple random local search around a candidate
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.03,
) -> np.ndarray:
    """
    Perform inexpensive pairwise mass‑transfer moves while the time budget
    permits.  Each move selects two indices i≠j and transfers a random
    fraction (up to ``max_transfer``) of the mass at i to j, then
    re‑normalises.  Moves are kept only if they improve ``fast_score``.
    """
    a = a0.copy()
    best_score = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # absolute amount to move (relative to a[i])
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        # re‑normalise (numerical stability)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            a[j] -= eps
            a[i] += eps
            a /= a.sum()   # restore sum=1 (tiny rounding errors)
    return a
