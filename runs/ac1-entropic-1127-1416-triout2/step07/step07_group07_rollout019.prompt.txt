# Utility functions – FFT convolution and simplex projection

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    self‑convolution.

def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f(a) = eps·log Σ_k exp(conv_k/eps)
    where conv = a⋆a.

    Returns  grad = 2 * (w_rev ⊛ a),
    with w = softmax(conv/eps).
    """

    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)                 # Σ w = 1

    # indices n‑1 … 2n‑2 correspond to shift i
    start = a.shape[0] - 1

    return grad

def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber that only accepts strictly improving moves.
    Moves:
        * swap two entries,
        * transfer a fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # transfer size adapts during the run
    step_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # numerical safeguard
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # become bolder
        else:
            step_frac *= 0.96                       # become more cautious

        max_steps -= 1

    return best_a, best_sc
