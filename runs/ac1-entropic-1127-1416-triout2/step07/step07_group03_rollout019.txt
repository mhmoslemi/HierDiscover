def solve_lp_step(f):
    n = len(f)
    M = np.max(convolution_fft(f, f))
    # Build A_ub and b_ub
    data = [np.full(n, f[d]) for d in range(n)]   # each is size n
    offsets = list(range(n))
    A_ub = scipy.sparse.diags(data, offsets, shape=(2*n-1, n), format='csc')
    b_ub = np.full(2*n-1, M)
    # solve: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback to current f
        return f
    g = res.x
    # Normalize sum to 1 (if sum >0)
    if np.sum(g) > 0:
        g = g / np.sum(g)
    else:
        g = f
    return g

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def convolution_fft(a, b):
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a):
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def lp_improve(f, max_iter=5, max_time=None):
    """
    Perform up to max_iter LP improvement steps starting from f.
    Stops early if time exceeds max_time.
    Returns improved vector.
    """
    start = time.time()
    n = f.size
    a = f.copy()
    for i in range(max_iter):
        if max_time is not None and time.time() - start > max_time:
            break
        # compute current max convolution
        conv = convolution_fft(a, a)
        M = float(np.max(conv))
        # Build constraints
        # A_ub shape (2n-1, n)
        # diagonal data
        data = [np.full(n, a[d]) for d in range(n)]
        offsets = list(range(n))
        A_ub = sparse.diags(data, offsets, shape=(2*n-1, n), format='csc')
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)  # maximize sum(g)
        bounds = [(0, None) for _ in range(n)]
        # Solve LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # if fails, break
            break
        g = res.x
        s = g.sum()
        if s <= 0:
            # fallback
            break
        # Normalize to sum 1
        g = g / s
        # Convex combination with a
        t = 0.2  # mixing factor, might adapt
        a = (1 - t) * a + t * g
        # Renormalize to sum 1
        a = a / a.sum()
    return a

def exponentiated_gradient_optimize(a0, deadline, max_iter=200_000, step0=0.1, jitter_interval=10000, jitter_scale=1e-3):
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.shape[0]
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)
        # reverse mask for convolution
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[2*n-2 - np.arange(n)]  # careful compute index
        # step schedule
        step = step0 / np.sqrt(it)
        a = a * np.exp(-step * subgrad)
        # small jitter to escape local minima
        if jitter_interval and (it % jitter_interval == 0):
            jitter = np.random.default_rng().normal(scale=jitter_scale, size=n)
            a = np.maximum(a + jitter, 0.0)
        # re-normalize
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        # optionally print progress
        if it % 20000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")
    return a

"""
Optimise a non‑negative step function (a finite sequence) for the
official ``evaluate_sequence`` objective.

Main ideas
----------

* **Diverse initial shapes** – arcsine envelope, uniform, decreasing
  power‑law, beta‑family and mixtures.  Every seed is normalised to
  Σa = 1 (the objective is scale‑invariant).

* **LP improvement** – for a current vector ``f`` compute
  ``M = max(conv(f,f))``.  Solve the linear programme

        maximise   Σ g_j
        subject to   g ≥ 0
                     conv(f,g) ≤ M                (2n‑1 linear constraints)

  The optimiser (SciPy‑``highs``) returns a feasible ``g`` with a larger
  ℓ₁‑norm while keeping the convolution bound.  Mixing a small amount
  of ``g`` into ``f`` (convex combination) provably does not increase
  the maximal convolution and often reduces it.

* **Entropic mirror‑descent (exponentiated‑gradient)** – after a few
  LP steps the non‑smooth problem is tackled directly on the simplex.
  The sub‑gradient of ``max(conv)`` is obtained from the set of active
  (maximal) convolution entries.  We use a decreasing step size
  ``η_t = η₀ / sqrt(t)`` and occasional tiny random jitters to escape
  shallow basins.

* **Exact projected sub‑gradient polishing** – a final pass with the
  classic sub‑gradient iteration (the routine used in the reference
  solution) squeezes any remaining improvement.

The whole search respects the 1000 s wall‑clock limit, always returns
a non‑empty sequence and keeps the best candidate seen across all
lengths.  Larger lengths (4096) are tried first because they provide a
larger optimisation space and usually lead to a smaller bound.
"""

import time
from typing import List

import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT, no circular wrap)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official objective for a vector that already
    satisfies Σa = 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Generators for diverse initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, Σ = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid, Σ = 1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape, Σ = 1."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_decreasing_power(n: int, p: float, eps: float) -> np.ndarray:
    """Monotonically decreasing power‑law (i+eps)^‑p, Σ = 1."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / ((i + eps) ** p + 1e-30)
    vals /= vals.sum()
    return vals


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform, Σ = 1."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def _lp_improve(f: np.ndarray, max_iter: int = 8,
                max_time: float | None = None) -> np.ndarray:
    """
    Perform a few LP steps starting from ``f``.
    Each step solves:
        maximise Σ g_j
        subject to g ≥ 0  and  conv(f,g) ≤ M,
    where M = max(conv(f,f)).
    The new vector is a convex combination  (1‑t)·f + t·g
    with a small mixing factor t.
    """
    start = time.time()
    n = f.size
    a = f.copy()

    for _ in range(max_iter):
        if max_time is not None and time.time() - start > max_time:
            break

        # current bound
        M = float(np.max(_convolution_fft(a, a)))

        # build the (2n‑1) × n constraint matrix:
        #   A_{r,j} = a_{r-j}  for 0 ≤ r-j < n, else 0.
        # Only offsets 0 … n‑1 appear, each as a constant diagonal.
        data = [np.full(n, a[d]) for d in range(n)]
        offsets = list(range(n))
        A_ub = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
        b_ub = np.full(2 * n - 1, M)

        # maximise Σ g   ==  minimise -Σ g
        c = -np.ones(n)
        bounds = [(0.0, None) for _ in range(n)]

        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0 or not np.isfinite(s):
            break
        g = g / s

        # convex mixing – a small step is enough to keep the bound non‑increasing
        t = 0.12
        a = (1.0 - t) * a + t * g
        a = a / a.sum()   # re‑normalise to Σ = 1

    return a


# ----------------------------------------------------------------------
# Entropic mirror descent (exponentiated‑gradient) on the true objective
# ----------------------------------------------------------------------


def _exp_grad_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
    jitter_interval: int = 25_000,
    jitter_scale: float = 1e-4,
) -> np.ndarray:
    """
    Mirror‑descent (KL) optimisation of the exact max‑convolution objective.
    Uses decreasing step size η_t = step0 / sqrt(t) and occasional random
    jitter to escape plateaus.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.shape[0]
    L = 2 * n - 1
    rng = np.random.default_rng(12345)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _convolution_fft(a, a)
        mx = float(np.max(conv))

        # tolerance for the set of active (maximal) indices
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = _convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # diminishing step size
        eta = step0 / np.sqrt(it)

        a *= np.exp(-eta * subgrad)

        # occasional tiny random perturbation (helps escape flat regions)
        if jitter_interval and (it % jitter_interval == 0):
            jitter = rng.normal(scale=jitter_scale, size=n)
            a = np.maximum(a + jitter, 0.0)

        # project back onto the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if it % 40_000 == 0:
            print(f"    EG it={it:,d}  score={_fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def _subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.05,
) -> np.ndarray:
    """
    Classic projected sub‑gradient descent on
        φ(a) = 2·n·max(conv(a,a))
    The iterate stays on the probability simplex.
    """
    a = init_a.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = _convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        # Euclidean projection onto the simplex
        if np.any(a < 0):
            u = np.sort(a)[::-1]
            cssv = np.cumsum(u) - 1
            rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
            if rho.size == 0:
                theta = 0.0
            else:
                rho = rho[-1]
                theta = cssv[rho] / (rho + 1)
            a = np.maximum(a - theta, 0.0)

        # renormalise (numerical drift)
        a = a / a.sum()

        if it % 30_000 == 0:
            print(f"    SubGrad it={it:,d}  score={_fast_score(a):.6f}")

    return a

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock bound (≈1000 s) and always
    returns a non‑empty list of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0.  Try the previously known construction if it exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Optimise several lengths, biggest first
    # --------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # -----------------------------
        # 1a. Build a diverse pool of seeds
        # -----------------------------
        seeds: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
        ]

        # random decreasing power‑law seeds
        for _ in range(5):
            p = rng.uniform(0.3, 4.0)
            eps = rng.uniform(0.0, 0.5)
            seeds.append(generate_decreasing_power(n, p, eps))

        # random beta‑shape seeds
        for _ in range(5):
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # mixture of arcsine & uniform
        for _ in range(2):
            w = rng.uniform(0.0, 1.0)
            seeds.append(generate_mixture_shape(n, w))

        # include the current global best (rescaled / tiled)
        if best_seq:
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            cand = np.clip(cand, 1e-12, None)
            cand = cand / cand.sum()
            seeds.append(cand)

        # -----------------------------
        # 1b. Keep only the most promising seeds (fast surrogate)
        # -----------------------------
        scored = [(float(_fast_score(s)), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:3]]   # at most three per length

        # -----------------------------
        # 1c. Allocate a time slice per seed and run the optimisation pipeline
        # -----------------------------
        remaining = deadline - time.time()
        per_seed = max(6.0, remaining / (len(top_seeds) * 2.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 3.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed:.1f}s")
            seed_deadline = min(time.time() + per_seed, deadline - 1.0)

            # ---- LP improvement phase (quick, few iterations) ----
            lp_time = (seed_deadline - time.time()) * 0.35
            a = seed_vec.copy()
            if lp_time > 0.8:
                a = _lp_improve(a, max_iter=10, max_time=lp_time)

            # ---- Exponentiated‑gradient polishing on the exact objective ----
            a = _exp_grad_optimize(a, seed_deadline,
                                   max_iter=250_000,
                                   step0=0.06,
                                   jitter_interval=30_000,
                                   jitter_scale=5e-5)

            # ---- Official evaluation ----
            try:
                val = evaluate_sequence(a.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

    # --------------------------------------------------------------
    # 2. Final global polishing using the exact sub‑gradient method
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = _subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=150_000,
            step0=0.04,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # --------------------------------------------------------------
    # 3. Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
