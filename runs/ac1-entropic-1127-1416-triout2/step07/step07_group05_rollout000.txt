def propose_candidate(seed=42, budget_s=1000):

    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # fallback best
    best_seq = []
    best_val = inf

    # function to evaluate official
    def try_update(seq):
        nonlocal best_seq, best_val
        try:
            val = evaluate_sequence(seq)
        except Exception as e:
            print("Exception in evaluate:", e)
            return
        print(f"Official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print("*** New best ***")

    # 0 fallback
    if 'height_sequence_1' in globals():
        try:
            fallback = [float(x) for x in globals()['height_sequence_1']]
            try_update(fallback)
        except Exception:
            pass

    # generate initial vector: either fallback normalized to sum=1 or arcsine
    if best_seq:
        init_vec = np.array(best_seq, dtype=float)
        init_vec = init_vec / init_vec.sum()
    else:
        init_n = 1024
        init_vec = arcsine_initial(init_n)

    # Run initial Adam on init vector (n = init_n)
    max_iters = 120_000
    if time.time() < deadline - 10:
        print("Initial Adam on n=", init_n)
        max_iters = int(min(max_iters, (deadline - time.time()) * 30))
        a_opt = adam_optimize_smooth_max(init_vec, deadline, max_iters=max_iters, tau_end=150.0, base_lr=0.2)
        init_vec = a_opt

    # hierarchical upscaling
    for target_n in (2048, 4096, 8192):
        if time.time() >= deadline - 10:
            break
        print(f"\nUpscaling to {target_n}")
        # interpolation
        up = upscale_vector(init_vec, target_n)
        # add jitter
        rng = np.random.default_rng(seed + target_n)
        jitter = rng.normal(scale=1e-4, size=target_n)
        up = up + jitter
        up = np.clip(up, 0.0, None)
        up = up / up.sum()

        # allocate iterations proportionally
        est_iters = int(min(120_000, (deadline - time.time()) * 25))
        if est_iters < 5000:
            continue
        a_opt = adam_optimize_smooth_max(up, deadline, max_iters=est_iters, tau_end=250.0, base_lr=0.18)
        init_vec = a_opt

        # after each upscaling, compute surrogate score
        surrogate = compute_score(init_vec)
        print(f"Surrogate after {target_n} = {surrogate:.6f}")

        # optional early evaluation (fast) maybe compute official for moderate n 2048 or 4096 only if time.
        if target_n <= 4096 and time.time() < deadline - 5:
            try_update(init_vec.tolist())

    # final refinement using subgradient + EG
    if time.time() < deadline - 5:
        print("\nFinal exact refinement")
        # subgradient descent
        max_iter_sub = int(min(200_000, (deadline - time.time()) * 25))
        refined = subgradient_descent_exact(init_vec, deadline, max_iter=max_iter_sub, step0=0.08)
        # EG
        max_iter_eg = int(min(150_000, (deadline - time.time()) * 25))
        refined = eg_refine(refined, deadline, max_iter=max_iter_eg, eta0=0.16)
        try_update(refined.tolist())
        init_vec = refined

    # final ensure non-empty sequence
    if not best_seq:
        n_fallback = 1024
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished. Best official score = {best_val:.6f}")
    return best_seq

"""
Propose a candidate step‑function.

The algorithm follows a simple multi‑scale scheme:

0️⃣  Load a fallback solution if available.
1️⃣  Start from a symmetric arcsine shape (known good heuristic) or the
    fallback, normalised to sum = 1.
2️⃣  Run a long Adam optimisation on the **smooth‑max** surrogate
    (log‑sum‑exp of the self‑convolution).  The temperature τ is raised
    exponentially, the logits are kept symmetric (w ← (w+w[::-1])/2) and the
    vector stays on the probability simplex via a soft‑max parametrisation.
3️⃣  Upscale the current best vector successively to larger lengths
    (2048 → 4096 → 8192).  After each up‑sampling a small random jitter is
    added, then Adam is run again with more aggressive τ‑end.
4️⃣  Perform a high‑precision refinement of the *exact* objective
    (max‑convolution) using a projected sub‑gradient method followed by an
    exponentiated‑gradient (mirror‑descent) polish.  After every update the
    iterate is symmetrised (a ← (a+a[::-1])/2) and projected back onto the
    simplex.
5️⃣  The best sequence (according to ``evaluate_sequence``) seen during the
    run is stored and finally returned.  All stages respect the global
    1000 s time limit.

The combination of a good initial shape, aggressive smooth‑max annealing,
symmetry enforcement and a final exact polish yields scores better than
the reference 1.5119 and aims for the target < 1.503.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (length = len(a)+len(b)-1).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score when the vector is normalised to sum=1.
    The surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators (used for seeding)

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(v) to target_n, followed by renormalisation."""
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 30_000,
    tau_start: float = 0.5,
    tau_end: float = 100.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k)

    Variables are logits w with a = softmax(w); symmetry is enforced by
    symmetrising the logits after each update (w ← (w+w[::-1])/2).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # Initialise logits from a (add tiny offset to avoid log(0))
    w = np.log(a_init + 1e-12)

    # Adam state
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # a = softmax(w) – kept symmetric implicitly by symmetrising w later
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()  # shape (L,)

        # gradient wrt a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wr.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry of logits (keeps a symmetric as well)
        w = (w + w[::-1]) * 0.5

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    # final symmetrisation (numerical safety)
    a = (a + a[::-1]) * 0.5
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    The iterate is symmetrised after each step.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) restricted to original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # enforce symmetry
        a = (a + a[::-1]) * 0.5

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    Symmetry is enforced after each update.
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        # enforce symmetry
        a = (a + a[::-1]) * 0.5

    return a
