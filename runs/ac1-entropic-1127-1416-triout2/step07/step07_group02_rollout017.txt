def hill_climb_refine(a, deadline, max_moves=5000, step_factor=0.2):

    best_val = evaluate_sequence(a.tolist())

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # small delta from i to j
        delta = step_factor * a[i]
        if delta < eps:
            continue
        a[i] -= delta
        a[j] += delta
        # Ensure nonneg
        if a[i] < -eps:
            a[i] += delta
            a[j] -= delta
            continue
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            # keep change
        else:
            # revert
            a[i] += delta
            a[j] -= delta
        max_moves -= 1
    return a, best_val

def lp_update(f, M, t=0.5):
    n = len(f)
    L = 2*n - 1
    # Build A as described
    A = np.zeros((L, n))
    for j in range(n):
        A[j:j+n, j] = f
    # set bounds
    bounds = [(0, None) for _ in range(n)]
    # solve
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(L, M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return original f
        return f
    b = res.x
    if b.sum() == 0:
        return f
    g = b / b.sum()
    new_f = (1 - t) * f + t * g
    # ensure nonnegativity and sum=1
    new_f = np.maximum(new_f, 0)
    s = new_f.sum()
    if s > 0:
        new_f /= s
    else:
        new_f = np.full_like(new_f, 1.0 / n)
    return new_f

"""
Propose a low‑value step sequence for the ``evaluate_sequence`` objective.

The algorithm mixes several ideas:
* A wide variety of (mostly asymmetric) analytical seeds (exponential,
  one‑sided power, Dirichlet draws, mixtures) are generated for each
  candidate length.
* Each seed is refined with a continuation Adam optimiser that minimises
  a smooth approximation of ``max(conv(a,a))`` (log‑sum‑exp with a
  linearly increasing temperature β).  The simplex constraint is imposed
  by a soft‑max re‑parameterisation.
* The best sequence found is polished with a projected sub‑gradient
  descent that uses the exact max‑convolution sub‑gradient.
* Finally a small number of linear‑programming refinement steps are
  performed (maximising the ℓ₁‑mass under the current convolution bound)
  which further reduces the objective.
All steps respect a global time budget (default 1000 s).  The routine
always returns a non‑empty, non‑negative sequence.
"""

import time
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate used for quick pre‑selection (scale‑invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Cheap surrogate: 2·n·max(conv(a,a))/sum(a)^2 (no clipping)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic initialisers (mostly asymmetric)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_one_sided_power(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Monotonically decreasing power law, normalised to sum 1."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / ((i + 0.5 + eps) ** p)
    return vals / vals.sum()


def generate_one_sided_exponential(n: int, rate: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-rate·i), normalised."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-rate * i)
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return vals / s


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of the symmetric arcsine and the uniform distribution."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration alpha."""
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Soft‑max utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns probabilities that sum to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth‑max + Adam optimiser (continuation on β)
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 800.0,
    ramp_iters: int = 80_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimise the smooth approximation
        L_β(a) = log‑sum‑exp(β·conv(a,a))

    ``z`` are unconstrained log‑probabilities; ``a = softmax(z)`` lives on
    the probability simplex.  β is increased linearly (continuation) to
    tighten the approximation.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # guarantees a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights (log‑sum‑exp trick)
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a  : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)               # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]          # slice of length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f}   surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for final polishing)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 180_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective
        f(a) = 2·n·max(conv(a,a)) / (sum a)^2 .
    The sub‑gradient of the max‑convolution is computed exactly.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    Sub‑grad it={it:,d}  surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# LP refinement (maximise ℓ₁‑mass under the current convolution bound)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, t: float = 0.5) -> np.ndarray:
    """
    One linear‑programming refinement step.
    Maximise sum(b) subject to  b ≥ 0  and  conv(f,b) ≤ M (elementwise).
    The resulting b is mixed with f:  f ← (1‑t)·f + t·(b / sum(b)).
    Returns the new f (still normalised to sum 1).  If the LP fails,
    ``f`` is returned unchanged.
    """
    n = f.size
    L = 2 * n - 1

    # Build convolution matrix A (size L × n) such that A @ b = conv(f,b)
    A = np.zeros((L, n), dtype=float)
    for j in range(n):
        A[j : j + n, j] = f  # column j = f shifted by j

    # Linear programme: maximise 1ᵀb  ⇔ minimise −1ᵀb
    c = -np.ones(n, dtype=float)
    b_ub = np.full(L, M, dtype=float)

    # Bounds: b_i ≥ 0
    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return f

    if not res.success:
        return f

    b = res.x
    s = b.sum()
    if s <= 1e-12:
        return f

    g = b / s                     # normalise to sum 1
    new_f = (1.0 - t) * f + t * g
    new_f = np.maximum(new_f, 0.0)
    new_f_sum = new_f.sum()
    if new_f_sum > 0.0:
        new_f /= new_f_sum
    else:
        new_f = np.full_like(new_f, 1.0 / n)
    return new_f


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (purely optional)
# ----------------------------------------------------------------------


def hill_climb_refine(a: np.ndarray, deadline: float, rng: np.random.Generator,
                     max_moves: int = 4000, step_factor: float = 0.05) -> Tuple[np.ndarray, float]:
    """
    Randomly move a small amount of mass between two coordinates and keep
    the move if it reduces the true objective.  Very cheap but sometimes
    squeezes out a tiny extra improvement.
    Returns the (possibly) improved array and its official score.
    """
    best_arr = a.copy()
    try:
        best_val = evaluate_sequence(best_arr.tolist())
    except Exception:
        best_val = float("inf")

    n = a.size
    eps = 1e-12

    moves = 0
    while time.time() < deadline and moves < max_moves:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best_arr[i] < eps:
            continue
        delta = step_factor * best_arr[i]
        if delta < eps:
            continue

        candidate = best_arr.copy()
        candidate[i] -= delta
        candidate[j] += delta
        if candidate[i] < -1e-12:
            continue

        try:
            val = evaluate_sequence(candidate.tolist())
        except Exception:
            continue

        if val < best_val:
            best_arr = candidate
            best_val = val
        moves += 1

    return best_arr, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The method combines:
    * a diverse pool of analytical seeds (including asymmetric shapes),
    * a continuation Adam optimiser on a smooth max‑approximation,
    * a final exact projected sub‑gradient polish,
    * a short series of LP refinement steps.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # a tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # --------------------------------------------------------------
    # Optional fallback to a previously discovered solution
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # --------------------------------------------------------------
    # Try several lengths (larger first – they usually give slightly
    # better asymptotic constants)
    # --------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ----------- generate a diverse set of seeds -----------------
        seed_vectors: List[np.ndarray] = [
            arcsine_initial(n),                               # symmetric baseline
            np.full(n, 1.0 / n),                              # uniform
            generate_mixture_shape(n, weight_arc=0.4),        # mix uniform+arcsine
            generate_one_sided_power(n, p=0.7),               # decreasing power law
            generate_one_sided_exponential(n, rate=0.0015),   # exponential decay
        ]

        # a few random Dirichlet draws (asymmetric)
        for _ in range(5):
            alpha = rng.uniform(0.1, 2.0)
            seed_vectors.append(generate_dirichlet_seed(n, alpha, rng))

        # keep only the most promising according to the cheap surrogate
        scored = [(fast_score(vec), vec) for vec in seed_vectors]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:4]]   # at most four per length

        # ----------- allocate a slice of the remaining budget per seed
        per_seed_budget = max(2.0,
                              (deadline - time.time()) / (len(top_seeds) * 3))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 3.0:
                break

            print(f"\n  Seed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_budget:.1f}s")
            seed_deadline = time.time() + per_seed_budget

            # initialise Adam via log‑softmax (avoids division by zero)
            z0 = np.log(seed_vec + 1e-12)

            # -----------------------------------------------------------------
            # 1. Continuation Adam optimisation (smooth‑max)
            # -----------------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=800.0,
                ramp_iters=80_000,
                lr=0.04,
            )

            # -----------------------------------------------------------------
            # 2. Official evaluation of the Adam result
            # -----------------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

    # ----------------------------------------------------------------------
    # 3. Final global polishing with projected sub‑gradient descent
    # ----------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nFinal global polishing (sub‑gradient descent)…")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=180_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Polished official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by polish ***")
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # 4. Optional LP refinement (a few deterministic improvement steps)
    # ----------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        n_best = len(best_seq)
        # LP refinement is memory‑intensive; restrict to n ≤ 4096
        if n_best <= 4096:
            print("\nLP refinement steps …")
            a_lp = np.array(best_seq, dtype=float)
            max_lp_iters = 5
            for it in range(max_lp_iters):
                if time.time() >= deadline - 0.5:
                    break
                conv = convolution_fft(a_lp, a_lp)
                M = float(np.max(conv))
                a_new = lp_refine_step(a_lp, M, t=0.5)
                try:
                    val = evaluate_sequence(a_new.tolist())
                except Exception:
                    break
                if val < best_val:
                    best_val, best_seq = val, a_new.tolist()
                    print(f"   LP step {it+1}: improved to {best_val:.6f}")
                a_lp = a_new
            # end of LP loop

    # ----------------------------------------------------------------------
    # 5. Tiny hill‑climbing tweak (very cheap, may squeeze a few 1e‑4)
    # ----------------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\nHill‑climb refinement …")
        a_arr = np.array(best_seq, dtype=float)
        a_hc, val_hc = hill_climb_refine(a_arr, deadline, rng,
                                         max_moves=3000, step_factor=0.04)
        if val_hc < best_val:
            best_val, best_seq = val_hc, a_hc.tolist()
            print(f"   Hill‑climb improved best to {best_val:.6f}")

    # ----------------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty return
    # ----------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
```

This implementation expands the original hybrid search by:

* adding genuinely asymmetric seeds (exponential, one‑sided power, Dirichlet draws);
* increasing the continuation‑Adam temperature range and ramp length;
* performing a final exact sub‑gradient polish;
* applying a few linear‑programming refinement steps that maximise the ℓ₁‑mass while respecting the current convolution bound;
* a lightweight hill‑climb to catch any remaining tiny improvements.

All steps respect the global time budget and always return a valid non‑negative sequence.
