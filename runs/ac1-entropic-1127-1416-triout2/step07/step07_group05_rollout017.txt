def lp_maximize_sum(f, M):
    n = f.shape[0]
    L = 2 * n - 1
    # Build A
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    b = np.full(L, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    return res.x

def lp_iterative_improvement(f, deadline, max_iters=5):
    for it in range(max_iters):
        if time.time() >= deadline - 2:
            break
        conv = convolution_fft(f, f)
        M = float(np.max(conv))
        try:
            g = lp_maximize_sum(f, M)
        except Exception as e:
            print(f"LP iteration {it} failed: {e}")
            break
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        improved = False
        for t in (0.5, 0.3, 0.1):
            f_new = (1 - t) * f + t * g_norm
            f_new = np.clip(f_new, 0.0, None)
            f_new /= f_new.sum()
            new_conv = convolution_fft(f_new, f_new)
            new_M = float(np.max(new_conv))
            # Accept if new M lower
            if new_M < M - 1e-12:
                f = f_new
                improved = True
                print(f"LP it {it} t={t} improved M from {M:.6e} to {new_M:.6e}")
                break
        if not improved:
            print(f"LP it {it} no improvement, stop.")
            break
    return f

"""
Propose a step‑function sequence with a very low  evaluate_sequence  score.

The search consists of three phases:

1️⃣  Generate a diverse pool of 1024‑point candidates (arcsine, beta‑like,
    power‑law, mixtures, triangular bumps, …) and keep the best few.

2️⃣  For each of these seeds run a short *linear‑program* improvement:
    given the current shape f and its peak   M = max(f*f),
    solve  max Σ g_j  subject to g≥0 and (f∗g)_k ≤ M  (linear constraints).
    The solution g pushes mass into positions that are “light” for f.
    A convex combination (1‑t)·f + t·ĝ (with ĝ = g/Σg) is taken;
    the step size t is chosen greedily (0.5 → 0.3 → 0.1) and accepted only
    if the peak strictly decreases.  A few LP iterations typically lower the
    surrogate 2·n·max(conv) by a noticeable amount.

3️⃣  Upsample the best 1024‑vector to 2048 and 4096 points, polish it with a
    smooth‑max Adam optimiser, and finish with an exact sub‑gradient descent
    followed by an exponentiated‑gradient (mirror‑descent) polish.
    All stages respect the global 1000 s budget and keep the best sequence
    according to the official ``evaluate_sequence`` function.

If a fallback solution (global variable ``height_sequence_1``) exists it is
used as a safety net.  The routine finally returns a plain list of floats.
"""

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (length = len(a)+len(b)-1).
    Both inputs are assumed to be 1‑D real arrays.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector a is expected to be normalised (sum = 1), so the denominator
    disappears and the score reduces to 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_asymmetric_power_shape(
    n: int, p: float, eps_left: float, eps_right: float
) -> np.ndarray:
    """
    Asymmetric generalised arcsine:
    a_i ∝ 1 / ((i+eps_left)*(n‑½‑i+eps_right))**p .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + eps_left) * (n - 0.5 - i + eps_right) + 1e-12) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(v) to target_n,
    followed by re‑normalisation to sum = 1.
    """
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 30_000,
    tau_start: float = 0.5,
    tau_end: float = 100.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimiser of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k)
    where τ (temperature) is increased exponentially from ``tau_start``
    to ``tau_end``.
    The variables are logits w with a = softmax(w), guaranteeing non‑negativity
    and Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # Initialise logits from the starting point (tiny offset for log)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # a = softmax(w)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # compute softmax weights over τ·conv
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()        # shape (L,)

        # gradient wrt a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wr.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (step size decays like 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging
        if it % 10_000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution objective.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # mask of entries attaining the maximum (within tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) restricted to original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror‑descent with KL geometry).
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A (shape (2n‑1, n)) such that A @ g = conv(f, g).
    Row k contains the coefficients f_i that multiply g_{k-i}.
    """
    n = f.shape[0]
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def lp_maximize_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximize   Σ g_j
        subject to g_j ≥ 0  and   (f * g)_k ≤ M   for all k.
    Returns the optimal g (may be very large in some components).
    """
    if not _SCIPY_AVAILABLE:        # pragma: no cover
        raise RuntimeError("SciPy not available – cannot run LP stage.")
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum → minimise -sum
    bounds = [(0.0, None) for _ in range(n)]

    # Use the HiGHS interior‑point solver (fast and reliable)
    res = _opt.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def lp_iterative_improvement(
    f: np.ndarray,
    deadline: float,
    max_iters: int = 6,
) -> np.ndarray:
    """
    Repeatedly solve the LP from the current shape f, then
    mix the current shape with the normalised LP solution.
    The mixing factor t is tried in decreasing order (0.5, 0.3, 0.1) and
    accepted only if the peak of the convolution strictly decreases.
    """
    for it in range(max_iters):
        if time.time() >= deadline - 2.0:
            break

        conv = convolution_fft(f, f)
        M = float(np.max(conv))
        try:
            g = lp_maximize_sum(f, M)
        except Exception as e:
            print(f"    LP iteration {it} failed: {e}")
            break

        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g

        improved = False
        for t in (0.5, 0.3, 0.1):
            f_new = (1.0 - t) * f + t * g_norm
            f_new = np.clip(f_new, 0.0, None)
            f_new /= f_new.sum()

            conv_new = convolution_fft(f_new, f_new)
            M_new = float(np.max(conv_new))
            if M_new < M - 1e-15:
                f = f_new
                improved = True
                print(f"    LP it {it} t={t:.2f} improved M {M:.6e} → {M_new:.6e}")
                break

        if not improved:
            print(f"    LP it {it} no improvement, stopping LP phase.")
            break
    return f

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  The routine follows the three‑phase plan described
    in the module docstring and returns the best sequence found before the
    global deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # a small safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation (n = 1024)
    # ------------------------------------------------------------------
    n_small = 1024
    seed_vectors: list[np.ndarray] = []

    # deterministic seeds
    seed_vectors.append(arcsine_initial(n_small))
    seed_vectors.append(np.full(n_small, 1.0 / n_small))

    # random parametric families
    for _ in range(250):
        if time.time() >= deadline - 20.0:
            break
        typ = rng.integers(0, 5)
        if typ == 0:  # beta‑like
            a = rng.uniform(0.05, 0.9)
            b = rng.uniform(0.05, 0.9)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n_small))
            cand = generate_beta_shape(n_small, a, b, eps)
        elif typ == 1:  # power‑shape
            p = rng.uniform(0.2, 3.0)
            eps = rng.uniform(0.0, 5.0)
            cand = generate_power_shape(n_small, p, eps)
        elif typ == 2:  # mixture arcsine/uniform
            w = rng.random()
            cand = generate_mixture_shape(n_small, w)
        elif typ == 3:  # triangular bump
            peak = rng.integers(0, n_small)
            cand = generate_triangular_shape(n_small, peak)
        else:           # asymmetric power shape
            p = rng.uniform(0.2, 3.0)
            eps_l = rng.uniform(0.0, 1.0)
            eps_r = rng.uniform(0.0, 1.0)
            cand = generate_asymmetric_power_shape(n_small, p, eps_l, eps_r)
        seed_vectors.append(cand)

    # Quick surrogate ranking
    scored = [(compute_score(v), v) for v in seed_vectors]
    scored.sort(key=lambda x: x[0])
    top_small = [vec for _, vec in scored[:3]]
    print(f"\nSeed ranking complete – best surrogate = {scored[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Linear‑program refinement of the top seeds
    # ------------------------------------------------------------------
    refined_candidates: list[tuple[float, np.ndarray]] = []
    for idx, vec in enumerate(top_small, start=1):
        if time.time() >= deadline - 10.0:
            break
        print(f"\n--- LP refinement for seed {idx}/{len(top_small)} ---")
        vec = vec / vec.sum()
        vec = lp_iterative_improvement(vec, deadline, max_iters=6)
        surrogate = compute_score(vec)
        print(f"    Surrogate after LP = {surrogate:.6f}")
        refined_candidates.append((surrogate, vec))

    refined_candidates.sort(key=lambda x: x[0])
    best_small_vec = refined_candidates[0][1]
    best_small_score = refined_candidates[0][0]

    # ------------------------------------------------------------------
    # 3️⃣  Upsample and smooth‑max Adam optimisation (2048 → 4096)
    # ------------------------------------------------------------------
    best_large_vec = None
    best_large_score = float("inf")
    for target_n in (2048, 4096):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Upscaling to n={target_n} and Adam optimisation ===")
        up = upscale_vector(best_small_vec, target_n)

        # a tiny random perturbation helps escape shallow minima
        jitter = rng.normal(scale=1e-4, size=target_n)
        up = up + jitter
        up = np.clip(up, 0.0, None)
        up /= up.sum()

        remaining = deadline - time.time()
        max_iters = int(min(150_000, remaining * 30))
        a_opt = adam_optimize_smooth_max(
            up,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=150.0,
            base_lr=0.22,
        )
        surrogate = compute_score(a_opt)
        print(f"    Surrogate after Adam = {surrogate:.6f}")
        if surrogate < best_large_score:
            best_large_score = surrogate
            best_large_vec = a_opt

    # ------------------------------------------------------------------
    # 4️⃣  Exact refinement and final polishing
    # ------------------------------------------------------------------
    if best_large_vec is not None:
        n_best = best_large_vec.shape[0]
        print(f"\n--- Exact refinement on n={n_best} ---")
        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, remaining * 30))
        refined = subgradient_descent_exact(
            best_large_vec,
            deadline,
            max_iter=max_iter_sub,
            step0=0.08,
        )

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(180_000, remaining * 30))
            refined = eg_refine(
                refined,
                deadline,
                max_iter=max_iter_eg,
                eta0=0.20,
            )

        # official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"Official score after final refinement = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print("*** New global best ***")
        except Exception as e:
            print(f"Evaluation failed after refinement: {e}")

    # ------------------------------------------------------------------
    # 5️⃣  Global polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Global polishing phase ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 30))
        a = subgradient_descent_exact(
            a, deadline, max_iter=max_iter_sub, step0=0.06
        )

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(200_000, remaining * 30))
            a = eg_refine(a, deadline, max_iter=max_iter_eg, eta0=0.17)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after polishing ***")

    # ------------------------------------------------------------------
    # 6️⃣  Guard – always return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print(f"Returned uniform fallback, score = {best_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
