    # Solve LP to maximize sum(b) subject to a*b <= M.
    n = len(a)
    # Build constraint matrix A_ub and b_ub = M*ones.
    # A_ub shape (2n-1, n)
    # Could store as csr.
    rows = []
    cols = []
    datas = []
    for i in range(n):
        # row indices for this a_i times each b_j: for j from 0 to n-1, row = i+j
        j_arr = np.arange(n)
        rows.append(i + j_arr)   # shape (n,)
        cols.append(j_arr)
        datas.append(np.full(n, a[i]))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    datas = np.concatenate(datas)
    A_ub = scipy.sparse.csr_matrix((datas, (rows, cols)), shape=(2*n-1, n))
    # b_ub = np.full(2*n-1, M)

    c = -np.ones(n)  # maximize sum => minimize -sum.
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')

def lp_improve(a, t, deadline, max_steps):
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    rows, cols = precompute rows, cols for n
    while steps < max_steps and time.time()<deadline:
        # compute current M
        conv = convolution_fft(a, a)
        M = conv.max()
        # solve LP
        data = np.repeat(a, n)
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        res = scipy.optimize.linprog(-np.ones(n), A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=[(0,None)]*n, method='highs')
        if not res.success:
            break
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = b_opt / sum_b
        a_new = (1 - t) * a + t * g
        # project to simplex
        a_new = a_new / a_new.sum()  # ensure sum=1
        a_new = np.maximum(a_new, 0)
        # compute new score
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
            print(f"LP step improved: score={best_score:.6f}")
        a = a_new
        # optionally adapt t: if improvement small reduce t
        steps += 1
        # maybe reduce t slowly after some steps
        if steps % 5 == 0:
            t = max(t * 0.8, 0.01)  # shrink step
    return best, best_score

"""
Multi‑phase optimizer for the step‑function bound.

Approach
--------
1. **Random warm‑starts** – draw a Dirichlet point on the probability simplex.
2. **Flattening** – minimise the variance of the self‑convolution (Adam on a
   quartic loss) – this yields a fairly flat autocorrelation.
3. **Smooth‑max surrogate** – minimise a differentiable approximation of the
   max‑convolution (ε·log‑sum‑exp) with Adam.
4. **LP‑based Frank‑Wolfe step** – given the current vector `a` with
   `M = max(a⊛a)`, solve the linear programme  

        maximise   Σ b_i
        subject to  a * b ≤ M  (component‑wise)
                    b_i ≥ 0

   (the convolution `a*b` is linear in `b`).  
   The optimiser returns `g = b / Σb` (∑g = 1) and moves
   `a ← (1‑t)·a + t·g`.  Repeating this step strictly reduces the peak
   of the autocorrelation.
5. **Hill‑climb (simulated annealing)** – stochastic mass transfers that
   occasionally accept worsening moves to escape local minima.
6. **Symmetry averaging** – replace `a` by `(a+reverse(a))/2` and re‑project
   onto the simplex; symmetric vectors often have flatter autocorrelations.

All vectors are projected onto the probability simplex after every update,
which guarantees a valid input for `evaluate_sequence`.  The routine respects
the supplied time budget (≈1000 s) and returns the best sequence found.

"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisation
# ----------------------------------------------------------------------


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Phase 2 – flattening the convolution (L2 variance) using Adam
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 500,
    step0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise   ∑ (conv_k - μ)²   (variance of the convolution)
    with Adam.  The gradient of this quartic loss is
        g = 4 * Σ_j a_j * (conv_{i+j} - μ)
    where μ = mean(conv).  After each step we project onto the simplex.
    Returns the best vector seen (exact score).
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)                 # length 2n‑1
        mu = np.mean(conv)                           # = sum_a**2/(2n‑1) = 1/(2n‑1)
        diff = conv - mu

        # gradient = 4 * Σ_j a_j * diff_{i+j}
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)      # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – smooth soft‑max (max‑surrogate) phase
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 1500,
    step0: float = 0.45,
    eps_factor: float = 0.22,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of the max‑convolution:
        f(a) = ε·log∑_k exp(conv_k/ε)
    Gradient is g_i = 2 * Σ_j a_j * w_{i+j},
    where w is the soft‑max over conv/ε.
    Returns the best vector (exact score) observed.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε shrinks but never below 1 % of maxc
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)           # soft‑max, ∑ w = 1

        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to help escape flat regions
        if t % 400 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – LP‑based Frank‑Wolfe step (linear program)
# ----------------------------------------------------------------------


def lp_frank_wolfe_step(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 15,
    step_size: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the linear programme

        maximise   Σ b_i
        subject to  a * b ≤ M   (M = max(a⊛a))
                    b_i ≥ 0

    and move `a ← (1‑t) a + t g` where `g = b / Σb`.
    The LP is solved with scipy's HiGHS solver using a sparse matrix
    representation of the convolution constraints.
    Returns the best vector seen and its exact score.
    """
    a = a0.astype(float).copy()
    n = len(a)

    # static sparse pattern: rows and columns of the convolution matrix
    i = np.arange(n).reshape(-1, 1)
    j = np.arange(n).reshape(1, -1)
    rows = (i + j).ravel().astype(np.int32)          # shape (n*n,)
    cols = np.tile(np.arange(n), n).astype(np.int32)  # shape (n*n,)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iter):
        if time.time() >= time_deadline:
            break

        # current peak of the autocorrelation
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # build RHS of the inequalities
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # data = a_i repeated for each column j
        data = np.repeat(a, n)  # length n*n

        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # solve linear program: minimise -Σb   (equivalent to maximise Σb)
        res = linprog(
            c=-np.ones(n, dtype=float),  # minimise -sum(b)
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # Abort the LP phase if the solver fails.
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b_opt = res.x
        sum_b = np.sum(b_opt)
        if sum_b <= 0:
            break

        # Normalise to a probability vector (∑g = 1)
        g = b_opt / sum_b

        # Convex combination
        a_new = (1.0 - step_size) * a + step_size * g
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = np.sum(a_new)
        if a_new_sum == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / a_new_sum

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            print(f"[LP] iter {it}, step {step_size:.3f}, score={best_sc:.6f}")

        a = a_new
        # shrink the step size a little as we converge
        step_size = max(step_size * 0.9, 0.02)

    return best_a, best_sc

# Phase 5 – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 150_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with simulated annealing.
    Moves:
        * swap two positions
        * transfer a fraction of mass from i → j
    Always accepts improving moves; otherwise accepts with probability
    exp(‑Δ / T) where T cools linearly.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12  # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        if step % 5000 == 0:
            # occasional larger perturbation
            rnd = random_dense_vector(n) * 0.02
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Symmetrise the vector (a + reverse(a))/2 and re‑project onto the simplex.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

            polished_seq, polished_sc = hill_climb_anneal(
                seq_to_polish,
                time_deadline=deadline - 2.0,
                max_steps=100_000,
            )
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished_seq
                print(f"[{outer_iter}] anneal (n={n}) improved score={best_score:.6f}")

            # final cheap symmetry averaging
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer_iter}] symmetry averaged (n={n}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
