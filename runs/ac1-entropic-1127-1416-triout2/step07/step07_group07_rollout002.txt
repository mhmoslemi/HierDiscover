def beta_initialisation(alpha, n):
    i = np.arange(n)
    x = i / (n-1)
    eps = 1e-12
    w = (x+eps)**(alpha-1) * (1 - x + eps)**(alpha-1)
    w = np.where(np.isfinite(w), w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones(n)
        w_sum = n
    w /= w_sum
    return w

"""
propose_candidate – a multi‑stage optimizer for the
self‑convolution supremum problem.

Algorithm outline
-----------------
1.  **Parametric seeding** – generate symmetric beta‑type
    distributions a_i ∝ (x_i)^{α‑1}(1‑x_i)^{α‑1} on a discrete
    grid (x_i = i/(n‑1)).  A coarse sweep over α gives a good
    starting shape for each chosen length n.

2.  **Smooth soft‑max (Adam) refinement** – minimise the
    log‑sum‑exp surrogate of the max‑convolution using Adam on
    the simplex.  The temperature ε is driven down during the run.

3.  **Exact sub‑gradient descent** – further polishing with the true
    sub‑gradient of  max(a∗a).

4.  **Stochastic hill‑climbing** – strict improving swaps / mass
    transfers, with an adaptive step‑size.

5.  **Symmetry averaging** – enforce (a+rev(a))/2 and re‑project.

All vectors are kept on the probability simplex (non‑negative,
sum = 1) so that the cheap fast score
    2·n·max(conv) equals the exact `evaluate_sequence` value.
The routine respects the supplied time budget (≈ 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a)=1.
    Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initialisation (symmetric Beta family)
# ----------------------------------------------------------------------


def beta_initialisation(alpha: float, n: int) -> np.ndarray:
    """
    Symmetric beta density on a discrete grid.
    a_i ∝ (x_i)^{α‑1}(1‑x_i)^{α‑1},  x_i = i/(n‑1).
    Result is projected onto the simplex (sum = 1).
    """
    i = np.arange(n, dtype=float)
    x = i / (n - 1)
    eps = 1e-12
    # work in log‑space to avoid overflow for α < 1
    log_w = (alpha - 1.0) * (np.log(x + eps) + np.log(1.0 - x + eps))
    # shift to keep numbers moderate
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w = project_to_simplex(w)
    return w


# ----------------------------------------------------------------------
# Smooth soft‑max gradient (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of    f_eps(a) = eps·log Σ_k exp(conv_k/eps)
    where conv = a ∗ a.

    Returns 2·(w_rev ⊛ a) with w = softmax(conv/eps).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)          # Σ w = 1
    w_rev = w[::-1]

    # cross‑correlation w_rev ⊛ a
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.4,
    eps_factor: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Runs until the supplied time deadline.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature – never lower than 1 % of the current maximum
        eps = maxc * max(eps_factor * (0.96 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        # Adam bias‑corrected estimates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if t % 1000 == 0:

    return best_a, best_sc

def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.09,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on 2·n·max(a∗a).
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2·a_{k‑i} (zero outside support)
        idx = np.arange(n)

            )

# ----------------------------------------------------------------------
# Stochastic hill‑climber (strictly improving moves)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber with swaps and mass‑transfer moves.
    Only strictly improving moves are kept.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_frac = 0.07  # initial transfer fraction

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # safeguard against tiny negatives caused by floating point
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.08, 0.5)   # become bolder after success
        else:
            step_frac *= 0.96                        # become more cautious

        max_steps -= 1

    return best_a, best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Enforce approximate symmetry: (a+rev(a))/2 and re‑project."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

            sc = compute_score(a0)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = a0
                best_alpha = alpha

        print(f"[grid] n={n:,d} best α≈{best_alpha:.4f}  score={local_best_sc:.6f}")

        # update global best if needed
        if local_best_sc < best_score:
            best_score = local_best_sc
            best_seq = local_best_a.tolist()
            print(f"  >>> new global best from grid (n={n})")

        # --------------------------------------------------------------
        # 1. Adam‑accelerated smooth soft‑max descent
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 20.0:
            # allocate roughly half the remaining time to Adam
            adam_deadline = time.time() + remaining * 0.5
            a_adam, sc_adam = adam_softmax_descent(
                local_best_a,
                time_deadline=adam_deadline,
                max_iter=200_000,
                step0=0.45,
                eps_factor=0.28,
            )
            if sc_adam < best_score:
                best_score = sc_adam
                best_seq = a_adam.tolist()
                print(f"[Adam] n={n:,d}  score={sc_adam:.6f}")

        # --------------------------------------------------------------
        # 2. Exact sub‑gradient descent for fine tuning
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 15.0:
            sub_deadline = time.time() + remaining * 0.4
            a_sub, sc_sub = subgradient_descent(
                a_adam if 'a_adam' in locals() else local_best_a,
                time_deadline=sub_deadline,
                max_iter=200_000,
                step0=0.09,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[Sub‑grad] n={n:,d}  score={sc_sub:.6f}")

        # --------------------------------------------------------------
        # 3. Stochastic hill‑climbing polishing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            hc_deadline = time.time() + remaining * 0.5
            start_vec = np.array(best_seq if best_seq is not None else local_best_a,
                                 dtype=float)
            polished, pol_sc = hill_climb(
                start_vec,
                time_deadline=hc_deadline,
                max_steps=300_000,
            )
            if pol_sc < best_score:
                best_score = pol_sc
                best_seq = polished.tolist()
                print(f"[Hill‑climb] n={n:,d}  score={pol_sc:.6f}")

            # symmetry averaging – inexpensive, often beneficial
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[Symmetry] n={n:,d}  score={sym_sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (guaranteed finite result)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}"
          f"   (length = {len(best_seq)})")
    return best_seq
