"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

The optimiser consists of many short‑budget restarts that each run a
pipeline of simple first‑order methods.  Compared with the reference
implementation we

* use longer inner iteration limits (more optimisation time),
* add a tiny “gradient‑descent‑on‑the‑max” phase that directly uses the
  current worst convolution index,
* try a small family of deterministic linear‑tilt starts (useful as a
  cheap fallback),
* keep the best sequence seen over the whole 1000 s budget.

All updates are projected onto the probability simplex, guaranteeing a
valid input for `evaluate_sequence`.  The code is self‑contained,
runs only on CPU and stays well under the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisation
# ----------------------------------------------------------------------


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


def linear_tilt_vector(n: int, tilt: float) -> np.ndarray:
    """
    Deterministic linear‑tilt start.
    `tilt` is the fraction of the total mass moved from the right end
    to the left end (0 ≤ tilt ≤ 1).  tilt=0 → uniform, tilt=1 → all mass
    on the first element (clipped to the simplex).
    """
    if tilt <= 0.0:
        return np.full(n, 1.0 / n)

    # a_i = 1/n + eps * ((n-1)/2 - i)
    eps = tilt * (2.0 / (n * (n - 1)))          # maximal admissible ε = 2/(n(n-1))
    a = np.full(n, 1.0 / n) + eps * ((n - 1) / 2 - np.arange(n))
    a = np.clip(a, 0, None)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Phase 2 – variance flattening (Adam)
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the self‑convolution with Adam.
    Returns the best vector seen.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        mu = np.mean(conv)
        diff = conv - mu

        # gradient of the variance loss
        diff_rev = diff[::-1]
        grad = 4.0 * convolution_fft(a, diff_rev)[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a = project_to_simplex(a - step * m_hat / (np.sqrt(v_hat) + eps))

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc

# Phase 3 – smooth‑max surrogate (Adam)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 8000,
    step0: float = 0.5,
    eps_factor: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Minimise ε·log∑exp(conv/ε), a smooth upper bound on max(conv).
    Returns the best exact‑score vector observed.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)                       # ∑ w = 1

        grad = 2.0 * convolution_fft(a, w[::-1])[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * m_hat / (np.sqrt(v_hat) + 1e-8))

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional jitter to shake out flat plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_sc

# Phase 4 – exact max sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the true objective
        f(a) = 2·n·max(a⊛a) .
    Uses the current worst shift to compute the sub‑gradient.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k-i}  (0 outside the overlap)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional small random shake
        if t % 5000 == 0:
            shake = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a = project_to_simplex(a + shake)

    return best_a, best_sc

def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 400_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with a linear cooling schedule.
    Moves:
        – swap two entries,
        – transfer a small fraction of mass from i→j.
    All moves are projected onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.14  # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.04 * best_a[i]                 # 4 % of source mass
                cand[i] = max(0.0, cand[i] - delta)
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # every 8000 steps apply a larger random perturbation
        if step % 8000 == 0:
            rnd = random_dense_vector(n) * 0.015          # 1.5 % random noise
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (cheap post‑process)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Average with the reversed version and re‑project.
    Often reduces the peak of the convolution.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget and returns a plain list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.5               # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback using a simple deterministic linear‑tilt construction
    # ------------------------------------------------------------------
    for tilt in (0.0, 0.3, 0.6):
        try:
            cand = linear_tilt_vector(2048, tilt).tolist()
            sc = evaluate_sequence(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand
                print(f"Fallback tilt={tilt:.2f} score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]     # keep FFT cheap
    outer = 0

    while time.time() < deadline - 5.0:
        outer += 1
        n = int(np.random.choice(lengths))

        # ------- 1. initial vector (mix of random & linear tilt) -------
        if best_seq is not None and np.random.rand() < 0.12:
            a0 = np.array(best_seq[:n], dtype=float)
            a0 = project_to_simplex(a0)
        else:
            if np.random.rand() < 0.33:
                a0 = linear_tilt_vector(n, tilt=np.random.rand() * 0.7)
            else:
                a0 = random_dense_vector(n)

        # quick filter
        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score, best_seq = cur_sc, a0.tolist()
            print(f"[{outer}] init  n={n}  score={best_score:.6f}")

        # ------- 2. flattening -------
        if time.time() < deadline - 8.0:
            a_f, sc_f = flatten_conv_adam(
                a0,
                time_deadline=deadline - 2.0,
                max_iter=8000,
                step0=0.15,
            )
            if sc_f < best_score:
                best_score, best_seq = sc_f, a_f.tolist()
                print(f"[{outer}] flatten  n={n}  score={best_score:.6f}")

        # ------- 3. smooth max surrogate -------
        if time.time() < deadline - 12.0:
            a_s, sc_s = smooth_max_adam(
                a_f,
                time_deadline=deadline - 3.0,
                max_iter=15000,
                step0=0.55,
                eps_factor=0.25,
            )
            if sc_s < best_score:
                best_score, best_seq = sc_s, a_s.tolist()
                print(f"[{outer}] surrogate n={n}  score={best_score:.6f}")

        # ------- 4. exact max sub‑gradient -------
        if time.time() < deadline - 16.0:
            a_g, sc_g = subgrad_optimize(
                a_s,
                time_deadline=deadline - 5.0,
                max_iter=250_000,
                step0=0.09,
            )
            if sc_g < best_score:
                best_score, best_seq = sc_g, a_g.tolist()
                print(f"[{outer}] sub‑grad n={n}  score={best_score:.6f}")

        # ------- 5. simulated‑annealing hill‑climb -------
        if time.time() < deadline - 6.0:
            seq_src = best_seq if best_seq is not None else a_g.tolist()
            a_h, sc_h = hill_climb_anneal(
                seq_src,
                time_deadline=deadline - 2.5,
                max_steps=300_000,
            )
            if sc_h < best_score:
                best_score, best_seq = sc_h, a_h
                print(f"[{outer}] anneal   n={n}  score={best_score:.6f}")

            # cheap symmetry averaging
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score, best_seq = sc_sym, a_sym.tolist()
                print(f"[{outer}] sym‑avg  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – fall back to a uniform vector if nothing else succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
