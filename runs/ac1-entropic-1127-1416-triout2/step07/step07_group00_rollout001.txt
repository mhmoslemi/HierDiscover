"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm (≈ 1000 s budget)
----------------------------
1. **Coarse – fine α‑grid** – search the one‑parameter family  
   a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α) on a modest size (n≈2048) to obtain a
   good starting shape.

2. **Heavy smooth optimisation** – minimise the log‑sum‑exp
   smoothed max of the self‑convolution using projected gradient descent
   with an exponential β‑schedule (β≈2 → 300).  The optimisation runs
   until the time budget for this phase is exhausted.

3. **Projected sub‑gradient refinement** – continue optimisation on the
   true max‑objective with a 1/√t stepsize.

4. **Multi‑scale up‑sampling** – interpolate the best small‑n vector to a
   larger size (n≈8192 if time permits) and perform a short
   smooth‑gradient pass to exploit the extra degrees of freedom.

5. **Final hill‑climb** – a cheap stochastic mass‑transfer walk that
   accepts only improving moves.

All heavy work uses only NumPy (FFT‑based convolutions).  The code respects
the 1000 s wall‑clock limit and returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of `evaluate_sequence` for sum(a)=1.
    Returns 2·n·max(a⊛a) (no clamping needed).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns a non‑negative vector whose entries sum to 1.
    """
    if np.all(v <= 0):
        # uniform point if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Edge‑heavy analytical families
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    Return the normalised vector a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α)
    lying on the probability simplex.
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Two‑stage grid search over α.
    Coarse grid [0, 3] step 0.2, then fine grid around the best point
    with step 0.02.  Returns (best_vector, exact_score).
    """
    best_a = None
    best_score = float("inf")

    # -------- coarse scan ----------
    alphas = np.arange(0.0, 3.0 + 1e-9, 0.2)
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_alpha = alpha

    # -------- fine scan around the best ----------
    if best_a is not None:
        low = max(0.0, best_alpha - 0.2)
        high = best_alpha + 0.2
        fine_alphas = np.arange(low, high + 1e-9, 0.02)
        for alpha in fine_alphas:
            if time.time() >= deadline:
                break
            a = alpha_start(n, alpha)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_alpha = alpha

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Smooth log‑sum‑exp (annealed) gradient descent
# ----------------------------------------------------------------------


def smooth_gradient_optimize(start_a: np.ndarray,
                             deadline: float,
                             max_iters: int = 2_000_000,
                             beta_start: float = 2.0,
                             beta_end: float = 300.0,
                             step0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using a decreasing step size and Euclidean projection onto the simplex.
    The smoothing parameter β is increased exponentially from `beta_start`
    to `beta_end`.  The routine returns the best exact score observed.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- β schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- smoothed max (log‑sum‑exp) -----
        conv = convolution_fft(a, a)            # length 2n‑1
        maxc = conv.max()
        # prevent overflow
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()     # probability vector, sum=1

        # ----- gradient of the smoothed objective -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)    # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]  # slice length n

        # ----- step & projection -----
        step = step0 / np.sqrt(it)
        a = project_to_simplex(a - step * grad)

        # ----- occasional exact evaluation -----
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient method for the true max‑objective
        2·n·max(a*a)
    using a 1/√t stepsize.
    Returns the best exact score encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                # index of the active max

        # sub‑gradient of max‑entry k:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        # occasional exact evaluation
        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Simple stochastic hill‑climb (mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
      – with probability 0.5 swap two entries,
      – otherwise transfer a small fraction of mass from one index to another.
    A move is kept only if the exact score improves.
    """
    a = start_seq.copy()
    n = a.size

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12  # initial mass‑transfer fraction

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.97                         # become timid

        max_steps -= 1

    return best_a, best_score

# 5. Upsampling helper (small → large n)
# ----------------------------------------------------------------------


def upsample(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Interpolate a_small (sum=1) to length n_large and renormalise.
    Linear interpolation on the unit interval preserves the shape.
    """
    x_small = np.linspace(0.0, 1.0, a_small.size)
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    # numerical safety – enforce non‑negativity then renormalise
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    return a_large

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows a multi‑stage strategy and respects the 1000 s
    wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 0. fallback – try a previously known good construction if present
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        # `height_sequence_1` is supplied by the evaluation harness
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Small‑scale (n≈2048) intensive optimisation
    # --------------------------------------------------------------
    n_small = 2048
    print(f"[{time.time():.1f}] using small size n = {n_small}")

    # 1a – α‑grid start
    a_alpha, sc_alpha = find_best_alpha_start(n_small, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] after α‑grid best = {best_score:.6f}")

    # 1b – smooth gradient descent (heavy)
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] smooth‑gradient (small n)")
        a_smooth, sc_smooth = smooth_gradient_optimize(
            np.array(best_seq, dtype=float),
            deadline - 200.0,
            max_iters=2_000_000,
            beta_start=2.0,
            beta_end=300.0,
            step0=0.8,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{time.time():.1f}] after smooth‑grad best = {best_score:.6f}")

    # 1c – sub‑gradient refinement (still small n)
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement (small n)")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=1_500_000,
            step0=0.05,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Upsample to larger n if enough time remains
    # --------------------------------------------------------------
    time_now = time.time()
    if best_seq is not None and time_now < deadline - 80.0:
        # Choose a larger size based on the remaining budget
        remaining = deadline - time_now
        n_large = 8192 if remaining > 200 else 4096
        print(f"[{time.time():.1f}] up‑sampling to n = {n_large}")

        a_large = upsample(np.array(best_seq, dtype=float), n_large)

        # short smooth‑gradient phase on the larger vector
        print(f"[{time.time():.1f}] final smooth‑gradient (large n)")
        a_big_smooth, sc_big = smooth_gradient_optimize(
            a_large,
            deadline - 30.0,
            max_iters=400_000,
            beta_start=5.0,
            beta_end=150.0,
            step0=0.6,
        )
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big_smooth.tolist()
            print(f"[{time.time():.1f}] after large‑n smooth best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Final cheap hill‑climb polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Safety net – if everything failed, fall back to a modest uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
