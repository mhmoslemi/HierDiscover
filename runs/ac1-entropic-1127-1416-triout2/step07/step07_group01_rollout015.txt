    def integer_sa(n, m, time_limit):
        # initial random set
        pos = np.random.choice(n, size=m, replace=False).tolist()
        present = np.zeros(n, dtype=bool)
        present[pos] = True

        conv_counts = conv.astype(np.int32)  # shape 2n-1
        cur_max = conv_counts.max()
        scaling = (2.0 * n) / (m * m)
        cur_score = scaling * cur_max
        best_local_max = cur_max
        best_pos = pos.copy()
        best_local_score = cur_score

        # temperature schedule
        start_temp = cur_score * 0.5  # half of initial score
        end_temp = cur_score * 0.001 # small
        max_iter = int(1e7)  # dummy
        iter_count = 0
        while time.time() < time_limit:
            iter_count +=1
            elapsed = time.time() - start_time
            # exponential decay
            t_frac = (deadline - time.time()) / (deadline - start_time)
            # we can compute temperature linearly decreasing:
            T = start_temp * (t_frac) + end_temp * (1 - t_frac)  # approximate

            # random swap
            # select i_out and its index in list
            idx_out = np.random.randint(len(pos))

            conv_counts[2 * i_out] += 1

"""
A two‑stage search for a step‑function that minimises
`evaluate_sequence`.  First a combinatorial annealer constructs a sparse
binary pattern with a very low maximum of the integer convolution
(the number of ordered pairs that share the same index sum).  With that
pattern we obtain a vector of equal non‑zero entries (sum = 1).  The
pattern is then refined with the smooth‑log‑sum‑exp Adam optimiser,
followed by a sub‑gradient clean‑up and a final stochastic hill‑climb.
All phases respect the 1000 s wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for many calls)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed max‑objective optimiser (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(start_a: np.ndarray,
                             time_deadline: float,
                             max_iters: int = 300_000,
                             beta_start: float = 5.0,
                             beta_end: float = 80.0,
                             step0: float = 0.8,
                             adam_beta1: float = 0.9,
                             adam_beta2: float = 0.999,
                             adam_eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam‑style optimisation of the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed objective
        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()              # probability vector

        # gradient (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]         # slice of length n

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser of the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 400_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb polishing (small mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq,
                           time_deadline,
                           max_steps: int = 500_000) -> tuple[list[float], float]:
    """Stochastic hill‑climb that swaps entries or transfers a tiny fraction of mass."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Integer‑Sparsity annealer (binary pattern optimisation)
# ----------------------------------------------------------------------


def integer_sa(n: int,
               m: int,
               time_deadline: float) -> tuple[list[int], float]:
    """
    Simulated annealing that searches for a set S ⊂ {0,…,n‑1} of size m
    minimising the maximal entry of the integer convolution of its
    indicator vector with itself.  The returned score is the exact
    `evaluate_sequence` value for the equal‑weight vector (weight = 1/m).
    """
    # start from a random set
    pos = list(np.random.choice(n, size=m, replace=False))
    present = np.zeros(n, dtype=bool)
    present[pos] = True

    # initialise integer convolution (ordered pairs)
    binary = np.zeros(n, dtype=np.int32)
    binary[pos] = 1
    conv_counts = np.convolve(binary, binary).astype(np.int32)  # length 2n‑1

    cur_max = int(conv_counts.max())
    scaling = (2.0 * n) / (m * m)          # constant factor for the final score
    cur_score = scaling * cur_max

    best_pos = pos.copy()
    best_score = cur_score
    best_max = cur_max

    # temperature schedule
    start_temp = cur_score * 0.5 + 1e-12  # avoid zero
    end_temp = cur_score * 0.001 + 1e-12

    # SA loop
    while time.time() < time_deadline:
        # linear cooling based on remaining proportion of the budget
        remaining = max(time_deadline - time.time(), 1e-12)
        total_allowed = max(time_deadline - (deadline - budget_s), 1e-12)
        t_frac = remaining / total_allowed
        T = start_temp * t_frac + end_temp * (1.0 - t_frac)

        # pick a random element to remove
        idx_out = np.random.randint(len(pos))
        i_out = pos[idx_out]

        # remove its contribution (ordered pairs)
        arr_before = np.array(pos, dtype=np.int32)          # includes i_out
        conv_counts[i_out + arr_before] -= 2
        conv_counts[2 * i_out] += 1                        # self‑pair correction

        # update the set
        pos.pop(idx_out)
        present[i_out] = False
        arr_without = np.array(pos, dtype=np.int32)        # size m‑1

        # propose a new element not currently used
        while True:
            j_in = np.random.randint(n)
            if not present[j_in]:
                break

        # add its contribution
        conv_counts[j_in + arr_without] += 2
        conv_counts[2 * j_in] += 1

        # commit the new element
        pos.append(j_in)
        present[j_in] = True

        new_max = int(conv_counts.max())
        new_score = scaling * new_max
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            cur_max = new_max
            if new_score < best_score:
                best_score = new_score
                best_max = new_max
                best_pos = pos.copy()
        else:
            # revert the move
            conv_counts[j_in + arr_without] -= 2
            conv_counts[2 * j_in] -= 1
            pos.pop()
            present[j_in] = False

            conv_counts[i_out + arr_before] += 2
            conv_counts[2 * i_out] -= 1
            pos.insert(idx_out, i_out)
            present[i_out] = True
            # cur_score, cur_max stay unchanged

    return best_pos, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises the supplied
    `evaluate_sequence`.  The routine respects the wall‑clock budget,

    a final hill‑climb.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Choose length (larger gives more freedom)
    # -----------------------------------------------------------------
    length = 8192
    if (deadline - time.time()) > 600 and budget_s >= 900:
        length = 8192
    else:
        length = 4096
    n = length
    print(f"Chosen sequence length = {n}")

    # -----------------------------------------------------------------
    # Phase 1 – integer simulated annealer for a sparse pattern
    # -----------------------------------------------------------------
    # we allocate roughly half the total budget for this combinatorial phase
    phase1_deadline = deadline - 500.0
    # pick a reasonable density – 5 % of entries gives a good trade‑off
    m = max(200, int(0.05 * n))

    print(f"Integer SA: n={n}, m={m} (≈{100*m/n:.2f}% density)")
    best_positions, approx_score = integer_sa(n, m, phase1_deadline)

    # build the equal‑weight vector from the best binary pattern

    exact_score = evaluate_sequence(a0.tolist())
    print(f"Integer SA exact score = {exact_score:.6f}")

    if exact_score < best_val:
        best_val = exact_score
        best_seq = a0.tolist()

    # -----------------------------------------------------------------
    # Phase 2 – smooth Adam optimisation (log‑sum‑exp)

    if best_seq is not None and time.time() < deadline - 250.0:

        smooth_val = evaluate_sequence(a_opt.tolist())

        if smooth_val < best_val:

            best_seq = a_opt.tolist()

    # -----------------------------------------------------------------
    # Phase 3 – sub‑gradient refinement
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print("\nSub‑gradient refinement")
        refined_a, _ = subgrad_optimize(np.array(best_seq, dtype=float),
                                        time_deadline=deadline - 120.0)
        refined_val = evaluate_sequence(refined_a.tolist())
        print(f"After sub‑gradient: score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_a.tolist()

    # -----------------------------------------------------------------
    # Phase 4 – final hill‑climb polishing
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_seq, _ = hill_climb_permutation(best_seq,
                                                 time_deadline=deadline,
                                                 max_steps=800_000)
        polished_val = evaluate_sequence(polished_seq)

        if polished_val < best_val:

    if best_seq is None:

        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
