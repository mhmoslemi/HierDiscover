"""
Optimiser for the “step‑function” challenge.

The algorithm proceeds in several stages while respecting the global
1000 s budget:

0️⃣  Load an optional fallback solution (``height_sequence_1``).

1️⃣  Build a large pool of parametrised families (beta‑like,
    power‑law, double‑exponential, Dirichlet, mixtures,…).
    Each candidate is normalised (sum = 1) and scored by the fast
    surrogate  ``2·n·max(conv(a,a))``.  Only the best few are kept.

2️⃣  Refine the top candidates with a long Adam optimiser that works on a
    smooth‑max surrogate (softmax of the convolution).  The temperature
    τ is increased exponentially from 0.5 up to a large value (≥ 150) so
    that the surrogate converges to the true max.

3️⃣  Upsample the current champion to larger lengths (2048 → 4096 →
    8192) and repeat the Adam refinement on each size.  Linear
    interpolation is used for up‑sampling and the vector is always
    re‑normalised to a probability simplex.

4️⃣  Run an exact Euclidean‑projected sub‑gradient descent on the true
    max‑convolution objective followed by an exponentiated‑gradient (EG)
    polish.  Both methods keep the vector on the simplex and use a
    diminishing step‑size schedule.

5️⃣  If time permits the sub‑gradient / EG pair is applied a few more
    times (alternating) – each pass usually yields a modest improvement.

Throughout the run the best sequence seen (according to the official
``evaluate_sequence`` function) is stored and finally returned as a plain
list of Python ``float`` objects.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    ``a`` must be normalised (sum = 1); the returned value equals the
    official objective.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
    (Duchi et al., 2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.25) -> np.ndarray:
    """Classic arcsine / beta(½,½) shape."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    total = vals.sum()
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_exponential_shape(n: int, lam: float) -> np.ndarray:
    """
    Symmetric double‑exponential shape:
    a_i ∝ exp(-λ·i) + exp(-λ·(n‑1‑i)).
    """
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_asymmetric_exponential_shape(
    n: int, lam_left: float, lam_right: float
) -> np.ndarray:
    """
    Asymmetric double‑exponential shape:
    a_i ∝ exp(-λ_left·i) + exp(-λ_right·(n‑1‑i)).
    """
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam_left * i) + np.exp(-lam_right * (n - 1 - i))
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_shape(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_dirichlet_shape(rng: np.random.Generator, n: int, conc: float) -> np.ndarray:
    """Draw a Dirichlet‑distributed vector with concentration ``conc``."""
    if conc <= 0:
        conc = 1.0
    v = rng.gamma(conc, 1.0, size=n)
    total = v.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return v / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(v) to target_n, followed by re‑normalisation."""
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up /= up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 30_000,
    tau_start: float = 0.5,
    tau_end: float = 150.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k)
    with an exponentially increasing temperature τ.
    The variables are logits w and a = softmax(w) guarantees non‑negativity
    and Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting point (add epsilon for log)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # a = softmax(w)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # soft‑max weights over τ·conv
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()  # shape (L,)

        # gradient wrt a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wr.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (learning‑rate decays like 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging
        if it % 10_000 == 0:
            surrogate = compute_score(a)
            print(
                f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}"
            )

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution
    objective.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries attaining the maximum (within tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) restricted to original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror‑descent with KL geometry).
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with minimal ``evaluate_sequence``.
    The routine performs a hierarchical optimisation:
      * diverse parametric seeding (beta, power, double‑exp, Dirichlet,…)
      * Adam refinement on the best few (smooth‑max surrogate)
      * progressive up‑sampling (2048 → 4096 → 8192) with further Adam
      * long Euclidean sub‑gradient descent + exponentiated‑gradient polish
      * optional additional alternating passes.
    All stages respect a global 1000 s budget and keep the best ever
    sequence according to the official ``evaluate_sequence`` function.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # ------------------------------------------------------------------
    # 0. Optional fallback
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_val: float = float("inf")
    if "height_sequence_1" in globals():
        try:
            fb = [float(x) for x in globals()["height_sequence_1"]]
            fb_val = evaluate_sequence(fb)
            print(f"Fallback (height_sequence_1) official score = {fb_val:.6f}")
            if fb_val < best_val:
                best_val, best_seq = fb_val, fb
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Helper to compute the fast surrogate on a *already* normalised vector
    # ------------------------------------------------------------------
    def surrogate_of(v: np.ndarray) -> float:
        s = v.sum()
        if s == 0.0:
            return float("inf")
        a = v / s
        return compute_score(a)

    # ------------------------------------------------------------------
    # 1. Generate a large pool of candidate shapes (n = 1024)
    # ------------------------------------------------------------------
    n_small = 1024
    pool: List[Tuple[float, np.ndarray]] = []

    # deterministic starters
    uniform = np.full(n_small, 1.0 / n_small)
    pool.append((surrogate_of(uniform), uniform.copy()))
    pool.append((surrogate_of(arcsine_shape(n_small)), arcsine_shape(n_small)))

    # randomised families
    while time.time() < deadline - 15.0:          # keep a margin for later stages
        typ = rng.integers(0, 9)
        if typ == 0:  # beta‑like
            a = rng.uniform(0.05, 3.0)
            b = rng.uniform(0.05, 3.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n_small))
            cand = generate_beta_shape(n_small, a, b, eps)
        elif typ == 1:  # power‑law
            p = rng.uniform(0.1, 4.0)
            eps = rng.uniform(0.0, 5.0)
            cand = generate_power_shape(n_small, p, eps)
        elif typ == 2:  # double‑exponential symmetric
            lam = rng.uniform(0.001, 3.0)
            cand = generate_exponential_shape(n_small, lam)
        elif typ == 3:  # double‑exponential asymmetric
            lam_l = rng.uniform(0.001, 3.0)
            lam_r = rng.uniform(0.001, 3.0)
            cand = generate_asymmetric_exponential_shape(n_small, lam_l, lam_r)
        elif typ == 4:  # Dirichlet
            conc = rng.uniform(0.05, 2.0)
            cand = generate_dirichlet_shape(rng, n_small, conc)
        elif typ == 5:  # mixture arcsine/uniform
            w = rng.random()
            cand = generate_mixture_shape(n_small, w)
        elif typ == 6:  # triangular bump
            peak = rng.integers(0, n_small)
            cand = generate_triangular_shape(n_small, peak)
        elif typ == 7:  # arcsine with random epsilon
            eps = rng.uniform(0.0, 1.0)
            cand = arcsine_shape(n_small, eps=eps)
        else:  # random jittered uniform
            jitter = rng.normal(scale=0.02, size=n_small)
            cand = uniform + jitter
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum()
        pool.append((surrogate_of(cand), cand))

    # keep the most promising seeds
    pool.sort(key=lambda x: x[0])
    top_k_small = 7
    top_seeds_small = [vec for _, vec in pool[:top_k_small]]
    print(f"\nSeeding complete – best surrogate (n={n_small}) = {pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the small seeds
    # ------------------------------------------------------------------
    refined_small: List[Tuple[float, np.ndarray]] = []
    for idx, seed_vec in enumerate(top_seeds_small, start=1):
        if time.time() >= deadline - 10.0:
            break
        print(f"\n--- Adam refinement of small seed {idx}/{len(top_seeds_small)} ---")
        # allocate a generous number of iterations while respecting time left
        remaining = deadline - time.time()
        max_iters = int(min(100_000, remaining * 30))
        a_opt = adam_optimize_smooth_max(
            seed_vec,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=180.0,
            base_lr=0.22,
        )
        sur = surrogate_of(a_opt)
        print(f"    Post‑Adam surrogate (n={n_small}) = {sur:.6f}")
        refined_small.append((sur, a_opt))

    refined_small.sort(key=lambda x: x[0])
    best_small = refined_small[0][1] if refined_small else top_seeds_small[0]

    # ------------------------------------------------------------------
    # 3. Progressive up‑sampling & Adam on larger sizes
    # ------------------------------------------------------------------
    best_large_vec = best_small
    best_large_score = surrogate_of(best_large_vec)

    for target_n in (2048, 4096, 8192):
        if time.time() >= deadline - 15.0:
            break
        print(f"\n=== Upscaling to n={target_n} and Adam refinement ===")
        up = upscale_vector(best_large_vec, target_n)

        # small random jitter before Adam (helps escape shallow traps)
        jitter = rng.normal(scale=1e-3, size=target_n)
        up = up + jitter
        up = np.clip(up, 0.0, None)
        up /= up.sum()

        remaining = deadline - time.time()
        max_iters = int(min(120_000, remaining * 30))
        a_opt = adam_optimize_smooth_max(
            up,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=250.0,
            base_lr=0.18,
        )
        sur = surrogate_of(a_opt)
        print(f"    Surrogate after up‑sample {target_n}: {sur:.6f}")

        if sur < best_large_score:
            best_large_score, best_large_vec = sur, a_opt

    # ------------------------------------------------------------------
    # 4. Exact refinement (sub‑gradient + EG) on the best large vector
    # ------------------------------------------------------------------
    if best_large_vec is not None:
        n_best = best_large_vec.shape[0]
        print(f"\n--- Exact refinement on n={n_best} ---")
        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, max(30_000, remaining * 30)))
        refined = subgradient_descent_exact(
            best_large_vec,
            deadline,
            max_iter=max_iter_sub,
            step0=0.09,
        )

        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(200_000, max(20_000, remaining * 30)))
            refined = eg_refine(
                refined,
                deadline,
                max_iter=max_iter_eg,
                eta0=0.20,
            )

        # one more alternating pass if time permits
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(200_000, max(20_000, remaining * 30)))
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=max_iter_sub,
                step0=0.07,
            )
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_eg = int(min(150_000, max(10_000, remaining * 30)))
                refined = eg_refine(
                    refined,
                    deadline,
                    max_iter=max_iter_eg,
                    eta0=0.18,
                )

        try:
            val = evaluate_sequence(refined.tolist())
            print(f"Official score after exact refinement = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print("*** New global best! ***")
        except Exception as e:
            print(f"Evaluation failed after refinement: {e}")

    # ------------------------------------------------------------------
    # 5. Final global polish (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n--- Final global polish ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, max(30_000, remaining * 30)))
        a = subgradient_descent_exact(
            a, deadline, max_iter=max_iter_sub, step0=0.05
        )

        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(180_000, max(20_000, remaining * 30)))
            a = eg_refine(a, deadline, max_iter=max_iter_eg, eta0=0.15)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6. Guard – always return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
