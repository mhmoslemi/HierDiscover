"""
A multi‑stage optimiser for the anti‑convolution max problem.

Stage 1 – a handful of deterministic “shapes” (uniform, triangular,
quadratic, sine², exponential) and many random restarts are refined
with a smooth log‑sum‑exp surrogate for the max‑convolution.
Stage 2 – the best candidate is polished by exact sub‑gradient descent.
Stage 3 – a simulated‑annealing hill‑climb that repeatedly transfers a
small amount of mass between two positions and accepts occasional
worsening moves.  All steps respect the 1000 s wall‑clock limit and
always keep the best vector seen so far.  The returned object is a plain
list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from math import log, exp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (faster than np.convolve)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the harness when ``a`` sums to 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimisation (Adam + projection)
# ----------------------------------------------------------------------


def smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 4.0,
    beta_end: float = 200.0,
    step0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a,a)) using Adam.
    Returns the best (exact) score and vector encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal β – controls smoothness of the log‑sum‑exp approximation
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed objective
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()   # weighting over all shifts

        # gradient of the smoothed max (cross‑correlation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 1000 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    # final check
    final = compute_score(a)
    if final < best_score:
        best_score = final
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact objective
    2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry (see analysis)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur = compute_score(a_new)
        if cur < best_score:
            best_score = cur
            best_a = a_new.copy()
        a = a_new

        # occasional jitter to leave plateaus
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final = compute_score(a)
    if final < best_score:
        best_score = final
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    seq: np.ndarray,
    deadline: float,
    max_moves: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a simple annealing schedule.
    Moves consist of transferring a small fraction of mass between two
    positions (or swapping two entries).  Always projects back to the simplex.
    """
    a = seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    # temperature schedule – start relatively hot to escape local minima
    start_temp = 0.02
    end_temp = 0.0002

    moves_done = 0
    while moves_done < max_moves and time.time() < deadline:
        moves_done += 1
        # linear temperature decay based on remaining budget
        frac = moves_done / max_moves
        temp = start_temp * (1 - frac) + end_temp * frac

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.08 * best_a[i]  # about 8 % of the source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score = cur
            best_a = cand
        else:
            # occasional uphill acceptance
            if np.random.rand() < exp(-(cur - best_score) / max(temp, 1e-12)):
                best_a = cand  # accept for exploration (do not update best_score)

    return best_a, best_score

def deterministic_starts(n: int) -> list[np.ndarray]:
    """Return a modest list of hand‑crafted probability vectors."""
    starts = []

    # 1. Uniform
    u = np.full(n, 1.0 / n)
    starts.append(u)

    # 2. Symmetric triangular (peak in centre)
    x = np.arange(n)
    centre = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - centre) / centre
    tri = np.maximum(tri, 0.0)
    starts.append(project_to_simplex(tri))

    # 3. Quadratic (parabolic, zero at ends)
    quad = (x * (n - 1 - x))
    starts.append(project_to_simplex(quad))

    # 4. Sine‑squared
    sine = np.sin(np.pi * (x + 0.5) / n) ** 2
    starts.append(project_to_simplex(sine))

    # 5. Exponential decay (mass near the left edge)
    tau = n / 8.0
    exp_decay = np.exp(-x / tau)
    starts.append(project_to_simplex(exp_decay))

    # 6. Reverse exponential (mass near right edge)
    exp_rev = np.exp(-(n - 1 - x) / tau)
    starts.append(project_to_simplex(exp_rev))

    # 7. Two‑Gaussian bumps
    sigma = n / 20.0
    gauss1 = np.exp(-((x - n * 0.3) ** 2) / (2 * sigma ** 2))
    gauss2 = np.exp(-((x - n * 0.7) ** 2) / (2 * sigma ** 2))
    bumps = gauss1 + gauss2
    starts.append(project_to_simplex(bumps))

    return starts

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  The routine respects the 1000 s wall‑clock
    budget and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback if a previous good construction exists
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a working length – larger gives more freedom, but we keep
    # it modest to stay inside the 1000 s budget.
    # ------------------------------------------------------------------
    n = 8192
    if (deadline - time.time()) > 300 and budget_s >= 900:
        # if we happen to have a lot of time, try a slightly larger problem
        n = 12288
    print(f"Working sequence length = {n}")

    # ------------------------------------------------------------------
    # Helper – generate a random start (dense or sparse)
    # ------------------------------------------------------------------
    def random_start() -> np.ndarray:
        """Return a random probability vector on the simplex."""
        if np.random.rand() < 0.65:               # dense start
            a = np.random.rand(n)
            a = a / a.sum()
        else:                                      # sparse start
            k = max(2, int(np.round(np.random.uniform(0.01, 0.2) * n)))
            idx = np.random.choice(n, k, replace=False)
            a = np.zeros(n, dtype=float)
            a[idx] = 1.0 / k
        return a

    # ------------------------------------------------------------------
    # 1. Deterministic + random restarts with smoothed optimisation
    # ------------------------------------------------------------------
    stage1_deadline = deadline - 250.0
    print("\n--- Stage 1: smoothed optimisation --------------------------------")
    start_vectors = deterministic_starts(n) + [random_start() for _ in range(8)]

    for idx, a0 in enumerate(start_vectors):
        if time.time() >= stage1_deadline:
            break
        # give each start a modest time slice
        slice_deadline = min(stage1_deadline,
                             time.time() + 30.0)  # ≤30 s per start
        a_opt, scr = smooth_optimize(
            a0,
            deadline=slice_deadline,
            max_iters=100_000,
            beta_start=5.0,
            beta_end=250.0,
            step0=0.95,
        )
        if scr < best_val:
            best_val = scr
            best_seq = a_opt.tolist()
            print(f"New best after deterministic/random start {idx}: score={best_val:.6f}")

    # Continue with pure random restarts as long as we have time
    while time.time() < stage1_deadline:
        a0 = random_start()
        slice_deadline = min(stage1_deadline,
                             time.time() + 35.0)  # keep a safe margin
        a_opt, scr = smooth_optimize(
            a0,
            deadline=slice_deadline,
            max_iters=150_000,
            beta_start=6.0,
            beta_end=300.0,
            step0=0.9,
        )
        if scr < best_val:
            best_val = scr
            best_seq = a_opt.tolist()
            print(f"Random restart improvement: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Exact sub‑gradient refinement on the best vector so far
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print("\n--- Stage 2: sub‑gradient refinement ---------------------------")
        refined_a, refined_score = subgradient_optimize(
            np.array(best_seq, dtype=float),
            deadline=deadline - 80.0,
            max_iter=400_000,
            step0=0.06,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Improved by sub‑grad: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        print("\n--- Stage 3: hill‑climb (simulated annealing) ---------------")
        polished_a, polished_score = hill_climb(
            np.array(best_seq, dtype=float),
            deadline=deadline - 5.0,
            max_moves=1_500_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_a.tolist()
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a modest uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (uses the harness‑provided function)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
