def logit_smooth_optimize(start_a, time_deadline, beta_start=1.0, beta_end=300.0, step0=0.8, adam_beta1=0.9, adam_beta2=0.999, adam_eps=1e-8):
    a = start_a.copy()
    # convert to logits
    y = np.log(np.maximum(a, 1e-12))
    m, v = np.zeros_like(y), np.zeros_like(y)
    best_y = y.copy()
    best_score = compute_score(a)
    for it in range(1, 1_000_000):
        if time.time() >= time_deadline:
            break
        beta = beta_start * (beta_end / beta_start) ** ((it - 1) / 100000)  # something
        # compute a from y
        max_y = np.max(y)  # for numerical stability
        exp_y = np.exp(y - max_y)
        a = exp_y / exp_y.sum()
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()
        # gradient wrt a
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]
        # gradient wrt y via softmax Jacobian
        dot = np.dot(grad_a, a)  # a·grad_a
        grad_y = a * (grad_a - dot)
        # Adam update
        m = adam_beta1 * m + (1 - adam_beta1) * grad_y
        v = adam_beta2 * v + (1 - adam_beta2) * (grad_y * grad_y)
        m_hat = m / (1 - adam_beta1 ** it)
        v_hat = v / (1 - adam_beta2 ** it)
        step = step0 / np.sqrt(it)
        y = y - step * m_hat / (np.sqrt(v_hat) + adam_eps)
        # Evaluate exact score periodically
        if it % 250 == 0:
            # compute only a from current y
            max_y = np.max(y)
            exp_y = np.exp(y - max_y)
            a_cur = exp_y / exp_y.sum()
            cur_score = compute_score(a_cur)
            if cur_score < best_score:
                best_score = cur_score
                best_y = y.copy()
    # final best
    max_y = np.max(best_y)
    exp_y = np.exp(best_y - max_y)
    best_a = exp_y / exp_y.sum()
    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = best_a
    return best_a, best_score

"""
A fast three‑stage optimiser for the “minimise max convolution’’ problem.

Stage 1 – a continuation of a smooth log‑sum‑exp objective.
          The probabilities are parametrised by logits (softmax), so the
          simplex constraint is enforced automatically.  Adam momentum
          with an annealed smoothing parameter (β from 1 → 300) drives the
          solution toward a flat self‑convolution.

Stage 2 – projected sub‑gradient descent on the exact ℓ∞‑objective.
          This cleans up the remaining overshoots of the smooth phase.

Stage 3 – a Simulated‑Annealing hill‑climber that moves small amounts
          of mass between two coordinates (or swaps them) while accepting
          occasional worsening moves.  The temperature cools slowly,
          allowing the algorithm to escape shallow local minima.

The code adapts the sequence length to the remaining budget
(4096, 8192 or 16384 entries) and always returns the best vector found
as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Stage 1 – Smooth log‑sum‑exp optimisation (softmax parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize_softmax(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 250_000,
    beta_start: float = 1.0,
    beta_end: float = 300.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)).

    The probability vector a is represented as softmax(logits).  The
    smoothing parameter β is increased gradually (continuation method).
    Adam momentum is used; the best exact score encountered is kept.
    """
    n = start_a.shape[0]

    # initialise logits safely (avoid log(0))
    eps = 1e-12
    y = np.log(np.maximum(start_a, eps))

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_y = y.copy()
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β (log‑linear schedule)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # probability vector from logits (numerically stable softmax)
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        a = exp_y / exp_y.sum()

        # ---- smoothed max (log‑sum‑exp) ----
        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                          # probability over sums

        # ---- gradient w.r.t. a (cross‑correlation) ----
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # ---- gradient w.r.t. logits (softmax Jacobian) ----
        dot = np.dot(grad_a, a)                          # a·∇a
        grad_y = a * (grad_a - dot)

        # ---- Adam update ----
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad_y
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        y = y - step * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ---- periodic exact evaluation ----
        if it % 250 == 0:
            y_max = np.max(y)
            exp_y = np.exp(y - y_max)
            a_cur = exp_y / exp_y.sum()
            cur_score = compute_score(a_cur)
            if cur_score < best_score:
                best_score = cur_score
                best_y = y.copy()
                best_a = a_cur.copy()

    # final evaluation of the best logits
    y_max = np.max(best_y)
    exp_y = np.exp(best_y - y_max)
    best_a = exp_y / exp_y.sum()
    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = best_a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))            # index of the maximal coefficient

        # sub‑gradient of the maximal entry (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # small jitter every 1000 iterations to leave plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – Simulated‑annealing hill‑climb (tiny mass moves)
# ----------------------------------------------------------------------


def hill_climb_sa(
    seq,
    time_deadline,
    max_steps: int = 600_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb with a simple Metropolis acceptance rule.
    Moves consist of transferring a small fraction of mass between two
    coordinates (or swapping them).  The “temperature’’ is the
    fraction of mass moved; it slowly cools.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    temperature = 0.10  # initial mass‑transfer fraction

    step = 0
    while step < max_steps and time.time() < time_deadline:
        cand = best_a.copy()

        # ---- propose a move -------------------------------------------------
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = temperature * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        # ---- Metropolis acceptance -----------------------------------------
        if cur_score < best_score:
            accept = True
        else:
            # probability of uphill move decreases with temperature
            prob = np.exp(-(cur_score - best_score) / (temperature * best_score + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            best_a = cand
            best_score = cur_score

        # ---- cooling --------------------------------------------------------
        if step % 1000 == 0 and step > 0:
            temperature = max(temperature * 0.995, 0.001)

        step += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Helper – generate deterministic start shapes
# ----------------------------------------------------------------------


def deterministic_starts(n: int) -> list[np.ndarray]:
    """Return a handful of hand‑crafted probability vectors."""
    starts = []

    # 1) uniform
    starts.append(np.full(n, 1.0 / n))

    # 2) triangular (peak in the middle)
    idx = np.arange(n)
    tri = np.minimum(idx, n - 1 - idx) + 1.0
    starts.append(project_to_simplex(tri))

    # 3) sine‑squared shape (more mass near the ends)
    sin = np.sin(np.pi * (idx + 0.5) / n)  # always positive
    starts.append(project_to_simplex(sin))

    # 4) exponential decay from the left edge
    exp_left = np.exp(-5.0 * idx / n)
    starts.append(project_to_simplex(exp_left))

    # 5) exponential decay from both edges (U‑shaped)
    exp_both = np.exp(-5.0 * np.abs(idx - (n - 1) / 2) / n)
    starts.append(project_to_simplex(exp_both))

    return starts

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises the harness‑provided
    `evaluate_sequence`.  The routine respects the 1000 s wall‑clock limit
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------------------
    # Choose a reasonable length (larger gives more freedom)
    # -------------------------------------------------------------
    # Very short runs -> 4096, medium -> 8192, long -> 16384
    if deadline - time.time() > 700:
        n = 16384
    elif deadline - time.time() > 300:
        n = 8192
    else:
        n = 4096
    print(f"[Info] Chosen sequence length = {n}")

    # -------------------------------------------------------------
    # Assemble a collection of start vectors
    # -------------------------------------------------------------
    starts = deterministic_starts(n)

    # add a few random starts (dense)
    for _ in range(12):
        rnd = np.random.rand(n)
        starts.append(rnd / rnd.sum())

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # 0. quick evaluation of the raw starts – pick the best one
    # -------------------------------------------------------------
    for a in starts:
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    print(f"[Info] Best raw start score = {best_val:.6f}")

    # -------------------------------------------------------------
    # 1. Smooth log‑sum‑exp optimisation (softmax + Adam)
    # -------------------------------------------------------------
    # we repeat the smooth phase several times, each time using the
    # current best as a warm start (plus an occasional jitter).
    while time.time() < deadline - 180.0:          # reserve ~3 min for later phases
        remaining = deadline - time.time()
        # allocate at most 30 s for a single run (or the remaining budget)
        time_for_run = min(30.0, remaining - 150.0)
        if time_for_run <= 0:
            break

        # jitter the start a little to diversify the search
        start_vec = best_seq.copy()
        if np.random.rand() < 0.25:
            start_vec += 0.01 * np.random.randn(n)
            start_vec = np.maximum(start_vec, 0.0)
            start_vec = project_to_simplex(start_vec)

        cand_seq, cand_score = smooth_optimize_softmax(
            start_vec,
            time_deadline=time.time() + time_for_run,
            max_iters=200_000,
            beta_start=1.0,
            beta_end=350.0,
            step0=0.9,
        )
        if cand_score < best_val:
            best_val = cand_score
            best_seq = cand_seq.copy()
            print(f"[Smooth] New best score = {best_val:.6f}")

    # -------------------------------------------------------------
    # 2. Sub‑gradient refinement (exact ℓ∞ objective)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print("\n[Info] Starting sub‑gradient refinement")
        refined_seq, refined_score = subgrad_optimize(
            best_seq,
            time_deadline=deadline - 60.0,
            max_iter=250_000,
            step0=0.09,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq.copy()
            print(f"[SubGrad] Improved best to {best_val:.6f}")

    # -------------------------------------------------------------
    # 3. Final hill‑climb polishing (simulated annealing)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n[Info] Starting hill‑climb polishing")
        polished_seq, polished_score = hill_climb_sa(
            best_seq,
            time_deadline=deadline,
            max_steps=800_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = np.array(polished_seq, dtype=float)
            print(f"[HillClimb] Improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safety fallback – if nothing succeeded return a modest uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small)
        best_val = compute_score(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation (uses the harness‑provided function)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[Done] Final reported score = {final_score:.6f}")

    # The harness expects a single list of floats.
    return best_seq.tolist()
