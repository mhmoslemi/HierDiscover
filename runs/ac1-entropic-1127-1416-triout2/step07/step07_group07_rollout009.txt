def adam_descent(a0, deadline, max_iter, step0):

    a = copy

    m, v = zeros.

    for t in range(1, max_iter+1):
        if time.time() >= deadline: break

        conv = conv_fft(a,a)

        maxc = np.max(conv)

        # epsilon schedule
        # start with epsilon0 = maxc * 0.5
        # decays by factor 0.999 each iter, min epsilon = maxc * 1e-6
        if t == 1:
            eps0 = maxc * 0.5
        eps = max(eps0 * (0.999**(t-1)), maxc * 1e-6)

        grad = softmax_gradient(a, eps)

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)

        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)

        step = step0 / (1+ t*0.0) # maybe keep constant? We'll keep decaying sqrt(t) as before.

        a -= step * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = project_to_simplex(a)

        # maybe enforce symmetry every 1000 steps

            a = final_symmetry_average(a)

        # track best
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

def adam_descent(a0, deadline, max_iter=150000, step0=0.5, eps_factor=0.5, eps_min_factor=1e-6):

    beta1=0.9; beta2=0.999; eps_adam=1e-8

    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter+1):
        if time.time() >= deadline: break
        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        eps = max(eps_factor * maxc * (0.99 ** (t-1)), maxc * eps_min_factor)

        grad = softmax_gradient(a, eps)

        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)

        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        step = step0 / np.sqrt(t)

        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        # enforce symmetry every 1000 iterations
        if t % 1000 == 0:
            a_new = final_symmetry_average(a_new)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            # optionally print
            if t % 5000 == 0:
                print(f"[Adam] iter {t} best_sc={best_sc:.6f}")

        a = a_new

    return best_a, best_sc

"""
Step‑function optimisation for `evaluate_sequence`.

The algorithm combines three ideas that proved useful for this task:

1.  **Parametric start** –  a symmetric power‑law family
    a_i ∝ ((i+ε)(n‑1‑i+ε))^{‑p}.  A cheap ternary search finds the exponent
    p that yields the smallest value of the smooth objective
    2·n·max(a⋆a).  This gives a more favourable seed than a plain arcsine.

2.  **Adam on a smooth soft‑max surrogate** –  we minimise
    f_ε(a)=ε·log∑_k exp((a⋆a)_k/ε).  The gradient is
    ∇f_ε(a)=2·(softmax(conv/ε) reversed ⊛ a).  ε is reduced
    exponentially (down to 10⁻⁶·max(conv)), which makes the surrogate
    converge to the true max while keeping a useful gradient.

3.  **Long sub‑gradient fine‑tuning** –  after the smooth phase we
    switch to the exact sub‑gradient of the true objective
    (taken at the shift that realises the maximum).  A large number of
    iterations with a diminishing step‑size finishes the optimisation.
    All iterates are projected onto the probability simplex and,
    every few thousand steps, forced to be (approximately) symmetric.

The code respects the 1000 s budget, prints progress, and finally
returns a plain Python list ready for `evaluate_sequence`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a)=1.
    Returns the value of the objective 2·n·max(a⋆a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Enforce (approx.) symmetry: (a+reverse(a))/2 and re‑project."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Parametric family (power‑law) initialisation
# ----------------------------------------------------------------------


def powerlaw_vector(n: int, p: float) -> np.ndarray:
    """Create a symmetric power‑law vector and project it onto the simplex."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return project_to_simplex(w)


def best_powerlaw_initialisation(n: int, time_deadline: float) -> np.ndarray:
    """
    Search the exponent p ∈ [0.1, 1.2] with a cheap ternary search.
    Returns the vector that gives the smallest `compute_score`.
    """
    lo, hi = 0.1, 1.2
    best_a = None
    best_sc = float("inf")
    # limit the number of evaluations (≈30) to stay well within the budget
    for _ in range(30):
        p1 = lo + (hi - lo) / 3.0
        p2 = hi - (hi - lo) / 3.0
        a1 = powerlaw_vector(n, p1)
        sc1 = compute_score(a1)
        a2 = powerlaw_vector(n, p2)
        sc2 = compute_score(a2)

        if sc1 < sc2:
            hi = p2
            if sc1 < best_sc:
                best_sc = sc1
                best_a = a1
        else:
            lo = p1
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a2

        # abort early if time is gone
        if time.time() >= time_deadline - 2.0:
            break

    # final refinement at the centre of the last interval
    p_mid = (lo + hi) / 2.0
    a_mid = powerlaw_vector(n, p_mid)
    sc_mid = compute_score(a_mid)
    if sc_mid < best_sc:
        best_sc, best_a = sc_mid, a_mid

    return best_a


# ----------------------------------------------------------------------
# Soft‑max (smooth max) gradient utilities
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f(a) = eps·log Σ_k exp(conv_k/eps)
    with conv = a⋆a.

    Returns  grad = 2·(w_rev ⊛ a), where w_i = softmax(conv_i/eps).
    """
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)                 # Σ w = 1

    # Cross‑correlation w_rev ⊛ a  (length n + (2n‑1) – 1 = 3n‑2)
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)

    start = a.shape[0] - 1               # index where shift 0 aligns
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def adam_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.5,
    eps_factor: float = 0.5,
    eps_min_factor: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth soft‑max objective.
    ε is driven exponentially down to a very small fraction of the current
    maximum convolution, which forces the surrogate to approximate the true
    max.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε schedule – exponential decay, never below eps_min_factor·maxc
        eps = max(eps_factor * maxc * (0.99 ** (t - 1)), maxc * eps_min_factor)

        grad = softmax_gradient(a, eps)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)

        # enforce symmetry regularly (helps the landscape)
        if t % 1000 == 0:
            a_new = final_symmetry_average(a_new)

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            if t % 5000 == 0:
                print(f"[Adam] iter {t:6d}  best_score={best_sc:.6f}")

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on 2·n·max(a⋆a).
    The sub‑gradient is taken at the shift that currently attains the maximum.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i}  (zero outside the support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        # re‑impose symmetry every few thousand steps
        if t % 2000 == 0:
            a_new = final_symmetry_average(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            if t % 10000 == 0:
                print(f"[Sub‑grad] iter {t:6d}  best_score={best_sc:.6f}")

        a = a_new

        # occasional small random perturbation to escape plateaus
        if t % 15000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc

def simulated_annealing_hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climber that occasionally accepts worsening moves.
    Temperature decays linearly; moves consist of a fractional mass transfer.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    start = time.time()

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule (starts at 10 % of current best)
        T = best_sc * 0.10 * (1.0 - step / max_steps)

        i, j = np.random.randint(n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move a random fraction (up to 30 %) of the source mass
        frac = np.random.rand() * 0.30
        delta = frac * a[i]

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cur_sc = compute_score(cand)
        delta_sc = cur_sc - best_sc

        if delta_sc < 0.0 or (T > 0.0 and np.random.rand() < np.exp(-delta_sc / T)):
            a = cand
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()
                if step % 25000 == 0:
                    print(f"[HC] step {step:6d}  new best={best_sc:.6f}")

        # enforce symmetry from time to time
        if step % 20000 == 0:
            a = final_symmetry_average(a)

    return best_a, best_sc
