"""
Propose a step‑function sequence that minimises the official evaluation score.

The optimiser works in several stages:

0️⃣  Load a previously saved solution (if present) as a safety fallback.
1️⃣  Choose a moderate length (default n=4096) and generate a diverse pool of
    analytically‑parameterised candidate shapes (arcsine, uniform,
    power‑type, beta‑type, mixtures).  All candidates are normalised to sum = 1.
2️⃣  Keep the best few candidates and refine each of them with a long Adam
    optimisation on a smooth‑max surrogate (log‑sum‑exp of the convolution).
    The temperature τ is raised exponentially up to a large value so that the
    surrogate converges to the true max.
3️⃣  Take the best refined vector and polish it with an exact sub‑gradient
    descent on the true max‑convolution objective, followed by an
    exponentiated‑gradient (mirror‑descent) stage.
4️⃣  If time permits, up‑sample the current best vector to a larger size
    (n=8192) and repeat the Adam + sub‑gradient + EG refinement.
5️⃣  A final cheap random‑local‑search (mass swaps) is performed while time
    remains, accepting only strictly improving moves.
6️⃣  The best ever sequence (according to the official ``evaluate_sequence``
    function) is returned.

All stages respect the global 1000 s budget (a small safety margin is kept)
and the algorithm stays entirely in‑memory, using only NumPy and the provided
``evaluate_sequence`` routine.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two 1‑D arrays via real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector a is expected to be normalised (sum = 1), so the denominator
    disappears and the score reduces to 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_asymmetric_power_shape(
    n: int, p: float, eps_left: float, eps_right: float
) -> np.ndarray:
    """
    Asymmetric generalised arcsine:
    a_i ∝ 1 / ((i+eps_left)*(n‑½‑i+eps_right))**p .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + eps_left) * (n - 0.5 - i + eps_right) + 1e-12) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from length len(v) to target_n and renormalise."""
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    tau_start: float = 0.5,
    tau_end: float = 200.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k)
    The temperature τ is increased exponentially from ``tau_start`` to ``tau_end``.
    The variables are logits w with a = softmax(w),
    guaranteeing non‑negativity and Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting point
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (exponential)
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()  # shape (L,)

        # gradient w.r.t. a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # back‑prop through softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress log
        if it % 20000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution
    objective.  The vector is kept on the probability simplex.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of (approximately) maximal entries
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) restricted to original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    eta0: float = 0.22,
) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent with KL geometry) polish.
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with minimal ``evaluate_sequence``.
    The routine follows the pipeline described in the module docstring and
    respects the global time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.5

    best_seq: List[float] = []
    best_val: float = float("inf")   # surrogate (same as official if sum=1)

    # ------------------------------------------------------------------
    # 0. Load a fallback if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Helper to keep the globally best candidate (surrogate)
    # ------------------------------------------------------------------
    def maybe_update(vec: np.ndarray) -> None:
        nonlocal best_seq, best_val
        s = compute_score(vec)          # surrogate = official when sum=1
        if s < best_val:
            best_val = s
            best_seq = vec.tolist()
            print(f"*** New best surrogate {s:.6f} ***")

    # ------------------------------------------------------------------
    # 1. Choose a working size and generate a pool of analytic shapes
    # ------------------------------------------------------------------
    n_main = 4096
    print(f"\n--- Generating initial pool for n={n_main} ---")
    pool: List[Tuple[float, np.ndarray]] = []

    # deterministic starters
    pool.append((compute_score(arcsine_initial(n_main)), arcsine_initial(n_main)))
    pool.append((compute_score(np.full(n_main, 1.0 / n_main)), np.full(n_main, 1.0 / n_main)))

    # systematic sweep of power‑type shapes
    for p in np.linspace(0.35, 1.10, 13):
        for eps in (0.0, 0.5, 1.0, 2.0):
            cand = generate_power_shape(n_main, p, eps)
            pool.append((compute_score(cand), cand))

    # random beta‑like shapes
    for _ in range(250):
        if time.time() >= deadline - 5.0:
            break
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        eps = rng.uniform(0.0, 2.0)
        cand = generate_beta_shape(n_main, a, b, eps)
        pool.append((compute_score(cand), cand))

    # random mixtures of arcsine & uniform
    for _ in range(150):
        if time.time() >= deadline - 5.0:
            break
        w = rng.random()
        cand = generate_mixture_shape(n_main, w)
        pool.append((compute_score(cand), cand))

    # keep the best few for expensive refinement
    pool.sort(key=lambda x: x[0])
    best_seeds = [vec for _, vec in pool[:5]]
    print(f"Best surrogate after seeding: {pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the selected seeds
    # ------------------------------------------------------------------
    refined_vectors: List[Tuple[float, np.ndarray]] = []
    for idx, seed_vec in enumerate(best_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Adam refinement on seed {idx}/{len(best_seeds)} ---")
        remaining = deadline - time.time()
        # rough estimate: aim for ~30 iterations per second
        max_iters = int(min(180_000, remaining * 30))
        opt_vec = adam_optimize_smooth_max(
            seed_vec,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=250.0,
            base_lr=0.25,
        )
        maybe_update(opt_vec)
        refined_vectors.append((compute_score(opt_vec), opt_vec))

    refined_vectors.sort(key=lambda x: x[0])
    current_vec = refined_vectors[0][1] if refined_vectors else best_seeds[0]

    # ------------------------------------------------------------------
    # 3. Exact sub‑gradient + exponentiated‑gradient polishing (same n)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\n--- Sub‑gradient descent (exact max) ---")
        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, remaining * 25))
        current_vec = subgradient_descent_exact(
            current_vec,
            deadline,
            max_iter=max_iter_sub,
            step0=0.10,
        )
        maybe_update(current_vec)

    if time.time() < deadline - 2.0:
        print("\n--- Exponentiated‑gradient polish ---")
        remaining = deadline - time.time()
        max_iter_eg = int(min(180_000, remaining * 25))
        current_vec = eg_refine(
            current_vec,
            deadline,
            max_iter=max_iter_eg,
            eta0=0.20,
        )
        maybe_update(current_vec)

    # ------------------------------------------------------------------
    # 4. Upscale to a larger size (if time remains) and repeat refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        for target_n in (8192,):
            if time.time() >= deadline - 5.0:
                break
            print(f"\n=== Upscaling to n={target_n} and refining ===")
            up_vec = upscale_vector(current_vec, target_n)

            # tiny random jitter before the next optimisation stage
            jitter = rng.normal(scale=1e-3, size=target_n)
            up_vec = np.clip(up_vec + jitter, 0.0, None)
            up_vec /= up_vec.sum()

            remaining = deadline - time.time()
            max_iters = int(min(120_000, remaining * 25))
            up_vec = adam_optimize_smooth_max(
                up_vec,
                deadline,
                max_iters=max_iters,
                tau_start=0.5,
                tau_end=300.0,
                base_lr=0.22,
            )
            maybe_update(up_vec)

            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_sub = int(min(150_000, remaining * 20))
                up_vec = subgradient_descent_exact(
                    up_vec,
                    deadline,
                    max_iter=max_iter_sub,
                    step0=0.08,
                )
                maybe_update(up_vec)

            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iter_eg = int(min(120_000, remaining * 20))
                up_vec = eg_refine(
                    up_vec,
                    deadline,
                    max_iter=max_iter_eg,
                    eta0=0.18,
                )
                maybe_update(up_vec)

            current_vec = up_vec  # keep the latest up‑scaled vector

    # ------------------------------------------------------------------
    # 5. Final cheap random local search (mass‑swap hill climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Final random local search ---")
        vec = np.array(best_seq, dtype=float) if best_seq else current_vec.copy()
        vec = vec / vec.sum()
        n = vec.shape[0]

        while time.time() < deadline - 1.0:
            # pick two distinct positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if vec[i] == 0.0:
                continue
            # move a random fraction of the mass at i to j
            frac = rng.random() * 0.4        # up to 40 % of the source mass
            delta = frac * vec[i]
            new_vec = vec.copy()
            new_vec[i] -= delta
            new_vec[j] += delta
            new_vec /= new_vec.sum()
            s_new = compute_score(new_vec)
            if s_new < best_val:
                vec = new_vec
                maybe_update(vec)

        # make sure the final best is stored
        current_vec = vec

    # ------------------------------------------------------------------
    # 6. Final official evaluation (just to report the true score)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nFinal official score = {final_official:.6f}")
    except Exception as exc:
        print(f"Official evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # 7. Guard – always return a non‑empty normalised list
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0 / n_main] * n_main
        _ = evaluate_sequence(best_seq)  # just to keep the same flow

    print(f"\nSearch finished. Best surrogate = {best_val:.6f}")
    return best_seq
