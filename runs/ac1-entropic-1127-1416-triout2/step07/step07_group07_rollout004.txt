```
def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start=time.time()
    deadline = start + budget_s - 2.0

    best_seq=None
    best_score=inf

    # fallback to known sequence
    try: 
        fb = list(height_sequence_1)
        fb_score = evaluate_sequence(fb)
        if fb_score < best_score:
            best_score=fb_score
            best_seq=fb
    except Exception: pass

    # length options - we can try 4096-32768.
    length_options=[16384,8192,4096,2048]
    outer_iter=0

    while time.time() < deadline - 5 and outer_iter < 30:
        outer_iter += 1

        n = int(np.random.choice(length_options))

        # random initialization method
        init_type = np.random.choice(['arcsine','beta','sparse'])
        if init_type == 'arcsine':
            a = arcsine_initialisation(n)
        elif init_type == 'beta':
            alpha = np.random.uniform(0.2, 0.8)
            beta = np.random.uniform(0.2, 0.8)
            a = beta_initialisation(n, alpha, beta)
        else:
            a = random_sparse_vector(n)

        # optionally perturb small random noise
        a = (a + np.random.rand(n)*1e-6)
        a = project_to_simplex(a)

        cur_score = compute_score(a)
```

The task is to minimise  

\[
\text{score}= \frac{2\,n\;\max (a\star a)}{(\sum a)^2},
\]

with a non‑negative step‑function `a`.  
All vectors are kept on the probability simplex (`∑a = 1`), therefore the
objective simplifies to  

\[
\text{score}=2\,n\;\max (a\star a) .
\]

The optimiser consists of four stages.

1. **Initialisation** – several families of probability vectors are sampled
   (arcsine, Beta‑shaped and sparse random).  
2. **Smooth soft‑max descent** – Adam optimiser on the smooth
   log‑sum‑exp surrogate of the maximum.  The temperature `ε` is reduced
   aggressively down to 0.1 % of the current maximum, providing a very
   accurate gradient of the true max.
3. **Exact multi‑sub‑gradient descent** – the true maximum is
   approximated by all conv‑positions that are within a tiny tolerance of
   the current max.  The averaged sub‑gradient is used together with a
   decreasing step‑size and projection onto the simplex.
4. **Simulated‑annealing hill‑climber** – stochastic mass transfers are
   accepted according to a temperature schedule, allowing the search to
   escape shallow local minima.  A final symmetry‑averaging step is applied.

All stages are time‑bounded; the algorithm keeps the best vector found
so far and returns it at the end of the 1000 s budget.  The implementation
uses only `numpy` and standard library modules, runs on a single CPU core
and respects the required function signature `propose_candidate`.

```python
"""
Step‑function optimiser for the evaluation metric

    score = 2·n·max(conv)   (the sum of the coefficients is fixed to 1)

The optimiser runs in four phases:

1.  Initialise a probability vector with one of several heuristics
    (arcsine, Beta‑shape, sparse random).
2.  Adam optimiser on the smooth log‑sum‑exp surrogate of the max.
    The temperature ε is driven down to 0.1 % of the current maximum,
    yielding a very accurate gradient of the true objective.
3.  Exact multi‑sub‑gradient descent: the true max is approximated by
    every convolution entry within a small relative tolerance; the
    averaged sub‑gradient is used together with a diminishing step size.
4.  Simulated‑annealing hill‑climbing (mass transfers) with a
    temperature schedule, followed by a symmetry‑averaging step.

Each phase respects the remaining time budget; the best sequence found
is finally evaluated with `evaluate_sequence` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic building blocks – FFT convolution and simplex projection
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast and memory‑efficient)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when the vector is on the simplex.
    Returns 2·n·max(conv)  (the exact objective value for sum(a)=1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine‑like density: a_i ∝ 1/√(i·(n‑1‑i)).
    Symmetric, already yields a relatively flat self‑convolution.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def beta_initialisation(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑shaped density: a_i ∝ (i/(n‑1))^{α‑1}·(1‑i/(n‑1))^{β‑1}.
    Allows exploration of a wide family of edge‑biased shapes.
    """
    i = np.arange(n, dtype=float)
    x = i / (n - 1)
    eps = 1e-12
    w = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    return project_to_simplex(w)


def random_sparse_vector(n: int) -> np.ndarray:
    """
    Random probability vector with a moderate number of non‑zero entries.
    The density is drawn uniformly from [0.15, 0.85].
    """
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 1. Smooth soft‑max descent (Adam optimiser)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f(a) = eps·log Σ_k exp(conv_k/eps)
    where conv = a⋆a.
    Returns    grad = 2·(w_rev ⊛ a)  with  w = softmax(conv/eps).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)                       # Σ w = 1

    # cross‑correlation w_rev ⊛ a gives Σ_j a_j·w_{i+j}
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)

    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 4000,
    step0: float = 0.2,
    eps_factor: float = 0.25,
    min_eps_factor: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    The temperature ε is decreased exponentially; it never falls below
    ``min_eps_factor``·current_max.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature – never smaller than *min_eps_factor* of the current max
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), min_eps_factor)

        grad = softmax_gradient(a, eps)

        # Adam update (biased estimates)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)          # basic decay
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to escape plateaux
        if t % 500 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact multi‑sub‑gradient descent (max‑conv only)
# ----------------------------------------------------------------------


def multistep_subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
    tol_factor: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true objective 2·n·max(a⋆a).
    The sub‑gradient is averaged over all convolution entries that are
    within ``tol_factor``·maxc of the current maximum, which yields a
    smoother descent direction.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        tol = maxc * tol_factor
        max_inds = np.where(conv >= maxc - tol)[0]

        # average sub‑gradient over the selected indices
        g = np.zeros_like(a)
        for k in max_inds:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += a[j[mask]]
        if max_inds.size > 0:
            g = 2.0 * g / max_inds.size

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random shake every few thousand iterations
        if t % 3000 == 0:
            a += np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Simulated‑annealing hill‑climber (mass transfers)
# ----------------------------------------------------------------------


def simulated_annealing_hillclimb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climber that accepts worsening moves with probability
    exp(−Δ/Temperature).  Temperature decays geometrically.
    Moves consist of swaps or small mass transfers.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    # step fraction controls how much mass is moved
    step_frac = 0.08
    # initial temperature (a few percent of the current score)
    temperature = max(1e-8, 0.01 * cur_sc)
    decay = 0.99996  # per step

    for _ in range(max_steps):
        if time.time() >= time_deadline:
            break

        # propose a candidate
        if np.random.rand() < 0.5:
            # swap two indices
            i, j = np.random.randint(n, size=2)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            cand = a.copy()
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta

        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)
        delta_sc = cand_sc - cur_sc

        accept = False
        if delta_sc < 0:
            accept = True
        else:
            # probabilistic acceptance of uphill moves
            if temperature > 0 and np.random.rand() < np.exp(-delta_sc / temperature):
                accept = True

        if accept:
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                # successful move → become a little bolder
                step_frac = min(step_frac * 1.07, 0.5)
            else:
                # unsuccessful move → become more cautious
                step_frac = max(step_frac * 0.96, 0.001)

        # temperature schedule
        temperature *= decay

    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Symmetry averaging (cheap but often beneficial)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a+reverse(a))/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` score.  The routine respects the given time budget
    (≈ 1000 s) and returns a plain Python list ready for the evaluation
    harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Optional fallback on a previously known construction
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score, best_seq = fb_score, fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Main stochastic search – several outer restarts
    # --------------------------------------------------------------
    lengths = [16384, 8192, 4096, 2048]   # larger sizes give more freedom
    max_outer = 25                       # capped by the time budget

    outer = 0
    while outer < max_outer and time.time() < deadline - 5.0:
        outer += 1

        n = int(np.random.choice(lengths))

        # ----------------------------------------------------------
        # 1a. Random initial vector
        # ----------------------------------------------------------
        init_kind = np.random.choice(["arcsine", "beta", "sparse"])
        if init_kind == "arcsine":
            a = arcsine_initialisation(n)
        elif init_kind == "beta":
            alpha = np.random.uniform(0.2, 0.8)
            beta = np.random.uniform(0.2, 0.8)
            a = beta_initialisation(n, alpha, beta)
        else:
            a = random_sparse_vector(n)

        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score, best_seq = cur_sc, a.tolist()
            print(f"[{outer}] init n={n} kind={init_kind}  score={cur_sc:.6f}")

        # ----------------------------------------------------------
        # 1b. Adam‑accelerated soft‑max gradient descent
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        # allocate roughly 35 % of the remaining time to this phase
        gd_deadline = time.time() + remaining * 0.35
        # Rough estimate: ~20–30 gradient evaluations per second
        max_iters = int(min(15000, (gd_deadline - time.time()) * 25))
        a_gd, _ = adam_softmax_descent(
            a0=a,
            time_deadline=gd_deadline,
            max_iter=max_iters,
            step0=0.30,
            eps_factor=0.25,
            min_eps_factor=0.001,
        )
        sc_gd = compute_score(a_gd)
        if sc_gd < best_score:
            best_score, best_seq = sc_gd, a_gd.tolist()
            print(f"[{outer}] after Adam GD n={n}  score={sc_gd:.6f}")

        # ----------------------------------------------------------
        # 1c. Exact multi‑sub‑gradient descent (fine‑tuning)
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 6.0:
            sub_deadline = time.time() + remaining * 0.30
            max_iters_sub = int(min(80000, (sub_deadline - time.time()) * 25))
            a_sub, _ = multistep_subgradient_descent(
                a0=a_gd,
                time_deadline=sub_deadline,
                max_iter=max_iters_sub,
                step0=0.07,
                tol_factor=1e-3,
            )
            sc_sub = compute_score(a_sub)
            if sc_sub < best_score:
                best_score, best_seq = sc_sub, a_sub.tolist()
                print(f"[{outer}] after sub‑grad n={n}  score={sc_sub:.6f}")

        # ----------------------------------------------------------
        # 1d. Simulated‑annealing hill‑climbing + symmetry averaging
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            hc_deadline = time.time() + remaining * 0.30
            steps_hc = int(min(250_000, (hc_deadline - time.time()) * 8000))
            seq_refine = np.array(best_seq, dtype=float)
            a_hc, _ = simulated_annealing_hillclimb(
                a_start=seq_refine,
                time_deadline=hc_deadline,
                max_steps=steps_hc,
            )
            sc_hc = compute_score(a_hc)
            if sc_hc < best_score:
                best_score, best_seq = sc_hc, a_hc.tolist()
                print(f"[{outer}] after SA hill‑climb score={sc_hc:.6f}")

            # symmetry averaging – cheap and often helps
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score, best_seq = sc_sym, a_sym.tolist()
                print(f"[{outer}] symmetry average improved score={sc_sym:.6f}")

    # --------------------------------------------------------------
    # 2. Fallback – uniform vector (guaranteed finite result)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 3. Final official evaluation (uses the provided evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
```
