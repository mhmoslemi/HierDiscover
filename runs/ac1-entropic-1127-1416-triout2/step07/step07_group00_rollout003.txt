```
def hill_climb_sa(start_seq, deadline, max_steps=400000):
    a = start_seq.copy()
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = a.size
    step_scale = 0.12
    step = 0
    while step < max_steps and time.time() < deadline:
        # propose
        cand = a.copy()
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * a[i]  # move from i
                cand[i] -= delta
                cand[j] += delta
                # ensure non-negative
                if cand[i] < 0:
                    cand[i] = 0.0
        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        # improvement?
        if cand_score < best_score:
            best_score, best_a = cand_score, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # occasional uphill acceptance
            # compute temperature
            # T decays linearly with steps
            t_ratio = step / max_steps
            T = max(1e-4, (1 - t_ratio) * 0.01)  # max 0.01 decreasing
            if np.random.rand() < np.exp(-(cand_score - best_score) / T):
                a = cand
            step_scale *= 0.97
        step += 1
    return best_a, best_score
```

assistantfinal**Solution Overview**

The candidate generator follows a four‑stage pipeline that fits into the 1000 s wall‑clock budget:

1. **Diversified warm‑starts** – a quick scan of several analytic families
   (inverse‑power, Gaussian / edge‑heavy exponentials and tiny random sparse
   vectors). The best vector from this stage becomes the seed for the next
   stage.

2. **Exponentiated‑gradient smoothing** – we minimise the smooth
   approximation  

   \[
   f_\beta(a)=\frac1\beta\log\!\left(\sum_k e^{\beta\,(a*a)_k}\right)
   \]

   by mirror‑descent on the simplex (multiplicative updates
   `a_i ← a_i·exp(-η·∇_i f_β)`).  
   The smoothing parameter β is increased exponentially from 5 to 200 while
   the step size η decays as 1/√t.  This phase produces a very good
   approximation of the true optimum.

3. **Projected sub‑gradient refinement** – a classic projected sub‑gradient
   method on the original max‑objective guarantees a monotone decrease of the
   exact score.

4. **Stochastic hill‑climb with simulated‑annealing acceptance** – random
   mass‑transfer moves are accepted either if they improve the exact score or
   with a small temperature‑dependent probability, allowing the search to
   escape shallow local minima.

All heavy work uses only NumPy and FFT‑based convolutions, runs comfortably
inside the time limit and never requires external solvers.

```python
"""
Non‑negative step‑function optimiser.

The algorithm consists of four phases:
 1) generate several analytic warm‑starts (inverse‑power, exponential,
    random sparse) and keep the best;
 2) minimise a smoothed max‑objective with exponentiated‑gradient (mirror
    descent) while annealing the smoothing parameter β;
 3) projected sub‑gradient descent on the true max‑objective;
 4) long stochastic hill‑climb with simulated‑annealing acceptance.

All computation is performed with NumPy/FFT; the final result is a plain
list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution a * b using a real‑valued FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for a vector already normalised
    to sum(a) == 1.  Returns the exact score 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for warm‑starts
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    Inverse‑power family:
        a_i(α) ∝ [(i+½)(n‑½‑i)]^(−α)   (symmetric, edge‑heavy)
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def exponential_start(n: int, gamma: float) -> np.ndarray:
    """
    Gaussian / edge‑heavy exponential family:
        a_i ∝ exp(−γ·(i−c)²)   with centre c = (n‑1)/2.
    Positive γ → mass in the centre, negative γ → mass at the edges.
    """
    i = np.arange(n, dtype=float)
    c = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - c) ** 2))
    if w.sum() == 0.0:               # fallback to uniform if under‑flow occurs
        w = np.ones_like(w)
    a = w / w.sum()
    return a


def random_sparse_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a sparse probability vector with a random support size."""
    sparsity = rng.uniform(0.04, 0.5)          # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed gradient (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------


def gradient_smoothed(a: np.ndarray, beta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Gradient of the smoothed objective f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    together with the underlying convolution (needed for the exact score).
    Returns (grad, conv) where grad has shape (n,).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    maxc = conv.max()
    # soft‑max weights for the smoothing
    exp_weights = np.exp(beta * (conv - maxc))
    w = exp_weights / exp_weights.sum()
    # gradient = 2·(a_rev ⊛ w) restricted to n entries
    grad = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]
    return grad, conv


def exp_gradient_optimize(start_a: np.ndarray,
                          deadline: float,
                          max_iters: int = 300_000,
                          beta_start: float = 5.0,
                          beta_end: float = 200.0,
                          eta0: float = 0.70) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using multiplicative updates:
        a_i ← a_i·exp(−η·∇_i f_β)   then renormalise.
    The smoothing parameter β is increased exponentially from `beta_start`
    to `beta_end`.  The routine returns the best exact score encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- annealing schedule for β ----
        t_frac = t / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # ---- gradient of the smoothed objective ----
        grad, conv = gradient_smoothed(a, beta)

        # ---- step size (1/√t) ----
        step = eta0 / np.sqrt(t)

        # Clip the gradient to avoid overflow in the exponentials
        grad = np.clip(grad, -10.0, 10.0)

        # ---- mirror‑descent update (multiplicative) ----
        a = a * np.exp(-step * grad)

        # ---- re‑normalise to the simplex ----
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ---- occasional exact evaluation ----
        if t % 250 == 0:
            sc = 2.0 * n * float(conv.max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    conv = convolution_fft(best_a, best_a)
    best_score = 2.0 * n * float(conv.max())
    return best_a, best_score


# ----------------------------------------------------------------------
# Classic sub‑gradient refinement (same as the reference implementation)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient method for 2·n·max(a*a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_sa(start_seq: np.ndarray,
                  deadline: float,
                  max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Very long stochastic walk on the simplex.
    - 50 % of the moves are swaps, the rest transfer a small fraction.
    - Worse moves are accepted with probability exp(−Δ/ T) where the
      temperature T decays linearly with the remaining budget.
    """
    a = start_seq.copy()
    n = a.size

    best_a = a.copy()
    best_score = compute_score(best_a)

    # initial “step scale” – typical transferred fraction of the source mass
    step_scale = 0.12
    step = 0
    while step < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑normalise (keeps us on the probability simplex)
        cand = cand / cand.sum()
        cand_score = compute_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # simulated‑annealing acceptance
            # temperature decays from ~0.01 to ~1e‑4 linearly
            T = max(1e-4, (max_steps - step) / max_steps * 0.01)
            if np.random.rand() < np.exp(-(cand_score - best_score) / T):
                a = cand
            step_scale *= 0.97

        step += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine respects the 1000 s wall‑clock limit and returns a plain
    Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the harness itself
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a vector length – 8192 if we have plenty of time, else 4096.
    # ------------------------------------------------------------------
    if (deadline - start_time) > 600.0:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] chosen length = {n}")

    # ------------------------------------------------------------------
    # 0️⃣ Fallback – if a previous construction is available.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]      # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Analytic warm‑starts (α‑grid)
    # ------------------------------------------------------------------
    # coarse scan
    alphas = np.linspace(0.0, 2.5, 26)          # step ≈0.1
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] new best (α‑grid) = {best_score:.6f}")

    # fine refinement around the best α (if time permits)
    if best_seq is not None and time.time() < deadline - 120.0:
        # retrieve the α that generated the current best_seq
        # (simple re‑evaluation – the overhead is negligible)
        best_alpha = None
        # we keep the α that gave the recorded best_score
        # by re‑scanning the grid (still cheap)
        for alpha in alphas:
            a = alpha_start(n, alpha)
            if compute_score(a) == best_score:
                best_alpha = alpha
                break
        if best_alpha is not None:
            refined_alphas = np.linspace(
                max(0.0, best_alpha - 0.12),
                best_alpha + 0.12,
                13,
            )
            for alpha in refined_alphas:
                if time.time() >= deadline:
                    break
                a = alpha_start(n, alpha)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.tolist()
                    print(f"[{time.time():.1f}] refined α‑grid best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Exponential / edge‑heavy family
    # ------------------------------------------------------------------
    gammas = np.linspace(-0.03, 0.03, 25)      # includes negative (edge heavy)
    for gamma in gammas:
        if time.time() >= deadline:
            break
        a = exponential_start(n, gamma)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] new best (exp) γ={gamma:.5f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Random sparse restarts (lightweight – only if time left)
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        a0 = random_sparse_start(n, rng)
        sc0 = compute_score(a0)
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.tolist()
            print(f"[{time.time():.1f}] random sparse start gave {best_score:.6f}")

    # ------------------------------------------------------------------
    # ------------------------------------------------------------------
    # From here on we assume a valid vector is present.
    # ------------------------------------------------------------------
    if best_seq is None:
        # final safeguard – uniform vector of modest size
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 4️⃣ Exponentiated‑gradient (smoothed log‑sum‑exp) optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting exponentiated‑gradient phase")
        a_exp, sc_exp = exp_gradient_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=200.0,
            eta0=0.70,
        )
        if sc_exp < best_score:
            best_score = sc_exp
            best_seq = a_exp.tolist()
            print(f"[{time.time():.1f}] after EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Projected sub‑gradient refinement (true max‑objective)
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 40.0,
            max_iters=500_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Final stochastic hill‑climb with simulated‑annealing acceptance
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_hc, sc_hc = hill_climb_sa(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=400_000,
        )
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will repeat this call)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats as required by the harness
    return best_seq
```
