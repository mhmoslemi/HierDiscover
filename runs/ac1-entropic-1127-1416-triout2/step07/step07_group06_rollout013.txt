"""
search_step_function_v2.py

A multi‑restart optimizer for the step‑function “max‑convolution” problem.
The pipeline for each restart (chosen length n) is:

1. **Diverse initialisation** – random Dirichlet, monotone power‑law, or a
   mixture of both (plus occasional reuse of the current best truncated).

2. **Flattening** – minimise the variance of the self‑convolution (quadratic
   loss) with Adam, projecting onto the probability simplex after each step.

3. **Smooth‑max descent** – minimise a log‑sum‑exp surrogate of the max‑convolution
   with a decreasing ε schedule (Adam).  This gives a much smoother landscape
   than the true max.

4. **Exact sub‑gradient descent** – polish the true objective using a classic
   sub‑gradient step on the current worst shift (Adam‑like step‑size 1/√t).

5. **Simulated‑annealing hill‑climb** – stochastic mass transfers and swaps,
   with a linear temperature schedule and occasional “big‑jump” perturbations.

6. **LP‑based improvement (optional, for n ≤ 1024)** – solve the linear program
   ``max Σ g_i  s.t.  f⋆g ≤ M`` (M = current max‑convolution) to obtain a
   partner vector `g`.  Mixing `g` with the current iterate often lowers the
   bound further.

7. **Symmetrisation** – replace the vector by (a+reverse(a))/2 and re‑project;
   symmetry tends to flatten the convolution.

All updates are projected onto the probability simplex (∑a=1, a_i≥0), which
guarantees a valid input for `evaluate_sequence`.  The optimiser respects the
global time budget (≈ 1000 s) and keeps the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Fast utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random / structured initialisation
# ----------------------------------------------------------------------


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


def monotone_decreasing_vector(n: int, power: float = 1.0) -> np.ndarray:
    """
    Generate a monotone decreasing vector a_i ∝ (n-i)^power and project onto the simplex.
    """
    x = (np.arange(n, 0, -1, dtype=float)) ** power
    return project_to_simplex(x)


# ----------------------------------------------------------------------
# Phase 2 – flattening the convolution (L2 variance) using Adam
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the self‑convolution using Adam.
    Returns the best vector seen and its exact score.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        mu = np.mean(conv)
        diff = conv - mu

        # Gradient of variance: 4 * Σ_j a_j * diff_{i+j}
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)  # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – smooth‑max (log‑sum‑exp) gradient descent with Adam
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 8000,
    step0: float = 0.45,
    eps_factor: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate  ε·log∑exp(conv/ε)  using Adam.
    The ε schedule shrinks geometrically but never below 1 % of the current max.
    Returns the best vector seen (exact score).
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε schedule
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)          # soft‑max, Σ w = 1

        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small jitter occasionally (helps escape plateaus)
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – exact sub‑gradient descent (max‑convolution only)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact objective 2·n·max(a⊛a).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient for the worst shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # tiny random shake every 1000 iterations
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 5 – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 180_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber with simulated annealing.
    Moves: swap, or transfer a fraction of mass between two positions.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12  # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]                # swap
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]                       # move 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional larger perturbation
        if step % 5000 == 0:
            rnd = random_dense_vector(n) * 0.02   # 2 % random noise
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 6 – LP‑based partner improvement (only for modest n)
# ----------------------------------------------------------------------


def lp_improve(
    f: np.ndarray,
    time_deadline: float,
    max_inner: int = 2,
) -> tuple[np.ndarray, float]:
    """
    Solve the linear program  max Σ g_i  s.t.  f⋆g ≤ M
    (where M = max(f⋆f)).  Mix the obtained `g` back into `f`.
    Returns an improved vector and its exact score.
    The LP is built with a sparse Toeplitz matrix; for n≈1024 the
    solve takes well under a second.
    """
    n = len(f)
    best = f.copy()
    best_sc = compute_score(best)

    for _ in range(max_inner):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(best, best)
        M = np.max(conv)

        # Build sparse constraint matrix A (rows = 2n‑1, cols = n)
        rows = []
        cols = []
        data = []
        for col in range(n):
            # contribution of f[col] appears in rows col … col+n‑1
            rows.extend(range(col, col + n))
            cols.extend([col] * n)
            data.extend([best[col]] * n)

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M)

        # Linear program: maximise Σ g_i  <=> minimise –Σ g_i
        c = -np.ones(n)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            break

        g = res.x
        g = project_to_simplex(g)

        # Mix with current best (convex combination)
        t_mix = 0.5
        new = (1 - t_mix) * best + t_mix * g
        new = project_to_simplex(new)

        new_sc = compute_score(new)
        if new_sc < best_sc:
            best_sc = new_sc
            best = new
        else:
            # No improvement → stop LP refinement
            break

    return best, best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (cheap but often beneficial)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a+reverse(a))/2 and project back onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a
    plain Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final official evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback: use the previously known construction if present
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore (larger lengths give more freedom but cost more)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    outer_iter = 0
    while time.time() < deadline - 5.0:
        outer_iter += 1
        n = int(np.random.choice(lengths))

        # --------------------------------------------------------------
        # 1. initial vector (mixed random / monotone / truncated best)
        # --------------------------------------------------------------
        r = np.random.rand()
        if best_seq is not None and np.random.rand() < 0.15:
            # reuse a truncated version of the incumbent
            a0 = np.array(best_seq[:n], dtype=float)
            a0 = project_to_simplex(a0)
        elif r < 0.4:
            a0 = random_dense_vector(n)
        else:
            power = np.random.uniform(0.5, 2.5)
            a0 = monotone_decreasing_vector(n, power)
            # occasionally blend with random noise
            if np.random.rand() < 0.3:
                a0 = 0.7 * a0 + 0.3 * random_dense_vector(n)
                a0 = project_to_simplex(a0)

        # quick filter – keep if it already beats the incumbent
        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[{outer_iter}] init: n={n}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2. Flattening phase
        # --------------------------------------------------------------
        if time.time() >= deadline - 4.0:
            break
        a_flat, sc_flat = flatten_conv_adam(
            a0,
            time_deadline=deadline - 2.0,
            max_iter=5000,
            step0=0.12,
        )
        if sc_flat < best_score:
            best_score = sc_flat
            best_seq = a_flat.tolist()
            print(f"[{outer_iter}] flatten: n={n}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # 3. Smooth‑max surrogate phase (with continuation)
        # --------------------------------------------------------------
        if time.time() >= deadline - 6.0:
            break
        a_smooth, sc_smooth = smooth_max_adam(
            a_flat,
            time_deadline=deadline - 3.0,
            max_iter=8000,
            step0=0.45,
            eps_factor=0.22,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{outer_iter}] smooth‑max: n={n}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # 4. Exact sub‑gradient descent
        # --------------------------------------------------------------
        if time.time() >= deadline - 10.0:
            break
        a_sub, sc_sub = subgrad_optimize(
            a_smooth,
            time_deadline=deadline - 5.0,
            max_iter=120_000,
            step0=0.06,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{outer_iter}] sub‑grad: n={n}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # 5. Simulated‑annealing hill‑climb
        # --------------------------------------------------------------
        if time.time() >= deadline - 8.0:
            break
        seq_to_polish = best_seq if best_seq is not None else a_sub.tolist()
        polished_seq, polished_sc = hill_climb_anneal(
            seq_to_polish,
            time_deadline=deadline - 2.0,
            max_steps=180_000,
        )
        if polished_sc < best_score:
            best_score = polished_sc
            best_seq = polished_seq
            print(f"[{outer_iter}] anneal improved: score={best_score:.6f}")

        # --------------------------------------------------------------
        # 6. Optional LP‑based refinement (only for moderate n)
        # --------------------------------------------------------------
        if n <= 1024 and time.time() < deadline - 2.0:
            a_lp, sc_lp = lp_improve(
                np.array(best_seq, dtype=float),
                time_deadline=deadline - 1.0,
                max_inner=2,
            )
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.tolist()
                print(f"[{outer_iter}] LP improved: score={best_score:.6f}")

        # --------------------------------------------------------------
        # 7. Symmetry averaging (cheap final touch)
        # --------------------------------------------------------------
        a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
        sym_sc = compute_score(a_sym)
        if sym_sc < best_score:
            best_score = sym_sc
            best_seq = a_sym.tolist()
            print(f"[{outer_iter}] symmetry averaged: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
