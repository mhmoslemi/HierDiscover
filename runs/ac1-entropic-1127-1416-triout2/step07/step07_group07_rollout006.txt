"""
Search algorithm for minimizing the convolution‐max metric.

The algorithm runs several stages within the given time budget:

1. **Parametric sweep** – a family of Beta‑shaped distributions
   a_i ∝ (i+1)^{α-1}·(n-i)^{β-1} (α,β∈(0,1)) is explored on a few
   lengths (1024, 2048, 4096).  The best candidate is kept.

2. **Adam soft‑max descent** – the smooth surrogate
   f_ε(a)=ε·log Σ exp(conv(a)/ε) is minimised with an Adam optimiser.
   ε is decreased geometrically to focus the gradient on the current
   maximum entry of the convolution.

3. **Exact sub‑gradient descent** – a projected sub‑gradient step on the
   true objective 2·n·max(a⋆a) refines the solution.

4. **Simulated‑annealing hill‑climber** – random mass transfers are
   accepted with a Metropolis rule that allows occasional uphill moves,
   enabling the search to escape shallow local minima.

5. **Symmetry averaging** – after each major phase the vector is
   symmetrised (a←(a+rev(a))/2) and re‑projected onto the simplex.

All vectors are kept on the probability simplex (non‑negative,
normalised to sum = 1); this makes the fast `compute_score`
function equal to the official `evaluate_sequence` value up to the final
scaling.  The routine respects the 1000 s time limit and returns the
best sequence found.
"""

import time
import numpy as np

# ---------------------------------------------------------------
# Helper utilities: FFT convolution, simplex projection, scoring
# ---------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σ x = 1}.
    """
    if np.all(v <= 0):
        # All non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------
# Parametric families for initialisation
# ---------------------------------------------------------------

def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑shaped U‑distribution on {0,…,n‑1}:
        w_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
    The support is symmetric when α==β.
    """
    i = np.arange(n, dtype=np.float64) + 1.0          # 1 … n
    # avoid zero powers at the boundaries by adding a tiny epsilon
    eps = 1e-12
    w = (i + eps) ** (alpha - 1.0) * (n - i + 1.0 + eps) ** (beta - 1.0)
    # Normalize to the simplex (project_to_simplex also handles possible tiny negatives)
    w = w / np.sum(w)
    return w


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Classic discrete arcsine shape: w_i ∝ 1 / sqrt(i·(n‑1‑i)).
    """
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    w = w / np.sum(w)
    return w


def random_sparse_vector(n: int, density: float = None) -> np.ndarray:
    """Create a random probability vector with a modest number of non‑zero entries."""
    if density is None:
        density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a

def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f_ε(a) = ε·log Σ exp(conv/ε)
    Returns 2·(w_rev ⊛ a) where w = softmax(conv/ε).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)
    # stable softmax
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)                # Σ w = 1
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 6000,
    step0: float = 0.35,
    eps_factor: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate.
    ε is exponentially decreased; the loop stops when the time budget expires.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature schedule – never smaller than 1 % of the current max,
        # and decays geometrically.
        eps = maxc * max(eps_factor * (0.93 ** (t - 1)), 0.005)

        grad = softmax_gradient(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)          # modest decay
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny random jitter to shake out of plateaus
        if t % 600 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.001 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_sc


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.09,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    The sub‑gradient is taken at a position attaining the current maximum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2·a_{k‑i} (zero outside [0,n‑1])
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # small random perturbation every few thousand steps
        if t % 2500 == 0:
            a = project_to_simplex(a + np.random.randn(n) * (step0 * 0.003 / np.sqrt(t)))

    return best_a, best_sc


def annealed_hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style hill‑climber.  Moves consist of a mass transfer between
    two coordinates.  The temperature T decays linearly to zero, allowing a
    controlled amount of uphill acceptance early on.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    # Temperature schedule – start relatively warm, finish cold.
    T0 = 0.02
    step_frac = 0.07

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # Linear cooling
        T = T0 * max(0.0, 1.0 - step / max_steps)

        # Random move: relocate a fraction of mass from i to j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_frac * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # Numerical safety
        if cand[i] < 0.0:
            cand[i] = 0.0

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        # Accept if improvement or with Metropolis probability
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            a = cand
            # become more daring after a win
            step_frac = min(step_frac * 1.06, 0.4)
        else:
            # Metropolis acceptance
            if T > 0 and np.random.rand() < np.exp(-(sc - best_sc) / T):
                a = cand
                # slightly increase step size after a successful uphill move
                step_frac = min(step_frac * 1.02, 0.4)
            else:
                # otherwise stay put and shrink step size slowly
                step_frac *= 0.995

    return best_a, best_sc


def enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Make the vector (approximately) symmetric and re‑project onto simplex."""
    sym = (a + a[::-1]) * 0.5
    return project_to_simplex(sym)

    a0: np.ndarray,
    deadline: float,
    max_iter: int = 6000,
    step0: float = 0.35,
    eps_factor: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    The temperature ε decays geometrically; the loop stops when the
    time budget expires.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε never smaller than 1 % of the current max – decays geometrically
        eps = maxc * max(eps_factor * (0.93 ** (t - 1)), 0.005)

        grad = softmax_gradient(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to shake out of plateaus
        if t % 600 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.001 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_sc


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.09,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(a⋆a).  The sub‑gradient is taken at a location attaining the
    current maximum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        if t % 2500 == 0:
            a = project_to_simplex(a + np.random.randn(n) * (step0 * 0.003 / np.sqrt(t)))

    return best_a, best_sc


def annealed_hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style hill‑climber.
    Moves transfer a fraction of mass between two coordinates.
    Temperature T decays linearly, permitting controlled uphill moves early
    on.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    T0 = 0.02           # initial temperature
    step_frac = 0.07    # initial transferred fraction

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_frac * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            cand[i] = 0.0
        cand = project_to_simplex(cand)

        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            a = cand
            step_frac = min(step_frac * 1.06, 0.4)   # become bolder after a win
        else:
            if T > 0 and np.random.rand() < np.exp(-(sc - best_sc) / T):
                a = cand
                step_frac = min(step_frac * 1.02, 0.4)
            else:
                step_frac *= 0.995

    return best_a, best_sc


def enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Symmetrise the vector and re‑project onto the simplex."""
    sym = (a + a[::-1]) * 0.5
    return project_to_simplex(sym)

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` score within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a couple of seconds for final eval

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optional fallback to a known construction (if supplied)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Parametric sweep over Beta‑shaped families
    # --------------------------------------------------------------
    lengths = [1024, 2048, 4096]                     # keep sweep cheap
    alphas = np.linspace(0.30, 0.70, 9)              # 0.30 … 0.70

    for n in lengths:
        if time.time() >= deadline - 4.0:
            break
        for alpha in alphas:
            a_vec = beta_distribution(n, alpha, alpha)
            sc = compute_score(a_vec)
            if sc < best_score:
                best_score = sc
                best_seq = a_vec.tolist()
                print(f"[sweep] n={n}, α={alpha:.2f} → score={sc:.6f}")

            # occasional jittered variant
            if np.random.rand() < 0.15:
                jitter = np.random.randn(n) * 0.001
                a_jit = project_to_simplex(a_vec + jitter)
                sc_j = compute_score(a_jit)
                if sc_j < best_score:
                    best_score = sc_j
                    best_seq = a_jit.tolist()
                    print(f"[jitter] n={n}, α={alpha:.2f} → score={sc_j:.6f}")

    # --------------------------------------------------------------
    # 2. Main optimisation loop – a few random restarts
    # --------------------------------------------------------------
    outer_max = 20
    for outer in range(outer_max):
        if time.time() >= deadline - 4.0:
            break

        # Choose a size (bigger sizes often bring the asymptotic constant
        # closer to its optimum)
        n = int(np.random.choice([2048, 4096, 8192]))

        # Initialise either from the current best (if lengths match) or
        # freshly from a random Beta exponent.
        if best_seq is not None and len(best_seq) == n:
            a0 = np.array(best_seq, dtype=np.float64)
            a0 = project_to_simplex(a0 + np.random.randn(n) * 5e-4)
        else:
            alpha = np.random.uniform(0.30, 0.70)
            a0 = beta_distribution(n, alpha, alpha)
            if np.random.rand() < 0.2:
                a0 = random_sparse_vector(n)

        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[{outer}] init (n={n})  score={cur_sc:.6f}")

        # ----------------------------------------------------------
        # 2.1 Adam soft‑max descent
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        iters = min(8000, int(remaining * 30))   # cheap estimate of grads per sec
        a_gd, _ = adam_softmax_descent(
            a0,
            deadline - 2.0,
            max_iter=iters,
            step0=0.30,
            eps_factor=0.25,
        )
        sc_gd = compute_score(a_gd)
        if sc_gd < best_score:
            best_score = sc_gd
            best_seq = a_gd.tolist()
            print(f"[{outer}] after Adam GD (n={n})  score={sc_gd:.6f}")

        # ----------------------------------------------------------
        # 2.2 Exact sub‑gradient descent
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            a_sub, _ = subgradient_descent(
                a_gd,
                deadline - 3.0,
                max_iter=100_000,
                step0=0.08,
            )
            sc_sub = compute_score(a_sub)
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{outer}] after sub‑grad (n={n})  score={sc_sub:.6f}")

        # ----------------------------------------------------------
        # 2.3 Annealed hill‑climb polishing
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 8.0:
            start_vec = np.array(best_seq, dtype=np.float64)
            a_hc, sc_hc = annealed_hill_climb(
                start_vec,
                deadline - 2.0,
                max_steps=200_000,
            )
            if sc_hc < best_score:
                best_score = sc_hc
                best_seq = a_hc.tolist()
                print(f"[{outer}] hill‑climb improved  score={sc_hc:.6f}")

            # Symmetry averaging – cheap, often beneficial
            a_sym = enforce_symmetry(np.array(best_seq, dtype=np.float64))
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = a_sym.tolist()
                print(f"[{outer}] symmetry averaging improved  score={sc_sym:.6f}")

    # --------------------------------------------------------------
    # 3. Fallback – uniform distribution (guaranteed finite)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 4. Final official evaluation
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
