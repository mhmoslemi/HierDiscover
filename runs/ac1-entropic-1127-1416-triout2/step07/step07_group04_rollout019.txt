```
def propose_candidate(seed=0, budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 1.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_val = float('inf')
    
    # set initial lengths options: try 8192 if we have enough time else 4096
    base_n = 8192 if (deadline - time.time() > 180) else 4096
    print(f"Using length {base_n}")
    
    # start shape: arcsine_initial for coarse 2048, then upsample.
    # we could also try multiple shapes: arcsine, beta, power.
    shape_generators = [
        lambda n: arcsine_initial(n),
        lambda n: generate_beta_shape(n, alpha=0.5, beta=0.5, eps=1e-6), # symmetric, heavy edges
        lambda n: generate_power_shape(n, p=1.0, eps=1e-6),
        lambda n: generate_mixture_shape(n, weight_arc=0.8)
    ]
    
    # For each shape start
    for gen_idx, gen in enumerate(shape_generators):
        if time.time() >= deadline: break
        print(f"\n--- Starting from generator {gen_idx} ---")
        a_initial = gen(base_n)
        # Ensure sum = 1
        a_initial = a_initial / a_initial.sum()
        
        # Adam smoother
        remaining = deadline - time.time()
        max_iters = int(min(500_000, remaining * 30))
        print(f"Adam iterations: {max_iters}")
        a_opt = adam_optimize_smooth_max(
            a_initial,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=500.0,
            base_lr=0.4,
        )
        # Polishing
        remaining = deadline - time.time()
        if remaining > 5:
            # subgradient
            max_sub_iter = int(min(200_000, remaining * 30))
            a_opt = subgradient_descent_exact(a_opt, deadline,
                                              max_iter=max_sub_iter,
                                              step0=0.06)
            # exponentiated gradient
            remaining = deadline - time.time()
            if remaining > 5:
                max_eg_iter = int(min(150_000, remaining * 30))
                a_opt = eg_refine(a_opt, deadline,
                                  max_iter=max_eg_iter,
                                  eta0=0.25)
        
        # Evaluate
        val = evaluate_sequence(a_opt.tolist())
        print(f"Candidate val = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_opt.tolist()
            print("** New best **")
        
        # Random local perturbation from best to explore
        if best_seq is not None and time.time() < deadline - 5:
            # try a few random perturbations
            for perturb in range(3):
```

The optimisation problem is  

\[
\min_{a\in\mathbb R_{\ge 0}^n}
\; \frac{2n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad
(a*a)_k=\sum_i a_i\,a_{k-i},
\]

where the convolution is linear (zero padding).  
The objective is *scale‑invariant* – only the *shape* of the step
function matters.  Good shapes are highly asymmetric: the vector and its
reverse have little overlap, which makes the maximum entry of the
convolution small while the total mass stays large.

The code below implements a three‑phase search that

1. **generates several diverse warm‑starts** (arcsine, beta‑skewed,
   power‑law, mixtures and a random Dirichlet), all normalised to a
   probability simplex.
2. **optimises the smooth‑max approximation** of the max‑convolution
   with Adam.  The temperature τ is increased from 0.5 to 500 so that
   the smooth surrogate becomes a very tight approximation of the true
   max.  The optimisation runs as long as the wall‑clock budget permits.
3. **polishes the final candidate** with a projected Euclidean
   sub‑gradient descent (exact max‑convolution) followed by an
   exponentiated‑gradient (mirror‑descent) step.

All phases respect the global time budget (default 1000 s) and keep the
best sequence found so far.  The final result is returned as a plain
`list[float]`, ready for `evaluate_sequence`.

```python
"""
Robust step‑function optimiser
==============================

This implementation searches for a non‑negative sequence that minimises
the official ``evaluate_sequence`` score.  The algorithm consists of

* several cheap warm‑starts (arcsine, beta‑skewed, power‑law,
  mixtures and random Dirichlet vectors);
* a long Adam optimisation of a smooth‑max surrogate of the max‑convolution
  (temperature τ is linearly increased up to 500);
* an exact polishing phase (projected sub‑gradient descent + exponentiated‑
  gradient mirror descent).

The whole routine is time‑aware: it continuously checks the remaining
wall‑clock time and adapts the number of iterations so that the function
always finishes before the 1000 s deadline.  The best sequence found is
returned as a list of floats.
"""

import time
from typing import List, Callable

import numpy as np

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        # all non‑positive → uniform simplex
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_surrogate(a: np.ndarray, tau: float) -> float:
    """
    Smooth‑max surrogate: 2·n·log‑sum‑exp(tau·conv(a,a))/tau.
    It is an upper bound on 2·n·max(conv) and converges to it as τ→∞.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    scaled = conv * tau
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    s = np.log(exp_scaled.sum()) + max_scaled
    return 2.0 * n * s / tau


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{-p}, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture: weight_arc·arcsine + (1‑weight_arc)·uniform, normalised."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points,
    followed by re‑normalisation to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform draw from the (n‑dimensional) simplex."""
    x = rng.exponential(scale=1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    tau_start: float = 0.5,
    tau_end: float = 200.0,
    base_lr: float = 0.5,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate.
    The optimisation variable is the logits ``w`` such that a = softmax(w).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)                 # numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)            # length L

        # smooth max via log‑sum‑exp (weights)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        # conv of a with the reversed weight vector
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if it % 50_000 == 0:
            surrogate = compute_surrogate(a, tau)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (nonsmooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.06,
) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.22,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a
```
