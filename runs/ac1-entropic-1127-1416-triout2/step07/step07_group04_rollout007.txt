"""
Step‑function optimizer with an LP‑based “mass‑reallocation” stage.

Algorithm
---------
1. **Coarse seeding** – generate a handful of 1024‑point candidates:
   * the classic arcsine shape,
   * uniform,
   * a few random Beta‑shapes and Dirichlet draws.
2. **LP mass‑reallocation** – for each coarse candidate we solve
   a linear program
        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f * g)_k ≤ M   (k = 0…2n‑2)
   where ``f`` is the current shape and ``M = max(f*f)``.
   The solution pushes probability mass into positions that do not
   increase the maximal self‑convolution.  After scaling the LP
   solution back to a probability vector we mix it with the current
   shape (mixing factor ≈ 0.2).  One or two LP‑iterations are enough
   to obtain a markedly better coarse shape.
3. **Resolution lift & smooth‑max Adam** – the improved shape is
   up‑sampled (linear interpolation) to a high resolution
   (4096 or 8192 points, depending on remaining time).  A long Adam
   optimisation is then performed on the smooth log‑sum‑exp
   approximation of ``max(conv(a,a))``.  The temperature τ is raised
   linearly from 0.5 to a large value (≈ 250) so that the surrogate
   converges to the true max.
4. **Exact polishing** – a projected Euclidean sub‑gradient descent
   (using the exact max‑convolution) followed by an exponentiated‑
   gradient (mirror‑descent) refines the solution to a locally optimal
   point.
5. **Selection** – every candidate is evaluated with the official
   ``evaluate_sequence`` routine; the best sequence found before the
   1000 s deadline is returned.

The whole pipeline is time‑aware: before each phase the remaining
wall‑clock time is checked and the number of iterations is limited so
that the function always finishes well before the deadline.
"""

import time
import numpy as np
from typing import List
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (fast for the sizes we use).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate used during optimisation.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (seed factories)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to Σ=1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points,
    re‑normalised to Σ=1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Linear‑programming mass re‑allocation
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ g)[k] = (f * g)[k].
    The matrix is very sparse (exactly one dense anti‑diagonal).
    """
    n = f.size
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # columns j where g_j participates
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_mass_reallocation(f: np.ndarray, mix_factor: float) -> np.ndarray:
    """
    One LP step:
        maximise Σ g_i  subject to g ≥ 0, (f * g) ≤ M,
    where M = max(f*f).  The solution g is scaled to sum‑1 and mixed
    with f using ``mix_factor`` (0 < mix_factor ≤ 1).
    Returns the mixed vector.
    """
    n = f.size
    M = float(np.max(convolution_fft(f, f)))
    A = build_conv_matrix(f)
    b_ub = np.full(A.shape[0], M, dtype=float)
    c_obj = -np.ones(n)                     # minimise -Σg ⇔ maximise Σg
    bounds = [(0, None)] * n

    # Using the high‑performance HiGHS interior‑point solver
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # If the LP fails we simply return the original vector.
        return f.copy()

    g = res.x
    sum_g = g.sum()
    if sum_g == 0.0:
        return f.copy()
    g_norm = g / sum_g
    # mixing – keep the total mass equal to 1
    mixed = (1.0 - mix_factor) * f + mix_factor * g_norm
    # re‑project onto the simplex to correct numerical drift
    mixed = project_to_simplex(mixed)
    return mixed


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (same idea as the reference code)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth log‑sum‑exp approximation of
    max(conv(a,a)).  The optimisation variable is the logits ``w`` (softmax → a)
    which automatically enforces a ≥ 0 and Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)               # initialise logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)                 # numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)             # length L

        # smooth max via log‑sum‑exp (scaled by τ)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp           # shape (L,)

        # gradient w.r.t. a : 2·conv(a, rev(weights))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)     # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if it % 30000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a

def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.06) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, rev(mask))
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

            if time.time() >= deadline:
                break
            mix = 0.25 / (lp_it + 1)                 # 0.25 then 0.125
            a = lp_mass_reallocation(a, mix_factor=mix)
        refined_coarse.append(a)

    # ------------------------------------------------------------------
    # 3️⃣  Upsample to a higher resolution and run smooth‑max Adam
    # ------------------------------------------------------------------
    # decide on the final size (more points give a tighter constant)
    remaining = deadline - time.time()
    target_n = 4096
    if remaining > 200.0:
        target_n = 8192                     # use the extra time if we have it

    print(f"\n--- Upscaling to n={target_n} and Adam optimisation ---")
    candidates = []

    for idx, a_coarse in enumerate(refined_coarse):
        if time.time() >= deadline:
            break
        a_high = upsample(a_coarse, target_n)

        # Adam – number of iterations is proportional to remaining time
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 35))
        print(f"  Candidate {idx+1}: Adam iters ≈ {max_iters}")

        a_refined = adam_optimize_smooth_max(
            a_high,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=250.0,
            base_lr=0.6,
        )

        # ------------------------------------------------------------------
        # 4️⃣  Exact polishing (sub‑gradient + EG)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 15.0:
            sub_iters = int(min(300_000, remaining * 30))
            a_refined = subgradient_descent_exact(
                a_refined,
                deadline,
                max_iter=sub_iters,
                step0=0.06,
            )
            remaining = deadline - time.time()
            if remaining > 5.0:
                eg_iters = int(min(200_000, remaining * 30))
                a_refined = eg_refine(
                    a_refined,
                    deadline,
                    max_iter=eg_iters,
                    eta0=0.24,
                )

        # ------------------------------------------------------------------
        # 5️⃣  Official evaluation
        # ------------------------------------------------------------------
        try:
            val = evaluate_sequence(a_refined.tolist())
            print(f"    Candidate {idx+1} official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_refined.tolist()
                print("*** New global best ***")
        except Exception as e:
            print(f"    Evaluation failed for candidate {idx+1}: {e}")

        candidates.append((a_refined, val if 'val' in locals() else None))

    # ------------------------------------------------------------------
    # 6️⃣  Fallback – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        print("No valid candidate found – falling back to uniform.")
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
