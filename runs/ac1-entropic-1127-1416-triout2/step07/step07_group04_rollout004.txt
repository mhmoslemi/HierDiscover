"""
Optimiser for the step‑function attack.

The algorithm follows a simple “coarse → fine → polish” scheme
but with a few changes compared with the previous state‑of‑the‑art
implementation:

*  A cheap parametric scan (Beta‑type symmetric densities and a
   arcsine/uniform mixture) on a modest grid (n=1024) is used to pick a
   good starting shape.
*  The main optimisation is performed at a relatively high resolution
   (n=4096 by default).  If time permits we go to 8192.
*  A short Adam phase with a warm‑up/annealing schedule for the
   smooth‑max is followed by a long Euclidean projected sub‑gradient
   descent and an exponentiated‑gradient polish.
*  Finally a tiny stochastic local‑search (Gaussian perturbations with a
   simplex projection) tries to squeeze out any remaining improvement.
*  All phases constantly check the remaining wall‑clock time so that the
   routine always finishes well before the 1000 s limit.

The returned list is normalised (sum = 1).  Because the official score
`evaluate_sequence` is scale‑invariant, only the *shape* of the vector
matters.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-6) -> np.ndarray:
    """Symmetric Beta‑type shape (α‑1,β‑1) on a grid, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points, re‑normalised.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth approximation of max(conv).
    The optimisation variable is the logits w (softmax → a) which
    automatically satisfies a ≥ 0, Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)                 # numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)            # length L

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (nonsmooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.06) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence`` value.
    The routine respects a global 1000 s wall‑clock budget and returns the best
    sequence it found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Coarse scouting (n=1024) – pick a promising shape
    # ------------------------------------------------------------------
    n_coarse = 1024
    print(f"\n--- Phase 0: shape scouting (n={n_coarse}) ---")
    candidates = []

    # deterministic arcsine
    candidates.append(arcsine_initial(n_coarse))

    # symmetric Beta shapes with several α
    for alpha in [0.35, 0.45, 0.55, 0.65, 0.75, 0.85]:
        candidates.append(generate_beta_shape(n_coarse, alpha, alpha))

    # mixture arcsine / uniform
    for w in [0.1, 0.3, 0.5, 0.7, 0.9]:
        candidates.append(generate_mixture_shape(n_coarse, w))

    # pick the one with smallest surrogate score
    best_coarse = None
    best_score = float("inf")
    for cand in candidates:
        s = compute_score(cand)
        if s < best_score:
            best_score = s
            best_coarse = cand
    print(f"Best coarse surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Warm‑up Adam on the coarse grid (fast)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    max_iter_coarse = int(min(80_000, remaining * 25))
    a_coarse = adam_optimize_smooth_max(
        best_coarse,
        deadline,
        max_iters=max_iter_coarse,
        tau_start=0.5,
        tau_end=80.0,
        base_lr=0.45,
    )
    print(f"Coarse Adam surrogate = {compute_score(a_coarse):.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample to the target resolution
    # ------------------------------------------------------------------
    # Decide on the ultimate size – 4096 is a good balance; go to 8192 only if we have ample time.
    target_n = 4096
    if deadline - time.time() > 200.0:
        target_n = 8192

    print(f"\n--- Phase 1: upsample to n={target_n} ---")
    a_high = upsample(a_coarse, target_n)

    # ------------------------------------------------------------------
    # 4. Long Adam phase (smooth‑max) on the high‑resolution grid
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    max_iter_high = int(min(200_000, remaining * 30))
    a_high = adam_optimize_smooth_max(
        a_high,
        deadline,
        max_iters=max_iter_high,
        tau_start=0.5,
        tau_end=250.0,
        base_lr=0.55,
    )
    print(f"High‑res Adam surrogate = {compute_score(a_high):.6f}")

    # ------------------------------------------------------------------
    # 5. Exact polishing: projected sub‑gradient descent
    # ------------------------------------------------------------------
    print("\n--- Phase 2: projected sub‑gradient descent ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(250_000, remaining * 30))
    a_refined = subgradient_descent_exact(a_high, deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.07)

    # ------------------------------------------------------------------
    # 6. Exact polishing: exponentiated‑gradient (mirror descent)
    # ------------------------------------------------------------------
    print("\n--- Phase 3: exponentiated‑gradient polish ---")
    remaining = deadline - time.time()
    max_iter_eg = int(min(180_000, remaining * 30))
    a_refined = eg_refine(a_refined, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.24)

    # ------------------------------------------------------------------
    # 7. Official evaluation of the polished candidate
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(a_refined.tolist())
        print(f"Official score after polishing = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_refined.tolist()
            print("*** New global best found! ***")
    except Exception as e:
        print(f"Evaluation failed after polishing: {e}")

    # ------------------------------------------------------------------
    # 8. Stochastic local search (tiny Gaussian tweaks)
    # ------------------------------------------------------------------
    print("\n--- Phase 4: stochastic local search ---")
    a_current = np.array(best_seq if best_seq else a_refined, dtype=float)
    n = a_current.size
    attempts = 0
    while time.time() < deadline - 0.5 and attempts < 20_000:
        attempts += 1
        # generate a small perturbation and re‑project to the simplex
        sigma = 0.02 / math.sqrt(attempts)   # gradually shrink
        noise = rng.normal(scale=sigma, size=n)
        cand = a_current + noise
        cand = project_to_simplex(cand)

        # cheap surrogate test first
        surrogate = compute_score(cand)
        if surrogate >= best_val:          # no chance to beat current best
            continue

        # official check
        val_cand = evaluate_sequence(cand.tolist())
        if val_cand < best_val:
            best_val = val_cand
            best_seq = cand.tolist()
            a_current = cand
            print(f"Local search improved official score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Final safety guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
