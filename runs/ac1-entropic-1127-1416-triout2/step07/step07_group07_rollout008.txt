"""
search_step_function.py

Search for a non‑negative step function that minimises

    score = 2·n·max(conv) / (sum(a)²)

where `conv = np.convolve(a, a)`.  Because the objective is
scale‑invariant we keep every candidate on the probability simplex
(`sum(a)=1`).  Then `score = 2·n·max(conv)` and the cheap function
`compute_score` can be used instead of the full `evaluate_sequence`.

The optimiser is a multi‑stage pipeline that respects a 1000 s time
budget:

1. **Initialisation** – arcsine‑type density (almost flat self‑convolution)
   or a random Dirichlet vector.
2. **Long Adam optimiser** on a smooth log‑sum‑exp surrogate of the max.
   The temperature is decreased slowly so that the gradient remains
   informative even when the current max is already small.
3. **Exact sub‑gradient descent** on the true `max(conv)`.
4. **Simulated‑annealing hill‑climber** that moves mass between two
   positions; it accepts worsening moves with a temperature that
   decays exponentially.
5. **Symmetry averaging** – `(a+reverse(a))/2` followed by a simplex
   projection.  The symmetry step is cheap and often reduces the peak.

All vectors stay on the simplex, therefore the cheap score returned by
`compute_score` is identical to the official `evaluate_sequence` value.
The routine runs as many independent restarts as the time budget allows
and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT.
    Result length = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of evaluate_sequence when sum(a) == 1.
    Returns 2·n·max(conv) where conv = a ⋆ a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine‑like distribution:
        a_i ∝ 1 / sqrt(i·(n‑1‑i))
    It is symmetric and already gives a fairly flat self‑convolution.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_dirichlet_initialisation(n: int) -> np.ndarray:
    """
    Random probability vector obtained from a Dirichlet(1,…,1) distribution.
    Equivalent to normalising i.i.d. exponential samples.
    """
    a = np.random.exponential(scale=1.0, size=n)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# 1. Adam optimiser on a smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed surrogate
        f_eps(a) = eps·log Σ_k exp(conv_k/eps)
    with conv = a ⋆ a.
    Returns  grad = 2 * (w_rev ⊛ a),
    where w = softmax(conv/eps).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)          # Σ w = 1

    # cross‑correlation w_rev ⊛ a
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)
    start = a.shape[0] - 1                 # index where shift 0 aligns
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
    step0: float = 0.3,
    eps_factor: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    The temperature ε is tied to the current maximal convolution value and
    decays exponentially.  The routine stops when the deadline is reached.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature – never smaller than 1 % of the current max
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to prevent stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.001 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent for the objective 2·n·max(a⋆a).
    The sub‑gradient is taken at the shift attaining the current maximum.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k-i} (zero outside the support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random shake every few thousand iterations
        if t % 3000 == 0:
            a += np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Simulated‑annealing hill‑climbing (mass transfers)
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climber that moves a fraction of mass from one index to another.
    It accepts worsening moves with a temperature that decays exponentially.
    """
    a = a_start.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    n = a.shape[0]
    # temperature schedule (starts relatively high, ends very low)
    T_start = 1e-3 * cur_sc
    T_end = 1e-7 * cur_sc

    step_frac = 0.12  # initial transfer fraction

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        frac = step / max_steps
        T = T_start * (T_end / T_start) ** frac

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose a random transfer of a fraction of the mass at i
        delta = step_frac * a[i] * np.random.rand()
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # safeguard – numerical round‑off may make i negative
        if cand[i] < -1e-12:
            continue

        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)

        delta_sc = cand_sc - cur_sc
        accept = False
        if delta_sc < 0.0:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-delta_sc / T):
                accept = True

        if accept:
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                # become more daring after a success
                step_frac = min(step_frac * 1.07, 0.5)
            else:
                # become more cautious after a rejection/neutral move
                step_frac = max(step_frac * 0.96, 0.001)

        # occasional symmetry enforcement (cheap and often helpful)
        if step % 5000 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = project_to_simplex(a_sym)
            a = a_sym
            cur_sc = compute_score(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper – enforce (approx.) symmetry
# ----------------------------------------------------------------------


def enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Replace a by the average of a and its reverse and re‑project."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` score.  The routine obeys the supplied time budget
    (≈ 1000 s) and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback – use a previously discovered construction if it
    # exists in the global namespace.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameter set for the outer loop (restarts)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]  # try a variety of sizes
    outer = 0

    while time.time() < deadline - 5.0:
        outer += 1
        n = int(np.random.choice(lengths))

        # ------------------------------------------------------------------
        # 0. Initial vector – arcsine or random Dirichlet
        # ------------------------------------------------------------------
        if np.random.rand() < 0.65:
            a0 = arcsine_initialisation(n)
        else:
            a0 = random_dirichlet_initialisation(n)

        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[{outer}] init (n={n})  score={cur_sc:.6f}")

        # ------------------------------------------------------------------
        # 1. Adam‑accelerated soft‑max descent (long run)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        # allocate roughly 30 % of the remaining budget to Adam
        adam_iters = int(min(30_000, remaining * 30))
        a_adam, sc_adam = adam_descent(
            a0,
            deadline=deadline - 1.0,
            max_iter=adam_iters,
            step0=0.35,
            eps_factor=0.18,
        )
        exact_sc = compute_score(a_adam)
        if exact_sc < best_score:
            best_score = exact_sc
            best_seq = a_adam.tolist()
            print(f"[{outer}] after Adam (n={n})  score={exact_sc:.6f}")

        # ------------------------------------------------------------------
        # 2. Exact sub‑gradient descent (fine‑tuning)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 12.0:
            sub_iters = int(min(80_000, remaining * 60))
            a_sub, sc_sub = subgradient_descent(
                a_adam,
                deadline=deadline - 2.0,
                max_iter=sub_iters,
                step0=0.07,
            )
            exact_sc = compute_score(a_sub)
            if exact_sc < best_score:
                best_score = exact_sc
                best_seq = a_sub.tolist()
                print(f"[{outer}] after sub‑grad (n={n})  score={exact_sc:.6f}")

        # ------------------------------------------------------------------
        # 3. Simulated‑annealing hill‑climb
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 8.0:
            # start from the best currently known vector for this outer loop
            start_vec = np.array(best_seq, dtype=float) if best_seq is not None else a_sub
            a_hill, sc_hill = annealing_hill_climb(
                start_vec,
                deadline=deadline - 1.0,
                max_steps=int(min(300_000, remaining * 250)),
            )
            if sc_hill < best_score:
                best_score = sc_hill
                best_seq = a_hill.tolist()
                print(f"[{outer}] hill‑climb improved  score={sc_hill:.6f}")

            # symmetry averaging – cheap, often improves the peak
            a_sym = enforce_symmetry(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer}] symmetry avg improved  score={sym_sc:.6f}")

    # ------------------------------------------------------------------
    # Final safety net – uniform distribution (guaranteed finite)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
