def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    best_seq = None
    best_val = inf

    # baseline uniform
    n0 = 1024
    uniform = np.full(n0, 1.0/n0)
    val = evaluate_sequence(uniform.tolist())
    if val < best_val:
        best_val = val
        best_seq = uniform.tolist()
    print("Uniform baseline:", val)
    # if time budget low return uniform maybe

    # Choose lengths to try, maybe 2048 and 1024
    lengths = [2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"Optimizing length {n}")
        # generate seeds
        seeds = []
        # deterministic seeds
        seeds.append(arcsine_initial(n))
        seeds.append(np.full(n, 1.0/n))
        seeds.append(generate_mixture_shape(n, 0.7))
        # power family scanning
        for p in np.linspace(0.2, 2.5, 9):
            seeds.append(generate_power_shape(n, p, eps=1e-8))
        # beta family random draws
        for _ in range(5):
            a = rng.uniform(0.5, 4.0)
            b = rng.uniform(0.5, 4.0)
            eps = rng.uniform(0, 0.01*n)
            seeds.append(generate_beta_shape(n, a, b, eps))
        # Dirichlet random
        for alpha in [0.2, 0.5, 1.0, 2.0]:
            dir_sample = rng.dirichlet([alpha]*n)
            seeds.append(dir_sample)

        # evaluate fast_score and sort to keep top k seeds
        scored = [(fast_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        topK = min(5, len(scored))
        top_seeds = [vec for _, vec in scored[:topK]]

        # allocate time for each seed approx
        per_seed_time = max(1.0, (deadline - time.time())/(len(top_seeds)*2.5))
        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 2:
                break
            print(f" Seed {idx}/{len(top_seeds)} time {per_seed_time:.1f}s")
            seed_deadline = time.time() + per_seed_time
            # warm start via Adam with moderate β
            z0 = np.log(seed_vec + 1e-12)
            a = smooth_adam_optimize(z0, seed_deadline, n=n,
                                     beta0=5.0, beta_max=800.0,
                                     ramp_iters=30_000, lr=0.05)
            # now exact gradient descent with line search
            # allocate remaining time for this seed
            remaining_deadline = deadline - 0.5
            a = exact_gradient_descent(a, remaining_deadline)
            # evaluate official
            try:
                val = evaluate_sequence(a.tolist())
                print(f"   Score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a.tolist()
                    print(f"   New best !! {best_val}")
            except Exception as e:
                print(" eval error", e)

    # final polishing with exact gradient descent long
    if best_seq is not None and time.time() < deadline - 0.3:

        a = np.array(best_seq, dtype=float)

        except:

    if not best_seq:

        best_seq = [1.0/n]*n

"""
Hybrid optimisation for the step‑function attack problem.

We search for a probability vector `a` (non‑negative, sum=1) that minimises

    f(a) = 2·n·max(conv(a,a))

where `conv` is the linear convolution of `a` with itself.
The algorithm proceeds as follows:

1. **Analytic & random seeds** – generate a diverse pool of
   candidates (arcsine, beta, power‑law, Dirichlet, mixtures).

2. **Fast pre‑selection** – evaluate every seed with a cheap surrogate
   (`fast_score`) and keep only the most promising few.

3. **Continuation Adam** – optimise a smooth approximation
   `Lβ(a)=log‑sum‑exp(β·conv(a,a))` using Adam while `β` is increased
   (β∈[5,800]).  The optimisation is performed on the unconstrained
   logits `z` where `a = softmax(z)`; this automatically respects the
   simplex constraints.

4. **Exact projected sub‑gradient descent** – starting from the Adam
   result we minimise the true objective using a projected sub‑gradient
   method with a diminishing step size and Euclidean projection onto
   the simplex.  This step works directly with the true `max(conv)`.

5. **Global polishing** – after all seeds have been processed we run a
   long sub‑gradient polishing on the best solution found so far.
   Small random perturbations followed by a short polishing are also
   tried while time permits.

All phases respect the global 1000 s budget and always return a valid
non‑negative sequence.  The routine prints progress and the best
official score (via `evaluate_sequence`) it has discovered.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(conv(a,a)) (sum(a)=1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic generators (all return a probability vector)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped seed."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on the discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (1 / ((i+0.5)(n‑0.5‑i)+eps)^p)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate Lβ
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 800.0,
    ramp_iters: int = 30_000,
    lr: float = 0.05,
) -> np.ndarray:
    """
    Minimise Lβ(a)=log‑sum‑exp(β·conv(a,a)) using Adam on logits `z`.
    β is linearly increased from `beta0` to `beta_max` (continuation).
    The returned `a` lies on the simplex.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of Σ w_i conv_i  w.r.t. a  (see analysis)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)                 # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]           # length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(
                f"    Adam iter={t:,d} β={beta:.1f}  surrogate≈{fast_score(a):.6f}"
            )
    return _softmax(z)

# Exact projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def projected_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.30,
) -> np.ndarray:
    """
    Projected sub‑gradient descent for f(a)=2·n·max(conv(a,a)).
    Uses Euclidean projection onto the simplex after each additive step.
    The step size follows η_t = step0 / √t.
    """
    a = _project_to_simplex(a0.copy())
    n = a.size
    L = 2 * n - 1
    best_a = a.copy()
    best_val = fast_score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # mask of entries attaining (approximately) the maximum
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        rev = convolution_fft(a, mask[::-1])          # length = 3n‑2
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # diminishing step size
        eta = step0 / np.sqrt(it)

        # additive update + projection
        a_new = _project_to_simplex(a - eta * subgrad)

        # evaluate new objective (fast, exact)
        val_new = fast_score(a_new)

        # keep the better of the two iterates
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            # even if not improving we still move (helps escape flat plateaus)
            a = a_new

        if it % 10_000 == 0:
            print(f"    Sub‑grad it={it:,d}  score≈{best_val:.6f}")

    return best_a

                z0,
                adam_deadline,

                beta_max=800.0,
                ramp_iters=30_000,
                lr=0.05,
            )

                    deadline,
                    max_iter=300_000,

                print(f"   Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3.  Final global polishing on the best candidate found so far
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_polished = projected_subgradient_descent(
            a_arr,
            deadline,
            max_iter=500_000,
            step0=0.30,
        )
        try:
            final_val = evaluate_sequence(a_polished.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_polished.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4.  Optional random‑perturbation restarts (if we still have time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0 and best_seq:
        n_cur = len(best_seq)
        # small Gaussian jitter
        perturb = np.array(best_seq, dtype=float) + rng.normal(scale=0.02, size=n_cur)
        perturb = np.maximum(perturb, 0.0)
        perturb /= perturb.sum() + 1e-16

        # very short sub‑gradient refinement (≈3 s)
        pert_deadline = time.time() + 3.0
        a_pert = projected_subgradient_descent(
            perturb,
            pert_deadline,
            max_iter=50_000,
            step0=0.25,
        )
        try:
            val = evaluate_sequence(a_pert.tolist())
            if val < best_val:
                best_val, best_seq = val, a_pert.tolist()
                print(f"   Perturbation improvement: {best_val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 5.  Guard – guarantee a non‑empty valid return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
