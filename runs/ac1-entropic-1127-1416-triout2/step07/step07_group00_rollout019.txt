```
def adam_smooth_optimize(start_a, deadline, max_iters=200_000, beta_start=5.0, beta_end=400.0, step=0.5, b1=0.9, b2=0.999, eps=1e-8):
    a = start_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_score = compute_score(a)
    best_a = a.copy()
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()
        grad = 2.0 * convolution_fft(a[::-1], w)
        # Adam updates
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** it)
        v_hat = v / (1 - b2 ** it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)
        # occasional evaluation
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score
```

```
def hill_climb_sa(start_seq, deadline, max_steps=1_000_000, initial_T=0.02, cooling_rate=0.99999):
    a = start_seq.copy()
    best_a = a.copy()
    best_score = compute_score(best_a)
    T = initial_T
    steps = 0
    while steps < max_steps and time.time() < deadline:
        # propose move
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(len(cand), size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(len(cand), size=2)
            delta = 0.05 * best_a[i]  # move small fraction
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        delta = sc - best_score
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            best_score = sc
            best_a = cand
            # Optionally increase step scale after improvement
        # decay temperature
        T *= cooling_rate
        steps += 1
    return best_a, best_score
```

```
    # Stage 4: refined subgradient (smaller step)
    if best_seq is not None and time.time() < deadline - 40:

        a_refine, sc_refine = subgrad_optimize(

        if sc_refine < best_score:

            best_seq = a_refine.tolist()
            print(f"[{time.time():.1f}] refined subgrad best = {best_score:.6f}")

    # Stage 5: hill climbing with SA
    if best_seq is not None and time.time() < deadline - 5:
        print(f"[{time.time():.1f}] hill climbing (SA)")
        a_hill, sc_hill = hill_climb_sa(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=600_000,

        if sc_hill < best_score:
            best_score = sc_hill
            best_seq = a_hill.tolist()
            print(f"[{time.time():.1f}] hill climbing improved best = {best_score:.6f}")

    # Fallback uniform if nothing
    if best_seq is None:

        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] using uniform fallback score = {best_score:.6f}")
```

The objective  
`score = 2·n·max(a⊛a) / (∑a)²`  
is invariant under a uniform scaling of the vector, therefore we can (and do) work on the probability simplex `∑a = 1`.  
For a probability vector the minimum possible score is `2` (the uniform vector).  
The previous implementation already reaches ≈ 1.511  by shaping the vector so that the
central term of the auto‑correlation `∑ a_i·a_{n‑1‑i}` becomes very small.  

The new solver improves the previous result by

* a **high‑precision smoothed‑max optimisation** using an Adam optimiser and a
  much larger annealing parameter `β` (up to 300);
* a **refined projected sub‑gradient phase** with a tiny step‑size;
* a **simulated‑annealing hill‑climber** that accepts occasional worsening moves;
* a slightly larger problem size (`n = 16384` when the time budget permits);
* a few clean‑up utilities (FFT based convolution, simplex projection).

All stages respect the 1000 s wall‑clock limit and communicate their progress via
`print`.  The final candidate is returned as a plain Python list of floats.

```python
"""
Fast search for a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm consists of

1.   a one‑parameter “edge‑heavy’’ analytical start,
2.   an Adam‑based smoothed‑max gradient descent with a high β‑schedule,
3.   two rounds of projected sub‑gradient descent (the second with a tiny step),
4.   a simulated‑annealing hill‑climber.

All heavy work uses only NumPy and runs inside the 1000 s budget.
The returned value is a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution a * b using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of `evaluate_sequence` when the vector already satisfies
    ∑a = 1.  Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive → uniform point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Edge‑heavy analytical start
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    Produce a normalised vector a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α).
    The result lies on the probability simplex.
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # safety – replace infinities
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a modest grid of α‑values, keep the one with the smallest exact score.
    Returns (best_vector, best_score).
    """
    best_a = None
    best_score = float("inf")
    alphas = np.linspace(0.0, 2.5, 26)          # step ≈0.1
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Smoothed‑max optimisation with Adam
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 5.0,
                         beta_end: float = 300.0,
                         step: float = 0.5,
                         b1: float = 0.9,
                         b2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    using Adam (first‑ and second‑moment adaptation) and an exponential β‑schedule.
    The routine returns the best exact score seen.
    """
    a = start_a.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- β schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- smoothed max (weights) -----
        conv = convolution_fft(a, a)                 # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))   # stabilised exponentials
        w = exp_weights / exp_weights.sum()          # probability vector, ∑w = 1

        # ----- gradient (cross‑correlation) -----
        grad = 2.0 * convolution_fft(a[::-1], w)

        # ----- Adam update -----
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** it)
        v_hat = v / (1 - b2 ** it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- stay on the simplex -----
        a = project_to_simplex(a)

        # ----- occasional exact check -----
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient method for   2·n·max(a*a)
    with a 1/√t stepsise.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))               # active index

        # sub‑gradient of the max‑convolution entry k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_seq: np.ndarray,
                  deadline: float,
                  max_steps: int = 600_000,
                  initial_T: float = 0.02,
                  cooling: float = 0.99995) -> tuple[np.ndarray, float]:
    """
    Long stochastic climb that accepts worsening moves with probability
    exp(-Δ/T).  The temperature decays geometrically.  Returns the best vector
    seen and its exact score.
    """
    a = start_seq.copy()
    best_a = a.copy()
    best_score = compute_score(best_a)

    T = initial_T
    steps = 0
    n = a.size

    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a small mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]              # 5 % of the source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        delta = sc - best_score

        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            best_score = sc
            best_a = cand

        T *= cooling
        steps += 1

    return best_a, best_score
```
