def l_bfgs_refine(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    def fun(v):
        # v is unconstrained real vector; compute a = softmax(v)
        exp_v = np.exp(v - np.max(v))
        a = exp_v / np.sum(exp_v)
        # compute smooth max and gradient wrt a with epsilon
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * 0.02  # maybe 2% of max
        w = np.exp((conv - maxc) / eps)
        w_sum = np.sum(w)
        smooth = maxc + eps * np.log(w_sum)
        # gradient w.r.t a
        w_norm = w / w_sum
        # grad_a = 2 * conv(a, w_norm) as earlier.
        w_rev = w_norm[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2 * n - 2
        grad_a = 2.0 * conv_rev[idx_start - np.arange(n)]
        # gradient w.r.t v: a_i * (grad_a_i - np.dot(a, grad_a))
        dot = np.dot(a, grad_a)
        grad_v = a * (grad_a - dot)
        return smooth, grad_v.astype(np.float64)

    # Perform minimize with L-BFGS-B
    v0 = np.log(a0 + 1e-12)  # initial v
    # Set options with maxiter based on remaining time
    maxiter = 200
    # Attempt real-time check inside callback? Can define custom callback to stop early based on time.
    # However SciPy minimize does not support time stop directly; we can set maxiter moderate and then check time externally.
    # We'll bound maxiter at 200.

    result = minimize(fun, v0, method='L-BFGS-B', jac=True,
               options={'maxiter': maxiter, 'disp': False})

    # get solution
    v_opt = result.x
    exp_v = np.exp(v_opt - np.max(v_opt))
    a_opt = exp_v / np.sum(exp_v)
    # compute exact score
    sc = compute_score(a_opt)
    return a_opt, sc

"""
search_step_function_v2.py

Multi‑phase optimizer for the step‑function objective used in the
evaluation harness.  The pipeline improves on the previous version by
* trying several parametric families (Gaussian, raised‑cosine) as
  deterministic seeds,
* adding a fast L‑BFGS refinement of the smooth‑max surrogate when
  SciPy is available,
* using more aggressive Adam schedules and longer iteration budgets,
* and finally polishing with a long simulated‑annealing hill‑climber.
All vectors are kept on the probability simplex (non‑negative,
∑a=1) by an Euclidean projection after every update.  The routine
runs for the given time budget (≈1000 s) and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS refinement)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import minimize
    SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Fast FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (non‑negative, sum=1). """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisation utilities
# ----------------------------------------------------------------------


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1) distribution)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Phase 2 – flattening the convolution (L2 variance) using Adam
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 5000,
    step0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the self‑convolution (quartic loss)
    with Adam, projecting onto the simplex after each step.
    Returns the best vector encountered and its exact score.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)                 # length 2n‑1
        mu = np.mean(conv)
        diff = conv - mu

        # gradient of variance: 4 * Σ_j a_j * diff_{i+j}
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)      # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – smooth‑max surrogate minimisation (Adam)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 8000,
    step0: float = 0.6,
    eps_factor: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp surrogate of max‑convolution using Adam.
    The smoothing parameter ε is multiplied by the current maximum
    and decays geometrically.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε shrinks but never below 0.5 % of maxc
        eps = maxc * max(eps_factor * (0.992 ** (t - 1)), 0.005)

        # soft‑max weights
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)

        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape flat plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – exact sub‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact objective
        J(a) = 2·n·max(a⊛a) .
    The sub‑gradient is taken at the shift attaining the maximum.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k-i} (zero outside support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random shake every thousand iterations
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc

        if time.time() >= deadline:
            raise RuntimeError("time out")
        # soft‑max to obtain a
        v_shift = v - np.max(v)            # improve numerical stability
        exp_v = np.exp(v_shift)
        a = exp_v / np.sum(exp_v)

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε is a small fraction of the current maximum
        eps = maxc * 0.02
        exp_vals = np.exp((conv - maxc) / eps)
        sumexp = np.sum(exp_vals)
        smooth = maxc + eps * np.log(sumexp)

        # gradient w.r.t a
        w = exp_vals / sumexp
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2 * n - 2
        grad_a = 2.0 * conv_rev[idx_start - np.arange(n)]

        # chain rule for soft‑max (v → a)
        dot = np.dot(a, grad_a)
        grad_v = a * (grad_a - dot)

        return smooth, grad_v.astype(np.float64)

    # Run L‑BFGS with a modest iteration budget; we rely on early exit
    # via the RuntimeError when time is up.
    try:
        res = minimize(
            obj_and_grad,
            v0,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 150, "disp": False},
        )
        v_opt = res.x
    except RuntimeError:                                                   # pragma: no cover
        # timeout – keep the current best (v0)
        v_opt = v0

    v_shift = v_opt - np.max(v_opt)
    exp_v = np.exp(v_shift)
    a_opt = exp_v / np.sum(exp_v)
    sc = compute_score(a_opt)
    return a_opt, sc


# ----------------------------------------------------------------------
# Phase 6 – simulated‑annealing hill‑climber
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    deadline,
    max_steps: int = 500_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with linear cooling.
    Moves:
        * swap two positions,
        * transfer a small fraction of mass from i → j.
    Accepts worsening moves with probability exp(−Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    # linear cooling schedule
    T0 = 0.13

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.04 * best_a[i]     # move ~4 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional diversification
        if step % 6000 == 0:
            rnd = random_dense_vector(n) * 0.015   # 1.5 % random noise
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# Symmetry averaging helper
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Average a vector with its reverse and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a
    plain Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # safety margin

    best_seq = None
    best_score = float("inf")

    # ---- optional fallback to a known construction (if present) ----
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score, best_seq = fb_score, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic parametric families (Gaussian & raised‑cosine)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]   # keep FFTs cheap enough
    sigma_factors = [0.06, 0.10, 0.14, 0.20, 0.30, 0.45]  # like 6‑45 % of n
    cosine_ps = [0.5, 1.0, 2.0, 3.0, 5.0, 8.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- Gaussian windows ----
        idx = np.arange(n, dtype=float)
        centre = (n - 1) / 2.0
        for sf in sigma_factors:
            if time.time() >= deadline:
                break
            sigma = sf * n
            a = np.exp(-((idx - centre) ** 2) / (2.0 * sigma ** 2))
            a = a / np.sum(a)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[Gaussian] n={n} sigma={sf:.2f} score={sc:.6f}")

        # ---- Raised‑cosine windows ----
        for p in cosine_ps:
            if time.time() >= deadline:
                break
            a = (1.0 - np.cos(np.pi * idx / (n - 1))) ** p
            a = a / np.sum(a)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[Cosine] n={n} p={p:.1f} score={sc:.6f}")

        # ---- random dense starts (light) ----
        for _ in range(3):
            if time.time() >= deadline:
                break
            a = random_dense_vector(n)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[Random] n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Long‑form refinement of the current best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        # extremely unlikely, but keep a safe fallback
        n0 = 256
        best_seq = [1.0 / n0] * n0
        best_score = evaluate_sequence(best_seq)

    a_current = np.array(best_seq, dtype=float)

    # ---- Phase 2: flattening (variance of convolution) ----
    if time.time() < deadline - 5.0:
        a_flat, sc_flat = flatten_conv_adam(
            a_current,
            deadline=deadline - 2.0,
            max_iter=6000,
            step0=0.18,
        )
        if sc_flat < best_score:
            best_score, best_seq = sc_flat, a_flat.tolist()
            a_current = a_flat
            print(f"[Flatten] improved score={best_score:.6f}")

    # ---- Phase 3: smooth‑max surrogate (Adam) ----
    if time.time() < deadline - 6.0:
        a_smax, sc_smax = smooth_max_adam(
            a_current,
            deadline=deadline - 3.0,
            max_iter=10000,
            step0=0.65,
            eps_factor=0.33,
        )
        if sc_smax < best_score:
            best_score, best_seq = sc_smax, a_smax.tolist()
            a_current = a_smax
            print(f"[Smooth‑max] improved score={best_score:.6f}")

    # ---- Phase 4: optional L‑BFGS refinement (if SciPy is present) ----
    if SCIPY_AVAILABLE and time.time() < deadline - 4.0:
        a_lbfgs, sc_lbfgs = lbfgs_refine(a_current, deadline - 2.0)
        if sc_lbfgs < best_score:
            best_score, best_seq = sc_lbfgs, a_lbfgs.tolist()
            a_current = a_lbfgs
            print(f"[L‑BFGS] improved score={best_score:.6f}")

    # ---- Phase 5: exact sub‑gradient descent (fine‑tune) ----
    if time.time() < deadline - 8.0:
        a_sub, sc_sub = subgrad_optimize(
            a_current,
            deadline=deadline - 5.0,
            max_iter=150_000,
            step0=0.055,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.tolist()
            a_current = a_sub
            print(f"[Sub‑grad] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing hill‑climb (most time‑consuming phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        seq_to_polish = best_seq if best_seq is not None else a_current.tolist()
        polished_seq, polished_sc = hill_climb_anneal(
            seq_to_polish,
            deadline=deadline - 2.0,
            max_steps=400_000,
        )
        if polished_sc < best_score:
            best_score, best_seq = polished_sc, polished_seq
            print(f"[Anneal] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final cheap symmetry averaging (often helpful)
    # ------------------------------------------------------------------
    a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
    sym_sc = compute_score(a_sym)
    if sym_sc < best_score:
        best_score = sym_sc
        best_seq = a_sym.tolist()
        print(f"[Symmetry] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
