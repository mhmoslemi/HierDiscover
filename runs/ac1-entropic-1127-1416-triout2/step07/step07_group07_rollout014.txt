def subgradient_descent_topk(a0, time_deadline, max_iter, step0, K):
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline: break
        # compute conv
        conv = convolution_fft(a, a)
        # Determine top K indices
        if K >= conv.size:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            idx = np.argpartition(conv, -K)[-K:]
            w = np.zeros_like(conv)
            w[idx] = 1.0 / K
        # compute gradient as 2 * conv(a, w_rev) (center part)
        w_rev = w[::-1]
        full = convolution_fft(a, w_rev)
        start = n - 1
        grad = 2.0 * full[start:start + n]

        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc, best_a = cur_sc, a_new.copy()

        # occasional jitter
        if t % 2000 == 0:
            a_new += np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a_new = project_to_simplex(a_new)

        a = a_new
    return best_a, best_sc

def simulated_annealing(a_start, time_deadline, max_steps=100000):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    # temperature schedule: start T0 = 1e-3 maybe? Actually scoring differences small, we need relative.
    T0 = (best_sc) * 0.1  # not sure
    for step in range(max_steps):
        if time.time() >= time_deadline:
            break
        # propose move: transfer epsilon mass from i to j
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        # choose transferable amount: fraction of a[i], random between 0 & step_frac.
        epsilon = np.random.rand() * a[i]  # up to full a[i] maybe too large, but we will bound
        if epsilon <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= epsilon
        a_new[j] += epsilon
        a_new = project_to_simplex(a_new)

        sc_new = compute_score(a_new)
        # acceptance probability
        # temperature schedule: T = T0 * (0.999**step) maybe
        T = T0 * (0.999**step)
        if sc_new < best_sc or np.random.rand() < np.exp(-(sc_new - best_sc) / max(T, 1e-12)):
            a = a_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
        # maybe occasionally reduce epsilon or adapt step size...
    return best_a, best_sc

"""
step‑function optimiser – a multi‑phase heuristic

The optimiser works on the probability simplex (∑a_i = 1, a_i ≥ 0) so that
the cheap proxy  ``score = 2·n·max(conv)`` is exactly the value returned by
``evaluate_sequence`` (the latter rescales the same way).

Phases
------
1.  Initialise a “beta‑shape’’ distribution (generalised arcsine) with random
    parameters – this gives a wide variety of smooth start‑points.

2.  Adam‑accelerated smooth‑max descent.
    A soft‑max with a temperature that is exponentially cooled yields a
    differentiable surrogate of the max‑convolution; Adam provides fast
    convergence.

3.  Exact sub‑gradient descent on the true max‑convolution.
    The sub‑gradient at the active shift(s) is computed efficiently via a
    convolution; a decreasing step‑size  α/√t is used.

4.  Simulated‑annealing style hill‑climber.
    Small mass transfers between two random positions are proposed; the move
    is accepted if it improves the score or with a Metropolis probability that
    decays with a temperature schedule.  This phase escapes shallow local
    minima discovered by the gradient steps.

5.  (Optional) symmetry averaging – (a+reversed(a))/2 – followed by a
    projection onto the simplex.

All phases are timed; the algorithm stops well before the 1000 s budget
to leave room for the final call to ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Imports and basic utilities
# ----------------------------------------------------------------------
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, exact up to floating error)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive – return the uniform vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Scoring – exact replica of ``evaluate_sequence`` for sum=1 vectors
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """2·n·max(conv) for a probability vector a (∑a=1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Initialisation helpers – families of smooth shapes
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """Discrete arcsine‑like density a_i ∝ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def beta_initialisation(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑shape density.
    a_i ∝ x_i**(alpha‑1)·(1‑x_i)**(beta‑1) with x_i = (i+0.5)/n.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    w = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    return project_to_simplex(w)


# ----------------------------------------------------------------------
# 1. Adam optimisation on the smooth soft‑max surrogate
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of f(a)=eps·log Σ_k exp(conv_k/eps) where conv = a⋆a.
    Returns 2·(w_rev ⊛ a) with w = softmax(conv/eps).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)          # Σ w = 1

    # cross‑correlation with reversed w
    w_rev = w[::-1]
    full = convolution_fft(a, w_rev)         # length 3·n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.2,
    eps_factor: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    ``eps`` is decreased exponentially; the loop stops when the time budget
    expires.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # temperature – never smaller than 1 % of current max
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), 0.01)

        grad = _softmax_gradient(a, eps)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to avoid stagnation
        if t % 500 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.002 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent (max‑conv only) – optional top‑K variant
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on 2·n·max(a⋆a).
    At each step the gradient is taken at the (single) shift attaining the
    maximal convolution value.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i}
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        # occasional random shake
        if t % 1500 == 0:
            a_new += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a_new = project_to_simplex(a_new)

        a = a_new

    return best_a, best_sc


def subgradient_descent_topk(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.09,
    K: int = 3,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent where the gradient is an average over the K
    largest convolution entries (active set).  This smoothes the direction
    and often yields better final values.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)

        # build weight vector w that puts equal mass on the K largest shifts
        if K >= conv.size:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            idx_top = np.argpartition(conv, -K)[-K:]
            w = np.zeros_like(conv)
            w[idx_top] = 1.0 / K

        # gradient = 2·(w_rev ⊛ a) (centered part)
        w_rev = w[::-1]
        full = convolution_fft(a, w_rev)
        start = n - 1
        grad = 2.0 * full[start:start + n]

        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        # occasional jitter
        if t % 2000 == 0:
            a_new += np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a_new = project_to_simplex(a_new)

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Simulated‑annealing style hill‑climber
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves with Metropolis acceptance.
    The temperature decays exponentially; the algorithm accepts downhill moves
    always and uphill moves with probability exp(-Δ/T).
    """
    a = a_start.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    n = a.shape[0]

    # initial temperature ≈ 5 % of the current score (tunable)
    T0 = max(cur_sc * 0.05, 1e-6)

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        # pick source and destination indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue

        # transfer a random fraction (up to 20 %) of the source mass
        frac = np.random.rand() * 0.20
        delta = frac * a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = project_to_simplex(a_new)

        sc_new = compute_score(a_new)
        Δ = sc_new - cur_sc
        # exponential cooling
        T = T0 * (0.9995 ** step)

        if Δ < 0 or np.random.rand() < np.exp(-Δ / max(T, 1e-12)):
            a = a_new
            cur_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()

    return best_a, best_sc

def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a+reverse(a))/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)
