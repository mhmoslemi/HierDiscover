"""
search_step_function.py

Optimiser for the “step‑function” challenge.

The routine builds on the previous hierarchy (random seeding → Adam on a smooth‑max
surrogate → up‑sampling → long Adam → exact sub‑gradient descent → exponentiated
gradient polish).  The main differences are:
  * much larger time budget for each stage (the 1000 s limit is respected);
  * many more diverse random seeds (including Dirichlet draws and Gaussian bumps);
  * a higher final temperature in Adam (≈ 250) for a tighter surrogate;
  * longer Adam runs (up to 200 k iterations) and more aggressive step sizes;
  * a longer Euclidean projected sub‑gradient phase (≈ 5·10⁵ iterations);
  * an additional local‑perturbation/annealing loop after polishing.
All stages keep the globally best sequence (according to the official
``evaluate_sequence``) and finally return it.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a*b (length len(a)+len(b)-1).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate used during optimisation.
    The vector must be normalised (sum = 1); the returned value equals
    the official score without the denominator term:
        2 * n * max(conv(a, a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for seeding)

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine law discretisation."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Discrete beta‑like density."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_asymmetric_power_shape(
    n: int, p: float, eps_left: float, eps_right: float
) -> np.ndarray:
    """Asymmetric generalised arcsine."""
    i = np.arange(n, dtype=float)
    denom = ((i + eps_left) * (n - 0.5 - i + eps_right) + 1e-12) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gaussian_shape(n: int, mu: float, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at mu."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_dirichlet_shape(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑distributed probability vector."""
    sample = rng.gamma(concentration, 1.0, size=n)
    total = sample.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return sample / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(v) to target_n, then re‑normalise."""
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 30_000,
    tau_start: float = 0.5,
    tau_end: float = 150.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k)
    with an exponential temperature schedule τ ∈ [tau_start, tau_end].
    Variables are logits w so that a = softmax(w) guaranteeing a ≥ 0,
    Σ a = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits safely
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax to obtain a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()          # shape (L,)

        # gradient wrt a  : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wr.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging
        if it % 20_000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.10,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution objective.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # entries attaining the maximum (within floating tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) restricted to original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """

      0️⃣  Load a fallback solution if available.
      1️⃣  Generate a large pool of diverse seeds (n≈1024) using many parametric families.
      2️⃣  Run a medium‑length Adam optimisation on the best few seeds.
      3️⃣  Upscale the champion to larger lengths (2048 → 4096 → 8192) and repeat Adam.
      4️⃣  Exact sub‑gradient descent followed by an exponentiated‑gradient polish.
      5️⃣  A final stochastic local‑search loop (small random perturbations) if time allows.
    The best sequence found (according to the official evaluator) is returned.

    start_time = time.time()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        elif typ == 4:    # asymmetric power

        elif typ == 5:    # Gaussian bump (varying width)

            cand = generate_gaussian_shape(n0, mu, sigma)
        else:             # simple exponential decay

            i = np.arange(n0, dtype=float)

                refined,

                print("*** New global best! ***")
        except Exception as exc:
            print(f"Official evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # 5.  Final stochastic polishing (small random perturbations)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0 and best_seq:
        a = np.array(best_seq, dtype=float)
        # noise magnitude shrinks as we get closer to the deadline
        time_left = deadline - time.time()
        scale = 0.01 * (time_left / budget_s)   # slowly decay
        jitter = rng.normal(scale=scale, size=a.size)
        a = a + jitter
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            continue
        a = a / a.sum()
        try:
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val, best_seq = val, a.tolist()
                print(f"Improved by perturbation: {best_val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6.  Guard – ensure a non‑empty sequence is always returned
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
