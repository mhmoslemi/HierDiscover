"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  The algorithm combines:

1.  Analytic seed families (arcsine, beta, power, mixtures) on several
    lengths (4096 → 1024).  A cheap surrogate ``fast_score`` selects the
    most promising shapes.

2.  Smooth‑max optimisation with Adam and a continuation on the
    temperature β (log‑sum‑exp approximation of the max).  The simplex
    constraint is enforced via a soft‑max re‑parameterisation.

3.  Exact sub‑gradient descent on the true objective (projected onto the
    probability simplex) for polishing.

4.  A cheap stochastic local‑search that repeatedly moves a tiny amount
    of mass between two coordinates and accepts improvements.

The whole routine respects a 1000 s wall‑time budget and always returns a
valid non‑negative list.  If nothing better than a uniform sequence is
found, the fallback is a uniform distribution of length 512.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Fast linear convolution (real FFT – O(N log N))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (scale‑invariant, cheap)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/sum(a)^2 (no clipping)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic seeds
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density on {0,…,n‑1} (unnormalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """1 / ((i+0.5)*(n-0.5-i)+eps)^p  (normalized)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Soft‑max helper (used by Adam)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns probabilities)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    beta0: float = 5.0,
    beta_max: float = 800.0,
    ramp_iters: int = 120_000,
    lr: float = 0.045,
) -> np.ndarray:
    """
    Minimises Lβ(a) = log‑sum‑exp(β·conv(a,a)) with a on the simplex.
    β is increased linearly from ``beta0`` to ``beta_max`` (continuation).
    ``z0`` are the un‑constrained logits (soft‑max(z0) = initial a).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights (log‑sum‑exp trick)
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)               # length = 3·n‑2
        n = a.size
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]          # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 10_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected onto simplex)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv(a,a))/sum(a)^2  (with sum(a)=1 after projection).
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    Sub‑grad it={it:,d} score≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Simple stochastic local search (mass‑transfer moves)
# ----------------------------------------------------------------------


def stochastic_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    init_step: float = 0.02,
    max_trials: int = 30_000,
) -> np.ndarray:
    """
    Repeatedly move a tiny amount of probability mass from one index to another.
    Accepts only improvements of the fast surrogate.
    """
    a = a0.copy()
    best_score = fast_score(a)
    n = a.size
    step = init_step

    trials = 0
    while time.time() < deadline and trials < max_trials:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to transfer (fraction of source mass)
        eps = step * a[i]
        if eps <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # keep feasibility (non‑negativity) – already guaranteed by eps ≤ a[i]
        s = fast_score(a_new)
        if s < best_score:
            a = a_new
            best_score = s
            # slightly increase aggressiveness after a success
            step = min(step * 1.02, 0.05)
        else:
            # gradually decay step size when moves are not accepted
            step = max(step * 0.9995, 0.001)

        trials += 1
        if trials % 5000 == 0:
            print(f"    Local search trial={trials:,d} step={step:.5f} score≈{best_score:.6f}")

    return a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The procedure consists of:

    * Analytic seed generation (arcsine, beta‑family, power‑family, mixtures)
    * Coarse pre‑selection via ``fast_score``
    * Smooth‑max Adam optimisation with a long continuation on β
    * Exact sub‑gradient polishing
    * Light stochastic local‑search fine‑tuning
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # If a previous candidate is available, use it as a safe fallback
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose lengths adaptively (larger lengths give finer discretisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    lengths = [4096, 2048, 1024] if remaining > 250 else [2048, 1024] if remaining > 130 else [1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1. Generate a rich pool of analytic seeds
        # --------------------------------------------------------------
        seed_vectors: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),                     # uniform
            generate_mixture_shape(n, 0.6),          # 60 % arcsine, 40 % uniform
        ]

        # power‑family sweep
        for p in np.linspace(0.15, 2.5, 14):
            seed_vectors.append(generate_power_shape(n, p, eps=1e-9))

        # systematic beta‑family grid (α,β in [0.2, 1.2])
        alphas = np.linspace(0.2, 1.2, 11)
        betas = np.linspace(0.2, 1.2, 11)
        for a_val in alphas:

        # random beta draws for extra diversity
        for _ in range(8):
            a_rand = rng.uniform(0.1, 1.5)
            b_rand = rng.uniform(0.1, 1.5)
            eps_rand = rng.uniform(0.0, 0.01 * n)
            seed_vectors.append(generate_beta_shape(n, a_rand, b_rand, eps_rand))

        # --------------------------------------------------------------
        # 2. Coarse pre‑selection using the cheap surrogate
        # --------------------------------------------------------------
        scored = [(fast_score(vec), vec) for vec in seed_vectors]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]   # keep up to five per length

        # --------------------------------------------------------------
        # 3. Optimise each selected seed (Adam → sub‑grad → local search)
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 4.0:
                break

            # allocate a slice of remaining time for this seed
            remaining = deadline - time.time()
            per_seed_budget = max(2.0, min(30.0, remaining / (len(top_seeds) - idx + 1)))
            seed_deadline = time.time() + per_seed_budget

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – time budget ≈{per_seed_budget:.1f}s")
            # initialise logits for the soft‑max (avoid log(0))
            z0 = np.log(seed_vec + 1e-12)

            # ------------------------------------------------------
            #   Adam optimisation on smooth‑max surrogate
            # ------------------------------------------------------
            a_adam = smooth_adam_optimize(
                z0,
                seed_deadline,
                beta0=5.0,
                beta_max=800.0,
                ramp_iters=140_000,
                lr=0.045,
            )

            # ------------------------------------------------------
            #   Official evaluation of the Adam result
            # ------------------------------------------------------
            try:
                val = evaluate_sequence(a_adam.tolist())
                print(f"   Official score after Adam = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_adam.tolist()
                    print(f"   *** New global best = {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed (Adam result): {e}")

            # ------------------------------------------------------
            #   Exact sub‑gradient polishing (if time permits)
            # ------------------------------------------------------
            if time.time() < deadline - 3.0:
                sub_deadline = deadline - 1.0
                a_sub = subgradient_descent_exact(
                    np.array(best_seq, dtype=float),
                    sub_deadline,
                    max_iter=200_000,
                    step0=0.07,

                try:

                    print(f"   Score after sub‑grad = {val_sub:.6f}")
                    if val_sub < best_val:

                except Exception:

            # ------------------------------------------------------

            # ------------------------------------------------------

        except Exception:

    # ------------------------------------------------------------------
    # 5. Ensure we always return a non‑empty valid sequence
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
