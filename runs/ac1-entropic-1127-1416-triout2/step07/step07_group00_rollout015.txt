"""
step‑function optimizer
-----------------------

This implementation searches for a non‑negative sequence `a` (sum(a)=1) that
minimises the evaluation function

    2·n·max(a * a)                     (the true objective after normalisation)

The core idea is to *flatten* the autocorrelation `a*a`.  For a continuous
distribution the arcsine law makes the convolution perfectly uniform, but the
discretisation error prevents the naïve arcsine vector from reaching the
theoretical lower bound 1.  We therefore:

1. Initialise with the arcsine‑type shape a_i ∝ [(i+½)(n‑½‑i)]⁻ᵅ  (α≈0.5).
2. Solve a smooth L²‑fit:
       minimise ‖conv(a,a) – u‖²   where u = 1/(2n‑1)⋅1
   using a soft‑max parametrisation and L‑BFGS (or a simple gradient descent
   if SciPy is unavailable).  This directly pushes the convolution towards the
   uniform target, which also reduces the max‑entry.
3. Polish the result with a long stochastic hill‑climb that only accepts
   moves that improve the *exact* `evaluate_sequence` score.
4. Return the best sequence found before the 1000 s wall‑clock deadline.

All heavy work uses only NumPy (FFT‑based convolutions) and fits comfortably
inside the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of `evaluate_sequence` for a probability vector (sum=1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    zz = z - np.max(z)
    e = np.exp(zz)
    return e / np.sum(e)


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """
    Analytic start vector a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α),
    already normalised to sum=1.
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect from any inf values (should not happen for α>0)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Very long stochastic walk on the simplex:
    - with probability 0.5 swap two entries,
    - otherwise transfer a fraction of mass from one index to another.
    Only moves that improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12   # starting transfer magnitude

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        # tiny numerical drift – re‑project onto simplex
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_score


# ----------------------------------------------------------------------
# L2 optimisation (conv → uniform)
# ----------------------------------------------------------------------


def l2_optimize(start_a: np.ndarray,
                deadline: float,
                max_iter_lbfgs: int = 200) -> tuple[np.ndarray, float]:
    """
    minimise ‖conv(a,a) – u‖²  where u is the uniform target.
    Uses SciPy's L‑BFGS‑B on the log‑softmax variables.
    Falls back to a simple manually‑implemented gradient descent if SciPy
    cannot be imported.
    Returns the best vector (still a probability vector) and its exact score.
    """
    n = start_a.shape[0]
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # initialise logits – softmax(z)=start_a
    z0 = np.log(start_a + 1e-30)

    # try to import SciPy
    try:
        import scipy.optimize as opt
        HAVE_SCIPY = True
    except Exception:  # pragma: no cover
        HAVE_SCIPY = False

    # --------------------------------------------------------------
    # auxiliary: loss + gradient given z (logits)
    # --------------------------------------------------------------
    def loss_and_grad(z):
        a = softmax(z)
        conv = convolution_fft(a, a)
        r = conv - target                     # residual
        loss = np.dot(r, r)                   # L2 loss

        # gradient w.r.t a
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(r, a_rev)[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        return loss, grad_z

    # --------------------------------------------------------------
    # SciPy L‑BFGS branch
    # --------------------------------------------------------------
    if HAVE_SCIPY:
        # we need a callback to stop early if time runs out
        class TimeLimiter:
            def __init__(self, deadline):
                self.deadline = deadline
                self.stop = False

            def __call__(self, xk):
                if time.time() >= self.deadline:
                    self.stop = True
                    return True
                return False

        limiter = TimeLimiter(deadline)

        try:
            res = opt.minimize(
                fun=loss_and_grad,
                x0=z0,
                method="L-BFGS-B",
                jac=True,
                callback=limiter,
                options={"maxiter": max_iter_lbfgs, "ftol": 1e-20, "gtol": 1e-12}
            )
            best_a = softmax(res.x)
            best_score = compute_score(best_a)
            return best_a, best_score
        except Exception:  # pragma: no cover
            # fall back to manual gradient descent
            pass

    # --------------------------------------------------------------
    # Manual gradient descent (soft‑max parametrisation)
    # --------------------------------------------------------------
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    step0 = 1.0
    for it in range(1, 2_000 + 1):
        if time.time() >= deadline:
            break

        # compute loss & gradient
        conv = convolution_fft(a, a)
        r = conv - target
        loss = np.dot(r, r)

        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(r, a_rev)[n - 1:n - 1 + n]

        # convert to logits
        z = np.log(a + 1e-30)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        step = step0 / np.sqrt(it)
        z_new = z - step * grad_z
        a = softmax(z_new)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Uses an L²‑based uniform‑convolution optimisation followed by a hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness itself
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # choose length – 8192 gives a good trade‑off between flexibility and speed
    # --------------------------------------------------------------
    n = 8192
    if deadline - start_time < 400.0:
        n = 4096
    print(f"[{time.time():.1f}] using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. optional fallback using a previously found construction
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. analytic arcsine start (α≈0.5)
    # --------------------------------------------------------------
    a_start = alpha_start(n, 0.5)
    sc_start = compute_score(a_start)
    if sc_start < best_score:
        best_score = sc_start
        best_seq = a_start.tolist()
        print(f"[{time.time():.1f}] arcsine start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. L2 optimisation (flatten convolution)
    # --------------------------------------------------------------
    if time.time() < deadline - 100.0:
        print(f"[{time.time():.1f}] starting L2 uniform‑convolution optimisation")
        a_opt, sc_opt = l2_optimize(np.array(best_seq, dtype=float), deadline - 80.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()

    # --------------------------------------------------------------
    # 3. final hill‑climb polishing (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:

        a_final, sc_final = hill_climb(np.array(best_seq, dtype=float),

            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. safety net – uniform vector if everything else failed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (the harness will also call evaluate_sequence)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # the harness expects a plain Python list of floats
    return best_seq
