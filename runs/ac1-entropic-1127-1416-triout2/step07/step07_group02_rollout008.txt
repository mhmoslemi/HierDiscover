def propose_candidate(seed=0, budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    rng = np.random.default_rng(seed)

    # 1. Generate seeds across lengths
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5:
            break
        # produce seed shapes
        seeds = []
        # arcsine exponent p grid
        for p in np.linspace(0.35, 0.9, 12):
            shape = generate_power_shape(n, p, eps=1e-8) # using denominator^p? Actually power shape defined earlier as 1/denom^p ; but for arcsine p maybe 0.5 corresponds to arcsine. That shape.
            seeds.append(shape)
        # uniform
        seeds.append(np.full(n, 1.0/n))
        # beta shapes grid
        for alpha in [0.4, 0.7, 1.0, 1.5, 2.0]:
            for beta in [0.4, 0.7, 1.0, 1.5, 2.0]:
                if alpha==beta==1.0: # uniform
                    continue
                shape = generate_beta_shape(n, alpha, beta, eps=1e-8)
                seeds.append(shape)
        # random Dirichlet draws
        for _ in range(8):
            vec = rng.dirichlet(np.ones(n))
            seeds.append(vec)

        # evaluate fast score and keep top k
        scored = [(fast_score(vec), vec) for vec in seeds]
        scored.sort(key=lambda x: x[0])
        top_k = 5
        top = [vec for _, vec in scored[:top_k]]

        # allocate per seed time: (deadline - time())/(len(top)*(some factor))
        per_seed_time = max(1.0, (deadline - time.time()) / (len(top) * 4))  # 4 phases per seed approx?

        for idx, init_vec in enumerate(top):
            if time.time() >= deadline - 2:
                break
            # Run Adam optimization
            sub_deadline = min(deadline - 0.5, time.time() + per_seed_time * 0.5)
            # Use log initialization: avoid 0
            eps = 1e-12
            z0 = np.log(init_vec + eps)
            a_opt = smooth_adam_optimize(z0, sub_deadline, n=n, beta0=5., beta_max=400., ramp_iters=80_000, lr=0.05)

            # Evaluate official
            try:
                val = evaluate_sequence(a_opt.tolist())
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                # Possibly print.
            except Exception as e:
                pass

            # Run exact subgradient descent for a while (maybe improved)
            subg_deadline = min(deadline - 0.5, time.time() + per_seed_time * 0.5)
            a_sub = subgradient_descent_exact(a_opt, subg_deadline, max_iter=200_000, step0=0.07)
            try:
                val2 = evaluate_sequence(a_sub.tolist())
                if val2 < best_val:
                    best_val, best_seq = val2, a_sub.tolist()
            except Exception:
                pass

            # Local random perturbation (optional)
            if time.time() < deadline - 1:
                a_pert = random_perturb(best_seq, rng, max_iters=int(5_000), time_limit=deadline - 1)

"""
Hybrid search for a low‑value step sequence (improved version).

Key ideas
---------
* Explore a richer family of analytic seeds (power‑law / beta / Dirichlet
  distributions) on several lengths (4096, 2048, 1024).  The seed pool
  is filtered with a cheap scale‑invariant surrogate
  ``fast_score`` (2·n·max(conv)/sum²).

* For each promising seed run a *smooth‑max* optimisation (log‑sum‑exp
  surrogate) using Adam with a continuation on the temperature β.
  The iterate lives on the simplex via a soft‑max re‑parameterisation.

* Refine the Adam output with an *exact* sub‑gradient descent on the
  true objective.  The sub‑gradient of the max‑convolution is computed
  analytically; each step is projected onto the probability simplex.

* Perform a cheap stochastic hill‑climbing phase that swaps small amounts
  of mass between random coordinates.  The surrogate guides the search
  and the official ``evaluate_sequence`` is called only for the current
  best candidate.

* Finally, a long run of the exact sub‑gradient method polishes the
  globally best sequence found.

All steps respect the global time budget (default 1000 s).  The routine
always returns a non‑negative sequence; if nothing better is found a
uniform fallback is used.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate used for pre‑selection (scale‑invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Cheap surrogate of the official metric:
    2·n·max(conv(a,a))/sum(a)²  (no clipping, no safety checks).
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families for seed generation
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    a_i ∝ 1 / ((i+0.5)*(n‑0.5‑i) + eps) ** p .
    For p=0 returns the uniform distribution; p>0 concentrates mass
    near the ends (p≈0.5 reproduces the arcsine shape).
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-8) -> np.ndarray:
    """
    Beta‑type density on the discrete grid.
    a_i ∝ (i+eps)^{alpha‑1}·(n‑1‑i+eps)^{beta‑1}.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on
    f(a)=2·n·max(conv(a,a))/sum(a)².
    The iterate stays on the simplex via Euclidean projection.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

    return a


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) + Adam optimiser (β‑continuation)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 400.0,
    ramp_iters: int = 80_000,
    lr: float = 0.05,
) -> np.ndarray:
    """
    Minimise the smooth approximation
        L_β(a) = log‑sum‑exp(β·conv(a,a))
    while keeping a on the simplex via a soft‑max map.
    β is increased linearly from beta0 to beta_max (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    [Adam] it={t:,d} β={beta:.1f}  surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Stochastic local perturbation (cheap hill‑climbing)
# ----------------------------------------------------------------------


def stochastic_perturb(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iter: int = 5_000,
) -> np.ndarray:
    """
    Simple random walk on the simplex:
    repeatedly move a tiny amount of mass from one coordinate to another.
    Acceptance is greedy w.r.t. the cheap fast_score surrogate.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = fast_score(best)

    # magnitude of a single move – a small fraction of the total mass
    eps = 1.0 / (n * 200.0)

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = eps * rng.random()
        if a[i] < delta:
            continue
        a[i] -= delta
        a[j] += delta
        sc = fast_score(a)
        if sc < best_score:
            best_score, best = sc, a.copy()
        else:
            # revert (greedy walk)
            a[i] += delta
            a[j] -= delta
        it += 1

    return best

                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ----------------------------------------------------------
            # 2d. Optional stochastic perturbation if time permits
            # ----------------------------------------------------------
            if time.time() < deadline - 1.0:
                pert_deadline = deadline - 0.5
                a_pert = stochastic_perturb(a_sub, rng, pert_deadline, max_iter=6_000)
                try:
                    val_pert = evaluate_sequence(a_pert.tolist())
                    print(f"   Perturb‑refined score = {val_pert:.6f}")
                    if val_pert < best_val:
                        best_val, best_seq = val_pert, a_pert.tolist()
                        print(f"   *** Updated best after perturbation: {best_val:.6f} ***")
                except Exception:
                    pass

    # ------------------------------------------------------------------
    # 5. Final global polishing (long exact sub‑gradient run)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        final_deadline = deadline - 0.2
        a_final = subgradient_descent_exact(
            a_arr,
            final_deadline,
            max_iter=300_000,
            step0=0.06,
        )
        try:
            final_val = evaluate_sequence(a_final.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_final.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty valid return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
