    i = np.arange(2*n-1)[:, None]

            # adjust previous best to length n
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()
            seeds.append(cand)

        # Evaluate seeds quickly via compute_score
        seed_scores = [(compute_score(s), s) for s in seeds]
        seed_scores.sort(key=lambda x: x[0])
        top_seeds = [s for _, s in seed_scores[:4]]

        # allocate per-seed time slice
        remaining_time = deadline - time.time()
        per_seed_time = max(2.0, remaining_time / (len(top_seeds) * 2.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"\nSeed {idx}/{len(top_seeds)}: length {n}, budget {per_seed_time:.1f}s")
            seed_deadline = min(time.time() + per_seed_time, deadline - 2.0)
            a = seed_vec.copy()
            a = a / a.sum()

            # --- LP iterative improvement ---
            max_lp_iters = 30
            t_mix = 0.5
            while time.time() < seed_deadline and max_lp_iters > 0:
                max_lp_iters -= 1
                conv = convolution_fft(a, a)
                M = float(conv.max())
                # Solve LP for g
                try:
                    from scipy.optimize import linprog
                except Exception as e:
                    print("SciPy linprog not available, skipping LP phase.")
                    break
                # Build convolution matrix
                i = np.arange(2 * n - 1)[:, None]
                j = np.arange(n)
                k = i - j
                mask = (k >= 0) & (k < n)
                A = np.zeros((2 * n - 1, n), dtype=float)
                A[mask] = a[k[mask]]
                # Linear program: maximize sum(g)  => minimize -sum(g)
                c = -np.ones(n, dtype=float)
                b_ub = np.full(2 * n - 1, M, dtype=float)
                bounds = [(0, None)] * n
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'tol':1e-9, 'presolve': True})
                if not res.success:
                    # No feasible improvement
                    break
                g_raw = res.x
                sum_g = g_raw.sum()
                if sum_g <= 0:
                    break
                g = g_raw / sum_g  # normalized to sum=1

                # try mixing
                t = t_mix
                improved = False
                while t > 1e-5:
                    new_a = (1 - t) * a + t * g
                    new_score = compute_score(new_a)
                    if new_score < compute_score(a) - 1e-12:
                        a = new_a
                        improved = True
                        print(f"   LP it improved score to {new_score:.6f} with t={t:.4f}")
                        break

                print(f"   Official score after LP = {val:.6f}")

                    print(f"   *** New global best {best_score:.6f} ***")
            except Exception as e:

            if time.time() < deadline - 5.0:

                                              beta_max=250.0, ramp_iters=30000, lr=0.07)
                try:

                    print(f"   Official after Adam = {val2:.6f}")
                    if val2 < best_score:
                        best_score, best_seq = val2, a_adam.tolist()
                        print(f"   *** New best after Adam {best_score:.6f} ***")
                except Exception as e:
                    print("   Adam evaluation error:", e)

            # --- Final EG polish if time remaining ---
            if time.time() < deadline - 3.0:
                a_eg = exponentiated_gradient_polish(np.array(best_seq, dtype=float), deadline, max_iter=20000, step=0.015)
                try:
                    val3 = official_score(a_eg.tolist())
                    print(f"   Final EG score = {val3:.6f}")
                    if val3 < best_score:
                        best_score, best_seq = val3, a_eg.tolist()
                        print(f"   *** New best after final EG {best_score:.6f} ***")
                except Exception as e:
                    print("   EG polish failed:", e)

        # After seeds, consider upscaling if time permits
        upscale_factors = [2, 2]  # scale to 2048 then to 4096
        current_seq = np.array(best_seq, dtype=float) if best_seq else None
        # If best_seq length still base length, attempt upscale
        for factor in upscale_factors:

            up_seq = np.repeat(current_seq, factor) / factor
            up_seq = up_seq[:target_len]  # trim if overshoot

            print(f"\nUpscaling to length {target_len}, running quick Adam")
            z0_up = np.log(up_seq + 1e-12)

            try:

"""
Propose a step function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Diverse seeds** – generate a handful of initial vectors of a given
   length (arcsine, uniform, random beta / power shapes).  The current
   best sequence (if any) is also added to the pool.

2. **Linear‑programming mixing** – for each seed we repeatedly solve the
   LP

        maximise   Σ g_i
        subject to conv(a, g) ≤ M      (M = max(conv(a, a)))
                    g_i ≥ 0

   where ``conv`` is the linear convolution (linear in ``g``).  The LP
   returns a feasible vector ``g`` whose own sum is then normalised to
   1.  We move a little towards ``g`` (exponential‑mixing with a
   tunable step size) and repeat; each successful mix provably does not
   increase the maximal auto‑convolution value.

3. **Smooth surrogate polishing** – after the LP phase the best vector
   is refined with Adam on the log‑sum‑exp smooth approximation of the
   max‑convolution.  The temperature β is annealed from 5 up to 250 so
   that optimisation first sees a smooth landscape and then a sharp
   version of the true objective.

4. **Exponentiated‑gradient polishing** – a few hundred iterations of
   KL‑mirror descent on the exact non‑smooth objective tighten the bound
   further.

5. **Upscaling** – the final 1024‑point solution is up‑scaled by
   integer repetition (each coefficient is split evenly among several
   neighbours).  The up‑scaled vector is again polished with a short
   Adam run; this often reduces the max‑auto‑correlation because the
   mass is spread more uniformly.

6. **Global sub‑gradient polishing** – a final projected sub‑gradient
   descent (the same routine used in the reference solution) is run
   until the wall‑clock budget is exhausted.

All steps respect the 1000 s budget and the return value is guaranteed
to be a non‑empty list of non‑negative numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective when ``a`` already sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Generators for diverse initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Projection utilities for the simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # identify all near‑maximal entries (numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 200.0,
    ramp_iters: int = 20000,
    lr: float = 0.05,
) -> np.ndarray:
    """
    Minimise the smooth log‑sum‑exp surrogate of max(conv) using Adam.
    The iterates are kept on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly raise the temperature β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp weighting
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2 * (a ⋆ w_rev) (restricted to original length)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final vector on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 50_000,
    step: float = 0.02,
) -> np.ndarray:
    """
    Mirror descent on the original max‑convolution objective using the
    KL divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for the set of active (maximal) indices
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming mixing utilities
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return the (2n‑1)×n matrix A such that    conv(a, g) = A @ g .
    Row i corresponds to the convolution entry at shift i.
    """
    n = a.shape[0]
    i = np.arange(2 * n - 1)[:, None]          # shape (2n‑1,1)
    j = np.arange(n)                           # shape (n,)
    k = i - j                                   # broadcasting → (2n‑1,n)
    mask = (k >= 0) & (k < n)
    A = np.zeros((2 * n - 1, n), dtype=float)
    A[mask] = a[k[mask]]
    return A


def lp_max_sum(a: np.ndarray, M: float, deadline: float):
    """
    Solve  maximise Σ g_i   subject to   conv(a, g) ≤ M,   g ≥ 0.
    Returns a normalised feasible vector (sum = 1) or None on failure.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return None

    n = a.shape[0]
    A = build_conv_matrix(a)                # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)            # minimise –Σ g_i
    bounds = [(0, None)] * n

    # Run the interior‑point “highs” solver – fast and robust.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'tol': 1e-9})

    if not res.success or res.x is None:
        return None

    g_raw = res.x
    s = g_raw.sum()
    if s <= 0:
        return None
    # normalise to the simplex (sum = 1) – the mixing step will preserve the unit sum
    return g_raw / s

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

            return evaluate_sequence(seq)

    if "height_sequence_1" in globals():
        try:

                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Base length optimisation (1024 points) – LP + Adam + EG

            b = rng.uniform(0.1, 5.0)

            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)

            if old.size >= base_n:

            a = seed_vec.copy()
            a = a / a.sum()                     # ensure unit sum

            # ------------------ LP iterative mixing ------------------------
            mix_step = 0.5
            lp_iters = 30
            while time.time() < seed_deadline and lp_iters > 0:
                lp_iters -= 1
                cur_score = compute_score(a)
                conv = convolution_fft(a, a)
                M = float(conv.max())

                g = lp_max_sum(a, M, seed_deadline)
                if g is None:
                    break

                # try to move a little towards g; shrink step if no improvement
                t = mix_step
                improved = False
                while t > 1e-5:
                    a_new = (1 - t) * a + t * g
                    new_score = compute_score(a_new)
                    if new_score < cur_score - 1e-12:

                        improved = True

                    t *= 0.5
                if not improved:
                    # No progress – stop the LP loop for this seed
                    break

            # ------------------ official evaluation ------------------------
            val = official_score(a.tolist())
            print(f"   Official score after LP = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a.tolist()
                print(f"   *** New global best {best_val:.6f} ***")

            # ------------------ Adam polishing (smooth surrogate) ---------
            if time.time() < deadline - 4.0:
                z0 = np.log(a + 1e-12)
                a_adam = smooth_adam_optimize(
                    z0,
                    deadline,
                    n=base_n,
                    beta0=5.0,
                    beta_max=250.0,
                    ramp_iters=30000,
                    lr=0.07,
                )
                val2 = official_score(a_adam.tolist())
                print(f"   Official after Adam = {val2:.6f}")
                if val2 < best_val:
                    best_val, best_seq = val2, a_adam.tolist()
                    print(f"   *** New best after Adam {best_val:.6f} ***")

            # ------------------ EG polishing -------------------------------
            if time.time() < deadline - 3.0:
                a_eg = exponentiated_gradient_polish(
                    np.array(best_seq, dtype=float),
                    deadline,
                    max_iter=20000,
                    step=0.015,
                )
                val3 = official_score(a_eg.tolist())
                print(f"   EG polished score = {val3:.6f}")
                if val3 < best_val:
                    best_val, best_seq = val3, a_eg.tolist()
                    print(f"   *** New best after EG {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Upscaling to larger lengths (2048 → 4096) and quick polishing
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 10.0:
        current = np.array(best_seq, dtype=float)
        for factor in (2, 2):               # 1024 → 2048 → 4096
            target_len = current.size * factor
            if target_len > 4096:
                break
            print(f"\nUpscaling to length {target_len}")
            up = np.repeat(current, factor) / factor
            up = up[:target_len]           # safety trimming
            up = up / up.sum()

            # a short Adam run on the up‑scaled vector
            z0 = np.log(up + 1e-12)
            a_up = smooth_adam_optimize(
                z0,
                deadline,
                n=target_len,
                beta0=5.0,
                beta_max=250.0,
                ramp_iters=20000,
                lr=0.07,
            )
            val_up = official_score(a_up.tolist())
            print(f"   Score after up‑scale+Adam = {val_up:.6f}")
            if val_up < best_val:
                best_val, best_seq = val_up, a_up.tolist()
                current = a_up
                print(f"   *** New best after up‑scale {best_val:.6f} ***")
            else:
                # keep the previous (better) vector for possible further up‑scaling
                current = np.array(best_seq, dtype=float)

    # ------------------------------------------------------------------
    # 3.  Final global sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=180_000,
            step0=0.07,
        )
        final_val = official_score(a_arr.tolist())
        print(f"   Final official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a_arr.tolist()
            print("*** Global best improved by final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = official_score(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
