def resize_sequence(a, new_n):
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_resized = np.interp(x_new, x_old, a)
    # ensure non-negative, re-normalize to sum=1
    a_resized = np.maximum(a_resized, 0)
    total = a_resized.sum()
    if total <= 0:
        a_resized = np.full(new_n, 1.0 / new_n)
    else:
        a_resized = a_resized / total
    return a_resized
```

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate (scale‑invariant) – used for pre‑selection
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Cheap proxy for the official objective (no clipping)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

# Analytic families for seed generation
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:

    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_linear_decrease(n: int, eps: float = 1e-12) -> np.ndarray:
    """Monotone decreasing linear profile."""
    vals = np.linspace(1.0, eps, n, dtype=float)
    return vals / vals.sum()


def generate_exponential_decay(n: int, rate: float = 0.001) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-rate * i)
    return vals / vals.sum()


def generate_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with concentration)."""
    v = rng.gamma(conc, 1.0, size=n)
    tot = v.sum()
    if tot <= 0:
        return np.full(n, 1.0 / n)
    return v / tot


def resize_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s <= 0:
        b = np.full(new_n, 1.0 / new_n)
    else:
        b = b / s
    return b


# ----------------------------------------------------------------------
# Projection utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max + Adam optimisation (continuation on β)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns probabilities that sum to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 1500.0,
    ramp_iters: int = 80000,
    lr: float = 0.03,
) -> np.ndarray:
    """
    Minimise the smooth approximation   L_β(a)=log ∑ exp(β·conv(a,a))
    while keeping a on the simplex via the soft‑max map.
    β grows linearly from ``beta0`` to ``beta_max`` (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥0 , Σa = 1
        conv = convolution_fft(a, a)
        # max element (needed for the numerics of the soft‑max)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    [Adam] iter={t:,d} β={beta:.1f}  surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Mirror‑descent (exact max sub‑gradient)
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.15,
    decay: float = 0.9995,
) -> np.ndarray:
    """
    Exponential‑gradient descent on the true objective
    f(a)=2·n·max(conv(a,a))/sum(a)^2.
    The iterate stays on the simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for the active set
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        if int(step * 1e6) % 200_000 == 0:
            print(f"    [Mirror] step={step:.6f}  surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.08,
) -> np.ndarray:
    """Projected sub‑gradient descent on the true (non‑smooth) objective."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    [SubGrad] it={it:,d}  surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Simple stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------


def stochastic_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    init_noise: float = 0.05,
    max_trials: int = 2000,
) -> np.ndarray:
    """
    Small‑scale random walk on the simplex.  Accepts a new point if the
    *official* objective improves.  The noise amplitude is slowly
    reduced.
    """
    best_a = a0.copy()
    try:
        best_val = evaluate_sequence(best_a.tolist())
    except Exception:
        best_val = float("inf")

    noise = init_noise
    it = 0
    while time.time() < deadline and it < max_trials:
        it += 1
        cand = best_a + noise * rng.standard_normal(best_a.shape)
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand)
        try:
            cand_val = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if cand_val < best_val:
            best_val, best_a = cand_val, cand
            noise = max(noise * 0.97, 1e-4)  # shrink a little
        else:
            noise = min(noise * 1.01, init_noise)  # slowly increase again

        if it % 500 == 0:
            print(f"    [Local] trial={it:,d}  best≈{best_val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine combines a large, diverse
    pool of analytic/random seeds, a continuation‑Adam phase with a
    high temperature schedule, a short stochastic refinement, mirror‑
    descent and a final projected sub‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.4   # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0.  Use a previously discovered construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"[Fallback] initial score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Choose problem sizes (the larger the better – we try 4096)
    # --------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ----------------------------------------------------------
        # 1a.  Build a rich pool of seeds
        # ----------------------------------------------------------
        seed_vectors: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_linear_decrease(n),
            generate_exponential_decay(n, rate=0.002),
        ]

        # power‑law family
        for p in np.linspace(0.2, 2.5, 8):
            seed_vectors.append(generate_power_shape(n, p, eps=1e-9))

        # random Dirichlet draws (sparse)
        for _ in range(6):
            conc = rng.uniform(0.05, 0.4)
            seed_vectors.append(generate_dirichlet(n, conc, rng))

        # random beta‑family draws
        for _ in range(4):
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            eps = rng.uniform(0.0, 0.01 * n)
            seed_vectors.append(generate_beta_shape(n, a, b, eps))

        # reuse the current best (if any) by interpolation
        if best_seq:
            seed_vectors.append(resize_sequence(np.array(best_seq, dtype=float), n))

        # ----------------------------------------------------------
        # 1b.  Keep the most promising seeds (fast surrogate)
        # ----------------------------------------------------------
        scored = [(fast_score(v), v) for v in seed_vectors]
        scored.sort(key=lambda x: x[0])
        top_seeds = [v for _, v in scored[:4]]   # at most four per length

        # ----------------------------------------------------------
        # 2.  Allocate a slice of the remaining budget for each seed
        # ----------------------------------------------------------
        remaining = max(0.0, deadline - time.time())
        per_seed = max(2.0, remaining / (len(top_seeds) * 2.5))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 2.0:
                break

            print(f"\n[Seed {idx}/{len(top_seeds)}] n={n} – budget≈{per_seed:.1f}s")
            seed_deadline = time.time() + per_seed

            # ------------------------------------------------------------------
            # 2a.  Adam optimisation on smooth‑max (continuation on β)
            # ------------------------------------------------------------------
            z0 = np.log(seed_vec + 1e-12)          # log‑softmax start
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=1500.0,
                ramp_iters=80000,
                lr=0.03,
            )

            # ------------------------------------------------------------------
            # 2b.  Official evaluation of the Adam result
            # ------------------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   [Adam] official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** Global best improved to {best_val:.6f} ***")
            except Exception as e:
                print(f"   [Adam] evaluation failed: {e}")

            # ------------------------------------------------------------------
            # 2c.  Short stochastic local search (Gaussian perturbations)
            # ------------------------------------------------------------------
            if time.time() < deadline - 1.0:
                local_deadline = time.time() + per_seed * 0.3
                a_local = stochastic_local_search(
                    np.array(best_seq, dtype=float),
                    local_deadline,
                    rng,
                    init_noise=0.04,
                    max_trials=800,
                )
                try:
                    val_loc = evaluate_sequence(a_local.tolist())
                    print(f"   [Local] refined score = {val_loc:.6f}")
                    if val_loc < best_val:
                        best_val, best_seq = val_loc, a_local.tolist()
                        print(f"   *** Global best after local search = {best_val:.6f} ***")
                except Exception:
                    pass

            # ------------------------------------------------------------------
            # 2d.  Mirror‑descent refinement (exact max sub‑gradient)
            # ------------------------------------------------------------------
            if time.time() < deadline - 1.0:
                md_deadline = deadline - 0.5
                a_mirror = mirror_descent_optimize(
                    np.array(best_seq, dtype=float),
                    md_deadline,
                    step0=0.15,
                    decay=0.9995,
                )
                try:
                    val_md = evaluate_sequence(a_mirror.tolist())
                    print(f"   [Mirror] refined score = {val_md:.6f}")
                    if val_md < best_val:
                        best_val, best_seq = val_md, a_mirror.tolist()
                        print(f"   *** Global best after mirror = {best_val:.6f} ***")
                except Exception:
                    pass

    # --------------------------------------------------------------
    # 3.  Final global polishing with projected sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\n[Final polishing] projected sub‑gradient descent")
        a_arr = np.array(best_seq, dtype=float)
        polish_deadline = deadline - 0.2
        a_polish = subgradient_descent_exact(
            a_arr,
            polish_deadline,
            max_iter=200_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_polish.tolist())
            print(f"   [Polish] final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_polish.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print("[Guard] fallback uniform sequence used")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
