   current = seq.copy()
   current_score = evaluate_sequence(current.tolist())
   best = current.tolist()
   best_score = current_score
   for i in range(steps):
       if time.time()>=deadline: break
       T = start_temp * (end_temp/start_temp) ** (i/steps)
       # propose new: add gaussian noise scaled to variance maybe 0.1*T?
       cand = current + (np.random.randn(len(seq)) * T)
       # project to simplex
       cand = _project_to_simplex(cand)
       cand_score = evaluate_sequence(cand.tolist())

def anneal_sequence(seq: np.ndarray, deadline: float, start_temp: float = 0.1, end_temp: float = 1e-4, max_iter: int = 200_000) -> np.ndarray:
    cur = seq.copy()
    cur_score = fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    n = len(seq)
    it = 0
    while time.time() < deadline and it < max_iter:
        t = start_temp * (end_temp / start_temp) ** (it / max_iter)
        # generate perturbation: add Gaussian noise scaled by t*scale_factor
        # scale_factor maybe 0.1 or 0.5?
        sigma = t * 0.5   # step size proportionate to temperature
        cand = cur + np.random.randn(n) * sigma
        # ensure non-negative (optional) and project to simplex
        cand = _project_to_simplex(cand)
        cand_score = fast_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / t):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        it += 1
    return best
```

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------
1.  For several lengths (4096, 2048, 1024) generate many analytic seeds:
    * arcsine law (several epsilon regularisations)
    * power‑law family
    * Beta family (random α,β,ε)
    * uniform / mixtures.
    All seeds are normalised to sum‑to‑1.

2.  Keep the few seeds with the smallest cheap surrogate score
    ``fast_score = 2·n·max(conv(a,a))/sum(a)²`` (computed with FFT).

3.  Refine every kept seed with a *smooth‑max* Adam optimiser.
    The smooth objective is ``log‑sum‑exp(β·conv(a,a))`` whose
    temperature β is increased from 5 to 2000 (continuation).  The
    iterate is kept on the probability simplex via a soft‑max mapping.

4.  After each Adam run keep the best‑scoring candidate (fast_score) and,
    if it improves the global best, store it.

5.  With the remaining time run a **simulated‑annealing** phase that
    repeatedly perturbs the current global best with Gaussian noise,
    re‑projects onto the simplex and accepts worse candidates with a
    temperature schedule.  The cheap surrogate ``fast_score`` guides the
    walk.

6.  Finally polish the best sequence with a projected sub‑gradient descent
    that uses the exact sub‑gradient of the true max‑convolution.

The whole routine respects the given time budget (default 1000 s) and
always returns a valid non‑empty list.  If nothing better than the fallback
uniform distribution is found, the uniform sequence is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate used for pre‑selection (scale‑invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/sum(a)² (no clipping, cheap)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic generators for diverse starting shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int, eps: float = 0.25) -> np.ndarray:
    """Discretised arcsine density on {0,…,n‑1} with regulariser eps."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """1 / ((i+½)(n-½-i)+eps)^p, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    vals = np.maximum(vals, 0.0)
    tot = vals.sum()
    if tot <= 0.0 or not np.isfinite(tot):
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family (i+eps)^{α‑1} (n‑1‑i+eps)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    base = np.maximum(base, 0.0)
    tot = base.sum()
    if tot <= 0.0 or not np.isfinite(tot):
        return np.full(n, 1.0 / n)
    return base / tot


def generate_mixture_shape(n: int, weight_arc: float, eps_arc: float = 0.25) -> np.ndarray:
    """Mixture arcsine/uniform."""
    arc = arcsine_initial(n, eps=eps_arc)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (continuation on β)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 2000.0,
    ramp_iters: int = 80_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimise the smooth approximation
          L_β(a) = log‑sum‑exp(β·conv(a,a))
    while keeping a on the simplex via the soft‑max map.
    β is increased linearly from β0 to β_max (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional status
        if t % 5_000 == 0:
            print(f"    Adam it={t:,d}  β={beta:.1f}  fast≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected to simplex)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))/sum(a)²."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    Sub‑grad it={it:,d}  fast≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Simulated annealing on the fast surrogate
# ----------------------------------------------------------------------


def anneal_sequence(
    seq: np.ndarray,
    deadline: float,
    start_temp: float = 0.1,
    end_temp: float = 1e-5,
    max_iter: int = 200_000,
) -> np.ndarray:
    """Simple annealing that perturbs seq and keeps the best fast_score."""
    cur = _project_to_simplex(seq.copy())
    cur_score = fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    n = cur.size
    it = 0
    while time.time() < deadline and it < max_iter:
        # temperature schedule (geometric)
        frac = it / max_iter
        T = start_temp * (end_temp / start_temp) ** frac

        # Gaussian perturbation whose std is proportional to temperature
        sigma = T * 0.5
        cand = cur + np.random.randn(n) * sigma
        cand = _project_to_simplex(cand)

        cand_score = fast_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        it += 1

        if it % 10_000 == 0:
            print(f"    Anneal it={it:,d}  T={T:.2e}  best fast≈{best_score:.6f}")

    return best

# Main entry point required by the harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine mixes analytic seeds, Adam continuation, simulated annealing,
    and a final projected sub‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq: List[float] = []
    best_fast = float("inf")   # fast surrogate value

    # ------------------------------------------------------------------
    # 0.  If a previously discovered sequence is available, use it as a
    #     starting point.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            cand_arr = np.array(cand, dtype=float)
            cand_arr = _project_to_simplex(cand_arr)  # ensure normalised
            val_fast = fast_score(cand_arr)
            print(f"Fallback (height_sequence_1) fast≈{val_fast:.6f}")
            if val_fast < best_fast:
                best_fast, best_seq = val_fast, cand_arr.tolist()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Scan a few lengths (larger first) and collect promising seeds.
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---- 1a. generate a rich pool of analytic seeds -----------------
        seed_vectors: List[np.ndarray] = []

        # deterministic families
        for eps in [1e-6, 1e-4, 0.001, 0.01, 0.1, 0.25, 0.5]:
            seed_vectors.append(arcsine_initial(n, eps=eps))
        seed_vectors.append(np.full(n, 1.0 / n))  # uniform

        # power‑law sweep
        for p in np.linspace(0.2, 3.0, 14):
            seed_vectors.append(generate_power_shape(n, p, eps=1e-8))

        # random Beta draws
        for _ in range(30):

            beta = rng.uniform(0.15, 5.0)

            seed_vectors.append(generate_beta_shape(n, alpha, beta, eps))

        # mixture arcsine / uniform
        for w in np.linspace(0.2, 0.8, 4):

            seed_deadline = time.time() + per_seed_budget

            # initialise Adam via log‑softmax (avoids division by zero)
            z0 = np.log(seed_vec + 1e-12)

            # ----- 2. Smooth‑max Adam optimisation ---------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=2000.0,
                ramp_iters=80_000,
                lr=0.04,
            )

            # ----- 3. Evaluate candidate with the cheap surrogate ----------
            val_fast = fast_score(a_opt)

            if val_fast < best_fast:

                print(f"   *** New global best (fast) = {best_fast:.6f} ***")

        # End of length loop
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:

        cur_arr = np.array(best_seq, dtype=float)
        refined = anneal_sequence(cur_arr, anneal_deadline,
                                  start_temp=0.2, end_temp=1e-5, max_iter=300_000)
        refined_fast = fast_score(refined)
        print(f"   After annealing fast≈{refined_fast:.6f}")
        if refined_fast < best_fast:

            print(f"   *** New global best after annealing = {best_fast:.6f} ***")

    # ------------------------------------------------------------------
    # 3. Final polish with projected sub‑gradient descent.
    # ------------------------------------------------------------------

    return best_seq
