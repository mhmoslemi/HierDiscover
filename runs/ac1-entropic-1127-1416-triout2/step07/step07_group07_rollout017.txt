"""
Propose a step‑function (non‑negative vector) that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation focuses on three ideas that helped push the bound
well below the previous 1.512… value:

* **Large scale optimisation** – we try lengths up to 65536.  The
  theoretical  O(log n / n) decay of the optimal objective becomes
  noticeable already for n≈2⁴–2⁶, so a larger n gives a stronger
  baseline.

* **Multi‑scale warm‑starts** – a solution found for a smaller n is
  linearly interpolated (and forced onto the simplex) before it is
  refined at the next larger scale.  This provides a good initial shape
  without having to start from scratch each time.

* **Three optimisation phases** –  
  1. *Adam* on a smooth soft‑max approximation of the max‑convolution,  
  2. exact *sub‑gradient* descent on the true max‑convolution, and  
  3. a lightweight *hill‑climber* that only accepts strictly improving
     moves.  

  All updates keep the vector on the probability simplex and enforce
  exact symmetry (aᵢ = aₙ₋₁₋ᵢ), which is known to reduce the peak of the
  self‑convolution.

The routine respects the supplied time budget (≈ 1000 s) and always
returns a valid sequence (a uniform fallback is provided for safety).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities – FFT‑based convolution and simplex handling
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (a and b are 1‑D arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of the public `evaluate_sequence` when sum(a) == 1.
    Returns 2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def normalize_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project onto the probability simplex by simple non‑negative clipping
    followed by renormalisation (O(n) and cheap).
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce exact mirror symmetry: (v + rev(v)) / 2."""
    v = (v + v[::-1]) * 0.5
    return normalize_simplex(v)


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int, eps: float = 1e-3) -> np.ndarray:
    """
    Discrete arcsine‑like density: a_i ∝ 1 / √((i+eps)·(n‑1‑i+eps)).
    eps > 0 prevents a huge spike at the borders.
    The result is forced onto the simplex and symmetrised.
    """
    i = np.arange(n, dtype=float)
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = normalize_simplex(w)
    return symmetrize(a)


def random_sparse_vector(n: int, density: float = 0.2) -> np.ndarray:
    """
    Random probability vector with a modest number of non‑zero entries.
    """
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return symmetrize(a)


def upsample_vector(old: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation of an older vector to a longer length.
    Afterwards the result is projected back onto the simplex and symmetrised.
    """
    old_len = old.shape[0]
    if old_len == new_len:
        return old.copy()

    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    a = np.interp(x_new, x_old, old)
    a = normalize_simplex(a)
    return symmetrize(a)


# ----------------------------------------------------------------------
# Gradient of the smooth soft‑max surrogate
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f_eps(a) = eps·log Σ_k exp(conv_k / eps) .
    Returns 2·(rev(w) ⊛ a), where
        w_i = softmax(conv_i / eps) and conv = a⋆a .
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max (log‑sum‑exp trick)
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)               # Σ w = 1

    # cross‑correlation of w_rev with a
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)

    # Extract the part that aligns with the original indices
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 10_000,
    step0: float = 0.4,
    eps_factor: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate.
    The iterate is kept on the simplex and forced symmetric at every step.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # current convolution (used for both gradient and score)
        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature – never smaller than 1 % of the current max
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        # Adam update (biased estimates)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)                         # mild decay
        a = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = normalize_simplex(a)
        a = symmetrize(a)

        cur_sc = 2.0 * n * float(np.max(conv))
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional tiny jitter to keep the dynamics alive
        if t % 800 == 0:
            jitter = np.random.randn(n) * (step0 * 0.001 / np.sqrt(t))
            a += jitter
            a = normalize_simplex(a)
            a = symmetrize(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 2 – Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on 2·n·max(a⋆a).
    The sub‑gradient at the current maximising shift k is
        g_i = 2·a_{k-i} (zero outside the support).
    The iterate is re‑projected onto the simplex and symmetrised.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build the sub‑gradient
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = normalize_simplex(a)
        a = symmetrize(a)

        cur_sc = 2.0 * n * float(np.max(conv))
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # gentle random shake every few thousand iterations
        if t % 3000 == 0:
            a += np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = normalize_simplex(a)
            a = symmetrize(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – Simple hill‑climber (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber that only accepts strictly improving moves.
    Moves are either a swap of two entries or a transfer of a fraction
    of the mass from one coordinate to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_frac = 0.1  # controls the size of a transfer move

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_frac * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = normalize_simplex(cand)
        cand = symmetrize(cand)

        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.08, 0.5)   # become bolder
        else:
            step_frac *= 0.96                       # become more cautious

        max_steps -= 1

    return best_a, best_sc

                best_seq = a_sub.tolist()
                current_best = best_seq
                print(f"[n={n}] after sub‑grad  score={sc_sub:.6f}")

        # ------------------------------------------------------------------
        # 1.C Hill‑climbing polishing (optional, cheap)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 8.0:
            hc_time = remaining * 0.20
            hc_deadline = time.time() + hc_time
            # number of candidate evaluations – very roughly 3000 per second
            max_steps_hc = max(10_000, int(hc_time * 3_000))

            a_hc, sc_hc = hill_climb(
                np.array(best_seq, dtype=float),
                time_deadline=hc_deadline,
                max_steps=max_steps_hc,
            )
            if sc_hc < best_score:
                best_score = sc_hc
                best_seq = a_hc.tolist()
                current_best = best_seq
                print(f"[n={n}] hill‑climb improved  score={sc_hc:.6f}")

            # symmetry averaging – cheap and often beneficial
            a_sym = symmetrize(np.array(best_seq, dtype=float))
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = a_sym.tolist()
                current_best = best_seq
                print(f"[n={n}] symmetry avg improved  score={sc_sym:.6f}")

    # ------------------------------------------------------------------
    # 2. Final fallback – uniform vector (guaranteed finite result)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 3. Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
