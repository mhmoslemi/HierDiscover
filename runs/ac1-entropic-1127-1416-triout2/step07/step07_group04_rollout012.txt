"""
Improved step‑function optimizer
--------------------------------

The search consists of three stages:

1. **Shape seeding** – a small set of analytically‑motivated families
   (arcsine with several ε‑smoothings, and mixtures with the uniform
   distribution) is generated on a target grid (default 4096 points).

2. **Fast sub‑gradient descent** – for each seed we run a short
   projected sub‑gradient descent that directly minimises
   `max(conv(a,a))`.  The sub‑gradient of `max_k (a*a)_k` at the
   currently maximal index `k*` is simply a *shift* of the vector `a`,
   which avoids a second FFT per iteration and therefore allows many
   more iterations within the 1000 s budget.

3. **Long refinement** – the best seed is refined with a long run of
   the same sub‑gradient method (hundreds of thousands of iterations)
   followed by an exponentiated‑gradient (mirror‑descent) polish.
   All steps respect a global wall‑clock deadline.

The algorithm works entirely with NumPy, uses only the supplied
`evaluate_sequence` for the final official score, and returns a
non‑negative list whose sum is 1 (the evaluation is scale‑invariant).
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact official objective for a vector a that already satisfies Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (seeds)
# ----------------------------------------------------------------------


def arcsine_eps_shape(n: int, eps: float) -> np.ndarray:
    """
    Discrete arcsine‑type density with a smoothing parameter eps.
    Normalised to Σ = 1.
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals_sum = vals.sum()
    if vals_sum == 0 or not np.isfinite(vals_sum):
        return np.full(n, 1.0 / n)
    return vals / vals_sum


def mixture_arcsine_uniform(n: int, weight_arc: float, eps: float = 0.25) -> np.ndarray:
    """
    Convex combination weight_arc·arcsine_eps + (1-weight_arc)·uniform.
    """
    arc = arcsine_eps_shape(n, eps)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Fast sub‑gradient optimisation (one‑FFT per iteration)
# ----------------------------------------------------------------------


def subgradient_descent_fast(a_init: np.ndarray,
                             deadline: float,
                             max_iter: int = 200_000,
                             step0: float = 0.07) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective
    `max(conv(a,a))`.  The sub‑gradient at the current maximal index
    k* is simply a shifted copy of a, which yields an O(n) gradient
    computation and only ONE FFT per iteration.
    """
    a = project_to_simplex(a_init.copy())
    n = a.size
    center = n - 1                     # index of zero shift in the convolution
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_star = int(np.argmax(conv))           # index where the max is attained
        offset = k_star - center                # offset ∈ [-(n‑1), …, n‑1]

        # gradient = 2 * shifted a (zero‑padded)
        grad = np.zeros_like(a)
        if offset >= 0:
            grad[offset:] = 2.0 * a[: n - offset]
        else:
            grad[: n + offset] = 2.0 * a[-offset:]

        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = project_to_simplex(a)

        # occasional progress report
        if it % 100_000 == 0:
            print(f"    Subgrad it={it:,d}  score={compute_score(a):.6f}")

    return a


def eg_refine_fast(a_init: np.ndarray,
                   deadline: float,
                   max_iter: int = 150_000,
                   eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent) polish.
    Uses the same single‑FFT shift‑gradient as above.
    """
    a = a_init.copy()
    n = a.size
    center = n - 1
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_star = int(np.argmax(conv))
        offset = k_star - center

        subgrad = np.zeros_like(a)
        if offset >= 0:
            subgrad[offset:] = 2.0 * a[: n - offset]
        else:
            subgrad[: n + offset] = 2.0 * a[-offset:]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a[:] = 1.0 / n
        else:
            a /= total

        if it % 100_000 == 0:
            print(f"    EG it={it:,d}  score={compute_score(a):.6f}")

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine follows the three‑stage scheme described at the top of
    the file.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback using a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback sequence official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 1 – generate a few analytic seeds and run a short sub‑gradient sweep
    # ------------------------------------------------------------------
    target_n = 4096
    # If we are already short on time, fall back to a smaller grid.
    if deadline - time.time() < 180.0:
        target_n = 2048
    print(f"\n--- Stage 1: seed generation on n={target_n} ---")

    # Parameter grid (kept deliberately small to respect the time budget)
    epsilons = [1e-4, 1e-3, 1e-2, 0.05, 0.25, 0.5]
    arc_weights = [0.0, 0.25, 0.5, 0.75, 1.0]   # 0 → uniform, 1 → pure arcsine

    seeds: List[np.ndarray] = []

    # Pure arcsine with different smoothings
    for eps in epsilons:
        seeds.append(arcsine_eps_shape(target_n, eps))

    # Mixtures of arcsine and uniform
    for w in arc_weights:
        seeds.append(mixture_arcsine_uniform(target_n, w, eps=0.25))

    # Add a random Dirichlet start (moderately sparse) for robustness
    random_dirichlet = rng.dirichlet(np.ones(target_n) * 0.5)
    seeds.append(random_dirichlet)

    # Keep only unique seeds (tiny deduplication)
    uniq_seeds = []
    for s in seeds:
        if not any(np.allclose(s, u) for u in uniq_seeds):
            uniq_seeds.append(s)

    # Run a very cheap sub‑gradient pass (≈20 k iterations) on each seed
    short_iters = 20_000
    for idx, seed_vec in enumerate(uniq_seeds, 1):
        if time.time() >= deadline - 200.0:   # leave about three minutes for the final polish
            break
        print(f"  Seed {idx}/{len(uniq_seeds)} – short refinement")
        a_opt = subgradient_descent_fast(seed_vec, deadline,
                                         max_iter=min(short_iters,
                                                      int((deadline - time.time()) * 30)),
                                         step0=0.12)
        score = compute_score(a_opt)
        print(f"    → short score = {score:.6f}")
        if score < best_val:
            best_val = score
            best_seq = a_opt.tolist()
            print("*** New best after short pass ***")

    # ------------------------------------------------------------------
    # Stage 2 – long refinement of the current best seed
    # ------------------------------------------------------------------
    if not best_seq:
        # safety net – uniform distribution
        best_seq = [1.0 / target_n] * target_n
        best_val = compute_score(np.array(best_seq))

    a_best = np.array(best_seq, dtype=float)

    print("\n--- Stage 2: long projected sub‑gradient descent ---")
    remaining = deadline - time.time()
    # Allocate the bulk of remaining time to this phase
    max_iters_long = int(min(800_000, remaining * 40))
    print(f"Running sub‑gradient for up to {max_iters_long:,d} iterations")
    a_best = subgradient_descent_fast(a_best, deadline,
                                      max_iter=max_iters_long,
                                      step0=0.07)

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\n--- Stage 3: exponentiated‑gradient polish ---")
        remaining = deadline - time.time()
        max_iters_eg = int(min(400_000, remaining * 40))
        a_best = eg_refine_fast(a_best, deadline,
                                 max_iter=max_iters_eg,
                                 eta0=0.25)

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = a_best.tolist()
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"\nFinal official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
    except Exception as exc:
        print(f"Evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
