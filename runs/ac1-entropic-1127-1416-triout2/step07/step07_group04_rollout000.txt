"""
Improved step‑function optimizer
================================

The optimizer follows a multi‑stage scheme that adds a few new ideas
to the previous best implementation:

1. **Diverse warm‑starts** – several analytically‑motivated shapes
   (arcsine, beta‑U‑shapes, arcsine‑uniform mixtures, a triangular “tent”)
   are generated.  A short Adam run on each shape quickly ranks them
   using a fast surrogate `2·n·max(conv(a,a))`.  The best‑scoring shape
   becomes the true starting point.

2. **Hierarchical refinement** – the optimisation proceeds in three
   resolutions: 1024 → 2048 → (up to) 8192 points.  Each up‑sample
   uses linear interpolation and the result is renormalised to the
   simplex.

3. **Two‑phase Adam** – first a moderate‑temperature schedule
   (τ≈0.5…50) is used to shape the distribution, then a high‑temperature
   schedule (τ≈0.5…300) drives the smooth‑max towards the true max.
   Adam works on the log‑softmax (logits) so the simplex constraints
   are satisfied automatically.

4. **Exact polishing** – projected Euclidean sub‑gradient descent
   on the true max‑convolution, followed by KL‑mirror‑descent
   (exponentiated‑gradient).  Both use a √(iteration) step‑size decay.

5. **Random perturbation sweep** – after polishing a cheap Monte‑Carlo
   local search perturbs the logits, keeps the best surrogate improvement,
   and finishes with another short polish.

All phases respect a global wall‑clock budget (default 1000 s) and
guarantee that a non‑empty sequence is returned.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    return np.fft.irfft(C, n=n_fft)[:m]


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 1e-8
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-8) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform, normalised."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def tent_shape(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape, normalised."""
    i = np.arange(n, dtype=float)
    vals = np.minimum(i + 1, n - i)
    return vals / vals.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points.
    The result is re‑normalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max approximation
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    tau_start: float = 0.5,
    tau_end: float = 200.0,
    base_lr: float = 0.5,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate.
    The optimisation variable lives in the log‑softmax space
    (logits w) which automatically respects non‑negativity and
    Σa = 1 after the softmax transform.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # initialise logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)          # numeric stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)     # length L

        # smooth‑max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp   # shape (L,)

        # gradient wrt a: 2 * conv(a, rev(weights))
        rev_w = weights[::-1]
        # convolution of a with rev_w gives length n+L-1
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wrt logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (nonsmooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.06,
) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.22,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        # renormalise (cannot be zero because exponentials are positive)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Small random‑perturbation sweep (cheap surrogate optimisation)
# ----------------------------------------------------------------------


def random_perturbation_sweep(
    best_a: np.ndarray,
    deadline: float,
    attempts: int = 2000,
    sigma: float = 0.02,
) -> np.ndarray:
    """
    Perform a Monte‑Carlo local search on the logits.
    Uses the fast surrogate score to accept improvements.
    """
    rng = np.random.default_rng()
    # work in log‑softmax space for easy perturbations
    logits = np.log(best_a + 1e-12)
    best_score = compute_score(best_a)
    best_logits = logits.copy()

    for _ in range(attempts):
        if time.time() >= deadline:
            break
        trial = best_logits + sigma * rng.normal(size=logits.shape)
        # softmax
        shift = trial - np.max(trial)
        a_trial = np.exp(shift)
        a_trial /= a_trial.sum()
        score = compute_score(a_trial)
        if score < best_score:
            best_score = score
            best_logits = trial
    # return the best found vector
    shift = best_logits - np.max(best_logits)
    a_best = np.exp(shift)
    a_best /= a_best.sum()
    return a_best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The algorithm proceeds through three resolutions,
    explores several analytic warm‑starts, runs a two‑phase Adam optimiser,
    applies exact polishing, performs a cheap random‑perturbation sweep,
    and finally returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Phase 0 – optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate quickly (surrogate) and print if it beats best
    # ------------------------------------------------------------------
    def maybe_update(candidate: np.ndarray, desc: str):
        nonlocal best_seq, best_val
        surrogate = compute_score(candidate)
        print(f"    {desc} surrogate = {surrogate:.6f}")
        # compute official only if we think it's promising
        # (only at very end we must call evaluate_sequence)
        if surrogate < best_val:  # keep best based on surrogate for now
            best_seq = candidate.tolist()
            best_val = surrogate

    # ------------------------------------------------------------------
    # Stage 1 – coarse optimisation at n=1024 (multiple warm‑starts)
    # ------------------------------------------------------------------
    n_coarse = 1024
    print(f"\n--- Stage 1: coarse grid (n={n_coarse}) ---")
    # generate a small portfolio of analytic initial shapes
    init_shapes = [
        arcsine_initial(n_coarse),
        generate_beta_shape(n_coarse, 0.35, 0.35, 1e-6),
        generate_beta_shape(n_coarse, 0.45, 0.45, 1e-6),
        generate_mixture_shape(n_coarse, 0.3),
        generate_mixture_shape(n_coarse, 0.7),
        tent_shape(n_coarse),
    ]

    # allocate a short Adam budget for ranking the shapes
    remaining = deadline - time.time()
    rank_iters = int(min(40_000, max(10_000, remaining * 8)))
    print(f"    Ranking {len(init_shapes)} shapes with {rank_iters} Adam iters each")
    scores = []
    for idx, shape in enumerate(init_shapes):
        if time.time() >= deadline:
            break
        a_tmp = adam_optimize_smooth_max(
            shape,
            deadline,
            max_iters=rank_iters,
            tau_start=0.5,
            tau_end=30.0,
            base_lr=0.4,
        )
        surrogate = compute_score(a_tmp)
        scores.append((surrogate, a_tmp))
        print(f"    Shape {idx+1}/{len(init_shapes)} surrogate = {surrogate:.6f}")

    # pick best based on surrogate
    scores.sort(key=lambda x: x[0])
    a_coarse = scores[0][1]
    maybe_update(a_coarse, "Coarse best after ranking")

    # ------------------------------------------------------------------
    # Stage 2 – longer Adam on the coarse grid (refine)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    max_iters_coarse = int(min(200_000, remaining * 30))
    print(f"\n--- Stage 2a: refine coarse grid (n={n_coarse}), {max_iters_coarse} iters ---")
    a_coarse = adam_optimize_smooth_max(
        a_coarse,
        deadline,
        max_iters=max_iters_coarse,
        tau_start=0.5,
        tau_end=50.0,
        base_lr=0.45,
    )
    maybe_update(a_coarse, "Coarse refined")

    # ------------------------------------------------------------------
    # Stage 3 – hierarchical up‑sampling to higher resolution
    # ------------------------------------------------------------------
    # Target resolution: try 8192 if time permits, otherwise 4096
    remaining = deadline - time.time()
    if remaining > 350.0:
        target_n = 8192
    elif remaining > 200.0:
        target_n = 4096
    else:
        target_n = 2048
    print(f"\n--- Stage 2b: upsample to n={target_n} ---")
    a_high = upsample(a_coarse, target_n)

    # ------------------------------------------------------------------
    # Stage 4 – long Adam on high‑resolution grid with high temperature
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    max_iters_high = int(min(400_000, remaining * 30))
    print(f"\n--- Stage 3: high‑res Adam (n={target_n}), {max_iters_high} iters ---")
    a_high = adam_optimize_smooth_max(
        a_high,
        deadline,
        max_iters=max_iters_high,
        tau_start=0.5,
        tau_end=300.0,
        base_lr=0.55,
    )
    maybe_update(a_high, "High‑res after Adam")

    # ------------------------------------------------------------------
    # Stage 5 – exact polishing (sub‑gradient + exponentiated‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    sub_iters = int(min(350_000, remaining * 30))
    print(f"\n--- Stage 4: Euclidean sub‑gradient descent ({sub_iters} iters) ---")
    a_refined = subgradient_descent_exact(
        a_high,
        deadline,
        max_iter=sub_iters,
        step0=0.05,
    )
    maybe_update(a_refined, "After sub‑gradient")

    remaining = deadline - time.time()
    eg_iters = int(min(250_000, remaining * 30))
    print(f"\n--- Stage 5: Exponentiated‑gradient polish ({eg_iters} iters) ---")
    a_refined = eg_refine(
        a_refined,
        deadline,
        max_iter=eg_iters,
        eta0=0.20,
    )
    maybe_update(a_refined, "After EG polish")

    # ------------------------------------------------------------------
    # Stage 6 – cheap random perturbation sweep (surrogate only)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        attempts = int(min(5000, remaining * 50))
        print(f"\n--- Stage 6: random perturbation sweep ({attempts} attempts) ---")
        a_refined = random_perturbation_sweep(
            a_refined,
            deadline,
            attempts=attempts,
            sigma=0.015,
        )
        maybe_update(a_refined, "After random sweep")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"\nOfficial score of final candidate = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** New global best (official) ***")
    except Exception as e:
        print(f"Official evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to uniform distribution of moderate length
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence official score = {best_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
