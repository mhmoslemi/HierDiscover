    rng = np.random.default_rng(seed)
    start = time.time()

    best_seq = None
    best_val = float('inf')
    lengths = [4096, 2048, 1024, 512, 256]  # order largest first

    # defined param for per-length num_seed and per-seed time

        if time.time() >= deadline - 5:
            break
        # generate seed vectors
        seeds = []
        # arcsine
        seeds.append(arcsine_initial(n))
        # uniform
        seeds.append(np.full(n, 1.0 / n))

        seeds.append(generate_mixture_shape(n, w))
        # power shape random exponent
        for _ in range(3):
            p = rng.uniform(0.1, 3.0)

        # beta shapes with random alphas,beta in log scale
        for _ in range(5):
            log_a = rng.uniform(-2, 2)  # exponent between -2 and 2 => 0.01-100

            a = 10**log_a

            seeds.append(generate_beta_shape(n, a, b, eps))
        # perturbed arcsine
        for _ in range(2):
            base = arcsine_initial(n)
            noise = rng.normal(scale=0.05, size=n)
            pert = base + noise
            pert = np.clip(pert, 0, None)
            if pert.sum() == 0:
                continue
            pert = pert / pert.sum()

        # evaluate fast_score for each and keep top few

def local_annealing_refine(a0, deadline, max_iter=200_000):
    a = a0.copy()
    n = a.size
    best_score = fast_score(a)
    best_a = a.copy()
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # random pair swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # determine epsilon to move: sample fraction of a[i] or small constant
        eps = min(a[i], 0.1) * rng.random() * 0.5   # move up to half of a[i] or smaller
        if eps <= 0:
            continue
        a[i] -= eps
        a[j] += eps
        # compute score
        sc = fast_score(a)
        # temperature schedule
        T = max(0.001, (deadline - time.time()) / (deadline - start))  # decreasing
        accept = False
        if sc < best_score:
            accept = True
        else:
            # acceptance probability
            if rng.random() < np.exp(-(sc - best_score) / (T + 1e-12)):
                accept = True
        if accept:
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            # revert move
            a[i] += eps
            a[j] -= eps
        # optionally print occasional progress
        if it % 5000 == 0:
            print(f"    Anneal it={it} best≈{best_score:.6f}")
    return best_a

"""
propose_candidate
-----------------
Search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

(the exact objective used by ``evaluate_sequence``).

The algorithm combines:
* a diverse pool of analytically‑motivated seeds (arcsine, beta,
  power‑law, mixtures, random perturbations);
* a long‑run continuation Adam optimiser on the smooth ‑log‑sum‑exp
  approximation of the max‑convolution, with a very high final β;
* a short mirror‑descent (exponential‑gradient) refinement;
* a stochastic “annealing” phase that randomly moves tiny mass
  between coordinates while accepting uphill moves with a decreasing
  temperature;
* a final projected sub‑gradient polish.

All steps respect the global 1000 s budget and use only the fast FFT‑based
convolution (`fast_score`) for the objective, which is exactly the same
as the official ``evaluate_sequence`` value (the latter only adds clipping
and a tiny sum‑lower‑bound check).  The returned list is guaranteed to be
non‑empty and contains only non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence for small values)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families for initial seeds
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Helper: projection onto probability simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize_v2(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 2000.0,
    ramp_iters: int = 120_000,
    lr: float = 0.025,
) -> np.ndarray:
    """
    Minimise the smooth approximation
        L_β(a) = log‑sum‑exp(β·conv(a,a))

    using Adam on the logits z (a = softmax(z)).
    β is increased linearly from beta0 to beta_max (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                   # a lies on the simplex
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights (log‑sum‑exp trick)
        shifted = conv - mx              # max‑shifted to avoid overflow
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)          # size = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]     # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if t % 5_000 == 0:
            print(
                f"    Adam it={t:,d} β={beta:.1f} score≈{fast_score(a):.6f}"
            )

    return _softmax(z)


# ----------------------------------------------------------------------
# Mirror descent (exponential‑gradient) with exact max‑subgradient
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.12,
    decay: float = 0.9995,
) -> np.ndarray:
    """
    Multiplicative (KL‑mirror) descent using the exact sub‑gradient of
    f(a)=2·n·max(conv(a,a))/sum(a)^2.
    The iterate stays on the simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient via convolution with reversed mask
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponential‑gradient update
        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay

        # very lightweight debug (printed only rarely)
        if int(step * 1e6) % 200_000 == 0:
            print(f"    Mirror step={step:.6f} score≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max, final polish)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    Sub‑grad it={it:,d} score≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Local stochastic annealing refinement
# ----------------------------------------------------------------------


def annealing_refine(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_moves: int = 200_000,
) -> np.ndarray:
    """
    Randomly moves a tiny amount of mass between two coordinates.
    Accepts worsening moves with a temperature that decays linearly
    to zero.  Uses the exact fast_score as the objective.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = fast_score(a)
    start = time.time()

    # temperature schedule: linear from 0.02 down to 0
    T0 = 0.02

    for it in range(1, max_moves + 1):
        if time.time() >= deadline:
            break

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # amount to relocate (fraction of a[i] up to a small cap)
        cap = min(a[i], 0.005)                # never move more than 0.5 % of mass
        if cap <= 0:
            continue
        eps = rng.random() * cap

        # propose move
        a[i] -= eps
        a[j] += eps

        sc = fast_score(a)

        # linearly decaying temperature
        T = T0 * max(0.0, (deadline - time.time()) / (deadline - start))

        accept = False
        if sc < best_score:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-(sc - best_score) / (T + 1e-12)):
                accept = True

        if accept:
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            # revert
            a[i] += eps
            a[j] -= eps

        if it % 10_000 == 0:
            print(f"    Anneal it={it:,d} best≈{best_score:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  Returns a list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Use a fallback if a previously discovered sequence is present
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = fast_score(np.array(cand, dtype=float))
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, np.array(cand, dtype=float)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Scan several lengths (larger first) with diverse seeds
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024, 512, 256]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---- generate a rich pool of seed vectors --------------------
        seed_vectors: list[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_mixture_shape(n, rng.random()),
        ]

        # random power‑law shapes
        for _ in range(3):
            p = rng.uniform(0.1, 3.0)
            seed_vectors.append(generate_power_shape(n, p, eps=1e-8))

        # random beta shapes – α,β drawn log‑uniform over [0.01, 100]
        for _ in range(5):
            log_a = rng.uniform(-2, 2)
            log_b = rng.uniform(-2, 2)
            a = 10 ** log_a
            b = 10 ** log_b
            eps = rng.uniform(0.0, 0.01 * n)
            seed_vectors.append(generate_beta_shape(n, a, b, eps))

        # perturbed arcsine (adds a little Gaussian noise)
        for _ in range(2):
            base = arcsine_initial(n)
            noise = rng.normal(scale=0.03, size=n)
            pert = base + noise
            pert = np.clip(pert, 0.0, None)
            if pert.sum() > 0:
                pert = pert / pert.sum()
                seed_vectors.append(pert)

        # ---- keep the most promising seeds according to fast_score ----
        scored = [(fast_score(vec), vec) for vec in seed_vectors]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:3]]

        # ---- optimise each top seed ----------------------------------
        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 2.0:
                break

            # allocate a per‑seed time slice (fair share of the remaining budget)
            remaining = deadline - time.time()
            per_seed_budget = max(2.0, remaining / (len(top_seeds) - idx + 1))
            seed_deadline = time.time() + per_seed_budget

            print(
                f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_budget:.1f}s"
            )

            # ---- continuation Adam (smooth max) --------------------
            z0 = np.log(seed_vec + 1e-12)
            a_opt = smooth_adam_optimize_v2(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=2000.0,
                ramp_iters=120_000,
                lr=0.025,
            )

            val = fast_score(a_opt)
            if val < best_val:
                best_val, best_seq = val, a_opt.copy()
                print(f"   New global best after Adam: {best_val:.6f}")

            # ---- short mirror‑descent refinement (if time left) ----
            if time.time() < seed_deadline - 0.5:
                md_a = mirror_descent_optimize(
                    a_opt, seed_deadline, step0=0.12, decay=0.9995
                )
                val_md = fast_score(md_a)
                if val_md < best_val:
                    best_val, best_seq = val_md, md_a.copy()
                    print(f"   New global best after mirror: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Global stochastic annealing refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nGlobal annealing refinement ...")
        best_seq = annealing_refine(best_seq, deadline, rng)

        val = fast_score(best_seq)
        if val < best_val:
            best_val = val
            print(f"   Best after annealing: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final polish with projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.2:
        print("\nFinal sub‑gradient polish ...")
        best_seq = subgradient_descent_exact(
            best_seq, deadline, max_iter=150_000, step0=0.05
        )
        val = fast_score(best_seq)
        if val < best_val:
            best_val = val
            print(f"   Best after final polish: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Safeguard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 512
        best_seq = np.full(n0, 1.0 / n0)
        best_val = fast_score(best_seq)

    print(f"\nSearch finished. Best reported score ≈ {best_val:.6f}")
    return best_seq.tolist()
