"""
Propose a step‐function (non‑negative sequence) that minimises the official
``evaluate_sequence`` objective.

Algorithm
---------
1. **Diversify** – generate a large set of candidate shapes for a few
   lengths (4096, 2048, 1024).  Families include arcsine, beta, power,
   exponential, Gaussian, triangular, sine and mixtures thereof.
   All candidates are normalised to a probability simplex
   (∑a_i = 1).

2. **Fast filtering** – rank the candidates with the cheap surrogate
   ``2·n·max(conv(a,a))`` (exact when the sum is 1).  Keep the best
   ~30 vectors.

3. **Quick Adam refinement** – each kept vector is refined for a short
   (≈5 s) window with Adam on a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The temperature β is linearly increased up to a
   high value (≈150) to gradually tighten the surrogate.

4. **Intensive polishing** – the current global best is repeatedly
   perturbed in log‑space and polished with exponentiated‑gradient
   (KL‑mirror) descent for a few seconds.  This updates the vector on
   the simplex and directly reduces the true non‑smooth objective.

5. **Final exact sub‑gradient pass** – a projected sub‑gradient descent
   on the original objective is run until the remaining time expires.
   This step finishes with a very precise solution.

6. **Fallback** – if nothing was produced, return a uniform vector.

All steps respect the 1000 s wall‑clock budget and guarantee a
non‑empty, non‑negative sequence.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact objective when the vector already sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Families of analytic shapes (all return a normalized vector)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre (lam∈(0,1))."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    d = np.abs(i - centre)
    vals = lam ** d
    return vals / vals.sum()


def generate_gaussian_shape(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, σ expressed as a fraction of n."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = max(sigma_frac * n, 1e-6)
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return vals / vals.sum()


def generate_triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular (linear rise then fall)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = centre - np.abs(i - centre) + 1.0
    return vals / vals.sum()


def generate_sine_shape(n: int, power: float = 1.0) -> np.ndarray:
    """Sine envelope sin(π·(i+0.5)/(n+1)) optionally raised to a power."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / (n + 1))
    if power != 1.0:
        vals = vals ** power
    return vals / vals.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Projection utilities for the simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for the final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) w.r.t. a
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 150.0,
    ramp_iters: int = 20_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Minimise a smooth log‑sum‑exp surrogate of max(conv) using Adam.
    Iterates are kept on the simplex via a soft‑max map.
    The temperature β climbs from ``beta0`` to ``beta_max``.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp weighting
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2 * (a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final vector on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The function respects the 1000 s wall‑clock budget and always returns
    a non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load a fallback from a previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Build a large pool of diverse seeds (three lengths)
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]
    raw_seeds = []   # list of (n, vector)

    for n in lengths:
        # deterministic bases
        raw_seeds.append((n, arcsine_initial(n)))
        raw_seeds.append((n, np.full(n, 1.0 / n)))
        raw_seeds.append((n, generate_triangular_shape(n)))
        raw_seeds.append((n, generate_sine_shape(n, power=0.8)))

        # random beta shapes
        for _ in range(6):
            a = rng.uniform(0.2, 4.0)
            b = rng.uniform(0.2, 4.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            raw_seeds.append((n, generate_beta_shape(n, a, b, eps)))

        # random power shapes
        for _ in range(6):
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            raw_seeds.append((n, generate_power_shape(n, p, eps)))

        # exponential family
        for _ in range(4):
            lam = rng.uniform(0.8, 0.99)
            raw_seeds.append((n, generate_exponential_shape(n, lam)))

        # Gaussian family
        for _ in range(4):
            sigma_frac = rng.uniform(0.08, 0.3)
            raw_seeds.append((n, generate_gaussian_shape(n, sigma_frac)))

        # mixture of arcsine and uniform
        for _ in range(3):
            w = rng.uniform(0.0, 1.0)
            raw_seeds.append((n, generate_mixture_shape(n, w)))

        # incorporate current global best if it exists
        if best_seq:
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()
            raw_seeds.append((n, cand))

    # ------------------------------------------------------------------
    # 2.  Fast surrogate ranking (keep the most promising seeds)
    # ------------------------------------------------------------------
    scored = []
    for n, vec in raw_seeds:
        v = np.maximum(vec, 1e-15)
        v = v / v.sum()
        s = compute_score(v)          # exact objective when sum=1
        scored.append((s, n, v))

    scored.sort(key=lambda x: x[0])
    top_k = 30                       # keep at most this many for quick Adam
    top_candidates = scored[:top_k]

    # ------------------------------------------------------------------
    # 3.  Quick Adam refinement on the surrogate (≈5 s per seed)
    # ------------------------------------------------------------------
    short_time = 5.0
    for idx, (approx, n, seed_vec) in enumerate(top_candidates, start=1):
        if time.time() >= deadline - 30.0:      # reserve time for polishing
            break
        print(f"\n[Quick] Seed {idx}/{len(top_candidates)} – length {n}")
        seed_deadline = min(time.time() + short_time, deadline - 5.0)

        # initialise Adam in log‑softmax coordinates
        eps = 1e-12
        z0 = np.log(seed_vec + eps)

        a_opt = smooth_adam_optimize(
            z0,
            seed_deadline,
            n=n,
            beta0=5.0,
            beta_max=150.0,
            ramp_iters=20_000,
            lr=0.07,
        )

        try:
            val = evaluate_sequence(a_opt.tolist())
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"   ** New best (quick Adam): {best_val:.6f} **")
        except Exception as e:
            print(f"   Evaluation error: {e}")

    # ------------------------------------------------------------------
    # 4.  Intensive polishing on the current best (stochastic perturbations)
    # ------------------------------------------------------------------
    polish_rounds = 5
    while polish_rounds > 0 and time.time() < deadline - 10.0:
        print(f"\n[Polish] Round {polish_rounds} – perturb‑and‑EG")
        a0 = np.array(best_seq, dtype=float)
        # perturb in log‑space
        loga = np.log(np.maximum(a0, 1e-15))
        noise = rng.normal(scale=0.2, size=loga.shape)
        perturbed = np.exp(loga + noise)
        perturbed = perturbed / perturbed.sum()

        # exponentiated‑gradient polishing (few seconds)
        eg_deadline = min(time.time() + 20.0, deadline - 3.0)
        a_polished = exponentiated_gradient_polish(
            perturbed,
            eg_deadline,
            max_iter=100_000,
            step=0.018,
        )
        try:
            val2 = evaluate_sequence(a_polished.tolist())
            if val2 < best_val:
                best_val, best_seq = val2, a_polished.tolist()
                print(f"   ** Improved after EG polish: {best_val:.6f} **")
        except Exception as e:
            print(f"   EG polish evaluation error: {e}")

        polish_rounds -= 1

    # ------------------------------------------------------------------
    # 5.  Final exact sub‑gradient polishing (uses remaining budget)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n[Final] Exact sub‑gradient polishing")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=200_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print(f"   ** Final improvement: {best_val:.6f} **")
        except Exception as e:
            print(f"   Final polishing evaluation error: {e}")

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
