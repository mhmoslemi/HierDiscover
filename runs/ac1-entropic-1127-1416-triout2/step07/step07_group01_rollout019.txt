def generate_candidate(n):
    typ = np.random.choice(['exp','pow','mixexp'])
    if typ == 'exp':
        # epsilon random between 0.5/n and 4/n
        c_factor = np.random.uniform(0.5, 5.0)  # c = epsilon*n
        epsilon = c_factor / n
        r = 1.0 - epsilon
        # generate a_i = r**i
        a = np.power(r, np.arange(n))
    elif typ == 'pow':
        p = np.random.uniform(0.1, 4.0)
        a = 1.0 / (np.arange(1, n+1) ** p)
    elif typ == 'mixexp':
        # mixture of two exponentials
        c1 = np.random.uniform(0.5, 5.0)
        c2 = np.random.uniform(0.5, 5.0)
        eps1 = c1 / n
        eps2 = c2 / n
        r1 = 1.0 - eps1
        r2 = 1.0 - eps2
        w = np.random.rand()
        a = w * (r1 ** np.arange(n)) + (1-w) * (r2 ** np.arange(n))
    else:
        a = np.random.rand(n)
    # normalize to sum 1
    a_sum = np.sum(a)
    a = a / a_sum
    return a.tolist()

def alternating_lp(a, time_deadline, max_iters=30, t=0.2):
    best_a = a.copy()
    best_score = compute_score(a)
    for it in range(max_iters):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        M = conv.max()
        b = solve_lp_for_b(a, M)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b
        a_new = (1 - t) * a + t * g
        # Ensure nonnegative and sum 1 (project to simplex)
        a_new = project_to_simplex(a_new)   # after mixing sum stays 1
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score
```

"""
Non‑negative step‑function optimisation.

The search consists of three stages.

1. **Parametric random probes** – fast generation of candidate vectors
   from simple families (exponential, power‑law, mixtures).  The best
   candidate is kept as a seed.

2. **Alternating linear‑programming refinement** – given a current
   vector `a` (sum = 1) we compute its worst convolution value `M`.
   Keeping `M` fixed we solve a linear programme that maximises the
   total mass of a new vector `b` while guaranteeing `a * b ≤ M`
   (component‑wise convolution).  The normalised solution `g` is mixed
   with `a`.  Repeating this step drives the maximal autocorrelation
   down.

3. **Smooth Adam optimisation + sub‑gradient polishing + hill‑climb**
   (identical to the previous best implementation) finishes the search.

The routine respects the 1000 s wall‑clock limit and always returns a
plain Python list of non‑negative floats.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# --------------------------------------------------------------
# Basic linear‑algebra utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = rfft(a, n=n_fft)
    B = rfft(b, n=n_fft)
    C = A * B
    conv = irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# 1.  Fast parametric random candidates
# --------------------------------------------------------------

def random_parametric_candidate(n: int) -> np.ndarray:
    """Create a random vector from a simple parametric family."""
    typ = np.random.choice(["exp", "pow", "mixexp"])
    if typ == "exp":
        # a_i ∝ (1‑ε)^i , ε ≈ c/n with c∈[0.5,4]
        c = np.random.uniform(0.5, 4.0)
        eps = c / n
        r = 1.0 - eps
        a = r ** np.arange(n, dtype=float)
    elif typ == "pow":
        # a_i ∝ (i+1)^(-p) , p∈[0.1,4]
        p = np.random.uniform(0.1, 4.0)
        a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
    else:   # "mixexp"
        # mixture of two exponentials
        c1 = np.random.uniform(0.5, 4.0)
        c2 = np.random.uniform(0.5, 4.0)
        eps1 = c1 / n
        eps2 = c2 / n
        r1 = 1.0 - eps1
        r2 = 1.0 - eps2
        w = np.random.rand()
        a = w * (r1 ** np.arange(n, dtype=float)) + (1.0 - w) * (r2 ** np.arange(n, dtype=float))
    # normalise to a probability vector
    a = a / a.sum()
    return a


# --------------------------------------------------------------
# 2.  Alternating LP refinement (requires scipy)
# --------------------------------------------------------------

# The LP part is optional – if SciPy is not available we simply skip it.
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    SCIPY_AVAILABLE = True
except Exception:                 # pragma: no cover
    SCIPY_AVAILABLE = False


def build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Return a sparse matrix A (shape (2n‑1, n)) such that for any vector b
    we have (A @ b)[k] = Σ_i a_i * b_{k‑i}  (linear convolution).
    """
    n = a.shape[0]
    m = 2 * n - 1
    rows = []
    cols = []
    data = []
    for i, ai in enumerate(a):
        start = i
        end = i + n
        rows.extend(range(start, end))
        cols.extend([i] * n)
        data.extend([ai] * n)
    A = csr_matrix((data, (rows, cols)), shape=(m, n))
    return A


def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t.   a * b ≤ M (component‑wise) ,   b ≥ 0.
    Returns the optimal b (or None on failure).
    """
    if not SCIPY_AVAILABLE:                     # pragma: no cover
        return None
    n = a.shape[0]
    A = build_conv_matrix(a)                    # (2n‑1) × n
    m = A.shape[0]
    c = -np.ones(n)                             # maximise Σb  ⇒ minimise -Σb
    b_ub = np.full(m, M)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=(0, None), method="highs")
    if not res.success:
        return None
    return res.x


def alternating_lp_optimize(start_a: np.ndarray,
                            time_deadline: float,
                            max_outer_iters: int = 25,
                            mix_factor: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Alternating scheme described in the problem statement.
    Returns the best vector encountered and its exact score.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for outer in range(max_outer_iters):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        M = float(conv.max())
        b = solve_lp_max_sum(a, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b == 0.0:
            break
        g = b / sum_b                               # now Σg = 1
        # convex combination (keeps the simplex)
        a = (1.0 - mix_factor) * a + mix_factor * g
        # numerical renormalisation
        a = a / a.sum()
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # slowly reduce the mixing factor
        mix_factor = mix_factor * 0.95

    return best_a, best_score


# --------------------------------------------------------------
# 3.  Smooth log‑sum‑exp optimiser (Adam)
# --------------------------------------------------------------

def smooth_gradient_optimize(start_a: np.ndarray,
                            time_deadline: float,
                            max_iters: int = 300_000,
                            beta_start: float = 5.0,
                            beta_end: float = 80.0,
                            step0: float = 0.8,
                            adam_beta1: float = 0.9,
                            adam_beta2: float = 0.999,
                            adam_eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam‑style minimisation of a log‑sum‑exp surrogate of max(conv)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smooth max
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smooth objective
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)               # length 3n‑2
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# --------------------------------------------------------------
# 4.  Sub‑gradient descent on the exact max‑objective
# --------------------------------------------------------------

def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 400_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient method on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# --------------------------------------------------------------
# 5.  Final hill‑climb polishing
# --------------------------------------------------------------

def hill_climb(seq,
               time_deadline,
               max_steps: int = 500_000) -> tuple[list[float], float]:
    """Small stochastic mass‑transfer moves on the simplex."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
