"""
Hybrid search that adds a linear‑programming “bound‑tightening” step.
The LP maximises the ℓ₁‑mass of a candidate g under the current
convolution bound  M = max((f*f)).  By mixing the LP‑solution back into
the current distribution we strictly reduce the bound.  After a few
LP iterations we polish the result with the mirror‑descent and exact
projected sub‑gradient methods from the previous state‑of‑the‑art.

The algorithm:
  1. Generate a few diverse seeds (arcsine, mixture, random Dirichlet).
  2. For each seed run a short LP‑loop:
        – compute M = max(conv(f,f))
        – solve   maximise   𝟙ᵀg   s.t.   f*g ≤ M   ,   g ≥ 0
        – normalise g to a probability vector and mix:
                f ← (1‑α)·f + α·g       (α≈0.3)
  3. Refine the best candidate with mirror‑descent and finally a
     projected sub‑gradient descent.
All work respects the overall 1000 s budget and always returns a
valid non‑negative sequence.
"""

import time
import numpy as np
import cvxpy as cp
from typing import List

# --------------------------------------------------------------
# Convolution utilities (FFT, fast surrogate)
# --------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """Cheap surrogate: 2·n·max(conv(a,a))/sum(a)² (no clipping)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# --------------------------------------------------------------
# Analytic seed generators (non‑negative, sum = 1)
# --------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """U‑shaped arcsine density (large at the ends, small in the centre)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_mixture_shape(n: int, weight_arc: float = 0.6) -> np.ndarray:
    """Convex mixture of the arcsine shape with the uniform distribution."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# --------------------------------------------------------------
# Projection onto the simplex (used by the optimiser)
# --------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# --------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent on
       f(a)=2·n·max(conv(a,a))/sum(a)² (exact max used)."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d}  score≈{fast_score(a):.6f}")

    return a


# --------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) with exact max‑subgradient
# --------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.12,
    decay: float = 0.9995,
) -> np.ndarray:
    """KL‑mirror descent (multiplicative updates) for the exact objective."""
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        if int(step * 1e6) % 100_000 == 0:
            print(f"    Mirror step={step:.6f}  score≈{fast_score(a):.6f}")

    return a


# --------------------------------------------------------------
# Linear‑programming improvement step
# --------------------------------------------------------------


def lp_improvement_step(f: np.ndarray, max_iter: int, deadline: float) -> np.ndarray:
    """
    Perform ``max_iter`` LP‑based improvement steps starting from f.
    Each step:
      – compute M = max(conv(f,f))
      – solve maximise Σ g_j  s.t. (f*g)_k ≤ M  ∀k , g ≥ 0
      – normalise g to a probability vector and mix: f ← (1‑α)f + αg
    Returns the improved f.
    """
    n = f.shape[0]
    a = _project_to_simplex(f)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current bound
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # ----- build the convolution matrix (dense, shape (2n‑1, n)) -----
        #   (f * g)_k = Σ_i f_i * g_{k‑i}
        A = np.zeros((2 * n - 1, n), dtype=float)
        for i, val in enumerate(a):
            A[i : i + n, i] = val  # each column i is shifted by i rows

        # ----- solve the LP ------------------------------------------------
        g = cp.Variable(n)
        constraints = [A @ g <= M + 1e-12, g >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except Exception as e:
            print(f"    LP solve failed (iteration {it}): {e}")
            break

        if g.value is None:
            print(f"    LP returned no solution at iteration {it}")
            break

        g_vec = np.maximum(g.value, 0.0)
        if np.sum(g_vec) <= 1e-12:
            break

        # ----- mix the new candidate ---------------------------------------
        g_norm = g_vec / np.sum(g_vec)         # back to a probability vector
        alpha = 0.30                           # mixing strength (can be tuned)
        a = (1.0 - alpha) * a + alpha * g_norm
        a = _project_to_simplex(a)             # safety projection

        if (it + 1) % 2 == 0:
            print(f"    LP it={it+1:3d}  score≈{fast_score(a):.6f}")

    return a
