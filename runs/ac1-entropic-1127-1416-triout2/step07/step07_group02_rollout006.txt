def propose_candidate(seed=… ,budget_s=1000):
   start= time.time()
   deadline = start + budget_s - 1.0
   rng = np.random.default_rng(seed)
   best_seq = None
   best_val = np.inf

   # optionally use previous best from global variable

   # generate seeds across lengths
   all_candidates = []  # list of (fast_score, seq, n)
   lengths = [4096, 2048, 1024, 512]
   for n in lengths:
       seed_vecs = []
       # Add arcsine and uniform
       seed_vecs.append(arcsine_initial(n))
       seed_vecs.append(np.full(n, 1/n))
       # Add mixtures
       for w in [0.25,0.5,0.75]:
           seed_vecs.append(generate_mixture_shape(n, w))
       # Beta family grid
       alphas = np.linspace(0.1, 0.9, 9)
       betas = np.linspace(0.1, 0.9, 9)
       for a in alphas:
          for b in betas:
               seed_vecs.append(generate_beta_shape(n, a, b, eps=1e-6))
       # Power family grid
       for p in np.linspace(0.2, 2.5, 11):
               seed_vecs.append(generate_power_shape(n, p, eps=1e-8))
       # Random seeds
       for _ in range(8):
           a = rng.uniform(0.1, 1.5)
           b = rng.uniform(0.1, 1.5)
           eps = rng.uniform(0,0.001*n)
           seed_vecs.append(generate_beta_shape(n, a, b, eps))
       # Evaluate fast_score
       for vec in seed_vecs:
            sc = fast_score(vec)
            all_candidates.append((sc, vec, n))
   # Keep top candidates
   all_candidates.sort(key=lambda x: x[0])
   top_k = 10  # keep e.g., 10 best overall
   top_candidates = all_candidates[:top_k]

   # iterate each candidate optimizing
   for idx, (init_score, vec, n) in enumerate(top_candidates, start=1):
       if time.time() >= deadline - 5: break
       print(f"\nCandidate {idx}/{len(top_candidates)} length {n} initial fast_score {init_score:.6f}")
       # allocate budget: more for earlier candidates
       remaining = deadline - time.time()
       # allocate proportionally: e.g., 0.6/(idx) maybe
       # Simpler: allocate up to 0.2*remaining for first candidate, decreasing later
       candidate_deadline = time.time() + remaining * 0.2
       # Run optimizer
       a_opt = optimize_sequence(vec, candidate_deadline, n)
       # Evaluate official
       val = evaluate_sequence(a_opt.tolist())
       print(f"   Official score: {val:.6f}")
       if val < best_val:
           best_val, best_seq = val, a_opt.tolist()
           print(f"   New best global: {best_val:.6f}")

   # final local perturbation loop
   while time.time() < deadline - 1.0:
       # perturb best_seq
       arr = np.array(best_seq, dtype=float)
       eps = 0.01 / arr.size # small epsilon
       noise = rng.normal(scale=eps, size=arr.shape)
       arr = arr + noise
       arr = np.clip(arr, 0, None) # nonnegative
       arr = arr / arr.sum()
       # optional short mirror descent
       md_end = time.time() + 0.5
       arr = mirror_descent_optimize(arr, md_end, step0=0.08, decay=0.999)
       val = evaluate_sequence(arr.tolist())
       if val < best_val:
           best_val, best_seq = val, arr.tolist()
           print(f"   Perturb improved to {best_val:.6f}")

   # final polishing
   if best_seq is not None and time.time() < deadline - 0.5:
       final_deadline = deadline
       a_arr = np.array(best_seq, dtype=float)
       a_arr = subgradient_descent_exact(a_arr, final_deadline, max_iter=500_000, step0=0.05)
       val = evaluate_sequence(a_arr.tolist())
       print(f"   Final polished score {val:.6f}")
       if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print("*** Global best improved by final polish ***")

   # guarantee non-empty
   if best_seq is None:
        n0=512
        best_seq = [1.0/n0]*n0
        best_val = evaluate_sequence(best_seq)
   print(f"Search finished. Best score {best_val:.6f}")
   return best_seq

"""
Propose a low‑value step sequence for the provided evaluate_sequence().

Algorithm Overview
------------------
1. **Coarse Parameter Scan (small size)** –   Search a dense grid of
   Beta‑family parameters (α,β) and Power‑family exponents on a modest
   length (n=512).  The cheap surrogate 2·n·max(conv)/sum² is used to keep
   the most promising shapes.

2. **Seed Generation for Larger Lengths** –   The best (α,β) and power
   exponents from the scan are transferred to the target lengths
   (4096, 2048, 1024, 512).  A few analytically‑motivated families
   (arcsine, uniform, mixtures) are also added.

3. **Continuation Adam on a Smooth‑max Surrogate** –   Each seed is
   optimised with Adam while the smooth‑max temperature β is increased
   from 5 to 500.  The iterate lives on the probability simplex via a
   soft‑max parametrisation.

4. **Mirror‑Descent with Exact Sub‑gradient** –   The Adam result is
   refined by exponential‑gradient descent using the true max‑convolution
   sub‑gradient.

5. **Projected Sub‑gradient Polishing** –   A final long run of exact
   sub‑gradient descent (Euclidean projection onto the simplex) gives
   the polish.

6. **Local Random Perturbations** –   If time remains we jitter the best
   sequence, re‑project and run a short mirror step, accepting any
   improvement.

All phases respect the global 1000 s budget.  The routine always returns
a valid non‑negative sequence; a uniform fallback is provided for safety.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate used for quick ranking (scale‑invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/sum(a)² (no clipping, cheap)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families for seed generation
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected) – final polish
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv)/sum²."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    [SubGrad] it={it:,d}  surrogate={fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam (continuation on β)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns probabilities)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 500.0,
    ramp_iters: int = 80_000,
    lr: float = 0.03,
) -> np.ndarray:
    """
    Minimise Lβ(a)=log‑sum‑exp(β·conv(a,a)) while staying on the simplex via soft‑max.
    β is increased linearly from beta0 to beta_max.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    [Adam] it={t:,d} β={beta:.1f} surrogate={fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) with exact max sub‑gradient
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.12,
    decay: float = 0.9995,
) -> np.ndarray:
    """
    Exponential‑gradient descent on the true objective.
    The iterate stays on the simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        if int(step * 1e6) % 100_000 == 0:
            print(f"    [Mirror] step={step:.6f} surrogate={fast_score(a):.6f}")

    return a

def optimise_candidate(init_seq: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """
    Optimise a candidate sequence inside the time budget `deadline`.
    Returns the (approximately) best sequence found for this seed.
    """
    now = time.time()
    if now >= deadline:
        return init_seq

    # --- Stage 1: smooth‑max Adam -------------------------------------------------
    stage1_deadline = min(deadline, now + (deadline - now) * 0.5)

        z0,

        n=n,
        beta0=5.0,
        beta_max=500.0,

        lr=0.03,
    )

    # --- Stage 2: mirror‑descent -------------------------------------------------
    now = time.time()
    if now < deadline:
        stage2_deadline = min(deadline, now + (deadline - now) * 0.5)

            stage2_deadline,

            a_mirror,
            deadline,

        )
    else:

            val = evaluate_sequence(arr.tolist())
            if val < best_val:
                best_val, best_seq = val, arr.tolist()
                print(f"   Perturbation improvement: {best_val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Final global polishing with projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\n=== Final polishing with exact sub‑gradient descent ===")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(a_arr, deadline, max_iter=400_000, step0=0.05)
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 7. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
