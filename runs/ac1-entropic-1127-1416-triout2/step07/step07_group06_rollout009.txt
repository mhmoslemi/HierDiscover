def lp_improve(a, M, time_deadline):
    # Build sparse constraint matrix A_ub (shape (2n-1, n))
    n = len(a)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows from i to i+n-1 inclusive
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
        data.extend([a[i]] * n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Right-hand side = M
    b_ub = np.full(2*n-1, M)
    # bounds: (0, None) for each variable
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    b = res.x
    # Scale to sum=1
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    b_scaled = b / sum_b
    return b_scaled

#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Multi‑phase optimiser for the step‑function evaluation problem.

The algorithm combines several complementary ideas:

1. **Random warm‑starts** on the probability simplex.
2. **Variance flattening** of the self‑convolution (Adam on an L2 loss).
3. **Smooth max (log‑sum‑exp) surrogate** minimisation (Adam with a
   decreasing temperature ε).
4. **Exact sub‑gradient descent** on the true max‑convolution objective.
5. **Linear‑programming refinement** – maximise the ℓ₁‑mass of a
   direction `b` while keeping the current convolution bound `M`
   (`a⊛b ≤ M`).  The resulting `b` is blended into the current
   iterate, which empirically lowers the peak of the autocorrelation.
6. **Stochastic hill‑climbing with simulated annealing** for fine‑grained escape
   from local minima.
7. **Symmetry averaging** (`(a+rev(a))/2`) as a cheap post‑process that often
   reduces the max‑convolution further.

All intermediate vectors are projected onto the probability simplex
(guaranteeing non‑negativity and unit‑sum).  The search respects a
1000 s time budget, repeatedly restarts with different lengths
(≤ 4096) and keeps the best sequence found.  The final score is
reported through the provided `evaluate_sequence` function.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# FFT‑based utilities (fast linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, sum(x)=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random simplex initialisation
# ----------------------------------------------------------------------


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with α=1)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Phase 2 – flattening (L2 variance of the convolution)
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 3000,
    step0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the variance of a⊛a.
    The quartic loss is  Σ (conv_k - μ)².
    Returns the best vector and its exact score.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)                     # length 2n‑1
        mu = np.mean(conv)                               # = sum(a)² / (2n‑1) = 1/(2n‑1)
        diff = conv - mu

        # gradient = 4 * Σ_j a_j * diff_{i+j}
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – smooth max (ε·log∑exp) optimiser
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 8000,
    step0: float = 0.5,
    eps_factor: float = 0.25,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth surrogate
        f(a) = ε·log∑_k exp(conv_k/ε)
    The temperature ε is decreased slowly (never below 1 % of max conv).
    Returns the best vector and its exact score.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature schedule
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        exp_vals = np.exp((conv - maxc) / eps)   # stable soft‑max
        w = exp_vals / np.sum(exp_vals)         # ∑ w = 1

        # gradient: g_i = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – exact sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 80000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for f(a)=2·n·max(a⊛a).
    Uses the classic sub‑gradient at the shift attaining the maximum.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k-i} (zero where out of range)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional random shake
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 5 – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with simulated annealing.
    Moves: swap two entries or transfer a small fraction of mass.
    Always accepts improving moves; otherwise accepts with probability
    exp(-Δ/T) where T cools linearly.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12  # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]          # swap
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]                 # move 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional large random perturbation
        if step % 5000 == 0:
            rnd = random_dense_vector(n) * 0.02
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# Phase 6 – symmetry averaging (a+rev(a))/2
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Return the symmetric version (a+reverse(a))/2, re‑projected to the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Phase 7 – LP‑based refinement
# ----------------------------------------------------------------------


def lp_improve(a: np.ndarray, M: float, time_deadline: float) -> np.ndarray | None:
    """
    Solve the linear program
        maximise   sum(b)
        subject to a ⊛ b ≤ M   (element‑wise)
                   b ≥ 0
    (only columns with a_i > 0 are kept to avoid unboundedness).
    The returned vector is rescaled to have unit ℓ₁‑norm.
    Returns None on failure or if time runs out.
    """
    if time.time() >= time_deadline:
        return None

    n = len(a)
    eps = 1e-12
    idx = np.where(a > eps)[0]
    if idx.size == 0:
        return None

    a_pos = a[idx]
    var_n = idx.size

    # Build sparse constraint matrix A (shape (2n‑1, var_n))
    rows = np.empty(var_n * n, dtype=np.int32)
    cols = np.empty(var_n * n, dtype=np.int32)
    data = np.empty(var_n * n, dtype=np.float64)
    pos = 0
    for col_id, (col_idx, a_val) in enumerate(zip(idx, a_pos)):
        rows[pos:pos + n] = np.arange(col_idx, col_idx + n, dtype=np.int32)
        cols[pos:pos + n] = col_id
        data[pos:pos + n] = a_val
        pos += n

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, var_n))

    # Linear program: maximise sum(b)  ⇔ minimise -∑b
    c = -np.ones(var_n, dtype=np.float64)          # objective
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # RHS of a⊛b ≤ M
    bounds = [(0, None)] * var_n

    # Solve using HiGHS (fast interior‑point / simplex)
    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": max(0.0, time_deadline - time.time())},
    )

    if not res.success or res.x is None:
        return None

    b = res.x
    s = b.sum()
    if s <= 0:
        return None
    b_scaled = b / s
    # embed back into full‑size vector (zeros where a was zero)
    full_b = np.zeros_like(a)
    full_b[idx] = b_scaled
    return full_b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a
    plain Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final `evaluate_sequence` call
    deadline = start_time + budget_s - 3.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # optional fallback to a known construction (if the global exists)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # search parameters
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]   # keep FFTs cheap while allowing variety
    outer_iter = 0

    while time.time() < deadline - 5.0:
        outer_iter += 1
        n = int(np.random.choice(lengths))

        # ----- 1. initialise a candidate
        if best_seq is not None and np.random.rand() < 0.2:
            # reuse a truncated/extended version of the incumbent
            a0 = np.array(best_seq[:n], dtype=float)
            if a0.shape[0] < n:
                a0 = np.pad(a0, (0, n - a0.shape[0]), "constant")
            a0 = project_to_simplex(a0)
        else:
            a0 = random_dense_vector(n)

        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[{outer_iter}] init   n={n}  score={best_score:.6f}")

        # ----- 2. flattening (L2 variance) phase
        if time.time() < deadline - 6.0:
            a_flat, sc_flat = flatten_conv_adam(
                a0,
                time_deadline=deadline - 3.0,
                max_iter=3000,
                step0=0.15,
            )
            if sc_flat < best_score:
                best_score = sc_flat
                best_seq = a_flat.tolist()
                print(f"[{outer_iter}] flatten n={n}  score={best_score:.6f}")

        # ----- 3. smooth‑max surrogate phase (Adam)
        if time.time() < deadline - 8.0:
            a_smax, sc_smax = smooth_max_adam(
                a_flat,
                time_deadline=deadline - 5.0,
                max_iter=8000,
                step0=0.55,
                eps_factor=0.30,
            )
            if sc_smax < best_score:
                best_score = sc_smax
                best_seq = a_smax.tolist()
                print(f"[{outer_iter}] smooth‑max n={n}  score={best_score:.6f}")

        # ----- 4. exact sub‑gradient descent
        if time.time() < deadline - 12.0:
            a_sub, sc_sub = subgrad_optimize(
                a_smax,
                time_deadline=deadline - 8.0,
                max_iter=100000,
                step0=0.06,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{outer_iter}] sub‑grad n={n}  score={best_score:.6f}")
        else:
            a_sub = a_smax

        # ----- 5. LP‑based refinement (few iterations)
        if time.time() < deadline - 12.0:
            a_lp = a_sub.copy()
            for lp_iter in range(3):
                M_val = np.max(convolution_fft(a_lp, a_lp))
                b_dir = lp_improve(a_lp, M_val, deadline - 4.0)
                if b_dir is None:
                    break
                # blend a little towards the LP direction
                blend = 0.20
                a_lp = (1 - blend) * a_lp + blend * b_dir
                a_lp = project_to_simplex(a_lp)
                sc_lp = compute_score(a_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.tolist()
                    print(f"[{outer_iter}] LP‑improve iter {lp_iter+1}  score={best_score:.6f}")

        else:
            a_lp = a_sub

        # ----- 6. stochastic hill‑climb (simulated annealing)
        if time.time() < deadline - 8.0:
            seq_to_polish = best_seq if best_seq is not None else a_lp.tolist()
            polished_seq, polished_sc = hill_climb_anneal(
                seq_to_polish,
                time_deadline=deadline - 4.0,
                max_steps=250_000,
            )
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished_seq
                print(f"[{outer_iter}] anneal improved  score={best_score:.6f}")

        # ----- 7. symmetry averaging (cheap and often beneficial)
        if best_seq is not None:
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer_iter}] symmetry averaged  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final safety net – if nothing succeeded, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # final call to the official evaluator (clamps, etc.)
    final_score = evaluate_sequence(best_seq)
    print(
        f"\nSearch finished after {outer_iter} outer loops.\n"
        f"Final reported score = {final_score:.6f}"
    )
    return best_seq
