"""
Optimiser for the “step‑function” challenge (v2).

Strategy
--------
1️⃣  Load a fallback solution (if any) and keep it as a baseline.
2️⃣  Initialise a modest pool (n=1024) with a few analytic shapes and a
    large random sample of parametric families (Beta, power‑law, mixtures,
    asymmetric power).  Keep the five best seeds (according to the fast
    surrogate 2·n·max(conv)).
3️⃣  For each seed run an Adam optimiser on the smooth‑max surrogate
    Φ_τ(a) = (1/τ)·log Σ_k exp(τ·(a∗a)_k).  The temperature τ is
    increased exponentially from 0.5 to a large final value (≈400‑500)
    so the surrogate turns into the true max.  The parameters are the
    logits of a soft‑max, guaranteeing a≥0 and Σa=1.
4️⃣  Upsample the current champion to larger sizes (2048 → 4096 → 8192
    when time permits).  After each up‑sampling a short Adam run is
    performed, followed by an exponentiated‑gradient (EG) polish that
    directly minimises the true max‑convolution.
5️⃣  A final exact sub‑gradient descent (Euclidean projection onto the
    simplex) and a second EG polishing are applied to the best large
    candidate.
6️⃣  The officially evaluated score is tracked throughout; the best
    ever sequence is returned.

All stages respect the global 1000 s budget, and the code prints a
few progress messages.  The implementation is deliberately vectorised
and uses FFT‑based convolutions for speed.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (fast for the symmetric case).
    Returns the full linear convolution of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_surrogate_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector a is assumed to be normalised (Σa = 1), hence
    score = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_asymmetric_power(
    n: int, p: float, eps_l: float, eps_r: float
) -> np.ndarray:
    """
    Asymmetric generalised arcsine:
    a_i ∝ 1 / ((i+ε_l)*(n‑½‑i+ε_r))**p .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + eps_l) * (n - 0.5 - i + eps_r) + 1e-12) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(v) to target_n,
    followed by re‑normalisation to Σ=1.
    """
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up /= up.sum()
    return up

def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 60_000,
    tau_start: float = 0.5,
    tau_end: float = 500.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k)
    The temperature τ is increased exponentially from ``tau_start`` to
    ``tau_end``.  The optimisation is performed in the log‑softmax space
    (logits w) so that a = softmax(w) automatically satisfies a≥0,
    Σa=1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from a_init
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax to obtain a (keeps Σa = 1)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a (full length)
        conv = convolution_fft(a, a)

        # soft‑max weights over τ·conv
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()          # shape (L,)

        # gradient w.r.t. a: 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (learning‑rate decays like 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (cheap surrogate)
        if it % 25_000 == 0:
            surrogate = compute_surrogate_score(a)
            print(
                f"    [Adam] it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}"
            )

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution
    objective.  The update is a = Π_simplex(a - η·g), where g is a
    sub‑gradient of max(conv).
    """
    a = project_to_simplex(a_init.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # tolerance to collect all indices attaining the max
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) restricted to original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 180_000,
    eta0: float = 0.22,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry)
    directly on the true max‑convolution objective.
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with minimal ``evaluate_sequence``.
    The routine runs a hierarchical stochastic‑gradient optimiser, upsamples
    the winner to larger sizes and finishes with exact sub‑gradient / EG
    polishing.  The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Load a fallback if one exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"[Fallback] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Initialise a seed pool at n=1024
    # ------------------------------------------------------------------
    n_small = 1024
    pool: List[Tuple[float, np.ndarray]] = []

    # deterministic starters
    uniform = np.full(n_small, 1.0 / n_small)
    pool.append((compute_surrogate_score(uniform), uniform.copy()))
    arc = arcsine_initial(n_small)
    pool.append((compute_surrogate_score(arc), arc.copy()))

    # random parametric seeds
    for _ in range(700):
        if time.time() >= deadline - 5.0:
            break
        typ = rng.integers(0, 5)
        if typ == 0:               # beta‑like
            a = rng.uniform(0.05, 0.8)
            b = rng.uniform(0.05, 0.8)
            eps = rng.uniform(0.0, 3.0)
            cand = generate_beta_shape(n_small, a, b, eps)
        elif typ == 1:             # power‑law
            p = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n_small, p, eps)
        elif typ == 2:             # arcsine‑/uniform mixture
            w = rng.random()
            cand = generate_mixture_shape(n_small, w)
        elif typ == 3:             # triangular bump
            peak = rng.integers(0, n_small)
            cand = generate_triangular_shape(n_small, peak)
        else:                      # asymmetric power
            p = rng.uniform(0.1, 2.5)
            eps_l = rng.uniform(0.0, 1.0)
            eps_r = rng.uniform(0.0, 1.0)
            cand = generate_asymmetric_power(n_small, p, eps_l, eps_r)

        pool.append((compute_surrogate_score(cand), cand))

    # keep a handful of the best seeds
    pool.sort(key=lambda x: x[0])
    top_seeds_small = [vec for _, vec in pool[:5]]
    print(f"[Seeding] best surrogate (n={n_small}) = {pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam on the top small seeds
    # ------------------------------------------------------------------
    refined_small: List[Tuple[float, np.ndarray]] = []
    for idx, seed_vec in enumerate(top_seeds_small, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n[Adam‑small] seed {idx}/{len(top_seeds_small)}")
        remaining = deadline - time.time()
        # allocate a generous number of iterations but respect time
        max_iters = int(min(80_000, remaining * 35))
        a_opt = adam_optimize_smooth_max(
            seed_vec,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=400.0,
            base_lr=0.25,
        )
        surrogate = compute_surrogate_score(a_opt)
        print(f"    surrogate after Adam = {surrogate:.6f}")
        refined_small.append((surrogate, a_opt))

    refined_small.sort(key=lambda x: x[0])
    best_small = refined_small[0][1]

    # ------------------------------------------------------------------
    # 3️⃣  Progressive up‑sampling and further Adam + EG
    # ------------------------------------------------------------------
    best_large_vec = best_small
    best_large_score = compute_surrogate_score(best_small)

    for target_n in (2048, 4096, 8192):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n[Upscale] n = {target_n}")

        up = upscale_vector(best_large_vec, target_n)

        # small random jitter to help escape plateaus
        jitter = rng.normal(scale=1e-3, size=target_n)
        up = np.clip(up + jitter, 0.0, None)
        up /= up.sum()

        remaining = deadline - time.time()
        max_iters = int(min(120_000, remaining * 30))
        a_opt = adam_optimize_smooth_max(
            up,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=500.0,
            base_lr=0.22,
        )
        surrogate = compute_surrogate_score(a_opt)
        print(f"    surrogate after Adam (n={target_n}) = {surrogate:.6f}")

        # EG polish (fast, cheap)
        remaining = deadline - time.time()
        max_iter_eg = int(min(100_000, remaining * 30))
        a_opt = eg_refine(
            a_opt,
            deadline,
            max_iter=max_iter_eg,
            eta0=0.20,
        )
        surrogate = compute_surrogate_score(a_opt)
        print(f"    surrogate after EG  (n={target_n}) = {surrogate:.6f}")

        # keep the best large candidate
        if surrogate < best_large_score:
            best_large_score = surrogate
            best_large_vec = a_opt

    # ------------------------------------------------------------------
    # 4️⃣  Exact sub‑gradient + final EG polish on the champion
    # ------------------------------------------------------------------
    if best_large_vec is not None:
        n_best = best_large_vec.shape[0]
        print(f"\n[Exact‑refine] starting n={n_best}")
        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 30))
        refined = subgradient_descent_exact(
            best_large_vec,
            deadline,
            max_iter=max_iter_sub,
            step0=0.09,
        )
        # final EG polish
        remaining = deadline - time.time()
        max_iter_eg = int(min(180_000, remaining * 30))
        refined = eg_refine(
            refined,
            deadline,
            max_iter=max_iter_eg,
            eta0=0.18,
        )

        try:
            val = evaluate_sequence(refined.tolist())
            print(f"[Official] score after exact+EG = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print("*** New global best ***")
        except Exception as e:
            print(f"Evaluation failed after polish: {e}")

    # ------------------------------------------------------------------
    # 5️⃣  Final global polishing if time permits
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n[Final‑polish] fine‑tuning the best vector")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(350_000, remaining * 30))
        a = subgradient_descent_exact(
            a, deadline, max_iter=max_iter_sub, step0=0.07
        )

        remaining = deadline - time.time()
        max_iter_eg = int(min(220_000, remaining * 30))
        a = eg_refine(a, deadline, max_iter=max_iter_eg, eta0=0.16)

        final_val = evaluate_sequence(a.tolist())
        print(f"[Final] official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6️⃣  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        print("[Guard] fallback to uniform distribution (n=1024)")
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\n[Done] best official score = {best_val:.6f}")
    return best_seq
