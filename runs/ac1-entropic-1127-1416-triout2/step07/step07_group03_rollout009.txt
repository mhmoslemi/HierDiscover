def adam_subgrad_optimize(a0, deadline, n, lr=0.05, beta1=0.9, beta2=0.999, epsilon=1e-12):
    eps = 1e-12
    z = np.log(a0 + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_a = a0.copy()
    best_score = compute_score(a0)
    # while time < deadline:
    while time.time() < deadline:
        t += 1
        a = softmax(z)
        mx, subgrad = max_conv_and_subgrad(a)
        # compute gradient wrt logits
        dot = np.dot(subgrad, a)
        grad_z = a * (subgrad - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        # track best
        if t % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:

"""
Propose a non‑negative step sequence that (approximately) minimizes

    evaluate_sequence(sequence)

The algorithm builds diverse seeds (arcsine, power‑law, beta‑like, Dirichlet,
mixtures) for a few lengths (1024, 2048, 4096).  Each seed is refined by a
sub‑gradient Adam optimiser that works directly on the true objective
``max(conv(a,a))`` (the smooth‐max surrogate is omitted – we use the exact
sub‑gradient of the max).  The optimisation is performed in the log‑softmax
parameterisation, which guarantees the simplex constraints automatically.
After the (fast) Adam phase a Nesterov‑style heavy‑ball sub‑gradient
polish is run for the remaining time.  The best sequence seen (according to
the official ``evaluate_sequence``) is returned; a fallback uniform sequence
is provided to guarantee a non‑empty answer.

The whole routine respects a 1000 s wall‑clock budget and prints progress
information.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT, zero‑padded, no wrap‑around)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (no circular wrap).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate for the objective when sum(a)=1 (2·n·max(conv))."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (max_conv, subgradient) of the max‑convolution objective
    evaluated at a (assumed to lie on the simplex).
    """
    n = a.size
    conv = convolution_fft(a, a)
    mx = float(np.max(conv))
    # tolerance to collect all maximal entries
    tol = mx * 1e-12 + 1e-14
    mask = (np.abs(conv - mx) <= tol).astype(float)

    mask_rev = mask[::-1]
    L = 2 * n - 1
    rev = convolution_fft(a, mask_rev)          # length = 3n‑2
    subgrad = 2.0 * rev[L - 1 - np.arange(n)]   # n‑vector
    return mx, subgrad


# ----------------------------------------------------------------------
# Shape generators for diverse initial seeds
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def dirichlet_seed(rng: np.random.Generator, n: int, concentration: float) -> np.ndarray:
    """Draw a Dirichlet vector (always sums to 1)."""
    if concentration <= 0.0:
        concentration = 1.0
    alpha = np.full(n, concentration)
    vec = rng.dirichlet(alpha)
    # avoid exact zeros which would cause log‑problems later
    eps = 1e-12
    vec = np.clip(vec, eps, None)
    vec /= vec.sum()
    return vec


# ----------------------------------------------------------------------
# Adam optimiser that uses the exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def adam_subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Adam optimisation on the true max‑convolution objective.
    Parameters are kept on the simplex via the soft‑max map.
    """
    n = a0.shape[0]
    # initialise logits
    log_eps = 1e-12
    z = np.log(a0 + log_eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_a = a0.copy()
    best_score = compute_score(a0)

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥ 0 , Σa = 1
        mx, subgrad = max_conv_and_subgrad(a)

        # chain rule through soft‑max
        dot = np.dot(subgrad, a)
        grad_z = a * (subgrad - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if t % 2000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"    Adam it={t:,d} best surrogate={best_score:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Nesterov‑style heavy‑ball sub‑gradient polishing (projected on simplex)
# ----------------------------------------------------------------------


def nesterov_subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.15,
    mu: float = 0.9,
    max_iter: int = 1_000_000,
) -> np.ndarray:
    """
    Heavy‑ball (Nesterov‑style) sub‑gradient descent on the original
    max‑convolution objective, with Euclidean projection onto the simplex.
    """
    a = project_to_simplex(a0.copy())
    v = np.zeros_like(a)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        mx, subgrad = max_conv_and_subgrad(a)
        step = step0 / np.sqrt(it)
        v = mu * v - step * subgrad
        a = a + v
        a = project_to_simplex(a)

        if it % 3000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"    Nesterov it={it:,d} best surrogate={best_score:.6f}")

    return best_a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The function respects the 1000 s wall‑clock budget and always returns
    a non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback from a previous run if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Optimise a few lengths (small → medium → large)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 15.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---------- 1a. build a rich pool of seeds ----------
        seeds: List[np.ndarray] = []

        # deterministic seeds
        seeds.append(arcsine_initial(n))
        seeds.append(np.full(n, 1.0 / n))

        # random power‑law seeds
        for _ in range(4):
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # random beta‑like seeds
        for _ in range(4):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # mixture of arcsine & uniform
        for _ in range(2):
            w = rng.uniform(0.0, 1.0)
            seeds.append(generate_mixture_shape(n, w))

        # Dirichlet draws (different concentration parameters)
        for _ in range(5):
            conc = rng.uniform(0.1, 2.0)
            seeds.append(dirichlet_seed(rng, n, conc))

        # include previous best (scaled / tiled) if we have one
        if best_seq:
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            cand = np.clip(cand, 1e-15, None)
            cand /= cand.sum()
            seeds.append(cand)

        # ---------- 1b. keep only the most promising seeds ----------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]   # keep at most five

        # ---------- 1c. allocate time per seed ----------
        remaining = deadline - time.time()
        per_seed_time = max(5.0, remaining / (len(top_seeds) * 1.5))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 10.0:
                break
            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_time:.1f}s")
            seed_deadline = min(time.time() + per_seed_time, deadline - 5.0)

            # ----------- 2. Adam optimisation on true max‑conv -----------
            a_adam = adam_subgrad_optimize(
                a0=seed_vec,
                deadline=seed_deadline,
                lr=0.07,
                beta1=0.9,
                beta2=0.999,
                eps=1e-12,
            )

            # ----------- 3. Official evaluation ----------
            try:
                val = evaluate_sequence(a_adam.tolist())
                print(f"   Official score after Adam = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_adam.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ----------- 4. Optional mirror‑descent polishing -----------
            if time.time() < deadline - 5.0:
                # give a small extra budget for a quick Nesterov polish
                polish_deadline = min(deadline - 2.0, time.time() + 2.5)
                a_polish = nesterov_subgrad_optimize(
                    a0=np.array(best_seq, dtype=float),
                    deadline=polish_deadline,
                    step0=0.12,
                    mu=0.9,
                )
                try:
                    val2 = evaluate_sequence(a_polish.tolist())
                    print(f"   Polished (Nesterov) score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_polish.tolist()
                        print(f"   *** New best after polish: {best_val:.6f} ***")
                except Exception as e:
                    print(f"   Polish failed: {e}")

    # ------------------------------------------------------------------
    # 5. Final global polishing (heavy‑ball) with remaining time
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global heavy‑ball polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = nesterov_subgrad_optimize(
            a0=a_arr,
            deadline=deadline,
            step0=0.14,
            mu=0.9,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
