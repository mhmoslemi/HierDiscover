    a = start_a.copy()
    n = a.shape[0]
    # track best as before
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute softmax weights
        # compute shift to avoid overflow
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()
        # gradient via cross-correlation
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n-1:n-1+n]

        # Determine step size
        eta = eta0 / np.sqrt(it)  # decreasing
        # exponentiated gradient update: a_i *= exp(-eta * grad_i)
        # Clip grad to prevent overflow
        # Using np.exp ensures positivity even if grad large negative.
        # But large positive grad may cause underflow -> set floor to some min.
        # Compute log update: log a <- log a - eta * grad
        log_a = np.log(np.maximum(a, eps))
        log_a -= eta * grad
        a = np.exp(log_a)
        # renormalize to sum 1
        sum_a = a.sum()
        if sum_a == 0:
            # all zero due to underflow, reinitialize to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a /= sum_a

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    # final evaluate
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

"""
Algorithm:
- Choose sequence length n based on remaining time.
- Generate an analytical start from a family a_i ∝ [(i+0.5)(n-0.5-i)]^{-α}
  scanning α over a modest grid and keep the best.
- Perform a series of random sparse restarts, each optimized by
  exponentiated‑gradient descent on the smooth log‑sum‑exp approximation
  of the max‑convolution.  The smoothing parameter β is annealed
  exponentially up to a large value, while the step size follows a
  1/√t schedule.
- Refine the current best with a longer run of the same optimiser,
  then a classic projected sub‑gradient descent on the true max‑objective,
  and finally a stochastic hill‑climb that moves small packets of mass.
All phases respect the global time budget and keep the best sequence seen.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]

def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a⊛a) for a normalized to sum=1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Family a_i ∝ [(i+½)(n‑½‑i)]^{‑α} on the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # replace possible infinities
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a

def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a coarse grid of α values."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 11), np.linspace(0.0, 2.5, 26)])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.4)   # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    # force a to be on the simplex (already is)
    return a

def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 400_000,
                        beta_start: float = 0.5,
                        beta_end: float = 300.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed objective
        f_β(a) = (1/β) log Σ_k exp(β (a*a)_k).
    β is annealed exponentially from beta_start to beta_end.
    The update is a_i ← a_i·exp(‑η·g_i) followed by normalization.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stabilised exponentials
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smooth max
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # step size (1/√t)
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of (a*a)_k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)

        if t % 500 == 0:

            if sc < best_sc:

                best_a = a.copy()
    sc = compute_score(a)

        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

def hill_climb(start_seq: np.ndarray,
               deadline: float,

"""
Algorithm
---------

1. **Choose a sensible length** – longer vectors allow a lower theoretical
   bound, but they also cost more FFT work.  The routine picks 8192,
   4096 or 2048 depending on the remaining wall‑clock budget.

2. **Analytical one‑parameter start** – a family
   a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ is scanned on a relatively fine grid (α∈[‑1,2.5]).
   The best vector is kept.

3. **Exponentiated‑gradient smoothing** – minimise the smooth
   log‑sum‑exp approximation of the max‑convolution

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k).

   The gradient of f_β is computed by a cross‑correlation (two FFTs per
   iteration).  Updates are performed with a KL‑mirror step
   `a ← a·exp(-η·∇f_β)`, followed by renormalisation onto the simplex.
   The temperature β is annealed exponentially from a small value
   (≈0.5) to a large one (≈300).  The step size follows  η_t = η₀/√t.

   Several cheap random‑sparse restarts are tried, each limited to a
   modest number of iterations, to diversify the search.

4. **Long dedicated exponentiated‑gradient run** – starting from the best
   vector found so far, a much longer optimisation (≈3·10⁵ iterations)
   is performed with the same scheme but tighter parameters.

5. **Projected sub‑gradient refinement** – classic sub‑gradient descent
   on the true max‑objective `2·n·max(a*a)` with a Euclidean projection
   onto the simplex.

6. **Final stochastic hill‑climb** – a long random walk that swaps
   entries or transfers a small packet of mass between two positions.
   Only moves that strictly improve the exact score are kept.

All phases respect the global 1000 s budget; the best sequence found
throughout is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalized to sum = 1.
    This is the value returned by `evaluate_sequence` when the input
    already satisfies the sum‑normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytical start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a modest α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    # grid includes a few negative alphas (centre‑heavy) and many positive ones
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 11),
                             np.linspace(0.0, 2.5, 26)])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Random sparse initialisation
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a  # already on the simplex


# ----------------------------------------------------------------------
# 3. Exponentiated‑gradient optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 300.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective f_β using a KL‑mirror (exponentiated‑gradient)
    step.  β is exponentially annealed, η follows η₀/√t.
    Returns the best exact score seen during the run.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smooth weights (soft‑max) ----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))    # stabilised
        w = expw / expw.sum()

        # ---- gradient of f_β ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ---- step size ----
        eta = eta0 / np.sqrt(it)

        # ---- exponentiated‑gradient update (KL‑projection) ----
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ---- occasional exact evaluation ----
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for 2·n·max(a*a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Random‑walk polishing: with prob 0.5 swap two entries,
    otherwise transfer a fraction of mass from one index to another.
    Only moves that improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
