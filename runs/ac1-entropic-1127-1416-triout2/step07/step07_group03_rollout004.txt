def shape_from_params(p, n):
    # p: array of length K (log values)
    # produce symmetric shape.
    K = len(p)
    half_len = (n + 1) // 2  # includes center for odd n
    # positions for knots: equally spaced across half_len.
    # For example, knot positions as float indices: np.linspace(0, half_len-1, K)
    knot_pos = np.linspace(0, half_len - 1, K)
    # Convert p to positive: np.exp(p)
    knot_vals = np.exp(p)  # positive
    # Interpolate across half positions
    idx = np.arange(half_len)
    half_vals = np.interp(idx, knot_pos, knot_vals)
    # Mirror to full
    if n % 2 == 0:
        # even n, half_len = n/2
        full = np.concatenate([half_vals, half_vals[::-1]])
    else:
        # odd n, we need to exclude the duplicate middle element
        full = np.concatenate([half_vals, half_vals[-2::-1]])
    return full

"""
Propose a step‐function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Parametric symmetric shape** –  the sequence is forced to be
   symmetric (a[i]=a[n‑1‑i]) and is described by a small number K of
   logarithmic knot values.  Between the knots the values are obtained
   by linear interpolation, exponentiated and finally normalised to
   Σa = 1.  This reduces the optimisation dimension from O(n) to K
   (K≈9‑13) while still representing a rich family of shapes (arcsine,
   power‑law, beta‑like, window functions, etc.).

2. **Diverse initialisations** – for each length we build several
   starting parameter vectors from analytically known shapes:
   arcsine, power‑law, beta and a few classical windows (Hann,
   Hamming, Blackman, Kaiser).  All are projected to the log‑knot
   space.

3. **Annealed random walk in log‑knot space** –  from each seed we
   perform a simple Gaussian walk whose step size σ is exponentially
   decreasing (σ₀≈0.5 → σ_min≈10⁻³).  The surrogate objective is the
   exact ``evaluate_sequence`` value (the denominator is 1 after
   normalisation), which is cheap to compute via an FFT‑based
   convolution.  Whenever a new point improves the best local or the
   global score it is kept.

4. **Polishing** – the best vector found for the current length is
   refined by two deterministic procedures:

   * *Exact projected sub‑gradient descent* on the true objective
     (max‑convolution).  The sub‑gradient is obtained analytically via
     a convolution with the mask of maximal entries.
   * *Exponentiated‑gradient (KL‑mirror) descent* which respects the
     simplex constraints automatically.

5. **Final local random perturbations** – after all lengths have been
   processed a cheap pairwise mass‑transfer hill‑climb is run on the
   current global best until time runs out.

The whole search respects the 1000 s wall‑clock limit, always returns
a non‑empty list of non‑negative numbers and is completely
self‑contained (no closures, no I/O).  It improves upon the previous
pipeline by reducing the optimisation dimension, exploiting symmetry,
and adding a simple simulated‑annealing‑style random walk that is
effective at escaping local minima.
"""

# FFT‑based convolution utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised sequence (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Analytic shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law envelope 1/((i+½)(n‑½‑i)+eps)⁽ᵖ⁾, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like envelope on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def hann_window(n: int) -> np.ndarray:
    """Classic Hann window, normalised to sum = 1."""
    w = np.hanning(n)
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full(n, 1.0 / n)
    return w / w.sum()


def hamming_window(n: int) -> np.ndarray:
    w = np.hamming(n)
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full(n, 1.0 / n)
    return w / w.sum()


def blackman_window(n: int) -> np.ndarray:
    w = np.blackman(n)
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full(n, 1.0 / n)
    return w / w.sum()


def kaiser_window(n: int, beta: float = 8.0) -> np.ndarray:
    w = np.kaiser(n, beta)
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full(n, 1.0 / n)
    return w / w.sum()

def shape_from_log_params(p: np.ndarray, n: int) -> np.ndarray:
    """
    Build a symmetric non‑negative sequence of length n from K log‑knot
    parameters p (size K).  The knots are placed uniformly on the first
    half of the interval, values are exp(p) and linearly interpolated.
    The result is normalised later.
    """
    K = p.size
    half_len = (n + 1) // 2                     # includes centre for odd n
    knot_pos = np.linspace(0, half_len - 1, K)  # knot positions (float)
    knot_vals = np.exp(p)                       # ensure positivity
    idx = np.arange(half_len, dtype=float)
    half_vals = np.interp(idx, knot_pos, knot_vals)

    if n % 2 == 0:
        # even length – simply mirror
        full = np.concatenate([half_vals, half_vals[::-1]])
    else:
        # odd length – skip duplicated centre element
        full = np.concatenate([half_vals, half_vals[-2::-1]])
    return full


def log_params_from_shape(shape: np.ndarray, K: int) -> np.ndarray:
    """
    Given a full symmetric shape (already normalised), extract K
    logarithmic knot parameters by sampling the first half at uniform
    positions and taking the log.
    """
    n = shape.size
    half_len = (n + 1) // 2
    first_half = shape[:half_len]
    knot_pos = np.linspace(0, half_len - 1, K)
    sampled = np.interp(knot_pos, np.arange(half_len), first_half)
    # add tiny epsilon to avoid log(0)
    return np.log(sampled + 1e-12)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact objective)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.  Stops when time expires.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # mask of (near) maximal entries
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2 * (a ⋆ mask_rev) restricted to length n
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # diminishing step‑size
        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a

def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 50_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative (exponentiated) update
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a

def random_local_improvement(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    eps_frac: float = 0.001,
    attempts: int = 500_000,
) -> np.ndarray:
    """
    Simple pairwise mass transfer: pick two indices i≠j,
    move a small fraction (≤ eps_frac) of a[i] to a[j] and accept if
    the objective improves.
    """
    current_score = compute_score(a)

    n = a.size
    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move: random fraction of a[i]
        delta = eps_frac * rng.random() * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # keep non‑negativity (should hold)
        if a_new[i] < 0:
            continue
        new_score = compute_score(a_new)
        if new_score < current_score:
            a = a_new
            current_score = new_score
    return a

            b = rng.uniform(0.2, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            beta_shape = generate_beta_shape(n, a, b, eps)
            seed_params.append(log_params_from_shape(beta_shape, K))

        # classical windows
        for win in (hann_window, hamming_window, blackman_window):
            w = win(n)
            seed_params.append(log_params_from_shape(w, K))

        # Kaiser window with a random beta
        beta = rng.uniform(5.0, 12.0)
        w = kaiser_window(n, beta)
        seed_params.append(log_params_from_shape(w, K))

        # Include the current global best (if any) projected onto this length
        if best_seq:
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()
            seed_params.append(log_params_from_shape(cand, K))

        # ---- Random walk (annealed) for each seed --------------------
        # Time slice for this length: split remaining budget equally among lengths
        remaining = deadline - time.time()
        # Rough estimate of how many iterations we can afford:
        # each iteration ≈ 0.002 s for n≈2048, a bit more for larger n.
        it_budget = max(2000, int(remaining / 0.002 / max(1, len(lengths) - lengths.index(n))))
        # Guard against insane numbers
        it_budget = min(it_budget, 25000)

        # Schedule for σ: geometric decay from σ₀ to σ_min
        sigma0 = 0.5
        sigma_min = 1e-3
        decay = (sigma_min / sigma0) ** (1.0 / it_budget)

        # Keep the best local (per length) candidate
        best_local_score = float("inf")
        best_local_vec = None

        for seed_id, p0 in enumerate(seed_params, start=1):
            if time.time() >= deadline - 6.0:
                break
            print(f"\n  Seed {seed_id}/{len(seed_params)} – σ₀={sigma0:.3f}")

            p_cur = p0.copy()
            sigma = sigma0
            # Run a short inner walk for this seed
            inner_iters = max(2000, int(it_budget / len(seed_params)))
            for it in range(inner_iters):
                if time.time() >= deadline - 3.0:
                    break

                # propose a new point
                p_new = p_cur + sigma * rng.standard_normal(K)

                # build the full vector and normalise
                a_new = shape_from_log_params(p_new, n)
                a_new = a_new / a_new.sum()
                score_new = compute_score(a_new)

                # accept if improvement (simple hill‑climb)
                if score_new < best_local_score:
                    best_local_score = score_new
                    best_local_vec = a_new.copy()
                    # update global best as well
                    if score_new < best_val:
                        best_val, best_seq = score_new, a_new.tolist()
                        print(f"   *** New global best (len {n}) = {best_val:.6f} ***")
                    p_cur = p_new  # move centre of search
                else:
                    # occasional random acceptance to keep exploration
                    if rng.random() < np.exp(-(score_new - best_local_score) / (sigma + 1e-12)):
                        p_cur = p_new

                sigma *= decay

                if it % 3000 == 0 and it > 0:
                    print(f"    it={it} sigma={sigma:.6e} best_local={best_local_score:.6f}")

        # ------------------------------------------------------------------
        # 2. Refine the best vector for this length with deterministic passes
        # ------------------------------------------------------------------
        if best_local_vec is None:
            continue

        if time.time() >= deadline - 8.0:
            break

        # a) Exact sub‑gradient descent
        print(f"\n  Polishing (sub‑gradient) length {n}, score={best_local_score:.6f}")
        a_refined = subgradient_descent_exact(
            best_local_vec,
            deadline,
            max_iter=120_000,
            step0=0.07,
        )
        try:
            val = evaluate_sequence(a_refined.tolist())
            print(f"   Sub‑grad polish official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()
                print(f"   *** New global best after sub‑grad polish = {best_val:.6f} ***")
        except Exception as e:
            print(f"   Sub‑grad polishing failed: {e}")

        # b) KL‑mirror (exponentiated‑gradient) polishing
        if time.time() < deadline - 4.0:
            print("   Polishing (exponentiated‑gradient)…")
            a_refined = exponentiated_gradient_polish(
                np.array(best_seq, dtype=float),
                deadline,
                max_iter=30000,
                step=0.018,
            )
            try:
                val = evaluate_sequence(a_refined.tolist())
                print(f"   EG polish official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_refined.tolist()
                    print(f"   *** New global best after EG polish = {best_val:.6f} ***")
            except Exception as e:
                print(f"   EG polishing failed: {e}")

    # ------------------------------------------------------------------
    # 3. Final cheap local random mass‑transfer improvement
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal random local improvement…")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = random_local_improvement(
            a_arr,
            rng,
            deadline,
            eps_frac=0.0008,
            attempts=200_000,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final local‑improvement score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final random walk ***")
        except Exception as e:
            print(f"   Final improvement failed: {e}")

    # ------------------------------------------------------------------
    # 4. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {n0}, score = {best_val}")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
