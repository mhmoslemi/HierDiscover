"""
Optimiser for the “step‑function” challenge (improved version).

The algorithm proceeds in three main phases:

0️⃣  Load a fallback solution if one is present.
1️⃣  Generate a diverse pool of candidate shapes for a relatively large
    length (default n=4096).  The pool consists of deterministic shapes
    (uniform, arcsine) and a systematic grid / random sweep over a few
    parametric families (beta‑like, power‑like, mixtures, asymmetric
    power).  The fast surrogate score  S(a)=2·n·max(conv(a,a))  (with
    ∑a=1) is used to keep the ten best candidates.
2️⃣  Each top candidate is refined by a long Adam optimisation on the
    smooth‑max surrogate (log‑sum‑exp of the convolution) with an
    exponential temperature schedule up to τ≈200.  The parametrisation
    a=softmax(w) guarantees non‑negativity and Σa=1.
3️⃣  The best refined vector is polished by a projected sub‑gradient
    descent on the true max‑convolution objective followed by an
    exponentiated‑gradient (mirror‑descent) pass.  The official
    ``evaluate_sequence`` is called only for the few final candidates,
    keeping the overall run well within the 1000 s budget.

All stages respect the global deadline, keep the globally best vector
according to the official score, and finally return that vector as a
plain list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (length = len(a)+len(b)-1).
    Returns a float array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector a is assumed to sum to 1, therefore the denominator
    disappears and the score reduces to 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x ≥ 0, Σ x = 1}
    Implementation of Duchi et al. (2008), works for any dimension.
    """
    if np.all(v <= 0):
        # all non‑positive: uniform distribution is the projection
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at index ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_asymmetric_power_shape(
    n: int, p: float, eps_left: float, eps_right: float
) -> np.ndarray:
    """
    Asymmetric generalised arcsine:
    a_i ∝ 1 / ((i+eps_left)*(n‑½‑i+eps_right))**p .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + eps_left) * (n - 0.5 - i + eps_right) + 1e-12) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total

# Optimisers
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    tau_start: float = 0.5,
    tau_end: float = 200.0,
    base_lr: float = 0.2,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log ∑_k exp(τ·conv_k).

    The temperature τ is raised exponentially from ``tau_start`` to
    ``tau_end`` over the course of the iteration.  Variables are stored
    as logits w with a = softmax(w) guaranteeing non‑negativity and
    Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits (add tiny offset for numerical stability)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # a = softmax(w)
        w_shift = w - np.max(w)  # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a * a
        conv = convolution_fft(a, a)

        # compute softmax weights over τ·conv
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()  # shape (L,)

        # gradient wrt a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wr.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (fast surrogate)
        if it % 50000 == 0:
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={compute_score(a):.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution
    objective.  The step size decays like 1/√t.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries attaining the maximum (within tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) restricted to original indices
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 50000 == 0:
            print(f"    Subgrad it={it:,d} surrogate={compute_score(a):.6f}")

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.15,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror‑descent with KL geometry) on the
    true max‑convolution objective.
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        if it % 50000 == 0:
            print(f"    EG it={it:,d} surrogate={compute_score(a):.6f}")

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence minimising ``evaluate_sequence``.
    The routine respects the 1000 s global budget, keeps the best ever
    sequence (according to the official score), and finally returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Optional fallback solution
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Choose a working size (large enough to beat the current bound)
    # ------------------------------------------------------------------
    n = 4096                     # a good compromise between quality & speed
    print(f"\nWorking length set to n={n}")

    # ------------------------------------------------------------------
    # 2️⃣  Build a pool of candidate shapes
    # ------------------------------------------------------------------
    pool: List[Tuple[float, np.ndarray]] = []

    # deterministic baseline shapes
    uniform = np.full(n, 1.0 / n)
    pool.append((compute_score(uniform), uniform.copy()))
    arc = arcsine_initial(n)
    pool.append((compute_score(arc), arc.copy()))

    # systematic sweep over the power‑shape exponent (greatly influences the constant)
    for p in np.linspace(0.4, 1.2, 31):  # 31 values gives a coarse but useful grid
        cand = generate_power_shape(n, p, eps=1e-9)
        pool.append((compute_score(cand), cand))

    # randomised parametric families (still cheap)
    for _ in range(300):
        if time.time() >= deadline - 5.0:
            break
        typ = rng.integers(0, 5)
        if typ == 0:  # beta‑like
            a = rng.uniform(0.05, 1.0)
            b = rng.uniform(0.05, 1.0)
            eps = rng.uniform(0.0, 3.0)
            cand = generate_beta_shape(n, a, b, eps)
        elif typ == 1:  # power‑like
            p = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 4.0)
            cand = generate_power_shape(n, p, eps)
        elif typ == 2:  # mixture arcsine / uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)
        elif typ == 3:  # triangular bump (peak anywhere)
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
        else:          # asymmetric power
            p = rng.uniform(0.2, 2.5)
            eps_l = rng.uniform(0.0, 2.0)
            eps_r = rng.uniform(0.0, 2.0)
            cand = generate_asymmetric_power_shape(n, p, eps_l, eps_r)

        pool.append((compute_score(cand), cand))

    # keep the best few for the expensive Adam stage
    pool.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in pool[:7]]
    print(f"\nShape seeding finished – best surrogate = {pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimisation on the top candidates
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_candidates, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Adam refinement on candidate {idx}/{len(top_candidates)} ---")
        remaining = deadline - time.time()
        # Rough heuristic: aim for ~1e-5 seconds per iteration → allocate
        # up to 250 k iterations if time permits.
        max_iters = int(min(250_000, remaining * 2000))
        a_opt = adam_optimize_smooth_max(
            seed_vec,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=200.0,
            base_lr=0.2,
        )
        # official evaluation (fast, but we call it only for the best)
        val = evaluate_sequence(a_opt.tolist())
        print(f"    Official score after Adam = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()
            print("*** New global best! ***")

    # ------------------------------------------------------------------
    # 4️⃣  Exact sub‑gradient + EG polish on the current global best
    # ------------------------------------------------------------------
    if best_seq:
        a = np.array(best_seq, dtype=float)
        # sub‑gradient stage
        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 2000))
        a = subgradient_descent_exact(a, deadline, max_iter=max_iter_sub, step0=0.07)

        # EG polish stage
        remaining = deadline - time.time()
        max_iter_eg = int(min(200_000, remaining * 2000))
        a = eg_refine(a, deadline, max_iter=max_iter_eg, eta0=0.15)

        final_val = evaluate_sequence(a.tolist())
        print(f"\nFinal polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5️⃣  Guard – always return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        print("\nNo valid candidate produced – falling back to uniform.")
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
