"""
Propose a step‑function (non‑negative sequence) with a minimal
``evaluate_sequence`` score.

Algorithm
---------
1. **Fallback** – if a previous solution (`height_sequence_1`) exists,
   evaluate it and keep it if it is the best seen so far.
2. **Small‑scale LP polishing** – start from the classic arcsine shape
   (n=1024, sum = 1).  Repeatedly solve the linear programme

        max   Σ_j g_j
        s.t.  (f * g)_k ≤ M   ∀k,   g_j ≥ 0 ,

   where ``M = max(f * f)`` and ``f`` is the current distribution.
   The solution ``g`` is scaled to unit sum and mixed with ``f`` using a
   diminishing Frank–Wolfe step.  This convex‑linear step quickly lowers
   the convolution peak.
3. **Upscaling + smooth‑max Adam** – linearly interpolate the best
   small vector to larger sizes (2048 → 4096).  For each size run a long
   Adam optimisation of the smooth‐max surrogate

        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·(a*a)_k),

   with an exponential temperature schedule (τ from 0.5 to ≈200).  The
   variables are soft‑max logits, guaranteeing non‑negativity and a sum
   of one.
4. **Exact sub‑gradient + exponentiated‑gradient polish** – on the
   best large vector perform projected sub‑gradient descent on the true
   max‑convolution objective followed by an EG polish.  This refines the
   solution in the non‑smooth regime.
5. **Local stochastic polishing** – while time permits, add small
   Gaussian jitter, re‑project onto the simplex and keep any improvement.
6. Return the best sequence found (as a plain ``list[float]``).

All stages respect a global 1000 s budget and keep the globally best
sequence according to the official ``evaluate_sequence`` function.
"""

import time
import numpy as np
from typing import List
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors via FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector a is assumed to sum to 1, so the result equals
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
    {x ≥ 0, Σ x = 1}.
    Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(v) to target_n,
    followed by re‑normalisation to sum = 1.
    """
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up = up / up.sum()
    return up


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 30_000,
    tau_start: float = 0.5,
    tau_end: float = 200.0,
    base_lr: float = 0.25,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = 1/τ·log Σ_k exp(τ·(a*a)_k)
    with an exponential temperature schedule from ``tau_start`` to
    ``tau_end``.  Variables are soft‑max logits (w) guaranteeing a ≥ 0,
    Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from a_init (add tiny epsilon for stability)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # softmax weights over τ·conv
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()  # shape (L,)

        # gradient wrt a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wr.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging
        if it % 10_000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true
    max‑convolution objective.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror‑descent with KL geometry).
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a


# ----------------------------------------------------------------------
# Linear‑programme helper (LP‑based Frank–Wolfe step)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the (2n‑1) × n matrix A such that A·g = f*g
    (linear convolution, non‑circular).  Row k corresponds to shift k.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(
    f: np.ndarray,
    deadline: float,
    max_iters: int = 25,
) -> tuple[np.ndarray, float]:
    """
    Perform a few Frank–Wolfe‑style LP refinements of ``f``.
    Returns the best vector seen (still normalised to sum = 1) and its
    surrogate score (2·n·max(conv)).
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline - 2.0:
            break

        conv = convolution_fft(f, f)
        M = float(np.max(conv))

        # Build linear constraints: (f * g)_k ≤ M
        A = build_conv_matrix(f)
        b = np.full(2 * n - 1, M)

        # maximise Σ g_j  → minimise -Σ g_j
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
        )
        if not res.success:
            print(f"LP iteration {it} failed: {res.message}")
            break

        g = res.x
        if g.sum() <= 0.0:
            break

        # scale to unit sum
        g = g / g.sum()

        # diminishing Frank–Wolfe step size
        t = 2.0 / (it + 2.0)
        f = (1.0 - t) * f + t * g
        f = f / f.sum()        # numeric safety

        score = compute_score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
            print(f"LP iteration {it}: improved surrogate {score:.6f}")

    return best_f, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """

      * optional fallback,
      * LP‑based refinement on a 1024‑length arcsine start,
      * up‑sampling + Adam smooth‑max optimisation (2048 → 4096),

    """

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Small‑scale LP refinement (n=1024)
    # ------------------------------------------------------------------
    n_small = 1024

    f = f / f.sum()                     # ensure exact normalisation

    f, small_score = lp_refine(f, deadline, max_iters=25)
    val = evaluate_sequence(f.tolist())
    print(f"LP‑refined small sequence official score = {val:.6f}")
    if val < best_val:
        best_val, best_seq = val, f.tolist()
        print("*** New global best after LP refinement ***")

    # ------------------------------------------------------------------

    current_vec = f

        current_vec = up_opt

    if current_vec is not None and time.time() < deadline - 5.0:

        max_iter_sub = int(min(250_000, remaining * 30))

        except Exception as e:
            print(f"Evaluation failed after refinement: {e}")
