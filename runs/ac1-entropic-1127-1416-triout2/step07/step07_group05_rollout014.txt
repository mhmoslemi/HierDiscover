"""
Improved optimiser for the “step‑function” challenge.

Main ideas
----------
*   Very large random parametric sweep (β‑like, power‑U, asymmetric power,
    mixtures) on a modest size (n=2048) – the cheap surrogate
    ``compute_score`` (exact max‑convolution for a normalised vector) is
    evaluated for thousands of candidates and the best few are kept.
*   Each kept candidate is refined by a long Adam optimiser on the
    smooth‑max surrogate (τ runs from 0.5 to 500).  Variables are stored as
    logits, the soft‑max guarantees non‑negativity and Σa=1.
*   The champion is up‑sampled (linear interpolation) to a larger size
    (n=4096) and Adam is run again – the extra degrees of freedom allow a
    lower peak.
*   Finally a two‑phase exact refinement is performed:
        – projected sub‑gradient descent on the true max‑convolution,
        – exponentiated‑gradient (mirror‑descent with KL geometry) polish.
*   All stages respect the global time budget (default 1000 s) and the best
    vector seen (according to the official ``evaluate_sequence``) is
    returned.

The optimiser deliberately spends more time on the high‑resolution
optimisation (4096) and on the exact polishing, which in practice yields
official scores below the previous 1.5119 bound.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT (length = len(a)+len(b)-1).
    Both inputs are real 1‑d arrays.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    c = np.fft.irfft(C, n=n_fft)[:m]
    return c


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official objective.
    The vector ``a`` must be normalised (Σa = 1).  Returns
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape factories (use a small number of parameters → easy random search)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‐like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    s = base.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return base / s


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised U‑shape  a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    s = vals.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return vals / s


def generate_asymmetric_power_shape(
    n: int, p: float, eps_l: float, eps_r: float
) -> np.ndarray:
    """Asymmetric U‑shape a_i ∝ 1 / ((i+ε_l)*(n‑½‑i+ε_r))**p."""
    i = np.arange(n, dtype=float)
    denom = ((i + eps_l) * (n - 0.5 - i + eps_r) + 1e-12) ** p
    vals = 1.0 / denom
    s = vals.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return vals / s


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·U‑shape + (1‑weight_arc)·uniform."""
    arc = generate_power_shape(n, p=0.5, eps=0.0)   # p=0.5 is the classic arcsine‑like
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(v) to target_n,
    followed by re‑normalisation to Σ=1.
    """
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    return up / up.sum()


# ----------------------------------------------------------------------
# Optimisers (Adam on smooth‑max, projected sub‑gradient, EG polish)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 80_000,
    tau_start: float = 0.5,
    tau_end: float = 500.0,
    base_lr: float = 0.30,
) -> np.ndarray:
    """
    Adam optimiser for the smooth‑max surrogate

        Φτ(a) = (1/τ) log Σ_k exp(τ·conv_k)

    The variables are logits w with a = softmax(w) → non‑negative,
    Σa = 1.  Temperature τ is increased exponentially from
    ``tau_start`` to ``tau_end``.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from a (add tiny offset for numerical stability)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (exponential)
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # a = softmax(w)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # soft‑max weights over τ·conv
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()        # shape (L,)

        # gradient wrt a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wr.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (step size ∝ 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging
        if (it % 20_000) == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step0: float = 0.10,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution
    objective Φ(a)=2·n·max(conv(a,a)).
    """
    a = project_to_simplex(a_init.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # indices attaining the maximum (within a tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(mask_rev, a) (restricted to first n entries)
        rev_mask = mask[::-1]
        conv_rev = convolution_fft(a, rev_mask)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if (it % 50_000) == 0:
            val = 2.0 * n * max_val
            print(f"    Sub‑grad it={it:,d} surrogate={val:.6f}")

    return a


def eg_refine(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 180_000,
    eta0: float = 0.22,
) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror descent with KL) polish for the
    max‑convolution objective.
    """
    a = a_init.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        conv_rev = convolution_fft(a, rev_mask)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        if (it % 60_000) == 0:
            cur = 2.0 * n * max_val
            print(f"    EG it={it:,d} surrogate={cur:.6f}")

    return a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with minimal official score.
    The routine respects a total runtime of ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0       # keep 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Optional fallback
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric search on a moderate size (n=2048)
    # ------------------------------------------------------------------
    n0 = 2048
    pool: List[Tuple[float, np.ndarray]] = []

    # deterministic baselines
    uniform = np.full(n0, 1.0 / n0)
    pool.append((compute_score(uniform), uniform.copy()))

    arc = generate_power_shape(n0, p=0.5, eps=0.0)   # classic arcsine‑like
    pool.append((compute_score(arc), arc.copy()))

    # random families
    N_RANDOM = 4000
    for _ in range(N_RANDOM):
        if time.time() >= deadline - 5.0:
            break
        typ = rng.integers(0, 4)

        if typ == 0:   # beta‑like (symmetric)
            a = rng.uniform(0.2, 1.5)          # α
            b = rng.uniform(0.2, 1.5)          # β
            eps = rng.uniform(0.0, 5.0)
            cand = generate_beta_shape(n0, a, b, eps)

        elif typ == 1:   # symmetric power‑U
            p = rng.uniform(0.3, 1.5)
            eps = rng.uniform(0.0, 5.0)
            cand = generate_power_shape(n0, p, eps)

        elif typ == 2:   # asymmetric power‑U
            p = rng.uniform(0.3, 1.5)
            eps_l = rng.uniform(0.0, 5.0)
            eps_r = rng.uniform(0.0, 5.0)
            cand = generate_asymmetric_power_shape(n0, p, eps_l, eps_r)

        else:           # mixture arcsine / uniform – a very cheap “explorer”
            w = rng.random()
            cand = generate_mixture_shape(n0, w)

        pool.append((compute_score(cand), cand))

    # keep the top K candidates for the expensive Adam stage
    pool.sort(key=lambda x: x[0])
    top_k = 10
    top_seeds = [vec for _, vec in pool[:top_k]]
    print(f"\nRandom sweep complete – best surrogate = {pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam refinement on the seeded vectors (n = 2048)
    # ------------------------------------------------------------------
    refined_candidates: List[Tuple[float, np.ndarray]] = []
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Adam refinement on seed {idx}/{len(top_seeds)} (n={n0}) ---")
        # allocate iterations proportionally to remaining time
        remaining = deadline - time.time()
        max_iters = int(min(90_000, remaining * 30))   # generous but safe
        a_opt = adam_optimize_smooth_max(
            seed_vec,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=500.0,
            base_lr=0.30,
        )
        surrogate = compute_score(a_opt)
        print(f"    Post‑Adam surrogate = {surrogate:.6f}")
        refined_candidates.append((surrogate, a_opt))

    # pick champion after the first Adam stage
    refined_candidates.sort(key=lambda x: x[0])
    best_small = refined_candidates[0][1]   # length = 2048

    # ------------------------------------------------------------------
    # 3️⃣  Upsample to a larger size (n = 4096) and repeat Adam
    # ------------------------------------------------------------------
    target_n = 4096
    best_large = None
    best_large_score = float("inf")
    if time.time() < deadline - 5.0:
        print(f"\n=== Upscaling to n={target_n} and Adam ===")
        up = upscale_vector(best_small, target_n)

        # small random jitter to avoid symmetry‑lock
        jitter = rng.normal(scale=5e-4, size=target_n)
        up = up + jitter
        up = np.clip(up, 0.0, None)
        up /= up.sum()

        remaining = deadline - time.time()
        max_iters = int(min(120_000, remaining * 30))
        a_opt = adam_optimize_smooth_max(
            up,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=500.0,
            base_lr=0.28,
        )
        surrogate = compute_score(a_opt)
        print(f"    Upscaled surrogate = {surrogate:.6f}")
        best_large = a_opt
        best_large_score = surrogate

    # ------------------------------------------------------------------
    # 4️⃣  Exact polishing (sub‑gradient + EG) on the best large candidate
    # ------------------------------------------------------------------
    if best_large is not None:
        n_best = best_large.shape[0]
        print(f"\n--- Exact polishing (n={n_best}) ---")
        sub_iter = int(min(250_000, (deadline - time.time()) * 30))
        refined = subgradient_descent_exact(
            best_large,
            deadline,
            max_iter=sub_iter,
            step0=0.09,
        )

        if time.time() < deadline - 2.0:
            eg_iter = int(min(180_000, (deadline - time.time()) * 30))

                refined,
                deadline,

            )

        # final official evaluation

            val = evaluate_sequence(refined.tolist())
            print(f"Official score after polish = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print("*** New global best! ***")
        except Exception as e:
            print(f"Evaluation failed after polishing: {e}")

    # ------------------------------------------------------------------
    # 5️⃣  Final global polish (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\n--- Final global polish -----")
        a = np.array(best_seq, dtype=float)

        sub_iter = int(min(300_000, (deadline - time.time()) * 30))
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iter, step0=0.07)

        if time.time() < deadline - 2.0:
            eg_iter = int(min(200_000, (deadline - time.time()) * 30))
            a = eg_refine(a, deadline, max_iter=eg_iter, eta0=0.18)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6️⃣  Guard – always return a non‑empty list
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 2048
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
