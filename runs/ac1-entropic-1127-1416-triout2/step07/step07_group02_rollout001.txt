def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    n = 2048  # start with baseline length

    # 1. Baseline families enumerations
    families = []

    families.append(arcsine_initial(n))
    families.append(np.full(n, 1.0/n))
    families.append(generate_mixture_shape(n, 0.5))

    # power families
    for p in np.linspace(0.5,5.0,10):
        families.append(generate_power_shape(n,p,eps=1e-9))

    # beta families grid
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    betas = [0.2, 0.5, 1.0, 2.0, 5.0]
    for a in alphas:
        for b in betas:
            families.append(generate_beta_shape(n, a, b, eps=1e-8))

    # mixture arcsine weight
    for w in np.linspace(0.0,1.0,11):
        families.append(generate_mixture_shape(n,w))

    # Quick scoring
    scored = []
    for vec in families:
        score = fast_score(vec)
        scored.append((score, vec))
    scored.sort(key=lambda x: x[0])
    top_k = 5
    candidates = [vec for _, vec in scored[:top_k]]

    # 2. For each candidate, apply refinement:
    for i, cand in enumerate(candidates):
        if time.time() >= deadline - 5.0:
            break
        # start mirror descent (exponential gradient) for up to some time
        sub_deadline = time.time() + min(30.0, (deadline - time.time())/ (len(candidates)-i+1))
        refined = mirror_descent_optimize(cand, sub_deadline, step0=0.2, decay=0.9998)
        # polish with subgradient descent for extra time
        if time.time() < deadline - 2.0:
            sub_deadline2 = time.time() + min(20.0, deadline - time.time() - 1.0)
            refined = subgradient_descent_exact(refined, sub_deadline2, max_iter=50000, step0=0.05)

        score = fast_score(refined)
        print(f"Candidate {i+1} final score approx {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = refined.tolist()
            print(f"*** New best: {best_score:.6f}")

    # Also maybe try random hill climb from best
    if best_seq is not None and time.time() < deadline - 5.0:
        # hill climb
        a = np.array(best_seq, dtype=float)
        # gradient steps with random directional perturbations to reduce max conv
        # Use simple Simulated annealing: propose a + epsilon (zero-mean) then renorm to simplex.
        while time.time() < deadline - 2.0:
            # perturbation
            noise = rng.normal(scale=0.01, size=a.shape)
            cand = a + noise
            cand = np.maximum(cand, 0.0)
            if cand.sum() == 0:
                cand = np.full_like(cand, 1.0/a.size)
            else:
                cand /= cand.sum()
            score = fast_score(cand)
            cur_score = fast_score(a)
            # accept if better
            if score < cur_score:
                a = cand
        best_seq = a.tolist()
        best_score = fast_score(a)

    if not best_seq:
        best_seq = [1.0 / n] * n
        best_score = fast_score(np.array(best_seq))

    print(f"Search finished. Best score {best_score:.6f}")
    return best_seq

"""
Propose a low‑value step‑function.

Algorithm
---------

1. **Exploratory grid** – For a few moderate lengths (1024, 2048,
   4096) we generate a diverse collection of analytic families:
   * arcsine, uniform, mixtures of arcsine & uniform,
   * power‑law shapes  (∝[(i+½)(n‑½‑i)]⁻ᵖ) for p∈[0.5,5],
   * Beta‑shapes  a^(α‑1)*(n‑1‑a)^(β‑1)   with α,β∈{0.2,0.5,1,2,5},
   * a handful of random Dirichlet draws (concentration 0.3–2).

   Every candidate is normalised to a probability vector
   (sum = 1) and scored with the cheap surrogate
   ``fast_score = 2·n·max(conv(a,a))`` – this equals the official
   objective when ``sum(a)=1`` because the denominator cancels.

2. **Selection** – The five best candidates (lowest surrogate score)
   across all lengths are kept.

3. **Refinement** – Each selected vector is further optimised on the
   simplex by a *mirror‑descent* (exponential‑gradient) step that uses
   the exact sub‑gradient of the max‑convolution.  The step‑size is
   deliberately generous (initial 0.2, exponential decay) to move out
   of shallow local minima.

4. **Polishing** – The refined vectors are finally polished with a
   projected sub‑gradient descent (the same routine used in the
   previous state‑of‑the‑art).  This guarantees feasibility and often
   yields a small extra drop in the score.

5. **Fallback** – If no candidate improves the initial uniform vector,
   a uniform distribution of length 512 is returned (guaranteeing a
   valid answer).

All work respects the supplied time budget (default 1000 s).  The
routine prints progress and the best official score found.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Fast convolution (real FFT) and surrogate score
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """Cheap surrogate: 2·n·max(conv(a,a)) (sum(a) is 1)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families of probability vectors
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Shape ∝ [(i+½)(n‑½‑i)+eps]⁻ᵖ."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))/sum(a)².
    Works directly with the true max‑convolution (mask‑based sub‑gradient).
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) w.r.t. a
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        # occasional debug output
        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d}  surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) with exact max‑subgradient
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.2,
    decay: float = 0.9998,
) -> np.ndarray:
    """
    Exponential‑gradient descent on the true objective.
    The iterate stays on the simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative KL‑mirror step
        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay

        # occasional progress line
        if int(step * 1e6) % 200_000 == 0:
            print(f"    Mirror step={step:.6f}  surrogate≈{fast_score(a):.6f}")

    return a

    """
    rng = np.random.default_rng(seed)

        scored = [(fast_score(v), v) for v in candidates]

            # Mirror‑descent phase (exact max sub‑gradient)

            a_md = mirror_descent_optimize(vec, md_deadline,

            a_final = subgradient_descent_exact(a_md, pg_deadline,
                                                max_iter=150_000,
                                                step0=0.05)

            # Evaluate the (exact) objective for bookkeeping
            try:

                surrogate_val = fast_score(a_final)
                print(f"   Official score = {official_val:.6f} (surrogate≈{surrogate_val:.6f})")
                if official_val < best_score:
                    best_score, best_seq = official_val, a_final.tolist()
                    print(f"   *** New global best: {best_score:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 4. Light random hill‑climb starting from the current best (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
