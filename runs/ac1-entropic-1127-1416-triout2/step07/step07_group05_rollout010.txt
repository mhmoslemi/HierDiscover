    A = diags(f, offsets=-np.arange(n), shape=(2*n-1, n), format='csc')
    c = -np.ones(n)  # minimize -sum(b) => maximize sum

    b = res.x
    return b

"""
Propose a step‑function vector that minimises ``evaluate_sequence``.

Approach
--------
*   Start from a few simple seeds (arcsine, uniform, optional fallback).
*   Repeatedly apply a *linear‑programming improvement* (Frank–Wolfe style):
    for the current vector f (∑f=1) compute M = max(f*f).
    Solve the LP  maximise   1ᵀb   subject to   f∗b ≤ M ,  b ≥ 0.
    The LP gives a direction that can increase the total mass while keeping the
    convolution peak unchanged.  By mixing the new direction with the old
    vector and renormalising we strictly decrease the objective
    2·n·max(f*f) (the surrogate of the official score).
*   Perform the LP‑step on a modest size (n≈1024) for a few iterations,
    then linearly up‑sample to larger lengths (2048, 4096) and repeat.
*   Finish with a short Euclidean‑projected sub‑gradient descent to polish
    the result.
*   Throughout the search keep the best vector seen according to the real
    ``evaluate_sequence`` call.

The whole routine respects the global 1000 s budget and returns a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional import of the optimisation stack
# ----------------------------------------------------------------------
try:                           # SciPy is usually available in the environment
    from scipy.sparse import diags, csc_matrix
    from scipy.optimize import linprog
    SCIPY_OK = True
except Exception:             # If SciPy is missing we fall back to a trivial seed
    SCIPY_OK = False

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, fast for many evaluations)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector a is assumed to be normalised (sum = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def uniform_initial(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def upscale_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(v) to target_n, re‑normalised to sum = 1."""
    if target_n == len(v):
        return v.copy()
    old_idx = np.arange(len(v))
    new_idx = np.linspace(0, len(v) - 1, target_n)
    up = np.interp(new_idx, old_idx, v)
    up /= up.sum()
    return up


# ----------------------------------------------------------------------
# Projection utilities (needed for sub‑gradient polishing)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming improvement (Frank‑Wolfe step)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> csc_matrix:
    """
    Construct the sparse matrix A such that A @ b = f * b (linear convolution).
    Shape of A is (2*n-1, n).
    """
    n = f.shape[0]
    L = 2 * n - 1
    offsets = -np.arange(n)                 # diagonal offsets = -d
    # each diagonal k (= -d) has constant value f[d] and length L-d
    data = [np.full(L - d, f[d], dtype=float) for d in range(n)]
    A = diags(data, offsets, shape=(L, n), format="csc")
    return A


def lp_step(f: np.ndarray, t_mix: float) -> np.ndarray | None:
    """
    One Frank‑Wolfe style LP step.
    Returns the mixed and normalised vector, or ``None`` on failure.
    """
    n = f.shape[0]
    # current maximal convolution value
    M = float(np.max(convolution_fft(f, f)))
    # build convolution matrix A so that A @ b = f * b
    A = build_conv_matrix(f)
    L = 2 * n - 1
    # inequality: A @ b ≤ M   (component‑wise)
    ub = np.full(L, M, dtype=float)
    # objective: maximise 1ᵀb  → minimise -1ᵀb
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n

    # solve the LP (highs is robust for dense right‑hand side)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception as e:
        print(f"   LP (highs) exception: {e}")
        return None

    if not res.success:
        # LP may fail if the problem is degenerate; just abort the step.
        print(f"   LP failed (status {res.status}): {res.message}")
        return None

    b = res.x
    if b.sum() <= 0.0:
        return None

    # direction g (normalised to sum = 1)
    g = b / b.sum()
    # mixing with the current iterate
    new_f = (1.0 - t_mix) * f + t_mix * g
    new_f /= new_f.sum()
    return new_f


def lp_improve(
    start_f: np.ndarray,
    max_iter: int,
    deadline: float,
    t0: float = 0.5,
    t_decay: float = 0.90,
    t_min: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Apply a short sequence of LP steps to improve the surrogate score.
    Returns the best vector seen and its surrogate value.
    """
    if not SCIPY_OK:
        # If the linear‑programming stack is unavailable we simply return the start.
        return start_f, compute_score(start_f)

    f = start_f.copy()
    best_f = f.copy()
    best_sc = compute_score(f)
    t = t0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline - 0.1:
            break
        new_f = lp_step(f, t)
        if new_f is None:
            break
        f = new_f
        t = max(t_min, t * t_decay)          # slowly shrink the mixing factor
        sc = compute_score(f)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            print(f"   LP iter {it:>2d}: surrogate = {sc:.6f}")
    return best_f, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient polishing (optional final stage)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 50000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the exact max‑convolution
    objective.  A modest number of iterations is enough for a final polish.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline - 0.1:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with minimal ``evaluate_sequence``.
    The routine combines a linear‑programming Frank‑Wolfe phase, up‑sampling,
    and a short sub‑gradient polish.  All stages respect the global 1000 s
    time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Load a fallback solution if present (previous run)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Build a small pool of seed vectors (n≈1024)
    # ------------------------------------------------------------------
    n_small = 1024
    seeds: list[np.ndarray] = [arcsine_initial(n_small), uniform_initial(n_small)]

    # add a few random Beta‑like shapes for diversity
    for _ in range(5):
        a = rng.uniform(0.1, 0.9)
        b = rng.uniform(0.1, 0.9)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_beta_shape(n_small, a, b, eps))

    # if we have a fallback, incorporate its normalised version as a seed
    if best_seq:
        fall_arr = np.array(best_seq, dtype=float)
        # truncate / interpolate to n_small entries
        if fall_arr.size != n_small:
            fall_arr = upscale_vector(fall_arr, n_small)
        fall_arr = np.clip(fall_arr, 0.0, None)
        fall_arr /= fall_arr.sum()
        seeds.append(fall_arr)

    # ------------------------------------------------------------------
    # 2️⃣  Run a few LP improvement cycles on each seed (small size)
    # ------------------------------------------------------------------
    best_local_f: np.ndarray | None = None
    best_local_sc: float = float("inf")

    for idx, seed_f in enumerate(seeds, start=1):
        if time.time() >= deadline - 0.5:
            break
        print(f"\n--- LP improvement on seed {idx}/{len(seeds)} (n={n_small}) ---")
        # each seed gets at most 8 LP iterations
        f_opt, sc_opt = lp_improve(
            seed_f, max_iter=8, deadline=deadline, t0=0.5, t_decay=0.85
        )
        print(f"   best surrogate after LP = {sc_opt:.6f}")
        if sc_opt < best_local_sc:
            best_local_sc = sc_opt
            best_local_f = f_opt.copy()

    if best_local_f is None:
        # fall back to a simple uniform vector
        best_local_f = uniform_initial(n_small)

    # ------------------------------------------------------------------
    # 3️⃣  Upsample and repeat LP improvement on larger sizes
    # ------------------------------------------------------------------
    for target_n in (2048, 4096):
        if time.time() >= deadline - 0.5:
            break
        print(f"\n=== Upscaling to n={target_n} and LP refinement ===")
        best_local_f = upscale_vector(best_local_f, target_n)

        # a tiny jitter can help escape pathological flats
        jitter = rng.normal(scale=1e-4, size=target_n)
        best_local_f = np.clip(best_local_f + jitter, 0.0, None)
        best_local_f /= best_local_f.sum()

        # perform a short LP run (fewer iterations for larger n)
        max_iters = 5 if target_n == 2048 else 4
        best_local_f, sc = lp_improve(
            best_local_f,
            max_iter=max_iters,
            deadline=deadline,
            t0=0.4,
            t_decay=0.80,
        )
        print(f"   surrogate after up‑scaled LP = {sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing with a brief sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final sub‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iter = int(min(20000, remaining * 30))
        polished = subgradient_descent_exact(
            best_local_f,
            deadline,
            max_iter=max_iter,
            step0=0.10,
        )
    else:
        polished = best_local_f

    # ------------------------------------------------------------------
    # 5️⃣  Evaluate the polished candidate (official score) and keep best
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(polished.tolist())
        print(f"Official score of polished candidate = {official_val:.6f}")
        if official_val < best_val:
            best_val, best_seq = official_val, polished.tolist()
    except Exception as e:
        print(f"Evaluation failed on polished candidate: {e}")

    # ------------------------------------------------------------------
    # 6️⃣  Guard – ensure we always return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
