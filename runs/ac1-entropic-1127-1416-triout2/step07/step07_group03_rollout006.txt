# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (no circular wrapping).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    conv = convolution_fft(a, a)
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective when the vector already
    sums to 1:  2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.25) -> np.ndarray:
    """Classic arcsine‑type envelope."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-9) -> np.ndarray:
    """Discrete Beta(α,β) shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    vals = left * right
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def power_shape(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """Shape proportional to 1/((i+0.5)*(n-0.5-i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine shape and the uniform distribution."""
    arc = arcsine_shape(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def dirichlet_shape(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑distributed vector (conc > 0)."""
    vec = rng.dirichlet(np.full(n, conc))
    # Dirichlet already sums to 1, but guard against numerical issues
    vec = np.clip(vec, 0.0, None)
    s = vec.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return vec / s


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    Reference: Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Smooth‑max surrogate + Adam optimiser (soft‑max parametrisation)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max converting ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 400.0,
    ramp_iters: int = 50_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam minimises the smooth log‑sum‑exp surrogate of the max‑convolution.
    The iterate stays on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(conv.max())

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp weighting
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a : 2 * (a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 10_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate={compute_score(a):.6f}")

    # final vector on the simplex
    a = softmax(z)
    return a

# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    init_step: float = 0.02,
) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective with KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0
    step = init_step

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(conv.max())
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative update
        a = a * np.exp(-step * subgrad)
        a_sum = a.sum()
        if a_sum <= 0 or not np.isfinite(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # simple step‑size decay
        if it % 10_000 == 0:
            step *= 0.7
            print(f"    EG it={it:,d} step={step:.5f} score={compute_score(a):.6f}")

    return a

def subgradient_descent_exact(

    Projected sub‑gradient descent on f(a)=2·n·max(a*a).
    The iterate stays on the simplex.

    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(conv.max())
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Small random walk around a reference vector (keeps sum = 1)
# ----------------------------------------------------------------------


def random_local_perturb(a: np.ndarray, rng: np.random.Generator, delta: float = 0.02) -> np.ndarray:
    """
    Move a tiny amount δ from a randomly chosen index i to another index j.
    Guarantees non‑negativity and keeps the sum unchanged.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a
    # amount we can take from i without making it negative
    take = min(delta, a[i])
    if take <= 0:
        return a
    a_new = a.copy()
    a_new[i] -= take
    a_new[j] += take
    return a_new

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 43, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine respects a 1000 s wall‑clock limit and always returns a
    non‑empty list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Load fallback if a previous solution exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Length schedule – try larger lengths first
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024, 512]

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1a. Generate a huge and diverse seed pool
        # --------------------------------------------------------------
        seeds: List[np.ndarray] = []

        # a) deterministic families
        seeds.append(arcsine_shape(n, eps=0.25))
        seeds.append(arcsine_shape(n, eps=1.0))
        seeds.append(np.full(n, 1.0 / n))

        # b) random beta families
        for _ in range(8):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(beta_shape(n, a, b, eps))

        # c) random power‑law families
        for _ in range(6):
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(power_shape(n, p, eps))

        # d) mixtures of arcsine & uniform
        for _ in range(4):
            w = rng.uniform(0.0, 1.0)
            seeds.append(mixture_shape(n, w))

        # e) Dirichlet draws with various concentration levels
        for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
            for _ in range(2):
                seeds.append(dirichlet_shape(n, conc, rng))

        # f) warm‑start from the current global best (if any)
        if best_seq:
            old = np.array(best_seq, dtype=float)
            if old.size >= n:
                cand = old[:n]
            else:
                reps = int(np.ceil(n / old.size))
                cand = np.tile(old, reps)[:n]
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()
            seeds.append(cand)

        # --------------------------------------------------------------
        # 1b. Fast pre‑selection – keep the most promising seeds
        # --------------------------------------------------------------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:8]]   # keep at most eight per length

        # --------------------------------------------------------------
        # 1c. Allocate a time slice per seed and run the optimisation pipeline
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        per_seed_time = max(3.0, remaining / (len(top_seeds) * 1.5))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 6.0:
                break
            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_time:.1f}s")
            seed_deadline = min(time.time() + per_seed_time, deadline - 2.0)

            # initialise Adam via log‑softmax (avoid log(0))
            eps_log = 1e-12
            z0 = np.log(seed_vec + eps_log)

            # ----------------------------------------------------------
            # 2. Smooth‑max Adam optimisation
            # ----------------------------------------------------------
            a_adam = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=400.0,
                ramp_iters=50_000,
                lr=0.07,
            )

            # ----------------------------------------------------------
            # 3. Official evaluation of the Adam output
            # ----------------------------------------------------------
            try:
                val = evaluate_sequence(a_adam.tolist())
                print(f"   Official score after Adam = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_adam.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed (Adam result): {e}")

            # ----------------------------------------------------------
            # 4. KL‑mirror polishing (exponentiated‑gradient)
            # ----------------------------------------------------------
            if time.time() < deadline - 4.0:
                eg_deadline = deadline - 2.0
                a_eg = exponentiated_gradient_polish(
                    a_adam,
                    eg_deadline,
                    max_iter=70_000,
                    init_step=0.02,
                )
                try:
                    val_eg = evaluate_sequence(a_eg.tolist())
                    print(f"   Polished (EG) score = {val_eg:.6f}")
                    if val_eg < best_val:
                        best_val, best_seq = val_eg, a_eg.tolist()
                        print(f"   *** New best after EG polish: {best_val:.6f} ***")
                except Exception as e:
                    print(f"   EG polishing failed: {e}")

            # ----------------------------------------------------------
            # 5. Exact sub‑gradient polishing (short)
            # ----------------------------------------------------------
            if time.time() < deadline - 2.0:
                subgrad_deadline = deadline - 1.0
                a_sub = subgradient_descent_exact(
                    np.array(best_seq, dtype=float),
                    subgrad_deadline,
                    max_iter=60_000,
                    step0=0.07,
                )
                try:
                    val_sub = evaluate_sequence(a_sub.tolist())
                    print(f"   Sub‑grad polished score = {val_sub:.6f}")
                    if val_sub < best_val:
                        best_val, best_seq = val_sub, a_sub.tolist()
                        print(f"   *** New best after sub‑grad polish: {best_val:.6f} ***")
                except Exception as e:
                    print(f"   Sub‑gradient polishing failed: {e}")

        # --------------------------------------------------------------
        # 6. Random local walk around the current best for this length
        # --------------------------------------------------------------
        if best_seq and time.time() < deadline - 3.0:
            a_current = np.array(best_seq, dtype=float)
            walk_deadline = deadline - 2.0
            it = 0
            while time.time() < walk_deadline:
                it += 1
                a_candidate = random_local_perturb(a_current, rng, delta=0.005)
                cand_score = compute_score(a_candidate)
                if cand_score < compute_score(a_current):
                    # Accept immediately – we use the cheap surrogate for speed
                    a_current = a_candidate
                if it % 10_000 == 0:
                    # occasional official check
                    try:
                        val_off = evaluate_sequence(a_current.tolist())
                        if val_off < best_val:
                            best_val, best_seq = val_off, a_current.tolist()
                            print(f"   *** New best after random walk (official) = {best_val:.6f} ***")
                    except Exception:
                        pass

            # store whatever we have after the walk
            best_seq = a_current.tolist()

    # ------------------------------------------------------------------
    # 7. Final global polishing with a long sub‑gradient run (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=250_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 8. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 256
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
