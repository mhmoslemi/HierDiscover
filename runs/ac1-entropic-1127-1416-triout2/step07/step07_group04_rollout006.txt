"""Step‑function optimizer (search for low evaluate_sequence score).

The algorithm proceeds in three stages:

1. **Coarse parametric sweep** – on a modest grid (n≈1024) we generate
   several families of symmetric shapes (arcsine, Beta(α,α), power‑law,
   mixtures with uniform).  Each candidate is projected to the simplex
   (∑a=1) and its exact score `2·n·max(conv(a,a))` (equal to the official
   objective because the sum is 1) is computed with an FFT‑based convolution.
   The best few shapes are kept.

2. **High‑resolution refinement** – the selected shape is up‑sampled to a
   larger resolution (n = 4096 or 8192 depending on remaining time) and
   further optimised by Adam on a smooth‑max approximation of the
   max‑convolution.  The temperature τ grows from 0.5 to a large value
   (≈ 400) so the surrogate approaches the true maximum.

3. **Exact polishing** – a projected Euclidean sub‑gradient descent on the
   true max‑convolution is performed, followed by an exponentiated‑gradient
   (mirror‑descent) phase.  Both loops use a √t step‑size decay.
   The best sequence (its official score evaluated by ``evaluate_sequence``)
   is returned.  The whole routine respects a 1000 s wall‑clock budget.

The search is deterministic apart from an optional random jitter used to
generate a second candidate from the best coarse shape.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-6) -> np.ndarray:
    """Symmetric Beta(α,β) shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    if not np.all(np.isfinite(base)) or base.sum() <= 0.0:
        return np.full(n, 1.0 / n)
    return base / base.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-6) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{-p}, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    if not np.all(np.isfinite(vals)) or vals.sum() <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / vals.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points.
    Result is re‑normalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (continuous phase)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    tau_start: float = 0.5,
    tau_end: float = 200.0,
    base_lr: float = 0.5,
) -> np.ndarray:
    """
    Adam optimisation of the smooth approximation of max(conv).
    The optimisation variable is the logits w (softmax → a) which
    automatically satisfies a ≥ 0, Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)          # numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)     # length L

        # smooth max via log‑sum‑exp (scaled by τ)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp   # shape (L,)

        # gradient w.r.t. a : 2·conv(a, rev(weights))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)  # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output (quiet in the final run)
        if it % 50000 == 0:
            surr = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surr:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact (nonsmooth) polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.06,
) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.22,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:

    Search for a non‑negative step sequence with a low ``evaluate_sequence`` value.
    The routine runs a cheap parametric sweep on n≈1024, selects the best shape,

    exact polishing (sub‑gradient + exponentiated‑gradient).  All phases respect
    the global wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – coarse parametric sweep (n=1024)
    # ------------------------------------------------------------------
    n_coarse = 1024
    print(f"\n--- Stage 1: coarse sweep (n={n_coarse}) ---")
    candidates = []   # list of (score, sequence)

    # arcsine baseline
    seq = arcsine_initial(n_coarse)
    cand_score = compute_score(seq)
    candidates.append((cand_score, seq.copy()))

    # symmetric Beta shapes
    alphas = np.linspace(0.20, 1.50, 66)    # fine grid around the arcsine α=0.5
    for a in alphas:
        seq = generate_beta_shape(n_coarse, a, a, eps=1e-6)
        cand_score = compute_score(seq)
        candidates.append((cand_score, seq.copy()))

    # power‑law (generalised arcsine) shapes
    for p in np.arange(0.4, 2.01, 0.06):
        seq = generate_power_shape(n_coarse, p, eps=1e-6)
        cand_score = compute_score(seq)
        candidates.append((cand_score, seq.copy()))

    # mixture arcsine + uniform
    arc = arcsine_initial(n_coarse)
    uniform = np.full(n_coarse, 1.0 / n_coarse)
    for w in np.linspace(0.0, 1.0, 21):
        seq = w * arc + (1.0 - w) * uniform
        seq = seq / seq.sum()
        cand_score = compute_score(seq)
        candidates.append((cand_score, seq.copy()))

    # keep the best few (including a random jitter of the overall best)
    candidates.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [seq for _, seq in candidates[:top_k]]

    jittered = jittered / jittered.sum()
    top_candidates.append(jittered)

    # optional fallback from a previously discovered construction
    if "height_sequence_1" in globals():
        try:
            fallback = np.array(globals()["height_sequence_1"], dtype=float)
            if fallback.size > 0:
                fallback = np.maximum(fallback, 0.0)
                fallback = fallback / fallback.sum()
                top_candidates.append(fallback)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 2 – upsample & smooth‑max Adam refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 250:
        target_n = 8192
    elif remaining > 120:
        target_n = 4096
    else:
        target_n = 2048
    print(f"\n--- Stage 2: upsample to n={target_n} and Adam refine ---")

    best_high_seq = None
    best_high_val = float("inf")

    for idx, cand in enumerate(top_candidates):
        if time.time() >= deadline - 10:
            break   # keep a small buffer for polishing

        # upsample to the target resolution
        a = upsample(cand, target_n)

        # Adam phase – allocate iterations based on remaining wall‑clock
        remain = deadline - time.time()
        max_iters = int(min(250_000, remain * 30))
        a = adam_optimize_smooth_max(
            a,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=400.0,
            base_lr=0.6,
        )
        print(f"    Candidate {idx}: post‑Adam surrogate = {compute_score(a):.6f}")

        # ------------------------------------------------------------------
        # Stage 3 – exact polishing (sub‑gradient + EG)
        # ------------------------------------------------------------------
        remain = deadline - time.time()
        max_iter_sub = int(min(350_000, remain * 25))
        a = subgradient_descent_exact(a, deadline, max_iter=max_iter_sub, step0=0.06)

        remain = deadline - time.time()
        if remain > 4.0:
            max_iter_eg = int(min(250_000, remain * 25))
            a = eg_refine(a, deadline, max_iter=max_iter_eg, eta0=0.25)

        # official evaluation
        try:
            val = evaluate_sequence(a.tolist())
        except Exception:
            val = float("inf")
        print(f"Candidate {idx} final official score = {val:.6f}")

        if val < best_high_val:
            best_high_val = val
            best_high_seq = a.tolist()
            print("*** New best candidate in this run ***")

    # ------------------------------------------------------------------
    # Guard – ensure a non‑empty return
    # ------------------------------------------------------------------
    if best_high_seq is None:
        # fallback to a simple uniform sequence (guaranteed finite score)
        fallback_n = 1024
        best_high_seq = [1.0 / fallback_n] * fallback_n
        best_high_val = evaluate_sequence(best_high_seq)

    print(f"\nSearch finished. Best official score = {best_high_val:.6f}")
    return best_high_seq
