    # Triangular
    tri = np.arange(1, n//2+1)

        tri_full = np.concatenate([tri, tri[::-1]])

        tri_full = np.concatenate([tri, tri[-2::-1]]) # maybe adjust

    # Hann window (cosine)
    i = np.arange(n)
    hann = 0.5 * (1 - np.cos(2*np.pi*i/(n-1)))
    seqs.append(hann / np.sum(hann))

    # Blackman
    blackman = 0.42 - 0.5*np.cos(2*np.pi*i/(n-1)) + 0.08*np.cos(4*np.pi*i/(n-1))

def smooth_max_lbfgs(a0, time_deadline, eps_factor_start=0.5, eps_factor_end=0.02, eps_decay=0.9):
    # Map a0 to x via log(a0) maybe plus small offset to avoid zeros.
    # Since a0 may have zeros (window functions may have zeros at ends), we need to add epsilon small then renormalize.
    eps0 = 1e-12
    a0 = np.maximum(a0, eps0)
    x0 = np.log(a0)
    # maybe shift to zero-mean? not needed.
    best_a = None
    best_sc = np.inf

    # schedule of epsilon factors for smoothing; we will run L-BFGS for each.
    eps_factor = eps_factor_start
    while time.time() < time_deadline - 2.0 and eps_factor >= eps_factor_end:
        # define objective for current eps
        def fun_x(x):
            a = softmax(x)
            conv = convolution_fft(a, a)
            maxc = np.max(conv)
            eps = maxc * eps_factor
            # stable log-sum-exp
            # compute L = maxc + eps * log sum exp( (conv - maxc) / eps )
            # Equivalent to L = maxc + eps * logsumexp
            z = (conv - maxc) / eps
            maxz = np.max(z)  # can be zero
            # but maxz is 0 because conv-maxc <=0 so maxz =0
            # logsumexp = maxz + log(np.sum(np.exp(z-maxz))) but maxz=0.
            logsum = np.log(np.sum(np.exp(z)))
            smooth = maxc + eps * logsum
            # compute gradient
            exp_vals = np.exp(z)
            w = exp_vals / np.sum(exp_vals)
            # gradient wrt a
            w_rev = w[::-1]
            conv_rev = convolution_fft(a, w_rev)
            n = a.shape[0]
            idx_start = 2*n-2
            grad_a = 2.0 * conv_rev[idx_start - np.arange(n)]
            # gradient wrt x
            dot = np.dot(grad_a, a)
            grad_x = a * (grad_a - dot)
            return smooth, grad_x.astype(np.float64)
        # use scipy.optimize.minimize
        opt_res = scipy.optimize.minimize(
            fun_x,
            x0,
            jac=True,
            method='L-BFGS-B',
            options={'maxiter': 200, 'disp': False}
        )
        a_opt = softmax(opt_res.x)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"lbfgs eps_factor={eps_factor:.4f} iter={opt_res.nit} score={best_sc:.6f}")
        # Prepare for next eps (smaller)
        x0 = opt_res.x  # use result as start for next smoothing
        eps_factor *= eps_decay
        # check time
    return best_a, best_sc

"""
Multi‑phase optimizer for the self‑convolution peak problem.

The search proceeds as follows:

1️⃣  Deterministic “window” initialisations (uniform, triangular,
    Hann, Hamming, Blackman, Gaussian, Kaiser).  They already give
    scores substantially better than a plain uniform vector.

2️⃣  Refine the best window by minimising a smooth approximation of the
    max‑convolution:
        f_ε(a) = ε·log∑_k exp((a⊛a)_k/ε) .
    The vector is parametrised by a soft‑max `a = softmax(x)`,
    which automatically satisfies non‑negativity and ∑a=1.
    For a fixed ε we run L‑BFGS‑B on the exact gradient.
    ε is decreased geometrically (ε = ε_factor·max(conv)),
    turning the smooth surrogate into the true max.

3️⃣  A short sub‑gradient descent on the exact objective
    `2·n·max(a⊛a)` (the same objective used by the original baseline).

4️⃣  A light simulated‑annealing hill‑climber to escape possible
    remaining local minima.

5️⃣  Symmetrisation `(a + a[::-1])/2` – a cheap trick that often lowers
    the peak further.

All phases respect the 1000 s wall‑time budget and keep the best vector
found.  If SciPy is unavailable the code falls back to the previous
Adam‑based optimiser.  The routine returns a plain Python list ready for
the evaluation harness.
"""

# Fast FFT based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex handling (soft‑max parametrisation removes the need for projection)
# ----------------------------------------------------------------------


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – maps ℝⁿ to the probability simplex."""
    xmax = np.max(x)
    e = np.exp(x - xmax)
    return e / np.sum(e)

# ----------------------------------------------------------------------


def window_candidates(n: int) -> list[np.ndarray]:
    """Generate a modest list of classic window shapes, all normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    seqs = []

    # 1. Uniform
    seqs.append(np.full(n, 1.0 / n))

    # 2. Triangular (linear rise‑fall)
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=float)
        tri = np.concatenate([left, left[::-1]])
    else:
        left = np.arange(1, half + 2, dtype=float)          # includes centre peak
        tri = np.concatenate([left, left[-2::-1]])
    seqs.append(tri / np.sum(tri))

    # 3. Hann (cosine) window
    hann = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    seqs.append(hann / np.sum(hann))

    # 4. Hamming window
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    seqs.append(hamming / np.sum(hamming))

    # 5. Blackman window
    blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
                + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    seqs.append(blackman / np.sum(blackman))

    # 6. Gaussian windows (several σ values)
    centre = (n - 1) / 2.0
    for sigma_factor in (0.15, 0.20, 0.25, 0.30):
        sigma = sigma_factor * n
        gauss = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        seqs.append(gauss / np.sum(gauss))

    # 7. Kaiser windows (several β values) – requires SciPy's i0
    if _SCIPY_OK:
        for beta in (2.0, 5.0, 8.0, 12.0):
            ratio = (2.0 * i) / (n - 1) - 1.0
            kaiser = i0(beta * np.sqrt(1.0 - ratio ** 2))
            seqs.append(kaiser / np.sum(kaiser))

    return seqs


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) objective – L‑BFGS refinement
# ----------------------------------------------------------------------


def _smooth_max_obj(x: np.ndarray, args) -> tuple[float, np.ndarray]:
    """
    Objective + gradient for a given smoothing factor ε.
    `args` = (n, eps_factor) where eps = eps_factor * max(conv).
    The vector a is obtained via soft‑max, guaranteeing feasibility.
    Returns (value, grad_wrt_x).
    """
    n, eps_factor = args
    a = softmax(x)

    # convolution a⊛a
    conv = convolution_fft(a, a)
    maxc = np.max(conv)
    eps = maxc * eps_factor if eps_factor > 0 else 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    #   f = maxc + eps * log( Σ exp((conv - maxc)/eps) )
    z = (conv - maxc) / eps                 # ≤ 0, max(z)=0
    exp_z = np.exp(z)                       # safe, under‑flow gives 0
    w = exp_z / np.sum(exp_z)              # soft‑max weights, Σw = 1
    logsum = np.log(np.sum(exp_z))          # = log Σ exp(z)
    smooth = maxc + eps * logsum

    # ----- gradient wrt a -----
    # grad_a = 2 * (a * w) convolved (as derived in the baseline code)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)           # length 2n‑1
    idx_start = 2 * n - 2
    grad_a = 2.0 * conv_rev[idx_start - np.arange(n)]

    # ----- chain rule: ∂f/∂x = ∂f/∂a · ∂a/∂x  (soft‑max Jacobian) -----
    dot = np.dot(grad_a, a)               # scalar = a·grad_a
    grad_x = a * (grad_a - dot)           # element‑wise

    return float(smooth), grad_x.astype(np.float64)


def smooth_max_lbfgs(a0: np.ndarray,
                     time_deadline: float,
                     start_eps_factor: float = 0.5,
                     end_eps_factor: float = 0.02,
                     decay: float = 0.7,
                     maxiter_per_phase: int = 150) -> tuple[np.ndarray, float]:
    """
    Refine `a0` by repeatedly solving the smooth‑max problem with L‑BFGS
    while gradually decreasing the smoothing factor ε.
    Returns the best vector seen and its exact score.
    """
    if not _SCIPY_OK:    # pragma: no cover
        # SciPy not available – skip this phase
        return a0, compute_score(a0)

    n = a0.shape[0]
    # initialise the optimisation variables in the unconstrained space
    eps = 1e-12
    a0 = np.maximum(a0, eps)
    x = np.log(a0)                     # soft‑max inverse (up to an additive constant)

    best_a = a0.copy()
    best_sc = compute_score(best_a)

    eps_factor = start_eps_factor

    while time.time() < time_deadline - 2.0 and eps_factor >= end_eps_factor:
        # L‑BFGS on the smoothed objective
        res = scipy.optimize.minimize(
            fun=_smooth_max_obj,
            x0=x,
            args=(n, eps_factor),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter_per_phase, 'disp': False}
        )
        a_opt = softmax(res.x)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] eps_factor={eps_factor:.4f} it={res.nit} score={best_sc:.6f}")

        # prepare for the next (tighter) smoothing step
        x = res.x
        eps_factor *= decay

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on f(a) = 2·n·max(a⊛a).
    Uses the exact sub‑gradient at the current maximal index.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k-i} (zero outside the support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        # re‑project onto the simplex (soft‑max not used here)
        a = np.maximum(a, 0.0)
        a_sum = np.sum(a)
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[sub‑grad] iter={t} score={best_sc:.6f}")

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = np.maximum(a + jitter, 0.0)
            a = a / np.sum(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Light simulated‑annealing hill‑climber (final polishing)
# ----------------------------------------------------------------------


def hill_climb_anneal(start_seq,
                      time_deadline: float,
                      max_steps: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber with a linear cooling schedule.
    Moves are either swaps or small mass transfers between two indices.
    """
    a = np.array(start_seq, copy=True)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    T0 = 0.12  # initial temperature

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]            # move 5 % of source mass
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        # re‑normalise (still on the simplex)
        cand_sum = np.sum(cand)
        if cand_sum == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / cand_sum

        cur_sc = compute_score(cand)
        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional random refresh to keep diversity
        if step % 5000 == 0:
            noise = np.random.randn(n) * 0.02
            cand = best_a + noise
            cand = np.maximum(cand, 0.0)
            cand = cand / np.sum(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (cheap final tweak)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + reversed(a))/2 and renormalise."""
    a_sym = (a + a[::-1]) * 0.5
    s = np.sum(a_sym)
    if s == 0.0:
        return np.full_like(a_sym, 1.0 / a_sym.shape[0])
    return a_sym / s


# ----------------------------------------------------------------------
# Main entry point – the harness will call this function
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the 1000 s budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback – the previously discovered construction, if present
    # ------------------------------------------------------------------
    try:                  # pragma: no cover
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – try deterministic window families on a few lengths
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    # we will keep at most three best candidates for the expensive refinement
    candidate_pool = []

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break

        for win in window_candidates(n):
            sc = compute_score(win)
            if sc < best_score:
                best_score = sc
                best_seq = win.tolist()
                print(f"[init] n={n} win_score={sc:.6f}")

            # keep a short list of promising vectors (store the raw ndarray)
            candidate_pool.append((sc, win.copy()))
        # keep only the top 5 by score to limit later work
        candidate_pool.sort(key=lambda tup: tup[0])
        candidate_pool = candidate_pool[:5]

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS smooth‑max refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if _SCIPY_OK:
        # allocate roughly half the remaining budget to this phase
        phase_deadline = time.time() + (deadline - time.time()) * 0.55
        for sc0, cand in candidate_pool:
            if time.time() >= phase_deadline:
                break
            a_ref, sc_ref = smooth_max_lbfgs(
                cand,
                time_deadline=phase_deadline,
                start_eps_factor=0.5,
                end_eps_factor=0.02,
                decay=0.7,
                maxiter_per_phase=180,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.tolist()
                print(f"[L‑BFGS] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent on the true max‑objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float) if best_seq is not None else np.full(256, 1.0 / 256)
        a_sub, sc_sub = subgrad_optimize(
            a_start,
            time_deadline=deadline - 4.0,
            max_iter=120_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[sub‑grad] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        a_start = np.array(best_seq, dtype=float)
        a_ann, sc_ann = hill_climb_anneal(
            a_start,
            time_deadline=deadline - 2.0,
            max_steps=100_000,
        )
        if sc_ann < best_score:
            best_score = sc_ann
            best_seq = a_ann.tolist()
            print(f"[anneal] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – symmetry averaging (tiny cheap improvement)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[symmetry] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
