def build_conv_constraint_matrix(a):
    n = a.size
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        for j in range(start, end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    return A_ub

def lp_iterative_optimize(a_init, max_iters, alpha=0.5, time_deadline=None):
    a = a_init.copy()
    n = a.size
    M = np.max(convolution_fft(a, a))
    target_norm = np.sqrt(2.0 * n)
    if np.abs(a.sum() - target_norm) > 1e-12:
        a = a * (target_norm / a.sum())
    for it in range(max_iters):
        if time_deadline is not None and time.time() >= time_deadline:
            break
        A_ub = build_conv_constraint_matrix(a)   # dynamic because a changes each iteration
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': 10})

            break
        g0 = res.x

        # rescale g to match target_norm

"""
Propose a non‑negative step sequence that minimizes ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Linear‑programming refinement on a modest size** –  
   Starting from an arcsine shape on *n_small* (default 256) we repeatedly
   solve a linear program  

       max  Σ g_i          (i.e. minimize ‖f*f‖∞)
       s.t. (f*g)_k ≤ M   for all k,
            g_i ≥ 0,

   where *M* = max(conv(f,f)).  The LP yields a vector *g* that pushes the
   L¹‑norm as high as possible while respecting the current worst‑case
   convolution bound.  Mixing the old vector with the scaled optimiser
   reduces the bound; we iterate until no further improvement or the time
   budget expires.

2. **Upscaling and smooth surrogate optimisation** –  
   The refined small‑scale shape is interpolated to a larger length
   (*n_large* = 2048).  Four to six diverse seeds (including the upscaled
   shape, an arcsine shape and a few random Beta/Power shapes) are fed to
   an Adam optimiser that minimises a smooth log‑sum‑exp approximation of
   ``max(conv)``.  The temperature β is raised from 5 to 250 during the
   run, turning the surrogate gradually into the true max.

3. **Polishing** –  
   The best candidate from the Adam phase is refined by
   KL‑mirror (exponentiated‑gradient) descent directly on the non‑smooth
   objective, followed by a long projected sub‑gradient descent pass.
   Throughout the run we keep track of the best official score obtained
   from ``evaluate_sequence`` and finally return that sequence.

The whole routine respects a 1000 s wall‑clock budget, always returns a
non‑empty sequence and never calls ``evaluate_sequence`` with an invalid
input.  It builds on the previous code base but introduces a fundamentally
different LP‑based refinement step, which in practice yields a lower bound
than the earlier approach.
"""

import time

import scipy.optimize

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (no circular wrap).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective when Σa = 1:
        score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to Σa = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid, normalised to Σa = 1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope), normalised to Σa = 1."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform, normalised to Σa = 1."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

# Projection utilities for the simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax that maps ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (surrogate of max(conv))
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 250.0,
    ramp_iters: int = 60000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    The iterates are kept on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                               # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # raise temperature linearly
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2 * (a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final vector on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 30000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Mirror descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a

def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a

def build_conv_constraint_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the sparse matrix C such that (C @ g)[k] = (a * g)[k] for all k.
    This matrix has shape (2·n‑1, n) and contains the coefficients a_{k‑j}.
    """
    n = a.size
    rows = []
    cols = []
    data = []

    for k in range(2 * n - 1):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        for j in range(start, end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])

    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_g(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ g_i   s.t.   (a * g)_k ≤ M   for all k,   g_i ≥ 0.
    Returns the optimal g (or a zero vector if the LP fails).
    """
    n = a.size
    A_ub = build_conv_constraint_matrix(a)
    c = -np.ones(n)                     # maximise Σg_i  → minimise -Σg_i
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None) for _ in range(n)]

    res = scipy.optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    else:
        # fall back to the zero vector – it is always feasible
        return np.zeros_like(a)


def lp_iterative_optimize(
    a_init: np.ndarray,
    max_iters: int,
    deadline: float,
) -> (np.ndarray, float):
    """
    Repeatedly apply the LP step and mix the result with the current vector.
    The vector is kept at the normalisation Σa = √(2·n) during the iterations.
    Returns the final (Σ=1‑normalised) vector and the final max‑convolution value.
    """
    a = a_init.copy()
    n = a.size
    target_norm = np.sqrt(2.0 * n)

    # bring a to the required ℓ₁‑norm
    a = a * (target_norm / a.sum())
    M = np.max(convolution_fft(a, a))

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        g0 = solve_lp_for_g(a, M)
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break

        # rescale g so that it has the same ℓ₁‑norm as a
        g = g0 * (target_norm / sum_g0)

        improved = False
        # try a few mixing coefficients, decreasing geometrically
        for t in [0.5, 0.25, 0.125, 0.0625, 0.03125]:
            a_new = (1.0 - t) * a + t * g
            M_new = np.max(convolution_fft(a_new, a_new))
            if M_new < M - 1e-12:
                a = a_new
                M = M_new
                improved = True
                break

        if not improved:
            break

    # final normalisation to Σ=1 (the official evaluator is scale‑invariant)
    a = a / a.sum()
    return a, M / (target_norm ** 2)   # convert M back to the Σ=1 scale


# ----------------------------------------------------------------------
# Upscaling utilities
# ----------------------------------------------------------------------


def upscale_shape(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation from a small shape to a larger length.
    The result is normalised to Σ=1.
    """
    x_small = np.linspace(0.0, 1.0, len(a_small))
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    return a_large

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The function respects the 1000 s wall‑clock budget and always returns
    a non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load a fallback sequence if one is available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  LP‑based refinement on a modest size (256)
    # ------------------------------------------------------------------
    n_small = 256
    print(f"\n--- LP refinement on n={n_small} ---")
    lp_deadline = min(deadline - 150.0, time.time() + 120.0)   # allocate ≈2 min
    a_small_start = arcsine_initial(n_small)   # Σ=1
    a_small_opt, _ = lp_iterative_optimize(
        a_small_start,
        max_iters=50,
        deadline=lp_deadline,
    )
    print("LP refinement finished.")

    # ------------------------------------------------------------------
    # 2.  Upscale to a larger working length and build a seed pool
    # ------------------------------------------------------------------
    n_large = 2048
    a_upscaled = upscale_shape(a_small_opt, n_large)

    seeds: List[np.ndarray] = [a_upscaled, arcsine_initial(n_large)]

    # random power‑law seeds
    for _ in range(4):
        p = rng.uniform(0.1, 4.0)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_power_shape(n_large, p, eps))

    # random beta‑shape seeds
    for _ in range(4):
        a_par = rng.uniform(0.1, 5.0)
        b_par = rng.uniform(0.1, 5.0)
        eps = rng.uniform(0.0, 0.5)
        seeds.append(generate_beta_shape(n_large, a_par, b_par, eps))

    # mixture seeds
    for _ in range(2):
        w = rng.uniform(0.0, 1.0)
        seeds.append(generate_mixture_shape(n_large, w))

    # include the current global best (if any) after resizing
    if best_seq:
        old = np.array(best_seq, dtype=float)
        if old.size != n_large:
            old = upscale_shape(old / old.sum(), n_large)
        seeds.append(old)

    # ------------------------------------------------------------------
    # 3.  Keep the most promising seeds (fast surrogate ranking)
    # ------------------------------------------------------------------
    scored = [(compute_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:5]]

    # ------------------------------------------------------------------
    # 4.  Optimize each seed with Adam → EG → Sub‑gradient polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    per_seed_time = max(10.0, remaining / (len(top_seeds) * 1.2))

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 8.0:
            break

        print(f"\n=== Optimising seed {idx}/{len(top_seeds)} (n={n_large}) ===")
        seed_deadline = min(deadline - 4.0, time.time() + per_seed_time)

        # ---- Adam on smooth surrogate ----
        z0 = np.log(seed_vec + 1e-12)
        a_opt = smooth_adam_optimize(
            z0,
            seed_deadline,
            n=n_large,
            beta0=5.0,
            beta_max=250.0,
            ramp_iters=60000,
            lr=0.07,
        )

        # ---- Official evaluation ----
        try:
            val = evaluate_sequence(a_opt.tolist())
            print(f"   Official score after Adam = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"   *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

        # ---- EG polishing (if time permits) ----
        if time.time() < deadline - 4.0:
            eg_deadline = deadline - 2.0
            a_eg = exponentiated_gradient_polish(
                np.array(best_seq, dtype=float),
                eg_deadline,
                max_iter=30000,
                step=0.015,
            )
            try:
                val_eg = evaluate_sequence(a_eg.tolist())
                print(f"   EG polished score = {val_eg:.6f}")
                if val_eg < best_val:
                    best_val, best_seq = val_eg, a_eg.tolist()
                    print(f"   *** New best after EG: {best_val:.6f} ***")
            except Exception as e:
                print(f"   EG polishing failed: {e}")

    # ------------------------------------------------------------------
    # 5.  Final global sub‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=250_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
