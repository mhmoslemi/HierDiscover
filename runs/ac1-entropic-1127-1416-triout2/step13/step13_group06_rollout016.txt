Propose an improved step‑function sequence.

Algorithm overview
------------------
1. **Initial scan of analytic families** (including new super‑Gaussian and
   cosine‑bump families) on a medium resolution (n = 8192).  The best
   candidate from this sweep is kept as a warm start.
2. **Random dense exploration** – ~12 000 random draws from the same
   families plus a few ad‑hoc transformations (power‑warp, uniform‑+‑Gaussian
   mixtures) with occasional zero‑padding.
3. **Local golden‑section refinement** of the best 1‑parameter family.
4. **Multi‑restart exponentiated‑gradient (mirror descent)** on the exact
   max‑convolution objective.  Three starts are used (best‑so‑far,
   uniform, and a random perturbation).  The step‑size schedule is
   ηₜ = η₀/(1+0.001 t) to allow a slower decay and finer convergence.
5. **Power‑warp post‑processing** – the current best vector is raised to
   several exponents (0.5 … 2.0) and re‑normalised; any improvement is kept.
6. **Final simulated‑annealing hill‑climb** with a high initial temperature
   and a linear cooling schedule (up to 12 M proposals).  Moves consist of
   swaps or fractional mass transfers; the acceptance rule follows the
   Metropolis criterion.
7. **Optional up‑sampling** – if time permits, the best 8192‑vector is
   linearly interpolated to 16384 entries and a short SA run is performed
   on the larger problem (this can reduce the objective further).
8. The final candidate is verified with the official ``evaluate_sequence``
   call and returned as a plain Python list.

All stages respect the global time budget (default 1000 s).  Progress

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (n‑½‑i)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return normalize(w)


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return normalize(w)


def super_gaussian_center_family(n: int, sigma: float, power: float) -> np.ndarray:
    """Super‑Gaussian (exp(−|x|^p)) centred on the middle."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * (np.abs(i - mu) / sigma) ** power)
    return normalize(w)


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump."""
    half_width = int(max(1, width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return normalize(w)


def cosine_bump_center_family(n: int, half_width_frac: float) -> np.ndarray:
    """Cosine bump (raised‑cosine) centred on the middle."""
    half_width = max(1, int(half_width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    d = np.abs(i - center)
    w = np.where(d <= half_width,
                 (np.cos(np.pi * d / (2.0 * half_width))) ** 2,
                 0.0)
    return normalize(w)


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    """Exponential decay from one side."""
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return normalize(w)


def mixed_uniform_gaussian_family(n: int, lam: float, sigma: float) -> np.ndarray:
    """Convex mixture of uniform and centred Gaussian."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    gaussian = gaussian_center_family(n, sigma)
    w = lam * uniform + (1.0 - lam) * gaussian
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return normalize(w)


def power_warp_family(base_vec: np.ndarray, power: float) -> np.ndarray:
    """Raise a base vector to a power and renormalise."""
    w = np.power(base_vec, power)
    return normalize(w)

def exp_gradient_descent(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 0.6) -> Tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using exponentiated gradients.
    Returns the best vector seen and its score.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # compute full convolution, locate max index
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step‑size schedule (slightly slower decay)
        eta = eta0 / (1.0 + 0.001 * t)
        a = a * np.exp(-eta * g)
        # renormalise (KL‑projection is just scaling)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] exp‑grad (iter {t}) improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 12_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        # two move types: swap or transfer
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 12) -> Tuple[float, float]:
    """Golden‑section search for the minimum of func on [lo,hi]."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

        elif choice == 2:     # super‑Gaussian

            power = np.random.choice([2.0, 4.0, 6.0])
            vec = super_gaussian_center_family(n, sigma, power)
        elif choice == 3:     # cosine bump
            half_w = np.random.uniform(0.05, 0.30)
            vec = cosine_bump_center_family(n, half_w)
        elif choice == 4:     # triangular
            width = np.random.uniform(0.1, 0.9)
            vec = triangular_center_family(n, width)
        elif choice == 5:     # left‑heavy power
            beta = np.random.uniform(0.0, 12.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 6:     # right‑heavy power
            beta = np.random.uniform(0.0, 12.0)
            vec = right_heavy_power_family(n, beta)
        elif choice == 7:     # product (U‑shaped)
            alpha = np.random.uniform(0.0, 12.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 8:     # exponential decay left
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n, gamma, left=True)
        elif choice == 9:     # exponential decay right
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n, gamma, left=False)
        else:                 # mixture uniform + Gaussian
            lam = np.random.uniform(0.0, 1.0)
            sigma = np.random.uniform(0.2 * n, 0.5 * n)
            vec = mixed_uniform_gaussian_family(n, lam, sigma)

        # occasional random zero‑padding (breaks symmetry)
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Local golden‑section refinement of the current best family
    # -----------------------------------------------------------
    if best_family is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining best family {best_family}")

        if best_family == "center_power":
            cur_alpha = best_params["alpha"]
            lo, hi = max(0.0, cur_alpha - 0.8), min(12.0, cur_alpha + 0.8)

            def f_alpha(a):
                return compute_score(symmetric_center_power_family(n, a))

            new_alpha, _ = golden_section_optimize(f_alpha, lo, hi, max_iter=14)
            vec = symmetric_center_power_family(n, new_alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_alpha
                print(f"[{time.time():.1f}] refined centre‑α={new_alpha:.4f} → {best_score:.6f}")

        elif best_family == "gaussian":
            cur_sigma = best_params["sigma"]
            lo, hi = max(0.2 * n, cur_sigma - 0.2 * n), min(0.5 * n, cur_sigma + 0.2 * n)

            def f_sigma(s):
                return compute_score(gaussian_center_family(n, s))

            new_sigma, _ = golden_section_optimize(f_sigma, lo, hi, max_iter=14)
            vec = gaussian_center_family(n, new_sigma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["sigma"] = new_sigma
                print(f"[{time.time():.1f}] refined gauss‑σ={new_sigma:.1f} → {best_score:.6f}")

        elif best_family == "super_gauss":
            cur_sigma = best_params["sigma"]
            cur_power = best_params["power"]
            lo_s, hi_s = max(0.1 * n, cur_sigma - 0.1 * n), min(0.5 * n, cur_sigma + 0.1 * n)

            def f_sigma(s):
                return compute_score(super_gaussian_center_family(n, s, cur_power))

            new_sigma, _ = golden_section_optimize(f_sigma, lo_s, hi_s, max_iter=12)
            vec = super_gaussian_center_family(n, new_sigma, cur_power)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["sigma"] = new_sigma
                print(f"[{time.time():.1f}] refined super‑gauss σ={new_sigma:.1f} (p={cur_power}) → {best_score:.6f}")

        elif best_family == "cosine_bump":
            cur_w = best_params["half_width_frac"]
            lo, hi = max(0.03, cur_w - 0.15), min(0.5, cur_w + 0.15)

            def f_w(w):
                return compute_score(cosine_bump_center_family(n, w))

            new_w, _ = golden_section_optimize(f_w, lo, hi, max_iter=12)
            vec = cosine_bump_center_family(n, new_w)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["half_width_frac"] = new_w
                print(f"[{time.time():.1f}] refined cosine‑w={new_w:.4f} → {best_score:.6f}")

        elif best_family == "product":
            cur_alpha = best_params["alpha"]
            lo, hi = max(0.0, cur_alpha - 0.6), min(12.0, cur_alpha + 0.6)

            def f_alpha(a):
                return compute_score(symmetric_product_family(n, a))

            new_alpha, _ = golden_section_optimize(f_alpha, lo, hi, max_iter=14)
            vec = symmetric_product_family(n, new_alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_alpha
                print(f"[{time.time():.1f}] refined product‑α={new_alpha:.4f} → {best_score:.6f}")

        elif best_family == "beta":
            p0, q0 = best_params["p"], best_params["q"]
            for _ in range(120):
                if time.time() >= deadline - 5.0:
                    break
                p = np.clip(p0 + np.random.randn() * 0.2, 0.1, 5.0)
                q = np.clip(q0 + np.random.randn() * 0.2, 0.1, 5.0)
                vec = beta_family(n, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    best_params["p"], best_params["q"] = p, q
                    print(f"[{time.time():.1f}] refined beta p={p:.3f},q={q:.3f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Exponentiated‑gradient descent – multiple restarts
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] exponentiated‑gradient optimisation (multiple starts)")
        starts = []

        # start 1: current best
        starts.append(np.array(best_seq, dtype=np.float64))

        # start 2: uniform
        starts.append(np.full(n, 1.0 / n, dtype=np.float64))

        # start 3: a random perturbation of the best (add small Gaussian noise)
        rnd = normalize(np.array(best_seq, dtype=np.float64) *
                        np.exp(0.05 * np.random.randn(n)))
        starts.append(rnd)

        for idx, a0 in enumerate(starts, 1):
            if time.time() >= deadline - 150.0:
                break
            print(f"[{time.time():.1f}]  exp‑grad restart #{idx}")
            a_opt, sc_opt = exp_gradient_descent(
                start_a=a0,
                deadline=deadline - 140.0,
                max_iters=120_000,
                eta0=0.6,
            )
            if sc_opt < best_score:
                best_score, best_seq = sc_opt, a_opt.tolist()
                print(f"[{time.time():.1f}]  after exp‑grad restart #{idx} best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Power‑warp post‑processing
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] power‑warp refinements")
        base = np.array(best_seq, dtype=np.float64)
        for pw in [0.5, 0.7, 0.8, 1.2, 1.5, 1.8, 2.0]:
            if time.time() >= deadline - 10.0:
                break
            vec = power_warp_family(base, pw)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] power‑warp p={pw:.2f} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Final SA hill‑climb (high temperature)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final simulated‑annealing hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=12_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Optional up‑sampling to a larger problem
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        # Try a quick up‑sampling to 16384 and a short SA run if time permits
        n_big = 16384
        try:
            print(f"[{time.time():.1f}] up‑sampling to n={n_big} for a final tweak")
            small = np.array(best_seq, dtype=np.float64)
            x_small = np.linspace(0.0, 1.0, n)
            x_big = np.linspace(0.0, 1.0, n_big)
            big = np.interp(x_big, x_small, small)
            big = normalize(big)
            # very short SA (2 M steps) on the larger vector
            a_big, sc_big = hill_climb_sa(
                start_a=big,
                deadline=deadline,
                max_steps=2_000_000,
                temp_factor=1.5e-3,
                step_scale_init=0.30,
            )
            if sc_big < best_score:
                best_score, best_seq = sc_big, a_big.tolist()
                print(f"[{time.time():.1f}] up‑sample SA improved best to {best_score:.6f}")
        except Exception as e:
            print(f"[{time.time():.1f}] up‑sampling failed: {e}")

    # -----------------------------------------------------------
    # 9. Fallback to a safe uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------
    # Final official evaluation (includes all security checks)
    # -----------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
