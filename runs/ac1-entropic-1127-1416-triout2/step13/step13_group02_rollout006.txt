def smooth_adam(a0, deadline, epsilon=1e-4, lr=0.03, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(a0)
    u = np.log(a0 + 1e-20)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    best_a = a0.copy()
    best_val = compute_score(a0)
    t=0
    while time.time() < deadline:
        t+=1
        # compute a from u softmax
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        a = exp_u / np.sum(exp_u)

        # compute conv and weighted terms
        conv = full_convolution(a)   # shape 2n-1
        maxc = conv.max()
        # compute weights for smoothing
        # to avoid overflow, shift by maxc
        shift = conv - maxc
        w = np.exp(shift / epsilon)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w = w / w_sum

        # gradient w.r.t a
        a_rev = a[::-1]
        conv_w_rev = full_convolution_general(w, a_rev)
        g_a = 2.0 * conv_w_rev[n-1 : n-1 + n]

        # gradient w.r.t u
        g_u = a * (g_a - np.dot(g_a, a))

        # Adam update
        m = beta1 * m + (1 - beta1) * g_u
        v = beta2 * v + (1 - beta2) * (g_u * g_u)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep best
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
    return best_a
```

"""
Non‑negative step‑function optimiser.

The routine proceeds in three stages:

1. **Global exploration** – builds a diverse pool of candidate
   distributions using several parametric families (power‑law,
   exponential, beta, triangular, sinusoid, semi‑circle) and a large
   set of Dirichlet draws.  The ten best vectors (by the exact objective
   2·n·max(conv)) are retained.

2. **Smoothed Adam polishing** – each retained vector is refined by
   minimising a smooth approximation of the max‑convolution:
       f_ε(a) = ε·log Σ_k exp(conv_k(a)/ε) .
   The gradient of f_ε is obtained analytically via FFT‑based
   convolution, and the variables are updated in the soft‑max
   parametrisation a = softmax(u) using an Adam optimiser.
   The temperature ε is slowly annealed, allowing the algorithm to
   escape plateaus and converge to a very low value of the original
   max‑convolution.

3. **Final annealing** – a short Metropolis‑style simulated‑annealing
   pass is applied to the best vector to try to break out of any
   remaining local minima.

The whole search respects a 1000 s wall‑clock budget and returns the
best sequence found as a plain list of floats suitable for the official
``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, length‑power‑of‑two for speed)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using a real FFT.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary vectors using a real FFT.
    Returns a vector of length len(x)+len(y)-1.
    """
    m = len(x) + len(y) - 1
    n_fft = _next_pow2(m)
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_triangular(n: int) -> np.ndarray:
    """Linear rise then fall (pyramid)."""
    a = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    total = a.sum()
    return a / total


def generate_sine(n: int) -> np.ndarray:
    """Half‑period sine, zero at the ends."""
    i = np.arange(n, dtype=float) + 0.5
    a = np.sin(np.pi * i / n)
    total = a.sum()
    return a / total


def generate_semicircle(n: int) -> np.ndarray:
    """Discrete semicircle (Wigner)."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.sqrt(np.maximum(0.0, 1.0 - (2 * x - 1) ** 2))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global baseline exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """
    Sweep a collection of families and a large pool of Dirichlet draws,
    retaining the ``max_keep`` best vectors (by the exact objective).
    """
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.05, 6.0, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(1e-5, 0.25, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep
    al_vals = np.concatenate([np.geomspace(0.01, 0.3, 12),
                              np.linspace(0.3, 4.0, 24)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.3, 12),
                              np.linspace(0.3, 4.0, 24)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) specialised shapes
    for gen in (generate_triangular, generate_sine, generate_semicircle):
        if time.time() >= deadline:
            break
        try_candidate(gen(n))

    # 5) massive Dirichlet draws (different concentration values)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = 1400   # ≈ 11 200 draws – still fast
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def smooth_adam(a0: np.ndarray, deadline: float,
                init_lr: float = 0.05,
                epsilon0: float = 2e-3,
                epsilon_min: float = 5e-7,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12) -> np.ndarray:
    """
    Adam optimiser for the smoothed objective
        f_ε(a) = ε·log Σ_k exp(conv_k/ε).

    The variables are stored as u with a = softmax(u).  The function
    returns the best vector seen before the deadline.
    """
    n = a0.shape[0]
    # initialise in log‑space (avoid log(0))
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float('inf')
    t = 0

    while time.time() < deadline:
        t += 1

        # ---- soft‑max projection ----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ---- convolution of a with itself ----
        conv = full_convolution(a)
        max_c = conv.max()

        # ---- temperature schedule (annealing) ----
        epsilon = max(epsilon_min, epsilon0 * (0.9995 ** t))

        # ---- smoothed weights ----
        shift = conv - max_c            # max term becomes 0
        w = np.exp(shift / epsilon)    # safe: max exponent == 0 → w≤1
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ---- gradient wrt a : g_a = 2·(w ⋆ a) ----
        a_rev = a[::-1]
        conv_wr = full_convolution_general(w, a_rev)
        g_a = 2.0 * conv_wr[n - 1: n - 1 + n]   # slice length n

        # ---- gradient wrt u (soft‑max Jacobian) ----
        dot = np.dot(g_a, a)
        g_u = a * (g_a - dot)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = init_lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ---- keep best (exact) objective seen ----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a if best_a is not None else a0


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """
    Metropolis hill‑climb that may accept worsening moves.
    Used only as a short final polishing step.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of a few seconds for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 offers a good trade‑off between expressivity
    # and runtime.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline - 300.0, start_time + 200.0)
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smoothed‑Adam polishing (primary + random restarts)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # primary polish
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_adam(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑adam‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # random‑restart polishing while time permits
        while time.time() < polish_deadline:
            conc = np.random.choice([0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_adam(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑adam‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback to uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
