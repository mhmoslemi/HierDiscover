def shape_power_product(n, p, delta):
    idx = np.arange(n, dtype=np.float64)
    a = ((idx + delta) * (n - 1 - idx + delta)) ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        return None
    a /= a_sum
    return a

    cur = a_start.copy()
    cur_score = compute_score(cur)
    best_seq = cur.copy()
    best_score = cur_score

    temp = max(1e-4 * best_score, 1e-6)

    last_improve_it = 0

        j = np.random.randint(n-1)
        if j >= i:

        if delta <= 0:

"""
Propose a coefficient sequence that minimises
evaluate_sequence(seq) = 2·n·max(conv(seq,seq))/sum(seq)^2.

The search consists of four stages:

1.  **Rich initialisation** – a handful of analytically‑motivated families
    (beta U‑shape, a symmetric “power‑product’’ shape, uniform, geometric,
    mixed plateau) are sampled and the best is kept.

2.  **Focused parametric sweep** – the two‑parameter family
        a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-p}
    (symmetric, edge‑heavy but regularised by δ) is explored on a fine
    grid (p≈0.35…0.85, δ≈0.001…3).  Small mixtures with a uniform vector are
    also examined.  The current best vector is updated whenever a better
    score (2·n·max(conv)) is found.

3.  **Sub‑gradient polishing** – a projected sub‑gradient descent on
    the non‑smooth objective (max of the convolution) is performed for a
    relatively long period (few hundred seconds).  After every 2000
    iterations a tiny random jitter is added to escape shallow plateaux.

4.  **Stochastic annealing + deterministic hill‑climb** – a simulated‑annealing
    mass‑transfer loop (move a small fraction of mass from one coordinate to
    another) accepts occasional deteriorations according to a decaying
    temperature schedule.  When no improvement is seen for many iterations
    a random jitter reset is performed.  Finally a greedy hill‑climb
    (deterministic greedy moves) polishes the solution.

All vectors are kept normalised to sum = 1, which makes the objective
scale‑invariant and allows us to use the fast FFT‑based convolution
(`full_convolution`).  The whole routine respects the 1000 s budget and
returns a plain Python list of non‑negative floats.
"""

import time

# Helper utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def uniform_vec(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{alpha} (1‑x_i)^{beta},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}, 0<q<1."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if not (0 < L < n):
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def shape_power_product(n: int, p: float, delta: float) -> np.ndarray:
    """
    Symmetric edge‑heavy shape:
        a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-p}
    delta > 0 avoids division by zero.
    """
    idx = np.arange(n, dtype=np.float64)
    prod = (idx + delta) * (n - 1 - idx + delta)
    # protect against overflow / under‑flow
    with np.errstate(divide='ignore'):
        a = prod ** (-p)
    a[np.isinf(a)] = 0.0
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mix_with_uniform(shape: np.ndarray, w: float) -> np.ndarray:
    """Convex combination w·shape + (1‑w)·uniform, followed by renormalisation."""
    if w >= 1.0:
        return shape
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n)
    a = w * shape + (1.0 - w) * uniform
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Stage 2 – focussed parametric sweep
# ----------------------------------------------------------------------
def parametric_sweep_shape(n: int, deadline: float,
                          best_vec: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """
    Fine‑grained search over the (p,δ) two‑parameter family,
    also trying a few convex mixes with the uniform vector.
    """
    # grids (chosen to respect typical running time)
    p_vals = np.arange(0.35, 0.86, 0.02)          # 0.35 … 0.85
    delta_vals = np.concatenate((
        np.arange(0.001, 0.101, 0.005),          # fine near zero
        np.arange(0.12, 1.01, 0.08),             # medium
        np.arange(1.2, 3.01, 0.4)                # larger offsets
    ))
    mix_weights = [0.6, 0.8, 1.0]                # 1.0 → pure shape

    for p in p_vals:
        for delta in delta_vals:
            if time.time() >= deadline:
                return best_vec, best_val
            shape = shape_power_product(n, p, delta)
            if shape is None:
                continue
            for w in mix_weights:
                a = mix_with_uniform(shape, w)
                val = compute_score(a)
                if val < best_val:
                    best_val, best_vec = val, a
    return best_vec, best_val


# ----------------------------------------------------------------------
# Stage 3 – projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # indices where the max is attained (or almost attained)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of each active entry: 2 * a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape flat regions
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stage 4 – simulated annealing with restarts
# ----------------------------------------------------------------------
def anneal_search(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing style stochastic optimisation.
    Moves a small amount of mass from one coordinate to another,
    accepts occasional deteriorations, and restarts after stagnation.
    """
    n = a_start.shape[0]
    cur = a_start.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # temperature proportional to the current score
    temp = max(1e-5 * best_score, 1e-8)
    temp_decay = 0.9999

    step_scale = 0.07               # typical fraction of source mass to move
    step_decay = 0.99995

    it = 0
    last_improve = 0
    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * cur[i] * np.random.rand()
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            accept = (np.random.rand() < prob)

        if accept:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                last_improve = it
                step_scale = min(step_scale * 1.02, 0.6)

        # decay temperature & step size
        temp = max(temp * temp_decay, 1e-12)
        step_scale = max(step_scale * step_decay, 1e-7)

        # restart with a small random jitter if stagnated too long
        if it - last_improve > 250_000:
            jitter = np.random.randn(n) * 0.002
            cur = best + jitter
            cur = project_to_simplex(cur)
            cur_score = compute_score(cur)
            last_improve = it

    return best


# ----------------------------------------------------------------------
# Stage 5 – deterministic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy greedy mass‑transfer moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous best construction, if available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Generate a diverse pool of starting vectors (≈ 50 s)
    # -----------------------------------------------------------------
    n = 4096
    pool = []

    # analytic families
    pool.append(uniform_vec(n))
    pool.append(beta_shape(n, -0.5, -0.5))          # arcsine‑like U‑shape
    pool.append(power_law(n, 0.5))
    pool.append(geometric(n, 0.99))
    pool.append(mixed_plateau_power(n, L_frac=0.1, p=0.6))

    # random Dirichlet samples
    for _ in range(200):
        conc = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        y = np.random.gamma(conc, 1.0, size=n)
        if y.sum() > 0:
            pool.append(y / y.sum())

    # keep the best from the pool
    for vec in pool:
        if vec is None:
            continue
        val = compute_score(vec)
        if val < best_val:
            best_val, best_seq = val, vec.tolist()
    print(f"[initial pool] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Focused parametric sweep over the (p,δ) family
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 500.0
    if best_seq is not None and time.time() < sweep_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_best, val_best = parametric_sweep_shape(
            n, sweep_deadline, a0, compute_score(a0)
        )
        if val_best < best_val:
            best_val, best_seq = val_best, a_best.tolist()
            print(f"[parametric sweep] improved score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 250 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 250.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.15)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing with restarts (≈ 250 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 100.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_annealed = anneal_search(a_start, anneal_deadline)
        val = compute_score(a_annealed)
        print(f"[anneal] post‑anneal score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_annealed.tolist()

    # -----------------------------------------------------------------
    # 5. Final hill‑climb (remaining time)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
