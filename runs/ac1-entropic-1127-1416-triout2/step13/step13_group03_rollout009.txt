"""
Propose a step‑function that minimises ``evaluate_sequence``.

Strategy
--------
1. **Global random exploration** – generate many candidates from a
   collection of parametric families (beta, power‑law, exponential,
   Gaussian, sine‑power, mixtures and Dirichlet draws).  Keep the best
   few vectors.

2. **Low‑dimensional optimisation** – run a cheap differential‑evolution
   search on the two‑parameter beta family (α,β) and on the power‑law
   exponent.  This refines the best shapes far beyond a simple grid.

3. **Smooth‑max Adam descent** – start Adam from the best candidates,
   using a smooth‑max surrogate of the max‑convolution.  The temperature
   schedule is more aggressive than the reference code.

4. **Projected sub‑gradient polishing** – exact max‑convolution gradient
   (sub‑gradient) descent with a diminishing step size.

5. **Simulated annealing hill‑climb** – occasional stochastic jumps
   improve the final result.

All phases respect the 1000 s wall‑clock budget; the best sequence found
is returned as a plain ``list`` of floats (sum is normalised to 1, so the
official ``evaluate_sequence`` returns the same value as ``compute_score``).
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum == 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max approximation of max(conv)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.12,
    tau_end: float = 1e-7,
    tau_decay: float = 0.90,
    lr_start: float = 0.10,
    lr_decay: float = 0.97,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-9,
) -> np.ndarray:
    """Adam optimiser using a smooth‑max (log‑sum‑exp) surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # temperature schedule
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # smooth‑max weights
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient of smooth‑max (cross‑correlation a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # learning‑rate decay
        if t % 500 == 0:
            lr = max(lr * lr_decay, 5e-5)

        # occasional jitter
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 5e-7,
                        cooling_factor: float = 0.9993,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 6000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def try_candidate(vec: np.ndarray, best_list: list, max_keep: int) -> None:
    """Normalise ``vec`` to sum‑1, evaluate, and maybe keep it."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    vec = project_to_simplex(vec)
    sc = compute_score(vec)
    if len(best_list) < max_keep:
        best_list.append((sc, vec.copy()))
        best_list.sort(key=lambda x: x[0])
    elif sc < best_list[-1][0]:
        best_list[-1] = (sc, vec.copy())
        best_list.sort(key=lambda x: x[0])

    # 1. Global random / parametric exploration (≈ 250 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 450.0
    best_pool = []  # list of (score, vector)
    max_keep = 12

    # a) Power‑law sweep
    for p in np.linspace(0.05, 7.0, 71):
        if time.time() >= phase_deadline:
            break
        try_candidate(generate_powerlaw(n, p), best_pool, max_keep)

    # b) Exponential sweep
    for lam in np.linspace(0.0001, 0.7, 120):
        if time.time() >= phase_deadline:
            break
        try_candidate(generate_exponential(n, lam), best_pool, max_keep)

    # c) Beta grid
    al_grid = np.geomspace(0.01, 5.0, 25)
    be_grid = np.geomspace(0.01, 5.0, 25)
    for a in al_grid:
        for b in be_grid:
            if time.time() >= phase_deadline:
                break
            try_candidate(generate_beta(n, a, b), best_pool, max_keep)

    # d) Random Beta draws (explore off‑grid)
    for _ in range(300):
        if time.time() >= phase_deadline:
            break
        alpha = 10 ** np.random.uniform(-3, 0.7)   # 0.001 … 5
        beta = 10 ** np.random.uniform(-3, 0.7)
        try_candidate(generate_beta(n, alpha, beta), best_pool, max_keep)

    # e) Gaussian family
    for sigma in np.linspace(0.12 * n, 0.5 * n, 40):
        if time.time() >= phase_deadline:
            break
        try_candidate(generate_gaussian(n, sigma), best_pool, max_keep)

    # f) Sine‑power family
    for p in np.linspace(0.2, 10.0, 40):
        if time.time() >= phase_deadline:
            break
        try_candidate(generate_sine_power(n, p), best_pool, max_keep)

    # g) Mixtures of two random families
    for _ in range(200):
        if time.time() >= phase_deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 6.0))
        b = generate_beta(n,
                          10 ** np.random.uniform(-3, 0.7),
                          10 ** np.random.uniform(-3, 0.7))
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam), best_pool, max_keep)

    # h) Dirichlet draws (wide concentration range)
    concs = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, 2000 // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= phase_deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a, best_pool, max_keep)

    best_pool.sort(key=lambda x: x[0])
    for sc, vec in best_pool:
        if sc < best_val:
            best_val, best_seq = sc, vec.tolist()
    print(f"[baseline] best score after random search = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Low‑dimensional optimisation with DE on Beta & Power‑law
    # --------------------------------------------------------------
    de_deadline = deadline - 300.0
    if time.time() < de_deadline:
        # ---- Beta family -------------------------------------------------
        def beta_obj(x):
            a, b = x
            vec = generate_beta(n, a, b)
            return compute_score(vec)

        bounds_beta = [(1e-4, 5.0), (1e-4, 5.0)]
        result_beta = differential_evolution(
            beta_obj,
            bounds_beta,
            seed=seed,
            maxiter=25,
            popsize=7,
            polish=False,
            updating='deferred',
            workers=1,
        )
        vec_beta = generate_beta(n, *result_beta.x)
        try_candidate(vec_beta, best_pool, max_keep)
        print(f"[DE‑beta] best = {result_beta.fun:.6f} (α={result_beta.x[0]:.4f}, "
              f"β={result_beta.x[1]:.4f})")

        # ---- Power‑law family -------------------------------------------
        def plaw_obj(p):
            vec = generate_powerlaw(n, p[0])
            return compute_score(vec)

        bounds_pl = [(0.01, 10.0)]
        result_pl = differential_evolution(
            plaw_obj,
            bounds_pl,
            seed=seed + 1,
            maxiter=25,
            popsize=7,
            polish=False,
            updating='deferred',
            workers=1,
        )
        vec_plaw = generate_powerlaw(n, result_pl.x[0])
        try_candidate(vec_plaw, best_pool, max_keep)
        print(f"[DE‑plaw] best = {result_pl.fun:.6f} (p={result_pl.x[0]:.4f})")

    # Keep the current best after DE
    best_pool.sort(key=lambda x: x[0])
    if best_pool:
        sc, vec = best_pool[0]
        if sc < best_val:
            best_val, best_seq = sc, vec.tolist()
    print(f"[post‑DE] best known score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Adam‑based smooth‑max optimisation from top candidates
    # --------------------------------------------------------------
    if best_seq is not None:
        adam_deadline = deadline - 150.0
        start_vecs = []
        # a) best from pool
        start_vecs.append(project_to_simplex(np.array(best_seq, dtype=float)))
        # b) uniform start
        start_vecs.append(np.full(n, 1.0 / n))
        # c) few random starts
        for _ in range(3):
            start_vecs.append(project_to_simplex(np.random.rand(n)))

        refined_best = start_vecs[0].copy()
        refined_val = compute_score(refined_best)

        for idx, svec in enumerate(start_vecs):
            if time.time() >= adam_deadline:
                break
            local_deadline = adam_deadline - (time.time() - start_time) - 0.5
            if local_deadline <= time.time():
                break
            a_ref = smooth_max_adam(
                svec,
                local_deadline,
                tau_start=0.12,
                tau_end=1e-7,
                tau_decay=0.90,
                lr_start=0.10,
                lr_decay=0.97,
            )
            val = compute_score(a_ref)
            print(f"[adam‑opt {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_ref.copy()

        best_val, best_seq = refined_val, refined_best.tolist()
        print(f"[adam‑opt] best after smooth optimisation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Sub‑gradient polishing (more iterations)
    # --------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # --------------------------------------------------------------
    # 5. Simulated annealing final hill‑climb
    # --------------------------------------------------------------
    anneal_deadline = deadline - 3.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # --------------------------------------------------------------
    # 6. Safeguard – fall back to a uniform vector if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 7. Final report using the official evaluation function
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
