def propose_candidate(seed=12345, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    n = 2048

    # Retrieve fallback or generate
    init_seq = None
    fallback = globals().get('height_sequence_1')
    if fallback is not None:
        arr = np.array(fallback, dtype=float)
        if np.isfinite(arr).all():
            init_seq = arr / arr.sum()
    if init_seq is None:
        init_seq = np.random.dirichlet(np.full(n, 0.2))
    # keep best vector and score
    best_vec = init_seq
    best_score = compute_score(best_vec)

    # Primary smoothing optimization
    cur_deadline = deadline - 30  # reserve final polish
    refined = smooth_adam_polish(best_vec, cur_deadline)
    val = compute_score(refined)
    if val < best_score:
        best_score, best_vec = val, refined

    # Random restarts
    while time.time() < cur_deadline:
        a0 = np.random.dirichlet(np.full(n, np.random.choice([0.05,0.1,0.2,0.5,1.0,2.0,4.0])))
        refined = smooth_adam_polish(a0, cur_deadline)
        val = compute_score(refined)
        if val < best_score:
            best_score, best_vec = val, refined
            print(f"[smooth‑adam‑restart] new best = {val:.6f}")

    # Final regular Adam polishing (active set subgradient)
    if time.time() < deadline:
        refined = adam_polish(best_vec, deadline)
        val = compute_score(refined)
        if val < best_score:
            best_score, best_vec = val, refined
            print(f"[final‑adam] final refined = {val:.6f}")

    # Scale to target sum sqrt(2n) and clip
    target_sum = np.sqrt(2 * n)
    final_seq = best_vec * target_sum / best_vec.sum()
    # enforce bounds 0..1000, but values ~ small.
    final_seq = [float(x) for x in final_seq]

    reported = evaluate_sequence(final_seq)

def smooth_adam_polish(a0, deadline, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                       alpha_start=10.0, alpha_factor=1.001, alpha_max=200.0):
   n = len(a0)
   u = np.log(a0 + 1e-20)
   m = np.zeros_like(u)
   v = np.zeros_like(u)
   best_a = None
   best_val = float('inf')
   t = 0
   alpha = alpha_start
   while time.time() < deadline:
        t += 1
        # compute a
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()
        # conv
        conv = full_convolution(a)
        # smooth max
        max_c = conv.max()
        # compute softmax weights
        shift = conv - max_c
        exp_alpha = np.exp(alpha * shift)
        w = exp_alpha / exp_alpha.sum()
        # gradient w.r.t a
        w_rev = w[::-1]
        # compute g via convolution
        # Use np.convolve
        g = 2.0 * np.convolve(a, w_rev)[n-1:2*n-1][::-1]
        # gradient w.r.t u (softmax)
        g_u = a * (g - np.dot(g, a))
        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        # step size schedule decaying
        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # update alpha
        alpha = min(alpha * alpha_factor, alpha_max)

        # evaluate actual score occasionally
        if t % 500 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
   # return best_a if found else final a via softmax
   if best_a is None:
       best_a = np.exp(u - np.max(u))
       best_a /= best_a.sum()
   return best_a

def grad_from_weights(a, w):
    w_rev = w[::-1]
    n = len(a)
    m = len(w_rev)
    n_fft = 1 << ((n + m - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    W = np.fft.rfft(w_rev, n=n_fft)
    G = A * W
    conv_rev = np.fft.irfft(G, n=n_fft)[:n+m-1]
    g = 2.0 * conv_rev[n-1:2*n-1][::-1]
    return g

"""
Non‑negative step‑function optimiser (≈1000 s budget).

Strategy
--------
1. Initialise with a fallback (previous best) or a few simple
   parametric families (beta, power‑law, exponential) and a random
   Dirichlet draw.
2. Refine each start with a *smooth‑Adam* optimiser.
   - The true objective is  max_k (a*a)_k  (the convolution peak).
   - We replace the hard max by a smooth log‑sum‑exp:
         φ_α(a) = log‑sum‑exp(α·conv(a))/α,
     where α starts small (≈10) and is slowly increased to ≈200.
   - Gradients of φ_α are obtained analytically:
         w_k = exp(α·conv_k)/Σ_j exp(α·conv_j)      (softmax weights)
         g = ∇_a φ_α = 2·conv_rev[n‑1:2n‑1][::-1],
         where conv_rev = a ⊛ w_rev  (⊛ = linear convolution,
               w_rev = w[::-1]).
   - Adam updates are performed on the soft‑max parametrisation
     a_i = exp(u_i)/Σ_j exp(u_j).  The best vector (according to the
     exact objective) is kept.
3. After the smooth phase, a short *sub‑gradient Adam* polish
   (using only the active constraints of the true max) is run to
   tighten the final peak.
4. The final unit‑sum vector is scaled to Σ a_i = √(2·n) (the
   normalisation required by ``evaluate_sequence``) and returned.

All heavy work uses FFT‑based convolutions (O(n log n)) and fits
comfortably into the 1000 s wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using an FFT of the next power‑of‑two."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max w.r.t. a.
    w  – softmax weights (length 2·n‑1).
    Returns g = ∇_a φ_α = 2·∑_k w_k·∂conv_k/∂a .
    """
    n = a.shape[0]
    w_rev = w[::-1]                 # reverse the weight vector
    m = w_rev.shape[0]              # = 2·n‑1
    n_fft = 1 << ((n + m - 2).bit_length())   # next power‑of‑two ≥ n+m‑1
    A = np.fft.rfft(a, n=n_fft)
    W = np.fft.rfft(w_rev, n=n_fft)
    G = A * W
    conv_rev = np.fft.irfft(G, n=n_fft)[: n + m - 1]   # length 3·n‑2
    # slice n‑1 … 2·n‑2 (inclusive) and reverse to obtain g_i
    g = 2.0 * conv_rev[n - 1: 2 * n - 1][::-1]
    return g


# ----------------------------------------------------------------------
# Exact objective (unit‑sum vectors)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (used for cheap initial seeds)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sineish(n: int) -> np.ndarray:
    """A simple sinusoidal shape: a_i ∝ sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=float) + 0.5
    a = np.sin(np.pi * i / n)
    a = np.maximum(a, 0.0)           # stay non‑negative
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Adam optimiser for the *smooth* surrogate objective
# ----------------------------------------------------------------------


def smooth_adam_polish(a0: np.ndarray, deadline: float,
                       lr: float = 0.05,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8,
                       alpha_start: float = 10.0,
                       alpha_factor: float = 1.001,
                       alpha_max: float = 200.0,
                       eval_every: int = 200) -> np.ndarray:
    """
    Adam optimiser minimizing the smoothed max of the convolution.
    a0 – initial unit‑sum vector.
    Returns the best unit‑sum vector observed before ``deadline``.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)          # log‑space (soft‑max)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0
    alpha = alpha_start

    while time.time() < deadline:
        t += 1

        # ---- soft‑max projection (ensures Σa = 1) ----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ---- convolution and smoothed weights ----
        conv = full_convolution(a)
        max_c = conv.max()
        shift = conv - max_c                     # ≤ 0, improves numerical stability
        exp_alpha = np.exp(alpha * shift)        # max entry becomes 1
        w = exp_alpha / exp_alpha.sum()          # softmax over 2·n‑1 entries

        # ---- gradient w.r.t. a (using FFT) ----
        g = gradient_from_weights(a, w)

        # ---- gradient w.r.t. the log‑parameters u (mirror‑descent) ----
        g_u = a * (g - np.dot(g, a))

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ---- anneal α (slowly tighten the approximation) ----
        alpha = min(alpha * alpha_factor, alpha_max)

        # ---- occasional exact evaluation (record the best) ----
        if t % eval_every == 0:
            cur_val = compute_score(a)           # exact “max‑conv” score
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[smooth‑adam] it {t:6d} | α={alpha:.1f} | score={cur_val:.6f}")

    # Return the best candidate seen (or the final one if none recorded)
    if best_a is None:
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        best_a = exp_shift / exp_shift.sum()
    return best_a


# ----------------------------------------------------------------------
# Adam optimiser for the *true* max‑convolution (sub‑gradient, as before)
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.05,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                tol: float = 1e-12) -> np.ndarray:
    """
    Adam optimiser of the exact objective max_k (a*a)_k on the simplex.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # soft‑max projection
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # convolution and active set
        conv = full_convolution(a)
        max_val = conv.max()
        active = np.where(conv >= max_val - max_val * tol)[0]

        # sub‑gradient of the true max
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # transform to gradient w.r.t. u (mirror‑descent)
        g_u = a * (g - np.dot(g, a))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaux
        if t % 2000 == 0:
            u += np.random.randn(n) * (lr * 0.001)

        # keep the best exact score
        cur_val = float(2.0 * n * max_val)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

    return best_a if best_a is not None else np.exp(u - np.max(u)) / np.exp(u - np.max(u)).sum()

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    """

    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------

            a_start = np.random.dirichlet(np.full(n, 0.2))

        a_start = np.random.dirichlet(np.full(n, 0.2))

    best_vec = a_start.copy()
