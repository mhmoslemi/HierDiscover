    a = np.exp(z)
    a /= a.sum()
    conv = convolution_fft(a,a)
    maxc = conv.max()
    exp_args = beta*(conv - maxc)
    w = np.exp(exp_args)
    w_sum = w.sum()
    smoothed = np.dot(w, conv)  # weighted average
    # return objective: 2*n * smoothed
    return 2.0 * a.shape[0] * smoothed

    a = np.exp(z)
    sum_a = a.sum()
    a = a / sum_a

    # gradient wrt a: sum_i w_i * d conv_i/d a = 2 * conv_rev (as derived)

    # gradient up to a
    dot_ag = np.dot(a, grad_a)

    grad_z = a * grad_a - dot_ag * a

Let's write final answer with code.assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function (a probability vector) that minimises  

\[
\text{score}(a)=\frac{2\,n\,\max(a*a)}{(\sum a)^2},
\]

where `*` denotes aperiodic convolution.  
All admissible vectors are scaled onto the simplex, so the denominator is always 1 and the problem reduces to minimising the maximal convolution entry.

The search consists of several complementary stages that together explore a large part of the search space while staying well within the 1000 s wall‑clock limit.

1. **Deterministic scan of analytic families** – product‑type families, a *shifted* product family (extra offset δ), left‑heavy power families, exponential‑U families, mixtures and the 2‑parameter Beta family.  
2. **Random dense sampling** of the same families (≈ 3000 evaluations).  
3. **Local refinement** of the best product‑type parameters (`α, δ`) and the best Beta parameters (`p,q`) by simple coordinate hill‑climbing.  
4. **Smooth‑max L‑BFGS‑B optimisation** – the non‑smooth `max` is replaced by a temperature‑controlled soft‑max; the variables are the logarithms of the probabilities, guaranteeing positivity and the simplex constraint automatically. A continuation on the temperature (β = 200 → 2000 → 20000) drives the solution toward the true objective.  
5. **Projected sub‑gradient descent** (as in the reference code) for a few more iterations.  
6. **Greedy mass‑transfer hill‑climb** with adaptive step‑size to polish the final vector.

Only `numpy` (and optionally `scipy.optimize`) are used.  All intermediate results are kept in NumPy arrays for speed, and every stage constantly checks the deadline.  The best vector found is returned as a plain Python list; the final `evaluate_sequence` call performs the official security checks.

```python
"""
Propose a step‑function with minimal 2·n·max(conv(a,a))/(sum(a)^2).

The search proceeds through:
  1. Deterministic grid over several analytic families (product,
     shifted product, left‑heavy power, exponential‑U, uniform‑mix,
     Beta).
  2. Random dense sampling of the same families.
  3. Local refinement of the best product (α,δ) and Beta (p,q)
     parameters via simple hill‑climbing.
  4. Continuation L‑BFGS‑B optimisation on a smooth‑max surrogate
     (variables are log‑probabilities, simplex constraint is built‑in).
  5. Projected sub‑gradient descent on the true max‑objective.
  6. Final greedy hill‑climb moving small mass fractions between
     entries.

All stages respect a common 1000 s deadline.  The best vector found
is returned as a plain ``list[float]``.
"""

import time
import math

# Core utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/(sum(a)^2).  a is assumed non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def shifted_product_family(n: int, alpha: float, delta: float) -> np.ndarray:
    """
    Product family with a symmetric offset δ>0:
        a_i ∝ [(i+½+δ)(n‑½‑i+δ)]^{‑α}.
    Reduces the extreme edge peaks.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5 + delta) + np.log(n - 0.5 - i + delta)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} (asymmetric, monotone decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Discrete Beta‑type density: a_i ∝ (i+½)^{p‑1}·(n‑½‑i)^{q‑1}.
    Works for any p,q>0 (U‑shapes when p,q<1).
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 5.0,
                         beta_end: float = 20_000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate of max(conv).
    Symmetry can be enforced every ``sym_interval`` steps.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 3000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_greedy(start_a: np.ndarray,
                      deadline: float,
                      max_steps: int = 10_000_000,
                      init_scale: float = 0.15) -> Tuple[np.ndarray, float]:
    """Greedy hill‑climb moving a small mass fraction from the largest entry."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        i = np.argmax(cur_a)                # donor (largest)
        j = np.random.randint(n)            # random receiver
        if i == j:
            continue

        delta = scale * cur_a[i]            # amount to move
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            scale = min(scale * 1.03, 0.5)
        else:
            scale = max(scale * 0.97, 1e-4)

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 12) -> Tuple[float, float]:
    """Golden‑section search on a 1‑D function."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

def hill_climb_alpha(alpha0: float, n: int, deadline: float) -> float:
    """Hill‑climb on the pure product family parameter α."""
    best = alpha0
    best_score = compute_score(symmetric_product_family(n, best))
    step = max(0.5, best * 0.5) if best > 0 else 0.5
    while time.time() < deadline and step > 1e-4:
        improved = False
        for sign in (-1, 1):
            cand = max(0.0, best + sign * step)
            sc = compute_score(symmetric_product_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] α refined to {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best

            if sc < best_score:
                best_score = sc
                best_d = cand_d
                improved = True
                print(f"[{time.time():.1f}] δ tuned to {best_d:.6f} (α={best_a:.6f}) score {best_score:.6f}")
                break
        if not improved:
            step_a *= 0.5
            step_d *= 0.5
    return best_a, best_d


def hill_climb_beta(p0: float, q0: float, n: int, deadline: float) -> Tuple[float, float]:
    """Hill‑climb on the 2‑parameter Beta family."""
    best_p, best_q = p0, q0
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = max(0.2, (p0 + q0) * 0.2)
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] β refined to p={best_p:.6f}, q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q

# Smooth‑max + L‑BFGS‑B optimisation
# ----------------------------------------------------------------------
try:
    from scipy.optimize import minimize  # type: ignore
except Exception:  # pragma: no cover
    minimize = None


def smooth_obj_and_grad(z: np.ndarray, beta: float) -> Tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate of max(conv) together with its gradient w.r.t. log‑variables.
    Uses the soft‑max weighted average with temperature β.
    """
    a = np.exp(z)
    a_sum = a.sum()
    a = a / a_sum
    n = a.shape[0]

    conv = convolution_fft(a, a)
    maxc = conv.max()
    # stable soft‑max weights
    exp_args = beta * (conv - maxc)      # ≤ 0
    w = np.exp(exp_args)
    w_sum = w.sum()
    w = w / w_sum

    # smooth value (weighted average)
    smooth_val = np.dot(w, conv)

    # gradient w.r.t. a
    conv_rev = convolution_fft(a[::-1], w)
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

    # back‑prop through softmax (log‑softmax) layer
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a

    obj = 2.0 * n * smooth_val
    grad_z_obj = 2.0 * n * grad_z
    return obj, grad_z_obj


def lbfgs_smooth_optimize(start_a: np.ndarray,
                          deadline: float,
                          betas: Tuple[float, ...] = (200.0, 2000.0, 20000.0),
                          maxiter_per_stage: int = 800) -> Tuple[np.ndarray, float]:
    """Continuation L‑BFGS‑B on the smooth‑max surrogate."""
    if minimize is None:
        # fallback – return the start vector unchanged
        return start_a, compute_score(start_a)

    z = np.log(start_a + 1e-12)

    for beta in betas:
        if time.time() >= deadline:
            break

        def fun(z_vec):
            val, g = smooth_obj_and_grad(z_vec, beta)
            return val, g

        res = minimize(fun,
                       x0=z,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_stage,
                                'disp': False})
        z = res.x
        # Debug information
        a_tmp = np.exp(z)
        a_tmp = a_tmp / a_tmp.sum()
        exact = compute_score(a_tmp)
        print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} → exact score {exact:.6f}")

        if time.time() + 2.0 >= deadline:
            # leave a tiny safety margin
            break

    a_opt = np.exp(z)
    a_opt = a_opt / a_opt.sum()
    return a_opt, compute_score(a_opt)

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the objective.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback (if a known good sequence exists)
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array([float(x) for x in height_sequence_1],
                                dtype=np.float64)  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working length (larger n gives a finer discretisation)
    # ------------------------------------------------------------------
    n = 32768   # 2¹⁵ – a good compromise between resolution and speed

    # ------------------------------------------------------------------
    # 2. Coarse deterministic scan over analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Starting coarse scan (n={n})")

    # ---- product family (no offset) ---------------------------------
    best_alpha = None
    for alpha in np.linspace(0.0, 20.0, 81):          # 0 … 20 step 0.25
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (product α={alpha:.4f}) = {best_score:.6f}")

    # ---- shifted product family (α, δ) ------------------------------
    best_alpha_delta = None
    for delta in np.linspace(0.0, 5.0, 11):          # 0 … 5 step 0.5
        if time.time() >= deadline - 30.0:
            break
        for alpha in np.linspace(0.0, 20.0, 41):    # coarser α for each δ
            if time.time() >= deadline - 30.0:
                break
            vec = shifted_product_family(n, alpha, delta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                best_alpha_delta = (alpha, delta)
                print(f"[{time.time():.1f}] new best (shifted α={alpha:.3f}, δ={delta:.2f}) = {best_score:.6f}")

    # ---- left‑heavy power family ------------------------------------
    for beta in np.linspace(0.0, 20.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] new best (left‑heavy β={beta:.4f}) = {best_score:.6f}")

    # ---- exponential‑U family ---------------------------------------
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] new best (exp‑U γ={gamma:.4f}) = {best_score:.6f}")

    # ---- uniform + exponential‑U mixture ----------------------------
    for lam in np.linspace(0.0, 1.0, 6):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

    # ---- Beta family (grid, includes very small parameters) -------
    p_grid = np.concatenate([np.linspace(0.01, 0.5, 10),
                             np.linspace(0.5, 2.0, 10)])
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                print(f"[{time.time():.1f}] new best (Beta p={p:.3f},q={q:.3f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Random dense exploration of the families
    # ------------------------------------------------------------------
    trials = 0
    max_trials = 3000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(6)
        if choice == 0:                     # product
            alpha = np.random.uniform(0.0, 20.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 1:                   # shifted product
            alpha = np.random.uniform(0.0, 20.0)
            delta = np.random.uniform(0.0, 5.0)
            vec = shifted_product_family(n, alpha, delta)
        elif choice == 2:                   # beta
            p = np.random.uniform(0.01, 2.5)
            q = np.random.uniform(0.01, 2.5)
            vec = beta_family(n, p, q)
        elif choice == 3:                   # left‑heavy
            beta = np.random.uniform(0.0, 20.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 4:                   # exponential‑U
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_u_family(n, gamma)
        else:                               # uniform‑mix of product
            lam = np.random.uniform(0.0, 1.0)
            alpha = np.random.uniform(0.0, 20.0)
            vec = lam * np.full(n, 1.0 / n) + (1.0 - lam) * symmetric_product_family(n, alpha)
            vec = normalize(vec)

        # occasional random right‑shift (helps with edge effects)
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Fine local refinement of the best analytic parameters
    # ------------------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product α (start {best_alpha:.4f})")
        refined_alpha = hill_climb_alpha(best_alpha, n, deadline - 10.0)
        vec = symmetric_product_family(n, refined_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = refined_alpha
            print(f"[{time.time():.1f}] refined product α={refined_alpha:.6f} → {best_score:.6f}")

    if best_alpha_delta is not None and time.time() < deadline - 20.0:
        a0, d0 = best_alpha_delta
        print(f"[{time.time():.1f}] refining shifted product (α={a0:.4f}, δ={d0:.4f})")
        ra, rd = hill_climb_alpha_delta(a0, d0, n, deadline - 10.0)
        vec = shifted_product_family(n, ra, rd)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha_delta = (ra, rd)
            print(f"[{time.time():.1f}] refined shifted (α={ra:.6f}, δ={rd:.6f}) → {best_score:.6f}")

    # Beta refinement (starting from a modest U‑shape)
    if time.time() < deadline - 20.0:
        start_p, start_q = 0.5, 0.5
        refined_p, refined_q = hill_climb_beta(start_p, start_q, n, deadline - 10.0)
        vec = beta_family(n, refined_p, refined_q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] refined Beta p={refined_p:.6f}, q={refined_q:.6f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Continuation L‑BFGS‑B on the smooth‑max surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] L‑BFGS‑B smooth‑max refinement")
        a_start = best_seq.astype(np.float64)
        a_opt, sc_opt = lbfgs_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 150.0,
            betas=(200.0, 2000.0, 20000.0),
            maxiter_per_stage=800,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] after L‑BFGS‑B best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Projected sub‑gradient descent (as a safety net)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 60.0,
            max_iters=2_500_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final greedy hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final greedy hill‑climb")
        a_final, sc_final = hill_climb_greedy(
            start_a=best_seq.astype(np.float64),
            deadline=deadline,
            max_steps=10_000_000,
            init_scale=0.15,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
            print(f"[{time.time():.1f}] greedy improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Fallback to a safe uniform vector (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    # Final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq.tolist()
```
