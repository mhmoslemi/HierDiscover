```python
"""
Search for a step‑function sequence that minimises
``evaluate_sequence`` (i.e. 2·n·max(a*a) for a normalised vector).
The algorithm proceeds in three stages:

1. **Coarse global sweep** – generate candidates from three
   parametric families (power‑law, exponential, beta) plus a pool of
   random Dirichlet draws.  The best few vectors (by the exact
   objective) are kept.

2. **Smoothed‑max gradient descent** – starting from each kept
   candidate we minimise the smooth Log‑Sum‑Exp surrogate of the
   max‑convolution using an exponentiated‑gradient update on the
   simplex (mirrored descent).  The smoothing parameter is increased
   gradually (continuation) to approach the true max.

3. **Optional random restarts** – while time permits we launch the
   same gradient procedure from fresh Dirichlet starts and keep the
   best solution found.

All heavy work is performed with FFT‑based convolutions, so the
method scales to a few‑thousand coefficients within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def linear_correlation(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear correlation (np.correlate(x, y, mode='full')) using FFT."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    corr = np.fft.irfft(X * np.conj(Y), n=n_fft)[:n]
    return corr


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (light)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """Quick sweep of a few parametric families and random Dirichlet draws.
    Keeps the best ``max_keep`` normalised vectors."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep (coarse)
    for p in np.linspace(0.05, 5.0, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(1e-5, 0.2, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep – include very sharp U‑shapes
    al_vals = np.concatenate([
        np.geomspace(1e-3, 0.2, 8),
        np.linspace(0.2, 3.0, 12)
    ])
    be_vals = al_vals
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) random Dirichlet draws with a wide range of concentrations
    concs = [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 400   # a few hundred draws per concentration
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent on the simplex
# ----------------------------------------------------------------------


def smoothed_gradient_descent(a0: np.ndarray,
                              deadline: float,
                              alphas: list = None,
                              init_lr: float = 0.5,
                              min_lr: float = 1e-5,
                              decay: float = 0.995,
                              max_iters_per_alpha: int = 3000) -> np.ndarray:
    """
    Minimises  max_k (a*a)_k  by optimising the smooth
    ``log‑sum‑exp`` surrogate.
    The variables live on the probability simplex; updates are performed
    in log‑space (exponentiated‑gradient) with a simple learning‑rate
    schedule and a continuation on the softening parameter ``alpha``.
    """
    n = a0.shape[0]
    a = a0.copy()
    # keep the best vector observed
    best_a = a.copy()
    best_val = compute_score(best_a)

    if alphas is None:
        alphas = [2.0, 5.0, 10.0, 20.0, 40.0]

    for alpha in alphas:
        lr = init_lr
        it = 0
        while time.time() < deadline and it < max_iters_per_alpha:
            it += 1
            # ----- convolution of the current a -----
            conv = full_convolution(a)                # length 2n‑1
            # ----- soft‑max weights (smooth max) -----
            maxc = conv.max()
            exp_term = np.exp(alpha * (conv - maxc))
            w = exp_term / exp_term.sum()             # length 2n‑1, non‑negative, sums to 1

            # ----- gradient of the smoothed objective -----
            corr = linear_correlation(w, a)            # length 3n‑2 (full correlation)
            grad = 2.0 * corr[(n - 1) - np.arange(n)]   # shape (n,)

            # ----- exponentiated‑gradient (mirrored) update -----
            log_a = np.log(a + 1e-20) - lr * grad
            a = np.exp(log_a)
            a_sum = a.sum()
            if a_sum == 0.0 or not np.isfinite(a_sum):
                a = np.full_like(a, 1.0 / n)
                a_sum = 1.0
            a /= a_sum

            # ----- keep track of the best objective value -----
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # occasional progress output
                print(f"[sgd] alpha {alpha:.1f}, it {it}, lr {lr:.6f}, score {cur_val:.6f}")

            # ----- learning‑rate decay -----
            lr = max(min_lr, lr * decay)

            # early exit if lr got extremely small

"""
Search for a step‑function sequence that minimises ``evaluate_sequence``  
(i.e. 2·n·max(a*a) for a normalised vector).  The algorithm consists of:

1. **Coarse global sweep** – generate candidates from three simple
   families (power‑law, exponential, beta) and a pool of random Dirichlet
   draws.  The best few normalised vectors (by the exact objective) are
   retained.

2. **Smoothed‑max gradient descent** – starting from each retained
   candidate we minimise the smooth Log‑Sum‑Exp surrogate of the
   max‑convolution using an exponentiated‑gradient update on the simplex
   (mirrored descent).  The smoothing parameter α is increased
   gradually (continuation) so the optimisation finally targets the real
   max.

3. **Random restarts** – while time permits we launch the same gradient
   procedure from fresh Dirichlet starts and keep the overall best
   solution.

All heavy work uses FFT‑based convolutions, allowing a few‑thousand
coefficients to be processed well within the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def linear_correlation(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear correlation (np.correlate(x, y, mode='full')) via FFT."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    corr = np.fft.irfft(X * np.conj(Y), n=n_fft)[:n]
    return corr


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Light global exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """Coarse sweep of a few parametric families and Dirichlet draws.
    Returns the best ``max_keep`` normalised vectors."""
    best = []  # [(score, vector), ...]

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep (coarse grid)
    for p in np.linspace(0.05, 5.0, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(1e-5, 0.2, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep – include very sharp U‑shapes
    al_vals = np.concatenate([
        np.geomspace(1e-3, 0.2, 8),
        np.linspace(0.2, 3.0, 12)
    ])
    be_vals = al_vals
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) random Dirichlet draws with a range of concentrations
    concs = [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 400
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent on the simplex
# ----------------------------------------------------------------------


def smoothed_gradient_descent(a0: np.ndarray,
                              deadline: float,
                              alphas: list = None,
                              init_lr: float = 0.5,
                              min_lr: float = 1e-5,
                              decay: float = 0.995,
                              max_iters_per_alpha: int = 3000) -> np.ndarray:
    """Minimise the smooth Log‑Sum‑Exp surrogate of max_k (a*a)_k.
    Updates are performed in log‑space (exponentiated‑gradient) with a
    continuation on the smoothing parameter ``alpha``."""
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(best_a)

    if alphas is None:
        alphas = [2.0, 5.0, 10.0, 20.0, 40.0]

    for alpha in alphas:
        lr = init_lr
        it = 0
        while time.time() < deadline and it < max_iters_per_alpha:
            it += 1

            # convolution of the current vector
            conv = full_convolution(a)                     # length 2n‑1

            # soft‑max weights (smooth approximation of the max)
            maxc = conv.max()
            exp_term = np.exp(alpha * (conv - maxc))
            w = exp_term / exp_term.sum()                 # length 2n‑1, sums to 1

            # gradient of the smoothed objective
            corr = linear_correlation(w, a)                # length 3n‑2
            grad = 2.0 * corr[(n - 1) - np.arange(n)]      # shape (n,)

            # exponentiated‑gradient (mirror‑descent) update
            log_a = np.log(a + 1e-20) - lr * grad
            a = np.exp(log_a)
            s = a.sum()
            if s == 0.0 or not np.isfinite(s):
                a = np.full_like(a, 1.0 / n)
                s = 1.0
            a /= s

            # evaluate and keep the best seen so far
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd] α={alpha:.1f}, it={it}, lr={lr:.6f}, score={cur_val:.6f}")

            # decay learning rate
            lr = max(min_lr, lr * decay)
            if lr <= min_lr:
                break

    return best_a
```
