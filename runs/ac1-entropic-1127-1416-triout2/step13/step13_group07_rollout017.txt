Optimise a non‑negative step‑function (probability vector) for the
objective  2·n·max(conv(a,a)) / (sum(a)**2).

The search consists of:

1.   A fast deterministic scan of a few analytic families (product‑U,
     left‑heavy power, exponential‑U, mixed uniform‑U, Beta).
2.   A large random Monte‑Carlo exploration of the same families, with
     random right‑shifts and random mixtures of two families.
3.   Local refinement of the best family’s parameters (α for the
     product family, (p,q) for the Beta family) and of the integer shift.
4.   A projected sub‑gradient descent on the true max‑objective (the
     most reliable final polishing step).
5.   A greedy mass‑transfer hill‑climb to squeeze the last bit of
     improvement.

All stages respect a single 1000 s wall‑clock budget.  The best
sequence found is returned as a plain ``list[float]``.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)**2).  Assumes a≥0."""
    s = np.sum(a)
    if s < 1e-15:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total <= 0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total


def shift_right(v: np.ndarray, d: int) -> np.ndarray:
    """Shift vector to the right by d (pad left with zeros) and renormalise."""
    if d <= 0:
        return v.copy()
    n = v.shape[0]
    if d >= n:
        return np.zeros_like(v)
    shifted = np.concatenate([np.zeros(d, dtype=v.dtype), v[: n - d]])
    return normalize(shifted)


# ----------------------------------------------------------------------
# Analytic families (all return a normalised non‑negative vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    # log‑product to avoid overflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid (p,q>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective.
    The sub‑gradient is taken at the index attaining max(conv).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        # Euclidean projection onto the simplex
        a = project_to_simplex(a)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb_alpha(alpha0: float, n: int, deadline: float) -> float:
    """Refine the product‑family exponent α (simple 1‑D hill‑climb)."""
    best = max(alpha0, 0.0)
    best_score = compute_score(symmetric_product_family(n, best))
    step = 1.0
    while time.time() < deadline and step > 1e-4:
        improved = False
        for sign in (-1, 1):
            cand = max(0.0, best + sign * step)
            sc = compute_score(symmetric_product_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] α refined to {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta(p0: float, q0: float, n: int, deadline: float) -> Tuple[float, float]:
    """Refine the two‑parameter Beta family (simple 2‑D hill‑climb)."""
    best_p, best_q = max(p0, 1e-6), max(q0, 1e-6)
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = 0.5
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(1e-6, best_p + dp)
                cand_q = max(1e-6, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] β refined to p={best_p:.6f}, q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q


def shift_refine(vec: np.ndarray,
                 current_shift: int,
                 n: int,
                 family_func,
                 family_params,
                 deadline: float) -> Tuple[np.ndarray, int, float]:
    """
    Refine the integer right‑shift of a given family.
    Returns (best_vec, best_shift, best_score).
    """
    best_shift = current_shift
    best_vec = vec.copy()
    best_score = compute_score(best_vec)
    step = max(1, n // 256)  # start with a relatively coarse step
    while time.time() < deadline and step >= 1:
        improved = False
        for d in (best_shift - step, best_shift + step):
            if 0 <= d < n:
                cand = shift_right(family_func(n, *family_params), d)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_shift = d
                    best_vec = cand
                    improved = True
                    print(f"[{time.time():.1f}] shift refined to {best_shift} score {best_score:.6f}")
                    break
        if not improved:
            step //= 2
    return best_vec, best_shift, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation function under a fixed wall‑clock budget.  Returns
    the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 5 s safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    best_seq: Optional[np.ndarray] = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback using a previously known construction
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array([float(x) for x in height_sequence_1], dtype=np.float64)  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size (larger n gives a finer discretisation)
    # ------------------------------------------------------------------
    n = 16384

    # ------------------------------------------------------------------
    # 2. Deterministic scan of a few analytically tractable families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] starting deterministic scan (n={n})")
    # product family
    for alpha in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (product α={alpha:.4f}) → {sc:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            best_beta = beta
            print(f"[{time.time():.1f}] new best (left‑heavy β={beta:.4f}) → {sc:.6f}")

    # exponential‑U family
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            best_gamma = gamma
            print(f"[{time.time():.1f}] new best (exp‑U γ={gamma:.4f}) → {sc:.6f}")

    # mixed uniform + exponential‑U
    for lam in np.linspace(0.0, 1.0, 6):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) → {sc:.6f}")

    # Beta family (coarse grid)
    for p in np.linspace(0.2, 2.0, 10):
        for q in np.linspace(0.2, 2.0, 10):
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec
                best_pq = (p, q)
                print(f"[{time.time():.1f}] new best (Beta p={p:.3f},q={q:.3f}) → {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Random dense exploration (≈10 k trials if time permits)
    # ------------------------------------------------------------------
    trials = 0
    max_trials = 20000
    print(f"[{time.time():.1f}] entering random search (target ≤{max_trials} trials)")
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        # pick a family uniformly at random
        family_choice = np.random.randint(5)
        if family_choice == 0:   # product family
            alpha = np.random.uniform(0.0, 15.0)
            base = symmetric_product_family(n, alpha)
            params = (alpha,)
            fam_func = symmetric_product_family
        elif family_choice == 1: # left‑heavy power
            beta = np.random.uniform(0.0, 15.0)
            base = left_heavy_power_family(n, beta)
            params = (beta,)
            fam_func = left_heavy_power_family
        elif family_choice == 2: # exponential‑U
            gamma = np.random.uniform(0.0, 6.0)
            base = exp_u_family(n, gamma)
            params = (gamma,)
            fam_func = exp_u_family
        elif family_choice == 3: # mixed uniform‑U
            lam = np.random.uniform(0.0, 1.0)
            gamma = np.random.uniform(0.0, 6.0)
            base = mixed_uniform_u_family(n, lam, gamma)
            params = (lam, gamma)
            fam_func = mixed_uniform_u_family
        else:                    # Beta family
            p = np.random.uniform(0.05, 2.5)
            q = np.random.uniform(0.05, 2.5)
            base = beta_family(n, p, q)
            params = (p, q)
            fam_func = beta_family

        # Random right‑shift with probability 0.35
        if np.random.rand() < 0.35:
            d = np.random.randint(0, n // 3 + 1)
            base = shift_right(base, d)

        # Occasionally try a mixture of two independent families
        if np.random.rand() < 0.07:
            # generate second component
            other_choice = np.random.randint(5)
            if other_choice == 0:
                a2 = symmetric_product_family(n, np.random.uniform(0.0, 15.0))
            elif other_choice == 1:
                a2 = left_heavy_power_family(n, np.random.uniform(0.0, 15.0))
            elif other_choice == 2:
                a2 = exp_u_family(n, np.random.uniform(0.0, 6.0))
            elif other_choice == 3:
                a2 = mixed_uniform_u_family(n,
                                            np.random.uniform(0.0, 1.0),
                                            np.random.uniform(0.0, 6.0))
            else:
                a2 = beta_family(n,
                                 np.random.uniform(0.05, 2.5),
                                 np.random.uniform(0.05, 2.5))
            lam_mix = np.random.rand()
            base = lam_mix * base + (1.0 - lam_mix) * a2
            base = normalize(base)

        sc = compute_score(base)
        if sc < best_score:
            best_score, best_seq = sc, base
            best_family = fam_func
            best_params = params
            best_shift = d if 'd' in locals() else 0
            print(f"[{time.time():.1f}] random trial #{trials} improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Local refinement of the best analytic family parameters & shift
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        # refine shift first (often gives quick gains)
        if 'best_family' in locals():
            vec_shifted, best_shift, best_score_shift = shift_refine(
                best_seq, best_shift, n, best_family, best_params, deadline - 15.0)
            if best_score_shift < best_score:
                best_score, best_seq = best_score_shift, vec_shifted

        # refine specific family parameters
        if best_family is symmetric_product_family and best_params:
            alpha0 = best_params[0]
            refined_alpha = hill_climb_alpha(alpha0, n, deadline - 10.0)
            vec = symmetric_product_family(n, refined_alpha)
            vec = shift_right(vec, best_shift) if best_shift else vec
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec
                print(f"[{time.time():.1f}] refined product α={refined_alpha:.6f} → {sc:.6f}")

        if best_family is left_heavy_power_family and best_params:
            beta0 = best_params[0]
            # simple 1‑D hill‑climb for β
            best = max(beta0, 0.0)
            best_sc = compute_score(left_heavy_power_family(n, best))
            step = 1.0
            while time.time() < deadline - 5.0 and step > 1e-4:
                improved = False
                for sgn in (-1, 1):
                    cand = max(0.0, best + sgn * step)
                    sc = compute_score(left_heavy_power_family(n, cand))
                    if sc < best_sc:
                        best_sc, best = sc, cand
                        improved = True
                        print(f"[{time.time():.1f}] β refined to {best:.6f} → {sc:.6f}")
                        break
                if not improved:
                    step *= 0.5
            vec = left_heavy_power_family(n, best)
            vec = shift_right(vec, best_shift) if best_shift else vec
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec

        if best_family is beta_family and best_params:
            p0, q0 = best_params
            p_opt, q_opt = hill_climb_beta(p0, q0, n, deadline - 10.0)
            vec = beta_family(n, p_opt, q_opt)
            vec = shift_right(vec, best_shift) if best_shift else vec
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec
                print(f"[{time.time():.1f}] refined Beta p={p_opt:.6f}, q={q_opt:.6f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent (the most reliable polishing step)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] starting sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 30.0,
            max_iters=2_500_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub
            print(f"[{time.time():.1f}] sub‑grad gave new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final greedy mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final greedy hill‑climb")
        cur_a = best_seq.astype(np.float64)
        cur_score = compute_score(cur_a)
        best_a = cur_a.copy()
        best_sc = cur_score
        scale = 0.15
        n = cur_a.size
        step = 0
        while time.time() < deadline - 2.0:
            # donor = index with largest value, receiver = random
            i = np.argmax(cur_a)
            j = np.random.randint(n)
            if i == j:
                continue
            delta = scale * cur_a[i]
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = project_to_simplex(cand)
            cand_sc = compute_score(cand)
            if cand_sc < cur_score:
                cur_a, cur_score = cand, cand_sc
                if cur_score < best_sc:
                    best_sc, best_a = cur_score, cand.copy()
                scale = min(scale * 1.03, 0.5)
            else:
                scale = max(scale * 0.97, 1e-4)
            step += 1
            if step % 500_000 == 0:
                print(f"[{time.time():.1f}] greedy step {step}, best {best_sc:.6f}")

        if best_sc < best_score:
            best_score, best_seq = best_sc, best_a
            print(f"[{time.time():.1f}] greedy final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Fallback to uniform if nothing succeeded (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 8. Final official evaluation (security checks included)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] search finished – final reported score = {final_score:.6f}")

    # Return a plain Python list
    return best_seq.tolist()
