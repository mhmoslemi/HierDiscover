from scipy.optimize import linprog

def solve_lp(a, M):
    n = a.shape[0]
    # Build A_ub matrix shape (2*n-1, n)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        # j range
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # for j in [j_start, j_end], a_{k-j} contributes
        # compute indices for a: i = k - j
        i = k - np.arange(j_start, j_end + 1)
        # Set row entries
        A[k, j_start:j_end+1] = a[i]
    b = np.full(rows, M)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return uniform
        return np.full(n, 1.0/n)
    b0 = res.x
    return b0
```

def solve_lp_step(a, M):

    a = a0.copy()

def solve_lp_step(a, M):
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # fill each column
    for j in range(n):
        A[j:j+n, j] = a
    b = np.full(rows, M, dtype=np.float64)
    # solve LP
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                  method='highs')
    if not res.success:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    b0 = res.x
    sum_b = b0.sum()
    if sum_b <= 1e-12:
        return np.full(n, 1.0 / n)
    return b0 / sum_b

"""
Search for a non‑negative step‑height sequence that minimises the official
``evaluate_sequence`` objective.

Algorithm
---------
1. **Fallback / uniform start** – use the previously known best sequence if
   available, otherwise the uniform vector.

2. **Linear‑programming driven refinement** – repeatedly solve a small LP:
   for the current vector `a` (with ∑a=1) compute M = max(conv(a,a)).
   Maximise Σ b_j subject to b≥0 and conv(a,b)≤M (a linear constraint in
   the unknown `b`).  The optimiser returns a dense candidate `b` that is
   normalised to a probability vector and mixed with the current `a`.
   The mixing weight is gradually reduced.  This step drives the worst
   convolution peak down quickly.

3. **Projected Adam optimisation** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp) while projecting onto the probability simplex.

4. **Projected sub‑gradient polishing** – directly minimise the exact
   ``2·n·max(conv)`` using a diminishing step‑size sub‑gradient method.

5. **Simulated annealing hill‑climb** – a final stochastic phase that can
   escape shallow local minima.

All phases respect the 1000 s wall‑clock budget; the best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·conv_k(a,a)   with respect to a.
    Returns g_i = 2·Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    m = len(w)
    rev_w = w[::-1]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is already normalised to sum 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) utilities for Adam
# ----------------------------------------------------------------------


def smooth_max(conv: np.ndarray, tau: float):
    """
    Returns the smooth approximation S = τ·log ∑ exp(conv/τ) and the soft‑max
    weight vector w (∑ w = 1) that yields the gradient.
    """
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.5,
                  tau_end: float = 1e-7,
                  tau_decay: float = 0.90,
                  lr: float = 0.04,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam minimising τ·log‑sum‑exp(conv) with exponential τ‑schedule."""
    a = a0.copy()
    n = a.shape[0]
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape flat regions
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact max‑convolution)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9993,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.04, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.96, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming step (core of the new search)
# ----------------------------------------------------------------------


def solve_lp_step(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ b_j  s.t. b≥0  and  conv(a,b) ≤ M (element‑wise).
    Returns the normalised solution b (sum(b)=1).  If the LP fails, falls back
    to the uniform distribution.
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix: column j contains a shifted copy of a
    A = np.empty((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a

    b_ub = np.full(rows, M, dtype=np.float64)
    # Objective: maximise Σ b   →   minimise -Σ b
    c = -np.ones(n, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs')
    except Exception:
        res = None

    # Clean up the large matrix ASAP
    del A

    if res is None or not res.success:
        return np.full(n, 1.0 / n)

    b0 = res.x
    s = b0.sum()
    if s <= 1e-12:
        return np.full(n, 1.0 / n)
    return b0 / s


def lp_iterative_optimize(a0: np.ndarray, deadline: float,
                          max_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve ``a`` by solving the LP described above and mixing the
    result back into ``a``.  Returns the best vector found and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    mix = 0.5  # initial mixing weight

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = full_convolution(a).max()
        b_norm = solve_lp_step(a, M)

        # convex combination and projection back onto the simplex
        a = (1.0 - mix) * a + mix * b_norm
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()

        # gradually shrink the mixing coefficient
        mix = max(mix * 0.92, 0.05)

    return best_a, best_score

        a = a_polished

    # -----------------------------------------------------------------
    # 5. Simulated annealing (final refinement, remaining time)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        print("[Anneal] start")
        a_final = simulated_annealing(a, deadline)
        cur = compute_score(a_final)
        print(f"[Anneal] final score = {cur:.6f}")
        if cur < best_score:
            best_score, best_seq = cur, a_final.tolist()

    # -----------------------------------------------------------------
    # 6. Final evaluation via the official function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
