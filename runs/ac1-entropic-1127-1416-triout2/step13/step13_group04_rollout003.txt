"""
Propose a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Broad parametric sweep** – evaluate a few well‑known families
   (power‑law, geometric, mixed plateau‑power, beta, logistic) on a
   fixed length (4096).  The best vector from this cheap sweep is kept
   as a seed.

2. **Exponential‑gradient (EG) optimisation** – minimise the peak of
   the self‑convolution by optimising a smooth approximation of the
   ``max`` (soft‑max with a temperature parameter).  The update is a
   mirror‑descent step on the probability simplex:
        a ← a·exp(‑η·g)   (componentwise)
   where g is the gradient of the smoothed max.  The temperature (α)
   is increased gradually so that the optimisation first explores
   globally and then focuses on the true maximum.

3. **Deterministic hill‑climb** – a fast greedy mass‑transfer pass that
   fine‑tunes the solution found by EG.

All vectors are normalised to Σa = 1 (scale‑invariance of the objective).
The routine respects the 1000 s budget by cutting the work into three
time‑bounded phases and always returns the best sequence discovered.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, length‑independent)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear full convolution a * a using real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of w (length 2n‑1) with a (length n):
        (w ⋆ a)[i] = Σ_k w_k * a_{k-i}
    The result has length n and is exactly the gradient of Σ w_k·conv_k.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    m = len(w) + len(a_rev) - 1          # = 3n‑2
    n_fft = _next_pow2(m)
    W = np.fft.rfft(w, n=n_fft)
    A = np.fft.rfft(a_rev, n=n_fft)
    conv = np.fft.irfft(W * A, n=n_fft)[:m]
    start = len(a_rev) - 1               # = n‑1
    return conv[start:start + n]


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the broad sweep)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def logistic_shape(n: int, center_frac: float, scale_frac: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    scale = max(scale_frac * n, 1e-6)
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def random_dirichlet(n: int, concentration: float) -> np.ndarray:
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    if y.sum() == 0.0:
        return None
    return y / y.sum()

# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:

    best_vec = None

    # power‑law

        if a is None:

        if a is None:

        if val < best_val:
            best_val, best_vec = val, a

    for L_frac in np.linspace(0.05, 0.30, 9):
        for p in np.linspace(0.4, 0.9, 11):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:

            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # beta shape
    alphas = np.linspace(-0.5, 2.5, 13)

# ----------------------------------------------------------------------
# Exponential‑gradient optimisation of the smoothed max
# ----------------------------------------------------------------------
def eg_optimize(a0: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """
    Mirror‑descent on the soft‑max of the self‑convolution.
    The temperature α is increased gradually; the step size η is
    decayed.  Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # algorithmic knobs
    alpha = 0.5          # soft‑max temperature
    alpha_max = 80.0
    eta = 0.5            # learning rate for EG
    eta_min = 1e-4
    beta = 0.9           # momentum coefficient (exponential moving average)
    m = np.zeros_like(a)  # momentum term
    it = 0

    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        maxc = conv.max()
        # soft‑max weights (stable)
        shifted = conv - maxc                # maxc subtracted for numerical stability
        exp_vals = np.exp(alpha * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of weighted sum = 2 * (w ⋆ a)
        g = 2.0 * cross_corr(w, a)

        # normalise gradient to keep the update magnitude under control
        gm = g.max()
        if gm > 0:
            g = g / gm

        # momentum
        m = beta * m + (1.0 - beta) * g

        # exponential‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # evaluate
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[EG] it={it:06d} score={cur_score:.6f} α={alpha:.2f} η={eta:.5f} time={time.time() - start_time:.1f}s")

        # schedule adjustments (every few thousand iterations)
        if it % 2000 == 0:
            # increase temperature → focus more on the true max
            alpha = min(alpha * 1.08, alpha_max)
            # decay learning rate
            eta = max(eta * 0.93, eta_min)
            # occasional jitter to escape plateaus
            jitter = np.random.randn(n) * (1e-3 / math.sqrt(it))
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_a


# ----------------------------------------------------------------------
# Simple deterministic hill‑climb (fine polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """
    Greedy mass‑transfer moves: pick i,j, move a small fraction of a[i]
    to a[j] if it lowers the peak of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()

        cur = compute_score(cand)
        if cur < best_score:
            best_score = cur
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        if no_improve > 120_000:
            # shake a little
            a = best_a + np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')
    n = 4096                       # length of the candidate vectors

    # ------------------------------------------------------------------
    # 0. fallback (if the global variable ``height_sequence_1`` exists)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. cheap parametric sweep to obtain a solid seed
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 550.0               # leave ~550 s for refinement
    if time.time() < sweep_deadline:
        seed_vec = parametric_sweep(n, sweep_deadline)
        if seed_vec is not None:
            val = compute_score(seed_vec)
            print(f"[sweep] best internal score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seed_vec.tolist()

    # ------------------------------------------------------------------
    # 2. Exponential‑gradient optimisation (major work)
    # ------------------------------------------------------------------
    eg_deadline = deadline - 170.0                  # reserve ~170 s for final hill‑climb
    if best_seq is not None and time.time() < eg_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_opt = eg_optimize(a0, eg_deadline, start_time)
        val = compute_score(a_opt)
        print(f"[EG‑final] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()

    # ------------------------------------------------------------------
    # 3. Final deterministic hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, deadline, start_time)
        val = compute_score(a_final)
        print(f"[hill‑climb] final internal score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # 4. safeguard – uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. final report using the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
