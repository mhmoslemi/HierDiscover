def hill_climb(a0, deadline, step_scale=0.01):
    a = a0.copy()
    cur_val = compute_score(a)
    n = len(a)
    while time.time() < deadline:
        # Randomly pick i (source) with a_i > threshold
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = step_scale * a[i] * np.random.rand()
        # delta <= a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand /= cand.sum()  # re-normalize but sum stays same? Actually sum unchanged. But after move delta removed from i and added to j, sum unchanged; no need to renorm but due to floating errors we could renorm.
        val = compute_score(cand)
        if val < cur_val or np.random.rand() < np.exp(-(val - cur_val) / (step_scale+1e-12)):
            a, cur_val = cand, val
            # Optionally increase step_scale gradually if success
            step_scale = min(step_scale * 1.02, 0.2)
        else:
            step_scale = max(step_scale * 0.98, 1e-5)
    return a

Advanced non‑negative step‑function optimiser.

The search proceeds in three stages:

1. **Global exploration** – a richer set of parametric families is
   scanned (symmetric power‑law, symmetric exponential, mixed
   power‑law+exponential, inverse‑linear, plus a large pool of Dirichlet
   draws, including specialised “edge‑biased’’ concentrations).  The

   is exhausted, always keeping the best vector seen.  Random‑restart Adam
   runs are launched while time permits.

3. **Final annealing** – a short Metropolis‑style simulated‑annealing
   pass (with adaptive temperature and step‑scale) is applied to the
   current best vector to escape any remaining local minima.

All stages respect the 1000 s wall‑clock budget.  The routine returns the
best sequence (plain list of floats) ready for the official
``evaluate_sequence`` call.
"""

# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector – sum = 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sym_powerlaw(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shaped power law:
       a_i ∝ (i+1)^{-p} + (n-i)^{-p}.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p) + (n - i + 1) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sym_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential tails:
       a_i ∝ exp(-lam·i) + exp(-lam·(n-1-i)).
    """
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_mixed_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """Mixture of power‑law and exponential tails:
       a_i ∝ (i+1)^{-p}·exp(-lam·i) + (n-i)^{-p}·exp(-lam·(n-1-i)).
    """
    i = np.arange(n, dtype=float)
    left = (i + 1) ** (-p) * np.exp(-lam * i)
    right = (n - i) ** (-p) * np.exp(-lam * (n - 1 - i))
    a = left + right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_inverse_linear(n: int) -> np.ndarray:
    """U‑shaped 1/(i+1) + 1/(n-i) shape."""
    i = np.arange(n, dtype=float)
    a = 1.0 / (i + 1.0) + 1.0 / (n - i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Directed Dirichlet draws (edge‑biased)
# ----------------------------------------------------------------------


def dirichlet_edge_draw(n: int, edge_frac: float,
                        conc_edge: float, conc_center: float) -> np.ndarray:
    """Draw from Dirichlet where the first/last `edge_len` coordinates have
    concentration `conc_edge` and the rest have `conc_center`.
    """
    edge_len = max(1, int(edge_frac * n))
    alpha = np.full(n, conc_center, dtype=float)
    alpha[:edge_len] = conc_edge
    alpha[-edge_len:] = conc_edge
    return np.random.dirichlet(alpha)


# ----------------------------------------------------------------------
# Global exploration (keeps the best `max_keep` candidates)
# ----------------------------------------------------------------------


def baseline_search_extended(n: int, deadline: float, max_keep: int = 20):
    """Sweeps richer families and draws, retaining the best `max_keep` vectors."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # project onto simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) symmetric power‑law sweep
    for p in np.linspace(0.02, 5.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_sym_powerlaw(n, p))

    # 2) symmetric exponential sweep
    for lam in np.linspace(1e-5, 0.25, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_sym_exponential(n, lam))

    # 3) mixed power‑law + exponential sweep (coarse grid)
    p_vals = np.linspace(0.1, 2.0, 30)
    lam_vals = np.linspace(1e-5, 0.15, 30)
    for p in p_vals:
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_mixed_powerexp(n, p, lam))

    # 4) inverse‑linear shape
    if time.time() < deadline:
        try_candidate(generate_inverse_linear(n))

    # 5) standard Dirichlet draws, several concentration levels
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1200  # ~9600 draws total
    for conc in concs:
        alpha = np.full(n, conc, dtype=float)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha)
            try_candidate(a)

    # 6) edge‑biased Dirichlet draws
    edge_fracs = [0.01, 0.02, 0.04]
    conc_edge_vals = [0.02, 0.05, 0.1]
    conc_center_vals = [0.5, 1.0, 2.0, 4.0]
    draws_per_setting = 300
    for ef in edge_fracs:
        for ce in conc_edge_vals:
            for cc in conc_center_vals:
                for _ in range(draws_per_setting):
                    if time.time() >= deadline:
                        break
                    a = dirichlet_edge_draw(n, ef, ce, cc)
                    try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam polishing (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.07,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                tol: float = 1e-12) -> np.ndarray:
    """
    Adam optimiser minimising `max_k (a*a)_k` on the simplex.
    Variables are stored as `u` with `a = softmax(u)`.
    Returns the best vector observed before the deadline.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)                    # log‑space initialisation

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float('inf')
    t = 0

    while time.time() < deadline:
        t += 1

        # ----- soft‑max projection -----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ----- convolution & active set -----
        conv = full_convolution(a)
        max_conv = conv.max()
        # indices where conv is within relative tolerance of the max
        active = np.where(conv >= max_conv - max_conv * tol)[0]

        # ----- sub‑gradient w.r.t. a -----
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # ----- mirror‑descent gradient (soft‑max Jacobian) -----
        g_u = a * (g - np.dot(g, a))

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Gentle decay (square‑root time scaling)
        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to break plateaux
        if t % 1500 == 0:
            u += np.random.randn(n) * (lr * 0.001)

        # ----- keep best solution -----
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # re‑normalise every few thousand steps (numerical safety)
        if t % 4000 == 0:
            max_u = np.max(u)
            exp_shift = np.exp(u - max_u)
            a = exp_shift / exp_shift.sum()
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()

    return best_a if best_a is not None else a0


# ----------------------------------------------------------------------
# Light simulated‑annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # No need to renormalise – total mass unchanged; just guard against
        # tiny negative rounding errors.
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to diversify the search
        if it % 2500 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # --------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Global baseline exploration (rich families)
    # --------------------------------------------------------------
    # Problem size – 4096 gives a finer discretisation while still
    # remaining tractable for the 1000 s budget.
    n = 4096

    baseline_deadline = deadline - 350.0   # reserve most of the budget for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search_extended(n, baseline_deadline, max_keep=20)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Adam polishing (primary + random restarts)
    # --------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # polish the current best with a generous time allotment
        a0 = np.array(best_seq, dtype=float)
        refined = adam_polish(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[adam‑polish‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # extra random‑restart polishing while time permits
        while time.time() < polish_deadline:
            # draw a fresh Dirichlet start with a random concentration
            conc = np.random.choice([0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = adam_polish(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[adam‑polish‑restart] new best = {val:.6f}")

    # --------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # --------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # --------------------------------------------------------------
    # 4. Fallback to a uniform distribution if nothing succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 5. Report using the official evaluation function
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
