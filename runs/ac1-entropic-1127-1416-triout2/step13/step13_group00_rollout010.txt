def refine_by_lp(a_initial, deadline):
    a = a_initial.copy()
    epsilon = 1e-12
    best_score = compute_score(a)
    best_a = a.copy()
    iteration = 0
    while time.time() < deadline:
        # ensure positivity
        a = np.maximum(a, epsilon)
        a = project_to_simplex(a)  # ensure sum 1

        # compute conv and M
        conv = full_convolution(a)
        M = conv.max()

        # Build A matrix
        n = a.shape[0]
        m = 2*n - 1
        A = np.zeros((m, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = a[i]

        # LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        # b vector
        b = M * np.ones(m)
        # bounds
        bounds = [(0, None)] * n
        # solve
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            # fallback or break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        # try blend with various t values to reduce max conv; we can use simple t=0.3
        t = 0.3
        a_new = (1 - t) * a + t * g_norm
        a_new = project_to_simplex(a_new)

        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        # update for next iteration
        a = a_new
        iteration += 1
        if iteration > 30:
            break

    return best_a

"""
LP‑based iterative refinement for the step‑function attack.

The objective is   score = 2·n·max(conv(a,a)) / (sum(a)²)
which for a probability vector (sum(a)=1) reduces to 2·n·max(conv(a,a)).
We seek a non‑negative vector a of length n that minimises this value.

Algorithm
---------
1. Construct a small pool of sensible starting vectors
   (uniform, exponential, power‑law, U‑shape and a few random Dirichlet draws)
   and keep the best of them w.r.t. the exact objective.

2. Repeatedly solve a linear programme

       maximise   Σ_i g_i
       subject to  a * g ≤ M        (M = max(conv(a,a)))
                   g_i ≥ 0

   where the convolution a * g is linear in the unknown g.
   The LP returns a direction g that adds as much total mass as possible
   while keeping the current max‑convolution bound M.

3. Mix the current vector a with the normalised LP solution g
   (a ← (1‑t)·a + t·ĝ,  t chosen from a short list).
   The mixture is projected back onto the probability simplex and the
   process repeats as long as time permits.  Each successful mix strictly
   lowers the objective.

4. A short projected sub‑gradient descent (the “polish’’ phase) finishes
   the optimisation.

5. The final vector is returned; the official ``evaluate_sequence`` routine
   is called only once at the very end to obtain the reported score.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (exact, no rounding)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families for initial candidates
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_ushape(n: int, p: float) -> np.ndarray:
    """U‑shaped: concentrate mass near the ends (p > 0)."""
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def random_dirichlet(n: int, conc: float) -> np.ndarray:
    return np.random.dirichlet([conc] * n)


# ----------------------------------------------------------------------
# LP‑based refinement
# ----------------------------------------------------------------------


def lp_refine(a0: np.ndarray, deadline: float, max_iter: int = 30) -> np.ndarray:
    """
    Iteratively improve a probability vector by solving the LP
    maximise Σ g_i  subject to  a * g ≤ M  and  g ≥ 0.
    After each LP we blend the current vector with the normalised solution.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    best_a = a.copy()
    best_val = compute_score(a)

    eps = 1e-12               # keep every entry strictly positive
    blend_factors = (0.5, 0.3, 0.2, 0.1, 0.05)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # enforce strict positivity and re‑normalise
        a = np.maximum(a, eps)
        a = a / a.sum()

        # current max‑convolution
        conv = full_convolution(a)
        M = conv.max()

        # Assemble the linear constraints a * g ≤ M.
        # Row k, column j contains a_{k‑j} if 0 ≤ k‑j < n, else 0.
        A = np.zeros((m, n), dtype=float)
        for i in range(n):
            A[i:i + n, i] = a[i]          # contribution of g_i to rows i … i+n‑1

        # Linear programme (maximise Σ g_i  ⇔  minimise −Σ g_i)
        c = -np.ones(n)                    # minimise -sum(g)
        b = np.full(m, M)                  # RHS = current bound M
        bounds = [(0, None)] * n           # g_i ≥ 0

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # In the unlikely event that the LP fails, abort the refinement loop.
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g                 # re‑normalise to a probability vector

        # Attempt several blending factors and keep the best reduction.
        improved = False
        for t in blend_factors:
            cand = (1.0 - t) * a + t * g_norm
            cand = project_to_simplex(cand)
            conv_cand = full_convolution(cand)
            M_cand = conv_cand.max()
            if M_cand < M - 1e-12:         # strict improvement
                a = cand
                M = M_cand
                improved = True
                break

        if not improved:
            # Even if no strict improvement, still move a little towards g.
            a = (a + g_norm) * 0.5
            a = project_to_simplex(a)
            conv = full_convolution(a)
            M = conv.max()

        cur_val = 2.0 * n * M               # exact objective for the candidate
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (short version)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.25) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        mx = conv.max()
        eps = mx * 1e-9
        active = np.where(conv >= mx - eps)[0]

        # gradient of the active max‑convolution entry
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a
