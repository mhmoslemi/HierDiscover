"""
Improved step‑function optimiser for ``evaluate_sequence`` (1000 s budget).

The method consists of three stages:

1. **Broad global sampling** – a large collection of parametric families
   (power‑law, exponential, beta, double‑exponential, double‑power‑law,
   cosine‑shape, triangular, random piecewise‑constant) and many Dirichlet
   draws are generated.  Every candidate is projected onto the probability
   simplex (∑a_i = 1, a_i ≥ 0) and scored with the exact objective
   ``2·n·max(conv(a,a))``.  The best 20 vectors are kept.

2. **High‑precision polishing** – each retained vector is refined by a
   mirror‑descent (soft‑max) Adam optimiser that minimizes the
   infinity‑norm of the self‑convolution.  The optimiser runs until the
   time budget for polishing expires, keeping the best observed vector.

3. **Escaping plateaux** – a fast simulated‑annealing hill‑climb is run
   from the current overall best to explore nearby configurations and
   possibly obtain a lower peak.

All stages respect the 1000 s wall‑clock limit.  The routine returns the
best sequence found as a plain Python list of floats, ready for the
official ``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities for the simplex and convolution
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    w_i ≥ 0, Σ w_i = 1.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT (length 2·n‑1).
    """
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())   # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=fft_len)
    C = A * A
    conv = np.fft.irfft(C, n=fft_len)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a non‑negative vector, NOT normalised)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return a


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    return a


def generate_double_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric sum of two exponentials from both ends."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    return a


def generate_double_powerlaw(n: int, p: float) -> np.ndarray:
    """Symmetric sum of two power‑laws from both ends."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p) + (n - i + 1) ** (-p)
    return a


def generate_cosine_shape(n: int, power: float) -> np.ndarray:
    """a_i ∝ (cos(π·i/(n‑1)))^{power}, clamped to non‑negative."""
    theta = np.pi * np.arange(n, dtype=float) / (n - 1)
    a = np.cos(theta) ** power
    a[a < 0] = 0.0
    return a


def generate_triangular(n: int) -> np.ndarray:
    """Linear increase to the centre then decrease (∧‑shape)."""
    i = np.arange(n, dtype=float)
    half = n // 2
    a = np.where(i <= half, i + 1.0, n - i)
    return a


def generate_piecewise_constant(n: int, segments: int) -> np.ndarray:
    """Random piecewise‑constant vector with ``segments`` blocks."""
    # Choose random break points
    cuts = np.sort(np.random.choice(n - 1, segments - 1, replace=False) + 1)
    lengths = np.diff(np.concatenate(([0], cuts, [n])))
    vals = np.random.rand(segments) + 1e-6          # avoid exact zeros
    a = np.repeat(vals, lengths)
    return a


# ----------------------------------------------------------------------
# Global exploration
# ----------------------------------------------------------------------


def try_candidate(vec: np.ndarray, best: list):
    """Project ``vec`` onto the simplex, evaluate and possibly store it."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return
    a = vec / total                     # project onto simplex
    sc = compute_score(a)
    # keep up to 20 best vectors
    if len(best) < 20:
        best.append((sc, a.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, a.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float):
    """Sweep families, random pieces and Dirichlet draws, returning the
    best 20 candidates (as numpy arrays, already normalised)."""
    best = []   # list of (score, vector)

    # ---------- power‑law ----------
    for p in np.linspace(0.01, 6.0, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p), best)

    # ---------- exponential ----------
    for lam in np.linspace(1e-6, 0.35, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam), best)

    # ---------- beta ----------
    al_vals = np.concatenate([np.geomspace(0.01, 0.3, 15),
                              np.linspace(0.3, 4.0, 25)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.3, 15),
                              np.linspace(0.3, 4.0, 25)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta), best)

    # ---------- double exponential ----------
    for lam in np.linspace(1e-5, 0.4, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_double_exponential(n, lam), best)

    # ---------- double power‑law ----------
    for p in np.linspace(0.01, 6.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_double_powerlaw(n, p), best)

    # ---------- cosine shape ----------
    for pw in np.linspace(0.5, 12.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_cosine_shape(n, pw), best)

    # ---------- triangular ----------
    try_candidate(generate_triangular(n), best)

    # ---------- random piecewise constant ----------
    for seg in [2, 3, 4, 5, 6]:
        for _ in range(1200):
            if time.time() >= deadline:
                break
            try_candidate(generate_piecewise_constant(n, seg), best)

    # ---------- massive Dirichlet draws ----------
    conc_levels = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per = 2000
    for conc in conc_levels:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a, best)

    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Adam polishing (soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.07,
                beta1: float = 0.85,
                beta2: float = 0.995,
                eps: float = 1e-9,
                tol: float = 1e-12) -> np.ndarray:
    """
    Adam optimiser for ``max(conv(a,a))`` on the probability simplex.
    Returns the best vector observed before ``deadline``.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)                 # log‑space parameters

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # ---- soft‑max projection ----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ---- convolution and active set ----
        conv = full_convolution(a)
        max_val = conv.max()
        active = np.where(conv >= max_val - max_val * tol)[0]

        # ---- sub‑gradient w.r.t. a ----
        g = np.zeros_like(a)
        # vectorised loop over active indices
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # ---- Adam update in log‑space (mirror descent) ----
        g_u = a * (g - np.dot(g, a))       # Jacobian of soft‑max
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to avoid exact stagnation
        if t % 2500 == 0:
            u += np.random.randn(n) * (lr * 0.001)

        cur_score = float(2.0 * n * max_val)
        if cur_score < best_val:
            best_val, best_a = cur_score, a.copy()

    return best_a if best_a is not None else a0

# Simulated annealing (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 6e-4,
                        final_temp: float = 2e-6,
                        cooling: float = 0.9993,
                        step_scale: float = 0.015) -> np.ndarray:
    """Metropolis hill‑climb with a simple temperature schedule."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            if val < best_val:
                best_val, best_a = val, cand.copy()
        # cool
        T = max(final_temp, T * cooling)

        # occasional jitter to shake out of deeper plateaux
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()

    return best_a
