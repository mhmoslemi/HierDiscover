def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (n‑½‑i)^{‑β} (mirror of left‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump. width_frac ∈ (0,1] controls half‑width."""
    half_width = max(1, int(width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    """Exponential decay from one side (U‑shaped if left=False)."""
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return w / w.sum()


def mixed_uniform_gaussian_family(n: int, lam: float, sigma: float) -> np.ndarray:
    """Convex mixture of uniform and centred Gaussian."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    gaussian = gaussian_center_family(n, sigma)
    w = lam * uniform + (1.0 - lam) * gaussian
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w

def exp_gradient_descent(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.6) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using exponentiated gradients.
    Uses the exact sub‑gradient of the max‑convolution at each step.
    Step‑size ηₜ = η₀ / t^{0.75} (slower decay for fine‑grained convergence).
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / (t ** 0.75)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # occasional true‑objective evaluation
        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] exp‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000,
                  init_temp_factor: float = 0.05,
                  step_scale_init: float = 0.3) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    Uses a higher initial temperature (≈ 5 % of the current score)
    and exponential cooling.
    Supported moves:
        * swap two entries,
        * transfer a fraction from i → j,
        * three‑way redistribution (i → j, i → k).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # (large) initial temperature
    T0 = max(cur_score * init_temp_factor, 1e-4)
    T = T0
    step_scale = step_scale_init

    # exponential cooling factor (final temperature ≈ 1e‑8)
    cooling = (1e-8 / T0) ** (1.0 / max_steps)

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        r = np.random.rand()
        if r < 0.4:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.8:        # simple transfer
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        else:                # three‑way redistribution
            i, j, k = np.random.randint(n, size=3)
            if i != j and i != k and cand[i] > 0:
                delta = step_scale * cand[i]
                part = np.random.rand()
                cand[i] -= delta
                cand[j] += delta * part
                cand[k] += delta * (1 - part)

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                # gradually increase the step scale when we improve
                step_scale = min(step_scale * 1.04, 0.6)

        # slowly shrink step scale when moves are rejected
        if delta >= 0:
            step_scale *= 0.995
            step_scale = max(step_scale, 1e-4)

        # exponential cooling
        T *= cooling

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 14) -> Tuple[float, float]:
    """Golden‑section search for the minimum of func on [lo,hi]."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the final call to evaluate_sequence()
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Optional fallback (previous best construction)
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Working discretisation
    # -----------------------------------------------------------
    n = 8192

    # -----------------------------------------------------------
    # 2. Coarse scan of analytic families (expanded ranges)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] coarse family scan (n={n})")
    best_family = None
    best_params = None

    # ---- symmetric product (U‑shaped) -------------------------
    for alpha in np.linspace(0.0, 20.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "product"
            best_params = {"alpha": alpha}
            print(f"[{time.time():.1f}] new best (U‑α={alpha:.3f}) = {best_score:.6f}")

    # ---- centre‑heavy power (positive exponent) ---------------
    for alpha in np.linspace(0.0, 20.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_center_power_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "center_power"
            best_params = {"alpha": alpha}
            print(f"[{time.time():.1f}] new best (center‑α={alpha:.3f}) = {best_score:.6f}")

    # ---- Gaussian centre ---------------------------------------
    for sigma in np.linspace(0.15 * n, 0.45 * n, 30):
        if time.time() >= deadline - 30.0:
            break
        vec = gaussian_center_family(n, sigma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "gaussian"
            best_params = {"sigma": sigma}
            print(f"[{time.time():.1f}] new best (gauss‑σ={sigma:.0f}) = {best_score:.6f}")

    # ---- Triangular centre --------------------------------------
    for width in np.linspace(0.1, 0.9, 33):
        if time.time() >= deadline - 30.0:
            break
        vec = triangular_center_family(n, width)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "triangular"
            best_params = {"width_frac": width}
            print(f"[{time.time():.1f}] new best (tri‑w={width:.3f}) = {best_score:.6f}")

    # ---- left‑heavy and right‑heavy power (asymmetric) ---------
    for beta in np.linspace(0.0, 20.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "left_power"
            best_params = {"beta": beta}
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")

        vec_r = right_heavy_power_family(n, beta)
        sc_r = compute_score(vec_r)
        if sc_r < best_score:
            best_score, best_seq = sc_r, vec_r.tolist()
            best_family = "right_power"
            best_params = {"beta": beta}
            print(f"[{time.time():.1f}] new best (right‑β={beta:.3f}) = {best_score:.6f}")

    # ---- exponential decay --------------------------------------
    for gamma in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_decay_family(n, gamma, left=True)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "exp_left"
            best_params = {"gamma": gamma}
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")

        vec_r = exp_decay_family(n, gamma, left=False)
        sc_r = compute_score(vec_r)
        if sc_r < best_score:
            best_score, best_seq = sc_r, vec_r.tolist()
            best_family = "exp_right"
            best_params = {"gamma": gamma}
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f} right) = {best_score:.6f}")

    # ---- beta family (two‑parameter) ----------------------------
    for _ in range(150):
        if time.time() >= deadline - 30.0:
            break
        p = np.random.uniform(0.1, 5.0)
        q = np.random.uniform(0.1, 5.0)
        vec = beta_family(n, p, q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "beta"
            best_params = {"p": p, "q": q}
            print(f"[{time.time():.1f}] new best (beta p={p:.3f},q={q:.3f}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Random dense search over families (≈15000 trials)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] random dense search")
    trials = 0
    max_trials = 15000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(9)
        if choice == 0:      # U‑shaped product
            alpha = np.random.uniform(0.0, 20.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 1:    # centre‑heavy power
            alpha = np.random.uniform(0.0, 20.0)
            vec = symmetric_center_power_family(n, alpha)
        elif choice == 2:    # Gaussian
            sigma = np.random.uniform(0.15 * n, 0.45 * n)
            vec = gaussian_center_family(n, sigma)
        elif choice == 3:    # triangular
            width = np.random.uniform(0.1, 0.9)
            vec = triangular_center_family(n, width)
        elif choice == 4:    # left‑heavy power
            beta = np.random.uniform(0.0, 20.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 5:    # right‑heavy power
            beta = np.random.uniform(0.0, 20.0)
            vec = right_heavy_power_family(n, beta)
        elif choice == 6:    # exponential decay left
            gamma = np.random.uniform(0.0, 8.0)
            vec = exp_decay_family(n, gamma, left=True)
        elif choice == 7:    # exponential decay right
            gamma = np.random.uniform(0.0, 8.0)
            vec = exp_decay_family(n, gamma, left=False)
        else:                # mixture uniform + Gaussian
            lam = np.random.uniform(0.0, 1.0)
            sigma = np.random.uniform(0.15 * n, 0.45 * n)
            vec = mixed_uniform_gaussian_family(n, lam, sigma)

        # occasional random shift (zero‑padding) that breaks symmetry
        if np.random.rand() < 0.25:
            d = np.random.randint(0, n // 50 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Local golden‑section refinement of the best family
    # -----------------------------------------------------------
    if best_family is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining best family {best_family}")

        if best_family == "product":
            cur_alpha = best_params["alpha"]
            lo, hi = max(0.0, cur_alpha - 1.0), min(20.0, cur_alpha + 1.0)

            def f_alpha(a):
                return compute_score(symmetric_product_family(n, a))

            new_alpha, _ = golden_section_optimize(f_alpha, lo, hi, max_iter=16)
            vec = symmetric_product_family(n, new_alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_alpha
                print(f"[{time.time():.1f}] refined product‑α={new_alpha:.4f} → {best_score:.6f}")

        elif best_family == "center_power":
            cur_alpha = best_params["alpha"]
            lo, hi = max(0.0, cur_alpha - 1.0), min(20.0, cur_alpha + 1.0)

            def f_alpha(a):
                return compute_score(symmetric_center_power_family(n, a))

            new_alpha, _ = golden_section_optimize(f_alpha, lo, hi, max_iter=16)
            vec = symmetric_center_power_family(n, new_alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_alpha
                print(f"[{time.time():.1f}] refined centre‑α={new_alpha:.4f} → {best_score:.6f}")

        elif best_family == "gaussian":
            cur_sigma = best_params["sigma"]
            lo, hi = max(0.15 * n, cur_sigma - 0.15 * n), min(0.45 * n, cur_sigma + 0.15 * n)

            def f_sigma(s):
                return compute_score(gaussian_center_family(n, s))

            new_sigma, _ = golden_section_optimize(f_sigma, lo, hi, max_iter=16)
            vec = gaussian_center_family(n, new_sigma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["sigma"] = new_sigma
                print(f"[{time.time():.1f}] refined gauss‑σ={new_sigma:.1f} → {best_score:.6f}")

        elif best_family == "triangular":
            cur_w = best_params["width_frac"]
            lo, hi = max(0.05, cur_w - 0.2), min(0.95, cur_w + 0.2)

            def f_w(w):
                return compute_score(triangular_center_family(n, w))

            new_w, _ = golden_section_optimize(f_w, lo, hi, max_iter=16)
            vec = triangular_center_family(n, new_w)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["width_frac"] = new_w
                print(f"[{time.time():.1f}] refined tri‑w={new_w:.4f} → {best_score:.6f}")

        elif best_family in ("left_power", "right_power"):
            cur_beta = best_params["beta"]
            lo, hi = max(0.0, cur_beta - 1.0), min(20.0, cur_beta + 1.0)

            if best_family == "left_power":
                fam = left_heavy_power_family
                name = "left‑β"
            else:
                fam = right_heavy_power_family
                name = "right‑β"

            def f_beta(b):
                return compute_score(fam(n, b))

            new_beta, _ = golden_section_optimize(f_beta, lo, hi, max_iter=16)
            vec = fam(n, new_beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["beta"] = new_beta
                print(f"[{time.time():.1f}] refined {name}={new_beta:.4f} → {best_score:.6f}")

        elif best_family == "beta":
            # simple local random refinement around (p,q)
            p0, q0 = best_params["p"], best_params["q"]
            for _ in range(300):
                if time.time() >= deadline - 5.0:
                    break
                p = np.clip(p0 + np.random.randn() * 0.15, 0.05, 6.0)
                q = np.clip(q0 + np.random.randn() * 0.15, 0.05, 6.0)
                vec = beta_family(n, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    best_params["p"], best_params["q"] = p, q
                    print(f"[{time.time():.1f}] refined beta p={p:.3f},q={q:.3f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Exponentiated‑gradient descent (mirror descent) refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] exponentiated‑gradient refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = exp_gradient_descent(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=300_000,
            eta0=0.5,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb (high temperature)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=5_000_000,
            init_temp_factor=0.07,
            step_scale_init=0.35,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Fallback to a simple uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
