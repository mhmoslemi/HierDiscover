"""
Optimisation of a non‑negative step‑function for the ``evaluate_sequence``
objective.

The search proceeds in three phases:

1. **Global exploration** – sweep a few classic parametric families
   (power‑law, exponential, beta, Gaussian) and a large set of random
   Dirichlet draws.  The best few candidates are retained.

2. **Hard‑max sub‑gradient descent** – minimise the true
   ``max(conv(a,a))`` directly.  At each iteration we locate the anti‑diagonal
   (index *k*) that currently attains the maximum, compute the exact
   sub‑gradient of this active term (which is just a shifted copy of ``a``),
   take a projected gradient step on the simplex (simple clipping + renormalisation)
   and decay the learning rate.  This method is more aggressive than the
   soft‑max/Adam approach used previously and usually pushes the score below
   1.51.

3. **Short simulated‑annealing polishing** – a final Metropolis walk
   (small mass transfers) to escape any remaining tiny plateaux.

The whole routine respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a * a (full length)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective used for internal comparison
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (for the global sweep)
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta_ - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global baseline search
# ----------------------------------------------------------------------
def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Sweep families and random Dirichlet draws, keep the best few."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # simplex projection
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law
    for p in np.linspace(0.1, 2.5, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential
    for lam in np.linspace(0.0001, 0.07, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta
    al_vals = np.linspace(0.2, 5.0, 30)
    be_vals = np.linspace(0.2, 5.0, 30)
    for alpha in al_vals:
        for beta_ in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta_))

    # 4) Gaussian
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 35)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # 5) Dirichlet draws (various concentrations)
    concs = [0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    draws_per_conc = 1300
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient descent (projected onto the simplex)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray, deadline: float,
                         init_lr: float = 0.12,
                         lr_decay: float = 0.9995,
                         min_lr: float = 1e-6,
                         momentum: float = 0.9,
                         eval_every: int = 500) -> np.ndarray:
    """
    Minimise max(conv(a,a)) via sub‑gradient descent.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    # momentum buffer
    v = np.zeros_like(a)

    it = 0
    lr = init_lr
    while time.time() < deadline:
        it += 1

        # full convolution to locate the active anti‑diagonal
        conv = full_convolution(a)
        k = int(np.argmax(conv))            # index where max is attained

        # compute exact sub‑gradient of the active term:
        # g_i = 2 * a_{k-i}  (if 0 <= k-i < n)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        g[idx_i] = 2.0 * a[idx_j]

        # Adam‑style momentum (simple exponential smoothing)
        v = momentum * v + (1.0 - momentum) * g

        # gradient step
        a = a - lr * v

        # project onto the non‑negative simplex (clip + renormalise)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # periodic exact evaluation
        if it % eval_every == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[subgrad] iter {it:7d}  score {val:.6f}  lr {lr:.2e}")

        # learning‑rate schedule
        lr = max(min_lr, lr * lr_decay)

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter to keep the walk from stagnating
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global exploration (≈ 180 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0   # keep ~5 min for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Sub‑gradient optimisation with several restarts (≈ 650 s)
    # -----------------------------------------------------------------
    opt_deadline = deadline - 120.0
    if best_seq is not None and time.time() < opt_deadline:
        # primary pass starting from current best
        a0 = np.array(best_seq, dtype=float)
        refined = subgradient_optimize(a0, opt_deadline)
        val = compute_score(refined)
        print(f"[subgrad‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional random restarts while time permits
        while time.time() < opt_deadline:
            remaining = opt_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)

            # choose a random start (Dirichlet) with a random concentration
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))

            refined = subgradient_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[subgrad‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
