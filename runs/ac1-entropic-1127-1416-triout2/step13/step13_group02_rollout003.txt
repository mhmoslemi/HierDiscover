def hill_climb(a_start, deadline, max_iter=50000):
    a = np.copy(a_start)
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val
    step_scale = 0.02
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = compute_score(cand)
        if val < cur_val:
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
        else:
            step_scale = max(step_scale * 0.98, 1e-6)
        if it % 5000 == 0:
            # random jitter
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = np.maximum(a, 0)
            a /= a.sum()
            cur_val = compute_score(a)
    return best_a

def baseline_search(n, deadline, max_keep=10):
    best = [] # list of (score, vector)

    def try_cand(vec):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # families
    for p in np.linspace(0.05, 5.0, 200):
        if time.time() >= deadline:
            break
        try_cand(generate_powerlaw(n, p))

    for lam in np.linspace(1e-5, 0.2, 200):
        if time.time() >= deadline:
            break
        try_cand(generate_exponential(n, lam))

    # beta
    al_vals = np.concatenate([np.geomspace(0.02, 0.5, 10), np.linspace(0.5, 3, 12)])
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_cand(generate_beta(n, alpha, beta))

    # sinusoidal shape: a_i = sin(π*(i+0.5)/n)^p (p varied)
    for p in np.linspace(0.5, 5.0, 50):
        if time.time() >= deadline:
            break
        i = np.arange(1, n+1)
        a = np.sin(np.pi * i / (n+1))**p
        try_cand(a)

    # random Dirichlet draws
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1000
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_cand(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

"""
Non‑negative step‑function optimiser.

The optimiser works in three stages:

1.  A global sweep over several parametric families (power‑law,
    exponential, beta, sinusoidal) together with many random Dirichlet
    draws.  The ten best candidates (according to the exact objective) are
    retained.

2.  Each retained candidate is refined by an Adam optimiser that
    minimises a *smoothed* version of the objective
        f_ε(a) = ε·log Σ_k exp( (a*a)_k / ε ).
    The smoothing parameter ε is decreased in a continuation schedule,
    allowing the optimiser to first see a smooth landscape and then home
    in on the true max‑convolution objective.

3.  The best vector obtained so far is finally polished by a cheap
    greedy hill‑climb (pairwise mass transfers) that can escape tiny
    local minima.

All work respects the 1000 s wall‑clock budget; the final list of floats
is returned in the format required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, works for arbitrary lengths)
# ----------------------------------------------------------------------


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power‑of‑two ≥ m
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (same‑length convolution)."""
    return conv_fft(a, a)


# ----------------------------------------------------------------------
# Objective helper (exact, for a normalized vector ∑a = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for the global sweep
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sinusoidal(n: int, power: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** power
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def baseline_search(
    n: int, deadline: float, max_keep: int = 12
) -> list[np.ndarray]:
    """Sweep families and many Dirichlet draws, keeping the best vectors."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law
    for p in np.linspace(0.05, 5.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential
    for lam in np.linspace(1e-5, 0.2, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta (covers U‑shapes and asymmetric shapes)
    al_vals = np.concatenate(
        [np.geomspace(0.02, 0.5, 10), np.linspace(0.5, 3.0, 12)]
    )
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) sinusoidal family (smooth bumps)
    for power in np.linspace(0.5, 5.0, 40):
        if time.time() >= deadline:
            break
        try_candidate(generate_sinusoidal(n, power))

    # 5) massive random Dirichlet draws
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1200
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed‑max gradient (used by the Adam optimiser)
# ----------------------------------------------------------------------


def grad_smoothed_max(a: np.ndarray, eps: float):
    """
    Returns the gradient of the smoothed max
        f_ε(a) = ε·log Σ exp( (a*a)_k / ε )
    with respect to a, together with the raw convolution vector.
    """
    n = a.shape[0]
    # raw convolution a*a
    conv = conv_fft(a, a)                     # length 2n‑1

    # soft‑max weights over convolution entries
    mx = np.max(conv)
    w = np.exp((conv - mx) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient ∂f/∂a = 2 * Σ_k w_k * a_{k‑j}
    # can be written as a cross‑correlation between a and w.
    a_rev = a[::-1]                           # reverse a
    corr = conv_fft(a_rev, w)                 # length 3n‑2
    start = (len(w) - 1) // 2                # = n‑1
    g = 2.0 * corr[start : start + n]         # shape (n,)

    return g, conv


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective, with continuation
# ----------------------------------------------------------------------


def adam_smooth_opt(
    a0: np.ndarray,
    deadline: float,
    eps_schedule=None,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
):
    """
    Adam optimisation of the true max‑convolution objective,
    but using a smoothed surrogate during optimisation.
    The smoothing parameter ε is decreased according to ``eps_schedule``.
    """
    if eps_schedule is None:
        eps_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    n = a0.shape[0]
    # initialise log‑space parameters (soft‑max)
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    t = 0

    best_a = None
    best_val = float("inf")

    for eps in eps_schedule:
        # simple iteration budget per ε
        max_iter_eps = 3000
        for _ in range(max_iter_eps):
            if time.time() >= deadline:
                break
            t += 1

            # soft‑max projection
            max_u = np.max(u)
            exp_shift = np.exp(u - max_u)
            a = exp_shift / exp_shift.sum()

            # gradient of smoothed max
            g_a, conv = grad_smoothed_max(a, eps)

            # chain rule through the soft‑max
            g_u = a * (g_a - np.dot(g_a, a))

            # Adam updates
            m = beta1 * m + (1.0 - beta1) * g_u
            v = beta2 * v + (1.0 - beta2) * (g_u ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr / (np.sqrt(t) + 1.0)
            u -= step * m_hat / (np.sqrt(v_hat) + eps_adam)

            # exact objective (no extra convolution needed)
            cur_val = float(2.0 * n * conv.max())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            # optional early stop for this ε if wall‑clock is near
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    return best_a if best_a is not None else a0


# ----------------------------------------------------------------------
# Final greedy hill‑climb (deterministic local search)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray, deadline: float, max_iter: int = 50000):
    """Simple pairwise mass‑transfer descent; accepts only improvements."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        if val < cur_val:
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if val < best_val:
                best_val, best_a = val, cand.copy()
        else:
            step_scale = max(step_scale * 0.98, 1e-6)

        if it % 5000 == 0:
            # occasional jitter to escape shallow plateaux
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allocated wall‑clock time.
    Returns a plain Python list of floats ready for the official
    ``evaluate_sequence`` call.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size – 2048 works well for the current budget
    n = 2048

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # --------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:

            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Global baseline exploration
    # --------------------------------------------------------------
    baseline_deadline = deadline - 300.0

        for vec in cand_vectors:

    # --------------------------------------------------------------
    polish_deadline = deadline - 120.0

        a0 /= a0.sum()
