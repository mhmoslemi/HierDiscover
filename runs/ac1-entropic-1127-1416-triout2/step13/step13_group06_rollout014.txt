"""
Improved step‑function search.

Algorithm
---------
* Scan a rich set of analytic families (including sine‑power windows)
  on a medium‑size grid (n=8192) and keep the best candidate.
* Perform a cheap random search over the same families and random
  mixtures, allowing occasional zero‑padding shifts.
* Upsample the best candidate to n=16384 (double length) and run a
  *smooth* exponentiated‑gradient descent on the simplex:
    - The objective is a smooth approximation of the max‑convolution
      using a soft‑max (log‑sum‑exp) with a temperature β that is
      increased from 0.5 to 20 during the run.
    - The gradient of the smooth objective can be written as a single
      convolution, so each iteration costs two real FFTs (O(N log N)).
    - After the multiplicative update we project back onto the
      probability simplex.
* The best exact score (computed with the original ``evaluate_sequence``
  routine) is returned.  All time‑checks respect the 1000 s budget.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using real FFT.
    Both a and b are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (without the clipping
    steps).  a is assumed to be non‑negative; if the sum is too small
    we return ``inf``.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to ≥0 and renormalise to a probability vector."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    half_width = int(max(1, width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return w / w.sum()


def mixed_uniform_gaussian_family(n: int, lam: float, sigma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    gaussian = gaussian_center_family(n, sigma)
    w = lam * uniform + (1.0 - lam) * gaussian
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w


def sin_power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π i/(n‑1))^p, zero at the ends, centre‑heavy."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * i / (n - 1))
    w = np.power(w, p)
    return w / w.sum()

# Smooth exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------


def smooth_exp_gradient_descent(start_a: np.ndarray,
                               deadline: float,
                               max_iters: int = 50_000,
                               eta0: float = 0.6,
                               beta_start: float = 0.5,
                               beta_end: float = 20.0,
                               eval_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a soft‑max approximation of the
    maximum convolution.  The update is multiplicative (exponentiated
    gradient) followed by normalisation.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    idx = (2 * n - 2) - np.arange(n)          # used for gradient extraction

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution a * a
        conv = convolution_fft(a, a)

        # temperature schedule (linear increase)
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / max_iters)

        # soft‑max weights
        scaled = beta * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()                          # shape (2n‑1,)

        # gradient = 2 * (a * w_rev) evaluated at the right indices
        w_rev = w[::-1]
        conv_g = convolution_fft(a, w_rev)    # length = 3n‑2
        g = 2.0 * conv_g[idx]

        # multiplicative update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # projection onto simplex (simple scaling, since a stays non‑negative)
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth‑exp‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] smooth‑exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score

# Simple hill‑climb (SA) – used only if time permits
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000,
                  temp_factor: float = 1.0e-3,
                  step_scale_init: float = 0.25) -> tuple[np.ndarray, float]:
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        d = cand_score - cur_score
        if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-18)):
            cur_a, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

         {"beta": np.linspace(0.0, 12.0, 61)}),
        ("product", symmetric_product_family,
         {"alpha": np.linspace(0.0, 12.0, 61)}),
        ("sin_power", sin_power_family,
         {"p": np.linspace(0.5, 10.0, 20)}),
        ("beta", beta_family,
         {"random": 100}),      # special handling – 100 random draws
    ]

    for name, func, param_dict in families:
        if time.time() >= deadline - 30.0:
            break
        if name == "beta":
            # 100 random (p,q) draws
            for _ in range(100):
                p = np.random.uniform(0.2, 5.0)
                q = np.random.uniform(0.2, 5.0)
                vec = beta_family(n_small, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best (beta p={p:.3f},q={q:.3f}) = {best_score:.6f}")
            continue

        # iterate over the supplied parameter grid
        for key, values in param_dict.items():
            for val in values:
                if time.time() >= deadline - 30.0:
                    break
                if key == "random":
                    continue
                kwargs = {key: val}
                vec = func(n_small, **kwargs)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best ({name} {key}={val:.4g}) = {best_score:.6f}")

    # ---------------------------------------------------------------
    # 2. Random dense search (≈8000 trials)
    # ---------------------------------------------------------------
    print(f"[{time.time():.1f}] random dense search (n={n_small})")
    trials = 0
    max_trials = 8000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(9)
        if choice == 0:      # centre‑heavy power
            alpha = np.random.uniform(0.0, 12.0)
            vec = symmetric_center_power_family(n_small, alpha)
        elif choice == 1:    # Gaussian
            sigma = np.random.uniform(0.2 * n_small, 0.5 * n_small)
            vec = gaussian_center_family(n_small, sigma)
        elif choice == 2:    # triangular
            width = np.random.uniform(0.1, 0.9)
            vec = triangular_center_family(n_small, width)
        elif choice == 3:    # left‑heavy power
            beta = np.random.uniform(0.0, 12.0)
            vec = left_heavy_power_family(n_small, beta)
        elif choice == 4:    # right‑heavy power
            beta = np.random.uniform(0.0, 12.0)
            vec = right_heavy_power_family(n_small, beta)
        elif choice == 5:    # product (U‑shaped)
            alpha = np.random.uniform(0.0, 12.0)
            vec = symmetric_product_family(n_small, alpha)
        elif choice == 6:    # exponential decay left
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n_small, gamma, left=True)
        elif choice == 7:    # exponential decay right
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n_small, gamma, left=False)
        else:                # sin‑power mixture
            p = np.random.uniform(0.5, 10.0)
            vec = sin_power_family(n_small, p)

        # occasional random zero‑padding shift (break symmetry)
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n_small // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ---------------------------------------------------------------
    # 3. Upsample to a larger size (n = 16384) and smooth‑grad optimisation
    # ---------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        n_big = 16384
        print(f"[{time.time():.1f}] up‑sampling best candidate to n={n_big}")

        # simple repeat‑up‑sampling (each entry duplicated)
        a_small = np.array(best_seq, dtype=np.float64)
        a_big = np.repeat(a_small, 2)
        a_big = normalize(a_big)          # ensure exact simplex constraint

        # smooth exponentiated‑gradient descent
        a_opt, sc_opt = smooth_exp_gradient_descent(
            start_a=a_big,
            deadline=deadline - 150.0,
            max_iters=80_000,
            eta0=0.6,
            beta_start=0.5,
            beta_end=25.0,
            eval_interval=3000,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] post‑upsample smooth‑grad best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # 4. Optional short simulated‑annealing refinement (if time left)
    # ---------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final short SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=2_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # 5. Fallback to a simple uniform vector if nothing succeeded
    # ---------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] fallback to uniform, score = {best_score:.6f}")

    # final official evaluation (includes security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
