def weighted_gradient(a, w):
    n = a.shape[0]
    N = 1 << ((n + len(w) - 1 - 1).bit_length())
    A = np.fft.fft(a, n=N)
    W = np.fft.fft(w, n=N)

    a = a0.copy()
    n = len(a)
    best_a = a.copy()

        t += 1
        # FFT of a
        A = np.fft.rfft(a, n=N)

"""
search_step_function_advanced.py

Advanced optimiser for ``evaluate_sequence``.
Algorithm:
  1. Broad parametric sweep over several analytic families
     (reverse power‑law, reverse‑power * geometric, shifted power‐law,
      plateau‑power, beta‑shapes, simple power‑law, geometric).
     Fine‑tuning around the best reverse‑power‐law exponent is performed.
  2. Small random convex combinations of a richer basis of shapes
     to discover hybrids.
  3. Smooth‑gradient descent on a soft‑max approximation of the
     convolution‑maximum (soft‑max temperature slowly increased).
     The gradient is obtained efficiently by an FFT based correlation.
  4. Final stochastic hill‑climb for tiny “last‑mile’’ improvements.
All vectors are projected onto the probability simplex after every
update (Euclidean projection).  The code respects the 1000 s budget
and prints intermediate scores.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    N = next_pow_two(m)
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)ᵖ (front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)            # n, n‑1, …, 1
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def reverse_power_geometric(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (n‑i)ᵖ·qⁱ, 0<q<1."""
    idx = np.arange(n, 0, -1, dtype=np.float64)
    geom = q ** np.arange(n, dtype=np.float64)
    a = (idx ** p) * geom
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def reverse_power_shift(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (n‑i+shift)ᵖ, shift ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64) + shift
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)⁻ᵖ."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ qⁱ (0<q<1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    return a / a.sum()


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ xᵅ·(1‑x)ᵝ,  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s

# Parametric sweep – broad family exploration
# ----------------------------------------------------------------------

    best_vec = None
    best_val = float('inf')
    best_p_coarse = None

    # ---- reverse power‑law (coarse) ----
    for p in np.linspace(0.5, 3.5, 61):                # step ≈0.05
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec, best_p_coarse = val, a, p

    # ---- reverse power‑law (fine) around the best coarse p ----
    if best_p_coarse is not None:
        low = max(0.5, best_p_coarse - 0.5)
        high = best_p_coarse + 0.5
        for p in np.linspace(low, high, 101):          # step ≈0.01
            if time.time() >= deadline:
                return best_vec
            a = reverse_power_law(n, p)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- reverse power * geometric (coarse) ----
    for p in np.linspace(0.5, 3.0, 31):
        for q in np.linspace(0.90, 0.999, 35):
            if time.time() >= deadline:
                return best_vec
            a = reverse_power_geometric(n, p, q)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- shifted reverse power‑law (very limited grid) ----
    for p in (1.5, 1.8, 2.0):
        for shift_frac in np.linspace(0.0, 0.5, 6):
            if time.time() >= deadline:
                return best_vec
            shift = shift_frac * n
            a = reverse_power_shift(n, p, shift)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- plain power‑law (decreasing) ----
    for p in np.linspace(0.10, 4.00, 391):
        if time.time() >= deadline:
            return best_vec
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- geometric ratios ----
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 110))):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- mixed plateau + power‑law ----
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- beta‑shape (extended range) ----
    alphas = np.linspace(-0.8, 5.0, 25)
    betas = np.linspace(-0.8, 5.0, 25)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- random convex combos of a richer basis ----
    # basic shapes (already normalised)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rev2 = reverse_power_law(n, 2.0)
    rev_geo = reverse_power_geometric(n, 2.0, 0.98)
    geom = geometric(n, 0.985)
    beta = beta_shape(n, 2.0, 2.0)
    basics = [uniform, rev2, rev_geo, geom, beta]

    for _ in range(3000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Smooth‑gradient descent on a soft‑max of the convolution
# ----------------------------------------------------------------------
def smooth_descent(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   start_alpha: float = 5.0,
                   end_alpha: float = 30.0,
                   step0: float = 0.3) -> np.ndarray:
    """
    Projected gradient descent on
        f(a) = (1/α)·log Σ_k exp(α·conv_k(a))
    using FFT‑based correlation for the gradient.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    N = next_pow_two(3 * n - 2)          # length sufficient for correlation
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0

    while time.time() < deadline:
        t += 1
        # FFT of current a
        A = np.fft.rfft(a, n=N)

        # full self‑convolution (needed for the soft‑max weights)
        conv_full = np.fft.irfft(A * A, n=N)[:conv_len]
        M = conv_full.max()

        # linearly increase the temperature parameter α
        elapsed = time.time() - start_time
        total = deadline - start_time
        alpha = start_alpha + (end_alpha - start_alpha) * (elapsed / total)

        # soft‑max weights (numerically stable)
        diff = conv_full - M                       # ≤ 0
        w = np.exp(alpha * diff)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_len)
        else:
            w /= w_sum

        # gradient = 2·corr(a, w)   (real correlation via FFT)
        W = np.fft.rfft(w, n=N)
        grad = 2.0 * np.fft.irfft(A * np.conj(W), n=N)[:n]

        # step size schedule
        step = step0 / math.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)

        # track improvement on the true max‑convolution
        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.008 / math.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (fine‑tuning)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                 # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

        it += 1

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of a few seconds
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')
    best_a = None

    # -----------------------------------------------------------------
    # 0. Optional fallback
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq, best_a = val, fallback, np.array(fallback)
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Parametric sweep (≈200 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 500.0
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            seq = a_initial.tolist()
            val = evaluate_sequence(seq)
            print(f"[parametric sweep] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq, best_a = val, seq, a_initial

    # -----------------------------------------------------------------
    # 2. Smooth‑gradient descent (bulk of remaining time)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 180.0
    if best_a is not None and time.time() < smooth_deadline:
        a_polished = smooth_descent(
            best_a,
            start_time=start_time,
            deadline=smooth_deadline,
            start_alpha=5.0,
            end_alpha=30.0,
            step0=0.3
        )
        seq = a_polished.tolist()
        val = evaluate_sequence(seq)
        print(f"[smooth‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq, best_a = val, seq, a_polished

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_a is not None and time.time() < hill_deadline:
        a_final = hill_climb(np.array(best_seq, dtype=float), hill_deadline)
        seq = a_final.tolist()
        val = evaluate_sequence(seq)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, seq

    # -----------------------------------------------------------------
    # Safeguard – fall back to a solid analytic construction
    # -----------------------------------------------------------------
    if best_seq is None:
        # a well‑behaved reverse‑power‑law (p≈1.85) is a reliable fallback
        a_fallback = reverse_power_law(n, 1.85)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
