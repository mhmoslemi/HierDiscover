"""
search_step_function_v3.py

A faster and more precise optimiser for the ``evaluate_sequence`` objective.

Main ideas
----------

*  Fine‑tune a *single* parametric family – a reverse power‑law
   a_i ∝ (n‑i)^p – using a cheap golden‑section line search.
   The previous code only scanned p on a coarse grid; a
   sub‑percent improvement in p is enough to beat the 1.503
   target.

*  Small mixtures of the reverse power‑law with a uniform background
   sometimes lower the peak at index 0 (a0²) without hurting the
   interior.  A tiny weight (≤0.15) is explored on a very cheap grid.

*  A longer projected sub‑gradient polishing phase with a larger
   initial step size (0.4) and a simple momentum term.

*  A stochastic hill‑climb with simulated‑annealing style acceptance
   (occasionally accepting non‑improving moves) that runs for the
   rest of the time budget.  Several restarts are performed if time
   permits.

All heavy lifting is done with NumPy (FFT‑based convolution) and the
``evaluate_sequence`` function is only called at the very end (and to
verify the best candidate).  The algorithm respects the 1000 s
budget – a 5 s safety margin is kept.

The entry point required by the harness is ``propose_candidate``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # tie‑break: uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law).
    p ≥ 0.
    """
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def uniform_vec(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def mix_vectors(v1: np.ndarray, v2: np.ndarray, w: float) -> np.ndarray:
    """
    Convex mixture w·v1 + (1‑w)·v2, projected back to the simplex.
    w∈[0,1].
    """
    a = w * v1 + (1.0 - w) * v2
    a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Fine search for the reverse‑power‑law exponent
# ----------------------------------------------------------------------
def golden_section_minimize(func, lo: float, hi: float, tol: float = 1e-5,
                            max_iter: int = 60):
    """Univariate minimisation on a bounded interval (no SciPy)."""
    phi = (math.sqrt(5) - 1) / 2  # ≈0.618
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = func(c)
    fd = func(d)
    it = 0
    while (hi - lo) > tol and it < max_iter:
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = func(d)
        it += 1
    # return the point with smallest function value among the final candidates
    if fc < fd:
        return c, fc
    else:
        return d, fd


def find_best_reverse_power_law(n: int, time_deadline: float):
    """Locate a near‑optimal exponent p for the reverse power‑law family."""
    # coarse scan to obtain a sensible bracketing interval
    best_p = None
    best_val = float('inf')
    for p in np.linspace(0.5, 4.0, 71):   # step ≈0.05
        if time.time() >= time_deadline:
            break
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_p = val, p

    # If we ran out of time during the coarse scan, just return what we have.
    if best_p is None:
        best_p = 1.0

    # refine with golden‑section search inside a ±0.2 window around best_p
    lo = max(0.0, best_p - 0.3)
    hi = best_p + 0.3
    def f(p):
        a = reverse_power_law(n, p)
        return compute_score(a)

    p_opt, val_opt = golden_section_minimize(f, lo, hi, tol=1e-5)
    return p_opt, val_opt

# Sub‑gradient polishing (with a larger initial step and simple momentum)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.4, momentum: float = 0.9) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a)) with momentum."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    v = np.zeros_like(a)            # momentum accumulator
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient (sum of shifted copies for each active index)
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        # momentum update
        v = momentum * v - step * g
        a = a + v
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# Stochastic hill‑climb with simulated‑annealing style acceptance
# ----------------------------------------------------------------------
def hill_climb_anneal(a_start: np.ndarray, time_deadline: float,
                      max_iter: int = 2_000_000) -> np.ndarray:
    """
    Greedy stochastic moves with occasional uphill steps.
    Returns the best vector seen.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.04
    it = 0
    T0 = 2e-3               # initial temperature
    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        if delta == 0.0:
            it += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.04, 0.6)   # become bolder after success
        else:
            # temperature decays linearly with iteration count
            T = T0 * max(0.0, 1.0 - it / max_iter)
            if T > 0 and np.random.rand() < math.exp(-(M_cand - best_M) / (T + 1e-14)):
                a = cand
                step_scale *= 0.99

        it += 1

        # every 150 k iterations we jitter around the best solution
        if it % 150_000 == 0:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)

    return best_a
