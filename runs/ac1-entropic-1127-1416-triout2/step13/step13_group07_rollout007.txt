    cur = start_a.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    temp = init_temp
    n = cur.size
    while time.time() < deadline:
        # propose neighbor: pick i, j differ, move mass
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i: j+=1
        delta = temp * np.random.rand() * cur[i] # small fraction
        if delta == 0: continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonnegative
        if cand[i] < 0: 
            # revert proportionally
            delta = cur[i]  # move all
            cand[i] = 0
            cand[j] = cur[j] + delta
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score)/temp):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{time.time():.1f}] SA improved to {best_score:.6f}")
        temp = max(temp * cooling, 1e-6)
    return best, best_score

"""
propose_candidate
-----------------
Searches for a non‑negative step‑function (probability vector) that
minimises the objective used in the competition:

    2·n·max(conv(a,a)) / (sum(a)**2)

The procedure mixes three phases:

1. **Global exploration** – deterministic scans of several analytic
   families (product‑power, left‑heavy power, exponential‑U, mixtures
   with uniform, and the two‑parameter Beta family) plus a large
   random sampling of the same families, with occasional right‑shifts.

2. **Local refinement** – hill‑climbing of the best parameters of the
   product and Beta families, a smooth‑max Adam optimisation (the loss
   is a log‑sum‑exp surrogate of the max‑convolution), a projected
   sub‑gradient descent on the true max‑objective, and finally a
   simulated‑annealing / greedy hill‑climb that moves small amounts of
   mass between entries.

3. **Safety fallback** – if nothing is found a uniform vector is used.

All steps respect a common 1000 s wall‑clock limit; the best sequence
found is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)**2).
    a is assumed non‑negative and summed to 1 (simplex).
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families (all return a normalised vector on the simplex)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_product_family(n: int, lam: float, alpha: float) -> np.ndarray:
    """Convex mixture of uniform and symmetric‑product family."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    prod = symmetric_product_family(n, alpha)
    w = lam * uniform + (1.0 - lam) * prod
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Discrete Beta‑type density.
    p,q>0.  When p,q<1 the shape is U‑shaped; p,q>1 gives a bell.
    """
    i = np.arange(n, dtype=np.float64)
    # Use logs to avoid overflow for extreme p,q
    logw = (p - 1.0) * np.log(i + 0.5) + (q - 1.0) * np.log(n - 0.5 - i)
    w = np.exp(logw)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)


def random_shift(vec: np.ndarray, max_shift: int) -> np.ndarray:
    """Right shift by a random amount ≤max_shift (fill with zeros)."""
    if max_shift <= 0:
        return vec
    d = np.random.randint(0, max_shift + 1)
    if d == 0:
        return vec
    shifted = np.concatenate([vec[d:], np.zeros(d, dtype=vec.dtype)])
    return normalize(shifted)

def hill_climb_alpha(alpha0: float,
                     n: int,
                     deadline: float,
                     step0: float = 0.8,
                     min_step: float = 1e-4) -> float:
    """One‑dimensional hill‑climb on the product‑family parameter."""
    best = max(0.0, alpha0)
    best_score = compute_score(symmetric_product_family(n, best))
    step = step0
    while time.time() < deadline and step > min_step:
        improved = False
        for sign in (-1, 1):
            cand = max(0.0, best + sign * step)
            sc = compute_score(symmetric_product_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] α refined to {best:.6f} → {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta(p0: float,
                    q0: float,
                    n: int,
                    deadline: float,
                    step0: float = 0.4,
                    min_step: float = 1e-3) -> Tuple[float, float]:
    """Hill‑climb on the two‑parameter Beta family."""
    best_p, best_q = max(0.01, p0), max(0.01, q0)
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = step0
    while time.time() < deadline and step > min_step:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] β refined to p={best_p:.6f}, q={best_q:.6f} → {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q

def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  max_iters: int = 500_000,
                  beta_start: float = 5.0,
                  beta_end: float = 20_000.0,
                  lr: float = 0.015,
                  sym_interval: int = 1) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    ``sym_interval`` forces symmetry every *sym_interval* steps.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise in log‑space to guarantee positivity
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = normalize(best_a)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = normalize(a)

        # optional symmetry enforcement
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = normalize(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective check
        if it % 5000 == 0:
            a_tmp = normalize(np.exp(z))
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = normalize(np.exp(z))
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                    deadline: float,
                    max_iters: int = 1_500_000,
                    step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = normalize(start_a.copy())
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def simulated_annealing_optimize(start_a: np.ndarray,
                                 deadline: float,
                                 init_temp: float = 0.02,
                                 cooling: float = 0.9995,
                                 min_temp: float = 1e-6,
                                 max_steps: int = 300_000) -> Tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing / greedy hill‑climb.
    Moves a small fraction of mass from a donor index to a receiver index.
    """
    n = start_a.shape[0]
    cur = normalize(start_a.copy())
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    temp = init_temp
    step_scale = 0.15  # initial transfer fraction

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # choose donor preferably among large entries
        donor = np.random.choice(n, p=cur)  # probability proportional to mass
        receiver = np.random.randint(n)
        if donor == receiver:
            continue
        delta = step_scale * cur[donor]  # amount to move
        cand = cur.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{time.time():.1f}] SA improved to {best_score:.6f}")
            # slowly increase step size when successful (but cap)
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            step_scale = max(step_scale * 0.97, 1e-4)

        temp = max(temp * cooling, min_temp)

        if step % 50_000 == 0:
            print(f"[{time.time():.1f}] SA step {step}, best {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    competition objective under a fixed wall‑clock budget.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # safety margin for final eval

    # final length – the competition expects a few thousand entries.
    L = 2048

    best_seq: Optional[np.ndarray] = None
    best_score = float("inf")
    best_alpha: Optional[float] = None
    best_beta_params: Optional[Tuple[float, float]] = None

    # ------------------------------------------------------------------
    # 0. optional fallback (if a known construction is available)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the outer scope
        fallback = globals()["height_sequence_1"]
        fb = evaluate_sequence([float(x) for x in fallback])
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array([float(x) for x in fallback], dtype=np.float64)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. deterministic scans of analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Starting deterministic scans (n={L})")

    # product family
    for alpha in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 20.0:
            break
        vec = symmetric_product_family(L, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best product α={alpha:.4f} → {best_score:.6f}")
        # random right‑shifts
        for _ in range(2):
            shifted = random_shift(vec, L // 30)
            sc2 = compute_score(shifted)
            if sc2 < best_score:
                best_score, best_seq = sc2, shifted.copy()
                print(f"[{time.time():.1f}] shifted product α={alpha:.4f} → {best_score:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 20.0:
            break
        vec = left_heavy_power_family(L, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            print(f"[{time.time():.1f}] new best left‑heavy β={beta:.4f} → {best_score:.6f}")

    # exponential‑U family
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 20.0:
            break
        vec = exp_u_family(L, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            print(f"[{time.time():.1f}] new best exp‑U γ={gamma:.4f} → {best_score:.6f}")

    # mixture uniform + product
    for lam in np.linspace(0.0, 1.0, 6):
        for alpha in np.linspace(0.0, 15.0, 11):
            if time.time() >= deadline - 20.0:
                break
            vec = mixed_uniform_product_family(L, lam, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.copy()
                print(f"[{time.time():.1f}] new best mixU prod λ={lam:.2f},α={alpha:.2f} → {best_score:.6f}")

    # mixture uniform + exponential‑U
    for lam in np.linspace(0.0, 1.0, 6):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 20.0:
                break
            vec = mixed_uniform_u_family(L, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.copy()
                print(f"[{time.time():.1f}] new best mixU exp λ={lam:.2f},γ={gamma:.2f} → {best_score:.6f}")

    # Beta family grid
    for p in np.linspace(0.2, 2.0, 10):
        for q in np.linspace(0.2, 2.0, 10):
            if time.time() >= deadline - 20.0:
                break
            vec = beta_family(L, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.copy()
                best_beta_params = (p, q)
                print(f"[{time.time():.1f}] new best Beta p={p:.3f},q={q:.3f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random dense exploration of families
    # ------------------------------------------------------------------
    trials = 0
    max_trials = 8000
    while time.time() < deadline - 15.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(5)
        if choice == 0:          # product family
            alpha = np.random.uniform(0.0, 15.0)
            vec = symmetric_product_family(L, alpha)
        elif choice == 1:        # beta family (log‑uniform)
            # sample p,q log‑uniform between 1e‑3 and 3
            logp = np.random.uniform(-3.0, 0.5)
            logq = np.random.uniform(-3.0, 0.5)
            p, q = 10 ** logp, 10 ** logq
            vec = beta_family(L, max(p, 0.01), max(q, 0.01))
        elif choice == 2:        # left‑heavy power
            beta = np.random.uniform(0.0, 15.0)
            vec = left_heavy_power_family(L, beta)
        elif choice == 3:        # exponential‑U
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_u_family(L, gamma)
        else:                    # mixture uniform + product
            lam = np.random.rand()
            alpha = np.random.uniform(0.0, 15.0)
            vec = mixed_uniform_product_family(L, lam, alpha)

        # occasional random shift
        if np.random.rand() < 0.2:
            vec = random_shift(vec, L // 30)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement of best product‑α and best Beta parameters
    # ------------------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] refining α (start {best_alpha:.4f})")
        refined_alpha = hill_climb_alpha(best_alpha, L,
                                         deadline - 5.0,
                                         step0=0.8)
        vec = symmetric_product_family(L, refined_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            best_alpha = refined_alpha
            print(f"[{time.time():.1f}] refined α={refined_alpha:.6f} → {best_score:.6f}")

    if best_beta_params is not None and time.time() < deadline - 10.0:
        p0, q0 = best_beta_params
        print(f"[{time.time():.1f}] refining β (start p={p0:.4f}, q={q0:.4f})")
        refined_p, refined_q = hill_climb_beta(p0, q0, L,
                                               deadline - 5.0,
                                               step0=0.4)
        vec = beta_family(L, refined_p, refined_q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            best_beta_params = (refined_p, refined_q)
            print(f"[{time.time():.1f}] refined β p={refined_p:.6f}, q={refined_q:.6f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam (smooth‑max) refinement – try a couple of restarts
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        # First run from the current best
        print(f"[{time.time():.1f}] Adam refinement (from best)")
        a_opt, sc_opt = adam_optimize(best_seq,
                                      deadline - 80.0,
                                      max_iters=400_000,
                                      beta_start=5.0,
                                      beta_end=20000.0,
                                      lr=0.015,
                                      sym_interval=1)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] Adam improved best → {best_score:.6f}")

        # Second restart from a uniform start (if time permits)
        if time.time() < deadline - 50.0:
            print(f"[{time.time():.1f}] Adam restart from uniform")
            uniform_start = np.full(L, 1.0 / L, dtype=np.float64)
            a_opt2, sc_opt2 = adam_optimize(uniform_start,
                                            deadline - 20.0,
                                            max_iters=300_000,
                                            beta_start=5.0,
                                            beta_end=20000.0,
                                            lr=0.015,
                                            sym_interval=1)
            if sc_opt2 < best_score:
                best_score, best_seq = sc_opt2, a_opt2.copy()
                print(f"[{time.time():.1f}] Adam (uniform) improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent on the true max‑objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(best_seq,
                                         deadline - 30.0,
                                         max_iters=1_200_000,
                                         step0=0.045)
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.copy()
            print(f"[{time.time():.1f}] sub‑grad improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing / greedy hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final simulated‑annealing")
        a_final, sc_final = simulated_annealing_optimize(best_seq,
                                                         deadline,
                                                         init_temp=0.02,
                                                         cooling=0.9997,
                                                         max_steps=200_000)
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.copy()
            print(f"[{time.time():.1f}] SA improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(L, 1.0 / L, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] fallback uniform used, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation (includes all security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq.tolist()
