"""
Improved search for a non‑negative step‑function that minimizes
evaluate_sequence (i.e. 2·n·max(conv(a,a)) when Σa = 1).

Algorithm
---------
1. **Fast global sweep** – a reduced sweep over power‑law,
   exponential, beta and Gaussian families together with a modest
   number of Dirichlet draws.  The top few candidates are kept.

2. **Linear‑programming refinement** – for a candidate a we solve
   the LP

       minimise   t
       s.t.       (a * b)_k ≤ t   for all k                (cross‑convolution)
                  Σ b_j = 1 ,   b_j ≥ 0                (simplex)

   The solution b is the best possible partner of a that stays under
   the current worst cross‑convolution value.  We then mix a and b
   (line‑search on the mixing coefficient) and repeat the LP step.
   This convex “alternating projection” quickly drives the maximum
   autocorrelation down.

3. **Light Adam fine‑tuning** – a few hundred Adam steps on the
   soft‑max parametrisation are applied to polish the result.

All phases respect a 1000 s wall‑clock budget and keep the sequence
on the simplex (Σ = 1, entries non‑negative).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for n up to a few k)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact (un‑scaled) objective: 2·n·max(conv)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a).  Input is assumed to be non‑negative and Σa = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Helper – build the linear convolution matrix for a fixed left factor
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build matrix A ∈ ℝ^{(2n‑1)×n} such that   (a * b) = A @ b .
    For each column i, the slice A[i:i+n, i] = a[i] and the rest is zero.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A


# ----------------------------------------------------------------------
# Linear‑programming based refinement
# ----------------------------------------------------------------------


def lp_refine(a0: np.ndarray, deadline: float, max_iters: int = 30) -> np.ndarray:
    """
    Iteratively improve a sequence by solving a small LP.
    Each iteration:
        – minimise t subject to (a * b)_k ≤ t,
        – b lies on the simplex,
        – mix the current a and b (line‑search) and keep the improvement.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = compute_score(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- 1) build the convolution matrix for the *current* a ----
        A_mat = build_conv_matrix(a)                     # shape (2n‑1, n)

        # ---- 2) formulate the LP ---------------------------------
        # variables: [b_0 … b_{n‑1}  t]
        rows = 2 * n - 1
        A_ub = np.concatenate([A_mat, -np.ones((rows, 1))], axis=1)
        b_ub = np.zeros(rows)

        A_eq = np.concatenate([np.ones((1, n)), np.zeros((1, 1))], axis=1)
        b_eq = np.array([1.0])

        c = np.concatenate([np.zeros(n), np.array([1.0])])   # minimise t

        bounds = [(0.0, None)] * n + [(0.0, None)]

        # ---- 3) solve ------------------------------------------------
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # LP failed – give up on further refinement
            print("[lp_refine] LP failed at iter", it)
            break

        b_opt = res.x[:n]
        # numerical drift – enforce simplex again
        b_opt = np.maximum(b_opt, 0.0)
        b_opt /= b_opt.sum()

        # ---- 4) line‑search on the mixing coefficient --------------
        # evaluate a few λ ∈ [0,1] and keep the best mixture
        lambdas = np.linspace(0.0, 1.0, 11)   # 0, 0.1, …, 1.0
        best_local_score = best_score
        best_mix = a

        for lam in lambdas:
            cand = (1.0 - lam) * a + lam * b_opt
            sc = compute_score(cand)
            if sc < best_local_score:
                best_local_score = sc
                best_mix = cand.copy()

        # ---- 5) accept improvement or stop -------------------------
        if best_local_score < best_score - 1e-9:
            a = best_mix
            best_score = best_local_score
            print(f"[lp_refine] iter {it:3d}  score {best_score:.6f}")
        else:
            # no improvement – stop early
            break

    return a


# ----------------------------------------------------------------------
# Light Adam optimiser on the soft‑max parametrisation (few thousand iters)
# ----------------------------------------------------------------------


def smooth_optimize_small(a0: np.ndarray, deadline: float,
                          init_lr: float = 0.12,
                          init_tau_factor: float = 5.0,
                          min_tau: float = 1e-9,
                          tau_decay: float = 0.998) -> np.ndarray:
    """
    Very short Adam run (≈ a few thousand steps) to polish a sequence.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits (log‑softmax trick)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initial temperature (smooth max)
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max → a
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution & soft‑max weights over the conv vector
        conv = full_convolution(a)
        max_c = conv.max()
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a (see derivation in the analysis)
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)            # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # chain rule to logits
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # occasional exact check
        if it % 500 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_small] iter {it:5d}  score {val:.6f}")

        # stop early if we are out of time
        if time.time() >= deadline:
            break

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Very short Metropolis‑style hill‑climb."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Global exploration (light‑weight version of the earlier baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 6):
    """Sweep a few parametric families and a modest set of Dirichlet draws."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec /= s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- power‑law -------------------------------------------------
    for p in np.linspace(0.1, 2.0, 120):
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        try_candidate(a)

    # ---- exponential ------------------------------------------------
    for lam in np.linspace(0.0001, 0.04, 180):
        if time.time() >= deadline:
            break
        a = np.exp(-lam * np.arange(n, dtype=float))
        a /= a.sum()
        try_candidate(a)

    # ---- beta -------------------------------------------------------
    al_vals = np.linspace(0.2, 5.0, 12)
    be_vals = np.linspace(0.2, 5.0, 12)
    i = np.arange(1, n + 1, dtype=float)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            left = np.power(i, alpha - 1.0)
            right = np.power(n - i + 1.0, beta - 1.0)
            a = left * right
            if not np.all(np.isfinite(a)):
                continue
            a /= a.sum()
            try_candidate(a)

    # ---- Gaussian ----------------------------------------------------
    for sigma in np.linspace(n / 20.0, n / 2.0, 20):
        if time.time() >= deadline:
            break
        x = np.arange(n, dtype=float)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        a /= a.sum()
        try_candidate(a)

    # ---- Dirichlet draws --------------------------------------------
    concs = [0.3, 0.6, 1.0, 2.0, 5.0]
    draws_per_conc = 500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # problem dimension – 2048 balances expressiveness & speed
    n = 2048

    best_seq = None
    best_val = float("inf")

    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Light global sweep (≈ 120 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 500.0
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=6)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. LP‑based refinement on the current best (≈ 300 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        a_start = np.array(best_seq, dtype=float)
        refine_deadline = deadline - 250.0
        refined = lp_refine(a_start, refine_deadline, max_iters=30)
        val = compute_score(refined)
        print(f"[lp_refine‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # a few random restarts to escape possible local minima
        while time.time() < deadline - 250.0:
            remaining = deadline - time.time() - 250.0
            if remaining <= 0:
                break
            sub_deadline = time.time() + min(30.0, remaining)
            # random Dirichlet start (varied concentration)
            conc = np.random.choice([0.1, 0.3, 0.7, 1.5, 3.0])
            a0 = np.random.dirichlet(np.full(n, conc))
            refined = lp_refine(a0, sub_deadline, max_iters=25)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[lp_refine‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Short Adam polish (≈ 150 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        a0 = np.array(best_seq, dtype=float)
        polish_deadline = deadline - 100.0
        refined = smooth_optimize_small(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[adam‑polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing pass (≈  80 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a0 = np.array(best_seq, dtype=float)
        final = simulated_annealing(a0, deadline - 5.0)
        val = compute_score(final)
        print(f"[anneal‑final] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:

    # -----------------------------------------------------------------
    # 6. Report official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
