def subgradient_descent(a_init, deadline, max_iters=200000):
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    step = 0.2
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute conv
        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        # early compute score
        # Determine max indices
        # tolerance to include all near max (e.g., diff < 1e-12)
        eps = max_val * 1e-12
        max_idx = np.where(conv >= max_val - eps)[0]
        # compute subgradient
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in max_idx:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2 * a[j[mask]]
        # if any non-zero
        if np.all(g == 0):
            break
        # compute candidate step: try step size = step / (np.linalg.norm(g) + 1e-12)
        eta = step
        for ls in range(10):
            cand = a - eta * g
            cand = np.maximum(cand, 0.0)
            cand = project_to_simplex(cand)
            cand = symmetrize(cand)

                best_a = cand.copy()
                a = cand

def convolution_fft(a, b):
    m = a.size + b.size - 1
    nfft = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv
```

def symmetric_product_family(n, alpha):
    i = np.arange(n, dtype=np.float64)
    # Avoid log of zero
    term1 = np.log(i + 0.5)
    term2 = np.log(n - 0.5 - i)
    logprod = term1 + term2
    w = np.exp(-alpha * logprod)
    # Replace infinities if any
    w[np.isinf(w)] = np.finfo(np.float64).max
    total = w.sum()
    if total == 0:
        w = np.full_like(w, 1.0/n)
    else:
        w = w / total
    return w

def golden_section_optimize(func, lo, hi, max_iter=30):
    phi = (1 + 5**0.5) / 2.0
    inv_phi = 1.0 / phi
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd
```

def subgradient_descent(a_init, deadline, max_iters=200_000):
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    step = 0.1  # initial step magnitude for gradient contribution
    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        iteration += 1
        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        # indices achieving max
        eps = max_val * 1e-12
        ks = np.where(conv >= max_val - eps)[0]
        # compute subgradient
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in ks:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        if np.all(g == 0):
            break
        # line search
        eta = step
        improved = False
        for _ in range(12):
            if time.time() >= deadline:
                break
            cand = a - eta * g
            # projection onto simplex ensures sum=1
            cand = np.maximum(cand, 0.0)
            cand = project_to_simplex(cand)
            cand = symmetrize(cand)
            sc = compute_score(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
                a = cand
                # increase step a bit for next iteration
                step = eta * 1.2
                improved = True
                break
            else:
                eta *= 0.5
        if not improved:
            # backtrack step
            step *= 0.5
            # occasional random perturbation to escape
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * 1e-5
                cand = a + jitter
                cand = np.maximum(cand, 0.0)
                cand = project_to_simplex(cand)
                cand = symmetrize(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    a = cand
    return best_a, best_score

def simulated_annealing(a_start, deadline, max_steps=5_000_000):
    n = a_start.shape[0]
    cur = a_start.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    T0 = cur_score * 1e-3  # temperature factor
    T = T0
    step_scale = 0.1
    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # perform move
        cand = cur.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        # project and symmetrize
        cand = project_to_simplex(cand)
        cand = symmetrize(cand)
        cand_score = compute_score(cand)
        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97
        # linear cooling
        T = T0 * (1.0 - step / max_steps)
    return best, best_score
```

"""
Propose a step‑function sequence that minimizes
evaluate_sequence(...).

Algorithm outline
-----------------
0. Load the previously‑found best sequence (height_sequence_1) if it exists.
1. Scan a few analytic families (U‑shaped product, beta) and
   refine the most promising parameters with a golden‑section line search.
2. Refine the current best candidate with a Euclidean projected sub‑gradient
   descent that always keeps the vector on the probability simplex and
   enforces symmetry (a_i = a_{n‑1‑i}).
   A simple back‑tracking line search is used to adapt the step size.
3. Finish with a modest simulated‑annealing hill‑climb that moves small
   amounts of mass between entries while keeping the simplex constraint.
4. Return the best sequence (as a plain list) after a final call to
   evaluate_sequence (which performs the required security checks).

The whole search respects a user‑supplied time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (sum = 1, v ≥ 0).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i}."""
    return (v + v[::-1]) / 2.0


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    total = w.sum()
    if total == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Objective wrappers for golden‑section searches
# ----------------------------------------------------------------------


def product_score(alpha: float, n: int) -> float:
    """Score of the symmetric product family with parameter alpha."""
    return compute_score(symmetric_product_family(n, alpha))


def beta_score(p: float, q: float, n: int) -> float:
    """Score of the beta family with parameters (p,q)."""
    return compute_score(beta_family(n, p, q))


# ----------------------------------------------------------------------
# One‑dimensional golden‑section optimizer
# ----------------------------------------------------------------------


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 30):
    """Golden‑section search for the minimum of func on [lo,hi]."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the simplex (with symmetry)
# ----------------------------------------------------------------------


def subgradient_descent(a_init: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the simplex.
    Uses the exact sub‑gradient of max(conv) at the current maximiser(s)
    and a simple back‑tracking line search.
    """
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    step = 0.1  # initial step size (will be scaled automatically)

    itr = 0
    while time.time() < deadline and itr < max_iters:
        itr += 1

        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        # indices that are (numerically) maximal
        eps = max_val * 1e-12
        ks = np.where(conv >= max_val - eps)[0]

        # compute sub‑gradient (sum of contributions from all maximal ks)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in ks:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        if np.all(g == 0):
            break

        # back‑tracking line search
        eta = step
        improved = False
        for _ in range(12):
            if time.time() >= deadline:
                break
            cand = a - eta * g
            cand = np.maximum(cand, 0.0)
            cand = project_to_simplex(cand)
            cand = symmetrize(cand)
            sc = compute_score(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
                a = cand
                step = eta * 1.2              # modestly increase step size
                improved = True
                break
            else:
                eta *= 0.5                     # shrink step size and retry

        if not improved:
            step *= 0.5                         # overall step size decay
            # occasional random jitter to escape shallow plateaus
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * 1e-5
                cand = a + jitter
                cand = np.maximum(cand, 0.0)
                cand = project_to_simplex(cand)
                cand = symmetrize(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        max_steps: int = 4_000_000,
                        temp_factor: float = 1.0e-3,
                        step_scale_init: float = 0.25) -> tuple[np.ndarray, float]:
    """SA on the simplex with symmetry enforcement."""
    n = a_start.size
    cur = a_start.copy()
    cur_score = compute_score(cur)

    best = cur.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur.copy()
        # move type: swap or transfer a fraction of mass
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # restore constraints
        cand = project_to_simplex(cand)
        cand = symmetrize(cand)

        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                step_scale = min(step_scale * 1.05, 0.5)   # gradually enlarge moves
        else:
            step_scale *= 0.97

        # linear cooling schedule
        T = T0 * (1.0 - step / max_steps)

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence(...).  Returns the best sequence found as a plain list.
    The routine respects a wall‑clock budget (default 1000 s).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Fallback to previous construction if available
    # -----------------------------------------------------------
    height_seq_1 = globals().get("height_sequence_1")
    if height_seq_1 is not None:
        try:
            seq0 = [float(x) for x in height_seq_1]
            sc0 = evaluate_sequence(seq0)
            best_seq = seq0
            best_score = sc0
            print(f"[{time.time():.1f}] fallback (height_sequence_1) score = {sc0:.6f}")
        except Exception:
            pass

    # -----------------------------------------------------------
    # 1. Analytic family scan (product & beta)
    # -----------------------------------------------------------
    n = 8192
    print(f"[{time.time():.1f}] scanning analytic families (n={n})")

    # ----- product family (U‑shaped) -----
    print(f"[{time.time():.1f}] golden‑section search on product family")
    a_opt, sc_opt = golden_section_optimize(
        lambda a: product_score(a, n), 0.0, 15.0, max_iter=30
    )
    vec = symmetric_product_family(n, a_opt)
    sc = compute_score(vec)
    if sc < best_score:
        best_score, best_seq = sc, vec.tolist()
        print(f"[{time.time():.1f}] new best product α={a_opt:.6f} → {sc:.6f}")

    # ----- beta family (two‑parameter) -----
    print(f"[{time.time():.1f}] random beta search")
    p_best, q_best = 1.0, 1.0
    for _ in range(250):
        if time.time() >= deadline - 30.0:
            break
        p = np.random.uniform(0.1, 4.0)
        q = np.random.uniform(0.1, 4.0)
        vec = beta_family(n, p, q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            p_best, q_best = p, q
            print(f"[{time.time():.1f}] beta random p={p:.3f}, q={q:.3f} → {sc:.6f}")

    # Local refinement of the best beta parameters
    def beta_refinement(p0, q0, time_limit):
        p, q = p0, q0
        step_p, step_q = 0.5, 0.5
        while time.time() < time_limit:
            improved = False
            for dp in (-step_p, step_p):
                p_new = p + dp
                if p_new <= 0:
                    continue
                sc = compute_score(beta_family(n, p_new, q))
                if sc < best_score:
                    best_score, best_seq = sc, beta_family(n, p_new, q).tolist()
                    p = p_new
                    improved = True
                    print(f"[{time.time():.1f}] beta refine p={p:.4f} → {sc:.6f}")
                    break
            if improved:
                continue
            for dq in (-step_q, step_q):
                q_new = q + dq
                if q_new <= 0:
                    continue
                sc = compute_score(beta_family(n, p, q_new))
                if sc < best_score:
                    best_score, best_seq = sc, beta_family(n, p, q_new).tolist()
                    q = q_new
                    improved = True
                    print(f"[{time.time():.1f}] beta refine q={q:.4f} → {sc:.6f}")
                    break
            if not improved:
                step_p *= 0.5
                step_q *= 0.5
                if step_p < 1e-4:
                    break

    beta_refinement(p_best, q_best, deadline - 20.0)

    # -----------------------------------------------------------
    # 2. Projected sub‑gradient descent refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] starting sub‑gradient descent (current best {best_score:.6f})")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = subgradient_descent(
            a_start,
            deadline - 10.0,
            max_iters=200_000,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after sub‑gradient descent best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb (best {best_score:.6f})")
        a_start = np.array(best_seq, dtype=np.float64)
        a_fin, sc_fin = simulated_annealing(
            a_start,
            deadline,
            max_steps=4_000_000,
            temp_factor=1.0e-3,
            step_scale_init=0.25,
        )
        if sc_fin < best_score:
            best_score, best_seq = sc_fin, a_fin.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Fallback to a uniform vector if everything failed
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes the security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
