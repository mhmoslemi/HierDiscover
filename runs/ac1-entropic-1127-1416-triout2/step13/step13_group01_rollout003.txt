def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Global exploration** – sweep several parametric families (power‑law,
   exponential, beta, Gaussian, triangle, sparse) and many Dirichlet draws.
   Keep the best few candidates.

2. **Smooth max optimisation** – minimise a smooth soft‑max approximation of
   the convolution max using Adam on the log‑softmax (exponentiated‑gradient)
   parametrisation.  The temperature τ is annealed very slowly (τ_dec = 0.9999)
   so the optimiser can escape shallow plateaus.

3. **Hard‑max projected gradient descent** – after the smooth phase,
   continue the search with a true max sub‑gradient (weights are a uniform
   distribution over the currently maximal convolution entries) and project
   back onto the probability simplex after each step.  This fine‑tunes the
   solution to a (possibly) better local minimum.

4. **Optional simulated‑annealing polish** – a short Metropolis‑style hill‑climb
   to escape any remaining tiny plateaux.

All work is performed on the *simplex* (∑a_i = 1, a_i ≥ 0), because the
evaluation function is invariant under uniform scaling (apart from clipping,
which never triggers for the generated vectors).  The internal objective
``compute_score`` therefore evaluates ``2·n·max(conv(a,a))`` directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real inputs)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution)."""
    return _linear_convolution(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective used during the search: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # All entries are zero or negative – return the uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return a / a.sum()


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a / a.sum()


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 1.0 / n)
    return a / a.sum()


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a / a.sum()


def generate_triangle(n: int, width: float) -> np.ndarray:
    """Symmetric triangle centred at the middle, width = half‑base length."""
    mu = (n - 1) / 2.0
    a = np.maximum(0.0, 1.0 - np.abs(np.arange(n, dtype=float) - mu) / width)
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def generate_sparse(n: int, k: int) -> np.ndarray:
    """k equally‑weighted spikes placed uniformly at random."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / len(idx)
    return a


# ----------------------------------------------------------------------
# Global baseline exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """Sweep families + random Dirichlet draws, keep the best `max_keep`."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        # Guard against NaNs / infinities – though families should not produce them.
        if not np.isfinite(vec).all():
            return
        # Normalise onto the simplex (non‑negative is already guaranteed)
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- power‑law ----------------------------------------------------
    for p in np.linspace(0.1, 3.0, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ----- exponential --------------------------------------------------
    for lam in np.linspace(0.00001, 0.08, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ----- beta ---------------------------------------------------------
    al_vals = np.linspace(0.2, 8.0, 20)
    be_vals = np.linspace(0.2, 8.0, 20)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ----- Gaussian ------------------------------------------------------
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ----- triangle ------------------------------------------------------
    width_vals = np.linspace(n / 64.0, n / 2.0, 25)
    for w in width_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_triangle(n, w))

    # ----- sparse --------------------------------------------------------
    for k in [2, 3, 5, 10, 20, 50, 100, 200]:
        if time.time() >= deadline:
            break
        for _ in range(50):
            if time.time() >= deadline:
                break
            try_candidate(generate_sparse(n, k))

    # ----- Dirichlet draws -----------------------------------------------
    concs = [0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per = 800
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.2,
                    init_tau_factor: float = 30.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9999) -> np.ndarray:
    """
    Minimise a smooth approximation of the max‑convolution objective.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise x = log a (tiny epsilon for safety)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature τ (based on current max)
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping (KL‑mirror)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution of the current a
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- smooth max weights
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient w.r.t. a (g_i = 2·∑_k w_k·a_{k-i})
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)  # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the soft‑max (x‑space)
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # ----- periodic exact evaluation
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Hard‑max projected gradient descent (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def hard_max_descent(a0: np.ndarray, deadline: float,
                     init_lr: float = 0.15,
                     min_lr: float = 5e-6,
                     lr_decay: float = 0.99995) -> np.ndarray:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution objective.
    Uses a uniform weight over all currently maximal convolution entries.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    lr = init_lr
    it = 0
    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        max_c = conv.max()
        # indices achieving (near) the maximum
        max_idx = np.where(conv >= max_c - 1e-12)[0]
        w = np.zeros_like(conv)
        if max_idx.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[max_idx] = 1.0 / max_idx.size

        # gradient (same form as in smooth version but with hard weights)
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # projected gradient step (Euclidean projection onto simplex)
        a = a - lr * grad_a
        a = np.maximum(a, 0.0)      # ensure non‑negativity before projection
        a = project_simplex(a)

        # occasional exact evaluation
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[hard_max] iter {it:7d}  score {val:.6f}")

        # decay learning rate
        lr = max(min_lr, lr * lr_decay)

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9999,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = project_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 offers a good trade‑off between expressivity
    # and speed for the 1000 s budget.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline - 400.0, start_time + 150.0)
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smoothed Adam optimisation with restarts (≈ 600 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if best_seq is not None and time.time() < smooth_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = smooth_optimize(a0, smooth_deadline)
        val = compute_score(a_refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

        # Additional random‑restart passes while time permits
        while time.time() < smooth_deadline:
            remaining = smooth_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            # random Dirichlet start (vary concentration)
            conc = np.random.choice([0.05, 0.1, 0.3, 0.7, 1.5, 3.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            a_refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(a_refined)
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Hard‑max projected gradient descent (≈ 150 s)
    # -----------------------------------------------------------------
    hard_deadline = deadline - 50.0
    if best_seq is not None and time.time() < hard_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_refined = hard_max_descent(a_start, hard_deadline)
        val = compute_score(a_refined)
        print(f"[hard‑max] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
