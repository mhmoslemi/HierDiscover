"""
Propose a low‑score step‑function sequence.

The algorithm mixes several cheap global searches over simple analytic families
(product‑power, left‑heavy power, exponential‑U, Beta and a piecewise‑power
family) with dedicated 1‑D refinements, random Dirichlet starts and a few
restarts of a smooth‑max Adam optimiser.  After the heuristic phase a short
projected sub‑gradient descent and a greedy mass‑transfer hill‑climb polish
the best candidate found.  All work is done under the global time budget
(≈ 1000 s) and the returned list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
# Core numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    a is assumed non‑negative; any scaling of a leaves the objective unchanged.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    # Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”.
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} (monotone decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Beta‑type density on the discrete grid.
    p,q>0.  When p,q<1 the shape is U‑shaped; p,q>1 gives a bell.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)


def piecewise_power_family(n: int, split: float, p: float, q: float) -> np.ndarray:
    """
    Asymmetric two‑sided power law.
    split ∈ (0,1) is the fraction of the support belonging to the left side.
    """
    i = np.arange(n, dtype=np.float64)
    left_len = int(round(split * (n - 1)))
    left = i <= left_len
    w = np.empty_like(i)
    w[left] = np.exp(-p * np.log(i[left] + 0.5))
    w[~left] = np.exp(-q * np.log(n - 0.5 - i[~left]))
    return normalize(w)

def hill_climb_1d(func, start: float, lo: float, hi: float,
                  deadline: float, init_step: float = 0.5,
                  min_step: float = 1e-5) -> float:
    """
    Simple pattern search for a unimodal 1‑D function.
    ``func`` must accept a single float and return a score (lower better).
    """
    best = start
    best_score = func(best)
    step = init_step
    while time.time() < deadline and step > min_step:
        improved = False
        for sgn in (-1.0, 1.0):
            cand = best + sgn * step
            cand = min(max(cand, lo), hi)
            sc = func(cand)
            if sc < best_score:
                best, best_score = cand, sc
                improved = True
                print(f"[{time.time():.1f}] 1‑D refine → {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_product_alpha(initial_alpha: float, n: int,

    """Refine the product‑family exponent α."""
    def f(alpha):
        return compute_score(symmetric_product_family(n, alpha))
    return hill_climb_1d(f, initial_alpha, 0.0, 30.0, deadline)

def hill_climb_beta(p0: float, q0: float, n: int,
                    deadline: float) -> Tuple[float, float]:
    """2‑D hill‑climb for the Beta family."""
    best_p, best_q = p0, q0
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = 0.4
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] Beta refined → p={best_p:.6f}, q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q


def hill_climb_piecewise(split0: float, p0: float, q0: float, n: int,

            cq = max(0.01, q + dq)
            sc = compute_score(piecewise_power_family(n, split, p, cq))
            if sc < best_score:
                q, best_score = cq, sc
                improved = True
                print(f"[{time.time():.1f}] piecewise q → {q:.4f} score {best_score:.6f}")
                break
        if not improved:
            step_split *= 0.5
            step_pq *= 0.5
    return split, p, q

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 5.0,
                         beta_end: float = 30_000.0,
                         lr: float = 0.008,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    ``beta`` (temperature) grows from ``beta_start`` to ``beta_end``.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 3000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved → {best_score:.6f}")

    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement → {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.07) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved → {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_greedy(start_a: np.ndarray,
                      deadline: float,
                      max_steps: int = 6_000_000,
                      init_scale: float = 0.12) -> Tuple[np.ndarray, float]:
    """
    Greedy hill‑climb moving a fraction of mass from the largest entry
    to a random other entry whenever the objective improves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        i = np.argmax(cur_a)            # donor (largest)
        j = np.random.randint(n)        # receiver
        if i == j:
            continue

        delta = scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
            scale = min(scale * 1.03, 0.5)
        else:
            scale = max(scale * 0.97, 1e-4)

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best candidate found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for final evaluation
    deadline = start_time + budget_s - 5.0

    best_seq: Optional[np.ndarray] = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Try the fallback known construction if it exists
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback baseline score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array([float(x) for x in height_sequence_1], dtype=np.float64)  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working resolution (larger n usually yields a better continuous approximation)
    # ------------------------------------------------------------------
    n = 16384  # power of two to keep FFT cheap

    # ------------------------------------------------------------------
    # 2. Coarse scan of analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] starting coarse scan with n={n}")

    # containers for best parameters
    best_alpha = None
    best_beta = None
    best_gamma = None
    best_pq: Tuple[float, float] = (0.5, 0.5)
    best_piecewise: Tuple[float, float, float] = (0.5, 1.0, 1.0)

    # product family
    for alpha in np.linspace(0.0, 20.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq, best_alpha = sc, vec, alpha
            print(f"[{time.time():.1f}] new best (product α={alpha:.4f}) → {best_score:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 20.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq, best_beta = sc, vec, beta
            print(f"[{time.time():.1f}] new best (left‑heavy β={beta:.4f}) → {best_score:.6f}")

    # exponential‑U family
    for gamma in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq, best_gamma = sc, vec, gamma
            print(f"[{time.time():.1f}] new best (exp‑U γ={gamma:.4f}) → {best_score:.6f}")

    # Beta family (grid)
    for p in np.linspace(0.2, 2.5, 13):
        for q in np.linspace(0.2, 2.5, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec
                best_pq = (p, q)
                print(f"[{time.time():.1f}] new best (Beta p={p:.3f}, q={q:.3f}) → {best_score:.6f}")

    # piecewise‑power family (very cheap coarse grid)
    for split in np.linspace(0.2, 0.8, 7):
        for p in np.linspace(0.5, 5.0, 6):
            for q in np.linspace(0.5, 5.0, 6):
                if time.time() >= deadline - 30.0:
                    break
                vec = piecewise_power_family(n, split, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec
                    best_piecewise = (split, p, q)
                    print(f"[{time.time():.1f}] new best (piecewise s={split:.2f}, p={p:.2f}, q={q:.2f}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Random dense search (Dirichlet + mixture families)
    # ------------------------------------------------------------------
    trials = 0
    max_trials = 2500
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(6)
        if choice == 0:  # product family
            alpha = np.random.uniform(0.0, 20.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 1:  # left‑heavy power
            beta = np.random.uniform(0.0, 20.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 2:  # exponential‑U
            gamma = np.random.uniform(0.0, 8.0)
            vec = exp_u_family(n, gamma)
        elif choice == 3:  # Beta family
            p = np.random.uniform(0.01, 3.0)
            q = np.random.uniform(0.01, 3.0)
            vec = beta_family(n, p, q)
        elif choice == 4:  # piecewise power
            split = np.random.uniform(0.2, 0.8)
            p = np.random.uniform(0.1, 6.0)
            q = np.random.uniform(0.1, 6.0)
            vec = piecewise_power_family(n, split, p, q)
        else:  # random Dirichlet (sparse heavy‑tail)
            concentration = np.random.uniform(0.05, 0.5)
            g = np.random.gamma(concentration, 1.0, size=n)
            vec = g / g.sum()

        # occasional random right‑shift to break symmetry
        if np.random.rand() < 0.15:
            d = np.random.randint(0, n // 40 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            print(f"[{time.time():.1f}] random trial #{trials} improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local refinement of the best analytic parameters
    # ------------------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product α (start {best_alpha:.4f})")
        refined_alpha = hill_climb_product_alpha(best_alpha, n, deadline - 10.0)
        vec = symmetric_product_family(n, refined_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq, best_alpha = sc, vec, refined_alpha
            print(f"[{time.time():.1f}] refined product α={refined_alpha:.6f} → {best_score:.6f}")

    if best_beta is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining left‑heavy β (start {best_beta:.4f})")
        refined_beta = hill_climb_left_beta(best_beta, n, deadline - 10.0)
        vec = left_heavy_power_family(n, refined_beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq, best_beta = sc, vec, refined_beta
            print(f"[{time.time():.1f}] refined left‑heavy β={refined_beta:.6f} → {best_score:.6f}")

    # refine beta family if we have an entry for it
    if best_pq != (0.5, 0.5) and time.time() < deadline - 20.0:
        p0, q0 = best_pq
        print(f"[{time.time():.1f}] refining Beta (start p={p0:.4f}, q={q0:.4f})")
        rp, rq = hill_climb_beta(p0, q0, n, deadline - 10.0)
        vec = beta_family(n, rp, rq)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq, best_pq = sc, vec, (rp, rq)
            print(f"[{time.time():.1f}] refined Beta → p={rp:.6f}, q={rq:.6f} score {best_score:.6f}")

    if best_piecewise != (0.5, 1.0, 1.0) and time.time() < deadline - 20.0:
        s0, p0, q0 = best_piecewise
        print(f"[{time.time():.1f}] refining piecewise power (start s={s0:.4f}, p={p0:.4f}, q={q0:.4f})")
        rs, rp, rq = hill_climb_piecewise(s0, p0, q0, n, deadline - 10.0)
        vec = piecewise_power_family(n, rs, rp, rq)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq, best_piecewise = sc, vec, (rs, rp, rq)
            print(f"[{time.time():.1f}] refined piecewise → s={rs:.4f}, p={rp:.4f}, q={rq:.4f} score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Multiple Adam restarts from diverse seeds
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline - 200.0:
                break
            print(f"[{time.time():.1f}] Adam restart {r+1}/{restarts}")
            start_vec = best_seq.copy()
            # inject a small random perturbation to diversify
            noise = np.random.normal(0, 0.02, size=n)
            start_vec = np.maximum(start_vec + noise, 0.0)
            start_vec = normalize(start_vec)
            a_opt, sc_opt = adam_smooth_optimize(
                start_a=start_vec,
                deadline=deadline - 150.0,
                max_iters=300_000,
                beta_start=5.0,
                beta_end=40_000.0,
                lr=0.009,
                sym_interval=0,
            )
            if sc_opt < best_score:
                best_score, best_seq = sc_opt, a_opt
                print(f"[{time.time():.1f}] Adam restart {r+1} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Sub‑gradient descent from the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq.copy(),
            deadline=deadline - 60.0,
            max_iters=1_800_000,
            step0=0.06,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final greedy hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final greedy hill‑climb")
        a_final, sc_final = hill_climb_greedy(
            start_a=best_seq.copy(),
            deadline=deadline,
            max_steps=6_000_000,
            init_scale=0.12,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final
            print(f"[{time.time():.1f}] greedy improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Fallback to a simple uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 512
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = compute_score(best_seq)
        print(f"[{time.time():.1f}] fallback to uniform, score = {best_score:.6f}")

    # final official evaluation (includes security checks)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq.tolist()
