"""
Search for a non‑negative step‑height vector that minimizes
evaluate_sequence (i.e. 2·n·max(conv(a,a)) with sum(a)=1).

The algorithm consists of several stages, each respecting the global
wall‑clock budget (default 1000 s):
  1) Broad generation of candidates from many parametric families and
     random Dirichlet draws (keeps the best few).
  2) Projected Adam minimisation of a smooth approximation of the
     max‑convolution (τ·log‑sum‑exp) with an exponential τ‑schedule.
  3) Projected sub‑gradient descent on the exact max‑convolution.
  4) A directed pair‑wise “flattening’’ heuristic that repeatedly moves
     a small fraction of mass from the index contributing most to the
     current peak towards an index contributing least.
  5) A short simulated‑annealing hill‑climb for final polishing.

All updates are projected onto the probability simplex and the vector
is forced to be symmetric (a_i = a_{n‑1‑i}); symmetry reduces the
search space and empirically improves the bound.  The best sequence
found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n‑1‑i}."""
    return 0.5 * (v + v[::-1])


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for zero‑padding (speed‑up)
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    # Numerical noise may produce tiny negatives – clip for safety
    conv[conv < 0] = 0.0
    return conv


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Returns g_i = 2·Σ_k w_k·a_{k‑i}.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2n‑1
    rev_w = w[::-1]

    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]

    # indices that map to gradient components
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smooth_max(conv: np.ndarray, tau: float):
    """
    Smooth approximation S = τ·log(∑ exp(conv/τ)).
    Returns (S, w) where w are the soft‑max weights (∑ w = 1).
    """
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_ushape(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_two_blocks(n: int, left_frac: float) -> np.ndarray:
    """
    Two uniform blocks: left_frac of the total mass in the first half,
    the rest in the second half.
    """
    a = np.zeros(n, dtype=float)
    half = n // 2
    a[:half] = (left_frac) / half
    a[half:] = (1.0 - left_frac) / (n - half)
    return a

# Global exploration – keep a few promising vectors
# ----------------------------------------------------------------------


def consider_vector(v: np.ndarray, best: list, keep: int):
    """Project, symmetrize, evaluate and possibly keep the vector."""
    if not np.isfinite(v).all():
        return
    v = np.maximum(v, 0.0)
    v = project_to_simplex(v)
    v = symmetrize(v)
    v = project_to_simplex(v)   # re‑project after symmetrisation
    sc = compute_score(v)
    if len(best) < keep:
        best.append((sc, v.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, v.copy())
        best.sort(key=lambda x: x[0])


def explore_candidates(n: int, deadline: float, keep: int = 12) -> list[np.ndarray]:
    """Broad generation of candidates; returns the best `keep` vectors."""
    best = []   # list of (score, vector)

    # 0) Uniform baseline
    consider_vector(np.full(n, 1.0 / n), best, keep)

    # 1) Power‑law (both decreasing/increasing)
    for p in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline:
            break
        consider_vector(generate_powerlaw(n, p), best, keep)
        consider_vector(generate_powerlaw(n, -p), best, keep)

    # 2) Exponential sweep
    for lam in np.linspace(-0.3, 0.3, 61):
        if time.time() >= deadline:
            break
        consider_vector(generate_exponential(n, lam), best, keep)

    # 3) Beta family
    al_vals = np.linspace(0.1, 5.0, 16)
    be_vals = np.linspace(0.1, 5.0, 16)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider_vector(generate_beta(n, a, b), best, keep)

    # 4) Gaussian widths
    for sigma in np.linspace(0.05 * n, 0.5 * n, 12):
        if time.time() >= deadline:
            break
        consider_vector(generate_gaussian(n, sigma), best, keep)

    # 5) Sine‑power shapes
    for p in np.linspace(0.1, 8.0, 16):
        if time.time() >= deadline:
            break
        consider_vector(generate_sine_power(n, p), best, keep)

    # 6) U‑shape families
    for p in np.linspace(0.1, 5.0, 12):
        if time.time() >= deadline:
            break
        consider_vector(generate_ushape(n, p), best, keep)

    # 7) Two‑block mixtures
    for lf in np.linspace(0.0, 1.0, 11):
        if time.time() >= deadline:
            break
        consider_vector(generate_two_blocks(n, lf), best, keep)

    # 8) Random Dirichlet draws with various concentrations
    concentrations = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = max(1, (keep * 200) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            consider_vector(np.random.dirichlet([conc] * n), best, keep)

    # Return the vectors only, sorted increasingly by score
    return [vec for _, vec in sorted(best, key=lambda x: x[0])]


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.5,
                  tau_end: float = 1e-7,
                  tau_decay: float = 0.90,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam minimising τ·log‑sum‑exp(conv) with exponential τ schedule."""
    a = a0.copy()
    n = a.shape[0]
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)          # soft‑max weights
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)
        a = symmetrize(a)
        a = project_to_simplex(a)   # tiny numerical drift fix

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape flats
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (lr * 0.002 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)
            a = symmetrize(a)
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact max) with adaptive step
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        eps = max(M * 1e-9, 1e-12)
        active = np.where(conv >= M - eps)[0]

        # binary weights for the active set
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = conv_with_weights(a, w)      # 2·Σ_active a_{k‑i}

        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)
        a = symmetrize(a)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)
            a = symmetrize(a)
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Pairwise flattening heuristic (direct mass transfer)
# ----------------------------------------------------------------------


def pairwise_flatten(a0: np.ndarray, deadline: float,
                     init_frac: float = 0.02,
                     min_frac: float = 1e-7,
                     decay: float = 0.9995) -> np.ndarray:
    """
    Repeatedly transfers a small fraction of mass from the index that
    contributes most to the current convolution peak to the index that
    contributes least.  The step fraction decays geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    frac = init_frac

    while time.time() < deadline:
        conv = full_convolution(a)
        M = conv.max()
        eps = max(M * 1e-9, 1e-12)
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:
            break

        # binary weights for the active set
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = conv_with_weights(a, w)          # 2·Σ_active a_{k‑i}
        donor = int(np.argmax(grad))
        receiver = int(np.argmin(grad))

        delta = frac * a[donor]
        if delta < min_frac:
            break

        a[donor] -= delta
        a[receiver] += delta

        # enforce non‑negativity (numerical safety)
        if a[donor] < 0:
            a[receiver] += a[donor]   # a[donor] is negative
            a[donor] = 0.0

        a = symmetrize(a)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # decay the transfer fraction slowly
        frac = max(min_frac, frac * decay)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Metropolis hill‑climbing with a temperature schedule."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = symmetrize(cand)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a + jitter
            a = symmetrize(a)
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑height sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin of a few seconds for the final evaluation/printing
    deadline = start + budget_s - 5.0

    n = 2048
    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Known fallback (if the global variable is present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad exploration (≈150 s)
    # -----------------------------------------------------------------
    explore_deadline = min(deadline - 800.0, start + 150.0)
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(n, explore_deadline, keep=15)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score = {best_score:.6f}")
    else:
        cand_vecs = []

    # -----------------------------------------------------------------
    # 2. Adam optimisation from a handful of seeds (≈300 s)
    # -----------------------------------------------------------------
    adam_deadline = min(deadline - 450.0, start + 480.0)
    if best_seq is not None and time.time() < adam_deadline:
        start_points = []

        # best found so far (projected & symmetrised)
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        best_vec = symmetrize(best_vec)
        best_vec = project_to_simplex(best_vec)
        start_points.append(best_vec)

        # uniform seed
        start_points.append(np.full(n, 1.0 / n))

        # a few random seeds
        for _ in range(3):
            rnd = np.random.rand(n)
            rnd = project_to_simplex(rnd)
            rnd = symmetrize(rnd)
            rnd = project_to_simplex(rnd)
            start_points.append(rnd)

        # top candidates from exploration
        for vec in cand_vecs[:4]:
            start_points.append(vec)

        refined_best = best_vec.copy()
        refined_val = best_score

        for idx, a0 in enumerate(start_points):
            if time.time() >= adam_deadline:
                break
            local_deadline = adam_deadline - (time.time() - start) - 0.5
            if local_deadline <= time.time():
                break
            print(f"[adam {idx}] start")
            a_opt = adam_optimize(a0, local_deadline)
            val = compute_score(a_opt)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()

        best_score, best_seq = refined_val, refined_best.tolist()
        print(f"[adam] best after Adam = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈200 s)
    # -----------------------------------------------------------------
    polish_deadline = min(deadline - 230.0, start + 700.0)
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Pairwise flattening heuristic (≈150 s)
    # -----------------------------------------------------------------
    flatten_deadline = min(deadline - 70.0, start + 860.0)
    if best_seq is not None and time.time() < flatten_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[flatten] start")
        a_flat = pairwise_flatten(a0, flatten_deadline)
        val = compute_score(a_flat)
        print(f"[flatten] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_flat.tolist()

    # -----------------------------------------------------------------
    # 5. Simulated annealing (final refinement, rest of budget)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – return uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation and report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
