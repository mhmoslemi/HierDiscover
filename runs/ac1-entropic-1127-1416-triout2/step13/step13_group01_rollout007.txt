import cvxpy as cp

def lp_step(f, M):
    n = f.shape[0]
    b = cp.Variable(n)
    # conv f*b <= M
    # we need to build linear constraints: for each k in range(2*n-1)
    # conv_k = sum_{i} f[i] * b[k-i] with b index valid.
    constraints = [b >= 0]
    # compute convolution matrix
    # use Toeplitz: each shift matrix row?
    # For speed we can precompute indices.
    for k in range(2*n-1):
        # indices i where both i in [0,n) and k-i in [0,n)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # sum_{i=i_min}^{i_max} f[i] * b[k-i]
        # this is dot of f[i_min:i_max+1] and b_rev?
        # implement as cp.sum(cp.multiply(f[i_range], b[k - i_range]))
        # we can use list comprehension
        expr = cp.sum(cp.multiply(f[i_min:i_max+1], b[k - np.arange(i_min, i_max+1)]))
        constraints.append(expr <= M)
    # objective sum(b)
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver='ECOS', warm_start=True, abstol=1e-8, reltol=1e-8, feastol=1e-8)
    return b.value
```

def hill_climb(a, deadline):
    n = len(a)
    cur_val = compute_score(a)
    best_val = cur_val
    best_a = a.copy()
    step = 0.01
    while time.time() < deadline:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        delta = step * a[i]
        cand = a.copy()
        if cand[i] < delta: continue
        cand[i] -= delta
        cand[j] += delta
        cand = cand / cand.sum()  # ensure sum=1 (approx)
        val = compute_score(cand)
        if val < cur_val:
            a = cand
            cur_val = val
            if val < best_val:
                best_val, best_a = val, cand.copy()
        # adapt step
        if val < cur_val:
            step = min(step * 1.02, 0.2)
        else:
            step = max(step * 0.98, 1e-5)
    return best_a
```

"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The algorithm consists of three stages:

1. **Enriched global exploration** – sweep many analytic families
   (power‑law, exponential, beta, Gaussian, Hann/Hamming/Blackman,
   Bartlett, cosine, Kaiser) and a large set of random Dirichlet draws.
   The top‑K candidates (K≈12) are kept.

2. **Annealed soft‑max Adam optimisation** – minimise the smooth proxy
   𝓁τ(a)=τ·log∑ₖexp(conv(a,a)ₖ/τ) using the soft‑max (exponential)
   parametrisation a=softmax(x).  τ is slowly annealed; Adam with
   momentum drives the search.  Several restarts from the best global
   candidates are performed until the time budget is exhausted.

3. **Greedy pairwise hill‑climb** – a cheap deterministic refinement
   that repeatedly transfers a small fraction of mass between two
   coordinates if the exact objective improves.  This removes tiny
   local plateaux that the smooth optimiser may miss.

All steps respect the 1000 s wall‑clock limit and return a plain
Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution (real FFT) – used everywhere
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for ``a*a`` (full convolution)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (no clipping – sum is always 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector, sum == 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    a[0] = a[-1] = 0.0
    return a / a.sum()


def generate_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a[0] = a[-1] = 0.0
    return a / a.sum()


def generate_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a[0] = a[-1] = 0.0
    return a / a.sum()


def generate_bartlett(n: int) -> np.ndarray:
    """Triangular (Bartlett) window."""
    i = np.arange(n, dtype=float)
    a = 1.0 - np.abs((i - (n - 1) / 2.0) / ((n - 1) / 2.0))
    return a / a.sum()


def generate_cosine_window(n: int) -> np.ndarray:
    """Raised cosine (pure cosine) that is zero at the ends."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * i / (n - 1))
    a[0] = a[-1] = 0.0
    return a / a.sum()


def generate_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window (numpy implementation)."""
    a = np.kaiser(n, beta)
    a[0] = a[-1] = 0.0
    return a / a.sum()


# ----------------------------------------------------------------------
# Global exploration – keep the best ``max_keep`` candidates
# ----------------------------------------------------------------------


def baseline_search(
    n: int, deadline: float, max_keep: int = 12
) -> list[np.ndarray]:
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- 1) power‑law -------------------------------------------------
    for p in np.linspace(0.05, 3.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ----- 2) exponential ------------------------------------------------
    for lam in np.linspace(1e-5, 0.08, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ----- 3) beta -------------------------------------------------------
    al_vals = np.linspace(0.1, 10.0, 30)
    be_vals = np.linspace(0.1, 10.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ----- 4) Gaussian ----------------------------------------------------
    sigma_vals = np.linspace(n / 32.0, n / 1.5, 40)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ----- 5) Classical windows -----------------------------------------
    for fn in (
        generate_hann,
        generate_hamming,
        generate_blackman,
        generate_bartlett,
        generate_cosine_window,
    ):
        if time.time() >= deadline:
            break
        try_candidate(fn(n))

    # ----- 6) Kaiser windows (multiple beta) ----------------------------
    for beta in np.linspace(2.0, 14.0, 12):
        if time.time() >= deadline:
            break
        try_candidate(generate_kaiser(n, beta))

    # ----- 7) Dirichlet draws -------------------------------------------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(
    a0: np.ndarray,
    deadline: float,
    init_lr: float = 0.25,
    init_tau_factor: float = 15.0,
    min_tau: float = 1e-10,
    tau_decay: float = 0.9997,
) -> np.ndarray:
    """
    Minimise  max_k (a*a)_k via a smooth log‑sum‑exp proxy.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)  # log‑softmax parameters

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature (smoothness)
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # --- soft‑max mapping -----------------------------------------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # --- convolution & smooth max weights -------------------------
        conv = full_convolution(a)
        max_c = conv.max()

        if tau < 1e-12:
            # hard max (single active index)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # --- gradient w.r.t. a (g = 2·conv(w, a_rev)) ----------------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)  # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start : start + n]

        # --- chain rule to x‑space (soft‑max) -------------------------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # --- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- temperature annealing ------------------------------------
        tau = max(min_tau, tau * tau_decay)

        # --- occasional exact evaluation -------------------------------
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(
                    f"[smooth_opt] iter {it:6d}  score {val:.6f}  τ {tau:.2e}"
                )
    return best_a


# ----------------------------------------------------------------------
# Simple deterministic hill‑climb (pairwise mass transfer)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy local refinement: repeatedly move a small fraction of the
    mass from one coordinate to another if the exact score improves.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    # step scale starts at 1 % of the donor mass
    step_scale = 0.01
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # project back onto the simplex (tiny numerical drift)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        val = compute_score(cand)
        if val < cur_val:
            a = cand
            cur_val = val
            if val < best_val:
                best_val, best_a = val, cand.copy()
            # increase step a little when we are making progress
            step_scale = min(step_scale * 1.02, 0.2)
        else:
            # otherwise shrink the step
            step_scale = max(step_scale * 0.97, 5e-5)

        if it % 5000 == 0:
            # occasional random jitter to escape stagnation
            jitter = np.random.randn(n) * (step_scale * 0.02)
            a = a * np.exp(jitter)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(
    seed: int = 12345, budget_s: int = 1000, **kwargs
) -> list[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # problem size – a balanced choice (larger values give more freedom)
    # --------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0. optional fallback from a previously saved solution
    # --------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Global baseline exploration (≈ 150 s)
    # --------------------------------------------------------------
    baseline_deadline = deadline - 350.0  # reserve for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # --------------------------------------------------------------
    # 2. Smoothed Adam optimisation with several restarts (≈ 600 s)
    # --------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # primary run from the current best
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # further restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            if remaining < 30.0:
                break
            sub_deadline = time.time() + min(30.0, remaining)
            # draw a random start – we vary concentration to explore
            conc = np.random.choice([0.03, 0.07, 0.2, 0.5, 1.0, 2.5])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # --------------------------------------------------------------
    # 3. Deterministic hill‑climb polish (≈ remaining time)
    # --------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = hill_climb(a_start, hill_deadline)
        val = compute_score(final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # --------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 5. Report the official evaluation
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
