def directed_descent(a, deadline):
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    eps = 1e-12
    step_scale = 0.1 # fraction of source mass to move
    decay = 0.9995
    it = 0
    while time.time() < deadline:
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - eps)[0]
        # compute gradient
        g = np.zeros(n)
        for k in active:
            # contributions from pairs i+j = k
            # for i from max(0, k-(n-1)) to min(n-1, k)
            i_min = max(0, k-(n-1))
            i_max = min(n-1, k)
            # contributions: each pair i, j=k-i.
            # gradient wrt a_i: 2*a_j (if i!=j) else 2*a_i (since derivative of a_i^2)
            # So loop:
            for i in range(i_min, i_max+1):
                j = k - i
                if j < 0 or j >= n:
                    continue
                if i == j:
                    g[i] += 2.0 * a[i]
                else:
                    g[i] += 2.0 * a[j]
        # choose source and target
        # source candidates: large gradient * a (impact)
        impact = g * a
        i_source = np.argmax(impact)
        i_target = np.argmin(g)  # or argmin a? but we want low gradient to add mass
        # adjust delta
        delta = step_scale * a[i_source] * np.random.rand()
        if delta <= 0:
            continue
        # apply move
        a_new = a.copy()
        a_new[i_source] -= delta
        a_new[i_target] += delta
        # ensure nonnegative
        if a_new[i_source] < 0:
            # adjust delta to avoid negativity
            delta = a[i_source]
            a_new[i_source] = 0.0
            a_new[i_target] = a[i_target] + delta
        # evaluate score
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
            # maybe increase step_scale a bit
            step_scale = min(step_scale * 1.05, 0.5)
        # acceptance rule: if new_score <= current_score accept; else accept with prob.
        cur_score = compute_score(a) # we could keep it
        # use Metropolis style acceptance
        cur_score = compute_score(a)
        if new_score <= cur_score:
            a = a_new
        else:
            prob = np.exp(-(new_score - cur_score) / max(1e-8, step_scale))
            if np.random.rand() < prob:
                a = a_new
        # decay parameters
        step_scale = max(step_scale * decay, 1e-6)
        it += 1
        if it % 5000 == 0:

                a[idx2] += eps

def mass_transfer_descent(a, deadline):
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    M = full_convolution(a).max()
    cur = a.copy()
    cur_score = 2 * n * M
    step = 0.2
    while time.time() < deadline:
        conv = full_convolution(cur)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        # compute gradient using active set
        g = np.zeros(n)
        for k in active:
            i_min = max(0, k-(n-1))
            i_max = min(n-1, k)
            idx = np.arange(i_min, i_max+1)
            j = k - idx
            # contributions
            g[idx] += 2.0 * cur[j]
        # select source and target
        src = np.argmax(g * cur)  # high impact
        tgt = np.argmin(g)        # low impact
        # ensure src != tgt and cur[src] >0
        if src == tgt or cur[src] == 0:
            # fallback random move
            src, tgt = np.random.choice(n, 2, replace=False)
        delta = step * cur[src] * np.random.rand()
        # move delta from src to tgt
        cur_new = cur.copy()
        cur_new[src] -= delta
        cur_new[tgt] += delta
        # enforce nonnegativity
        if cur_new[src] < 0:
            cur_new[tgt] += cur_new[src]  # add back negative part
            cur_new[src] = 0.0
        # compute new score
        conv_new = full_convolution(cur_new)
        new_score = 2 * n * conv_new.max()
        # If improvement, accept
        if new_score < cur_score:
            cur = cur_new
            cur_score = new_score
            if new_score < best_score:
                best = cur_new.copy()
                best_score = new_score
                # increase step a bit
                step = min(step * 1.02, 0.6)
        else:
            # maybe accept with small prob decreasing with step
            prob = np.exp(-(new_score - cur_score) / max(step, 1e-8))
            if np.random.rand() < prob:
                cur = cur_new
                cur_score = new_score
            # reduce step
            step = max(step * 0.995, 1e-6)
        # occasional random jitter
        if np.random.rand() < 0.001:
            idx1, idx2 = np.random.choice(n, 2, replace=False)
            eps = step * cur[idx1]
            cur[idx1] = max(cur[idx1] - eps, 0.0)
            cur[idx2] += eps
            cur = cur / cur.sum()
            cur_score = 2 * n * full_convolution(cur).max()
    return best

"""
Search for a non‑negative step‑function (vector) that minimises

    2·n·max(conv(a,a)) / (sum(a))²

where ``conv`` denotes the full linear convolution.
The algorithm works in three phases:

1. **Exploratory parametric sweep** – a few fast families (exponential,
   quadratic, sinusoidal) and random Dirichlet draws give a decent starting
   point.
2. **Mass‑transfer active‑set descent** – repeatedly moves a small amount of
   probability mass from indices that contribute most to the current
   maximal convolution value towards indices that contribute least.
   The step size is adapted and a Metropolis acceptance rule (simulated
   annealing) helps to escape shallow plateaus.
3. **Sub‑gradient polishing** – a final projected sub‑gradient walk (as in
   the reference implementation) cleans up the remaining tiny errors.

All vectors are kept on the probability simplex (sum = 1) which makes the
objective scale‑invariant.  The code is written for a single CPU and respects
the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast for n up to several ten‑thousands)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())        # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The factor Σa is kept equal to one, therefore the objective is scale‑invariant.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families used for the quick sweep
# ----------------------------------------------------------------------
def exp_power(n: int, c: float, p: float) -> np.ndarray:
    """a_i ∝ exp(-c·(i+1)^p)  (c>0, p>0)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = np.exp(-c * (i ** p))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def quad_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1 - (i/(n‑1))^p,   p>0."""
    if n == 1:
        return None
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 - np.power(x, p)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def sinusoid(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p  (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(np.pi * x) ** p
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a random point from Dirichlet(conc,…,conc)."""
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    if s == 0.0:
        return None
    return y / s


# ----------------------------------------------------------------------
# Mass‑transfer active‑set descent
# ----------------------------------------------------------------------
def mass_transfer_descent(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Stochastic descent that moves a small amount of probability mass from indices
    that largely contribute to the current maximal convolution value towards
    indices that contribute the least.  An annealing‑style acceptance rule is
    used to avoid getting stuck in local plateaus.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    step = 0.15               # typical fraction of source mass to move
    step_decay = 0.9999
    temp = max(1e-4 * cur_score, 1e-8)
    temp_decay = 0.9997
    eps_active = 1e-12       # tolerance for identifying the active set
    it = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - eps_active)[0]

        # ---- gradient of the active max‑convolution (sub‑gradient) ----
        g = np.zeros(n, dtype=np.float64)
        for k in active:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            j = k - idx
            # contribution: 2 * a_j (including the diagonal case)
            g[idx] += 2.0 * a[j]

        # ---- choose source (large impact) and target (small impact) ----
        impact = g * a                     # larger impact → better source
        src = int(np.argmax(impact))
        tgt = int(np.argmin(g))

        if a[src] <= 0 or src == tgt:
            # fallback: random pair
            src, tgt = np.random.choice(n, 2, replace=False)

        delta = step * a[src] * np.random.rand()
        if delta <= 0:
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta
        # Numerical safety – keep the vector on the simplex exactly
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate /= a_candidate.sum()

        cand_score = compute_score(a_candidate)

        # ---- Metropolis acceptance ----
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            accept = (np.random.rand() < prob)

        if accept:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                # when we improve, allow a slightly larger step
                step = min(step * 1.02, 0.6)

        # ---- decay parameters ----
        temp = max(temp * temp_decay, 1e-12)
        step = max(step * step_decay, 1e-7)

        # occasional jitter to shake off stagnation
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step * 0.1)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            cur_score = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (the reference method)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float, step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]
    eps_active = 1e-12

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - eps_active)[0]

        # sub‑gradient of each active entry: 2 * a shifted
        g = np.zeros_like(a)
        for k in active:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            j = k - idx
            g[idx] += 2.0 * a[j]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

        # occasional random jitter to escape tiny plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a
