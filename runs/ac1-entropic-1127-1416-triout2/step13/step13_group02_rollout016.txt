def smooth_adam_polish(a0, deadline, p_start=2.0, p_end=30.0, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8, eval_every=200):
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    best_a = None
    best_val = float('inf')
    t = 0
    while time.time() < deadline:
        t += 1
        # softmax
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()
        # schedule p
        # linearly increase from p_start to p_end over approx 5000 iterations
        # after that stay at p_end
        p = p_start + (p_end - p_start) * min(1.0, t / 5000.0)
        conv, g_a = compute_grad_lp(a, p)
        # mirror gradient
        g_u = a * (g_a - np.dot(g_a, a))
        # Adam updates
        m = beta1 * m + (1 - beta1) * g_u
        v = beta2 * v + (1 - beta2) * (g_u * g_u)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)
        # occasional jitter
        if t % 2000 == 0:
            u += np.random.randn(n) * (lr * 0.001)
        # evaluate periodically
        if t % eval_every == 0:
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
    # final check
    if best_a is None:
        best_a = a
    return best_a, best_val

    for conc in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:

            top.sort(key=lambda x: x[0])

    print(f"[baseline] top {K} scores: {[sc for sc,_ in top]}")

            best_seq = vec.tolist()

    while time.time() < polish_deadline:

        if np.random.rand() < 0.3 and top:
            _, start_vec = top[np.random.randint(len(top))]

        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq

def compute_grad_softmax(a, alpha):
    n = a.shape[0]
    L = 1 << ((3*n - 2).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n-1]
    maxc = conv.max()
    # compute softmax weights
    # subtract maxc for numerical stability
    exp_w = np.exp(alpha * (conv - maxc))
    w = exp_w / exp_w.sum()
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(W * np.conj(A), n=L)[:3*n - 2]
    idx = n - 1 - np.arange(n)
    g = 2.0 * corr[idx]
    return conv, g

"""
Search for a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The core optimiser performs Adam on a smooth
approximation of the max‑convolution ‑‑ a soft‑max (log‑sum‑exp)
over the linear convolution a*a.  The smoothing parameter α is raised
gradually so that the optimiser first sees a gentle landscape and
finally concentrates on the true ℓ∞‑norm.  Candidates are drawn from a
few parametric families and many Dirichlet samples; the best few are
polished repeatedly until the time budget runs out.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (linear, not circular)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for a quick global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Smoothed “max‑convolution’’ gradient (log‑sum‑exp / softmax)
# ----------------------------------------------------------------------


def _grad_softmax_max(a: np.ndarray, alpha: float):
    """
    Returns (conv, g) where
      conv = a*a  (linear convolution)
      g    = ∂ (soft‑max_{α}(conv)) / ∂ a
    The soft‑max is   (1/α)·log( Σ_k exp(α·conv_k) ).
    """
    n = a.shape[0]
    # zero‑pad length for the correlation needed later
    L = 1 << ((3 * n - 2).bit_length())

    # Fourier transform of a (used for both convolution and correlation)
    A = np.fft.rfft(a, n=L)

    # linear convolution a*a (full length 2n‑1)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

    # soft‑max weights (stable implementation)
    maxc = conv.max()
    # exp(α·(conv‑maxc)) ≤ 1, no overflow
    w = np.exp(alpha * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        # degenerate case – fall back to uniform weights
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # correlation needed for the gradient:
    #   g_i = 2· Σ_k w_k·a_{k‑i}
    # which is a cross‑correlation of w with a.
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(W * np.conj(A), n=L)[: 3 * n - 2]

    # pick the slice corresponding to indices 0 … n‑1:
    idx = n - 1 - np.arange(n)
    g = 2.0 * corr[idx]
    return conv, g


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def smooth_adam_polish(a0: np.ndarray, deadline: float,
                       alpha_start: float = 5.0,
                       alpha_end: float = 80.0,
                       alpha_iters: int = 20000,
                       lr: float = 0.06,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8,
                       eval_every: int = 250) -> np.ndarray:
    """
    Adam optimisation of the smoothed max‑convolution.
    The vector is stored in log‑space (u) and projected to the simplex
    by the soft‑max `a = exp(u) / Σ exp(u)`.  The smoothing parameter α
    is increased linearly from `alpha_start` to `alpha_end` over the
    first `alpha_iters` steps, after which it stays at `alpha_end`.
    """
    n = a0.shape[0]
    # initialise log‑probabilities (avoid log(0))
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # ----- simplex projection (soft‑max) -----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ----- smoothing schedule -----
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / alpha_iters)

        # ----- gradient of the smooth max -----
        conv, g_a = _grad_softmax_max(a, alpha)

        # mirror‑descent gradient (Jacobian of soft‑max)
        g_u = a * (g_a - np.dot(g_a, a))

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)   # mild decay
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaux
        if t % 3000 == 0:
            u += np.random.randn(n) * (lr * 0.001)

        # ----- keep best solution seen -----
        if t % eval_every == 0:
            cur_val = float(2.0 * n * conv.max())
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                # tiny print for monitoring
                print(f"[polish] iter {t:,d} – α={alpha:.2f} – score={cur_val:.6f}")

    # final check – one last evaluation in case loop never hit the
    # `if t % eval_every` condition after the final iteration.
    if best_a is None:
        # evaluate one more time explicitly
        conv = full_convolution(np.exp(u - np.max(u)) / np.exp(u - np.max(u)).sum())
        best_val = float(2.0 * n * conv.max())
        best_a = np.exp(u - np.max(u)) / np.exp(u - np.max(u)).sum()

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    evaluate_sequence() within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # -----------------------------------------------------------------
    # Determine problem size (use length of a known fallback if present)
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        n = len(fallback)
    else:
        n = 2048                      # a good trade‑off for speed vs. quality

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # -----------------------------------------------------------------
    if fallback is not None:
        try:
            val = evaluate_sequence(fallback)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, list(fallback)
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Light global sweep (few parametric families + Dirichlet draws)
    # -----------------------------------------------------------------
    candidates = []

    # a) simple deterministic families
    candidates.append(("uniform", np.full(n, 1.0 / n)))
    for p in (0.5, 1.0, 2.0, 4.0):
        candidates.append((f"powerlaw_{p}", generate_powerlaw(n, p)))
    for lam in (0.05, 0.2, 0.5, 1.0, 2.0):
        candidates.append((f"exp_{lam}", generate_exponential(n, lam)))
    for a in (0.5, 1.0, 2.0):
        candidates.append((f"beta_{a}_{a}", generate_beta(n, a, a)))
    # sine‑window (a classic smooth shape)
    i = np.arange(n, dtype=float)
    sine = np.sin(np.pi * (i + 0.5) / n) ** 2
    sine = sine / sine.sum()
    candidates.append(("sine", sine))

    # b) random Dirichlet draws (moderate amount – fast)
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per = 500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per):
            a = np.random.dirichlet(alpha_vec)
            candidates.append(("dirichlet", a))

    # keep the ten best candidates (by the exact objective)
    top = []   # list of (score, vector)
    def _maybe_keep(vec):
        sc = compute_score(vec)
        if len(top) < 10:
            top.append((sc, vec.copy()))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, vec.copy())
            top.sort(key=lambda x: x[0])

    for name, vec in candidates:
        # normalise to sum‑1 and guard against pathologies
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            continue
        vec = vec / s
        _maybe_keep(vec)

    print("[baseline] top scores :", [f"{sc:.6f}" for sc, _ in top])

    # initialise best from the baseline
    for sc, vec in top:
        if sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[baseline] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam polishing using the smooth‑max surrogate
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0            # reserve a short final pass
    while time.time() < polish_deadline:
        # pick a start vector: 30 % chance from the elite pool,
        # otherwise a fresh Dirichlet draw.
        if np.random.rand() < 0.3 and top:
            start_vec = top[np.random.randint(len(top))][1]
        else:
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0])
            start_vec = np.random.dirichlet(np.full(n, conc))

        run_deadline = min(time.time() + 30.0, polish_deadline)
        refined = smooth_adam_polish(start_vec, run_deadline)
        sc = compute_score(refined)
        if sc < best_val:
            best_val, best_seq = sc, refined.tolist()
            print(f"[smooth‑adam] improved → {sc:.6f}")

    # -----------------------------------------------------------------
    # 3. Very short simulated‑annealing (escape possible plateaux)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)

        def _anneal(a, t_end):
            n_local = a.shape[0]
            cur_val = compute_score(a)
            best_local, best_local_val = a.copy(), cur_val
            T = 5e-4
            step = 0.02
            while time.time() < t_end:
                i = np.random.randint(n_local)
                if a[i] <= 0.0:
                    continue
                j = np.random.randint(n_local)
                if i == j:
                    continue
                delta = step * a[i]
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
                val = compute_score(cand)
                d = val - cur_val
                if d < 0 or np.random.rand() < np.exp(-d / max(T, 1e-12)):
                    a, cur_val = cand, val
                    T = max(1e-6, T * 0.9995)
                    step = min(0.5, step * 1.02)
                    if cur_val < best_local_val:
                        best_local_val, best_local = cur_val, a.copy()
                else:
                    step = max(1e-7, step * 0.98)
                # occasional jitter to keep movement alive
                if np.random.rand() < 0.001:
                    jitter = np.random.randn(n_local) * (step * 0.01)
                    a = a * np.exp(jitter)
                    a /= a.sum()
                    cur_val = compute_score(a)
            return best_local

        annealed = _anneal(a_start, anneal_deadline)
        sc = compute_score(annealed)
        if sc < best_val:
            best_val, best_seq = sc, annealed.tolist()
            print(f"[anneal] final improvement → {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Fallback to uniform if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
