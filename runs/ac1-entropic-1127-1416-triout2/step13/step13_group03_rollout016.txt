"""
Optimiser for step‑function heights.

Strategy
--------
1. **Coarse global exploration** on a moderate dimension (n_small = 1024).
   A variety of parametric families (power‑law, exponential, beta,
   Gaussian, sine‑power, tent, cosine) and many Dirichlet draws are tried.
   The best few candidates (according to the exact objective) are retained.

2. **Smooth‑max Adam optimisation** on the small dimension.
   The surrogate  τ·log ∑ exp(conv/τ)  is minimised with an Adam update
   and a geometric temperature schedule.  Each start vector is refined
   and the best refined vector is kept.

3. **Upscaling** – the best small‑size vector is interpolated to the
   target dimension (n_large = 4096) and projected back onto the simplex.

4. **Local refinement on the full size** – a longer Adam run, a
   projected sub‑gradient polish and a simulated‑annealing hill‑climb
   are applied, all respecting the remaining time budget.

The algorithm constantly checks the wall‑clock and stops well before the
deadline, returning the best sequence found (a plain list of floats).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v>=0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (no zero‑padding artefacts)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = rfft(a, n=n_fft)
    C = A * A
    conv = irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector with sum(a)=1:
       score = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a vector with sum 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 1.0 / n)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 1.0 / n)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 1.0 / n)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 1.0 / n)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 1.0 / n)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 1.0 / n)


def generate_tent(n: int) -> np.ndarray:
    # symmetric triangular shape (peak in the middle)
    half = n // 2
    left = np.arange(1, half + 1, dtype=float)
    right = left[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=float)
    a = np.concatenate([left, right])
    total = a.sum()
    return a / total if total > 0 else np.full(n, 1.0 / n)


def generate_cosine(n: int) -> np.ndarray:
    # 0 at the ends, cosine‑bump in the centre (non‑negative)
    x = np.linspace(0, np.pi, n)
    a = 1.0 - np.cos(x)  # lies in [0,2]
    total = a.sum()
    return a / total if total > 0 else np.full(n, 1.0 / n)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1‑λ)·vec2, projected onto the simplex."""
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)

# Global exploration on a modest size

def baseline_search_small(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 6000,
):
    """Generate many parametric candidates and Dirichlet draws on size n.
    Returns a list of up to max_keep best vectors (as numpy arrays)."""
    best = []  # list of (score, vector)

    def maybe_add(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1. power‑law sweep
    for p in np.linspace(0.1, 5.0, 41):
        if time.time() >= deadline:
            break
        maybe_add(generate_powerlaw(n, p))

    # 2. exponential sweep
    for lam in np.linspace(0.0005, 0.4, 40):
        if time.time() >= deadline:
            break
        maybe_add(generate_exponential(n, lam))

    # 3. beta grid
    al_grid = np.linspace(0.05, 2.0, 25)
    be_grid = np.linspace(0.05, 2.0, 25)
    for alpha in al_grid:
        for beta in be_grid:
            if time.time() >= deadline:
                break
            maybe_add(generate_beta(n, alpha, beta))

    # 4. random beta draws
    for _ in range(300):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)
        beta = 10 ** np.random.uniform(-2, 0.3)
        maybe_add(generate_beta(n, alpha, beta))

    # 5. Gaussian family
    for sigma in np.linspace(0.05 * n, 0.4 * n, 30):
        if time.time() >= deadline:
            break
        maybe_add(generate_gaussian(n, sigma))

    # 6. sine‑power family
    for p in np.linspace(0.2, 8.0, 35):
        if time.time() >= deadline:
            break
        maybe_add(generate_sine_power(n, p))

    # 7. tent & cosine
    maybe_add(generate_tent(n))
    maybe_add(generate_cosine(n))

    # 8. mixtures of random families
    for _ in range(120):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(n,
                          10 ** np.random.uniform(-2, 0.3),
                          10 ** np.random.uniform(-2, 0.3))
        lam = np.random.rand()
        maybe_add(mixture(a, b, lam))

    # 9. Dirichlet draws (wide range of concentration)
    conc_options = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(conc_options))
    for conc in conc_options:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            maybe_add(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-6,
    tau_decay: float = 0.93,
    lr_start: float = 0.10,
    lr_decay: float = 0.97,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """Adam optimisation of the smooth‑max surrogate:
       fτ(a) = τ·log Σ_k exp(conv_k(a)/τ)
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # temperature schedule
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # compute smooth‑max weights
        conv = full_convolution(a)
        shift = conv.max()  # for numerical stability
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()  # size m

        # gradient of the smooth‑max (2 * a ⋆ w_rev)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = rfft(a, n=size)
        B = rfft(rev_w, n=size)
        corr = irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # learning‑rate annealing
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-5)

        # occasional jitter to avoid plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.30) -> np.ndarray:
    """Exact sub‑gradient descent on  max(conv) with projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

# Simulated annealing (stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 3e-3,
                        final_temp: float = 5e-7,
                        cooling_factor: float = 0.9993,
                        step_scale_init: float = 0.08,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    """Metropolis hill‑climb that accepts worsening moves with probability exp(-Δ/ T)."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.015)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def upscale_vector(vec_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size,
    followed by projection back onto the simplex."""
    n_small = vec_small.shape[0]
    # positions in small vector (0 … n_small‑1)
    xs_small = np.linspace(0, 1, n_small)
    xs_large = np.linspace(0, 1, n_large)
    interpolated = np.interp(xs_large, xs_small, vec_small)
    return project_to_simplex(interpolated)

    except Exception:
        pass

    # -----------------------------------------------------------------

            n_small, small_deadline, max_keep=12, random_draws=8000
        )

    # -----------------------------------------------------------------
    # 2. Adam optimisation on the small vectors
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 350.0:
        # select a few promising starts (the already best plus a few random)
        start_vectors = [np.array(best_seq, dtype=float)]
        start_vectors.append(np.full(n_small, 1.0 / n_small))
        for _ in range(3):
            start_vectors.append(project_to_simplex(np.random.rand(n_small)))

        refined_best = start_vectors[0].copy()
        refined_val = best_val

        for idx, start_vec in enumerate(start_vectors):
            if time.time() >= deadline - 350.0:
                break
            local_deadline = min(deadline - 250.0,
                                 time.time() + 120.0)       # about 2 min per start
            a_ref = smooth_max_adam(start_vec, local_deadline)
            val = compute_score(a_ref)
            print(f"[adam‑small {idx}] score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_ref.copy()

        best_val, best_seq = refined_val, refined_best.tolist()
        print(f"[adam‑small] best after refinement = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Upscale to the target size (n_large) and final refinement
    # -----------------------------------------------------------------
    n_large = 4096
    if best_seq is not None and time.time() < deadline - 120.0:
        a_large = upscale_vector(np.array(best_seq, dtype=float), n_large)

        # Adam on the large vector
        large_adam_deadline = time.time() + 100.0
        a_large = smooth_max_adam(a_large, large_adam_deadline)

        # Sub‑gradient polish (more iterations)
        polish_deadline = time.time() + 80.0
        a_large = subgradient_polish(a_large, polish_deadline)

        # Simulated annealing for fine‑grained tweaks
        anneal_deadline = time.time() + 80.0
        a_large = simulated_annealing(a_large, anneal_deadline)

        val = compute_score(a_large)
        print(f"[final‑large] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_large.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n_large] * n_large
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final reporting using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
