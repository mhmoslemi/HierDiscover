``2·n·max(conv(a,a)) / (sum(a)^2)``.
The search combines:

* A fast scan of a rich set of analytic families (U‑shaped,
  centre‑heavy, Gaussian, triangular, exponential, mixed).

  – Dirichlet‑sparse draws,

  uses a sub‑gradient built from *all* convolution indices that are
  within 0.1 % of the current maximum (instead of a single index).
* A longer simulated‑annealing hill‑climb with adaptive temperature.

All steps respect a 1000 s wall‑clock limit (a few seconds are kept

The routine returns the best sequence found as a plain ``list`` of

"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)^2).
    a is assumed to be a 1‑D NumPy array with non‑negative entries.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (sum=1, v≥0).  The implementation follows the classic O(n)
    algorithm of Duchi et al.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (n‑½‑i)^{‑β} (mirror of left‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """
    Symmetric triangular bump.
    width_frac ∈ (0,1] controls half‑width relative to n.
    """
    half_width = int(max(1, width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    """Exponential decay from one side."""
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return w / w.sum()


def mixed_uniform_gaussian_family(n: int, lam: float, sigma: float) -> np.ndarray:
    """Convex mixture of uniform and centred Gaussian."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    gaussian = gaussian_center_family(n, sigma)
    w = lam * uniform + (1.0 - lam) * gaussian
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w


def random_sparse_vector(n: int, concentration: float = 0.3) -> np.ndarray:
    """
    Dirichlet draw with small concentration to generate a sparse vector.
    """
    x = np.random.gamma(shape=concentration, scale=1.0, size=n)
    return normalize(x)


def mixture_two_families(
    n: int,
    fam1_func,
    fam2_func,
    pars1: Tuple,
    pars2: Tuple,
    lam: float,
) -> np.ndarray:
    """
    Convex mixture of two families (lam * fam1 + (1-lam) * fam2),
    returned as a normalised probability vector.
    """
    v1 = fam1_func(n, *pars1)
    v2 = fam2_func(n, *pars2)
    w = lam * v1 + (1.0 - lam) * v2
    return normalize(w)

def exp_gradient_descent(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.6,
    tol: float = 1e-4,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using exponentiated gradients.
    The sub‑gradient is formed from *all* convolution indices whose
    value is within 0.1 % of the current maximum (instead of a single
    index).  This provides a smoother descent direction.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    # pre‑allocate an index array for speed
    idxs_full = np.arange(n, dtype=np.int64)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = conv.max()
        # indices whose value is within 0.1 % of the maximum
        close = np.where(conv >= mx * (1.0 - 1e-3))[0]

        # build sub‑gradient as sum of gradients from the selected indices
        g = np.zeros_like(a)
        for k in close:
            j = k - idxs_full
            mask = (j >= 0) & (j < n)
            # contribution 2 * a_j for each valid pair (i, j)
            g[mask] += 2.0 * a[j[mask]]

        # exponentiated‑gradient step with diminishing stepsize
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # project back onto the simplex (KL‑projection = renormalisation)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional true‑objective evaluation
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc + tol < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] exp‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc + tol < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 12_000_000,
    temp_factor: float = 1.2e-3,
    step_scale_init: float = 0.25,
) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    The move set consists of a random swap of two entries or a
    transfer of a fraction of mass from one index to another.
    Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 30) -> Tuple[float, float]:
    """Golden‑section search for the minimum of func on [lo,hi]."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. optional fallback from a previous run
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. discretisation size
    # -----------------------------------------------------------
    n = 8192

    # -----------------------------------------------------------
    # 2. coarse scan of analytic families
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] coarse family scan (n={n})")
    best_family = None
    best_params = None

    # centre‑heavy power (alpha ≥ 0)
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_center_power_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family, best_params = "center_power", {"alpha": alpha}
            print(f"[{time.time():.1f}] new best (center‑α={alpha:.3f}) = {best_score:.6f}")

    # Gaussian centre (sigma)
    for sigma in np.linspace(0.2 * n, 0.5 * n, 30):
        if time.time() >= deadline - 30.0:
            break
        vec = gaussian_center_family(n, sigma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family, best_params = "gaussian", {"sigma": sigma}
            print(f"[{time.time():.1f}] new best (gauss‑σ={sigma:.0f}) = {best_score:.6f}")

    # Triangular centre (width_frac)
    for width in np.linspace(0.1, 0.9, 33):
        if time.time() >= deadline - 30.0:
            break
        vec = triangular_center_family(n, width)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family, best_params = "triangular", {"width_frac": width}
            print(f"[{time.time():.1f}] new best (tri‑w={width:.3f}) = {best_score:.6f}")

    # left‑heavy power
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family, best_params = "left_power", {"beta": beta}
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")

        vec_r = right_heavy_power_family(n, beta)
        sc_r = compute_score(vec_r)
        if sc_r < best_score:
            best_score, best_seq = sc_r, vec_r.tolist()
            best_family, best_params = "right_power", {"beta": beta}
            print(f"[{time.time():.1f}] new best (right‑β={beta:.3f}) = {best_score:.6f}")

    # product (U‑shaped) family
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family, best_params = "product", {"alpha": alpha}
            print(f"[{time.time():.1f}] new best (U‑α={alpha:.3f}) = {best_score:.6f}")

    # beta family (two‑parameter)
    for _ in range(100):
        if time.time() >= deadline - 30.0:
            break
        p = np.random.uniform(0.2, 3.0)
        q = np.random.uniform(0.2, 3.0)
        vec = beta_family(n, p, q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family, best_params = "beta", {"p": p, "q": q}
            print(f"[{time.time():.1f}] new best (beta p={p:.3f},q={q:.3f}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Random dense search (≈15000 trials, extended families)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] random dense search")
    trials = 0
    max_trials = 15000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        # pick a random construction strategy
        choice = np.random.randint(12)
        if choice == 0:          # centre‑heavy power
            alpha = np.random.uniform(0.0, 12.0)
            vec = symmetric_center_power_family(n, alpha)
        elif choice == 1:        # left‑heavy power
            beta = np.random.uniform(0.0, 12.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 2:        # right‑heavy power
            beta = np.random.uniform(0.0, 12.0)
            vec = right_heavy_power_family(n, beta)
        elif choice == 3:        # product (U‑shaped)
            alpha = np.random.uniform(0.0, 12.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 4:        # Gaussian centre
            sigma = np.random.uniform(0.2 * n, 0.5 * n)
            vec = gaussian_center_family(n, sigma)
        elif choice == 5:        # triangular centre
            width = np.random.uniform(0.1, 0.9)
            vec = triangular_center_family(n, width)
        elif choice == 6:        # exponential decay left
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n, gamma, left=True)
        elif choice == 7:        # exponential decay right
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n, gamma, left=False)
        elif choice == 8:        # mixed uniform + Gaussian
            lam = np.random.uniform(0.0, 1.0)
            sigma = np.random.uniform(0.2 * n, 0.5 * n)
            vec = mixed_uniform_gaussian_family(n, lam, sigma)
        elif choice == 9:        # random sparse Dirichlet
            conc = np.random.uniform(0.05, 0.5)
            vec = random_sparse_vector(n, concentration=conc)
        else:                    # mixture of two families (left vs right heavy)
            beta1 = np.random.uniform(0.0, 12.0)
            beta2 = np.random.uniform(0.0, 12.0)
            lam = np.random.uniform(0.0, 1.0)
            vec = mixture_two_families(
                n,
                left_heavy_power_family,
                right_heavy_power_family,
                (beta1,),
                (beta2,),
                lam,
            )

        # occasional random shift (zero‑padding) that breaks symmetry
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Fine golden‑section refinement of the best one‑parameter family
    # -----------------------------------------------------------
    if best_family is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining best family {best_family}")

        if best_family == "center_power":
            cur = best_params["alpha"]
            lo, hi = max(0.0, cur - 0.8), min(12.0, cur + 0.8)

            def f(a):
                return compute_score(symmetric_center_power_family(n, a))

            new_a, _ = golden_section_optimize(f, lo, hi, max_iter=20)
            vec = symmetric_center_power_family(n, new_a)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_a
                print(f"[{time.time():.1f}] refined centre‑α={new_a:.4f} → {best_score:.6f}")

        elif best_family == "gaussian":
            cur = best_params["sigma"]
            lo, hi = max(0.2 * n, cur - 0.2 * n), min(0.5 * n, cur + 0.2 * n)

            def f(s):
                return compute_score(gaussian_center_family(n, s))

            new_s, _ = golden_section_optimize(f, lo, hi, max_iter=20)
            vec = gaussian_center_family(n, new_s)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["sigma"] = new_s
                print(f"[{time.time():.1f}] refined gauss‑σ={new_s:.1f} → {best_score:.6f}")

        elif best_family == "triangular":
            cur = best_params["width_frac"]
            lo, hi = max(0.05, cur - 0.2), min(0.95, cur + 0.2)

            def f(w):
                return compute_score(triangular_center_family(n, w))

            new_w, _ = golden_section_optimize(f, lo, hi, max_iter=20)
            vec = triangular_center_family(n, new_w)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["width_frac"] = new_w
                print(f"[{time.time():.1f}] refined tri‑w={new_w:.4f} → {best_score:.6f}")

        elif best_family == "product":
            cur = best_params["alpha"]
            lo, hi = max(0.0, cur - 0.6), min(12.0, cur + 0.6)

            def f(a):
                return compute_score(symmetric_product_family(n, a))

            new_a, _ = golden_section_optimize(f, lo, hi, max_iter=20)
            vec = symmetric_product_family(n, new_a)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_a
                print(f"[{time.time():.1f}] refined product‑α={new_a:.4f} → {best_score:.6f}")

        elif best_family == "beta":
            # simple local random refinement
            p0, q0 = best_params["p"], best_params["q"]
            for _ in range(300):
                if time.time() >= deadline - 5.0:
                    break
                p = np.clip(p0 + np.random.randn() * 0.15, 0.1, 5.0)
                q = np.clip(q0 + np.random.randn() * 0.15, 0.1, 5.0)
                vec = beta_family(n, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    best_params["p"], best_params["q"] = p, q
                    print(f"[{time.time():.1f}] refined beta p={p:.3f},q={q:.3f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Exponentiated‑gradient descent (mirror descent) refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] exponentiated‑gradient refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        # allocate a generous budget for the gradient steps
        grad_deadline = deadline - 150.0
        max_iters = int(min(400_000, (grad_deadline - time.time()) * 1000))
        a_opt, sc_opt = exp_gradient_descent(
            start_a=a_start,
            deadline=grad_deadline,
            max_iters=max_iters,
            eta0=0.7,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb (high temperature)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=15_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Fallback to a uniform small vector if everything failed
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
