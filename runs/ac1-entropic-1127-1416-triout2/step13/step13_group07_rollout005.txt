"""
propose_candidate
-----------------
Searches for a non‑negative step‑function (probability vector) that
minimises the competition objective

    2·n·max(conv(a,a)) / (∑a)²

The driver performs the following stages (all respect the global
wall‑clock budget of 1000 s):

1. **Coarse deterministic scan** of several analytic families
   (product‑α, left‑heavy‑β, exponential‑U, mixed‑uniform‑U,
   and the 2‑parameter Beta family) – the scan range is widened
   compared with the reference implementation.

2. **Random dense exploration** of the same families (a few thousand
   random draws, occasional random right‑shifts).

3. **Local refinement** of the best product‑α and Beta‑(p,q) parameters
   using simple hill‑climbing in one‑ and two‑dimensional parameter
   spaces.

4. **Adam optimisation** of the smooth‑max surrogate of the
   max‑convolution.  The temperature β is raised geometrically from
   5 to 3·10⁴; symmetry is enforced every few dozen steps and the
   learning‑rate schedule is tuned for large vectors.

5. **Projected sub‑gradient descent** on the exact max‑objective
   (the same update as the reference but with a slightly larger
   iteration budget).

6. **Greedy hill‑climb** that moves a small fraction of mass from a
   random donor to a random receiver whenever this reduces the score.

7. **Simulated annealing** fine‑tuning – small random perturbations
   are accepted with a Metropolis probability that slowly cools.

All stages keep the current best sequence and return it (as a plain
``list[float]``) when the time budget expires.  The implementation
uses only NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (∑a)².
    a is assumed non‑negative (the optimiser keeps it on the simplex).
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families (all return a normalised non‑negative vector)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid (p,q>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)


def dirichlet_random(n: int, conc: float = 0.3) -> np.ndarray:
    """Sample a random point on the simplex from a Dirichlet(α) distribution."""
    draws = np.random.gamma(conc, 1.0, size=n)
    return draws / draws.sum()

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 5.0,
                         beta_end: float = 30_000.0,
                         lr: float = 0.011,
                         sym_interval: int = 30) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    ``beta`` grows geometrically from ``beta_start`` to ``beta_end``.
    Every ``sym_interval`` steps the vector is symmetrised.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights (log‑sum‑exp trick)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 4000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_800_000,
                     step0: float = 0.047) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_greedy(start_a: np.ndarray,
                      deadline: float,
                      max_steps: int = 9_000_000,
                      init_scale: float = 0.18) -> Tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb that transfers a small fraction of mass
    from a random donor to a random receiver whenever this reduces
    the objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # donor – pick a relatively large entry (probability ~a_i)
        donor_idx = np.random.choice(n, p=cur_a)
        receiver_idx = np.random.randint(n)
        if donor_idx == receiver_idx:
            continue

        delta = scale * cur_a[donor_idx]
        cand = cur_a.copy()
        cand[donor_idx] -= delta
        cand[receiver_idx] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            scale = min(scale * 1.04, 0.5)
        else:
            scale = max(scale * 0.96, 1e-5)

        if step % 600_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score


def simulated_annealing(start_a: np.ndarray,
                        deadline: float,
                        init_temp: float = 0.12,
                        decay: float = 0.998,
                        inner_steps: int = 1500) -> Tuple[np.ndarray, float]:
    """
    Very lightweight Metropolis‑style annealing.
    Small random Gaussian perturbations are added, the vector is
    re‑normalised and accepted according to the temperature schedule.
    """
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    temp = init_temp

    n = cur_a.size
    while time.time() < deadline and temp > 1e-5:
        for _ in range(inner_steps):
            # propose a perturbation proportional to the current mass
            noise = np.random.randn(n) * (0.03 * cur_a + 1e-6)
            cand = cur_a + noise
            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)

            if cand_score < cur_score:
                cur_a, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_score, best_a = cand_score, cand.copy()
            else:
                # Metropolis acceptance
                prob = math.exp(-(cand_score - cur_score) / temp)
                if np.random.rand() < prob:
                    cur_a, cur_score = cand, cand_score

        temp *= decay
        # occasional logging
        print(f"[{time.time():.1f}] anneal T={temp:.5f}, best={best_score:.6f}")

        if time.time() >= deadline:
            break

    return best_a, best_score

def hill_climb_alpha(alpha0: float, n: int, deadline: float) -> float:
    """Hill‑climb on the product‑family parameter α."""
    best = alpha0
    best_score = compute_score(product_family(n, best))
    step = 0.9
    while time.time() < deadline and step > 5e-4:
        improved = False
        for sgn in (-1, 1):
            cand = max(0.0, best + sgn * step)
            sc = compute_score(product_family(n, cand))
            if sc < best_score:
                best, best_score = cand, sc
                improved = True
                print(f"[{time.time():.1f}] α refined to {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta(p0: float, q0: float, n: int, deadline: float) -> Tuple[float, float]:
    """Hill‑climb on the 2‑parameter Beta family."""
    best_p, best_q = p0, q0
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = 0.45
    while time.time() < deadline and step > 5e-4:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_p, best_q, best_score = cand_p, cand_q, sc
                    improved = True
                    print(f"[{time.time():.1f}] β refined to p={best_p:.6f}, q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` under the 1000 s wall‑clock budget.
    Returns the best sequence found as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # Choose a working length (larger n typically gives a lower bound)
    # -----------------------------------------------------------
    n = 32768            # 2^15 – a good compromise between quality and speed

    print(f"[{time.time():.1f}] Starting search, n={n}")

    # -----------------------------------------------------------
    # 1. Coarse deterministic scan over analytic families
    # -----------------------------------------------------------
    # product family
    best_alpha = None
    for alpha in np.linspace(0.0, 30.0, 61):
        if time.time() >= deadline - 45.0:
            break
        vec = product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best product α={alpha:.4f} → {best_score:.6f}")

        # random right‑shifts (to break symmetry a little)
        d = np.random.randint(0, n // 20 + 1)
        shifted = np.concatenate([vec[d:], np.zeros(d)])
        shifted = normalize(shifted)
        sc2 = compute_score(shifted)
        if sc2 < best_score:
            best_score, best_seq = sc2, shifted
            print(f"[{time.time():.1f}] shifted product α={alpha:.4f}, d={d} → {best_score:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 30.0, 61):
        if time.time() >= deadline - 45.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            print(f"[{time.time():.1f}] new best left‑heavy β={beta:.4f} → {best_score:.6f}")

    # exponential‑U family
    for gamma in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 45.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            print(f"[{time.time():.1f}] new best exp‑U γ={gamma:.4f} → {best_score:.6f}")

    # mixture uniform + exponential‑U
    for lam in np.linspace(0.0, 1.0, 7):
        for gamma in np.linspace(0.0, 8.0, 17):
            if time.time() >= deadline - 45.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec
                print(f"[{time.time():.1f}] new best mix λ={lam:.2f},γ={gamma:.2f} → {best_score:.6f}")

    # Beta family (coarse grid)
    for p in np.linspace(0.05, 5.0, 12):
        for q in np.linspace(0.05, 5.0, 12):
            if time.time() >= deadline - 45.0:
                break
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec
                print(f"[{time.time():.1f}] new best Beta p={p:.3f},q={q:.3f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 2. Random dense search (≈4000 evaluations)
    # -----------------------------------------------------------
    trials = 0
    max_trials = 4000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(5)
        if choice == 0:  # product
            alpha = np.random.uniform(0.0, 30.0)
            vec = product_family(n, alpha)
        elif choice == 1:  # beta
            p = np.random.uniform(0.02, 6.0)
            q = np.random.uniform(0.02, 6.0)
            vec = beta_family(n, p, q)
        elif choice == 2:  # left‑heavy
            beta = np.random.uniform(0.0, 30.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 3:  # exponential‑U
            gamma = np.random.uniform(0.0, 8.0)
            vec = exp_u_family(n, gamma)
        else:  # Dirichlet random
            conc = np.random.uniform(0.05, 0.6)
            vec = dirichlet_random(n, conc)

        # occasional random shift
        if np.random.rand() < 0.3:
            d = np.random.randint(0, n // 20 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Local refinement of analytic‑family parameters
    # -----------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product‑α (start {best_alpha:.4f})")
        refined_alpha = hill_climb_alpha(best_alpha, n, deadline - 10.0)
        vec = product_family(n, refined_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            best_alpha = refined_alpha
            print(f"[{time.time():.1f}] refined product‑α={refined_alpha:.6f} → {best_score:.6f}")

    # Beta refinement (start from a moderate U‑shape)
    if time.time() < deadline - 20.0:
        start_p, start_q = 0.6, 0.6
        refined_p, refined_q = hill_climb_beta(start_p, start_q, n, deadline - 10.0)
        vec = beta_family(n, refined_p, refined_q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec
            print(f"[{time.time():.1f}] refined Beta p={refined_p:.6f}, q={refined_q:.6f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Adam smooth‑max optimisation (no enforced symmetry, long run)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Adam refinement (smooth‑max)")
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 150.0,
            max_iters=500_000,
            beta_start=5.0,
            beta_end=30_000.0,
            lr=0.011,
            sym_interval=30,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Projected sub‑gradient descent on the true max‑objective
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 60.0,
            max_iters=1_600_000,
            step0=0.047,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final greedy hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] final greedy hill‑climb")
        a_greedy, sc_greedy = hill_climb_greedy(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 5.0,
            max_steps=8_000_000,
            init_scale=0.18,
        )
        if sc_greedy < best_score:
            best_score, best_seq = sc_greedy, a_greedy
            print(f"[{time.time():.1f}] greedy improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Light simulated annealing for very fine tuning
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] simulated annealing fine‑tune")
        a_anneal, sc_anneal = simulated_annealing(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 1.0,
            init_temp=0.12,
            decay=0.997,
            inner_steps=1200,
        )
        if sc_anneal < best_score:
            best_score, best_seq = sc_anneal, a_anneal
            print(f"[{time.time():.1f}] annealing improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Safety fallback (should never be needed)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = compute_score(best_seq)

    # Final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq.tolist()
