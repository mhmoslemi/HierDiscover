def proj_simplex(v):
    # v: np.ndarray shape (n,)
    # based on algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def global_explore(n, deadline, max_keep=15):
    best = []
    def consider(vec):
        if not np.isfinite(vec).all(): return
        # ensure positivity after clipping
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0: return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])
    # parametric families
    # powerlaw
    for p in np.linspace(0.01, 10.0, 400):
        if time.time() >= deadline: break
        consider(generate_powerlaw(n, p))
    # exponential
    for lam in np.linspace(1e-6, 0.5, 400):
        if time.time() >= deadline: break
        consider(generate_exponential(n, lam))
    # beta
    al_vals = np.concatenate([np.geomspace(0.01, 0.5, 20), np.linspace(0.5, 5.0, 20)])
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline: break
            consider(generate_beta(n, alpha, beta))
    # Dirichlet draws
    for conc in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]:
        n_draws = 2000
        alpha_vec = np.full(n, conc)
        for _ in range(n_draws):
            if time.time() >= deadline: break
            a = np.random.dirichlet(alpha_vec)
            consider(a)
    # Sparse random draws
    while time.time() < deadline:
        s = np.random.randint(int(0.1*n), n)  # support size at least 10% of n
        idx = np.random.choice(n, size=s, replace=False)
        vals = np.random.rand(s)
        a = np.zeros(n)
        a[idx] = vals
        a /= a.sum()
        consider(a)
    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

    while time.time() < deadline:
        t+=1
        # compute a via softmax
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()
        # convolution
        conv = full_convolution(a)
        # compute gamma
        if t < max_iters_est:
            gamma = gamma_start + (gamma_end - gamma_start) * (t / max_iters_est)
        else:
            gamma = gamma_end
        # stable softmax of conv
        max_c = np.max(conv)
        exp_c = np.exp(gamma * (conv - max_c))
        p = exp_c / exp_c.sum()
        # gradient wrt a
        p_rev = p[::-1]
        conv_full = conv_fft(a, p_rev)  # full convolution
        idx_start = 2 * n - 2
        g = 2.0 * conv_full[idx_start - np.arange(n)]
        # grad wrt u (softmax Jacobian)
        g_u = a * (g - np.dot(g, a))
        # Adam steps
        m = beta1 * m + (1 - beta1) * g_u
        v = beta2 * v + (1 - beta2) * (g_u * g_u)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.05 / (np.sqrt(t) + 1.0)  # learning rate schedule
        u -= step * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate real objective using max conv
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # Occasionally adjust gamma more rapidly if stagnation
        if t % 5000 == 0:
            # occasional jitter
            u += np.random.randn(n) * 1e-3
    return best_a

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The optimiser proceeds in three stages:

1. **Broad global exploration** – scans three analytic families
   (power‑law, exponential, beta) over wide parameter ranges and draws
   many random Dirichlet and sparse vectors.  The ten best candidates
   (by the exact objective 2·n·max(conv)) are retained.

2. **Soft‑max Adam polishing** – each retained candidate is refined by
   minimising a smooth surrogate of the max‑convolution:
   `Sγ(a)= (1/γ)·log(∑_k exp(γ·(a∗a)_k))`.  The surrogate converges to the
   true max when γ grows.  Gradients of `Sγ` are obtained analytically,
   and an Adam‑style mirror‑descent on the soft‑max (log‑parameterised)
   simplex is performed.  γ is increased linearly from 1 to ≈60 during
   optimisation, allowing a gradual tightening of the bound.

3. **Final simulated‑annealing polish** – a short Metropolis‐style
   hill‑climb tries to escape any remaining local minima.

All stages respect the 1000 s wall‑clock budget.  The routine returns the
best sequence found as a plain list of floats, ready for the official
``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((size - 1).bit_length())      # next power‑of‑two ≥ size
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:size]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a (length 2·n‑1) via FFT."""
    return conv_fft(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, ∑x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (keeps the best vectors)
# ----------------------------------------------------------------------


def global_explore(n: int, deadline: float, max_keep: int = 15):
    """Wide sweep over families + massive random draws, keeping the best."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # project onto simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep – very wide log‑grid
    for p in np.linspace(0.01, 10.0, 400):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) Exponential sweep – from almost flat to sharply decaying
    for lam in np.linspace(1e-6, 0.5, 400):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep – covers U‑shapes, bowls and asymmetric forms
    al_vals = np.concatenate([np.geomspace(0.01, 0.5, 20),
                              np.linspace(0.5, 5.0, 20)])
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Massive Dirichlet draws with many concentration levels
    concs = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = 1500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    # 5) Sparse random vectors – random support, random weights
    while time.time() < deadline:
        supp = np.random.randint(int(0.1 * n), n)          # at least 10 % support
        idx = np.random.choice(n, size=supp, replace=False)
        vals = np.random.rand(supp)
        a = np.zeros(n, dtype=float)
        a[idx] = vals
        a /= a.sum()
        try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Soft‑max Adam polishing (smooth surrogate of the max)
# ----------------------------------------------------------------------


def softmax_adam_polish(a0: np.ndarray, deadline: float,
                       init_lr: float = 0.05,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8) -> np.ndarray:
    """
    Adam‑style optimisation of the smoothed objective
        Sγ(a) = (1/γ)·log(∑_k exp(γ·(a∗a)_k))
    on the simplex.  The temperature γ is increased linearly from 1 to
    60 during the run.  The variables are stored as log‑probabilities
    ``u`` with a = softmax(u).  Returns the best vector observed.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)          # log‑parametrisation (strictly positive)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = a0.copy()
    best_val = compute_score(a0)

    t = 0
    max_iters_est = 20000          # used to schedule γ

    while time.time() < deadline:
        t += 1

        # ---- soft‑max projection (always on the simplex) ----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ---- convolution and smoothed max ----
        conv = full_convolution(a)

        # schedule γ (linear growth)
        if t < max_iters_est:
            gamma = 1.0 + (60.0 - 1.0) * (t / max_iters_est)
        else:
            gamma = 60.0

        # stable soft‑max of the convolution values
        max_c = np.max(conv)
        exp_c = np.exp(gamma * (conv - max_c))
        p = exp_c / exp_c.sum()               # p_k ≥ 0, ∑ p_k = 1

        # ---- gradient of the surrogate w.r.t. a ----
        # g = ∂S/∂a = 2 * (a * (p_rev ∗ a))_reversed
        p_rev = p[::-1]
        conv_full = conv_fft(a, p_rev)        # length 3·n‑2
        idx_start = 2 * n - 2
        g = 2.0 * conv_full[idx_start - np.arange(n)]

        # ---- transform gradient to the log‑parameter space ----
        # Jacobian of softmax:  ∂a/∂u = diag(a) - a·a^T
        g_u = a * (g - np.dot(g, a))

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = init_lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ---- keep the best exact objective seen so far ----
        cur_val = compute_score(a)            # true max‑based score
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to shake out of plateaux
        if t % 3000 == 0:
            u += np.random.randn(n) * (init_lr * 1e-3)

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a
