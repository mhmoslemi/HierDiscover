def grad_smoothed(a, conv, alpha):
    # compute weights for each k
    maxi = conv.max()
    # compute softmax weights stable:
    shift = np.max(alpha * conv)
    exp_weights = np.exp(alpha * conv - shift)
    w = exp_weights / exp_weights.sum()
    # compute gradient g
    n = len(a)
    g = np.zeros_like(a)
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        i_idx = np.arange(i_start, i_end+1)
        j_idx = k - i_idx
        # each pair contributes 2*a[j_idx] (subgradient) weighted by w[k]
        g[i_idx] += 2 * w[k] * a[j_idx]
    return g

"""
Non‑negative step‑function optimiser (≈ 1000 s budget).

The core idea
---------------
*   Work with a probability vector – the denominator of the official
    objective cancels out, so only the shape of the vector matters.
*   Minimise the *soft‑max* of the auto‑convolution
    `M_α(a)=log‑sum‑exp(α·(a*a))/α`.  For α→∞ the smooth function
    converges to the true max, but for moderate α it supplies a
    differentiable surrogate that avoids the plateaux of a pure
    max‑gradient.
*   The gradient of a weighted convolution can be obtained by a
    single FFT‑based convolution:
        ∂/∂a M_α = 2·corr(a,w) ,   w_i = softmax(α·(a*a)_i)
    where `corr(a,w)[k] = Σ_j a_j·w_{j+k}`.
*   Adam (mirror‑descent on the simplex via the soft‑max
    parametrisation) is applied to the surrogate gradient.
*   A cheap global sweep (parametric families + many Dirichlet
    draws) provides a handful of promising seeds.
*   Random‑restart Adam runs are performed while time permits, and a
    very short simulated‑annealing pass is used as a final polish.

The code follows the schedule:
    1. global sweep → keep the best 5 vectors,
    2. for each vector: smooth‑Adam optimisation,
    3. additional random‑restart smooth‑Adam runs,
    4. final annealing (optional),
    5. return the best sequence (plain Python list).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (all lengths are powers of two for speed)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT of length L (≥ 2·len(a)‑1).
    Returns the first 2·n‑1 entries.
    """
    A = np.fft.rfft(a, n=L)
    C = A * A
    conv = np.fft.irfft(C, n=L)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def weighted_gradient_fft(A: np.ndarray, a: np.ndarray, w: np.ndarray,
                         L: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements  g = 2·Σ_k w_k·∂(a*a)_k/∂a  = 2·corr(a,w) .
    Uses a pre‑computed FFT of a (A) and a single FFT of rev(w).
    """
    # reverse w (corr = conv with reversed)
    rev_w = w[::-1]
    W_rev = np.fft.rfft(rev_w, n=L)
    grad_fft = A * W_rev
    # full convolution length = n + (2n‑1) - 1 = 3n‑2
    conv_full = np.fft.irfft(grad_fft, n=L)
    n = a.shape[0]
    m = 2 * n - 1               # length of w
    # the relevant slice is conv_full[m‑1‑i] for i=0..n‑1
    # (see analysis for a derivation)
    idx = (m - 1) - np.arange(n)
    g = 2.0 * conv_full[idx]
    return g


# ----------------------------------------------------------------------
# Scoring utilities (fast, no clipping)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used during the search.
    Assumes a ≥ 0 and sum(a) = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    L = next_pow2(3 * n - 2)          # enough for both convolutions
    conv = full_convolution_fft(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families for the global sweep
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a /= a.sum()
    return a


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    a /= a.sum()
    return a

# Global sweep (keeps the top k candidates)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, keep: int = 5):
    best = []   # list of (score, vector)

    def try_vec(v):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0:
            return
        v /= s                      # project to simplex
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # --- parametric families -------------------------------------------------
    for p in np.linspace(0.05, 5.0, 200):
        if time.time() >= deadline:
            break
        try_vec(generate_powerlaw(n, p))

    for lam in np.linspace(1e-5, 0.2, 200):
        if time.time() >= deadline:
            break
        try_vec(generate_exponential(n, lam))

    al_vals = np.concatenate([np.geomspace(0.01, 0.2, 10),
                              np.linspace(0.2, 3.0, 15)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.2, 10),
                              np.linspace(0.2, 3.0, 15)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_vec(generate_beta(n, alpha, beta))

    # --- massive Dirichlet draws --------------------------------------------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per = 1200
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            v = np.random.dirichlet(alpha_vec)
            try_vec(v)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def smooth_adam(a0: np.ndarray, deadline: float,
                lr: float = 0.12,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                start_alpha: float = 0.5,
                alpha_inc: float = 0.001,
                max_alpha: float = 300.0,
                tol: float = 1e-12) -> np.ndarray:
    """
    Adam optimiser on the smooth surrogate
          M_α(a)=log‑sum‑exp(α·(a*a))/α .
    The variables are stored in log‑space (u) with a = softmax(u).
    """
    n = a0.shape[0]
    L = next_pow2(3 * n - 2)          # length sufficient for all FFTs
    # initialise in log‑space (avoid log(0))
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        # ---- soft‑max projection ------------------------------------------------
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ---- convolution a*a (FFT) -------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        # current (exact) objective – used for bookkeeping
        cur_val = float(2.0 * n * conv.max())
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # ---- soft‑max weights (smooth max) ------------------------------------
        alpha = min(max_alpha, start_alpha + alpha_inc * t)
        # stable soft‑max
        shifted = conv * alpha
        max_shift = np.max(shifted)
        exp_w = np.exp(shifted - max_shift)
        w = exp_w / exp_w.sum()                     # Σ w = 1

        # ---- weighted gradient (FFT) -----------------------------------------
        # gradient of Σ w_k·(a*a)_k w.r.t a
        g = weighted_gradient_fft(A, a, w, L)

        # ---- transform to gradient on the log‑parameters (mirror‑descent) -----
        g_u = a * (g - np.dot(g, a))

        # ---- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)   # gentle decay
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape extremely flat regions
        if t % 3000 == 0:
            u += np.random.randn(n) * (lr * 0.001)

        # bail out early if the improvement is negligible
        if t % 2000 == 0 and best_val < 1.52:
            # when we have already passed the target region we can be
            # a little more aggressive with the learning rate
            lr = min(0.25, lr * 1.02)

    return best_a if best_a is not None else np.exp(u) / np.exp(u).sum()


# ----------------------------------------------------------------------
# Very short simulated‑annealing polish (optional)
# ----------------------------------------------------------------------


def anneal(a_start: np.ndarray, deadline: float,
           init_temp: float = 4e-4,
           final_temp: float = 1e-6,
           cooling: float = 0.9995,
           step_scale: float = 0.02) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best, best_val = a.copy(), cur_val
    T = init_temp
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = compute_score(cand)
        d = val - cur_val
        if d < 0.0 or np.random.rand() < np.exp(-d / max(T, 1e-12)):
            a, cur_val = cand, val
            if val < best_val:
                best, best_val = cand.copy(), val
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            step_scale = max(step_scale * 0.98, 1e-7)
        T = max(final_temp, T * cooling)
        if it % 3000 == 0:
            # occasional random perturbation
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)
    return best


# ----------------------------------------------------------------------
# Main entry point – the harness will call this
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function with a low value of
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and returns a plain list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # reserve a few seconds for safety (the harness adds its own margin)
    deadline = start + budget_s - 4.0

    # Problem size – a larger n gives more degrees of freedom.
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global sweep (≈ 250 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    if time.time() < sweep_deadline:
        cand_vecs = baseline_search(n, sweep_deadline, keep=5)
        for v in cand_vecs:
            val = compute_score(v)
            if val < best_val:
                best_val, best_seq = val, v.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smoothed Adam polishing (the heavy part)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_adam(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑adam‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # extra random restarts while time permits
        while time.time() < polish_deadline:
            conc = np.random.choice([0.03, 0.07, 0.15, 0.3, 0.6, 1.2])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_adam(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑adam‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Tiny simulated‑annealing polish (optional)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = anneal(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be used)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report the official score
    # -----------------------------------------------------------------
    reported = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {reported:.6f}")
    return best_seq
